## [html_aria_hidden]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>aria-hidden Attribute Demo</title>
    <style>
        .demo-container {
            padding: 20px;
            border: 2px solid #333;
            margin: 10px 0;
        }
        .hidden-visually {
            /* PROBLEM: Hide decorative content from screen readers
               TIME: O(1) SPACE: O(1) */
            aria-hidden: true;
        }
        .icon {
            display: inline-block;
            width: 24px;
            height: 24px;
            background: #007bff;
            margin-right: 8px;
        }
    </style>
</head>
<body>
    <h1>aria-hidden Attribute</h1>
    
    <!-- PROBLEM: How to hide decorative elements from assistive technologies
         TIME: O(1) SPACE: O(1) -->
    
    <div class="demo-container">
        <!-- Icon is decorative, hidden from screen readers -->
        <button>
            <span class="icon" aria-hidden="true"></span>
            <span>Search</span>
        </button>
    </div>
    
    <div class="demo-container">
        <!-- Decorative image hidden from accessibility tree -->
        <div>
            <img src="decoration.png" alt="" aria-hidden="true">
            <p>This is the main content.</p>
        </div>
    </div>
    
    <div class="demo-container">
        <!-- Toggle visibility with JavaScript -->
        <p id="toggle-text">This text can be toggled.</p>
        <button onclick="toggleHidden()">Toggle Hidden</button>
    </div>

    <script>
        /**
         * Toggle aria-hidden attribute on an element
         * PROBLEM: Dynamically show/hide content from screen readers
         * TIME: O(1) SPACE: O(1)
         * 
         * Why: Useful for modals, dropdowns, and dynamic content
         *      that should be excluded from accessibility when not visible
         */
        function toggleHidden() {
            const element = document.getElementById('toggle-text');
            const isHidden = element.getAttribute('aria-hidden') === 'true';
            element.setAttribute('aria-hidden', !isHidden);
        }

        /**
         * Hide an element from screen readers
         * PROBLEM: Programmatically exclude content from accessibility tree
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {HTMLElement} element - The element to hide
         * Why: Clean API for managing accessibility state
         */
        function hideFromScreenReaders(element) {
            element.setAttribute('aria-hidden', 'true');
        }

        /**
         * Show an element to screen readers
         * PROBLEM: Restore element to accessibility tree
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {HTMLElement} element - The element to show
         * Why: Symmetric API for toggling accessibility
         */
        function showToScreenReaders(element) {
            element.removeAttribute('aria-hidden');
        }

        /**
         * Check if element is hidden from screen readers
         * PROBLEM: Query accessibility visibility state
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {HTMLElement} element - The element to check
         * @returns {boolean} True if hidden from screen readers
         * Why: Useful for conditional accessibility logic
         */
        function isHiddenFromScreenReaders(element) {
            return element.getAttribute('aria-hidden') === 'true';
        }
    </script>
</body>
</html>
```

**Why**: `aria-hidden="true"` removes elements from the accessibility tree while keeping them visible on screen. Essential for decorative icons, redundant content, and complex UI patterns where visual and accessible content differ.

## [html_aria_live_assertive]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>aria-live="assertive" Demo</title>
    <style>
        .notification {
            padding: 15px;
            margin: 10px 0;
            border-radius: 4px;
        }
        .notification.error {
            background: #f8d7da;
            border: 1px solid #f5c6cb;
            color: #721c24;
        }
        .demo-container {
            padding: 20px;
            border: 2px solid #333;
            margin: 10px 0;
        }
    </style>
</head>
<body>
    <h1>aria-live="assertive" - Urgent Announcements</h1>
    
    <!-- PROBLEM: How to announce critical updates immediately to screen readers
         TIME: O(1) SPACE: O(1) -->
    
    <div class="demo-container">
        <!-- Assertive live region - interrupts current announcements -->
        <div id="alert-region" aria-live="assertive" class="notification"></div>
        
        <button onclick="showError()">Trigger Error</button>
        <button onclick="showSuccess()">Trigger Success</button>
    </div>

    <script>
        /**
         * Show assertive notification that interrupts screen reader
         * PROBLEM: Announce critical messages immediately
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} message - The message to announce
         * Why: Use for errors, warnings, and urgent status changes
         *      that require immediate user attention
         */
        function showError() {
            const region = document.getElementById('alert-region');
            region.className = 'notification error';
            region.textContent = 'Critical error: Form validation failed!';
        }

        /**
         * Show success notification
         * PROBLEM: Announce important success states
         * TIME: O(1) SPACE: O(1)
         * 
         * Why: Assertive mode should be used sparingly to avoid
         *      overwhelming screen reader users
         */
        function showSuccess() {
            const region = document.getElementById('alert-region');
            region.className = 'notification';
            region.style.background = '#d4edda';
            region.textContent = 'Operation completed successfully!';
        }

        /**
         * Create assertive live region programmatically
         * PROBLEM: Dynamically add urgent announcement regions
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} id - The ID for the region
         * @param {HTMLElement} parent - Parent container
         * @returns {HTMLElement} The created live region
         * Why: Factory function for consistent live region creation
         */
        function createAssertiveRegion(id, parent) {
            const region = document.createElement('div');
            region.id = id;
            region.setAttribute('aria-live', 'assertive');
            region.setAttribute('role', 'alert');
            parent.appendChild(region);
            return region;
        }

        /**
         * Announce message assertively with timeout
         * PROBLEM: Show temporary urgent notifications
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} message - Message to announce
         * @param {number} timeout - Duration in milliseconds
         * Why: Auto-clearing notifications improve UX
         */
        function announceAssertive(message, timeout = 5000) {
            const region = document.getElementById('alert-region');
            region.textContent = message;
            
            if (timeout > 0) {
                setTimeout(() => {
                    region.textContent = '';
                }, timeout);
            }
        }
    </script>
</body>
</html>
```

**Why**: `aria-live="assertive"` immediately interrupts screen readers to announce urgent content. Use for critical errors, validation failures, and time-sensitive alerts. Overuse can create poor accessibility experience.

## [html_meta_refresh_redirect]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Meta Refresh Redirect Demo</title>
    
    <!-- PROBLEM: How to redirect users after a delay without JavaScript
         TIME: O(1) SPACE: O(1) -->
    
    <!-- Redirect after 5 seconds -->
    <!-- <meta http-equiv="refresh" content="5;url=https://example.com"> -->
    
    <!-- Immediate redirect (0 seconds) -->
    <!-- <meta http-equiv="refresh" content="0;url=https://example.com"> -->
    
    <style>
        .demo-container {
            padding: 20px;
            border: 2px solid #333;
            margin: 10px 0;
        }
        .countdown {
            font-size: 24px;
            font-weight: bold;
            color: #007bff;
        }
    </style>
</head>
<body>
    <h1>Meta Refresh Redirect</h1>
    
    <div class="demo-container">
        <h2>How It Works</h2>
        <p>The meta refresh tag redirects users after a specified time:</p>
        <code>&lt;meta http-equiv="refresh" content="5;url=https://example.com"&gt;</code>
        
        <h3>Parameters:</h3>
        <ul>
            <li><strong>content</strong>: Seconds before redirect; URL after semicolon</li>
            <li><strong>http-equiv</strong>: Set to "refresh"</li>
        </ul>
    </div>
    
    <div class="demo-container">
        <h2>JavaScript Countdown Alternative</h2>
        <p>Redirecting in <span id="countdown" class="countdown">5</span> seconds...</p>
        <button onclick="cancelRedirect()">Cancel Redirect</button>
    </div>

    <script>
        /**
         * Meta refresh alternative using JavaScript
         * PROBLEM: Provide redirect with countdown and cancel option
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} url - Destination URL
         * @param {number} seconds - Delay before redirect
         * Why: JavaScript redirect allows user cancellation and feedback
         */
        let redirectTimer = null;
        let countdownInterval = null;
        
        function startRedirect(url, seconds) {
            let remaining = seconds;
            const countdownEl = document.getElementById('countdown');
            
            // Update countdown every second
            countdownInterval = setInterval(() => {
                remaining--;
                countdownEl.textContent = remaining;
                
                if (remaining <= 0) {
                    clearInterval(countdownInterval);
                }
            }, 1000);
            
            // Set redirect timer
            redirectTimer = setTimeout(() => {
                window.location.href = url;
            }, seconds * 1000);
        }

        /**
         * Cancel the pending redirect
         * PROBLEM: Allow users to abort automatic navigation
         * TIME: O(1) SPACE: O(1)
         * 
         * Why: User control over navigation improves accessibility
         */
        function cancelRedirect() {
            if (redirectTimer) {
                clearTimeout(redirectTimer);
                redirectTimer = null;
            }
            if (countdownInterval) {
                clearInterval(countdownInterval);
                countdownInterval = null;
            }
            document.getElementById('countdown').textContent = 'cancelled';
        }

        /**
         * Generate meta refresh tag as string
         * PROBLEM: Programmatically create redirect HTML
         * TIME: O(1) SPACE: O(n) where n is URL length
         * 
         * @param {string} url - Destination URL
         * @param {number} seconds - Delay in seconds
         * @returns {string} HTML meta tag string
         * Why: Useful for server-side rendering and dynamic pages
         */
        function generateMetaRefresh(url, seconds = 0) {
            return `<meta http-equiv="refresh" content="${seconds};url=${url}">`;
        }

        /**
         * Apply meta refresh dynamically to document
         * PROBLEM: Add redirect after page load
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} url - Destination URL
         * @param {number} seconds - Delay in seconds
         * Why: Dynamic redirect control based on conditions
         */
        function applyMetaRefresh(url, seconds) {
            const meta = document.createElement('meta');
            meta.httpEquiv = 'refresh';
            meta.content = `${seconds};url=${url}`;
            document.head.appendChild(meta);
        }

        // Demo: Start countdown (commented out to prevent actual redirect)
        // startRedirect('https://example.com', 5);
    </script>
</body>
</html>
```

**Why**: Meta refresh provides server-side redirects without JavaScript. Use sparingly as it's considered a poor practice for SEO and accessibility. Prefer HTTP 301/302 redirects or JavaScript with user confirmation.

## [html_link_rel_help]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Link rel="help" Demo</title>
    
    <!-- PROBLEM: Provide contextual help documentation link
         TIME: O(1) SPACE: O(1) -->
    
    <!-- Link to help documentation for this page -->
    <link rel="help" href="/docs/user-guide.html">
    
    <style>
        .demo-container {
            padding: 20px;
            border: 2px solid #333;
            margin: 10px 0;
        }
        .help-link {
            display: inline-flex;
            align-items: center;
            padding: 8px 16px;
            background: #007bff;
            color: white;
            text-decoration: none;
            border-radius: 4px;
        }
        .help-link:hover {
            background: #0056b3;
        }
    </style>
</head>
<body>
    <h1>Link rel="help" - Help Documentation</h1>
    
    <div class="demo-container">
        <h2>What It Does</h2>
        <p>The <code>rel="help"</code> attribute links to help documentation relevant to the current page.</p>
        
        <h3>HTML Example:</h3>
        <pre>&lt;link rel="help" href="/docs/user-guide.html"&gt;</pre>
        <pre>&lt;a rel="help" href="/faq"&gt;Get Help&lt;/a&gt;</pre>
    </div>
    
    <div class="demo-container">
        <h2>Accessible Help Link</h2>
        <a rel="help" href="/help" class="help-link" aria-label="Get help with this page">
            ❓ Help
        </a>
    </div>

    <script>
        /**
         * Create help link element programmatically
         * PROBLEM: Add help documentation links dynamically
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} href - URL to help documentation
         * @param {string} title - Optional title attribute
         * @returns {HTMLLinkElement} The created link element
         * Why: Semantic linking improves navigation and SEO
         */
        function createHelpLink(href, title = '') {
            const link = document.createElement('link');
            link.rel = 'help';
            link.href = href;
            if (title) {
                link.title = title;
            }
            return link;
        }

        /**
         * Add help link to document head
         * PROBLEM: Inject help resource into page metadata
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} href - Help documentation URL
         * Why: Dynamic help link based on page context
         */
        function addHelpToHead(href) {
            const existing = document.querySelector('link[rel="help"]');
            if (existing) {
                existing.href = href;
            } else {
                document.head.appendChild(createHelpLink(href));
            }
        }

        /**
         * Create help anchor with accessibility attributes
         * PROBLEM: Generate accessible help navigation
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} href - Help URL
         * @param {string} text - Link text
         * @returns {HTMLAnchorElement} Accessible help link
         * Why: Consistent accessible help link generation
         */
        function createHelpAnchor(href, text = 'Help') {
            const anchor = document.createElement('a');
            anchor.rel = 'help';
            anchor.href = href;
            anchor.textContent = text;
            anchor.setAttribute('aria-label', `Get help: ${text}`);
            return anchor;
        }

        /**
         * Get all help links in document
         * PROBLEM: Query help resources for navigation
         * TIME: O(n) SPACE: O(k) where k is number of help links
         * 
         * @returns {Array} Array of help link elements
         * Why: Useful for building help navigation menus
         */
        function getHelpLinks() {
            return Array.from(document.querySelectorAll('[rel="help"]'));
        }
    </script>
</body>
</html>
```

**Why**: `rel="help"` semantically links to help documentation. Screen readers and browsers can use this to provide contextual assistance. Improves navigation discoverability and user support.

## [css_grid_align_items]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>CSS Grid align-items Demo</title>
    <style>
        .demo-container {
            padding: 20px;
            border: 2px solid #333;
            margin: 10px 0;
        }
        
        /* PROBLEM: Control vertical alignment of grid items
           TIME: O(1) SPACE: O(1) */
        
        .grid-container {
            display: grid;
            grid-template-columns: repeat(3, 100px);
            grid-template-rows: 200px;
            gap: 10px;
            background: #f0f0f0;
            padding: 10px;
            margin: 10px 0;
        }
        
        .grid-item {
            background: #007bff;
            color: white;
            padding: 10px;
        }
        
        /* align-items values */
        .grid-start { align-items: start; }
        .grid-end { align-items: end; }
        .grid-center { align-items: center; }
        .grid-stretch { align-items: stretch; }
        .grid-baseline { align-items: baseline; }
    </style>
</head>
<body>
    <h1>CSS Grid align-items</h1>
    
    <div class="demo-container">
        <h2>align-items: start</h2>
        <div class="grid-container grid-start">
            <div class="grid-item">Item 1</div>
            <div class="grid-item">Item 2</div>
            <div class="grid-item">Item 3</div>
        </div>
    </div>
    
    <div class="demo-container">
        <h2>align-items: center</h2>
        <div class="grid-container grid-center">
            <div class="grid-item">Item 1</div>
            <div class="grid-item">Item 2</div>
            <div class="grid-item">Item 3</div>
        </div>
    </div>

    <script>
        /**
         * Set align-items on a grid container
         * PROBLEM: Programmatically control grid item alignment
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {HTMLElement} container - Grid container element
         * @param {string} value - Alignment value (start|end|center|stretch|baseline)
         * Why: Dynamic layout control based on content or user preference
         */
        function setGridAlignItems(container, value) {
            const validValues = ['start', 'end', 'center', 'stretch', 'baseline'];
            if (validValues.includes(value)) {
                container.style.alignItems = value;
            }
        }

        /**
         * Toggle between alignment values
         * PROBLEM: Cycle through alignment options
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {HTMLElement} container - Grid container
         * @returns {string} The new alignment value
         * Why: Interactive alignment switching for demos
         */
        function toggleGridAlignItems(container) {
            const values = ['start', 'center', 'end', 'stretch'];
            const current = getComputedStyle(container).alignItems;
            const currentIndex = values.indexOf(current);
            const nextIndex = (currentIndex + 1) % values.length;
            container.style.alignItems = values[nextIndex];
            return values[nextIndex];
        }

        /**
         * Create a grid container with alignment
         * PROBLEM: Factory function for consistent grid creation
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} align - Initial align-items value
         * @param {number} columns - Number of columns
         * @returns {HTMLElement} Configured grid container
         * Why: Reusable grid container generation
         */
        function createGridContainer(align = 'stretch', columns = 3) {
            const container = document.createElement('div');
            container.style.display = 'grid';
            container.style.gridTemplateColumns = `repeat(${columns}, 100px)`;
            container.style.alignItems = align;
            container.style.gap = '10px';
            return container;
        }

        /**
         * Get current align-items value
         * PROBLEM: Query computed alignment style
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {HTMLElement} container - Grid container
         * @returns {string} Current align-items value
         * Why: Useful for responsive alignment checks
         */
        function getGridAlignItems(container) {
            return getComputedStyle(container).alignItems;
        }
    </script>
</body>
</html>
```

**Why**: `align-items` controls vertical (block-axis) alignment of all grid items within their cells. Essential for centering content, stretching to fill, or aligning to edges. Works with flexbox too.

## [css_grid_column_start]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">

        /**

         * TIME: O(1) SPACE: O(1)

         * Get current column-start value
         * PROBLEM: Query grid position
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {HTMLElement} item - Grid item

            return getComputedStyle(item).gridColumnStart;

         * Move item to new column position
         * PROBLEM: Reposition grid item dynamically
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {HTMLElement} item - Grid item
         * @param {number} newStart - New starting column
         * Why: Animation and reordering support
         */
        function moveGridColumn(item, newStart) {
            item.style.gridColumnStart = newStart;
        }

        /**
         * Create grid item with column position
         * PROBLEM: Factory for positioned grid items
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} content - Item content
         * @param {number} start - Column start position
         * @returns {HTMLElement} Configured grid item
         * Why: Consistent item creation with positioning
         */
        function createGridItem(content, start) {
            const item = document.createElement('div');
            item.className = 'grid-item';
            item.textContent = content;
            item.style.gridColumnStart = start;
            return item;
        }
    </script>
</body>
</html>
```

**Why**: `grid-column-start` specifies the grid line where an item begins. Essential for magazine-style layouts, spanning elements, and precise grid positioning. Works with named lines for maintainability.

## [css_selector_descendant]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>CSS Descendant Selector Demo</title>
    <style>
        .demo-container {
            padding: 20px;
            border: 2px solid #333;
            margin: 10px 0;
        }
        
        /* PROBLEM: Style elements nested anywhere inside another element
           TIME: O(n) SPACE: O(1) - where n is DOM depth */
        
        /* Descendant combinator: space between selectors */
        .article p {
            color: #333;
            line-height: 1.6;
        }
        
        /* Targets all spans inside divs inside .article */
        .article div span {
            color: #007bff;
            font-weight: bold;
        }
        
        /* Nested at any depth */
        .article em {
            background: yellow;
            padding: 2px 4px;

            background: #fff3cd;
            border: 1px solid #ffc107;
        }

    <h1>CSS Descendant Selector (Space)</h1>
    
    <div class="demo-container">
        <h2>How It Works</h2>
        <p>The space character selects <strong>any</strong> nested element at any depth:</p>
        <code>.parent .child { }</code>
        
        <div class="article">
            <p>This paragraph is styled by <em>.article p</em></p>
            <div>
                <p>This paragraph is also styled (nested deeper)</p>
                <span>This span is styled by <em>.article div span</em></span>
            </div>
        </div>
    </div>

    <script>
        /**
         * Query all descendants matching a selector
         * PROBLEM: Find elements at any nesting level
         * TIME: O(n) SPACE: O(k) - where k is number of matches
         * 
         * @param {HTMLElement} parent - Parent element
         * @param {string} selector - CSS selector
         * @returns {NodeList} All matching descendants
         * Why: JavaScript equivalent of descendant selector
         */
        function getDescendants(parent, selector) {
            return parent.querySelectorAll(selector);
        }

        /**
         * Check if element is descendant of another
         * PROBLEM: Determine ancestry relationship
         * TIME: O(d) SPACE: O(1) - where d is depth
         * 
         * @param {HTMLElement} potentialAncestor - Possible ancestor
         * @param {HTMLElement} element - Element to check
         * @returns {boolean} True if element is a descendant
         * Why: Useful for event delegation and DOM traversal
         */
        function isDescendantOf(potentialAncestor, element) {
            let current = element.parentNode;
            while (current !== null) {
                if (current === potentialAncestor) {
                    return true;
                }
                current = current.parentNode;
            }
            return false;
        }

        /**
         * Apply styles to all descendants matching selector
         * PROBLEM: Programmatically style nested elements
         * TIME: O(n) SPACE: O(1)
         * 
         * @param {HTMLElement} parent - Parent container
         * @param {string} selector - Descendant selector
         * @param {Object} styles - Style object
         * Why: Dynamic styling based on data or state
         */
        function styleDescendants(parent, selector, styles) {
            const elements = parent.querySelectorAll(selector);
            elements.forEach(el => {
                Object.assign(el.style, styles);
            });
        }

        /**
         * Find closest ancestor matching selector
         * PROBLEM: Traverse up the DOM tree
         * TIME: O(d) SPACE: O(1)
         * 
         * @param {HTMLElement} element - Starting element
         * @param {string} selector - CSS selector to match
         * @returns {HTMLElement|null} Matching ancestor or null
         * Why: Reverse of descendant selection
         */
        function getClosestAncestor(element, selector) {
            return element.closest(selector);
        }
    </script>
</body>
</html>
```

**Why**: The descendant selector (space) matches elements at any depth within another element. Most commonly used selector pattern. Use child combinator (>) for direct children only for better performance.

         * TIME: O(1) SPACE: O(1)

         * @param {number} duration - Duration in milliseconds
         * Why: Testing and programmatic interactions

        function simulateActive(element, duration = 200) {
            element.classList.add('js-active');
            setTimeout(() => {
                element.classList.remove('js-active');
            }, duration);
        }

        /**
         * Add active state listener
         * PROBLEM: Track when element is being activated
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {HTMLElement} element - Target element
         * @param {Function} callback - Callback on active
         * Why: Custom handling of activation events
         */
        function onActive(element, callback) {
            element.addEventListener('mousedown', () => callback(true));
            element.addEventListener('mouseup', () => callback(false));
            element.addEventListener('mouseleave', () => callback(false));
            
            // Touch support
            element.addEventListener('touchstart', () => callback(true));
            element.addEventListener('touchend', () => callback(false));
        }

         * PROBLEM: Factory for consistent buttons
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} text - Button text
         * @param {Function} onClick - Click handler
         * @returns {HTMLButtonElement} Styled button
         * Why: Consistent button creation with active states
         */
        function createButton(text, onClick) {
            const btn = document.createElement('button');
            btn.className = 'btn';
            btn.textContent = text;
            btn.addEventListener('click', onClick);

         * PROBLEM: Determine what color :active produces
         * TIME: O(1) SPACE: O(1)

         * @returns {string} Computed background color

        function getActiveColor(element) {
            // Note: This gets the current state, not the :active rule
            return getComputedStyle(element).backgroundColor;
        }

**Why**: `:active` styles elements while being clicked or activated. Creates tactile feedback for buttons and links. Must appear after `:hover` in CSS for proper cascade. Duration is brief—use subtle transforms or color changes.

## [js_string_charAt]

```html
<!DOCTYPE html>
<html lang="en">
<head>

    <div class="demo-container">
        <h2>Basic Usage</h2>

    </div>
    
    <div class="demo-container">
        <h2>Interactive Demo</h2>

    <script>

        const sampleString = "JavaScript";
        document.getElementById('output-basic').innerHTML = `
            String: "${sampleString}"<br>
            charAt(0): "${sampleString.charAt(0)}"<br>

        /**
         * Get character at specified index
         * PROBLEM: Access individual character in string
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} str - Input string
         * @param {number} index - Character position
         * @returns {string} Character at index or empty string
         * Why: Safe character access - returns '' for out of bounds
         */
        function getCharAt(str, index) {
            return str.charAt(index);
        }

        /**
         * Get first character of string
         * PROBLEM: Common pattern for string processing
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} str - Input string
         * @returns {string} First character or empty string
         * Why: Convenient shortcut for charAt(0)
         */
        function getFirstChar(str) {
            return str.charAt(0);
        }

        /**
         * Get last character of string
         * PROBLEM: Common pattern for parsing
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} str - Input string
         * @returns {string} Last character or empty string
         * Why: Useful for checking extensions, suffixes
         */
        function getLastChar(str) {
            return str.charAt(str.length - 1);
        }

        /**
         * Safely get character with bounds check
         * PROBLEM: Validate index before access
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} str - Input string
         * @param {number} index - Character position
         * @returns {string|null} Character or null if invalid
         * Why: Explicit handling of invalid indices
         */
        function getCharSafe(str, index) {
            if (index < 0 || index >= str.length) {
                return null;
            }
            return str.charAt(index);
        }

        /**
         * Iterate string with charAt
         * PROBLEM: Process each character individually
         * TIME: O(n) SPACE: O(n)
         * 
         * @param {string} str - Input string
         * @param {Function} callback - Process each char
         * Why: Alternative to for...of with index access
         */
        function forEachChar(str, callback) {
            for (let i = 0; i < str.length; i++) {
                callback(str.charAt(i), i);
            }
        }

         * Display charAt result interactively
         */
        function showCharAt() {
            const str = document.getElementById('input-string').value;

                Unicode: U+${code.toString(16).toUpperCase().padStart(4, '0')}
            `;
        }
    </script>
</body>
</html>
```

**Why**: `charAt(index)` returns the character at a position, safely returning empty string for out-of-bounds indices. Prefer bracket notation `str[i]` for brevity unless you need the safe behavior for negative indices.

## [js_array_shift]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>JavaScript Array.shift() Demo</title>
    <style>
        .demo-container {
            padding: 20px;
            border: 2px solid #333;
            margin: 10px 0;
        }
        .output {
            background: #f8f9fa;
            padding: 10px;
            margin: 10px 0;
            border-radius: 4px;
            font-family: monospace;
        }
        .queue-item {
            padding: 5px 10px;
            margin: 2px;
            background: #007bff;
            color: white;
            display: inline-block;
            border-radius: 4px;
        }
    </style>
</head>
<body>
    <h1>JavaScript Array.shift()</h1>
    
    <div class="demo-container">
        <h2>Basic Usage</h2>
        <div class="output" id="output-basic"></div>
    </div>
    
    <div class="demo-container">
        <h2>Queue Demo</h2>
        <p>Queue: <span id="queue-display"></span></p>
        <button onclick="dequeueItem()">Dequeue (Shift)</button>
        <button onclick="enqueueItem()">Enqueue (Push)</button>
        <button onclick="resetQueue()">Reset</button>
        <div class="output" id="output-queue"></div>
    </div>

    <script>
        // Basic demonstration
        let demoArray = [1, 2, 3, 4, 5];
        const first = demoArray.shift();
        document.getElementById('output-basic').innerHTML = `
            Original: [1, 2, 3, 4, 5]<br>
            After shift(): [${demoArray}]<br>
            Returned value: ${first}<br>
            <br>
            <strong>Important:</strong> shift() re-indexes all elements! TIME: O(n)
        `;

        // Queue implementation
        let queue = ['Task A', 'Task B', 'Task C', 'Task D'];
        
        function updateQueueDisplay() {
            const display = document.getElementById('queue-display');
            display.innerHTML = queue.map(item => 
                `<span class="queue-item">${item}</span>`
            ).join(' ');
        }
        
        updateQueueDisplay();

        /**
         * Remove first element from array
         * PROBLEM: Extract and return first item, modifying array
         * TIME: O(n) SPACE: O(1) - re-indexes all remaining elements
         * 
         * @param {Array} arr - Input array
         * @returns {*} The removed first element
         * Why: Queue dequeue operation, FIFO data structure
         */
        function shift(arr) {
            return arr.shift();
        }

        /**
         * Dequeue item from queue
         * PROBLEM: Remove and return next item in queue
         * TIME: O(n) SPACE: O(1)
         * 
         * @returns {*} Next item in queue or undefined if empty
         * Why: FIFO processing order
         */
        function dequeueItem() {
            const item = queue.shift();
            updateQueueDisplay();
            document.getElementById('output-queue').innerHTML = 
                `Dequeued: ${item !== undefined ? item : 'Queue is empty!'}`;
        }

        /**
         * Enqueue item to queue
         * PROBLEM: Add item to end of queue
         * TIME: O(1) SPACE: O(1)
         */
        let counter = queue.length + 1;
        function enqueueItem() {
            queue.push(`Task ${String.fromCharCode(64 + counter++)}`);
            updateQueueDisplay();
            document.getElementById('output-queue').innerHTML = 'Item added to queue.';
        }

        /**
         * Reset queue to initial state
         */
        function resetQueue() {
            queue = ['Task A', 'Task B', 'Task C', 'Task D'];
            counter = 5;
            updateQueueDisplay();
            document.getElementById('output-queue').innerHTML = 'Queue reset.';
        }

        /**
         * Shift multiple elements
         * PROBLEM: Remove first n elements
         * TIME: O(n*m) SPACE: O(m) where m is count
         * 
         * @param {Array} arr - Input array
         * @param {number} count - Number of elements to shift
         * @returns {Array} Array of removed elements
         * Why: Batch dequeue operations
         */
        function shiftMultiple(arr, count) {
            const result = [];
            for (let i = 0; i < count && arr.length > 0; i++) {
                result.push(arr.shift());
            }
            return result;
        }

        /**
         * Safe shift with default value
         * PROBLEM: Handle empty arrays gracefully
         * TIME: O(n) SPACE: O(1)
         * 
         * @param {Array} arr - Input array
         * @param {*} defaultValue - Value to return if empty
         * @returns {*} First element or default
         * Why: Prevent undefined in processing pipelines
         */
        function shiftSafe(arr, defaultValue = null) {
            return arr.length > 0 ? arr.shift() : defaultValue;
        }

        /**
         * Peek without removing
         * PROBLEM: View first element without modification
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {Array} arr - Input array
         * @returns {*} First element or undefined
         * Why: Check queue head without dequeueing
         */
        function peek(arr) {
            return arr[0];
        }
    </script>
</body>
</html>
```

**Why**: `shift()` removes the first element and returns it, with O(n) time complexity due to re-indexing. Use for queue (FIFO) operations. For performance-critical code with frequent additions/removals at both ends, consider using a linked list.
