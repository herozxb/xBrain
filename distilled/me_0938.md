## Maximum Sum Subarray (Sliding Window)

```python
# PROBLEM: Find the maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)
def max_sum_subarray(arr, k):
    # Handle edge case: array too small for window
    if len(arr) < k:
        return None
    
    # Compute sum of first window
    window_sum = sum(arr[:k])
    max_sum = window_sum
    
    # Slide window across array, updating sum in O(1) per step
    for i in range(k, len(arr)):
        # Add new element, remove old element from left
        window_sum += arr[i] - arr[i - k]
        max_sum = max(max_sum, window_sum)
    
    return max_sum
```
**Why**: Classic sliding window technique for fixed-size subarray problems; avoids recomputing sum from scratch each time.

## Two Sum II (Two Pointers)

```python
# PROBLEM: Find two numbers in sorted array that sum to target; return indices (1-based)
# TIME: O(n) SPACE: O(1)
def two_sum_sorted(nums, target):
    # Initialize two pointers at opposite ends
    left, right = 0, len(nums) - 1
    
    while left < right:
        current_sum = nums[left] + nums[right]
        
        if current_sum == target:
            # Return 1-based indices
            return [left + 1, right + 1]
        elif current_sum < target:
            # Need larger sum, move left pointer right
            left += 1
        else:
            # Need smaller sum, move right pointer left
            right -= 1
    
    return None  # No solution exists
```
**Why**: Two-pointer technique exploits sorted property to achieve O(n) without extra space.

## Subsets (Backtracking)

```python
# PROBLEM: Generate all possible subsets (power set) of a set
# TIME: O(n * 2^n) SPACE: O(n) for recursion stack
def subsets(nums):
    result = []
    
    def backtrack(start, current):
        # Add copy of current subset to result
        result.append(current[:])
        
        # Try adding each remaining element
        for i in range(start, len(nums)):
            # Include nums[i] in current subset
            current.append(nums[i])
            # Recurse with next starting index
            backtrack(i + 1, current)
            # Backtrack: remove last element
            current.pop()
    
    backtrack(0, [])
    return result
```
**Why**: Backtracking explores all combinations by building up solutions incrementally and undoing choices.

## Number of Connected Components (Union-Find)

```python
# PROBLEM: Count connected components in an undirected graph
# TIME: O(E * α(n)) ≈ O(E) SPACE: O(n)
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.components = n
    
    def find(self, x):
        # Path compression: point directly to root
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        # Union by rank to keep tree balanced
        px, py = self.find(x), self.find(y)
        if px == py:
            return False  # Already connected
        
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        self.components -= 1
        return True

def count_components(n, edges):
    uf = UnionFind(n)
    for u, v in edges:
        uf.union(u, v)
    return uf.components
```
**Why**: Union-Find with path compression and union by rank provides near-constant-time connectivity queries.

## Longest Common Subsequence (Dynamic Programming)

```python
# PROBLEM: Find length of longest common subsequence between two strings
# TIME: O(m * n) SPACE: O(m * n) or O(min(m,n)) with optimization
def longest_common_subsequence(text1, text2):
    m, n = len(text1), len(text2)
    
    # dp[i][j] = LCS length of text1[:i] and text2[:j]
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if text1[i - 1] == text2[j - 1]:
                # Characters match: extend previous LCS
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                # Take best of skipping char from either string
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    
    return dp[m][n]
```
**Why**: Classic 2D DP demonstrating optimal substructure and overlapping subproblems.

## Dijkstra's Shortest Path (Graphs)

```python
# PROBLEM: Find shortest path distances from source to all nodes in weighted graph
# TIME: O((V + E) log V) with heap SPACE: O(V)
import heapq

def dijkstra(graph, start):
    # Distance dictionary: node -> shortest distance
    distances = {node: float('inf') for node in graph}
    distances[start] = 0
    
    # Priority queue: (distance, node)
    pq = [(0, start)]
    visited = set()
    
    while pq:
        dist, node = heapq.heappop(pq)
        
        # Skip if already processed
        if node in visited:
            continue
        visited.add(node)
        
        # Relax all neighbors
        for neighbor, weight in graph[node]:
            new_dist = dist + weight
            if new_dist < distances[neighbor]:
                distances[neighbor] = new_dist
                heapq.heappush(pq, (new_dist, neighbor))
    
    return distances
```
**Why**: Greedy approach with min-heap efficiently finds shortest paths in graphs with non-negative weights.

## Lowest Common Ancestor (Trees)

```python
# PROBLEM: Find lowest common ancestor of two nodes in binary tree
# TIME: O(n) SPACE: O(h) where h is tree height
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def lowest_common_ancestor(root, p, q):
    # Base case: reached leaf or found target
    if root is None or root == p or root == q:
        return root
    
    # Search in both subtrees
    left = lowest_common_ancestor(root.left, p, q)
    right = lowest_common_ancestor(root.right, p, q)
    
    # If both sides returned nodes, root is LCA
    if left and right:
        return root
    
    # Otherwise return non-None side
    return left if left else right
```
**Why**: Elegant recursive tree traversal leveraging the definition of LCA; single pass solution.

## Merge K Sorted Lists (Heap)

```python
# PROBLEM: Merge k sorted linked lists into one sorted list
# TIME: O(N log k) where N = total nodes, k = number of lists
# SPACE: O(k) for heap
import heapq

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next
    
    # Enable comparison for heap
    def __lt__(self, other):
        return self.val < other.val

def merge_k_lists(lists):
    # Min-heap containing head of each list
    heap = []
    for node in lists:
        if node:
            heapq.heappush(heap, node)
    
    # Dummy head simplifies edge cases
    dummy = ListNode()
    current = dummy
    
    while heap:
        # Get smallest node
        smallest = heapq.heappop(heap)
        current.next = smallest
        current = current.next
        
        # Push next node from same list
        if smallest.next:
            heapq.heappush(heap, smallest.next)
    
    return dummy.next
```
**Why**: Heap efficiently maintains k candidates and always extracts minimum; better than merge-sort approach.

## Word Search II (Trie)

```python
# PROBLEM: Find all words from dictionary that exist in 2D board
# TIME: O(m * n * 4^L) where L is max word length SPACE: O(total chars in words)
class TrieNode:
    def __init__(self):
        self.children = {}
        self.word = None  # Store complete word at end node

def find_words(board, words):
    # Build trie from word list
    root = TrieNode()
    for word in words:
        node = root
        for ch in word:
            if ch not in node.children:
                node.children[ch] = TrieNode()
            node = node.children[ch]
        node.word = word
    
    result = []
    rows, cols = len(board), len(board[0)
    
    def dfs(r, c, node):
        ch = board[r][c]
        
        # No word with this prefix
        if ch not in node.children:
            return
        
        next_node = node.children[ch]
        
        # Found a complete word
        if next_node.word:
            result.append(next_node.word)
            next_node.word = None  # Avoid duplicates
        
        # Mark visited
        board[r][c] = '#'
        
        # Explore 4 directions
        for dr, dc in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
            nr, nc = r + dr, c + dc
            if 0 <= nr < rows and 0 <= nc < cols:
                dfs(nr, nc, next_node)
        
        # Restore cell
        board[r][c] = ch
    
    # Start DFS from each cell
    for r in range(rows):
        for c in range(cols):
            dfs(r, c, root)
    
    return result
```
**Why**: Trie provides efficient prefix lookup during board traversal; avoids searching for non-existent prefixes.

## Climbing Stairs with Memoization (LRU Cache)

```python
# PROBLEM: Count ways to climb n stairs taking 1 or 2 steps at a time
# TIME: O(n) SPACE: O(n)
from functools import lru_cache

@lru_cache(maxsize=None)
def climb_stairs(n):
    # Base cases
    if n <= 1:
        return 1
    if n == 2:
        return 2
    
    # Recurrence: ways(n) = ways(n-1) + ways(n-2)
    return climb_stairs(n - 1) + climb_stairs(n - 2)

# Alternative: iterative with O(1) space
def climb_stairs_iterative(n):
    if n <= 2:
        return n
    
    prev, curr = 1, 2
    for _ in range(3, n + 1):
        prev, curr = curr, prev + curr
    
    return curr
```
**Why**: `@lru_cache` decorator provides elegant memoization; automatic caching of expensive recursive calls.
