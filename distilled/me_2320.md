## Suffix Tree Construction

```python
# PROBLEM: Build a suffix tree for efficient pattern matching and substring queries
# TIME: O(n²) SPACE: O(n²)
class SuffixTreeNode:
    def __init__(self):
        self.children = {}  # Maps characters to child nodes
        self.start = None   # Start index of the edge label
        self.end = None     # End index of the edge label
        self.suffix_link = None  # Link for Ukkonen's algorithm

class SuffixTree:
    def __init__(self, text):
        self.text = text
        self.root = SuffixTreeNode()
        self._build_suffix_tree()
    
    def _build_suffix_tree(self):
        """Build suffix tree using naive O(n²) approach."""
        n = len(self.text)
        # Add all suffixes to the tree
        for i in range(n):
            self._add_suffix(i)
    
    def _add_suffix(self, start_idx):
        """Add a single suffix starting at start_idx to the tree."""
        current = self.root
        i = start_idx
        n = len(self.text)
        
        while i < n:
            char = self.text[i]
            if char not in current.children:
                # Create new leaf node
                new_node = SuffixTreeNode()
                new_node.start = i
                new_node.end = n - 1
                current.children[char] = new_node
                return
            else:
                # Traverse existing edge
                child = current.children[char]
                edge_start = child.start
                edge_end = child.end
                
                # Match characters along the edge
                j = edge_start
                while i < n and j <= edge_end and self.text[i] == self.text[j]:
                    i += 1
                    j += 1
                
                if j > edge_end:
                    # Continue to child node
                    current = child
                else:
                    # Split the edge
                    split_node = SuffixTreeNode()
                    split_node.start = edge_start
                    split_node.end = j - 1
                    
                    # Update existing child
                    child.start = j
                    split_node.children[self.text[j]] = child
                    
                    # Create new leaf for remaining suffix
                    if i < n:
                        new_leaf = SuffixTreeNode()
                        new_leaf.start = i
                        new_leaf.end = n - 1
                        split_node.children[self.text[i]] = new_leaf
                    
                    current.children[char] = split_node
                    return
    
    def search(self, pattern):
        """Check if pattern exists in the text. O(m) where m = len(pattern)."""
        current = self.root
        i = 0
        n = len(pattern)
        
        while i < n:
            char = pattern[i]
            if char not in current.children:
                return False
            child = current.children[char]
            # Match along the edge
            j = child.start
            while i < n and j <= child.end and pattern[i] == self.text[j]:
                i += 1
                j += 1
            if i < n and j <= child.end:
                return False  # Mismatch
            current = child
        return True
```
**Why**: Suffix trees enable O(m) pattern matching after O(n) preprocessing, crucial for bioinformatics, text editors, and search engines.

## Prime Factorization

```python
# PROBLEM: Find all prime factors of a number with their exponents
# TIME: O(√n) SPACE: O(log n)
def prime_factorization(n):
    """
    Return prime factors of n as a dictionary {prime: exponent}.
    Uses trial division optimized with small primes.
    """
    if n < 2:
        return {}
    
    factors = {}
    
    # Check for factor of 2
    while n % 2 == 0:
        factors[2] = factors.get(2, 0) + 1
        n //= 2
    
    # Check odd factors from 3 to √n
    divisor = 3
    while divisor * divisor <= n:
        while n % divisor == 0:
            factors[divisor] = factors.get(divisor, 0) + 1
            n //= divisor
        divisor += 2
    
    # If n is still greater than 1, it's a prime factor
    if n > 1:
        factors[n] = factors.get(n, 0) + 1
    
    return factors

def prime_factorization_with_primes(n, primes):
    """
    Factorize using precomputed prime list for better performance.
    TIME: O(π(√n)) where π is prime counting function.
    """
    factors = {}
    
    for p in primes:
        if p * p > n:
            break
        while n % p == 0:
            factors[p] = factors.get(p, 0) + 1
            n //= p
    
    if n > 1:
        factors[n] = 1
    
    return factors

# Example usage:
# prime_factorization(360) → {2: 3, 3: 2, 5: 1}
# prime_factorization(1000000007) → {1000000007: 1} (prime)
```
**Why**: Prime factorization is fundamental to cryptography (RSA), number theory applications, and simplifying mathematical expressions.

## Rod Cutting Problem

```python
# PROBLEM: Maximize revenue by cutting a rod of length n, given prices for each length
# TIME: O(n²) SPACE: O(n)
def rod_cutting(prices, n):
    """
    Given price[i] = price for rod of length i,
    find maximum revenue for rod of length n.
    
    Args:
        prices: List where prices[i-1] is price for length i
        n: Length of rod to cut
    
    Returns:
        Maximum revenue achievable
    """
    # dp[i] = max revenue for rod of length i
    dp = [0] * (n + 1)
    
    # Build solution bottom-up
    for i in range(1, n + 1):
        max_val = float('-inf')
        # Try all possible first cuts
        for j in range(1, i + 1):
            if j <= len(prices):
                max_val = max(max_val, prices[j - 1] + dp[i - j])
        dp[i] = max_val
    
    return dp[n]

def rod_cutting_with_solution(prices, n):
    """
    Returns both maximum revenue and the optimal cuts.
    """
    dp = [0] * (n + 1)
    cut_choice = [0] * (n + 1)  # First cut length for optimal solution
    
    for i in range(1, n + 1):
        max_val = float('-inf')
        for j in range(1, min(i, len(prices)) + 1):
            if prices[j - 1] + dp[i - j] > max_val:
                max_val = prices[j - 1] + dp[i - j]
                cut_choice[i] = j
        dp[i] = max_val
    
    # Reconstruct the cuts
    cuts = []
    length = n
    while length > 0:
        cuts.append(cut_choice[length])
        length -= cut_choice[length]
    
    return dp[n], cuts

# Example:
# prices = [1, 5, 8, 9, 10, 17, 17, 20]
# rod_cutting(prices, 8) → 22 (cut into lengths 2 and 6)
```
**Why**: Classic DP problem demonstrating optimal substructure; applications in resource allocation and manufacturing optimization.

## Longest Repeating Subsequence

```python
# PROBLEM: Find longest subsequence that appears at least twice in a string (non-overlapping indices)
# TIME: O(n²) SPACE: O(n²)
def longest_repeating_subsequence(s):
    """
    Find the length of longest subsequence that repeats in the string.
    The two occurrences must use different indices.
    
    This is a variation of LCS where we compare the string with itself,
    but require different indices.
    """
    n = len(s)
    # dp[i][j] = length of LRS for s[0:i] and s[0:j]
    dp = [[0] * (n + 1) for _ in range(n + 1)]
    
    # Fill the DP table
    for i in range(1, n + 1):
        for j in range(1, n + 1):
            if s[i - 1] == s[j - 1] and i != j:
                # Characters match and at different positions
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                # Take max of excluding one character from either string
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    
    return dp[n][n]

def get_longest_repeating_subsequence(s):
    """
    Return one of the longest repeating subsequences.
    """
    n = len(s)
    dp = [[0] * (n + 1) for _ in range(n + 1)]
    
    for i in range(1, n + 1):
        for j in range(1, n + 1):
            if s[i - 1] == s[j - 1] and i != j:
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    
    # Backtrack to find the subsequence
    result = []
    i, j = n, n
    while i > 0 and j > 0:
        if s[i - 1] == s[j - 1] and i != j:
            result.append(s[i - 1])
            i -= 1
            j -= 1
        elif dp[i - 1][j] > dp[i][j - 1]:
            i -= 1
        else:
            j -= 1
    
    return ''.join(reversed(result))

# Example: longest_repeating_subsequence("aabebcdd") → 3 ("abd")
```
**Why**: Used in DNA sequence analysis, plagiarism detection, and finding repeated patterns in data.

## Letter Combinations of Phone Number

```python
# PROBLEM: Generate all possible letter combinations for a phone number
# TIME: O(4^n * n) SPACE: O(4^n * n)
def letter_combinations(digits):
    """
    Given a string of digits 2-9, return all possible letter combinations
    that the number could represent (like phone keypad).
    
    2: abc, 3: def, 4: ghi, 5: jkl, 6: mno, 7: pqrs, 8: tuv, 9: wxyz
    """
    if not digits:
        return []
    
    # Mapping of digit to letters
    phone_map = {
        '2': 'abc', '3': 'def', '4': 'ghi', '5': 'jkl',
        '6': 'mno', '7': 'pqrs', '8': 'tuv', '9': 'wxyz'
    }
    
    result = []
    
    def backtrack(index, current):
        # Base case: combination complete
        if index == len(digits):
            result.append(''.join(current))
            return
        
        # Try each letter for current digit
        for letter in phone_map[digits[index]]:
            current.append(letter)
            backtrack(index + 1, current)
            current.pop()  # Backtrack
    
    backtrack(0, [])
    return result

def letter_combinations_iterative(digits):
    """
    Iterative approach using queue/accumulator pattern.
    """
    if not digits:
        return []
    
    phone_map = {
        '2': 'abc', '3': 'def', '4': 'ghi', '5': 'jkl',
        '6': 'mno', '7': 'pqrs', '8': 'tuv', '9': 'wxyz'
    }
    
    combinations = ['']
    
    for digit in digits:
        new_combinations = []
        for combo in combinations:
            for letter in phone_map[digit]:
                new_combinations.append(combo + letter)
        combinations = new_combinations
    
    return combinations

# Example: letter_combinations("23") → ["ad","ae","af","bd","be","bf","cd","ce","cf"]
```
**Why**: Classic backtracking problem teaching combinatorial generation; practical use in T9 predictive text systems.

## Word Search II

```python
# PROBLEM: Find all words from a dictionary that exist in a 2D board (Boggle-style)
# TIME: O(m * n * 4^L) where L is max word length SPACE: O(total chars in words)
class TrieNode:
    def __init__(self):
        self.children = {}
        self.word = None  # Store complete word at leaf

def word_search_ii(board, words):
    """
    Given a 2D board of characters and a list of words,
    find all words that can be formed by adjacent cells (horizontal/vertical).
    Each cell can be used at most once per word.
    """
    if not board or not board[0] or not words:
        return []
    
    # Build trie from words
    root = TrieNode()
    for word in words:
        node = root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.word = word  # Mark end of word
    
    rows, cols = len(board), len(board[0])
    result = set()
    
    def dfs(r, c, node):
        char = board[r][c]
        
        # Check if this path can continue
        if char not in node.children:
            return
        
        next_node = node.children[char]
        
        # Found a word
        if next_node.word:
            result.add(next_node.word)
        
        # Mark cell as visited
        board[r][c] = '#'
        
        # Explore all 4 directions
        directions = [(0, 1), (0, -1), (1, 0), (-1, 0)]
        for dr, dc in directions:
            nr, nc = r + dr, c + dc
            if 0 <= nr < rows and 0 <= nc < cols and board[nr][nc] != '#':
                dfs(nr, nc, next_node)
        
        # Restore cell
        board[r][c] = char
    
    # Start DFS from each cell
    for r in range(rows):
        for c in range(cols):
            dfs(r, c, root)
    
    return list(result)

# Example:
# board = [["o","a","a","n"],["e","t","a","e"],["i","h","k","r"],["i","f","l","v"]]
# words = ["oath","pea","eat","rain"]
# word_search_ii(board, words) → ["oath","eat"]
```
**Why**: Combines Trie with DFS for efficient multi-word search; used in word games, Boggle solvers, and search optimization.

## HyperLogLog (Cardinality Estimation)

```python
# PROBLEM: Estimate the number of distinct elements in a massive dataset
# TIME: O(1) per element SPACE: O(m) where m = number of registers
import hashlib

class HyperLogLog:
    """
    Probabilistic data structure for cardinality estimation.
    Uses ~1.04 * m bytes for m registers with ~1.04/√m error rate.
    """
    
    def __init__(self, precision=14):
        """
        precision: number of bits for register index (4-16)
        Number of registers = 2^precision
        Error rate ≈ 1.04 / √(2^precision)
        """
        self.precision = precision
        self.num_registers = 1 << precision  # 2^precision
        self.registers = [0] * self.num_registers
        self.alpha = self._get_alpha()
    
    def _get_alpha(self):
        """Correction constant based on number of registers."""
        m = self.num_registers
        if m == 16:
            return 0.673
        elif m == 32:
            return 0.697
        elif m == 64:
            return 0.709
        else:
            return 0.7213 / (1 + 1.079 / m)
    
    def _hash(self, element):
        """Return 64-bit hash of element."""
        if isinstance(element, str):
            element = element.encode('utf-8')
        return int(hashlib.md5(element).hexdigest()[:16], 16)
    
    def add(self, element):
        """Add an element to the set."""
        h = self._hash(element)
        
        # Use first 'precision' bits for register index
        register_idx = h & (self.num_registers - 1)
        
        # Use remaining bits for counting leading zeros
        remaining = h >> self.precision
        
        # Count leading zeros (plus 1 for the first 1 bit position)
        leading_zeros = self._count_leading_zeros(remaining) + 1
        
        # Keep maximum
        self.registers[register_idx] = max(
            self.registers[register_idx], 
            leading_zeros
        )
    
    def _count_leading_zeros(self, n):
        """Count leading zeros in binary representation."""
        if n == 0:
            return 64 - self.precision
        count = 0
        while (n >> (63 - self.precision - count)) == 0:
            count += 1
        return count
    
    def count(self):
        """Estimate the cardinality."""
        # Harmonic mean of 2^register_values
        sum_inv = sum(2 ** (-r) for r in self.registers)
        
        estimate = (self.alpha * self.num_registers ** 2) / sum_inv
        
        # Small range correction
        if estimate < 2.5 * self.num_registers:
            zeros = self.registers.count(0)
            if zeros > 0:
                estimate = self.num_registers * (
                    0.6931471805599453  # ln(2)
                ) / zeros
        
        # Large range correction (for very large sets)
        elif estimate > (1/30) * (1 << 32):
            estimate = -(1 << 32) * (1 - estimate / (1 << 32))
        
        return int(estimate)
    
    def merge(self, other):
        """Merge another HyperLogLog into this one."""
        for i in range(self.num_registers):
            self.registers[i] = max(self.registers[i], other.registers[i])

# Example:
# hll = HyperLogLog(precision=12)  # ~4096 registers, ~1.6% error
# for i in range(1000000):
#     hll.add(f"user_{i}")
# hll.count() → ≈1000000 (with ~1.6% error)
```
**Why**: Enables counting unique items in big data with minimal memory; used by Reddit, Google, and streaming platforms for analytics.

## K-d Tree (k-dimensional tree)

```python
# PROBLEM: Efficiently search for nearest neighbors in k-dimensional space
# TIME: O(n log n) build, O(log n) average search SPACE: O(n)
class KDNode:
    def __init__(self, point, axis, left=None, right=None):
        self.point = point      # The k-dimensional point
        self.axis = axis        # Split axis (0, 1, ..., k-1)
        self.left = left
        self.right = right

class KDTree:
    """
    Space-partitioning data structure for organizing points in k dimensions.
    """
    
    def __init__(self, points, k=2):
        """
        Build KD-tree from list of points.
        k: number of dimensions
        """
        self.k = k
        self.root = self._build(points, depth=0)
    
    def _build(self, points, depth):
        """Recursively build the tree."""
        if not points:
            return None
        
        # Select axis based on depth
        axis = depth % self.k
        
        # Sort points by the current axis and choose median
        points.sort(key=lambda p: p[axis])
        median_idx = len(points) // 2
        
        # Create node and recurse
        return KDNode(
            point=points[median_idx],
            axis=axis,
            left=self._build(points[:median_idx], depth + 1),
            right=self._build(points[median_idx + 1:], depth + 1)
        )
    
    def nearest_neighbor(self, target):
        """
        Find the nearest neighbor to target point.
        Returns (nearest_point, distance).
        """
        self.best = None
        self.best_dist = float('inf')
        
        def search(node):
            if node is None:
                return
            
            # Calculate distance to current node
            dist = self._distance(node.point, target)
            if dist < self.best_dist:
                self.best = node.point
                self.best_dist = dist
            
            # Determine which subtree to search first
            axis = node.axis
            diff = target[axis] - node.point[axis]
            
            # Search the closer subtree first
            first = node.left if diff < 0 else node.right
            second = node.right if diff < 0 else node.left
            
            search(first)
            
            # Check if we need to search the other subtree
            if diff ** 2 < self.best_dist:
                search(second)
        
        search(self.root)
        return self.best, self.best_dist ** 0.5
    
    def _distance(self, p1, p2):
        """Calculate squared Euclidean distance."""
        return sum((a - b) ** 2 for a, b in zip(p1, p2))
    
    def range_search(self, lower, upper):
        """
        Find all points within the hyper-rectangle defined by lower and upper.
        """
        results = []
        
        def search(node):
            if node is None:
                return
            
            # Check if current point is in range
            if all(l <= p <= u for l, p, u in zip(lower, node.point, upper)):
                results.append(node.point)
            
            axis = node.axis
            
            # Check left subtree
            if node.point[axis] >= lower[axis]:
                search(node.left)
            
            # Check right subtree
            if node.point[axis] <= upper[axis]:
                search(node.right)
        
        search(self.root)
        return results

# Example:
# points = [[2,3], [5,4], [9,6], [4,7], [8,1], [7,2]]
# tree = KDTree(points, k=2)
# tree.nearest_neighbor([9, 2]) → ([8,1], 1.414...)
```
**Why**: Essential for spatial databases, nearest neighbor search in ML, and computer graphics; enables O(log n) queries in multi-dimensional space.

## Cuckoo Filter

```python
# PROBLEM: Set membership with better deletion support than Bloom filter
# TIME: O(1) insert/lookup/delete SPACE: O(n) with ~2-3 bits per element
import hashlib

class CuckooFilter:
    """
    Probabilistic filter supporting add, lookup, and delete operations.
    False positive rate ~ 2-3%, no false negatives.
    """
    
    def __init__(self, capacity, bucket_size=4, fingerprint_size=8):
        """
        capacity: expected number of elements
        bucket_size: entries per bucket
        fingerprint_size: bits for fingerprint (affects false positive rate)
        """
        self.capacity = capacity
        self.bucket_size = bucket_size
        self.fingerprint_size = fingerprint_size
        self.fingerprint_mask = (1 << fingerprint_size) - 1
        self.num_buckets = self._next_power_of_2(capacity // bucket_size)
        self.buckets = [[] for _ in range(self.num_buckets)]
        self.max_kicks = 500  # Maximum displacement attempts
    
    def _next_power_of_2(self, n):
        """Find next power of 2 >= n."""
        power = 1
        while power < n:
            power <<= 1
        return power
    
    def _hash(self, data):
        """Generate hash value."""
        if isinstance(data, str):
            data = data.encode('utf-8')
        return int(hashlib.sha256(data).hexdigest(), 16)
    
    def _fingerprint(self, item):
        """Generate fingerprint for item."""
        h = self._hash(item)
        fp = h & self.fingerprint_mask
        # Ensure fingerprint is never 0 (0 means empty)
        return fp if fp != 0 else 1
    
    def _hash1(self, item):
        """First hash function."""
        return self._hash(item) % self.num_buckets
    
    def _hash2(self, fp, h1):
        """Second hash function (uses fingerprint)."""
        # XOR-based alternate position
        return (h1 ^ (self._hash(fp) % self.num_buckets)) % self.num_buckets
    
    def add(self, item):
        """Add item to filter. Returns False if filter is too full."""
        fp = self._fingerprint(item)
        h1 = self._hash1(item)
        h2 = self._hash2(fp, h1)
        
        # Try to insert in either bucket
        buckets = [h1, h2]
        for h in buckets:
            if len(self.buckets[h]) < self.bucket_size:
                self.buckets[h].append(fp)
                return True
        
        # Kick out existing entry
        h = h1
        for _ in range(self.max_kicks):
            # Randomly select an entry to kick
            import random
            idx = random.randint(0, len(self.buckets[h]) - 1)
            kicked_fp = self.buckets[h][idx]
            self.buckets[h][idx] = fp
            fp = kicked_fp
            
            # Find alternate position
            h = self._hash2(fp, h)
            if len(self.buckets[h]) < self.bucket_size:
                self.buckets[h].append(fp)
                return True
        
        return False  # Filter is full
    
    def contains(self, item):
        """Check if item might be in filter."""
        fp = self._fingerprint(item)
        h1 = self._hash1(item)
        h2 = self._hash2(fp, h1)
        return fp in self.buckets[h1] or fp in self.buckets[h2]
    
    def remove(self, item):
        """Remove item from filter. Returns True if item was present."""
        fp = self._fingerprint(item)
        h1 = self._hash1(item)
        h2 = self._hash2(fp, h1)
        
        if fp in self.buckets[h1]:
            self.buckets[h1].remove(fp)
            return True
        if fp in self.buckets[h2]:
            self.buckets[h2].remove(fp)
            return True
        return False

# Example:
# cf = CuckooFilter(capacity=10000)
# cf.add("hello")        # True
# cf.contains("hello")   # True
# cf.remove("hello")     # True
# cf.contains("hello")   # False
```
**Why**: Supports deletion unlike Bloom filter; used in caching systems, network routers, and databases for membership testing.

## Fixed Window Rate Limiter

```python
# PROBLEM: Limit number of requests per fixed time window
# TIME: O(1) SPACE: O(n) where n = number of unique clients
import time
from collections import defaultdict
from threading import Lock

class FixedWindowRateLimiter:
    """
    Rate limiter using fixed time windows.
    Simple but allows burst at window boundaries.
    """
    
    def __init__(self, max_requests, window_seconds):
        """
        max_requests: maximum requests allowed per window
        window_seconds: window duration in seconds
        """
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self.windows = defaultdict(int)  # client -> count
        self.window_starts = defaultdict(float)  # client -> window start time
        self.lock = Lock()
    
    def is_allowed(self, client_id):
        """
        Check if request from client_id is allowed.
        Returns (allowed: bool, remaining: int, reset_time: float).
        """
        with self.lock:
            current_time = time.time()
            window_start = self.window_starts.get(client_id, 0)
            
            # Check if window has expired
            if current_time - window_start >= self.window_seconds:
                # Start new window
                self.windows[client_id] = 0
                self.window_starts[client_id] = current_time
                window_start = current_time
            
            # Calculate remaining requests and reset time
            current_count = self.windows[client_id]
            remaining = max(0, self.max_requests - current_count)
            reset_time = window_start + self.window_seconds
            
            # Check if allowed
            if current_count < self.max_requests:
                self.windows[client_id] = current_count + 1
                return True, remaining - 1, reset_time
            else:
                return False, 0, reset_time
    
    def get_stats(self, client_id):
        """Get current rate limit stats for a client."""
        current_time = time.time()
        window_start = self.window_starts.get(client_id, current_time)
        
        if current_time - window_start >= self.window_seconds:
            return {
                'count': 0,
                'remaining': self.max_requests,
                'reset_time': current_time + self.window_seconds
            }
        
        return {
            'count': self.windows[client_id],
            'remaining': max(0, self.max_requests - self.windows[client_id]),
            'reset_time': window_start + self.window_seconds
        }


class SlidingWindowRateLimiter:
    """
    Rate limiter using sliding window for smoother rate limiting.
    More accurate but requires O(requests) space per client.
    """
    
    def __init__(self, max_requests, window_seconds):
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self.requests = defaultdict(list)  # client -> [timestamps]
        self.lock = Lock()
    
    def is_allowed(self, client_id):
        """Check if request is allowed using sliding window."""
        with self.lock:
            current_time = time.time()
            cutoff = current_time - self.window_seconds
            
            # Remove expired timestamps
            self.requests[client_id] = [
                t for t in self.requests[client_id] if t > cutoff
            ]
            
            # Check if allowed
            if len(self.requests[client_id]) < self.max_requests:
                self.requests[client_id].append(current_time)
                return True
            return False
```
