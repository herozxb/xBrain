## css_transform_style

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>CSS Transform Style - 3D Perspective</title>
  <style>
    /* 
     * PROBLEM: Preserve 3D transformations in nested elements
     * TIME: O(1) SPACE: O(1)
     * 
     * transform-style: flat - children flattened into 2D plane
     * transform-style: preserve-3d - children maintain 3D position
     */
    
    .scene {
      width: 200px;
      height: 200px;
      perspective: 800px;
      margin: 50px;
    }
    
    .parent-flat {
      width: 100%;
      height: 100%;
      transform-style: flat;
      transform: rotateX(30deg) rotateY(30deg);
      background: rgba(255, 0, 0, 0.5);
      border: 2px solid #333;
    }
    
    .parent-3d {
      width: 100%;
      height: 100%;
      transform-style: preserve-3d;
      transform: rotateX(30deg) rotateY(30deg);
      background: rgba(0, 255, 0, 0.5);
      border: 2px solid #333;
    }
    
    .child {
      position: absolute;
      width: 80px;
      height: 80px;
      top: 60px;
      left: 60px;
      background: rgba(0, 0, 255, 0.8);
      transform: translateZ(50px);
      display: flex;
      align-items: center;
      justify-content: center;
      color: white;
      font-weight: bold;
    }
    
    .container {
      display: flex;
      gap: 50px;
    }
  </style>
</head>
<body>
  <!-- 
    Why: Essential for creating true 3D effects in CSS.
    Without preserve-3d, nested elements lose their 3D depth.
    Used in card flips, 3D carousels, and complex animations.
  -->
  
  <h2>transform-style: flat (2D)</h2>
  <div class="container">
    <div class="scene">
      <div class="parent-flat">
        Parent
        <div class="child">Child</div>
      </div>
    </div>
    
    <div class="scene">
      <div class="parent-3d">
        Parent
        <div class="child">Child</div>
      </div>
    </div>
  </div>
  
  <p>The right cube preserves 3D - child maintains its Z-offset!</p>
</body>
</html>
```

**Why**: Required for creating authentic 3D CSS effects like card flips, 3D carousels, and layered animations. Without `preserve-3d`, nested elements collapse into the parent's 2D plane.

## js_shorthand_method_names

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>ES6 Shorthand Method Names</title>
</head>
<body>
  <script>
    /* 
     * PROBLEM: Define methods in objects with cleaner syntax
     * TIME: O(1) SPACE: O(1)
     * 
     * ES6 allows omitting 'function' keyword in object methods.
     * Also works for computed property names.
     */
    
    // ============================================
    // OLD SYNTAX (ES5)
    // ============================================
    const oldObject = {
      name: 'Old Style',
      
      // Traditional function expression
      greet: function() {
        return `Hello, I'm ${this.name}`;
      },
      
      // Method with parameters
      add: function(a, b) {
        return a + b;
      }
    };
    
    // ============================================
    // NEW SYNTAX (ES6 Shorthand)
    // ============================================
    const newObject = {
      name: 'New Style',
      
      // Shorthand method - no 'function' keyword needed!
      greet() {
        return `Hello, I'm ${this.name}`;
      },
      
      // Works with parameters too
      add(a, b) {
        return a + b;
      },
      
      // Computed method name
      ['compute' + 'Sum'](arr) {
        return arr.reduce((acc, val) => acc + val, 0);
      },
      
      // Generator shorthand
      *generator() {
        yield 1;
        yield 2;
        yield 3;
      },
      
      // Async method shorthand
      async fetchData(url) {
        const response = await fetch(url);
        return response.json();
      }
    };
    
    // ============================================
    // DEMONSTRATION
    // ============================================
    console.log('=== Shorthand Method Names Demo ===\n');
    
    console.log(oldObject.greet());  // "Hello, I'm Old Style"
    console.log(newObject.greet());  // "Hello, I'm New Style"
    
    console.log(oldObject.add(5, 3)); // 8
    console.log(newObject.add(5, 3)); // 8
    
    console.log(newObject.computeSum([1, 2, 3, 4, 5])); // 15
    
    // Generator usage
    for (const value of newObject.generator()) {
      console.log('Generator value:', value);
    }
    
    // ============================================
    // PRACTICAL USE CASE: Module Pattern
    // ============================================
    const Calculator = {
      result: 0,
      
      add(n) {
        this.result += n;
        return this; // Chainable
      },
      
      subtract(n) {
        this.result -= n;
        return this;
      },
      
      multiply(n) {
        this.result *= n;
        return this;
      },
      
      getResult() {
        return this.result;
      },
      
      reset() {
        this.result = 0;
        return this;
      }
    };
    
    // Chaining with shorthand methods
    const result = Calculator
      .add(10)
      .multiply(2)
      .subtract(5)
      .getResult();
    
    console.log('Calculator result:', result); // 15
    
    // Output to page
    document.body.innerHTML = `
      <h2>ES6 Shorthand Method Names</h2>
      <pre>
Old: greet: function() { }
New: greet() { }

Results:
- newObject.greet(): ${newObject.greet()}
- newObject.add(5, 3): ${newObject.add(5, 3)}
- newObject.computeSum([1,2,3,4,5]): ${newObject.computeSum([1,2,3,4,5])}
- Calculator chain result: ${result}
      </pre>
    `;
  </script>
</body>
</html>
```

**Why**: Cleaner, more readable object method definitions. Eliminates verbose `function` keyword. Essential for modern JavaScript classes, modules, and method chaining patterns.

## js_array_reverse

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>JavaScript Array Reverse</title>
</head>
<body>
  <script>
    /* 
     * PROBLEM: Reverse the order of elements in an array
     * TIME: O(n) SPACE: O(1) - in-place mutation
     * 
     * Array.prototype.reverse() reverses array IN-PLACE.
     * Returns reference to the same array (mutated).
     * WARNING: Modifies original array!
     */
    
    // ============================================
    // BASIC USAGE
    // ============================================
    const numbers = [1, 2, 3, 4, 5];
    console.log('Original:', numbers);
    
    const reversed = numbers.reverse();
    console.log('Reversed:', reversed);       // [5, 4, 3, 2, 1]
    console.log('Original after:', numbers);  // [5, 4, 3, 2, 1] - MUTATED!
    
    // They're the same reference
    console.log('Same reference:', reversed === numbers); // true
    
    // ============================================
    // NON-DESTRUCTIVE REVERSE (preserve original)
    // ============================================
    const fruits = ['apple', 'banana', 'cherry'];
    
    // Method 1: Spread operator + reverse
    const reversedFruits = [...fruits].reverse();
    console.log('Fruits preserved:', fruits);           // Original intact
    console.log('Reversed copy:', reversedFruits);
    
    // Method 2: slice() + reverse
    const reversedCopy = fruits.slice().reverse();
    
    // Method 3: toReversed() - ES2023 (non-mutating)
    // const reversedNew = fruits.toReversed();
    
    // ============================================
    // REVERSING STRINGS (via array conversion)
    // ============================================
    function reverseString(str) {
      // Split into array, reverse, join back
      return str.split('').reverse().join('');
    }
    
    console.log(reverseString('Hello')); // 'olleH'
    console.log(reverseString('JavaScript')); // 'tpircSavaJ'
    
    // ============================================
    // PRACTICAL USE CASES
    // ============================================
    
    // 1. Get last N items from array
    function getLastN(arr, n) {
      return [...arr].reverse().slice(0, n).reverse();
    }
    
    const items = [1, 2, 3, 4, 5, 6, 7];
    console.log('Last 3 items:', getLastN(items, 3)); // [5, 6, 7]
    
    // 2. Check if string is palindrome
    function isPalindrome(str) {
      const normalized = str.toLowerCase().replace(/[^a-z]/g, '');
      return normalized === normalized.split('').reverse().join('');
    }
    
    console.log('Palindrome "racecar":', isPalindrome('racecar')); // true
    console.log('Palindrome "A man a plan":', isPalindrome('A man, a plan, a canal, Panama')); // true
    
    // 3. Reverse sorting order (when combined with sort)
    const scores = [85, 92, 78, 96, 88];
    scores.sort((a, b) => b - a); // Descending
    console.log('Descending scores:', scores);
    
    // ============================================
    // REVERSE 2D ARRAY (matrix)
    // ============================================
    const matrix = [
      [1, 2, 3],
      [4, 5, 6],
      [7, 8, 9]
    ];
    
    const reversedMatrix = matrix.map(row => [...row].reverse()).reverse();
    console.log('Reversed matrix:', reversedMatrix);
    // [[9, 8, 7], [6, 5, 4], [3, 2, 1]]
    
    // ============================================
    // OUTPUT TO PAGE
    // ============================================
    document.body.innerHTML = `
      <h2>Array.reverse() Examples</h2>
      <pre>
Basic Reverse:
  Original: [1, 2, 3, 4, 5]
  After reverse(): [5, 4, 3, 2, 1]
  
Non-Destructive (preserve original):
  [...arr].reverse() creates a new reversed array
  
String Reverse:
  'Hello' → '${reverseString('Hello')}'
  
Palindrome Check:
  'racecar': ${isPalindrome('racecar')}
  'A man, a plan, a canal, Panama': ${isPalindrome('A man, a plan, a canal, Panama')}
  
Last N Items [1-7], n=3:
  ${JSON.stringify(getLastN(items, 3))}
      </pre>
    `;
  </script>
</body>
</html>
```

**Why**: Fundamental array manipulation for reversing order. Commonly used in UI (newest-first lists), string processing, and algorithmic problems. Remember: it mutates the original array!

## js_string_replace

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>JavaScript String Replace</title>
</head>
<body>
  <script>
    /* 
     * PROBLEM: Replace occurrences in a string
     * TIME: O(n) SPACE: O(n)
     * 
     * str.replace(pattern, replacement)
     * - With string: replaces ONLY first occurrence
     * - With regex + /g flag: replaces all occurrences
     * - Use replaceAll() for global string replacement (ES2021)
     */
    
    // ============================================
    // BASIC STRING REPLACEMENT
    // ============================================
    const sentence = 'The cat sat on the cat mat';
    
    // String pattern - replaces ONLY FIRST match
    const replaced1 = sentence.replace('cat', 'dog');
    console.log(replaced1);
    // "The dog sat on the cat mat" - only first 'cat' changed!
    
    // ============================================
    // REGEX GLOBAL REPLACEMENT
    // ============================================
    const replaced2 = sentence.replace(/cat/g, 'dog');
    console.log(replaced2);
    // "The dog sat on the dog mat" - ALL 'cat' changed
    
    // Case-insensitive replacement
    const text = 'Hello HELLO hello';
    const replaced3 = text.replace(/hello/gi, 'hi');
    console.log(replaced3); // "hi hi hi"
    
    // ============================================
    // REPLACE WITH CALLBACK FUNCTION
    // ============================================
    const prices = 'Item: $10, Tax: $2, Total: $12';
    
    // Convert all dollar amounts to euros (multiply by 0.85)
    const inEuros = prices.replace(/\$(\d+)/g, (match, amount) => {
      const euros = Math.round(parseInt(amount) * 0.85);
      return `€${euros}`;
    });
    console.log(inEuros); // "Item: €9, Tax: €2, Total: €10"
    
    // ============================================
    // SPECIAL REPLACEMENT PATTERNS
    // ============================================
    const str = 'Hello World';
    
    // $$ - inserts a literal $
    console.log(str.replace('World', '$$$')); // "Hello $$$"
    
    // $& - inserts the matched substring
    console.log('hello'.replace(/l/g, '[$&]')); // "he[l][l]o"
    
    // $` - inserts portion before match
    console.log('hello'.replace(/l/, '$`')); // "heheo"
    
    // $' - inserts portion after match
    console.log('hello'.replace(/l/, "$'")); // "heolo"
    
    // $n - nth captured group
    const date = '2024-12-25';
    const formatted = date.replace(/(\d{4})-(\d{2})-(\d{2})/, '$3/$2/$1');
    console.log(formatted); // "25/12/2024"
    
    // Named groups (ES2018)
    const nameStr = 'John Doe';
    const swapped = nameStr.replace(/(?<first>\w+) (?<last>\w+)/, '$<last>, $<first>');
    console.log(swapped); // "Doe, John"
    
    // ============================================
    // PRACTICAL USE CASES
    // ============================================
    
    // 1. Sanitize HTML (basic XSS prevention)
    function escapeHtml(text) {
      return text
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }
    console.log(escapeHtml('<script>alert("XSS")</script>'));
    
    // 2. Format phone number
    function formatPhone(digits) {
      return digits.replace(/(\d{3})(\d{3})(\d{4})/, '($1) $2-$3');
    }
    console.log(formatPhone('5551234567')); // "(555) 123-4567"
    
    // 3. Trim extra whitespace
    function normalizeWhitespace(text) {
      return text.replace(/\s+/g, ' ').trim();
    }
    console.log(normalizeWhitespace('  Multiple   spaces   here  '));
    
    // 4. Mask sensitive data
    function maskCreditCard(cardNumber) {
      return cardNumber.replace(/\d(?=\d{4})/g, '*');
    }
    console.log(maskCreditCard('1234567890123456')); // "************3456"
    
    // ============================================
    // OUTPUT TO PAGE
    // ============================================
    document.body.innerHTML = `
      <h2>String.replace() Examples</h2>
      <pre>
String pattern (first only):
  "${sentence}"
  → "${sentence.replace('cat', 'dog')}"

Regex with /g (all):
  → "${sentence.replace(/cat/g, 'dog')}"

Callback function (price conversion):
  "${prices}"
  → "${inEuros}"

Date reformatting with groups:
  "${date}" → "${formatted}"

Phone formatting:
  "5551234567" → "${formatPhone('5551234567')}"

Credit card masking:
  "1234567890123456" → "${maskCreditCard('1234567890123456')}"
      </pre>
    `;
  </script>
</body>
</html>
```

**Why**: Powerful text manipulation for search/replace operations. Supports regex patterns, callback functions, and special tokens. Essential for data sanitization, formatting, and text processing.

## js_style_property

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>JavaScript Style Property Manipulation</title>
  <style>
    .box {

      display: flex;
      align-items: center;
      justify-content: center;

    }
    
    .controls {

      background: #f5f5f5;
      border-radius: 8px;
    }
    
    button {
      padding: 10px 15px;
      margin: 5px;
      cursor: pointer;
    }
  </style>
</head>
<body>
  <div id="targetBox" class="box">Style Me!</div>
  
  <div class="controls">
    <button onclick="changeColor()">Random Color</button>
    <button onclick="changeSize()">Random Size</button>
    <button onclick="rotateBox()">Rotate</button>
    <button onclick="addShadow()">Add Shadow</button>
    <button onclick="resetStyles()">Reset</button>
  </div>

  <script>
    /* 
     * PROBLEM: Dynamically modify CSS styles via JavaScript
     * TIME: O(1) SPACE: O(1)
     * 
     * element.style.property = value
     * - Uses camelCase (backgroundColor, not background-color)
     * - Modifies INLINE styles only
     * - Use getComputedStyle() for computed values
     */
    
    const box = document.getElementById('targetBox');
    
    // ============================================
    // BASIC STYLE MODIFICATION
    // ============================================
    
    // Single property
    box.style.color = 'white';
    
    // Multiple properties
    function changeColor() {
      const randomColor = '#' + Math.floor(Math.random() * 16777215).toString(16);
      box.style.backgroundColor = randomColor;
      box.textContent = randomColor;
    }
    
    function changeSize() {
      const randomWidth = Math.floor(Math.random() * 200) + 100;
      const randomHeight = Math.floor(Math.random() * 200) + 100;
      
      // Note: Must include units!
      box.style.width = randomWidth + 'px';
      box.style.height = randomHeight + 'px';
    }
    
    function rotateBox() {
      const currentTransform = box.style.transform || '';
      const currentRotation = parseInt(currentTransform.replace(/[^0-9]/g, '')) || 0;
      const newRotation = currentRotation + 45;
      
      box.style.transform = `rotate(${newRotation}deg)`;
    }
    
    function addShadow() {
      box.style.boxShadow = '0 10px 30px rgba(0, 0, 0, 0.3)';
    }
    
    // ============================================
    // CSS TEXT (multiple properties at once)
    // ============================================
    
    function applyMultipleStyles() {
      // Method 1: cssText (overwrites all inline styles!)
      box.style.cssText = `
        background: linear-gradient(45deg, #ff6b6b, #feca57);
        border-radius: 50%;
        transform: scale(1.2);
        box-shadow: 0 20px 40px rgba(0,0,0,0.4);
      `;
    }
    
    // ============================================
    // READING STYLES
    // ============================================
    
    // element.style only returns INLINE styles
    console.log('Inline width:', box.style.width); // May be empty!
    
    // getComputedStyle returns ACTUAL computed values
    const computed = window.getComputedStyle(box);
    console.log('Computed width:', computed.width);
    console.log('Computed background:', computed.backgroundColor);
    console.log('Computed display:', computed.display);
    
    // ============================================
    // RESET STYLES
    // ============================================
    
    function resetStyles() {
      // Clear all inline styles
      box.style.cssText = '';
      box.textContent = 'Style Me!';
    }
    
    // ============================================
    // UTILITY: Apply styles object
    // ============================================
    
    function applyStyles(element, styles) {
      Object.assign(element.style, styles);
    }
    
    // Usage:
    applyStyles(box, {
      border: '3px dashed #e74c3c',
      padding: '20px',
      fontSize: '18px'
    });
    
    // ============================================
    // CSS CUSTOM PROPERTIES (Variables)
    // ============================================
    
    function changeTheme(color) {
      // Modify CSS variable on root
      document.documentElement.style.setProperty('--primary-color', color);
    }
    
    // Read CSS variable
    function getThemeColor() {
      return getComputedStyle(document.documentElement)
        .getPropertyValue('--primary-color');
    }
    
    // ============================================
    // DEMO OUTPUT
    // ============================================
    console.log('=== Style Property Demo ===');
    console.log('Click buttons to modify the box styles!');
  </script>
</body>
```
