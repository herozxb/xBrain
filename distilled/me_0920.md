## 1. Doubly Linked List Circular
```python
# PROBLEM: Implement a circular doubly linked list with insert, delete, and traversal operations
# TIME: O(1) for insert/delete at known position, O(n) for search SPACE: O(n)
class Node:
    """Node for circular doubly linked list."""
    def __init__(self, data):
        self.data = data
        self.prev = None
        self.next = None

class CircularDoublyLinkedList:
    """Circular doubly linked list implementation."""
    
    def __init__(self):
        # Sentinel node simplifies edge cases
        self.head = Node(None)
        self.head.prev = self.head
        self.head.next = self.head
        self.size = 0
    
    def insert_after(self, node, data):
        """Insert new node after given node. O(1)"""
        new_node = Node(data)
        new_node.prev = node
        new_node.next = node.next
        node.next.prev = new_node
        node.next = new_node
        self.size += 1
        return new_node
    
    def insert_front(self, data):
        """Insert at front of list. O(1)"""
        return self.insert_after(self.head, data)
    
    def insert_back(self, data):
        """Insert at back of list. O(1)"""
        return self.insert_after(self.head.prev, data)
    
    def remove(self, node):
        """Remove given node from list. O(1)"""
        if node is self.head:
            return None
        node.prev.next = node.next
        node.next.prev = node.prev
        self.size -= 1
        return node.data
    
    def traverse(self):
        """Traverse entire list. O(n)"""
        result = []
        current = self.head.next
        while current != self.head:
            result.append(current.data)
            current = current.next
        return result
```
**Why**: Circular doubly linked lists enable O(1) insertions/deletions at any position when node reference is known, commonly used in LRU caches and Fibonacci heaps.

## 2. Skip List Variants

```python
# PROBLEM: Implement a skip list for O(log n) search, insert, and delete operations
# TIME: O(log n) average for search/insert/delete SPACE: O(n)
import random

class SkipNode:
    """Node in skip list with multiple forward pointers."""
    def __init__(self, value=None, level=0):
        self.value = value
        self.forward = [None] * (level + 1)

class SkipList:
    """Probabilistic data structure for ordered elements."""
    
    def __init__(self, max_level=16, p=0.5):
        self.max_level = max_level
        self.p = p  # Probability for level promotion
        self.header = SkipNode(level=max_level)
        self.level = 0
    
    def random_level(self):
        """Generate random level for new node."""
        level = 0
        while random.random() < self.p and level < self.max_level:
            level += 1
        return level
    
    def search(self, value):
        """Find value in skip list. O(log n) average."""
        current = self.header
        # Start from top level, work down
        for i in range(self.level, -1, -1):
            while current.forward[i] and current.forward[i].value < value:
                current = current.forward[i]
        current = current.forward[0]
        return current if current and current.value == value else None
    
    def insert(self, value):
        """Insert value into skip list. O(log n) average."""
        update = [None] * (self.max_level + 1)
        current = self.header
        
        for i in range(self.level, -1, -1):
            while current.forward[i] and current.forward[i].value < value:
                current = current.forward[i]
            update[i] = current
        
        new_level = self.random_level()
        if new_level > self.level:
            for i in range(self.level + 1, new_level + 1):
                update[i] = self.header
            self.level = new_level
        
        new_node = SkipNode(value, new_level)
        for i in range(new_level + 1):
            new_node.forward[i] = update[i].forward[i]
            update[i].forward[i] = new_node
```
**Why**: Skip lists provide balanced tree performance without complex rebalancing logic, used in Redis sorted sets and Java ConcurrentSkipListMap.

## 4. Active Record Pattern
```python
# PROBLEM: Encapsulate database access within domain objects using Active Record
# TIME: O(1) for CRUD operations SPACE: O(n) for data storage
class ActiveRecord:
    """Base class providing CRUD operations."""
    
    _table_name = None
    _primary_key = "id"
    _columns = []
    _connection = None
    
    def __init__(self, **kwargs):
        # Store attributes dynamically
        for key, value in kwargs.items():
            setattr(self, key, value)
    
    @classmethod
    def set_connection(cls, connection):
        """Set database connection for all records."""
        cls._connection = connection
    
    @classmethod
    def find(cls, pk):
        """Find record by primary key. O(1)."""
        cursor = cls._connection.execute(
            f"SELECT * FROM {cls._table_name} WHERE {cls._primary_key} = ?",
            (pk,)
        )
        row = cursor.fetchone()
        if row:
            return cls._instantiate(row)
        return None
    
    @classmethod
    def where(cls, **conditions):
        """Find records matching conditions. O(n)."""
        where_clause = " AND ".join(f"{k} = ?" for k in conditions)
        values = tuple(conditions.values())
        cursor = cls._connection.execute(
            f"SELECT * FROM {cls._table_name} WHERE {where_clause}",
            values
        )
        return [cls._instantiate(row) for row in cursor.fetchall()]
    
    def save(self):
        """Insert or update record. O(1)."""
        if hasattr(self, self._primary_key):
            return self._update()
        return self._insert()
    
    def _insert(self):
        """Insert new record. O(1)."""
        columns = ", ".join(self._columns)
        placeholders = ", ".join("?" * len(self._columns))
        values = tuple(getattr(self, col) for col in self._columns)
        cursor = self._connection.execute(
            f"INSERT INTO {self._table_name} ({columns}) VALUES ({placeholders})",
            values
        )
        setattr(self, self._primary_key, cursor.lastrowid)
    
    def delete(self):
        """Delete record from database. O(1)."""
        pk_value = getattr(self, self._primary_key)
        self._connection.execute(
            f"DELETE FROM {self._table_name} WHERE {self._primary_key} = ?",
            (pk_value,)
        )
    
    @classmethod
    def _instantiate(cls, row):
        """Create instance from database row."""
        obj = cls()
        for i, col in enumerate(cls._columns + [cls._primary_key]):
            setattr(obj, col, row[i])
        return obj


class Product(ActiveRecord):
    """Product entity using Active Record pattern."""
    _table_name = "products"
    _columns = ["name", "price", "category"]
```
**Why**: Active Record provides simple, intuitive database access for smaller applications, popularized by Ruby on Rails and Django ORM.

## 5. Paxos Algorithm
```python
# PROBLEM: Achieve consensus in distributed system with faulty nodes using Paxos
# TIME: O(n) messages per round SPACE: O(n) for state storage
import threading
from enum import Enum

class PaxosState(Enum):
    PREPARE = "prepare"
    PROMISE = "promise"
    ACCEPT = "accept"
    ACCEPTED = "accepted"

class PaxosNode:
    """Single node in Paxos consensus protocol."""
    
    def __init__(self, node_id, quorum_size):
        self.node_id = node_id
        self.quorum_size = quorum_size
        self.proposal_number = 0
        self.promised_proposal = 0
        self.accepted_proposal = None
        self.accepted_value = None
        self.lock = threading.Lock()
    
    def prepare(self, proposal_num):
        """Phase 1a: Proposer sends prepare request."""
        with self.lock:
            if proposal_num > self.promised_proposal:
                self.promised_proposal = proposal_num
                return {
                    "type": PaxosState.PROMISE,
                    "proposal_num": proposal_num,
                    "accepted": (self.accepted_proposal, self.accepted_value),
                    "from": self.node_id
                }
            return None  # Rejected
    
    def accept(self, proposal_num, value):
        """Phase 2a: Proposer sends accept request."""
        with self.lock:
            if proposal_num >= self.promised_proposal:
                self.promised_proposal = proposal_num
                self.accepted_proposal = proposal_num
                self.accepted_value = value
                return {
                    "type": PaxosState.ACCEPTED,
                    "proposal_num": proposal_num,
                    "value": value,
                    "from": self.node_id
                }
            return None  # Rejected


class Proposer:
    """Paxos proposer coordinating consensus."""
    
    def __init__(self, proposer_id, acceptors):
        self.proposer_id = proposer_id
        self.acceptors = acceptors
        self.proposal_counter = 0
    
    def propose(self, value):
        """Execute full Paxos protocol. O(n) messages."""
        # Generate unique proposal number
        self.proposal_counter += 1
        proposal_num = (self.proposal_counter, self.proposer_id)
        
        # Phase 1: Prepare
        promises = []
        for acceptor in self.acceptors:
            response = acceptor.prepare(proposal_num)
            if response:
                promises.append(response)
        
        # Check quorum
        if len(promises) < len(self.acceptors) // 2 + 1:
            return None  # Failed to reach quorum
        
        # Use highest accepted value if exists
        accepted_values = [p["accepted"] for p in promises if p["accepted"][0]]
        if accepted_values:
            value = max(accepted_values, key=lambda x: x[0])[1]
        
        # Phase 2: Accept
        accepts = []
        for acceptor in self.acceptors:
            response = acceptor.accept(proposal_num, value)
            if response:
                accepts.append(response)
        
        # Check quorum for acceptance
        if len(accepts) >= len(self.acceptors) // 2 + 1:
            return value  # Consensus reached
        return None
```
**Why**: Paxos is the foundation of distributed consensus, used in Google Chubby, Spanner, and Apache ZooKeeper for fault-tolerant state replication.

## 6. Database Sharding

```python
# PROBLEM: Distribute data across multiple database servers using sharding
# TIME: O(1) for shard routing SPACE: O(n) distributed across shards
import hashlib
from collections import defaultdict

class ShardRouter:
    """Routes queries to appropriate shard based on shard key."""
    
    def __init__(self, num_shards):
        self.num_shards = num_shards
        self.shards = {i: defaultdict(dict) for i in range(num_shards)}
    
    def get_shard_id(self, key):
        """Determine shard for given key using consistent hashing. O(1)."""
        # Hash-based sharding for even distribution
        hash_value = int(hashlib.md5(str(key).encode()).hexdigest(), 16)
        return hash_value % self.num_shards
    
    def insert(self, table, key, data):
        """Insert data into correct shard. O(1)."""
        shard_id = self.get_shard_id(key)
        self.shards[shard_id][table][key] = data
        return shard_id
    
    def select(self, table, key):
        """Retrieve data from correct shard. O(1)."""
        shard_id = self.get_shard_id(key)
        return self.shards[shard_id][table].get(key)
    
    def get_shard_stats(self):
        """Return distribution statistics. O(n)."""
        return {
            shard_id: {
                "tables": len(tables),
                "total_keys": sum(len(rows) for rows in tables.values())
            }
            for shard_id, tables in self.shards.items()
        }


class RangeBasedSharding:
    """Sharding based on key ranges."""
    
    def __init__(self, ranges):
        # ranges: [(start, end, shard_id), ...]
        self.ranges = sorted(ranges, key=lambda x: x[0])
    
    def get_shard_id(self, key):
        """Find shard for key using binary search. O(log n)."""
        left, right = 0, len(self.ranges) - 1
        while left <= right:
            mid = (left + right) // 2
            start, end, shard_id = self.ranges[mid]
            if start <= key <= end:
                return shard_id
            elif key < start:
                right = mid - 1
            else:
                left = mid + 1
        return None  # Key outside all ranges


class DirectoryBasedSharding:
    """Sharding using lookup directory."""
    
    def __init__(self):
        self.directory = {}  # key -> shard_id
        self.shards = defaultdict(dict)
    
    def register_mapping(self, key, shard_id):
        """Register key to shard mapping. O(1)."""
        self.directory[key] = shard_id
    
    def get_shard_id(self, key):
        """Look up shard for key. O(1)."""
        return self.directory.get(key)
    
    def rebalance(self, from_shard, to_shard, keys):
        """Move keys between shards. O(k) for k keys."""
        for key in keys:
            self.directory[key] = to_shard
```
**Why**: Database sharding enables horizontal scaling by distributing data across multiple servers, essential for high-traffic applications like social networks and e-commerce platforms.

## 7. Hash Indexing
```python
# PROBLEM: Implement hash-based index for O(1) average lookups
# TIME: O(1) average for insert/lookup/delete SPACE: O(n)
class HashIndex:
    """Hash index using separate chaining for collision resolution."""
    
    def __init__(self, initial_size=16, load_factor=0.75):
        self.size = initial_size
        self.count = 0
        self.load_factor = load_factor
        self.buckets = [[] for _ in range(initial_size)]
    
    def _hash(self, key):
        """Compute bucket index for key."""
        return hash(key) % self.size
    
    def _resize(self):
        """Double the hash table size. O(n)."""
        old_buckets = self.buckets
        self.size *= 2
        self.buckets = [[] for _ in range(self.size)]
        self.count = 0
        
        for bucket in old_buckets:
            for key, value in bucket:
                self.insert(key, value)
    
    def insert(self, key, value):
        """Insert or update key-value pair. O(1) average."""
        if self.count / self.size >= self.load_factor:
            self._resize()
        
        index = self._hash(key)
        bucket = self.buckets[index]
        
        # Check if key exists
        for i, (k, v) in enumerate(bucket):
            if k == key:
                bucket[i] = (key, value)
                return
        
        bucket.append((key, value))
        self.count += 1
    
    def lookup(self, key):
        """Find value for key. O(1) average."""
        index = self._hash(key)
        bucket = self.buckets[index]
        
        for k, v in bucket:
            if k == key:
                return v
        return None
    
    def delete(self, key):
        """Remove key from index. O(1) average."""
        index = self._hash(key)
        bucket = self.buckets[index]
        
        for i, (k, v) in enumerate(bucket):
            if k == key:
                del bucket[i]
                self.count -= 1
                return True
        return False
    
    def range_query(self, start, end):
        """Hash indexes don't support efficient range queries. O(n)."""
        results = []
        for bucket in self.buckets:
            for key, value in bucket:
                if start <= key <= end:
                    results.append((key, value))
        return results


class LinearProbeHashIndex:
    """Hash index using linear probing for collision resolution."""
    
    def __init__(self, size=16):
        self.size = size
        self.keys = [None] * size
        self.values = [None] * size
        self.deleted = [False] * size
    
    def insert(self, key, value):
        """Insert using linear probing. O(1) average."""
        index = hash(key) % self.size
        first_deleted = -1
        
        while self.keys[index] is not None:
            if self.keys[index] == key:
                self.values[index] = value
                return
            if self.deleted[index] and first_deleted == -1:
                first_deleted = index
            index = (index + 1) % self.size
        
        insert_at = first_deleted if first_deleted != -1 else index
        self.keys[insert_at] = key
        self.values[insert_at] = value
        self.deleted[insert_at] = False
    
    def lookup(self, key):
        """Find value using linear probing. O(1) average."""
        index = hash(key) % self.size
        while self.keys[index] is not None:
            if self.keys[index] == key and not self.deleted[index]:
                return self.values[index]
            index = (index + 1) % self.size
        return None
```
**Why**: Hash indexes provide constant-time lookups for equality queries, used in PostgreSQL hash indexes and in-memory databases like Redis.

## 8. Snapshot Isolation
```python
# PROBLEM: Implement snapshot isolation for concurrent transactions without locks
# TIME: O(n) for conflict detection SPACE: O(n) for version storage
import time
from collections import defaultdict

class Version:
    """Represents a single version of a data item."""
    
    def __init__(self, value, start_ts, end_ts=float('inf')):
        self.value = value
        self.start_ts = start_ts
        self.end_ts = end_ts


class SnapshotIsolationManager:
    """Manages snapshot isolation for transactions."""
    
    def __init__(self):
        self.data = defaultdict(list)  # key -> [Version, ...]
        self.lock = threading.Lock()
        self.timestamp_counter = 0
        self.active_transactions = {}
    
    def begin_transaction(self):
        """Start new transaction with snapshot timestamp. O(1)."""
        with self.lock:
            self.timestamp_counter += 1
            snapshot_ts = self.timestamp_counter
            self.active_transactions[snapshot_ts] = set()
            return snapshot_ts
    
    def read(self, key, snapshot_ts):
        """Read value visible at snapshot timestamp. O(log v) for v versions."""
        versions = self.data.get(key, [])
        # Binary search for visible version
        for version in reversed(versions):
            if version.start_ts <= snapshot_ts < version.end_ts:
                return version.value
        return None
    
    def write(self, key, value, tx_ts):
        """Stage write in transaction's private workspace. O(1)."""
        if tx_ts not in self.active_transactions:
            raise ValueError("Unknown transaction")
        self.active_transactions[tx_ts].add((key, value))
    
    def commit(self, tx_ts):
        """Commit transaction with first-committer-wins rule. O(k) for k writes."""
        with self.lock:
            writes = self.active_transactions.get(tx_ts, set())
            
            # Check for write-write conflicts
            for key, _ in writes:
                versions = self.data.get(key, [])
                for version in versions:
                    if version.start_ts > tx_ts and version.end_ts == float('inf'):
                        del self.active_transactions[tx_ts]
                        return False  # Conflict, abort
            
            # Create new versions
            self.timestamp_counter += 1
            commit_ts = self.timestamp_counter
            
            for key, value in writes:
                # Close old versions
                for version in self.data[key]:
                    if version.end_ts == float('inf'):
                        version.end_ts = commit_ts
                # Add new version
                self.data[key].append(Version(value, commit_ts))
            
            del self.active_transactions[tx_ts]
            return True
    
    def rollback(self, tx_ts):
        """Rollback transaction. O(1)."""
        with self.lock:
            self.active_transactions.pop(tx_ts, None)
```
**Why**: Snapshot isolation provides consistent reads without blocking writers, used in PostgreSQL, Oracle, and MySQL (InnoDB) for better concurrency than strict serializability.

## 9. Process Pool Executor
```python
# PROBLEM: Execute CPU-bound tasks in parallel using process pool
# TIME: O(n/p) for n tasks with p processes SPACE: O(n) for task queue
from concurrent.futures import ProcessPoolExecutor, as_completed
import multiprocessing

class ParallelProcessor:
    """Manages parallel task execution using process pool."""
    
    def __init__(self, max_workers=None):
        self.max_workers = max_workers or multiprocessing.cpu_count()
    
    def map_parallel(self, func, items, chunksize=1):
        """Apply function to items in parallel. O(n/p) time."""
        with ProcessPoolExecutor(max_workers=self.max_workers) as executor:
            results = list(executor.map(func, items, chunksize=chunksize))
        return results
    
    def map_with_progress(self, func, items, callback=None):
        """Execute with progress tracking. O(n/p) time."""
        results = []
        with ProcessPoolExecutor(max_workers=self.max_workers) as executor:
            futures = {executor.submit(func, item): item for item in items}
            
            for future in as_completed(futures):
                item = futures[future]
                try:
                    result = future.result()
                    results.append((item, result))
                    if callback:
                        callback(item, result)
                except Exception as e:
                    results.append((item, e))
        
        return results
    
    def parallel_reduce(self, mapper, reducer, items):
        """Map-reduce pattern implementation. O(n/p + log p)."""
        with ProcessPoolExecutor(max_workers=self.max_workers) as executor:
            # Map phase
            mapped = list(executor.map(mapper, items))
            
            # Reduce phase (can be parallelized with tree reduction)
            result = mapped[0]
            for value in mapped[1:]:
                result = reducer(result, value)
            
            return result


def parallel_sort(data, num_processes=None):
    """Sort large dataset using parallel merge sort. O(n log n / p)."""
    num_processes = num_processes or multiprocessing.cpu_count()
    
    def merge(left, right):
        result = []
        i = j = 0
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                result.append(left[i])
                i += 1
            else:
                result.append(right[j])
                j += 1
        result.extend(left[i:])
        result.extend(right[j:])
        return result
    
    if len(data) <= 1000 or num_processes == 1:
        return sorted(data)
    
    # Split and sort chunks in parallel
    chunk_size = len(data) // num_processes
    chunks = [data[i:i+chunk_size] for i in range(0, len(data), chunk_size)]
    
    with ProcessPoolExecutor(max_workers=num_processes) as executor:
        sorted_chunks = list(executor.map(sorted, chunks))
    
    # Merge sorted chunks
    while len(sorted_chunks) > 1:
        merged = []
        for i in range(0, len(sorted_chunks), 2):
            if i + 1 < len(sorted_chunks):
                merged.append(merge(sorted_chunks[i], sorted_chunks[i+1]))
            else:
                merged.append(sorted_chunks[i])
        sorted_chunks = merged
    
    return sorted_chunks[0]
```
**Why**: Process pools enable true parallelism for CPU-bound tasks in Python, bypassing GIL limitations, essential for data processing and scientific computing.

## 10. asyncio Patterns
```python
# PROBLEM: Implement common async patterns for concurrent I/O operations
# TIME: O(n/c) for n concurrent operations with c concurrency limit SPACE: O(c)
import asyncio
from asyncio import Semaphore, Queue, gather, create_task

class AsyncRateLimiter:
    """Rate limiter using token bucket algorithm for async code."""
    
    def __init__(self, rate, capacity):
        self.rate = rate  # Tokens per second
        self.capacity = capacity
        self.tokens = capacity
        self.last_update = asyncio.get_event_loop().time()
        self.lock = asyncio.Lock()
    
    async def acquire(self):
        """Wait until a token is available. O(1) when tokens available."""
        async with self.lock:
            now = asyncio.get_event_loop().time()
            elapsed = now - self.last_update
            self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)
            self.last_update = now
            
            if self.tokens < 1:
                wait_time = (1 - self.tokens) / self.rate
                await asyncio.sleep(wait_time)
                self.tokens = 0
            else:
                self.tokens -= 1


async def fetch_with_retry(session, url, max_retries=3, backoff=1):
    """Fetch URL with exponential backoff retry. O(2^r) worst case."""
    for attempt in range(max_retries):
        try:
            async with session.get(url) as response:
                return await response.json()
        except Exception as e:
            if attempt == max_retries - 1:
                raise
            await asyncio.sleep(backoff * (2 ** attempt))


async def bounded_gather(tasks, limit=10):
    """Execute tasks with concurrency limit. O(n/c) for c limit."""
    semaphore = Semaphore(limit)
    
    async def bounded_task(task):
        async with semaphore:
            return await task
    
    return await gather(*[bounded_task(t) for t in tasks])


class AsyncProducerConsumer:
    """Producer-consumer pattern with async queue."""
    
    def __init__(self, num_producers=2, num_consumers=3, queue_size=100):
        self.queue = Queue(maxsize=queue_size)
        self.num_producers = num_producers
        self.num_consumers = num_consumers
    
    async def producer(self, producer_id, items):
        """Produce items to queue. O(n)."""
        for item in items:
            await self.queue.put((producer_id, item))
            await asyncio.sleep(0.01)  # Simulate work
        await self.queue.put(None)  # Signal completion
    
    async def consumer(self, consumer_id, process_func):
        """Consume items from queue. O(n/c) for c consumers."""
        while True:
            item = await self.queue.get()
            if item is None:
                self.queue.task_done()
                break
            producer_id, data = item
            await process_func(consumer_id, producer_id, data)
            self.queue.task_done()
    
    async def run(self, producer_items, process_func):
        """Run producer-consumer system. O(n/c)."""
        producers = [
            create_task(self.producer(i, producer_items[i]))
            for i in range(self.num_producers)
        ]
        consumers = [
            create_task(self.consumer(i, process_func))
            for i in range(self.num_consumers)
        ]
        
        await gather(*producers)
        # Send one None per consumer
        for _ in range(self.num_consumers - self.num_producers):
            await self.queue.put(None)
        
        await gather(*consumers)


async def async_context_pool(resources, max_concurrent=5):
    """Pool of async context managers. O(c) for c concurrent."""
    semaphore = Semaphore(max_concurrent)
    pool = asyncio.Queue()
    
    # Initialize pool
    for resource in resources:
        await pool.put(resource)
    
    async def use_resource(func):
        async with semaphore:
            resource = await pool.get()
            try:
                result = await func(resource)
                return result
            finally:
                await pool.put(resource)
    
    return use_resource
```
