## LCP Array Construction

```python
# PROBLEM: Build Longest Common Prefix array from suffix array in O(n)
# TIME: O(n) SPACE: O(n)
def build_lcp_array(text: str, suffix_array: list[int]) -> list[int]:
    """
    Kasai's algorithm to construct LCP array.
    LCP[i] = longest common prefix between suffix at SA[i] and SA[i-1].
    """
    n = len(text)
    rank = [0] * n  # rank[i] = position of suffix starting at i in SA
    
    # Build inverse suffix array (rank array)
    for i, sa_idx in enumerate(suffix_array):
        rank[sa_idx] = i
    
    lcp = [0] * n
    k = 0  # current LCP length
    
    for i in range(n):
        if rank[i] == 0:
            k = 0
            continue
        
        # Find the suffix that precedes suffix[i] in SA
        j = suffix_array[rank[i] - 1]
        
        # Extend the common prefix
        while i + k < n and j + k < n and text[i + k] == text[j + k]:
            k += 1
        
        lcp[rank[i]] = k
        
        # Next suffix will have LCP at least k-1
        if k > 0:
            k -= 1
    
    return lcp

# Example usage
text = "banana"
sa = [5, 3, 1, 0, 4, 2]  # suffix array for "banana"
lcp = build_lcp_array(text, sa)
# lcp = [0, 1, 3, 0, 0, 2]
```
**Why**: Essential for string processing, pattern matching, and substring queries in O(n log n) or O(n) time.

## Suffix Array Construction

```python
# PROBLEM: Build suffix array in O(n log² n) using sorting
# TIME: O(n log² n) SPACE: O(n)
def build_suffix_array(text: str) -> list[int]:
    """
    Build suffix array using prefix doubling algorithm.
    SA[i] = starting index of i-th lexicographically smallest suffix.
    """
    n = len(text)
    # Initial ranks based on single characters
    suffixes = [(text[i], i) for i in range(n)]
    suffixes.sort()
    
    # Map suffix index to its rank
    rank = [0] * n
    for i, (_, idx) in enumerate(suffixes):
        rank[idx] = i
    
    sa = [idx for _, idx in suffixes]
    
    k = 1
    while k < n:
        # Sort by (rank[i], rank[i+k])
        suffixes.sort(key=lambda x: (rank[x], rank[x + k] if x + k < n else -1))
        
        # Update suffix array
        sa = [idx for _, idx in suffixes]
        
        # Recompute ranks
        new_rank = [0] * n
        new_rank[sa[0]] = 0
        for i in range(1, n):
            prev, curr = sa[i - 1], sa[i]
            prev_key = (rank[prev], rank[prev + k] if prev + k < n else -1)
            curr_key = (rank[curr], rank[curr + k] if curr + k < n else -1)
            new_rank[curr] = new_rank[prev] + (1 if prev_key != curr_key else 0)
        
        rank = new_rank
        
        # All ranks unique means we're done
        if rank[sa[-1]] == n - 1:
            break
        k *= 2
    
    return sa

# Example
text = "banana"
print(build_suffix_array(text))  # [5, 3, 1, 0, 4, 2]
```
**Why**: Compact alternative to suffix tree. Enables fast substring search, longest repeated substring, and more.

## Edmonds-Karp Maximum Flow

```python
# PROBLEM: Find maximum flow from source to sink in a flow network
# TIME: O(VE²) SPACE: O(V²)
from collections import deque

def edmonds_karp(capacity: list[list[int]], source: int, sink: int) -> int:
    """
    Edmonds-Karp algorithm using BFS to find augmenting paths.
    Returns the maximum flow value.
    """
    n = len(capacity)
    flow = [[0] * n for _ in range(n)]
    max_flow = 0
    
    def bfs() -> tuple[int, list[int]]:
        """Find augmenting path using BFS. Returns flow and parent array."""
        parent = [-1] * n
        parent[source] = source
        min_capacity = [float('inf')] * n
        
        queue = deque([source])
        while queue:
            u = queue.popleft()
            
            for v in range(n):
                # Residual capacity > 0 and not visited
                residual = capacity[u][v] - flow[u][v]
                if residual > 0 and parent[v] == -1:
                    parent[v] = u
                    min_capacity[v] = min(min_capacity[u], residual)
                    
                    if v == sink:
                        return min_capacity[v], parent
                    
                    queue.append(v)
        
        return 0, parent  # No augmenting path
    
    while True:
        augment, parent = bfs()
        
        if augment == 0:
            break  # No more augmenting paths
        
        # Augment flow along the path
        v = sink
        while v != source:
            u = parent[v]
            flow[u][v] += augment
            flow[v][u] -= augment  # Back edge for residual graph
            v = u
        
        max_flow += augment
    
    return max_flow

# Example: Network with 4 nodes
cap = [
    [0, 3, 2, 0],
    [0, 0, 5, 2],
    [0, 0, 0, 3],
    [0, 0, 0, 0]
]
print(edmonds_karp(cap, 0, 3))  # Output: 5
```
**Why**: Fundamental network flow algorithm. Used in bipartite matching, min-cut, and resource allocation.

## Point in Polygon

```python
# PROBLEM: Determine if a point lies inside, outside, or on boundary of polygon
# TIME: O(n) SPACE: O(1)
def point_in_polygon(point: tuple[float, float], 
                     polygon: list[tuple[float, float]]) -> str:
    """
    Ray casting algorithm. Cast horizontal ray from point to infinity.
    Count intersections with polygon edges.
    - Odd count: inside
    - Even count: outside
    """
    x, y = point
    n = len(polygon)
    inside = False
    
    def on_segment(px, py, x1, y1, x2, y2) -> bool:
        """Check if point (px, py) lies on segment from (x1,y1) to (x2,y2)."""
        cross = (y2 - y1) * (px - x1) - (x2 - x1) * (py - y1)
        if abs(cross) > 1e-9:
            return False
        return (min(x1, x2) <= px <= max(x1, x2) and 
                min(y1, y2) <= py <= max(y1, y2))
    
    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]
        
        # Check if point is on current edge
        if on_segment(x, y, x1, y1, x2, y2):
            return "BOUNDARY"
        
        # Check if ray from point crosses this edge
        if ((y1 > y) != (y2 > y)):  # Edge straddles horizontal line at y
            # Compute x-coordinate of intersection
            x_intersect = x1 + (y - y1) * (x2 - x1) / (y2 - y1)
            
            if x < x_intersect:
                inside = not inside
    
    return "INSIDE" if inside else "OUTSIDE"

# Example: Triangle
polygon = [(0, 0), (2, 0), (1, 2)]
print(point_in_polygon((1, 0.5), polygon))   # INSIDE
print(point_in_polygon((1, 0), polygon))     # BOUNDARY
print(point_in_polygon((3, 1), polygon))     # OUTSIDE
```
**Why**: Core computational geometry operation. Used in GIS, collision detection, and graphics.

## Minimum Spanning Circle (Welzl's Algorithm)

```python
# PROBLEM: Find smallest circle enclosing all points
# TIME: O(n) expected SPACE: O(n)
import random
import math

def minimum_spanning_circle(points: list[tuple[float, float]]) -> tuple[tuple[float, float], float]:
    """
    Welzl's randomized algorithm for minimum enclosing circle.
    Returns (center, radius).
    """
    def circle_from_1(a: tuple[float, float]) -> tuple[tuple[float, float], float]:
        return a, 0.0
    
    def circle_from_2(a: tuple[float, float], b: tuple[float, float]) -> tuple[tuple[float, float], float]:
        cx = (a[0] + b[0]) / 2
        cy = (a[1] + b[1]) / 2
        r = math.dist(a, b) / 2
        return (cx, cy), r
    
    def circle_from_3(a: tuple[float, float], b: tuple[float, float], 
                      c: tuple[float, float]) -> tuple[tuple[float, float], float]:
        """Circle passing through three points."""
        ax, ay = a
        bx, by = b
        cx, cy = c
        
        d = 2 * (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by))
        
        if abs(d) < 1e-9:
            # Points are collinear, return circle from farthest pair
            return max([circle_from_2(a, b), circle_from_2(b, c), 
                       circle_from_2(a, c)], key=lambda x: x[1])
        
        ux = ((ax**2 + ay**2) * (by - cy) + (bx**2 + by**2) * (cy - ay) + 
              (cx**2 + cy**2) * (ay - by)) / d
        uy = ((ax**2 + ay**2) * (cx - bx) + (bx**2 + by**2) * (ax - cx) + 
              (cx**2 + cy**2) * (bx - ax)) / d
        
        center = (ux, uy)
        radius = math.dist(center, a)
        return center, radius
    
    def is_inside(point, center, radius):
        return math.dist(point, center) <= radius + 1e-9
    
    def welzl(pts: list, boundary: list, n: int) -> tuple[tuple[float, float], float]:
        if n == 0 or len(boundary) == 3:
            if len(boundary) == 0:
                return ((0, 0), 0)
            elif len(boundary) == 1:
                return circle_from_1(boundary[0])
            elif len(boundary) == 2:
                return circle_from_2(boundary[0], boundary[1])
            else:
                return circle_from_3(boundary[0], boundary[1], boundary[2])
        
        # Randomly pick a point
        idx = random.randint(0, n - 1)
        p = pts[idx]
        pts[idx], pts[n - 1] = pts[n - 1], pts[idx]
        
        # Get circle without this point
        circle = welzl(pts, boundary, n - 1)
        
        if is_inside(p, circle[0], circle[1]):
            return circle
        
        # Point is outside, must be on boundary
        boundary.append(p)
        return welzl(pts, boundary, n - 1)
    
    shuffled = points.copy()
    random.shuffle(shuffled)
    return welzl(shuffled, [], len(shuffled))

# Example
points = [(0, 0), (1, 0), (0, 1), (1, 1)]
center, radius = minimum_spanning_circle(points)
print(f"Center: {center}, Radius: {radius:.3f}")
```
**Why**: Optimal solution for facility location, clustering, and collision detection problems.

## Convex Polygon Intersection

```python
# PROBLEM: Find intersection of two convex polygons
# TIME: O(n + m) SPACE: O(n + m)
def convex_polygon_intersection(
    poly1: list[tuple[float, float]], 
    poly2: list[tuple[float, float]]
) -> list[tuple[float, float]]:
    """
    Sutherland-Hodgman algorithm for clipping convex polygons.
    Returns vertices of intersection polygon.
    """
    def cross(o, a, b):
        """Cross product of OA and OB vectors."""
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
    
    def line_intersection(p1, p2, p3, p4):
        """Find intersection of lines p1-p2 and p3-p4."""
        d1 = (p2[0] - p1[0], p2[1] - p1[1])
        d2 = (p4[0] - p3[0], p4[1] - p3[1])
        
        cross_d = d1[0] * d2[1] - d1[1] * d2[0]
        if abs(cross_d) < 1e-9:
            return None  # Parallel
        
        t = ((p3[0] - p1[0]) * d2[1] - (p3[1] - p1[1]) * d2[0]) / cross_d
        
        return (p1[0] + t * d1[0], p1[1] + t * d1[1])
    
    def clip_polygon(subject, clip_edge_start, clip_edge_end):
        """Clip subject polygon against a single edge."""
        if not subject:
            return []
        
        output = []
        n = len(subject)
        
        for i in range(n):
            current = subject[i]
            next_v = subject[(i + 1) % n]
            
            # Determine if points are inside (left of clip edge)
            current_inside = cross(clip_edge_start, clip_edge_end, current) >= 0
            next_inside = cross(clip_edge_start, clip_edge_end, next_v) >= 0
            
            if current_inside:
                output.append(current)
                if not next_inside:
                    # Exiting: add intersection
                    inter = line_intersection(current, next_v, clip_edge_start, clip_edge_end)
                    if inter:
                        output.append(inter)
            elif next_inside:
                # Entering: add intersection
                inter = line_intersection(current, next_v, clip_edge_start, clip_edge_end)
                if inter:
                    output.append(inter)
        
        return output
    
    result = poly1[:]
    
    # Clip against each edge of poly2
    for i in range(len(poly2)):
        edge_start = poly2[i]
        edge_end = poly2[(i + 1) % len(poly2)]
        result = clip_polygon(result, edge_start, edge_end)
    
    return result

# Example
poly1 = [(0, 0), (2, 0), (2, 2), (0, 2)]  # Square
poly2 = [(1, 1), (3, 1), (3, 3), (1, 3)]  # Overlapping square
intersection = convex_polygon_intersection(poly1, poly2)
print(intersection)  # [(1, 1), (2, 1), (2, 2), (1, 2)]
```
**Why**: Critical for collision detection, computer graphics, and GIS applications.

## Maximum Clique (Bron-Kerbosch)

```python
# PROBLEM: Find maximum clique in a graph (largest complete subgraph)
# TIME: O(3^(n/3)) worst case SPACE: O(n)
def maximum_clique(graph: dict[int, set[int]]) -> set[int]:
    """
    Bron-Kerbosch algorithm with pivoting.
    graph: adjacency list as dict {vertex: set of neighbors}
    """
    max_clique = set()
    
    def bron_kerbosch(R: set, P: set, X: set):
        nonlocal max_clique
        
        if not P and not X:
            # R is a maximal clique
            if len(R) > len(max_clique):
                max_clique = R.copy()
            return
        
        # Choose pivot to minimize branching
        pivot = max(P | X, key=lambda v: len(P & graph.get(v, set())))
        
        for v in P - graph.get(pivot, set()):
            neighbors = graph.get(v, set())
            bron_kerbosch(R | {v}, P & neighbors, X & neighbors)
            P = P - {v}
            X = X | {v}
    
    vertices = set(graph.keys())
    bron_kerbosch(set(), vertices, set())
    return max_clique

# Example: Triangle with an extra vertex
graph = {
    0: {1, 2},
    1: {0, 2, 3},
    2: {0, 1},
    3: {1}
}
print(maximum_clique(graph))  # {0, 1, 2}
```
**Why**: NP-hard but fundamental for social network analysis, bioinformatics, and scheduling.

## Traveling Salesman Branch and Bound

```python
# PROBLEM: Find shortest Hamiltonian cycle visiting all cities
# TIME: O(n!) worst, O(n²·2^n) with DP SPACE: O(n)
import math
from typing import Optional

def tsp_branch_and_bound(dist: list[list[float]]) -> tuple[float, list[int]]:
    """
    Branch and bound TSP solver.
    dist[i][j] = distance from city i to city j.
    Returns (min_cost, path).
    """
    n = len(dist)
    best_cost = [float('inf')]
    best_path = [None]
    
    def reduced_cost(matrix: list[list[float]], rows: set, cols: set) -> float:
        """Calculate lower bound using row/column reduction."""
        cost = 0
        
        # Row reduction
        for i in range(n):
            if i not in rows:
                min_val = min(matrix[i][j] for j in range(n) if j not in cols)
                if min_val != float('inf') and min_val > 0:
                    cost += min_val
                    for j in range(n):
                        if j not in cols:
                            matrix[i][j] -= min_val
        
        # Column reduction
        for j in range(n):
            if j not in cols:
                min_val = min(matrix[i][j] for i in range(n) if i not in rows)
                if min_val != float('inf') and min_val > 0:
                    cost += min_val
                    for i in range(n):
                        if i not in rows:
                            matrix[i][j] -= min_val
        
        return cost
    
    def solve(path: list, cost: float, visited: set):
        if len(path) == n:
            # Complete tour, return to start
            total = cost + dist[path[-1]][path[0]]
            if total < best_cost[0]:
                best_cost[0] = total
                best_path[0] = path[:]
            return
        
        # Create reduced matrix for lower bound
        matrix = [row[:] for row in dist]
        
        # Branch on unvisited cities
        last = path[-1]
        candidates = []
        
        for next_city in range(n):
            if next_city not in visited:
                new_cost = cost + dist[last][next_city]
                if new_cost < best_cost[0]:  # Pruning
                    candidates.append((next_city, new_cost))
        
        # Sort by edge cost for better pruning
        candidates.sort(key=lambda x: x[1])
        
        for next_city, new_cost in candidates:
            if new_cost >= best_cost[0]:
                break  # Prune
            path.append(next_city)
            visited.add(next_city)
            solve(path, new_cost, visited)
            path.pop()
            visited.remove(next_city)
    
    # Start from city 0
    solve([0], 0, {0})
    
    return best_cost[0], best_path[0]

# Example
dist = [
    [0, 10, 15, 20],
    [10, 0, 35, 25],
    [15, 35, 0, 30],
    [20, 25, 30, 0]
]
cost, path = tsp_branch_and_bound(dist)
print(f"Cost: {cost}, Path: {path}")  # Cost: 80, Path: [0, 1, 3, 2]
```
**Why**: Classic NP-hard optimization. Branch and bound provides exact solutions for small-medium instances.

## Chinese Postman Problem

```python
# PROBLEM: Find shortest closed walk covering all edges (minimum Eulerian circuit)
# TIME: O(n³) SPACE: O(n²)
from collections import defaultdict
import heapq

def chinese_postman(graph: dict[int, list[tuple[int, float]]]) -> tuple[float, list[int]]:
    """
    Find shortest closed walk that traverses every edge at least once.
    1. Find odd-degree vertices
    2. Pair them optimally (minimum weight perfect matching)
    3. Add duplicate edges for pairs
    4. Find Eulerian circuit
    """
    # Build degree map
    degree = defaultdict(int)
    edges = []
    for u in graph:
        for v, w in graph[u]:
            degree[u] += 1
            edges.append((u, v, w))
    
    # Find odd-degree vertices
    odd_vertices = [v for v in degree if degree[v] % 2 == 1]
    
    # Floyd-Warshall for shortest paths between all pairs
    all_vertices = list(graph.keys())
    idx = {v: i for i, v in enumerate(all_vertices)}
    n = len(all_vertices)
    
    dist_matrix = [[float('inf')] * n for _ in range(n)]
    for i in range(n):
        dist_matrix[i][i] = 0
    
    for u in graph:
        for v, w in graph[u]:
            dist_matrix[idx[u]][idx[v]] = min(dist_matrix[idx[u]][idx[v]], w)
            dist_matrix[idx[v]][idx[u]] = min(dist_matrix[idx[v]][idx[u]], w)
    
    for k in range(n):
        for i in range(n):
            for j in range(n):
                dist_matrix[i][j] = min(dist_matrix[i][j], 
                                       dist_matrix[i][k] + dist_matrix[k][j])
    
    # Minimum weight perfect matching on odd vertices (greedy approximation)
    def min_matching_greedy(vertices):
        if not vertices:
            return [], 0
        
        vertices = vertices[:]
        matching = []
        total = 0
        
        while len(vertices) >= 2:
            # Find closest pair
            best_pair = None
            best_dist = float('inf')
            
            for i in range(len(vertices)):
                for j in range(i + 1, len(vertices)):
                    d = dist_matrix[idx[vertices[i]]][idx[vertices[j]]]
                    if d < best_dist:
                        best_dist = d
                        best_pair = (vertices[i], vertices[j])
            
            matching.append(best_pair)
            total += best_dist
            vertices.remove(best_pair[0])
            vertices.remove(best_pair[1])
        
        return matching, total
    
    matching, extra_cost = min_matching_greedy(odd_vertices)
    
    # Build Eulerian multigraph by duplicating matched edge paths
    # For simplicity, just compute total cost
    original_cost = sum(w for _, _, w in edges) // 2  # Each edge counted twice
    
    # Hierholzer's algorithm for Eulerian circuit (simplified)
    def find_eulerian_circuit():
        adj = defaultdict(list)
        for u in graph:
            for v, w in graph[u]:
                adj[u].append((v, w))
        
        # Add duplicate edges for matching
        for u, v in matching:
            adj[u].append((v, 0))  # Placeholder weight
            adj[v].append((u, 0))
        
        stack = [all_vertices[0]]
        circuit = []
        
        while stack:
            v = stack[-1]
            if adj[v]:
                u, _ = adj[v].pop()
                # Remove reverse edge
                for i, (node, _) in enumerate(adj[u]):
                    if node == v:
                        adj[u].pop(i)
                        break
                stack.append(u)
            else:
                circuit.append(stack.pop())
        
        return circuit[::-1]
    
    circuit = find_eulerian_circuit()
    total_cost = original_cost + extra_cost
    
    return total_cost, circuit

# Example
graph = {
    0: [(1, 1), (2, 2)],
    1: [(0, 1), (2, 1), (3, 3)],
    2: [(0, 2), (1, 1), (3, 1)],
    3: [(1, 3), (2, 1)]
}
cost, path = chinese_postman(graph)
print(f"Cost: {cost}, Path: {path}")
```
**Why**: Route optimization for mail delivery, garbage collection, and street sweeping.

## Minimum Cost Flow (Successive Shortest Path)

```python
# PROBLEM: Find min-cost flow from source to sink with capacity constraints
# TIME: O(F·E·logV) SPACE: O(V²)
import heapq

def min_cost_flow(
    capacity: list[list[int]], 
    cost: list[list[int]], 
    source: int, 
    sink: int, 
    flow_needed: int
) -> tuple[int, int]:
    """
    Successive shortest path algorithm.
    Returns (actual_flow, total_cost).
    """
    n = len(capacity)
    flow = [[0] * n for _ in range(n)]
    total_cost = 0
    total_flow = 0
    
    def dijkstra() -> tuple[int, list[int]]:
        """Find shortest augmenting path by cost."""
        dist = [float('inf')] * n
        parent = [-1] * n
        dist[source] = 0
        pq = [(0, source)]
        
        while pq:
            d, u = heapq.heappop(pq)
            
            if d > dist[u]:
                continue
            
            for v in range(n):
                residual = capacity[u][v] - flow[u][v]
                if residual > 0:
                    edge_cost = cost[u][v]
                    # Handle negative costs with potentials in full implementation
                    if dist[u] + edge_cost < dist[v]:
                        dist[v] = dist[u] + edge_cost
                        parent[v] = u
                        heapq.heappush(pq, (dist[v], v))
        
        return dist[sink], parent
    
    while total_flow < flow_needed:
        path_cost, parent = dijkstra()
        
        if parent[sink] == -1:
            break  # No more augmenting paths
        
        # Find bottleneck capacity
        augment = flow_needed - total_flow
        v = sink
        while v != source:
            u = parent[v]
            augment = min(augment, capacity[u][v] - flow[u][v])
            v = u
        
        # Augment flow
        v = sink
        while v != source:
            u = parent[v]
            flow[u][v] += augment
            flow[v][u] -= augment
            total_cost += augment * cost[u][v]
            v = u
        
        total_flow += augment
    
    return total_flow, total_cost

# Example: Send 4 units from 0 to 3
capacity = [
    [0, 3, 2, 0],
    [0, 0, 5, 2],
    [0, 0, 0, 3],
    [0, 0, 0, 0]
]
cost = [
    [0, 1, 5, 0],
    [0, 0, 1, 3],
    [0, 0, 0, 1],
    [0, 0, 0, 0]
]
flow, total = min_cost_flow(capacity, cost, 0, 3, 4)
print(f"Flow: {flow}, Cost: {total}")  # Flow: 4, Cost: 15
```
**Why**: Generalizes max-flow with costs. Used in transportation, assignment, and supply chain optimization.
