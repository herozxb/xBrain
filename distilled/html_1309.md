  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>:nth-child Selector</title>
  <style>
    /*
      :nth-child(an+b) selects elements based on position
      a = cycle size, n = counter (0,1,2...), b = offset
      
      Common patterns:
      - :nth-child(odd) or :nth-child(2n+1) - odd items
      - :nth-child(even) or :nth-child(2n) - even items
      - :nth-child(3n) - every 3rd item
      - :nth-child(n+4) - items 4 and beyond
      - :nth-child(-n+3) - first 3 items
    */
    
    .list {
      list-style: none;
      padding: 0;
    }
    
    .list li {
      padding: 12px 16px;
      margin: 4px 0;
      border-radius: 4px;

    }
    
    /* Highlight every 5th item */

      background: #fff3e0;

    /* Style first 3 items differently */

    /* Select items from position 4 onwards */

## css_box_sizing_border_box

<!-- PROBLEM: How to include padding and border in element's total width/height -->
<!-- TIME: O(1) SPACE: O(1) -->

    /*
      box-sizing controls how total width/height is calculated
      
      content-box (default):
        width = content only
        total width = width + padding + border
        
      border-box:
        width = content + padding + border
        total width = specified width
        content shrinks to accommodate padding/border
        
      Best practice: Apply border-box to all elements
    */
    
    /* Universal border-box reset */
    *, *::before, *::after {
      box-sizing: border-box;
    }
    
    .comparison {
      display: flex;
      gap: 20px;
      margin: 20px 0;
    }
    
    .content-box {
      box-sizing: content-box;
      width: 200px;
      padding: 20px;
      border: 5px solid #e91e63;
      background: #fce4ec;
      
      /* Total width = 200 + 40 (padding) + 10 (border) = 250px */
    }
    
    .border-box {
      box-sizing: border-box;
      width: 200px;
      padding: 20px;
      border: 5px solid #4caf50;
      background: #e8f5e9;
      
      /* Total width = 200px exactly */
      /* Content width = 200 - 40 (padding) - 10 (border) = 150px */
    }

      display: flex;

      background: #e3f2fd;
      text-align: center;
    }
  </style>

      - ...values: the interpolated expressions
      
      The strings array has a raw property with unprocessed escape sequences
    */

    // Basic tagged template
    function highlight(strings, ...values) {
      // strings: ["The ", " is ", " years old"]
      // values: ["Alice", 30]
      
      let result = strings[0];
      for (let i = 0; i < values.length; i++) {
        result += `<mark>${values[i]}</mark>` + strings[i + 1];
      }
      return result;
    }

    const name = "Alice";
    const age = 30;
    const highlighted = highlight`The ${name} is ${age} years old`;

    // Safe HTML escaping
    function safeHTML(strings, ...values) {
      let result = strings[0];
      for (let i = 0; i < values.length; i++) {
        const escaped = String(values[i])
          .replace(/&/g, '&amp;')
          .replace(/</g, '&lt;')
          .replace(/>/g, '&gt;')
          .replace(/"/g, '&quot;')
          .replace(/'/g, '&#039;');
        result += escaped + strings[i + 1];
      }
      return result;
    }
