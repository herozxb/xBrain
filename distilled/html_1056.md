## js_element_hasAttribute

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">

        // before attempting to get or manipulate attribute values
    </script>
```

**Why**: hasAttribute() is essential for defensive programming - checking if an attribute exists before accessing it prevents errors and enables conditional attribute-based logic.

```html

<body>
    <!-- PROBLEM: Stop all subsequent event handlers from executing -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <button id="myButton">Click Me</button>
    <div id="log"></div>

    <script>
        const button = document.getElementById('myButton');
        const logDiv = document.getElementById('log');
        
        function log(message) {
            logDiv.innerHTML += message + '<br>';
        }
        
        // First click handler
        button.addEventListener('click', function(e) {
            log('Handler 1 executed');
            // Stop all other handlers on this element from running
            e.stopImmediatePropagation();
        });
        
        // Second click handler - will NOT execute
        button.addEventListener('click', function(e) {
            log('Handler 2 - NEVER SHOWN');
        });
        
        // Third click handler - will NOT execute
        button.addEventListener('click', function(e) {
            log('Handler 3 - NEVER SHOWN');
        });
        
        // Why: Unlike stopPropagation (which only stops bubbling),
        // stopImmediatePropagation also prevents other handlers on same element
    </script>
</body>
</html>
```

**Why**: stopImmediatePropagation() is crucial when you need to completely halt event processing, preventing not just bubbling but also other handlers on the same element from executing.

## js_sessionStorage_clear

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>sessionStorage.clear() Demo</title>
</head>
<body>
    <!-- PROBLEM: Clear all session storage data -->
    <!-- TIME: O(n) SPACE: O(1) where n = number of items -->
    
    <h2>Session Storage Manager</h2>
    
    <input type="text" id="keyInput" placeholder="Key">
    <input type="text" id="valueInput" placeholder="Value">
    <button onclick="addItem()">Add Item</button>
    <button onclick="clearStorage()">Clear All</button>
    
    <div id="display"></div>

    <script>
        function addItem() {
            const key = document.getElementById('keyInput').value;
            const value = document.getElementById('valueInput').value;
            
            if (key && value) {
                sessionStorage.setItem(key, value);
                updateDisplay();
            }
        }
        
        function clearStorage() {
            // Clear all session storage for this origin
            sessionStorage.clear();
            updateDisplay();
        }
        
        function updateDisplay() {
            const display = document.getElementById('display');
            display.innerHTML = '<h3>Session Storage Contents:</h3>';
            
            if (sessionStorage.length === 0) {
                display.innerHTML += '<p>Empty</p>';
            } else {
                for (let i = 0; i < sessionStorage.length; i++) {
                    const key = sessionStorage.key(i);
                    const value = sessionStorage.getItem(key);
                    display.innerHTML += `<p>${key}: ${value}</p>`;
                }
            }
        }
        
        updateDisplay();
        
        // Why: Essential for user logout, form resets, or clearing
        // temporary page state without affecting other tabs
    </script>
</body>
</html>
```

**Why**: sessionStorage.clear() provides a clean way to reset all session data, useful for logout flows, form resets, or clearing temporary application state.

## js_fetch_credentials

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Fetch Credentials Demo</title>
</head>
<body>
    <!-- PROBLEM: Send cookies and authentication with fetch requests -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <button onclick="fetchWithCredentials()">Fetch with Credentials</button>
    <button onclick="fetchWithoutCredentials()">Fetch without Credentials</button>
    <div id="result"></div>

    <script>
        async function fetchWithCredentials() {
            const resultDiv = document.getElementById('result');
            
            try {
                // 'include' sends cookies even for cross-origin requests
                // 'same-origin' (default) only sends cookies for same-origin
                const response = await fetch('https://api.example.com/data', {
                    method: 'GET',
                    credentials: 'include', // Options: 'omit', 'same-origin', 'include'
                    headers: {
                        'Content-Type': 'application/json'
                    }
                });
                
                const data = await response.json();
                resultDiv.innerHTML = `<p>Success with credentials: ${JSON.stringify(data)}</p>`;
            } catch (error) {
                resultDiv.innerHTML = `<p>Error: ${error.message}</p>`;
            }
        }
        
        async function fetchWithoutCredentials() {
            const resultDiv = document.getElementById('result');
            
            try {
                // 'omit' never sends cookies
                const response = await fetch('https://api.example.com/data', {
                    method: 'GET',
                    credentials: 'omit'
                });
                
                const data = await response.json();
                resultDiv.innerHTML = `<p>Success without credentials: ${JSON.stringify(data)}</p>`;
            } catch (error) {
                resultDiv.innerHTML = `<p>Error: ${error.message}</p>`;
            }
        }
        
        // Why: Controls cookie/authorization header behavior in fetch,
        // essential for authenticated API requests and CORS handling
    </script>
</body>
</html>
```

**Why**: The credentials option in fetch is critical for handling authentication cookies, especially in cross-origin scenarios where security and proper credential handling are essential.

## js_input_setCustomValidity

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>setCustomValidity Demo</title>
    <style>
        input:invalid { border-color: red; }
        input:valid { border-color: green; }
    </style>
</head>
<body>
    <!-- PROBLEM: Set custom validation messages for form inputs -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <form onsubmit="return validateForm(event)">
        <div>
            <label>Password:</label>
            <input type="password" id="password" required>
        </div>
        
        <div>
            <label>Confirm Password:</label>
            <input type="password" id="confirmPassword" required>
        </div>
        
        <div>
            <label>Username:</label>
            <input type="text" id="username" required>
        </div>
        
        <button type="submit">Submit</button>
    </form>

    <script>
        const confirmPassword = document.getElementById('confirmPassword');
        const username = document.getElementById('username');
        
        // Password match validation
        confirmPassword.addEventListener('input', function() {
            const password = document.getElementById('password').value;
            
            if (this.value !== password) {
                // Set custom error message
                this.setCustomValidity('Passwords do not match!');
            } else {
                // Clear custom error - must call with empty string
                this.setCustomValidity('');
            }
        });
        
        // Username validation (no spaces)
        username.addEventListener('input', function() {
            if (this.value.includes(' ')) {
                this.setCustomValidity('Username cannot contain spaces');
            } else if (this.value.length < 3) {
                this.setCustomValidity('Username must be at least 3 characters');
            } else {
                this.setCustomValidity('');
            }
        });
        
        function validateForm(event) {
            event.preventDefault();
            
            // Check if form is valid
            const form = event.target;
            if (form.checkValidity()) {
                alert('Form is valid and ready to submit!');
            } else {
                // Show validation messages
                form.reportValidity();
            }
            
            return false;
        }
        
        // Why: Enables custom validation logic beyond built-in constraints,
        // providing user-friendly error messages for complex validation rules
    </script>
</body>
</html>
```

**Why**: setCustomValidity() enables sophisticated custom validation with user-friendly error messages, going beyond basic HTML5 validation attributes.

## html5_geolocation_clearWatch

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Geolocation clearWatch Demo</title>
</head>
<body>
    <!-- PROBLEM: Stop continuous location tracking -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <h2>Location Tracker</h2>
    
    <button id="startBtn" onclick="startTracking()">Start Tracking</button>
    <button id="stopBtn" onclick="stopTracking()" disabled>Stop Tracking</button>
    
    <div id="location"></div>

    <script>
        let watchId = null;
        
        function startTracking() {
            const locationDiv = document.getElementById('location');
            
            if ('geolocation' in navigator) {
                // watchPosition() returns an ID that can be used to stop tracking
                watchId = navigator.geolocation.watchPosition(
                    // Success callback
                    function(position) {
                        const { latitude, longitude, accuracy } = position.coords;
                        const timestamp = new Date(position.timestamp);
                        
                        locationDiv.innerHTML = `
                            <p>Latitude: ${latitude.toFixed(6)}</p>
                            <p>Longitude: ${longitude.toFixed(6)}</p>
                            <p>Accuracy: ${accuracy.toFixed(2)} meters</p>
                            <p>Last Update: ${timestamp.toLocaleTimeString()}</p>
                        `;
                    },
                    // Error callback
                    function(error) {
                        locationDiv.innerHTML = `<p>Error: ${error.message}</p>`;
                    },
                    // Options
                    {
                        enableHighAccuracy: true,
                        timeout: 5000,
                        maximumAge: 0
                    }
                );
                
                document.getElementById('startBtn').disabled = true;
                document.getElementById('stopBtn').disabled = false;
            } else {
                locationDiv.innerHTML = '<p>Geolocation not supported</p>';
            }
        }
        
        function stopTracking() {
            if (watchId !== null) {
                // clearWatch() stops the position watching
                navigator.geolocation.clearWatch(watchId);
                watchId = null;
                
                document.getElementById('startBtn').disabled = false;
                document.getElementById('stopBtn').disabled = true;
                
                document.getElementById('location').innerHTML += 
                    '<p><strong>Tracking stopped</strong></p>';
            }
        }
        
        // Why: Essential for battery conservation and privacy - 
        // stops continuous GPS usage when tracking is no longer needed
    </script>
</body>
</html>
```

**Why**: clearWatch() is essential for resource management and privacy, stopping continuous GPS tracking when location updates are no longer needed.

```html
                    <p>Name: ${obj.name}</p>
                    <p>Age: ${obj.age}</p>
                    <p>Skills: ${obj.skills.join(', ')}</p>
                    <p>City: ${obj.address.city}</p>
                `;
            } catch (error) {
                document.getElementById('result').innerHTML = 
                    `<p style="color: red;">Error: ${error.message}</p>`;
            }
        }
        
        function parseWithReviver() {
            const jsonString = document.getElementById('jsonInput').value;
            
            try {
                // JSON.parse with reviver function
                // Reviver can transform or filter values
                const obj = JSON.parse(jsonString, function(key, value) {
                    // Convert age to a specific format
                    if (key === 'age') {
                        return value + ' years old';
                    }
                    // Convert all string values to uppercase
                    if (typeof value === 'string' && key !== 'age') {
                        return value.toUpperCase();
                    }
                    return value;
                });
                
                document.getElementById('result').innerHTML = `
                    <h4>With Reviver (transformed):</h4>
                    <pre>${JSON.stringify(obj, null, 2)}</pre>
                `;
            } catch (error) {
                document.getElementById('result').innerHTML = 
                    `<p style="color: red;">Error: ${error.message}</p>`;
            }
        }
        
        // Initial parse
        parseJSON();
        
        // Why: Fundamental for parsing API responses, reading JSON files,
        // and data deserialization with optional transformation via reviver
    </script>
</body>
</html>
```

**Why**: JSON.parse() is fundamental for working with APIs, configuration files, and data interchange, with the reviver function enabling powerful data transformation during parsing.

## js_map_get

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Map.get() Demo</title>
</head>
<body>
    <!-- PROBLEM: Retrieve value from Map by key -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <h2>User Database (Map)</h2>
    
    <input type="text" id="userId" placeholder="Enter user ID (e.g., user1)">
    <button onclick="getUser()">Get User</button>
    <button onclick="showAll()">Show All Users</button>
    
    <div id="result"></div>

    <script>
        // Create a Map to store user data
        const userDatabase = new Map([
            ['user1', { name: 'Alice', age: 28, role: 'Admin' }],
            ['user2', { name: 'Bob', age: 34, role: 'User' }],
            ['user3', { name: 'Charlie', age: 22, role: 'User' }],
            ['user4', { name: 'Diana', age: 30, role: 'Moderator' }]
        ]);
        
        // Map can use any value as key (even objects)
        const objectKey = { id: 'special' };
        userDatabase.set(objectKey, { name: 'Special User', note: 'Object key!' });
        
        function getUser() {
            const userId = document.getElementById('userId').value;
            const resultDiv = document.getElementById('result');
            
            // map.get() returns the value or undefined if key doesn't exist
            const user = userDatabase.get(userId);
            
            if (user) {
                resultDiv.innerHTML = `
                    <h3>User Found:</h3>
                    <p><strong>ID:</strong> ${userId}</p>
                    <p><strong>Name:</strong> ${user.name}</p>
                    <p><strong>Age:</strong> ${user.age}</p>
                    <p><strong>Role:</strong> ${user.role}</p>
                `;
            } else {
                resultDiv.innerHTML = `
                    <p style="color: red;">User "${userId}" not found</p>
                    <p>Available IDs: user1, user2, user3, user4</p>
                `;
            }
        }
        
        function showAll() {
            const resultDiv = document.getElementById('result');
            let html = '<h3>All Users:</h3>';
            
            // Map.forEach() or for...of to iterate
            userDatabase.forEach((user, key) => {
                if (typeof key === 'string') {
                    html += `<p><strong>${key}:</strong> ${user.name} (${user.role})</p>`;
                } else {
                    html += `<p><strong>Object key:</strong> ${user.name}</p>`;
                }
            });
            
            resultDiv.innerHTML = html;
        }
        
        // Demonstrate has() method
        function demonstrateMapMethods() {
            console.log('Has user1:', userDatabase.has('user1')); // true
            console.log('Has user99:', userDatabase.has('user99')); // false
            console.log('Get non-existent:', userDatabase.get('user99')); // undefined
        }
        
        demonstrateMapMethods();
        
        // Why: Map.get() provides O(1) key-value lookup with any key type,
        // unlike objects which only support string/Symbol keys
    </script>
</body>
</html>
```

**Why**: Map.get() provides efficient O(1) lookup with support for any key type, making it superior to plain objects for complex key-value scenarios.
