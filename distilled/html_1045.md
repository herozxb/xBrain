## 1. js_element_removeAttribute
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>removeAttribute Demo</title>
</head>
<body>
    <!-- PROBLEM: How to dynamically remove attributes from DOM elements
         TIME: O(1) SPACE: O(1) -->
    
    <button id="myBtn" disabled class="btn-primary" data-info="temp">
        Click Me
    </button>
    
    <script>
        // removeAttribute() - Removes a specified attribute from an element
        // Returns undefined, modifies element in-place
        
        const button = document.getElementById('myBtn');
        
        // Remove multiple attributes to enable the button
        function enableButton() {
            // Remove 'disabled' attribute - makes button clickable
            button.removeAttribute('disabled');
            
            // Remove custom data attribute
            button.removeAttribute('data-info');
            
            // Remove class attribute (be careful - removes ALL classes)
            button.removeAttribute('class');
        }
        
        // Enable button after 2 seconds
        setTimeout(enableButton, 2000);
        
        // Note: removeAttribute returns undefined even if attribute doesn't exist
        // No error thrown - safe to call on non-existent attributes
        
        // Alternative: Check if attribute exists first
        function safeRemove(element, attr) {
            if (element.hasAttribute(attr)) {
                element.removeAttribute(attr);
                return true; // Successfully removed
            }
            return false; // Attribute didn't exist
        }
    </script>
</body>
</html>
```
**Why:** Essential for dynamic UI changes - enabling/disabling elements, clearing custom data, toggling states.

## 2. js_event_currentTarget
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>currentTarget Demo</title>
    <style>
        .card {
            padding: 20px;
            background: #f0f0f0;
            cursor: pointer;
        }
        .card-content {
            background: white;
            padding: 10px;
        }
    </style>
</head>
<body>
    <!-- PROBLEM: How to reference the element that the listener is attached to
         TIME: O(1) SPACE: O(1) -->
    
    <div class="card" id="card1">
        <div class="card-content">
            <span>Click anywhere in card</span>
        </div>
    </div>
    
    <script>
        // event.currentTarget - Always references the element 
        // that the event listener is attached to
        // Unlike event.target which references the clicked element
        
        const card = document.getElementById('card1');
        
        card.addEventListener('click', function(event) {
            // currentTarget is ALWAYS the card (listener attached here)
            console.log('currentTarget:', event.currentTarget.id); // "card1"
            
            // target is the actual clicked element (could be span, div.card-content, or card)
            console.log('target:', event.target.tagName);
            
            // Use currentTarget when you need the bound element
            event.currentTarget.style.background = '#e0e0e0';
            
            // Important: currentTarget is only available during event handler
            // If you need to reference it later, store it in a variable
            const boundElement = event.currentTarget;
        });
        
        // Real-world example: Event delegation with multiple buttons
        document.body.addEventListener('click', function(event) {
            if (event.target.matches('button')) {
                // currentTarget is body, target is the clicked button
                console.log('Clicked button:', event.target.textContent);
            }
        });
    </script>
</body>
</html>
```
**Why:** Critical for event delegation patterns and accessing the bound element regardless of which child was actually clicked.

## 3. js_sessionStorage_removeItem
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>sessionStorage.removeItem Demo</title>
</head>
<body>
    <!-- PROBLEM: How to remove specific items from session storage
         TIME: O(1) average SPACE: O(1) -->
    
    <div>
        <input type="text" id="username" placeholder="Username">
        <button onclick="saveSession()">Save Session</button>
        <button onclick="clearSession()">Clear Session</button>
        <button onclick="checkSession()">Check Session</button>
    </div>
    
    <script>
        // sessionStorage.removeItem(key) - Removes specific item from session storage
        // Data persists until tab/window is closed
        
        function saveSession() {
            const username = document.getElementById('username').value;
            sessionStorage.setItem('username', username);
            sessionStorage.setItem('timestamp', Date.now().toString());
            console.log('Session saved');
        }
        
        function clearSession() {
            // Remove specific item
            sessionStorage.removeItem('username');
            console.log('Username removed from session');
            
            // Note: removeItem returns undefined, doesn't throw if key doesn't exist
            
            // To clear ALL session data, use clear()
            // sessionStorage.clear();
        }
        
        function checkSession() {
            const username = sessionStorage.getItem('username');
            const timestamp = sessionStorage.getItem('timestamp');
            
            if (username) {
                const elapsed = Date.now() - parseInt(timestamp);
                console.log(`Welcome back, ${username}! Session age: ${elapsed}ms`);
            } else {
                console.log('No active session found');
            }
        }
        
        // Monitor storage changes (only fires in OTHER tabs/windows)
        window.addEventListener('storage', function(event) {
            if (event.key === 'username' && event.newValue === null) {
                console.log('Username was removed in another tab');
            }
        });
        
        // Best practice: Clean up sensitive data
        window.addEventListener('beforeunload', function() {
            sessionStorage.removeItem('sensitiveToken');
            sessionStorage.removeItem('tempPassword');
        });
    </script>
</body>
</html>
```
**Why:** Essential for managing session state, clearing sensitive data, and controlling what persists across page refreshes.

## 4. js_fetch_mode
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>fetch mode Demo</title>
</head>
<body>
    <!-- PROBLEM: How to control CORS behavior in fetch requests
         TIME: O(n) for network SPACE: O(1) -->
    
    <button onclick="testModes()">Test Fetch Modes</button>
    <div id="output"></div>
    
    <script>
        // fetch mode options:
        // 'cors' - default, allows cross-origin requests
        // 'no-cors' - limited to opaque responses (can't read content)
        // 'same-origin' - only allows same-origin requests
        // 'navigate' - for navigation requests (rarely used manually)
        
        async function testModes() {
            const output = document.getElementById('output');
            const apiUrl = 'https://jsonplaceholder.typicode.com/posts/1';
            
            // 1. CORS mode (default) - Full access to response
            try {
                const response = await fetch(apiUrl, {
                    mode: 'cors'  // Explicitly set (this is default)
                });
                const data = await response.json();
                console.log('CORS mode:', data.title);
            } catch (error) {
                console.error('CORS failed:', error);
            }
            
            // 2. no-cors mode - Opaque response (can't read body)
            try {
                const response = await fetch(apiUrl, {
                    mode: 'no-cors'
                });
                // response.type === 'opaque'
                // response.status === 0
                // Cannot call .json() or .text()
                console.log('no-cors mode: Response received but opaque');
                console.log('Type:', response.type); // "opaque"
                console.log('Status:', response.status); // 0
            } catch (error) {
                console.error('no-cors failed:', error);
            }
            
            // 3. same-origin mode - Fails on cross-origin
            try {
                const response = await fetch(apiUrl, {
                    mode: 'same-origin'
                });
            } catch (error) {
                console.log('same-origin failed as expected:', error.message);
            }
            
            // Real-world example: Loading images with no-cors
            async function loadImage(url) {
                const response = await fetch(url, { mode: 'no-cors' });
                const blob = await response.blob();
                return URL.createObjectURL(blob);
            }
        }
    </script>
</body>
</html>
```
**Why:** Critical for understanding CORS restrictions, controlling cross-origin access, and handling different API scenarios.

## 5. html5_drag_events
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>HTML5 Drag Events Demo</title>
    <style>
        .draggable {
            width: 100px;
            height: 100px;
            background: #3498db;
            margin: 10px;
            cursor: move;
        }
        .dropzone {
            width: 200px;
            height: 200px;
            border: 2px dashed #ccc;
            padding: 20px;
        }
        .drag-over {
            background-color: #e0f7fa !important;
            border-color: #00bcd7 !important;
        }
    </style>
</head>
<body>
    <!-- PROBLEM: How to implement native drag-and-drop functionality
         TIME: O(1) for handlers SPACE: O(1) -->
    
    <div class="draggable" draggable="true" id="drag1">Drag Me</div>
    <div class="draggable" draggable="true" id="drag2">Drag Me Too</div>
    
    <div class="dropzone" id="dropzone">Drop here</div>
    
    <script>
        const draggables = document.querySelectorAll('.draggable');
        const dropzone = document.getElementById('dropzone');
        
        // DRAGGABLE ELEMENT EVENTS
        
        // 1. dragstart - Fires when dragging starts
        draggables.forEach(el => {
            el.addEventListener('dragstart', function(event) {
                // Store the element's ID for later retrieval
                event.dataTransfer.setData('text/plain', event.target.id);
                
                // Optional: Set drag image
                event.dataTransfer.effectAllowed = 'move';
                
                // Visual feedback
                event.target.style.opacity = '0.5';
            });
            
            // 2. dragend - Fires when dragging ends (whether dropped or not)
            el.addEventListener('dragend', function(event) {
                event.target.style.opacity = '1';
            });
            
            // 3. drag - Fires continuously during drag (high frequency)
            el.addEventListener('drag', function(event) {
                // Use sparingly - fires many times per second
                console.log('Dragging...');
            });
        });
        
        // DROPZONE EVENTS
        
        // 4. dragenter - Fires when dragged element enters drop zone
        dropzone.addEventListener('dragenter', function(event) {
            event.preventDefault(); // Required to allow dropping
            event.target.classList.add('drag-over');
        });
        
        // 5. dragover - Fires continuously while element is over drop zone
        dropzone.addEventListener('dragover', function(event) {
            event.preventDefault(); // REQUIRED - without this, drop won't fire
            event.dataTransfer.dropEffect = 'move';
        });
        
        // 6. dragleave - Fires when dragged element leaves drop zone
        dropzone.addEventListener('dragleave', function(event) {
            event.target.classList.remove('drag-over');
        });
        
        // 7. drop - Fires when element is dropped
        dropzone.addEventListener('drop', function(event) {
            event.preventDefault();
            event.target.classList.remove('drag-over');
            
            // Get the dragged element's ID
            const id = event.dataTransfer.getData('text/plain');
            const draggable = document.getElementById(id);
            
            // Move the element into the dropzone
            event.target.appendChild(draggable);
        });
        
        // Event sequence: dragstart → drag → dragenter → dragover → drop → dragend
    </script>
</body>
</html>
```
**Why:** Native drag-and-drop provides better UX than mouse events alone, with built-in visual feedback and cross-browser support.

## 6. css_variable_fallback
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>CSS Variable Fallback Demo</title>
    <style>
        /* PROBLEM: How to provide default values for CSS custom properties
           TIME: O(1) SPACE: O(1) */
        
        :root {
            --primary-color: #3498db;
            --secondary-color: #e74c3c;
        }
        
        .box1 {
            /* Use defined variable */
            background: var(--primary-color);
            padding: 20px;
            margin: 10px;
        }
        
        .box2 {
            /* Fallback: Use --undefined-color if exists, otherwise use orange */
            background: var(--undefined-color, orange);
            padding: 20px;
            margin: 10px;
        }
        
        .box3 {
            /* Multiple fallbacks (nested) */
            background: var(--theme-color, var(--primary-color, #999));
            padding: 20px;
            margin: 10px;
        }
        
        .box4 {
            /* Fallback with multiple values */
            margin: var(--custom-margin, 10px 20px);
            padding: var(--custom-padding, 15px);
            background: var(--nonexistent, var(--also-nonexistent, #f39c12));
        }
        
        .dynamic {
            /* Fallback can be another CSS property or function */
            width: var(--custom-width, calc(100% - 40px));
            height: var(--custom-height, 100px);
            background: var(--dynamic-bg, linear-gradient(45deg, red, blue));
            margin: 10px;
        }
        
        /* Fallbacks don't prevent invalid values */
        .box5 {
            /* If --my-color is "invalid", the fallback won't save it */
            color: var(--my-color, blue);
            /* If --my-color is set to an invalid value, color stays default */
        }
    </style>
</head>
<body>
    <div class="box1">Using defined --primary-color</div>
    <div class="box2">Using fallback (orange) for undefined variable</div>
    <div class="box3">Nested fallback (uses --primary-color as fallback)</div>
    <div class="box4">Multiple property fallbacks</div>
    <div class="dynamic">Dynamic fallback with calc()</div>
    
    <script>
        // Manipulate CSS variables via JavaScript
        const dynamicBox = document.querySelector('.dynamic');
        
        // Set a custom property
        dynamicBox.style.setProperty('--custom-height', '150px');
        
        // Fallback value demonstration
        console.log('Fallbacks ensure graceful degradation');
        
        // Note: Fallback values are NOT used if the variable exists
        // even if the variable's value is invalid
        document.documentElement.style.setProperty('--my-color', 'invalid-value');
        // This will NOT fall back to blue - it will be invalid
    </script>
</body>
</html>
```
**Why:** Fallback values provide graceful degradation and sensible defaults when CSS variables are undefined or being loaded.

## 7. js_regexp_exec
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>RegExp.exec() Demo</title>
</head>
<body>
    <!-- PROBLEM: How to execute regex and get detailed match information
         TIME: O(n) where n is string length SPACE: O(m) where m is match size -->
    
    <script>
        // regexp.exec(string) - Executes search for match in string
        // Returns array with match details or null if no match
        
        // 1. Basic exec with capturing groups
        const regex1 = /(\d{4})-(\d{2})-(\d{2})/;
        const dateStr = 'Today is 2026-02-27 and tomorrow is 2026-02-28';
        
        const match1 = regex1.exec(dateStr);
        if (match1) {
            console.log('Full match:', match1[0]);     // "2026-02-27"
            console.log('Year:', match1[1]);          // "2026"
            console.log('Month:', match1[2]);         // "02"
            console.log('Day:', match1[3]);           // "27"
            console.log('Index:', match1.index);      // 9 (position in string)
            console.log('Input:', match1.input);      // original string
            console.log('Groups:', match1.groups);    // undefined (no named groups)
        }
        
        // 2. Named capture groups (ES2018+)
        const regex2 = /(?<year>\d{4})-(?<month>\d{2})-(?<day>\d{2})/;
        const match2 = regex2.exec(dateStr);
        if (match2) {
            console.log('Named groups:', match2.groups);
            // { year: "2026", month: "02", day: "27" }
        }
        
        // 3. Global flag - exec maintains lastIndex state
        const regex3 = /\d{4}-\d{2}-\d{2}/g;
        let match;
        
        // Find all matches using exec in a loop
        while ((match = regex3.exec(dateStr)) !== null) {
            console.log(`Found ${match[0]} at index ${match.index}`);
            // First: "2026-02-27" at 9
            // Second: "2026-02-28" at 37
        }
        
        // 4. Without global flag - exec always returns first match
        const regex4 = /hello/;
        const str4 = 'hello world hello';
        console.log(regex4.exec(str4)); // Always matches first "hello"
        
        // 5. Real-world example: Extract all email addresses
        const emailRegex = /([a-zA-Z0-9._-]+)@([a-zA-Z0-9._-]+\.[a-zA-Z0-9._-]+)/g;
        const text = 'Contact: alice@example.com, bob@test.org, charlie@domain.net';
        const emails = [];
        
        while ((match = emailRegex.exec(text)) !== null) {
            emails.push({
                full: match[0],
                username: match[1],
                domain: match[2]
            });
        }
        console.log('Emails:', emails);
        
        // exec() vs match():
        // - exec is a regex method, match is a string method
        // - exec provides more control with global flag
        // - exec can be called repeatedly to find all matches
    </script>
</body>
</html>
```
**Why:** exec() provides detailed match information including groups and positions, essential for complex text parsing and extraction.

## 8. js_date_getFullYear
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Date.getFullYear() Demo</title>
</head>
<body>
    <!-- PROBLEM: How to extract the year from a Date object
         TIME: O(1) SPACE: O(1) -->
    
    <script>
        // date.getFullYear() - Returns the year (4 digits for years 1000-9999)
        // Returns local time year, not UTC
        
        // 1. Current year
        const now = new Date();
        const currentYear = now.getFullYear();
        console.log('Current year:', currentYear); // 2026
        
        // 2. Specific date
        const independence = new Date('1776-07-04');
        console.log('Independence year:', independence.getFullYear()); // 1776
        
        // 3. Handle different eras
        const ancient = new Date('0500-01-01');
        console.log('Ancient year:', ancient.getFullYear()); // 500
        
        const future = new Date('3000-01-01');
        console.log('Future year:', future.getFullYear()); // 3000
        
        // 4. Important: getFullYear() vs getYear()
        // getYear() is DEPRECATED - returns year - 1900
        // getFullYear() should ALWAYS be used
        
        // 5. Real-world example: Age calculator
        function calculateAge(birthDate) {
            const today = new Date();
            const birth = new Date(birthDate);
            
            let age = today.getFullYear() - birth.getFullYear();
            const monthDiff = today.getMonth() - birth.getMonth();
            
            // Adjust if birthday hasn't occurred this year
            if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birth.getDate())) {
                age--;
            }
            
            return age;
        }
        
        console.log('Age:', calculateAge('1990-06-15'));
        
        // 6. Real-world: Year validation
        function isValidYear(year) {
            const date = new Date();
            const inputDate = new Date(year, 0, 1); // Jan 1st of that year
            
            // Check if the year was parsed correctly
            return inputDate.getFullYear() === year;
        }
        
        // 7. Extract year from various date formats
        function extractYear(dateInput) {
            const date = new Date(dateInput);
            if (isNaN(date.getTime())) {
                return 'Invalid date';
            }
            return date.getFullYear();
        }
        
        console.log(extractYear('2026-02-27'));    // 2026
        console.log(extractYear('02/27/2026'));    // 2026
        console.log(extractYear('Feb 27 2026'));   // 2026
        console.log(extractYear(1740633600000));   // 2026 (timestamp)
        
        // 8. Compare years
        function isSameYear(date1, date2) {
            return new Date(date1).getFullYear() === new Date(date2).getFullYear();
        }
        
        // For UTC year, use getUTCFullYear()
        console.log('Local year:', now.getFullYear());
        console.log('UTC year:', now.getUTCFullYear());
    </script>
</body>
</html>
```
**Why:** Year extraction is fundamental for date calculations, age validation, and time-based logic in applications.

## 9. css_focus_visible

## 10. web_components_custom_elements
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Custom Elements Demo</title>
</head>
<body>
    <!-- PROBLEM: How to create reusable HTML elements with encapsulated behavior
         TIME: O(1) for lifecycle hooks SPACE: O(1) -->
    
    <!-- Use the custom element -->
    <user-card name="Alice" email="alice@example.com"></user-card>
    <user-card name="Bob" email="bob@example.com"></user-card>
    
    <!-- Counter custom element -->
    <counter-element></counter-element>
    
    <script>
        // 1. Define a custom element
        class UserCard extends HTMLElement {
            // Called when element is added to the DOM
            connectedCallback() {
                const name = this.getAttribute('name') || 'Anonymous';
                const email = this.getAttribute('email') || 'No email';
                
                this.innerHTML = `
                    <style>
                        .card {
                            border: 1px solid #ddd;
                            padding: 15px;
                            margin: 10px;
                            border-radius: 8px;
                            background: #f9f9f9;
                        }
                        .name {
                            font-weight: bold;
                            color: #333;
                        }
                        .email {
                            color: #666;
                            font-size: 0.9em;
                        }
                    </style>
                    <div class="card">
                        <div class="name">${name}</div>
                        <div class="email">${email}</div>
                    </div>
                `;
            }
            
            // Observe attribute changes
            static get observedAttributes() {
                return ['name', 'email'];
            }
            
            // Called when observed attribute changes
            attributeChangedCallback(name, oldValue, newValue) {
                if (oldValue !== newValue) {
                    console.log(`${name} changed from ${oldValue} to ${newValue}`);
                }
            }
            
            // Called when element is removed from DOM
            disconnectedCallback() {
                console.log('User card removed from DOM');
            }
        }
        
        // Register the custom element (must contain a hyphen)
        customElements.define('user-card', UserCard);
        
        // 2. More complex example with shadow DOM
        class CounterElement extends HTMLElement {
            constructor() {
                super();
                this.count = 0;
            }
            
            connectedCallback() {
                // Create shadow DOM for style encapsulation
                const shadow = this.attachShadow({ mode: 'open' });
                
                shadow.innerHTML = `
                    <style>
                        .counter {
                            display: inline-flex;
                            align-items: center;
                            gap: 10px;
                        }
                        button {
                            padding: 5px 15px;
                            font-size: 16px;
                            cursor: pointer;
                        }
                        span {
                            font-size: 20px;
                            font-weight: bold;
                        }
                    </style>
                    <div class="counter">
                        <button id="dec">-</button>
                        <span id="count">${this.count}</span>
                        <button id="inc">+</button>
                    </div>
                `;
                
                // Add event listeners
                shadow.getElementById('inc').addEventListener('click', () => {
                    this.count++;
                    shadow.getElementById('count').textContent = this.count;
                });
                
                shadow.getElementById('dec').addEventListener('click', () => {
                    this.count--;
                    shadow.getElementById('count').textContent = this.count;
                });
            }
        }
        
        customElements.define('counter-element', CounterElement);
        
        // 3. Lifecycle callbacks summary:
        // - constructor(): Called when element is created
        // - connectedCallback(): Called when added to DOM
        // - disconnectedCallback(): Called when removed from DOM
        // - adoptedCallback(): Called when moved to new document
        // - attributeChangedCallback(): Called when attribute changes
        
        // 4. Best practices:
        // - Always include a hyphen in element name
        // - Use shadow DOM for style encapsulation
        // - Keep constructor minimal
        // - Do DOM manipulation in connectedCallback
    </script>
</body>
```
