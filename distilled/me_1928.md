## Binary Indexed Tree (Fenwick Tree)

```python
"""
Binary Indexed Tree (Fenwick Tree) for efficient prefix sum queries.
Supports point updates and range sum queries in O(log n) time.
"""
from typing import List, Optional

class BinaryIndexedTree:
    # PROBLEM: Efficiently compute prefix sums with point updates
    # TIME: O(log n) for both update and query, SPACE: O(n)
    
    def __init__(self, size: int) -> None:
        """Initialize BIT with given size (1-indexed internally)."""
        self.n: int = size
        self.tree: List[int] = [0] * (size + 1)  # 1-indexed array
    
    def update(self, index: int, delta: int) -> None:
        """
        Add delta to element at index (0-indexed input).
        Time: O(log n)
        """
        i: int = index + 1  # Convert to 1-indexed
        while i <= self.n:
            self.tree[i] += delta
            i += i & (-i)  # Move to parent
    
    def query(self, index: int) -> int:
        """
        Get sum of elements from index 0 to index (inclusive).
        Time: O(log n)
        """
        i: int = index + 1  # Convert to 1-indexed
        result: int = 0
        while i > 0:
            result += self.tree[i]
            i -= i & (-i)  # Move to previous range
        return result
    
    def range_query(self, left: int, right: int) -> int:
        """Get sum of elements from left to right (inclusive)."""
        if left == 0:
            return self.query(right)
        return self.query(right) - self.query(left - 1)

# Example usage
if __name__ == "__main__":
    bit = BinaryIndexedTree(10)
    bit.update(2, 5)  # arr[2] += 5
    bit.update(5, 3)  # arr[5] += 3
    print(bit.range_query(2, 5))  # Sum from index 2 to 5
```

**Why**: BIT is essential for scenarios requiring frequent range queries and point updates. Used in competitive programming, computational geometry, and real-time analytics where O(log n) performance is critical.

## Graph Coloring (Backtracking)

```python
"""
Graph coloring using backtracking algorithm.
Assigns colors to vertices such that no adjacent vertices share the same color.
"""
from typing import List, Dict, Set, Optional

class GraphColoring:
    # PROBLEM: Color graph vertices with minimum colors (NP-hard)
    # TIME: O(m^n) worst case (m colors, n vertices), SPACE: O(n)
    
    def __init__(self, vertices: int) -> None:
        """Initialize graph with given number of vertices."""
        self.v: int = vertices
        self.graph: Dict[int, Set[int]] = {i: set() for i in range(vertices)}
    
    def add_edge(self, u: int, v: int) -> None:
        """Add undirected edge between vertices u and v."""
        self.graph[u].add(v)
        self.graph[v].add(u)
    
    def is_safe(self, vertex: int, color: int, colors: List[int]) -> bool:
        """Check if color assignment is valid for vertex."""
        for neighbor in self.graph[vertex]:
            if colors[neighbor] == color:
                return False
        return True
    
    def color_graph(self, num_colors: int) -> Optional[List[int]]:
        """
        Attempt to color graph with num_colors.
        Returns color assignment or None if impossible.
        """
        colors: List[int] = [0] * self.v
        
        def backtrack(vertex: int) -> bool:
            """Recursively assign colors using backtracking."""
            if vertex == self.v:
                return True  # All vertices colored successfully
            
            for color in range(1, num_colors + 1):
                if self.is_safe(vertex, color, colors):
                    colors[vertex] = color
                    
                    if backtrack(vertex + 1):
                        return True
                    
                    colors[vertex] = 0  # Backtrack
            
            return False  # No valid color found
        
        if backtrack(0):
            return colors
        return None

# Example usage
if __name__ == "__main__":
    gc = GraphColoring(5)
    gc.add_edge(0, 1)
    gc.add_edge(0, 2)
    gc.add_edge(1, 2)
    gc.add_edge(1, 3)
    gc.add_edge(2, 3)
    gc.add_edge(3, 4)
    
    result = gc.color_graph(3)  # Try with 3 colors
    if result:
        print(f"Coloring: {result}")
    else:
        print("No solution with 3 colors")
```

```python
Useful for lazy loading, access control, logging, and caching.

from typing import Optional, Dict, Any

class Image(ABC):
    """Abstract interface for image operations."""
    
    @abstractmethod

        """Display the image."""
        pass
    
    @abstractmethod
    def get_metadata(self) -> Dict[str, Any]:
        """Get image metadata."""
        pass

# Real Subject
class RealImage(Image):
    """The actual image object that loads from disk."""
    
    def __init__(self, filename: str) -> None:
        self.filename: str = filename
        self._metadata: Optional[Dict[str, Any]] = None
        self._load_from_disk()
    
    def _load_from_disk(self) -> None:
        """Simulate expensive disk loading operation."""
        print(f"Loading {self.filename} from disk...")
        time.sleep(1)  # Simulate load time
        self._metadata = {
            "filename": self.filename,
            "size": "2.5MB",
            "resolution": "1920x1080"
        }
    
    def display(self) -> None:
        print(f"Displaying {self.filename}")
    
    def get_metadata(self) -> Dict[str, Any]:
        return self._metadata if self._metadata else {}

# Virtual Proxy (Lazy Loading)
class ImageProxy(Image):
    """Proxy that defers loading until actually needed."""
    
    def __init__(self, filename: str) -> None:
        self.filename: str = filename
        self._real_image: Optional[RealImage] = None
        self._cached_metadata: Optional[Dict[str, Any]] = None
    
    def display(self) -> None:
        """Lazy load and display the image."""
        if self._real_image is None:
            self._real_image = RealImage(self.filename)
        self._real_image.display()
    
    def get_metadata(self) -> Dict[str, Any]:
        """Return cached metadata without loading image."""
        if self._cached_metadata is None:
            if self._real_image is not None:
                self._cached_metadata = self._real_image.get_metadata()
            else:
                # Return lightweight metadata without loading
                self._cached_metadata = {"filename": self.filename, "loaded": False}
        return self._cached_metadata

    """Proxy that controls access based on user permissions."""

        self.user_role: str = user_role

        """Check if user has permission for the action."""
        permissions = {
            "admin": ["display", "metadata"],
            "user": ["display"],
            "guest": []
        }

    def display(self) -> None:
        if not self._check_permission("display"):
            print(f"Access denied: {self.user_role} cannot display images")
            return

            self._real_image = RealImage(self.filename)
        self._real_image.display()
    
    def get_metadata(self) -> Dict[str, Any]:
        if not self._check_permission("metadata"):
            print(f"Access denied: {self.user_role} cannot view metadata")
            return {}
        
        if self._real_image is None:
            self._real_image = RealImage(self.filename)
        return self._real_image.get_metadata()
```

## CQRS Pattern (Command Query Responsibility Segregation)

```python
"""
CQRS: Separate read and write operations into different models.
Optimizes performance and scalability for complex domains.
"""
from typing import Dict, List, Any, Optional
from dataclasses import dataclass, field
from datetime import datetime
from abc import ABC, abstractmethod
import uuid

# Domain Model
@dataclass
class User:
    """User entity."""
    id: str
    email: str
    name: str
    created_at: datetime = field(default_factory=datetime.now)
    version: int = 0

# Commands (Write Model)
@dataclass
class Command(ABC):
    """Base command interface."""
    command_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    timestamp: datetime = field(default_factory=datetime.now)

@dataclass
class CreateUserCommand(Command):
    """Command to create a new user."""
    email: str = ""
    name: str = ""

@dataclass
class UpdateUserNameCommand(Command):
    """Command to update user name."""
    user_id: str = ""
    new_name: str = ""

@dataclass
class DeleteUserCommand(Command):
    """Command to delete a user."""
    user_id: str = ""

# Queries (Read Model)
@dataclass
class Query(ABC):
    """Base query interface."""
    query_id: str = field(default_factory=lambda: str(uuid.uuid4()))

@dataclass
class GetUserQuery(Query):
    """Query to get a single user."""
    user_id: str = ""

@dataclass
class ListUsersQuery(Query):
    """Query to list users with optional filtering."""
    email_filter: Optional[str] = None
    limit: int = 100

# Write Model (Command Handler)
class UserWriteRepository:
    """Handles all write operations - optimized for consistency."""
    
    def __init__(self) -> None:
        self._users: Dict[str, User] = {}
        self._event_store: List[Dict[str, Any]] = []
    
    def handle(self, command: Command) -> Optional[str]:
        """Dispatch command to appropriate handler."""
        if isinstance(command, CreateUserCommand):
            return self._create_user(command)
        elif isinstance(command, UpdateUserNameCommand):
            return self._update_user_name(command)
        elif isinstance(command, DeleteUserCommand):
            return self._delete_user(command)
        return None
    
    def _create_user(self, cmd: CreateUserCommand) -> str:
        """Create new user and emit event."""
        user_id = str(uuid.uuid4())
        user = User(
            id=user_id,
            email=cmd.email,
            name=cmd.name
        )
        self._users[user_id] = user
        
        # Emit event for read model synchronization
        self._emit_event({
            "type": "UserCreated",
            "user_id": user_id,
            "email": cmd.email,
            "name": cmd.name,
            "timestamp": datetime.now().isoformat()
        })
        
        return user_id
    
    def _update_user_name(self, cmd: UpdateUserNameCommand) -> str:
        """Update user name and emit event."""
        if cmd.user_id not in self._users:
            raise ValueError(f"User {cmd.user_id} not found")
        
        user = self._users[cmd.user_id]
        user.name = cmd.new_name
        user.version += 1
        
        self._emit_event({
            "type": "UserNameUpdated",
            "user_id": cmd.user_id,
            "new_name": cmd.new_name,
            "version": user.version,
            "timestamp": datetime.now().isoformat()
        })
        
        return cmd.user_id
    
    def _delete_user(self, cmd: DeleteUserCommand) -> str:
        """Delete user and emit event."""
        if cmd.user_id in self._users:
            del self._users[cmd.user_id]
            
            self._emit_event({
                "type": "UserDeleted",
                "user_id": cmd.user_id,
                "timestamp": datetime.now().isoformat()
            })
        
        return cmd.user_id
    
    def _emit_event(self, event: Dict[str, Any]) -> None:
        """Store event for event sourcing."""
        self._event_store.append(event)
        print(f"Event emitted: {event['type']}")

# Read Model (Query Handler)
@dataclass
class UserReadModel:
    """Denormalized read model optimized for queries."""
    id: str
    email: str
    name: str
    created_at: str
    search_index: str = ""  # Pre-computed for fast search

class UserReadRepository:
    """Handles all read operations - optimized for query performance."""
    
    def __init__(self) -> None:
        self._read_models: Dict[str, UserReadModel] = {}
        self._email_index: Dict[str, str] = {}  # Email -> User ID mapping
    
    def handle(self, query: Query) -> Any:
        """Dispatch query to appropriate handler."""
        if isinstance(query, GetUserQuery):
            return self._get_user(query)
        elif isinstance(query, ListUsersQuery):
            return self._list_users(query)
        return None
    
    def _get_user(self, query: GetUserQuery) -> Optional[UserReadModel]:
        """Get user by ID - direct lookup."""
        return self._read_models.get(query.user_id)
    
    def _list_users(self, query: ListUsersQuery) -> List[UserReadModel]:
        """List users with optional filtering."""
        users = list(self._read_models.values())
        
        if query.email_filter:
            users = [u for u in users if query.email_filter.lower() in u.email.lower()]
        
        return users[:query.limit]
    
    def apply_event(self, event: Dict[str, Any]) -> None:
        """Update read model based on domain event."""
        event_type = event["type"]
        
        if event_type == "UserCreated":
            self._read_models[event["user_id"]] = UserReadModel(
                id=event["user_id"],
                email=event["email"],
                name=event["name"],
                created_at=event["timestamp"],
                search_index=f"{event['name']} {event['email']}".lower()
            )
            self._email_index[event["email"]] = event["user_id"]
        
        elif event_type == "UserNameUpdated":
            if event["user_id"] in self._read_models:
                user = self._read_models[event["user_id"]]
                user.name = event["new_name"]
                user.search_index = f"{event['new_name']} {user.email}".lower()
        
        elif event_type == "UserDeleted":
            user_id = event["user_id"]
            if user_id in self._read_models:
                email = self._read_models[user_id].email
                del self._read_models[user_id]
                if email in self._email_index:
                    del self._email_index[email]

# CQRS Mediator
class UserCQRS:
    """Mediator that coordinates command and query sides."""
    
    def __init__(self) -> None:
        self.write_repo = UserWriteRepository()
        self.read_repo = UserReadRepository()
    
    def execute_command(self, command: Command) -> Optional[str]:
        """Execute command and synchronize read model."""
        result = self.write_repo.handle(command)
        
        # Apply new events to read model
        for event in self.write_repo._event_store:
            self.read_repo.apply_event(event)
        
        return result
    
    def execute_query(self, query: Query) -> Any:
        """Execute query on read model."""
        return self.read_repo.handle(query)

# Example usage
if __name__ == "__main__":
    cqrs = UserCQRS()
    
    # Commands (Write)
    user_id = cqrs.execute_command(CreateUserCommand(
        email="alice@example.com",
        name="Alice Smith"
    ))
    
    cqrs.execute_command(UpdateUserNameCommand(
        user_id=user_id,
        new_name="Alice Johnson"
    ))
    
    # Queries (Read)
    user = cqrs.execute_query(GetUserQuery(user_id=user_id))
    print(f"User: {user.name}, Email: {user.email}")
    
    users = cqrs.execute_query(ListUsersQuery(email_filter="alice"))
    print(f"Found {len(users)} user(s)")
```

**Why**: CQRS enables independent scaling of read and write workloads, optimized data models for each operation type, and is essential in event-sourced systems. Used extensively in high-traffic applications (e-commerce, social media) where read/write patterns differ significantly.

## Database Sharding (Horizontal Partitioning)

```python
"""
Database Sharding: Distribute data across multiple database instances.
Enables horizontal scaling for large-scale applications.
"""
from typing import Dict, Any, Optional, List, Callable
from dataclasses import dataclass, field
from abc import ABC, abstractmethod
import hashlib

@dataclass
class Shard:
    """Represents a single database shard."""
    id: int
    host: str
    port: int
    database: str
    data: Dict[str, Any] = field(default_factory=dict)  # In-memory simulation
    
    def insert(self, key: str, value: Any) -> None:
        """Insert data into shard."""
        self.data[key] = value
    
    def get(self, key: str) -> Optional[Any]:
        """Retrieve data from shard."""
        return self.data.get(key)
    
    def delete(self, key: str) -> bool:
        """Delete data from shard."""
        if key in self.data:
            del self.data[key]
            return True
        return False
    
    def count(self) -> int:
        """Get number of records in shard."""
        return len(self.data)

# Sharding Strategies
class ShardingStrategy(ABC):
    """Abstract base class for sharding strategies."""
    
    @abstractmethod
    def get_shard(self, key: str, num_shards: int) -> int:
        """Determine which shard to use for given key."""
        pass

class HashSharding(ShardingStrategy):
    """Hash-based sharding - consistent distribution."""
    
    def get_shard(self, key: str, num_shards: int) -> int:
        """
        Use hash function to determine shard.
        Time: O(1), ensures even distribution.
        """
        hash_value = int(hashlib.md5(key.encode()).hexdigest(), 16)
        return hash_value % num_shards

class RangeSharding(ShardingStrategy):
    """Range-based sharding - ordered data."""
    
    def __init__(self, boundaries: List[int]) -> None:
        """
        boundaries: List of values defining shard boundaries.
        Example: [1000, 5000, 10000] creates 4 shards:
          - Shard 0: key < 1000
          - Shard 1: 1000 <= key < 5000
          - Shard 2: 5000 <= key < 10000
          - Shard 3: key >= 10000
        """
        self.boundaries = boundaries
    
    def get_shard(self, key: str, num_shards: int) -> int:
        """
        Determine shard based on key value ranges.
        Best for numeric keys or ordered data.
        """
        try:
            numeric_key = int(key)
        except ValueError:
            # Fallback to hash if key is not numeric
            return HashSharding().get_shard(key, num_shards)
        
        for i, boundary in enumerate(self.boundaries):
            if numeric_key < boundary:
                return i
        return len(self.boundaries)

class DirectorySharding(ShardingStrategy):
    """Directory-based sharding - explicit mapping."""
    
    def __init__(self) -> None:
        """Initialize with empty lookup table."""
        self.lookup: Dict[str, int] = {}
    
    def add_mapping(self, key: str, shard_id: int) -> None:
        """Add explicit key-to-shard mapping."""
        self.lookup[key] = shard_id
    
    def get_shard(self, key: str, num_shards: int) -> int:
        """
        Look up shard from directory.
        Flexible but requires maintaining lookup table.
        """
        if key in self.lookup:
            return self.lookup[key]
        # Fallback to hash for unmapped keys
        return HashSharding().get_shard(key, num_shards)

# Sharded Database
class ShardedDatabase:
    """Manages multiple shards with pluggable sharding strategy."""
    
    def __init__(
        self,
        shards: List[Shard],
        strategy: ShardingStrategy
    ) -> None:
        """
        Initialize sharded database.
        
        Args:
            shards: List of Shard instances
            strategy: Sharding strategy to use
        """
        self.shards = shards
        self.strategy = strategy
        self.num_shards = len(shards)
    
    def _get_shard_for_key(self, key: str) -> Shard:
        """Get the appropriate shard for a key."""
        shard_id = self.strategy.get_shard(key, self.num_shards)
        return self.shards[shard_id]
    
    def insert(self, key: str, value: Any) -> None:
        """Insert data into appropriate shard."""
        shard = self._get_shard_for_key(key)
        shard.insert(key, value)
        print(f"Inserted '{key}' into Shard {shard.id}")
    
    def get(self, key: str) -> Optional[Any]:
        """Retrieve data from appropriate shard."""
        shard = self._get_shard_for_key(key)
        return shard.get(key)
    
    def delete(self, key: str) -> bool:
        """Delete data from appropriate shard."""
        shard = self._get_shard_for_key(key)
        return shard.delete(key)
    
    def get_stats(self) -> Dict[int, int]:
        """Get record count for each shard."""
        return {shard.id: shard.count() for shard in self.shards}
    
    def query_all_shards(self, predicate: Callable[[Any], bool]) -> List[Any]:
        """
        Query across all shards (fan-out query).
        Used when shard key is not known.
        """
        results = []
        for shard in self.shards:
            for value in shard.data.values():
                if predicate(value):
                    results.append(value)
        return results

# Consistent Hashing (Advanced Sharding)
class ConsistentHash:
    """
    Consistent hashing for minimal rebalancing during shard changes.
    Uses virtual nodes for better distribution.
    """
    
    def __init__(self, virtual_nodes: int = 150) -> None:
        """
        Initialize consistent hash ring.
        
        Args:
            virtual_nodes: Number of virtual nodes per physical node
        """
        self.virtual_nodes = virtual_nodes
        self.ring: Dict[int, int] = {}  # hash -> shard_id
        self.sorted_keys: List[int] = []
    
    def add_shard(self, shard_id: int) -> None:
        """Add a shard to the hash ring."""
        for i in range(self.virtual_nodes):
            virtual_key = f"shard_{shard_id}_vnode_{i}"
            hash_key = int(hashlib.md5(virtual_key.encode()).hexdigest(), 16)
            self.ring[hash_key] = shard_id
        
        self.sorted_keys = sorted(self.ring.keys())
    
    def remove_shard(self, shard_id: int) -> None:
        """Remove a shard from the hash ring."""
        keys_to_remove = [
            k for k, v in self.ring.items() if v == shard_id
        ]
        for key in keys_to_remove:
            del self.ring[key]
        
        self.sorted_keys = sorted(self.ring.keys())
    
    def get_shard(self, key: str, num_shards: int) -> int:
        """
        Find the shard responsible for a key.
        Uses binary search on the ring.
        """
        if not self.ring:
            return HashSharding().get_shard(key, num_shards)
        
        hash_key = int(hashlib.md5(key.encode()).hexdigest(), 16)
        
        # Binary search for first node >= hash_key
        left, right = 0, len(self.sorted_keys)
        while left < right:
            mid = (left + right) // 2
            if self.sorted_keys[mid] < hash_key:
                left = mid + 1
            else:
                right = mid
        
        # Wrap around to first node if needed
        if left == len(self.sorted_keys):
            left = 0
        
        return self.ring[self.sorted_keys[left]]
```

```python
from typing import (
    TypeVar, Generic, Protocol, Optional, List, Dict,
    Callable, Union, Any, Tuple, Type, runtime_checkable,
    overload, Final, Literal, TypedDict, NamedTuple
)
from dataclasses import dataclass
from abc import ABC, abstractmethod
from collections.abc import Iterator

# 1. Generics
T = TypeVar('T')
K = TypeVar('K')
V = TypeVar('V')

class Container(Generic[T]):
    """Generic container that holds items of type T."""
    
    def __init__(self) -> None:
        self._items: List[T] = []
    
    def add(self, item: T) -> None:
        """Add item to container."""
        self._items.append(item)
    
    def get(self, index: int) -> Optional[T]:
        """Get item at index."""
        if 0 <= index < len(self._items):
            return self._items[index]
        return None
    
    def get_all(self) -> List[T]:
        """Get all items."""
        return self._items.copy()

# Bounded TypeVar
Numeric = TypeVar('Numeric', int, float)

def add_numbers(a: Numeric, b: Numeric) -> Numeric:
    """Add two numeric values."""
    return a + b  # type: ignore

# 2. Protocols (Structural Subtyping)
@runtime_checkable
class Drawable(Protocol):
    """Protocol for objects that can be drawn."""
    
    def draw(self) -> str:
        """Draw the object."""
        ...

class Circle:
    """Circle class (implicitly implements Drawable)."""
    
    def __init__(self, radius: float) -> None:
        self.radius = radius
    
    def draw(self) -> str:
        return f"Circle with radius {self.radius}"

class Square:
    """Square class (implicitly implements Drawable)."""
    
    def __init__(self, side: float) -> None:
        self.side = side
    
    def draw(self) -> str:
        return f"Square with side {self.side}"

def render(obj: Drawable) -> str:
    """Render any drawable object."""
    return obj.draw()

# 3. Covariance and Contravariance
# Covariant TypeVar (can only appear in return positions)
T_co = TypeVar('T_co', covariant=True)

class Producer(Generic[T_co]):
    """Covariant generic - only produces values."""
    
    def __init__(self, value: T_co) -> None:
        self._value = value
    
    def get(self) -> T_co:
        """Produce a value."""
        return self._value

# Contravariant TypeVar (can only appear in argument positions)
T_contra = TypeVar('T_contra', contravariant=True)

class Consumer(Generic[T_contra]):
    """Contravariant generic - only consumes values."""
    
    def process(self, value: T_contra) -> None:
        """Consume a value."""
        print(f"Processing: {value}")

# 4. Function Overloading
class Calculator:
    """Calculator with overloaded methods."""
    
    @overload
    def calculate(self, value: int) -> int: ...
    
    @overload
    def calculate(self, value: float) -> float: ...
    
    @overload
    def calculate(self, value: str) -> str: ...
    
    def calculate(self, value: Union[int, float, str]) -> Union[int, float, str]:
        """Calculate based on input type."""
        if isinstance(value, int):
            return value * 2
        elif isinstance(value, float):
            return value * 2.0
        else:
            return value + value

# 5. Literal Types
Direction = Literal['up', 'down', 'left', 'right']

def move(direction: Direction, steps: int) -> str:
    """Move in a direction (type-safe)."""
    return f"Moving {direction} by {steps} steps"

# Status code literal
Status = Literal[200, 404, 500]

def handle_response(status: Status) -> str:
    """Handle HTTP status codes."""
    responses: Dict[int, str] = {
        200: "OK",
        404: "Not Found",
        500: "Internal Server Error"
    }
    return responses[status]

# 6. TypedDict
class UserDict(TypedDict):
    """Type-safe dictionary structure."""
    id: int
    name: str
    email: str
    age: NotRequired[int]  # Optional field (Python 3.11+)

def create_user(data: UserDict) -> str:
    """Create user from typed dict."""
    return f"User {data['name']} <{data['email']}>"

# 7. NamedTuple with type hints
class Point(NamedTuple):
    """Immutable point with x, y coordinates."""
    x: float
    y: float
    
    def distance_from_origin(self) -> float:
        """Calculate distance from origin."""
        return (self.x ** 2 + self.y ** 2) ** 0.5

# 8. Callable Types
EventHandler = Callable[[str, Dict[str, Any]], None]
Comparator = Callable[[T, T], int]

def sort_with_comparator(
    items: List[T],
    comparator: Comparator[T]
) -> List[T]:
    """Sort items using custom comparator."""
    return sorted(items, key=lambda x: comparator(x, x))

# 9. Final and Type Guards
class Config:
    """Configuration with final constants."""
    
    MAX_RETRIES: Final[int] = 3
    API_VERSION: Final[str] = "v2"
    
    def __init__(self) -> None:
        self.timeout: Final[int] = 30

def is_string_list(value: List[Any]) -> bool:
    """Type guard for list of strings."""
    return all(isinstance(v, str) for v in value)

# 10. Advanced Union Types and Type Aliases
JSON = Union[None, bool, int, float, str, List['JSON'], Dict[str, 'JSON']]
Result[T] = Union[T, Exception]  # Python 3.12+ syntax

# Type alias (recommended way in Python 3.10+)
type Vector = List[float]
type Matrix = List[Vector]

# Example usage
if __name__ == "__main__":
    # Generics
    int_container: Container[int] = Container()
    int_container.add(42)
    print(f"Item: {int_container.get(0)}")
    
    # Protocols
    circle = Circle(5.0)
    square = Square(3.0)
    print(render(circle))
    print(render(square))
    
    # Function overloading
    calc = Calculator()
    print(calc.calculate(10))      # int
    print(calc.calculate(3.14))    # float
    
    # Literal types
    print(move('up', 5))
    print(handle_response(200))
    
    # TypedDict
    user_data: UserDict = {
        'id': 1,
        'name': 'Alice',
        'email': 'alice@example.com',
        'age': 30
    }
    print(create_user(user_data))
    
    # NamedTuple
    point = Point(3.0, 4.0)
    print(f"Distance: {point.distance_from_origin()}")
```
