## 1. Radix Trie (Patricia Trie)

```python
# PROBLEM: Efficient prefix-based string storage and lookup with path compression
# TIME: O(k) for insert/search/delete where k is key length, SPACE: O(n * k)
class RadixTrieNode:
    """Node with single string label and children mapping."""
    
    def __init__(self):
        self.label: str = ""
        self.children: dict[str, "RadixTrieNode"] = {}
        self.is_end: bool = False


class RadixTrie:
    """
    Compressed trie that merges nodes with single child.
    Reduces space by collapsing chains into single edges.
    """
    
    def __init__(self):
        self.root = RadixTrieNode()
    
    def _common_prefix(self, a: str, b: str) -> str:
        """Find longest common prefix between two strings."""
        min_len = min(len(a), len(b))
        for i in range(min_len):
            if a[i] != b[i]:
                return a[:i]
        return a[:min_len]
    
    def insert(self, word: str) -> None:
        """Insert word into radix trie with path compression."""
        node = self.root
        
        while word:
            # Find child with matching prefix
            for key, child in list(node.children.items()):
                prefix = self._common_prefix(word, key)
                
                if prefix:
                    if prefix == key:
                        # Full key matches, continue to child
                        node = child
                        word = word[len(prefix):]
                        break
                    else:
                        # Partial match, split the node
                        new_node = RadixTrieNode()
                        new_node.label = prefix
                        new_node.children[key[len(prefix):]] = child
                        child.label = key[len(prefix):]
                        
                        del node.children[key]
                        node.children[prefix] = new_node
                        node = new_node
                        word = word[len(prefix):]
                        break
            else:
                # No matching child, create new node
                new_node = RadixTrieNode()
                new_node.label = word
                new_node.is_end = True
                node.children[word] = new_node
                return
        
        node.is_end = True
    
    def search(self, word: str) -> bool:
        """Check if exact word exists in trie."""
        node = self.root
        
        while word:
            for key, child in node.children.items():
                if word.startswith(key):
                    node = child
                    word = word[len(key):]
                    break
            else:
                return False
        
        return node.is_end
    
    def starts_with(self, prefix: str) -> bool:
        """Check if any word starts with given prefix."""
        node = self.root
        
        while prefix:
            for key, child in node.children.items():
                if prefix.startswith(key):
                    node = child
                    prefix = prefix[len(key):]
                    break
                elif key.startswith(prefix):
                    return True
            else:
                return False
        
        return True
```

**Why**: Radix tries compress paths with single children, reducing memory by 30-50% compared to standard tries. Ideal for autocomplete and IP routing tables.

## 2. Min-Max Heap (Double-Ended Priority Queue)

```python
# PROBLEM: Support O(log n) retrieval of both min and max elements
# TIME: O(log n) insert/delete, O(1) find_min/find_max, SPACE: O(n)
class MinMaxHeap:
    """
    Alternating min and max levels for efficient double-ended priority queue.
    Level i is min-level if floor(log2(i+1)) is even, else max-level.
    """
    
    def __init__(self):
        self.data: list[int] = []
    
    def _is_min_level(self, i: int) -> bool:
        """Check if index is at a min-level."""
        level = (i + 1).bit_length() - 1
        return level % 2 == 0
    
    def _parent(self, i: int) -> int | None:
        return (i - 1) // 2 if i > 0 else None
    
    def _children(self, i: int) -> list[int]:
        """Get children and grandchildren indices."""
        left = 2 * i + 1
        right = 2 * i + 2
        result = []
        for child in [left, right]:
            if child < len(self.data):
                result.append(child)
                # Add grandchildren
                gc1, gc2 = 2 * child + 1, 2 * child + 2
                if gc1 < len(self.data):
                    result.append(gc1)
                if gc2 < len(self.data):
                    result.append(gc2)
        return result
    
    def _compare(self, i: int, j: int) -> bool:
        """Compare based on level: < for min-level, > for max-level."""
        if self._is_min_level(i):
            return self.data[i] < self.data[j]
        return self.data[i] > self.data[j]
    
    def _get_extreme_descendant(self, i: int) -> int:
        """Find min or max among children and grandchildren."""
        candidates = self._children(i)
        if not candidates:
            return i
        extreme = i
        for c in candidates:
            if self._compare(c, extreme):
                extreme = c
        return extreme
    
    def _bubble_up(self, i: int) -> None:
        """Restore heap property upward."""
        parent = self._parent(i)
        if parent is None:
            return
        
        grandparent = self._parent(parent)
        if grandparent is None:
            return
        
        # Compare with grandparent based on current level
        if self._compare(i, grandparent):
            self.data[i], self.data[grandparent] = self.data[grandparent], self.data[i]
            self._bubble_up(grandparent)
    
    def push(self, val: int) -> None:
        """Insert value into min-max heap."""
        self.data.append(val)
        i = len(self.data) - 1
        
        parent = self._parent(i)
        if parent is None:
            return
        
        # Determine comparison direction with parent
        parent_is_min = self._is_min_level(parent)
        if parent_is_min:
            if self.data[i] < self.data[parent]:
                self.data[i], self.data[parent] = self.data[parent], self.data[i]
                self._bubble_up(parent)
            elif self.data[i] > self.data[parent]:
                self._bubble_up(i)
        else:
            if self.data[i] > self.data[parent]:
                self.data[i], self.data[parent] = self.data[parent], self.data[i]
                self._bubble_up(parent)
            elif self.data[i] < self.data[parent]:
                self._bubble_up(i)
    
    def find_min(self) -> int | None:
        """Return minimum element (always at root)."""
        return self.data[0] if self.data else None
    
    def find_max(self) -> int | None:
        """Return maximum element (at level 1)."""
        if not self.data:
            return None
        if len(self.data) == 1:
            return self.data[0]
        if len(self.data) == 2:
            return self.data[1]
        return max(self.data[1], self.data[2])
```

**Why**: Single data structure supporting O(1) min/max access. Perfect for sliding window median, priority scheduling with both urgent and batch jobs.

## 3. Service Locator Pattern

```python
# PROBLEM: Decouple service consumers from concrete implementations
# TIME: O(1) for lookup, SPACE: O(n) for registered services
from typing import Callable, TypeVar, Generic, Any
from dataclasses import dataclass
from functools import wraps

T = TypeVar("T")


@dataclass
class ServiceDescriptor(Generic[T]):
    """Holds service registration metadata."""
    factory: Callable[[], T]
    singleton: bool = True
    instance: T | None = None


class ServiceLocator:
    """
    Centralized registry for service resolution.
    Supports both transient and singleton lifetimes.
    """
    
    _services: dict[type, ServiceDescriptor] = {}
    _resolving: set[type] = set()  # Circular dependency detection
    
    @classmethod
    def register(
        cls,
        interface: type[T],
        factory: Callable[[], T],
        singleton: bool = True
    ) -> None:
        """Register a service with its factory function."""
        cls._services[interface] = ServiceDescriptor(factory, singleton)
    
    @classmethod
    def resolve(cls, interface: type[T]) -> T:
        """Resolve service instance, handling singletons and circular deps."""
        if interface not in cls._services:
            raise KeyError(f"Service not registered: {interface.__name__}")
        
        descriptor = cls._services[interface]
        
        # Return cached singleton if exists
        if descriptor.singleton and descriptor.instance is not None:
            return descriptor.instance
        
        # Detect circular dependencies
        if interface in cls._resolving:
            raise RuntimeError(f"Circular dependency detected: {interface.__name__}")
        
        cls._resolving.add(interface)
        try:
            instance = descriptor.factory()
            if descriptor.singleton:
                descriptor.instance = instance
            return instance
        finally:
            cls._resolving.discard(interface)
    
    @classmethod
    def clear(cls) -> None:
        """Clear all registrations (useful for testing)."""
        cls._services.clear()
        cls._resolving.clear()


def inject(*interfaces: type) -> Callable:
    """Decorator for dependency injection via service locator."""
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            services = [ServiceLocator.resolve(i) for i in interfaces]
            return func(*services, *args, **kwargs)
        return wrapper
    return decorator


# Usage example
class DatabaseService:
    def query(self, sql: str) -> list:
        return [{"id": 1, "name": "Alice"}]


class CacheService:
    def __init__(self):
        self._cache: dict[str, Any] = {}
    
    def get(self, key: str) -> Any | None:
        return self._cache.get(key)
    
    def set(self, key: str, value: Any) -> None:
        self._cache[key] = value


# Registration
ServiceLocator.register(DatabaseService, DatabaseService)
ServiceLocator.register(CacheService, CacheService)
```

**Why**: Enables loose coupling and testability. Services can be swapped for mocks without changing consumer code. Common in plugin architectures and legacy code modernization.

## 4. Unit of Work Pattern

```python
# PROBLEM: Coordinate multiple repository operations in a single transaction
# TIME: O(1) commit/rollback, O(n) for tracked entities, SPACE: O(n)
from dataclasses import dataclass, field
from enum import Enum
from typing import Generic, TypeVar, Protocol, Any
from contextlib import contextmanager

T = TypeVar("T")


class EntityState(Enum):
    """Track entity lifecycle state."""
    ADDED = "added"
    MODIFIED = "modified"
    DELETED = "deleted"
    UNCHANGED = "unchanged"


@dataclass
class TrackedEntity(Generic[T]):
    """Wraps entity with change tracking metadata."""
    entity: T
    state: EntityState = EntityState.UNCHANGED
    original: dict[str, Any] = field(default_factory=dict)


class Repository(Protocol[T]):
    """Repository interface for data access."""
    
    def add(self, entity: T) -> None: ...
    def update(self, entity: T) -> None: ...
    def delete(self, entity: T) -> None: ...
    def find_by_id(self, id: int) -> T | None: ...


class UnitOfWork:
    """
    Manages transaction scope for multiple repositories.
    Tracks changes and coordinates commit/rollback.
    """
    
    def __init__(self):
        self._tracked: dict[int, TrackedEntity] = {}
        self._repositories: dict[type, Repository] = {}
        self._committed: bool = False
    
    def register_repository(self, entity_type: type, repo: Repository) -> None:
        """Associate repository with entity type."""
        self._repositories[entity_type] = repo
    
    def track(self, entity: Any, state: EntityState) -> None:
        """Start tracking an entity with given state."""
        entity_id = id(entity)
        if entity_id not in self._tracked:
            # Capture original state for rollback
            original = {k: v for k, v in vars(entity).items()}
            self._tracked[entity_id] = TrackedEntity(entity, state, original)
        else:
            # Only escalate state (deleted > modified > added > unchanged)
            tracked = self._tracked[entity_id]
            if state.value > tracked.state.value:
                tracked.state = state
    
    def commit(self) -> None:
        """Persist all tracked changes to repositories."""
        if self._committed:
            raise RuntimeError("Unit of Work already committed")
        
        # Group by entity type for batch operations
        by_type: dict[type, list[TrackedEntity]] = {}
        for tracked in self._tracked.values():
            entity_type = type(tracked.entity)
            by_type.setdefault(entity_type, []).append(tracked)
        
        # Apply changes to repositories
        for entity_type, tracked_list in by_type.items():
            repo = self._repositories.get(entity_type)
            if not repo:
                continue
            
            for tracked in tracked_list:
                if tracked.state == EntityState.ADDED:
                    repo.add(tracked.entity)
                elif tracked.state == EntityState.MODIFIED:
                    repo.update(tracked.entity)
                elif tracked.state == EntityState.DELETED:
                    repo.delete(tracked.entity)
        
        self._committed = True
    
    def rollback(self) -> None:
        """Restore all entities to original state."""
        for tracked in self._tracked.values():
            for key, value in tracked.original.items():
                setattr(tracked.entity, key, value)
        
        self._tracked.clear()
    
    @contextmanager
    def transaction(self):
        """Context manager for automatic commit/rollback."""
        try:
            yield self
            self.commit()
        except Exception:
            self.rollback()
            raise


# Usage with entity classes
@dataclass
class User:
    id: int
    name: str
    email: str
```

**Why**: Ensures atomicity across multiple aggregates. Prevents partial updates and maintains consistency. Essential for complex business operations spanning multiple tables.

## 5. Circuit Breaker State Machine

```python
# PROBLEM: Prevent cascading failures by stopping calls to failing services
# TIME: O(1) for state checks, SPACE: O(1)
from enum import Enum
from dataclasses import dataclass, field
from typing import Callable, Any
from time import time
from functools import wraps


class CircuitState(Enum):
    """Circuit breaker states."""
    CLOSED = "closed"        # Normal operation
    OPEN = "open"            # Failing, reject all calls
    HALF_OPEN = "half_open"  # Testing if service recovered


@dataclass
class CircuitStats:
    """Track call statistics for state transitions."""
    failures: int = 0
    successes: int = 0
    last_failure_time: float = 0
    last_success_time: float = 0


class CircuitBreaker:
    """
    Implements circuit breaker pattern with three states.
    Protects system from cascading failures.
    """
    
    def __init__(
        self,
        failure_threshold: int = 5,
        recovery_timeout: float = 30.0,
        success_threshold: int = 3,
        name: str = "default"
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.success_threshold = success_threshold
        self.name = name
        
        self._state = CircuitState.CLOSED
        self._stats = CircuitStats()
        self._state_changed_at = time()
    
    @property
    def state(self) -> CircuitState:
        """Get current state, with automatic transition from OPEN."""
        if self._state == CircuitState.OPEN:
            elapsed = time() - self._stats.last_failure_time
            if elapsed >= self.recovery_timeout:
                self._transition(CircuitState.HALF_OPEN)
        return self._state
    
    def _transition(self, new_state: CircuitState) -> None:
        """Handle state transition and reset counters."""
        self._state = new_state
        self._state_changed_at = time()
        
        if new_state == CircuitState.CLOSED:
            self._stats.failures = 0
            self._stats.successes = 0
        elif new_state == CircuitState.HALF_OPEN:
            self._stats.successes = 0
    
    def _on_success(self) -> None:
        """Handle successful call."""
        self._stats.successes += 1
        self._stats.last_success_time = time()
        
        if self._state == CircuitState.HALF_OPEN:
            if self._stats.successes >= self.success_threshold:
                self._transition(CircuitState.CLOSED)
    
    def _on_failure(self) -> None:
        """Handle failed call."""
        self._stats.failures += 1
        self._stats.last_failure_time = time()
        
        if self._state == CircuitState.HALF_OPEN:
            self._transition(CircuitState.OPEN)
        elif self._state == CircuitState.CLOSED:
            if self._stats.failures >= self.failure_threshold:
                self._transition(CircuitState.OPEN)
    
    def call(self, func: Callable[[], Any]) -> Any:
        """Execute function through circuit breaker."""
        current_state = self.state
        
        if current_state == CircuitState.OPEN:
            raise RuntimeError(f"Circuit '{self.name}' is OPEN - rejecting call")
        
        try:
            result = func()
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise


def circuit_breaker(
    failure_threshold: int = 5,
    recovery_timeout: float = 30.0,
    name: str | None = None
) -> Callable:
    """Decorator to wrap function with circuit breaker."""
    def decorator(func: Callable) -> Callable:
        cb = CircuitBreaker(
            failure_threshold=failure_threshold,
            recovery_timeout=recovery_timeout,
            name=name or func.__name__
        )
        
        @wraps(func)
        def wrapper(*args, **kwargs):
            return cb.call(lambda: func(*args, **kwargs))
        
        wrapper.circuit_breaker = cb  # Allow external inspection
        return wrapper
    
    return decorator
```

**Why**: Prevents resource exhaustion from repeated failing calls. Allows failing services time to recover. Key pattern for microservices resilience.

## 6. Adaptive Replacement Cache (ARC)

```python
# PROBLEM: Self-tuning cache that balances recency and frequency
# TIME: O(1) for all operations, SPACE: O(capacity)
from collections import OrderedDict
from dataclasses import dataclass
from enum import Enum


class ListType(Enum):
    T1 = "t1"  # Recent items
    T2 = "t2"  # Frequent items
    B1 = "b1"  # Recent ghosts (evicted from T1)
    B2 = "b2"  # Frequent ghosts (evicted from T2)


@dataclass
class CacheEntry:
    value: any
    list_type: ListType


class ARCache:
    """
    Adaptive Replacement Cache.
    Automatically tunes ratio between recency (LRU) and frequency (LFU).
    """
    
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.p = 0  # Target size for T1
        
        # Active caches
        self.t1: OrderedDict[str, CacheEntry] = OrderedDict()  # Recent
        self.t2: OrderedDict[str, CacheEntry] = OrderedDict()  # Frequent
        
        # Ghost lists (eviction history)
        self.b1: OrderedDict[str, None] = OrderedDict()  # Recent ghosts
        self.b2: OrderedDict[str, None] = OrderedDict()  # Frequent ghosts
    
    def _replace(self, key: str | None) -> None:
        """Evict from T1 or T2 based on current target p."""
        t1_size = len(self.t1)
        
        if t1_size > 0 and (t1_size > self.p or (t1_size == self.p and key in self.b2)):
            # Evict from T1
            oldest_key, _ = self.t1.popitem(last=False)
            self.b1[oldest_key] = None
        elif self.t2:
            # Evict from T2
            oldest_key, _ = self.t2.popitem(last=False)
            self.b2[oldest_key] = None
    
    def get(self, key: str) -> any | None:
        """Retrieve value and promote if found."""
        # Check T1
        if key in self.t1:
            entry = self.t1.pop(key)
            self.t2[key] = entry  # Promote to frequent
            return entry.value
        
        # Check T2
        if key in self.t2:
            entry = self.t2.pop(key)
            self.t2[key] = entry  # Refresh position
            return entry.value
        
        return None
    
    def put(self, key: str, value: any) -> None:
        """Insert or update entry, managing cache sizes."""
        # Case 1: Key exists in T1 - move to T2
        if key in self.t1:
            self.t1.pop(key)
            self.t2[key] = CacheEntry(value, ListType.T2)
            return
        
        # Case 2: Key exists in T2 - update
        if key in self.t2:
            self.t2.pop(key)
            self.t2[key] = CacheEntry(value, ListType.T2)
            return
        
        # Case 3: Key in B1 ghost - increase p, fetch to T2
        if key in self.b1:
            self.b1.pop(key)
            # Adapt: increase target for T1
            delta = max(1, len(self.b2) // max(1, len(self.b1)))
            self.p = min(self.capacity, self.p + delta)
            self._replace(key)
            self.t2[key] = CacheEntry(value, ListType.T2)
            return
        
        # Case 4: Key in B2 ghost - decrease p, fetch to T2
        if key in self.b2:
            self.b2.pop(key)
            # Adapt: decrease target for T1
            delta = max(1, len(self.b1) // max(1, len(self.b2)))
            self.p = max(0, self.p - delta)
            self._replace(key)
            self.t2[key] = CacheEntry(value, ListType.T2)
            return
        
        # Case 5: New key - insert into T1
        if len(self.t1) + len(self.t2) >= self.capacity:
            self._replace(key)
        
        self.t1[key] = CacheEntry(value, ListType.T1)
    
    @property
    def size(self) -> int:
        return len(self.t1) + len(self.t2)
    
    @property
    def hit_rate(self) -> float:
        total = self.size + len(self.b1) + len(self.b2)
        return self.size / total if total > 0 else 0.0
```

**Why**: Self-optimizing for different access patterns. Outperforms both LRU and LFU in mixed workloads. IBM patent now freely available.

## 7. Two-Phase Commit (2PC)

```python
# PROBLEM: Ensure atomic transaction across distributed nodes
# TIME: O(n) for n participants with 2 network rounds, SPACE: O(n)
from dataclasses import dataclass
from enum import Enum
from typing import Protocol, Any
from asyncio import gather


class Vote(Enum):
    """Participant vote in prepare phase."""
    COMMIT = "commit"
    ABORT = "abort"


class TransactionState(Enum):
    """Coordinator transaction state."""
    INIT = "init"
    PREPARING = "preparing"
    PREPARED = "prepared"
    COMMITTING = "committing"
    COMMITTED = "committed"
    ABORTING = "aborting"
    ABORTED = "aborted"


class Participant(Protocol):
    """Interface for 2PC participants."""
    
    async def prepare(self, txn_id: str, data: Any) -> Vote: ...
    async def commit(self, txn_id: str) -> None: ...
    async def rollback(self, txn_id: str) -> None: ...


@dataclass
class ParticipantRecord:
    """Track participant state in transaction."""
    participant: Participant
    vote: Vote | None = None
    completed: bool = False


class TwoPhaseCoordinator:
    """
    Coordinates atomic commit across distributed participants.
    Implements classic 2PC protocol with logging for recovery.
    """
    
    def __init__(self, coordinator_id: str):
        self.coordinator_id = coordinator_id
        self._transactions: dict[str, TransactionState] = {}
        self._participants: dict[str, list[ParticipantRecord]] = {}
        self._log: list[str] = []  # Simplified write-ahead log
    
    def _log_event(self, txn_id: str, event: str) -> None:
        """Persist event for crash recovery."""
        entry = f"[{self.coordinator_id}] TXN={txn_id} EVENT={event}"
        self._log.append(entry)
        # In production: flush to durable storage
    
    async def begin_transaction(
        self,
        txn_id: str,
        participants: list[Participant],
        data: Any
    ) -> bool:
        """
        Execute two-phase commit protocol.
        Returns True if committed, False if aborted.
        """
        # Initialize transaction
        self._transactions[txn_id] = TransactionState.INIT
        self._participants[txn_id] = [
            ParticipantRecord(p) for p in participants
        ]
        self._log_event(txn_id, "BEGIN")
        
        # Phase 1: Prepare
        self._transactions[txn_id] = TransactionState.PREPARING
        prepare_tasks = [
            record.participant.prepare(txn_id, data)
            for record in self._participants[txn_id]
        ]
        
        try:
            votes = await gather(*prepare_tasks)
            
            # Record votes
            for record, vote in zip(self._participants[txn_id], votes):
                record.vote = vote
            
            # Check if all voted COMMIT
            all_commit = all(v == Vote.COMMIT for v in votes)
            
            if all_commit:
                self._transactions[txn_id] = TransactionState.PREPARED
                self._log_event(txn_id, "PREPARED")
                return await self._commit_phase(txn_id)
            else:
                self._log_event(txn_id, "ABORT_VOTE")
                return await self._abort_phase(txn_id)
        
        except Exception as e:
            self._log_event(txn_id, f"ABORT_ERROR: {e}")
            return await self._abort_phase(txn_id)
    
    async def _commit_phase(self, txn_id: str) -> bool:
        """Phase 2: Commit all participants."""
        self._transactions[txn_id] = TransactionState.COMMITTING
        self._log_event(txn_id, "COMMIT_START")
        
        commit_tasks = [
            record.participant.commit(txn_id)
            for record in self._participants[txn_id]
        ]
        
        try:
            await gather(*commit_tasks)
            self._transactions[txn_id] = TransactionState.COMMITTED
            self._log_event(txn_id, "COMMITTED")
            return True
        except Exception as e:
            # Must retry commit until success (blocking)
            self._log_event(txn_id, f"COMMIT_ERROR: {e}")
            raise RuntimeError(f"Commit failed, manual intervention needed: {e}")
    
    async def _abort_phase(self, txn_id: str) -> bool:
        """Rollback all participants that prepared."""
        self._transactions[txn_id] = TransactionState.ABORTING
        self._log_event(txn_id, "ABORT_START")
        
        abort_tasks = [
            record.participant.rollback(txn_id)
            for record in self._participants[txn_id]
            if record.vote == Vote.COMMIT
        ]
        
        if abort_tasks:
            await gather(*abort_tasks, return_exceptions=True)
        
        self._transactions[txn_id] = TransactionState.ABORTED
        self._log_event(txn_id, "ABORTED")
        return False
```

**Why**: Guarantees atomicity across distributed systems. Standard for distributed databases. Note: blocking protocol - consider 3PC or Saga for long transactions.

## 8. Consistent Hashing with Virtual Nodes

```python
# PROBLEM: Distribute keys across nodes with minimal remapping on changes
# TIME: O(log n) lookup with n virtual nodes, SPACE: O(n * replicas)
import hashlib
from bisect import bisect_right
from typing import Any


class ConsistentHash:
    """
    Consistent hash ring with virtual nodes for balanced distribution.
    Each physical node maps to multiple positions on the ring.
    """
    
    def __init__(self, replicas: int = 150, hash_func: Any = None):
        """
        Args:
            replicas: Number of virtual nodes per physical node.
                     Higher values = better distribution but more memory.
            hash_func: Hash function (default: MD5 -> int).
        """
        self.replicas = replicas
        self.hash_func = hash_func or self._default_hash
        
        self.ring: dict[int, str] = {}  # hash -> node
        self.sorted_keys: list[int] = []  # sorted ring positions
        self.nodes: set[str] = set()
    
    @staticmethod
    def _default_hash(key: str) -> int:
        """MD5-based hash for consistent distribution."""
        return int(hashlib.md5(key.encode()).hexdigest(), 16)
    
    def _get_ring_positions(self, node: str) -> list[int]:
        """Generate ring positions for all virtual nodes."""
        positions = []
        for i in range(self.replicas):
            virtual_key = f"{node}:vnode:{i}"
            positions.append(self.hash_func(virtual_key))
        return positions
    
    def add_node(self, node: str) -> int:
        """Add node with virtual nodes to the ring."""
        if node in self.nodes:
            return 0
        
        self.nodes.add(node)
        positions = self._get_ring_positions(node)
        
        for pos in positions:
            self.ring[pos] = node
            self.sorted_keys.append(pos)
        
        self.sorted_keys.sort()
        return len(positions)
    
    def remove_node(self, node: str) -> int:
        """Remove node and all its virtual nodes from ring."""
        if node not in self.nodes:
            return 0
        
        self.nodes.remove(node)
        positions = self._get_ring_positions(node)
        removed = 0
        
        for pos in positions:
            if pos in self.ring:
                del self.ring[pos]
                self.sorted_keys.remove(pos)
                removed += 1
        
        return removed
    
    def get_node(self, key: str) -> str | None:
        """Find node responsible for given key."""
        if not self.ring:
            return None
        
        key_hash = self.hash_func(key)
        
        # Find first node >= key_hash
        idx = bisect_right(self.sorted_keys, key_hash)
        
        # Wrap around to first node if past end
        if idx >= len(self.sorted_keys):
            idx = 0
        
        return self.ring[self.sorted_keys[idx]]
    
    def get_nodes(self, key: str, count: int = 3) -> list[str]:
        """Get multiple nodes for replication (clockwise on ring)."""
        if not self.ring or count <= 0:
            return []
        
        key_hash = self.hash_func(key)
        idx = bisect_right(self.sorted_keys, key_hash)
        
        result = []
        seen = set()
        
        for i in range(len(self.sorted_keys)):
            pos_idx = (idx + i) % len(self.sorted_keys)
            node = self.ring[self.sorted_keys[pos_idx]]
            
            if node not in seen:
                seen.add(node)
                result.append(node)
                
                if len(result) >= count:
                    break
        
        return result
    
    def get_distribution(self, keys: list[str]) -> dict[str, int]:
        """Calculate key distribution across nodes."""
        distribution = {node: 0 for node in self.nodes}
        
        for key in keys:
            node = self.get_node(key)
            if node:
                distribution[node] += 1
        
        return distribution
```

**Why**: Only K/N keys need remapping when node joins/leaves (vs all with modulo hashing). Virtual nodes ensure even distribution. Foundation for distributed caches and databases.

## 9. Saga Pattern with Compensating Transactions

```python
# PROBLEM: Maintain consistency across services without distributed transactions
# TIME: O(n) for n steps with sequential execution, SPACE: O(n)
from dataclasses import dataclass, field
from enum import Enum
from typing import Callable, Any, Awaitable
from asyncio import gather


class SagaState(Enum):
    """Saga execution state."""
    RUNNING = "running"
    COMPLETED = "completed"
    COMPENSATING = "compensating"
    COMPENSATED = "compensated"
    FAILED = "failed"


@dataclass
class SagaStep:
    """Single saga step with forward and compensating actions."""
    name: str
    action: Callable[[], Awaitable[Any]]  # Forward action
    compensate: Callable[[Any], Awaitable[None]]  # Rollback action
    result: Any = None
    executed: bool = False


@dataclass
class SagaExecution:
    """Tracks saga state and executed steps."""
    saga_id: str
    steps: list[SagaStep]
    state: SagaState = SagaState.RUNNING
    current_step: int = 0
    completed_steps: list[int] = field(default_factory=list)
    error: Exception | None = None


class SagaOrchestrator:
    """
    Orchestrates saga execution with compensation on failure.
    Coordinates distributed transactions across services.
    """
    
    def __init__(self):
        self._executions: dict[str, SagaExecution] = {}
    
    async def execute_saga(
        self,
        saga_id: str,
        steps: list[SagaStep]
    ) -> tuple[bool, Any | None]:
        """
        Execute saga steps sequentially with rollback on failure.
        Returns (success, final_result_or_error).
        """
        execution = SagaExecution(saga_id, steps)
        self._executions[saga_id] = execution
        
        try:
            # Execute forward actions
            for i, step in enumerate(steps):
                execution.current_step = i
                
                try:
                    step.result = await step.action()
                    step.executed = True
                    execution.completed_steps.append(i)
                except Exception as e:
                    execution.error = e
                    execution.state = SagaState.COMPENSATING
                    
                    # Rollback completed steps in reverse order
                    await self._compensate(execution)
                    
                    execution.state = SagaState.COMPENSATED
                    return (False, str(e))
            
            execution.state = SagaState.COMPLETED
            return (True, steps[-1].result if steps else None)
        
        except Exception as e:
            execution.state = SagaState.FAILED
            execution.error = e
            return (False, str(e))
    
    async def _compensate(self, execution: SagaExecution) -> None:
        """Execute compensating transactions in reverse order."""
        for step_idx in reversed(execution.completed_steps):
            step = execution.steps[step_idx]
            
            try:
                await step.compensate(step.result)
            except Exception as e:
                # Log but continue compensating
                print(f"Compensation failed for {step.name}: {e}")
    
    def get_execution(self, saga_id: str) -> SagaExecution | None:
        """Retrieve saga execution state."""
        return self._executions.get(saga_id)


# Example: Order fulfillment saga
async def create_order_saga():
    """Demonstrates saga for order fulfillment."""
    
    orchestrator = SagaOrchestrator()
    
    # Shared state between steps
    context = {"order_id": None, "payment_id": None, "shipment_id": None}
    
    steps = [
        SagaStep(
            name="create_order",
            action=lambda: _create_order(context),
            compensate=lambda _: _cancel_order(context)
        ),
        SagaStep(
            name="reserve_inventory",
            action=lambda: _reserve_inventory(context),
            compensate=lambda _: _release_inventory(context)
        ),
        SagaStep(
            name="process_payment",
            action=lambda: _process_payment(context),
            compensate=lambda _: _refund_payment(context)
        ),
        SagaStep(
            name="create_shipment",
            action=lambda: _create_shipment(context),
            compensate=lambda _: _cancel_shipment(context)
        ),
    ]
    
    return await orchestrator.execute_saga("order-123", steps)


# Mock service functions
async def _create_order(ctx: dict) -> str:
    ctx["order_id"] = "ORD-123"
    return ctx["order_id"]

async def _cancel_order(ctx: dict) -> None:
    print(f"Cancelling order {ctx['order_id']}")

async def _reserve_inventory(ctx: dict) -> bool:
    return True

async def _release_inventory(ctx: dict) -> None:
    print("Releasing inventory")

async def _process_payment(ctx: dict) -> str:
    ctx["payment_id"] = "PAY-123"
    return ctx["payment_id"]

async def _refund_payment(ctx: dict) -> None:
    print(f"Refunding payment {ctx['payment_id']}")

async def _create_shipment(ctx: dict) -> str:
    ctx["shipment_id"] = "SHIP-123"
    return ctx["shipment_id"]

async def _cancel_shipment(ctx: dict) -> None:
    print(f"Cancelling shipment {ctx['shipment_id']}")
```

**Why**: Alternative to 2PC for long-running transactions. Each service handles its own compensation. No global locks. Popular in microservices for order processing, travel booking.

## 10. Sliding Window Rate Limiter

```python
# PROBLEM: Limit requests with smooth sliding window, not fixed buckets
# TIME: O(log n) for n stored timestamps, SPACE: O(n) per client
from collections import deque
from time import time
from dataclasses import dataclass
import bisect


@dataclass
class RequestRecord:
    """Timestamp with optional metadata."""
    timestamp: float
    metadata: dict | None = None


class SlidingWindowRateLimiter:
    """
    Exact sliding window rate limiting.
    Counts requests in [now - window_size, now].
    """
    
    def __init__(self, max_requests: int, window_seconds: float):
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self._requests: dict[str, deque[RequestRecord]] = {}
    
    def _cleanup(self, client_id: str, now: float) -> None:
        """Remove expired timestamps from window."""
        if client_id not in self._requests:
            return
        
        cutoff = now - self.window_seconds
        requests = self._requests[client_id]
        
        while requests and requests[0].timestamp < cutoff:
            requests.popleft()
    
    def is_allowed(self, client_id: str) -> bool:
        """Check if request is allowed (does not consume quota)."""
        now = time()
        self._cleanup(client_id, now)
        
        count = len(self._requests.get(client_id, []))
        return count < self.max_requests
    
    def consume(self, client_id: str, metadata: dict | None = None) -> tuple[bool, int]:
        """
        Attempt to consume one request from quota.
        Returns (allowed, remaining_count).
        """
        now = time()
        self._cleanup(client_id, now)
        
        if client_id not in self._requests:
            self._requests[client_id] = deque()
        
        count = len(self._requests[client_id])
        
        if count >= self.max_requests:
            return (False, 0)
        
        self._requests[client_id].append(RequestRecord(now, metadata))
        return (True, self.max_requests - count - 1)
    
    def get_stats(self, client_id: str) -> dict:
        """Get current window statistics for client."""
        now = time()
        self._cleanup(client_id, now)
        
        requests = self._requests.get(client_id, [])
        count = len(requests)
        
        return {
            "current_count": count,
            "max_requests": self.max_requests,
            "remaining": max(0, self.max_requests - count),
            "window_seconds": self.window_seconds,
            "oldest_request": requests[0].timestamp if requests else None,
            "reset_at": requests[0].timestamp + self.window_seconds if requests else now
        }


class SlidingLogRateLimiter:
    """
    Memory-efficient sliding log using sorted list + binary search.
    Better for sparse traffic patterns.
    """
    
    def __init__(self, max_requests: int, window_seconds: float):
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self._logs: dict[str, list[float]] = {}
    
    def is_allowed(self, client_id: str) -> bool:
        """Check quota using binary search for efficiency."""
        now = time()
        cutoff = now - self.window_seconds
        
        if client_id not in self._logs:
            return True
        
        logs = self._logs[client_id]
        
        # Binary search for cutoff position
        idx = bisect.bisect_left(logs, cutoff)
        count = len(logs) - idx
        
        return count < self.max_requests
    
    def record_request(self, client_id: str) -> tuple[bool, int]:
        """Record request and return (allowed, remaining)."""
        now = time()
        cutoff = now - self.window_seconds
        
        if client_id not in self._logs:
            self._logs[client_id] = []
        
        logs = self._logs[client_id]
        
        # Prune old entries
        idx = bisect.bisect_left(logs, cutoff)
        if idx > 0:
            self._logs[client_id] = logs[idx:]
            logs = self._logs[client_id]
        
        # Check limit
        if len(logs) >= self.max_requests:
            return (False, 0)
        
        # Insert maintaining sorted order
        bisect.insort(logs, now)
        return (True, self.max_requests - len(logs) - 1)


class RateLimiterMiddleware:
    """
    Rate limiter as reusable middleware component.
    Integrates with web frameworks.
    """
    
    def __init__(
        self,
        max_requests: int = 100,
        window_seconds: float = 60.0,
        key_func: callable = None
    ):
        self.limiter = SlidingWindowRateLimiter(max_requests, window_seconds)
        self.key_func = key_func or (lambda req: req.get("client_id", "default"))
    
    def check(self, request: dict) -> tuple[bool, dict]:
        """
        Check rate limit for request.
        Returns (allowed, headers_dict).
        """
        client_id = self.key_func(request)
        allowed, remaining = self.limiter.consume(client_id)
        stats = self.limiter.get_stats(client_id)
        
        headers = {
            "X-RateLimit-Limit": str(self.limiter.max_requests),
            "X-RateLimit-Remaining": str(remaining),
            "X-RateLimit-Reset": str(int(stats.get("reset_at", 0))),
        }
        
        return (allowed, headers)
```

**Why**: Exact rate limiting without fixed bucket boundaries. Smoother than token bucket for API quotas. Binary search variant optimizes memory for sparse traffic.
