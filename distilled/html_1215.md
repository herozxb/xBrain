```html
        const form = document.getElementById('userForm');
        
        // Custom validation messages
        const validators = {
            username: (value) => {
                if (!value) return 'Username is required';
                if (value.length < 3) return 'Username must be at least 3 characters';
                if (!/^[a-zA-Z0-9_]+$/.test(value)) return 'Only letters, numbers, underscores allowed';
                return '';
            },
            email: (value) => {
                if (!value) return 'Email is required';
                if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) return 'Invalid email format';
                return '';
            },
            password: (value) => {
                if (!value) return 'Password is required';
                if (value.length < 8) return 'Password must be at least 8 characters';
                if (!/(?=.*\d)(?=.*[a-z])(?=.*[A-Z])/.test(value)) return 'Must contain uppercase, lowercase, and number';
                return '';
            },
            age: (value) => {
                if (!value) return 'Age is required';
                const num = parseInt(value);
                if (num < 18 || num > 120) return 'Age must be between 18 and 120';
                return '';
            },
            website: (value) => {
                if (!value) return ''; // Optional field
                try {
                    new URL(value);
                    return '';
                } catch {
                    return 'Invalid URL format';
                }
            }
        };

        // Real-time validation on input
        Object.keys(validators).forEach(fieldName => {
            const input = document.getElementById(fieldName);
            const errorSpan = document.getElementById(`${fieldName}Error`);
            
            input.addEventListener('input', () => {
                const error = validators[fieldName](input.value);
                errorSpan.textContent = error;
                input.setCustomValidity(error);
            });
        });

        // Form submission validation
        form.addEventListener('submit', (e) => {
            e.preventDefault();
            
            let isValid = true;
            Object.keys(validators).forEach(fieldName => {
                const input = document.getElementById(fieldName);
                const errorSpan = document.getElementById(`${fieldName}Error`);
                const error = validators[fieldName](input.value);
                
                errorSpan.textContent = error;
                input.setCustomValidity(error);
                
                if (error) isValid = false;
            });

            if (isValid) {
                console.log('Form submitted:', new FormData(form));
                alert('Form submitted successfully!');
            }
        });
    </script>
</body>
</html>
```

**Why**: Client-side validation with HTML5 attributes (`required`, `pattern`, `min`, `max`) and JavaScript provides immediate feedback, reduces server load, and improves UX while server-side validation remains essential for security.

## html_image_lazy_loading

```html
<!DOCTYPE html>
<html lang="en">
<!-- PROBLEM: Implement lazy loading for images to improve page performance -->
<!-- TIME: O(n) SPACE: O(n) where n = number of images -->
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Lazy Loading Images</title>
    <style>
        .image-container {
            margin: 2rem 0;
            min-height: 300px;
        }
        
        img {
            width: 100%;
            height: auto;
            display: block;
            background: #f0f0f0;
            transition: opacity 0.3s;
        }
        
        img[data-src] {
            opacity: 0;
        }
        
        img.loaded {
            opacity: 1;
        }
        
        /* Placeholder shimmer effect */
        .placeholder {
            background: linear-gradient(90deg, #f0f0f0 25%, #e0e0e0 50%, #f0f0f0 75%);
            background-size: 200% 100%;
            animation: shimmer 1.5s infinite;
        }
        
        @keyframes shimmer {
            0% { background-position: 200% 0; }
            100% { background-position: -200% 0; }
        }
    </style>
</head>
<body>
    <h1>Lazy Loading Demo</h1>
    
    <!-- Native lazy loading (modern browsers) -->
    <div class="image-container">
        <h3>Native Lazy Loading</h3>
        <img 
            loading="lazy" 
            src="https://picsum.photos/800/400?random=1" 
            alt="Lazy loaded image 1"
        >
    </div>

    <!-- Intersection Observer API (more control) -->
    <div class="image-container">
        <h3>Intersection Observer</h3>
        <img 
            data-src="https://picsum.photos/800/400?random=2" 
            alt="Lazy loaded image 2"
            class="placeholder"
        >
    </div>

    <div class="image-container">
        <img 
            data-src="https://picsum.photos/800/400?random=3" 
            alt="Lazy loaded image 3"
            class="placeholder"
        >
    </div>

    <div class="image-container">
        <img 
            data-src="https://picsum.photos/800/400?random=4" 
            alt="Lazy loaded image 4"
            class="placeholder"
        >
    </div>

    <!-- Responsive images with lazy loading -->
    <div class="image-container">
        <h3>Responsive + Lazy Loading</h3>
        <img 
            loading="lazy"
            src="https://picsum.photos/400/300?random=5"
            srcset="https://picsum.photos/400/300?random=5 400w,
                    https://picsum.photos/800/600?random=5 800w,
                    https://picsum.photos/1200/900?random=5 1200w"
            sizes="(max-width: 600px) 400px,
                   (max-width: 1200px) 800px,
                   1200px"
            alt="Responsive lazy image"
        >
    </div>

    <script>
        // Intersection Observer implementation
        class LazyLoader {
            constructor(options = {}) {
                this.rootMargin = options.rootMargin || '50px';
                this.threshold = options.threshold || 0.01;
                this.observer = null;
                this.init();
            }

            init() {
                // Check for native support first
                if ('loading' in HTMLImageElement.prototype) {
                    this.loadNative();
                    return;
                }

                // Fall back to Intersection Observer
                if ('IntersectionObserver' in window) {
                    this.observer = new IntersectionObserver(
                        this.handleIntersection.bind(this),
                        {
                            rootMargin: this.rootMargin,
                            threshold: this.threshold
                        }
                    );
                    this.observeImages();
                } else {
                    // Fallback: load all images
                    this.loadAllImages();
                }
            }

            handleIntersection(entries, observer) {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        this.loadImage(entry.target);
                        observer.unobserve(entry.target);
                    }
                });
            }

            loadImage(img) {
                const src = img.dataset.src;
                if (!src) return;

                img.src = src;
                img.onload = () => {
                    img.classList.remove('placeholder');
                    img.classList.add('loaded');
                };
                img.onerror = () => {
                    console.error('Failed to load:', src);
                    img.classList.remove('placeholder');
                };
            }

            observeImages() {
                const images = document.querySelectorAll('img[data-src]');
                images.forEach(img => this.observer.observe(img));
            }

            loadNative() {
                const images = document.querySelectorAll('img[data-src]');
                images.forEach(img => {
                    img.loading = 'lazy';
                    img.src = img.dataset.src;
                    img.removeAttribute('data-src');
                });
            }

            loadAllImages() {
                const images = document.querySelectorAll('img[data-src]');
                images.forEach(img => this.loadImage(img));
            }
        }

        // Initialize lazy loader
        const lazyLoader = new LazyLoader({
            rootMargin: '100px', // Start loading 100px before visible
            threshold: 0.01
        });

        // Utility: Preload critical images
        function preloadImage(src) {
            return new Promise((resolve, reject) => {
                const img = new Image();
                img.onload = resolve;
                img.onerror = reject;
                img.src = src;
            });
        }

        // Example: Preload above-the-fold images
        preloadImage('https://picsum.photos/800/400?random=1')
            .then(() => console.log('Critical image preloaded'));
    </script>
</body>
</html>
```

**Why**: Lazy loading defers offscreen image loading, reducing initial page weight, improving Time to Interactive (TTI), and saving bandwidth. Intersection Observer provides fine-grained control over when images start loading.

```html
                } else {
                    element[key] = value;
                }

                    element.appendChild(document.createTextNode(child));
```

```html
        const log = (message) => {
            const logDiv = document.getElementById('log');
            const entry = document.createElement('div');
            entry.textContent = `[${new Date().toLocaleTimeString()}] ${message}`;
            logDiv.appendChild(entry);
            logDiv.scrollTop = logDiv.scrollHeight;
        };

        // ===== EVENT DELEGATION IMPLEMENTATION =====

        class EventDelegator {
            constructor(rootElement) {
                this.root = rootElement;
                this.handlers = new Map();
                this.boundHandler = this.handleEvent.bind(this);
            }

            // Register handler for selector
            on(eventType, selector, handler, options = {}) {
                const key = `${eventType}:${selector}`;
                
                if (!this.handlers.has(key)) {
                    this.handlers.set(key, {
                        selector,
                        handlers: [],
                        options
                    });
                    
                    this.root.addEventListener(eventType, this.boundHandler, options);
                }
                
                this.handlers.get(key).handlers.push(handler);
                
                return () => this.off(eventType, selector, handler);
            }

            // Remove handler
            off(eventType, selector, handler) {
                const key = `${eventType}:${selector}`;
                const entry = this.handlers.get(key);
                
                if (entry) {
                    entry.handlers = entry.handlers.filter(h => h !== handler);
                    
                    if (entry.handlers.length === 0) {
                        this.handlers.delete(key);
                        this.root.removeEventListener(eventType, this.boundHandler);
                    }
                }
            }

            // Handle delegated events
            handleEvent(event) {
                const { type, target } = event;
                
                for (const [key, entry] of this.handlers) {
                    if (!key.startsWith(type + ':')) continue;
                    
                    // Find matching element up the tree
                    let element = target;
                    while (element && element !== this.root) {
                        if (element.matches && element.matches(entry.selector)) {
                            // Call all handlers for this selector
                            entry.handlers.forEach(handler => {
                                handler.call(element, event, element);
                            });
                            break;
                        }
                        element = element.parentElement;
                    }
                }
            }

            // One-time event
            once(eventType, selector, handler) {
                const wrapper = (event, element) => {
                    this.off(eventType, selector, wrapper);
                    handler.call(element, event, element);
                };
                return this.on(eventType, selector, wrapper);
            }

            // Trigger custom event
            emit(eventType, detail = {}) {
                const event = new CustomEvent(eventType, { 
                    bubbles: true, 
                    detail 
                });
                this.root.dispatchEvent(event);
            }
        }

                parts.push(event.key.toLowerCase());
```
