## Edmonds-Karp Algorithm

```python
# PROBLEM: Find maximum flow in a network using BFS-based Ford-Fulkerson
# TIME: O(V * E^2) SPACE: O(V^2)
from collections import deque

def edmonds_karp(capacity: list[list[int]], source: int, sink: int) -> int:
    """
    Edmonds-Karp algorithm for computing maximum flow in a flow network.
    Uses BFS to find augmenting paths, guaranteeing polynomial time complexity.
    
    Args:
        capacity: Adjacency matrix where capacity[u][v] is edge capacity
        source: Source vertex index
        sink: Sink vertex index
    
    Returns:
        Maximum flow value from source to sink
    """
    n = len(capacity)
    # Residual capacity matrix (copy of original capacities)
    residual = [row[:] for row in capacity]
    max_flow = 0
    
    def bfs_find_path() -> tuple[list[int], int]:
        """Find augmenting path using BFS, return path and bottleneck capacity."""
        parent = [-1] * n
        parent[source] = source
        min_capacity = [float('inf')] * n
        queue = deque([source])
        
        while queue:
            u = queue.popleft()
            for v in range(n):
                # Check if edge has residual capacity and v is unvisited
                if parent[v] == -1 and residual[u][v] > 0:
                    parent[v] = u
                    min_capacity[v] = min(min_capacity[u], residual[u][v])
                    if v == sink:
                        # Reconstruct path
                        path = []
                        curr = sink
                        while curr != source:
                            path.append(curr)
                            curr = parent[curr]
                        path.append(source)
                        return path[::-1], min_capacity[sink]
                    queue.append(v)
        return [], 0
    
    # Main loop: keep finding augmenting paths
    while True:
        path, flow = bfs_find_path()
        if flow == 0:
            break  # No more augmenting paths
        
        # Update residual capacities along the path
        for i in range(len(path) - 1):
            u, v = path[i], path[i + 1]
            residual[u][v] -= flow  # Forward edge decreases
            residual[v][u] += flow  # Backward edge increases
        
        max_flow += flow
    
    return max_flow
```
**Why**: Fundamental max-flow algorithm with guaranteed polynomial time; BFS ensures shortest augmenting paths, preventing infinite loops on irrational capacities.

## Dinic's Algorithm

```python
# PROBLEM: Find maximum flow efficiently using level graphs and blocking flows
# TIME: O(V^2 * E) or O(E * sqrt(V)) for bipartite SPACE: O(V + E)
from collections import deque

class Dinic:
    """
    Dinic's algorithm for maximum flow with level graph optimization.
    Uses BFS for level assignment and DFS with current edge optimization.
    """
    
    def __init__(self, n: int):
        self.n = n
        self.adj = [[] for _ in range(n)]  # Edge list per vertex
        # Each edge: [to, capacity, reverse_edge_index]
    
    def add_edge(self, fr: int, to: int, cap: int) -> None:
        """Add directed edge with given capacity."""
        forward = [to, cap, len(self.adj[to])]
        backward = [fr, 0, len(self.adj[fr])]
        self.adj[fr].append(forward)
        self.adj[to].append(backward)
    
    def _bfs_level(self, source: int, sink: int, level: list[int]) -> bool:
        """Build level graph using BFS. Returns True if sink is reachable."""
        level[:] = [-1] * self.n
        level[source] = 0
        queue = deque([source])
        
        while queue:
            u = queue.popleft()
            for to, cap, _ in self.adj[u]:
                if level[to] == -1 and cap > 0:
                    level[to] = level[u] + 1
                    queue.append(to)
        
        return level[sink] != -1
    
    def _dfs_flow(self, u: int, sink: int, flow: int, 
                  level: list[int], ptr: list[int]) -> int:
        """Send flow using DFS with current edge pointer optimization."""
        if u == sink:
            return flow
        
        for i in range(ptr[u], len(self.adj[u])):
            ptr[u] = i  # Update current edge pointer
            to, cap, rev = self.adj[u][i]
            
            # Only traverse edges that go to next level
            if level[to] == level[u] + 1 and cap > 0:
                pushed = self._dfs_flow(to, sink, min(flow, cap), level, ptr)
                if pushed > 0:
                    # Update residual capacities
                    self.adj[u][i][1] -= pushed
                    self.adj[to][rev][1] += pushed
                    return pushed
        
        return 0
    
    def max_flow(self, source: int, sink: int) -> int:
        """Compute maximum flow from source to sink."""
        total = 0
        level = [-1] * self.n
        
        # Phase 1: Build level graph
        while self._bfs_level(source, sink, level):
            ptr = [0] * self.n  # Reset current edge pointers
            
            # Phase 2: Find blocking flow
            while True:
                pushed = self._dfs_flow(source, sink, float('inf'), level, ptr)
                if pushed == 0:
                    break
                total += pushed
            
            level = [-1] * self.n  # Reset for next iteration
        
        return total
```
**Why**: Fastest practical max-flow algorithm; level graph prevents cycles, current-edge optimization avoids revisiting dead ends.

## Z-Algorithm

```python
# PROBLEM: Compute Z-array for pattern matching and string analysis
# TIME: O(n) SPACE: O(n)
def z_algorithm(s: str) -> list[int]:
    """
    Compute Z-array where Z[i] = longest substring starting at i that matches prefix.
    
    Z-array properties:
    - Z[0] is typically undefined (or len(s) by convention)
    - Z[i] = longest common prefix of s and s[i:]
    
    Applications: pattern matching (concatenate pattern + '$' + text),
    finding periods, palindrome detection, string compression.
    """
    n = len(s)
    z = [0] * n
    l, r = 0, 0  # Current rightmost Z-box [l, r)
    
    for i in range(1, n):
        # Case 1: i is outside current Z-box
        if i >= r:
            l = r = i
            while r < n and s[r - l] == s[r]:
                r += 1
            z[i] = r - l
            r -= 1
        else:
            # Case 2: i is inside Z-box
            k = i - l  # Mirror position relative to l
            if z[k] < r - i:
                # Z[k] doesn't reach Z-box boundary
                z[i] = z[k]
            else:
                # Z[k] reaches or exceeds boundary, need to extend
                l = i
                while r < n and s[r - l] == s[r]:
                    r += 1
                z[i] = r - l
                r -= 1
    
    z[0] = n  # Convention: Z[0] = full string length
    return z


def z_search(text: str, pattern: str) -> list[int]:
    """Find all occurrences of pattern in text using Z-algorithm."""
    combined = pattern + '$' + text
    z = z_algorithm(combined)
    matches = []
    m = len(pattern)
    
    for i in range(m + 1, len(combined)):
        if z[i] == m:
            matches.append(i - m - 1)
    
    return matches
```
**Why**: Linear-time string matching without preprocessing pattern; Z-box trick reuses previous computations for efficiency.

## Huffman Coding

```python
# PROBLEM: Construct optimal prefix-free codes for lossless data compression
# TIME: O(n log n) SPACE: O(n)
import heapq
from dataclasses import dataclass
from typing import Optional

@dataclass
class HuffmanNode:
    """Node in Huffman tree."""
    freq: int
    char: Optional[str] = None
    left: Optional['HuffmanNode'] = None
    right: Optional['HuffmanNode'] = None
    
    # For heap comparison (nodes with same frequency)
    def __lt__(self, other: 'HuffmanNode') -> bool:
        return self.freq < other.freq


def build_huffman_tree(freq_map: dict[str, int]) -> HuffmanNode:
    """
    Build Huffman tree from character frequency map.
    
    Args:
        freq_map: Dictionary mapping characters to their frequencies
    
    Returns:
        Root of Huffman tree
    """
    # Create leaf nodes for each character
    heap = [HuffmanNode(freq=f, char=c) for c, f in freq_map.items()]
    heapq.heapify(heap)
    
    # Build tree by combining lowest frequency nodes
    while len(heap) > 1:
        left = heapq.heappop(heap)
        right = heapq.heappop(heap)
        merged = HuffmanNode(
            freq=left.freq + right.freq,
            left=left,
            right=right
        )
        heapq.heappush(heap, merged)
    
    return heap[0]


def generate_codes(node: HuffmanNode, prefix: str = "", 
                   codes: dict[str, str] = None) -> dict[str, str]:
    """Generate Huffman codes by traversing tree."""
    if codes is None:
        codes = {}
    
    if node.char is not None:
        # Leaf node: store code for character
        codes[node.char] = prefix or "0"  # Handle single character case
    else:
        # Internal node: recurse with extended prefix
        if node.left:
            generate_codes(node.left, prefix + "0", codes)
        if node.right:
            generate_codes(node.right, prefix + "1", codes)
    
    return codes


def huffman_encode(text: str) -> tuple[str, dict[str, str]]:
    """Encode text using Huffman coding."""
    if not text:
        return "", {}
    
    # Count frequencies
    freq = {}
    for c in text:
        freq[c] = freq.get(c, 0) + 1
    
    # Build tree and generate codes
    root = build_huffman_tree(freq)
    codes = generate_codes(root)
    
    # Encode text
    encoded = "".join(codes[c] for c in text)
    return encoded, codes


def huffman_decode(encoded: str, root: HuffmanNode) -> str:
    """Decode Huffman-encoded text using tree."""
    if not encoded:
        return ""
    
    result = []
    node = root
    
    for bit in encoded:
        node = node.left if bit == '0' else node.right
        
        if node.char is not None:
            result.append(node.char)
            node = root
    
    return "".join(result)
```
**Why**: Optimal prefix-free coding with provable compression ratio; heap ensures lowest-frequency symbols get longest codes.

## Leaky Bucket Rate Limiter

```python
# PROBLEM: Rate limiting with constant processing rate and burst capacity
# TIME: O(1) per request SPACE: O(1)
import time
from threading import Lock

class LeakyBucket:
    """
    Leaky bucket rate limiter implementation.
    
    Water analogy:
    - Requests add water to bucket
    - Bucket leaks at constant rate
    - Bucket overflows if requests exceed leak rate
    
    Properties:
    - Smooths out burst traffic
    - Guarantees constant output rate
    - Provides strict rate enforcement
    """
    
    def __init__(self, capacity: int, leak_rate: float):
        """
        Args:
            capacity: Maximum bucket size (burst tolerance)
            leak_rate: Requests processed per second
        """
        self.capacity = capacity
        self.leak_rate = leak_rate  # requests per second
        self.water = 0.0  # Current water level
        self.last_leak = time.time()
        self.lock = Lock()
    
    def _leak(self) -> None:
        """Remove water that has leaked since last check."""
        now = time.time()
        elapsed = now - self.last_leak
        leaked = elapsed * self.leak_rate
        self.water = max(0.0, self.water - leaked)
        self.last_leak = now
    
    def allow_request(self) -> bool:
        """
        Try to add a request to the bucket.
        
        Returns:
            True if request is allowed, False if bucket would overflow
        """
        with self.lock:
            self._leak()
            
            if self.water < self.capacity:
                self.water += 1
                return True
            return False
    
    def get_wait_time(self) -> float:
        """Calculate time until next request can be allowed."""
        with self.lock:
            self._leak()
            if self.water < self.capacity:
                return 0.0
            # Time for bucket to drain one unit
            return 1.0 / self.leak_rate


class LeakyBucketQueue:
    """
    Leaky bucket that queues requests instead of rejecting them.
    Useful when you want to process all requests at a controlled rate.
    """
    
    def __init__(self, capacity: int, leak_rate: float):
        self.capacity = capacity
        self.leak_rate = leak_rate
        self.queue = []
        self.last_process = time.time()
        self.lock = Lock()
    
    def add_request(self, request) -> bool:
        """Add request to queue if space available."""
        with self.lock:
            if len(self.queue) < self.capacity:
                self.queue.append(request)
                return True
            return False
    
    def process_ready(self) -> list:
        """Get all requests ready for processing based on leak rate."""
        with self.lock:
            now = time.time()
            elapsed = now - self.last_process
            can_process = int(elapsed * self.leak_rate)
            
            ready = []
            while can_process > 0 and self.queue:
                ready.append(self.queue.pop(0))
                can_process -= 1
            
            self.last_process = now
            return ready
```
**Why**: Provides smooth traffic shaping; ideal for APIs and network traffic where constant output rate matters more than exact request timing.

## Treap (Tree + Heap)

```python
# PROBLEM: Self-balancing BST with probabilistic guarantees using heap priorities
# TIME: O(log n) expected for all operations SPACE: O(n)
import random
from typing import Optional, Generic, TypeVar

T = TypeVar('T')

class TreapNode(Generic[T]):
    """Node in treap with key (BST property) and priority (heap property)."""
    def __init__(self, key: T):
        self.key = key
        self.priority = random.random()  # Random heap priority
        self.left: Optional['TreapNode[T]'] = None
        self.right: Optional['TreapNode[T]'] = None
        self.size = 1  # Subtree size for order statistics
    
    def update_size(self):
        """Recalculate subtree size."""
        self.size = 1
        if self.left:
            self.size += self.left.size
        if self.right:
            self.size += self.right.size


class Treap(Generic[T]):
    """
    Treap: Randomized BST combining BST and heap properties.
    
    - BST property: left.key < key < right.key
    - Heap property: priority > left.priority and priority > right.priority
    
    Expected O(log n) operations due to random priorities preventing
    adversarial worst cases. No rebalancing logic needed!
    """
    
    def __init__(self):
        self.root: Optional[TreapNode[T]] = None
    
    def _rotate_right(self, y: TreapNode[T]) -> TreapNode[T]:
        """Right rotation for maintaining heap property."""
        x = y.left
        y.left = x.right
        x.right = y
        y.update_size()
        x.update_size()
        return x
    
    def _rotate_left(self, x: TreapNode[T]) -> TreapNode[T]:
        """Left rotation for maintaining heap property."""
        y = x.right
        x.right = y.left
        y.left = x
        x.update_size()
        y.update_size()
        return y
    
    def insert(self, key: T) -> None:
        """Insert key into treap."""
        self.root = self._insert(self.root, key)
    
    def _insert(self, node: Optional[TreapNode[T]], key: T) -> TreapNode[T]:
        if node is None:
            return TreapNode(key)
        
        if key < node.key:
            node.left = self._insert(node.left, key)
            # Fix heap property violation
            if node.left.priority > node.priority:
                node = self._rotate_right(node)
        elif key > node.key:
            node.right = self._insert(node.right, key)
            # Fix heap property violation
            if node.right.priority > node.priority:
                node = self._rotate_left(node)
        # Duplicate keys ignored
        
        node.update_size()
        return node
    
    def delete(self, key: T) -> None:
        """Delete key from treap."""
        self.root = self._delete(self.root, key)
    
    def _delete(self, node: Optional[TreapNode[T]], key: T) -> Optional[TreapNode[T]]:
        if node is None:
            return None
        
        if key < node.key:
            node.left = self._delete(node.left, key)
        elif key > node.key:
            node.right = self._delete(node.right, key)
        else:
            # Found node to delete
            if node.left is None:
                return node.right
            if node.right is None:
                return node.left
            
            # Rotate with higher priority child and continue deletion
            if node.left.priority > node.right.priority:
                node = self._rotate_right(node)
                node.right = self._delete(node.right, key)
            else:
                node = self._rotate_left(node)
                node.left = self._delete(node.left, key)
        
        if node:
            node.update_size()
        return node
    
    def search(self, key: T) -> bool:
        """Check if key exists in treap."""
        return self._search(self.root, key)
    
    def _search(self, node: Optional[TreapNode[T]], key: T) -> bool:
        while node:
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                return True
        return False
    
    def kth_smallest(self, k: int) -> Optional[T]:
        """Find kth smallest element (1-indexed)."""
        return self._kth(self.root, k)
    
    def _kth(self, node: Optional[TreapNode[T]], k: int) -> Optional[T]:
        if node is None:
            return None
        
        left_size = node.left.size if node.left else 0
        
        if k <= left_size:
            return self._kth(node.left, k)
        elif k == left_size + 1:
            return node.key
        else:
            return self._kth(node.right, k - left_size - 1)
```
**Why**: Simpler than AVL/Red-Black trees; random priorities provide probabilistic balance without explicit rebalancing rules.
