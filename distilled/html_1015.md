# HTML/JavaScript Functions - DOM Navigation & Class Manipulation

## firstElementChild

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>firstElementChild Demo</title>
</head>
<body>
    <!-- PROBLEM: Get the first ELEMENT child (ignoring text nodes) -->
    <!-- TIME: O(1) SPACE: O(1) -->

    <div id="container">
        Text node here
        <p>First paragraph</p>
        <span>Second element</span>
    </div>

    <script>
        /**
         * firstElementChild returns the first child ELEMENT node
         * Ignores text, comment, and other non-element nodes
         * Read-only property, returns null if no element children
         */
        
        const container = document.getElementById('container');
        
        // Get first element child (skips text nodes)
        const firstElement = container.firstElementChild;
        console.log(firstElement); // <p>First paragraph</p>
        
        // Compare with firstChild (includes text nodes)
        const firstNode = container.firstChild;
        console.log(firstNode); // Text node "Text node here"
        
        // Safe navigation example
        function getFirstElementContent(element) {
            const first = element.firstElementChild;
            return first ? first.textContent : 'No child elements';
        }
        
        console.log(getFirstElementContent(container)); // "First paragraph"
    </script>
</body>
</html>
```

**Why**: Use `firstElementChild` when you need to traverse only element nodes, avoiding whitespace text nodes that can complicate DOM navigation. More reliable than `firstChild` for element traversal.

## nextElementSibling

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>nextElementSibling Demo</title>
</head>
<body>
    <!-- PROBLEM: Get the next sibling ELEMENT (ignoring text nodes) -->
    <!-- TIME: O(1) SPACE: O(1) -->

    <div id="items">
        <div class="item" data-index="1">Item 1</div>
        Text between items
        <div class="item" data-index="2">Item 2</div>
        <div class="item" data-index="3">Item 3</div>
    </div>

    <script>
        /**
         * nextElementSibling returns the next ELEMENT sibling
         * Skips text, comment, and other non-element nodes
         * Read-only, returns null if no next element sibling
         */
        
        const firstItem = document.querySelector('.item[data-index="1"]');
        
        // Navigate to next element sibling
        const secondItem = firstItem.nextElementSibling;
        console.log(secondItem.dataset.index); // "2"
        
        // Traverse all element siblings
        function collectSiblings(element) {
            const siblings = [];
            let current = element.nextElementSibling;
            
            while (current) {
                siblings.push(current);
                current = current.nextElementSibling;
            }
            
            return siblings;
        }
        
        const nextSiblings = collectSiblings(firstItem);
        console.log(nextSiblings.length); // 2
        
        // Navigate through form fields
        function focusNextField(currentField) {
            const next = currentField.nextElementSibling;
            if (next && next.tagName === 'INPUT') {
                next.focus();
            }
        }
    </script>
</body>
</html>
```

**Why**: `nextElementSibling` enables forward traversal through DOM elements without intermediate text nodes. Essential for navigating lists, forms, tab interfaces, and implementing keyboard navigation.

## previousElementSibling

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>previousElementSibling Demo</title>
</head>
<body>
    <!-- PROBLEM: Get the previous sibling ELEMENT (ignoring text nodes) -->
    <!-- TIME: O(1) SPACE: O(1) -->

    <nav id="breadcrumb">
        <a href="/">Home</a>
        <span> / </span>
        <a href="/products">Products</a>
        <span> / </span>
        <a href="/products/laptop" class="current">Laptop</a>
    </nav>

    <script>
        /**
         * previousElementSibling returns the previous ELEMENT sibling
         * Skips text, comment, and other non-element nodes
         * Read-only, returns null if no previous element sibling
         */
        
        const current = document.querySelector('.current');
        
        // Navigate backwards through breadcrumb
        function getBreadcrumbPath(currentElement) {
            const path = [currentElement.textContent];
            let prev = currentElement.previousElementSibling;
            
            while (prev) {
                // Skip separator spans
                if (prev.tagName === 'A') {
                    path.unshift(prev.textContent);
                }
                prev = prev.previousElementSibling;
            }
            
            return path.join(' > ');
        }
        
        console.log(getBreadcrumbPath(current)); // "Home > Products > Laptop"
        
        // Move focus to previous form field
        function focusPreviousField(currentField) {
            const prev = currentField.previousElementSibling;
            if (prev && prev.tagName === 'INPUT') {
                prev.focus();
            }
        }
        
        // Check if element has previous sibling
        function isFirstElement(element) {
            return element.previousElementSibling === null;
        }
    </script>
</body>
</html>
```

**Why**: `previousElementSibling` allows backward navigation through elements, perfect for breadcrumb trails, undo operations, backward tab navigation, and checking if an element is first in a container.

## children

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>children Demo</title>
</head>
<body>
    <!-- PROBLEM: Get all child ELEMENTS as HTMLCollection -->
    <!-- TIME: O(1) SPACE: O(1) -->

    <div id="gallery">
        Text ignored
        <img src="photo1.jpg" alt="Photo 1">
        <img src="photo2.jpg" alt="Photo 2">
        Comment ignored
        <img src="photo3.jpg" alt="Photo 3">
    </div>

    <script>
        /**
         * children returns an HTMLCollection of all child ELEMENTS
         * Excludes text, comment, and other non-element nodes
         * Live collection - updates automatically when DOM changes
         * Read-only property
         */
        
        const gallery = document.getElementById('gallery');
        
        // Get all child elements (images only, no text/comment)
        const images = gallery.children;
        console.log(images.length); // 3
        
        // Convert to array for array methods
        const imageArray = Array.from(gallery.children);
        const altTexts = imageArray.map(img => img.alt);
        console.log(altTexts); // ["Photo 1", "Photo 2", "Photo 3"]
        
        // Iterate through children
        function applyToChildren(parent, callback) {
            const children = parent.children;
            for (let i = 0; i < children.length; i++) {
                callback(children[i], i);
            }
        }
        
        applyToChildren(gallery, (img, index) => {
            img.style.opacity = index === 0 ? 1 : 0.5;
        });
        
        // Check if element has children
        function hasChildren(element) {
            return element.children.length > 0;
        }
        
        // Get child at specific index
        function getChildAt(parent, index) {
            return parent.children[index] || null;
        }
        
        console.log(getChildAt(gallery, 1).alt); // "Photo 2"
    </script>
</body>
</html>
```

**Why**: `children` provides a clean collection of only element nodes, unlike `childNodes` which includes all node types. Live collection automatically reflects DOM changes, making it ideal for dynamic content manipulation.

## parentElement

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>parentElement Demo</title>
</head>
<body>
    <!-- PROBLEM: Get the parent ELEMENT node -->
    <!-- TIME: O(1) SPACE: O(1) -->

    <div id="outer">
        <div id="middle">
            <span id="inner">Click me</span>
        </div>
    </div>

    <script>
        /**
         * parentElement returns the parent ELEMENT node
         * Returns null if parent is not an element or node has no parent
         * Read-only property
         * Difference from parentNode: parentElement returns null for document.documentElement
         */
        
        const inner = document.getElementById('inner');
        
        // Get direct parent
        const parent = inner.parentElement;
        console.log(parent.id); // "middle"
        
        // Traverse up the DOM tree
        function getAncestors(element) {
            const ancestors = [];
            let current = element.parentElement;
            
            while (current) {
                ancestors.push(current);
                current = current.parentElement;
            }
            
            return ancestors;
        }
        
        const ancestors = getAncestors(inner);
        console.log(ancestors.map(el => el.id)); // ["middle", "outer", ""]
        
        // Find closest ancestor matching selector
        function findAncestor(element, selector) {
            let current = element.parentElement;
            
            while (current) {
                if (current.matches(selector)) {
                    return current;
                }
                current = current.parentElement;
            }
            
            return null;
        }
        
        // Alternative: use built-in closest()
        const outer = inner.closest('#outer');
        console.log(outer.id); // "outer"
        
        // Check depth in DOM tree
        function getDepth(element) {
            let depth = 0;
            let current = element.parentElement;
            
            while (current) {
                depth++;
                current = current.parentElement;
            }
            
            return depth;
        }
        
        console.log(getDepth(inner)); // Depth from inner to html
    </script>
</body>
</html>
```

**Why**: `parentElement` provides safe upward traversal by returning only element nodes. Use for bubbling up event handlers, finding containers, calculating DOM depth, or removing elements from their parents.

## classList.toggle

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>classList.toggle Demo</title>
    <style>
        .active { background: green; color: white; }
        .hidden { display: none; }
    </style>
</head>
<body>
    <!-- PROBLEM: Toggle a class on/off -->
    <!-- TIME: O(1) SPACE: O(1) -->

    <button id="toggleBtn">Toggle Active</button>
    <div id="panel" class="hidden">Panel Content</div>

    <script>
        /**
         * classList.toggle(className) adds class if absent, removes if present
         * classList.toggle(className, force) uses boolean to add/remove
         * Returns true if class added, false if removed
         */
        
        const btn = document.getElementById('toggleBtn');
        const panel = document.getElementById('panel');
        
        // Simple toggle
        btn.addEventListener('click', () => {
            const isActive = btn.classList.toggle('active');
            console.log(isActive ? 'Activated' : 'Deactivated');
        });
        
        // Toggle panel visibility
        function togglePanel() {
            panel.classList.toggle('hidden');
        }
        
        // Force toggle with second parameter
        function setPanelVisibility(show) {
            // true = add class, false = remove class
            panel.classList.toggle('hidden', !show);
        }
        
        // Toggle with animation
        function toggleWithAnimation(element, className) {
            element.style.transition = 'all 0.3s';
            return element.classList.toggle(className);
        }
        
        // Toggle multiple states
        const states = ['idle', 'loading', 'success', 'error'];
        let currentStateIndex = 0;
        
        function cycleState(element) {
            // Remove current state
            element.classList.remove(states[currentStateIndex]);
            
            // Move to next state
            currentStateIndex = (currentStateIndex + 1) % states.length;
            
            // Add new state
            element.classList.add(states[currentStateIndex]);
        }
        
        // Conditional toggle based on viewport
        function toggleMobileMenu() {
            const menu = document.getElementById('mobileMenu');
            const shouldOpen = !menu.classList.contains('open');
            
            menu.classList.toggle('open', shouldOpen);
            document.body.classList.toggle('menu-open', shouldOpen);
        }
        
        // Example: Toggle all checkboxes
        document.getElementById('selectAll')?.addEventListener('change', (e) => {
            const checkboxes = document.querySelectorAll('.item-checkbox');
            checkboxes.forEach(cb => {
                cb.classList.toggle('checked', e.target.checked);
            });
        });
    </script>
</body>
</html>
```

**Why**: `classList.toggle` simplifies class state management by handling add/remove logic automatically. Perfect for UI states (open/closed, active/inactive), theme switching, and interactive elements without manual class checking.

## classList.contains

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>classList.contains Demo</title>
    <style>
        .selected { background: #007bff; color: white; }
        .disabled { opacity: 0.5; pointer-events: none; }
    </style>
</head>
<body>
    <!-- PROBLEM: Check if element has a specific class -->
    <!-- TIME: O(1) SPACE: O(1) -->

    <ul id="itemList">
        <li class="item selected">Item 1</li>
        <li class="item">Item 2</li>
        <li class="item disabled">Item 3</li>
    </ul>

    <script>
        /**
         * classList.contains(className) checks if class exists
         * Returns true if class is present, false otherwise
         * Case-sensitive class name matching
         */
        
        const items = document.querySelectorAll('.item');
        
        // Check if item is selected
        items.forEach(item => {
            if (item.classList.contains('selected')) {
                console.log('Found selected item:', item.textContent);
            }
        });
        
        // Conditional action based on class
        function toggleItem(element) {
            // Don't allow toggle on disabled items
            if (element.classList.contains('disabled')) {
                console.log('Item is disabled');
                return;
            }
            
            // Toggle selection
            if (element.classList.contains('selected')) {
                element.classList.remove('selected');
            } else {
                element.classList.add('selected');
            }
        }
        
        // Validate form field state
        function validateField(field) {
            const isValid = field.classList.contains('valid');
            const hasError = field.classList.contains('error');
            
            if (hasError) {
                showError(field);
            } else if (isValid) {
                showSuccess(field);
            }
            
            return isValid && !hasError;
        }
        
        // Filter elements by class
        function getElementsByClass(parent, className) {
            const elements = parent.children;
            const filtered = [];
            
            for (let i = 0; i < elements.length; i++) {
                if (elements[i].classList.contains(className)) {
                    filtered.push(elements[i]);
                }
            }
            
            return filtered;
        }
        
        // Check multiple classes
        function hasAllClasses(element, classNames) {
            return classNames.every(name => element.classList.contains(name));
        }
        
        function hasAnyClass(element, classNames) {
            return classNames.some(name => element.classList.contains(name));
        }
        
        // Example usage
        const firstItem = document.querySelector('.item');
        console.log(hasAnyClass(firstItem, ['selected', 'disabled'])); // true
        console.log(hasAllClasses(firstItem, ['selected', 'active'])); // false
        
        // State machine pattern
        function getCurrentState(element) {
            const states = ['idle', 'loading', 'success', 'error'];
            
            for (const state of states) {
                if (element.classList.contains(state)) {
                    return state;
                }
            }
            
            return null;
        }
    </script>
</body>
</html>
```

**Why**: `classList.contains` provides clean boolean checks for class presence. Essential for conditional logic, state validation, filtering elements, and implementing state machines without regex or string manipulation.

## classList.replace

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>classList.replace Demo</title>
    <style>
        .theme-light { background: white; color: black; }
        .theme-dark { background: #222; color: white; }
        .size-small { font-size: 12px; }
        .size-large { font-size: 24px; }
    </style>
</head>
<body>
    <!-- PROBLEM: Replace one class with another -->
    <!-- TIME: O(1) SPACE: O(1) -->

    <div id="content" class="theme-light size-small">
        Content area
    </div>

    <button id="themeToggle">Switch Theme</button>
    <button id="sizeToggle">Switch Size</button>

    <script>
        /**
         * classList.replace(oldClass, newClass) swaps one class for another
         * Returns true if replacement succeeded, false if oldClass wasn't found
         * Does nothing if oldClass doesn't exist
         */
        
        const content = document.getElementById('content');
        
        // Theme switching
        document.getElementById('themeToggle').addEventListener('click', () => {
            const isLight = content.classList.contains('theme-light');
            const isDark = content.classList.contains('theme-dark');
            
            if (isLight) {
                content.classList.replace('theme-light', 'theme-dark');
            } else if (isDark) {
                content.classList.replace('theme-dark', 'theme-light');
            }
        });
        
        // Size switching with fallback
        document.getElementById('sizeToggle').addEventListener('click', () => {
            const replaced = content.classList.replace('size-small', 'size-large');
            
            // If small wasn't found, try opposite
            if (!replaced) {
                content.classList.replace('size-large', 'size-small');
            }
        });
        
        // Safe replace with add fallback
        function safeReplace(element, oldClass, newClass) {
            const replaced = element.classList.replace(oldClass, newClass);
            
            // If old class didn't exist, just add new class
            if (!replaced) {
                element.classList.add(newClass);
            }
            
            return true;
        }
        
        // State transition pattern
        const states = ['idle', 'loading', 'success', 'error'];
        let currentStateIndex = 0;
        
        function transitionState(element, nextState) {
            const current = states[currentStateIndex];
            const replaced = element.classList.replace(current, nextState);
            
            if (replaced) {
                currentStateIndex = states.indexOf(nextState);
            }
            
            return replaced;
        }
        
        // Batch replace multiple classes
        function replaceClasses(element, replacements) {
            // replacements = { 'old-class': 'new-class', ... }
            const results = {};
            
            for (const [oldClass, newClass] of Object.entries(replacements)) {
                results[oldClass] = element.classList.replace(oldClass, newClass);
            }
            
            return results;
        }
        
        // Example: Switch breakpoint classes
        function updateResponsiveClasses(element, breakpoint) {
            const classMap = {
                'mobile': 'desktop',
                'tablet': 'desktop',
                'desktop': 'mobile'
            };
            
            for (const [oldClass, newClass] of Object.entries(classMap)) {
                if (element.classList.contains(oldClass)) {
                    element.classList.replace(oldClass, newClass);
                    break;
                }
            }
        }
        
        // Animation state replacement
        function animateTransition(element) {
            element.classList.replace('fade-in', 'fade-out');
            
            setTimeout(() => {
                element.classList.replace('fade-out', 'hidden');
            }, 300);
        }
    </script>
</body>
</html>
```

**Why**: `classList.replace` atomically swaps one class for another in a single operation, avoiding the add/remove dance. Perfect for theme switching, state transitions, responsive design changes, and animation states.

## className

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>className Demo</title>
    <style>
        .box { padding: 20px; border: 1px solid #ccc; }
        .highlight { background: yellow; }
        .important { font-weight: bold; }
    </style>
</head>
<body>
    <!-- PROBLEM: Get or set all classes as a string -->
    <!-- TIME: O(1) SPACE: O(1) -->

    <div id="target" class="box highlight">
        Target Element
    </div>

    <script>
        /**
         * className gets/sets the entire class attribute as a string
         * Getter: returns space-separated class names
         * Setter: completely replaces all classes
         * Useful for complete class replacement or string manipulation
         */
        
        const element = document.getElementById('target');
        
        // GET: Read all classes
        console.log(element.className); // "box highlight"
        
        // SET: Replace all classes
        element.className = 'new-class another-class';
        console.log(element.className); // "new-class another-class"
        
        // Clear all classes
        function clearAllClasses(element) {
            element.className = '';
        }
        
        // Copy classes from one element to another
        function copyClasses(source, target) {
            target.className = source.className;
        }
        
        // Append class to existing (safer: use classList.add)
        function addClassWithString(element, newClass) {
            if (element.className) {
                element.className += ' ' + newClass;
            } else {
                element.className = newClass;
            }
        }
        
        // Check if has class (simpler: use classList.contains)
        function hasClass(element, className) {
            return (' ' + element.className + ' ').includes(' ' + className + ' ');
        }
        
        // Remove class (simpler: use classList.remove)
        function removeClassWithString(element, className) {
            const classes = element.className.split(' ');
            const filtered = classes.filter(c => c !== className);
            element.className = filtered.join(' ');
        }
        
        // Template-based class assignment
        function setClassesFromTemplate(element, template) {
            // template = 'button {{size}} {{theme}}'
            const classes = template
                .replace('{{size}}', 'large')
                .replace('{{theme}}', 'dark');
            
            element.className = classes;
        }
        
        // Clone element with modified classes
        function cloneWithClasses(element, newClasses) {
            const clone = element.cloneNode(true);
            clone.className = newClasses;
            return clone;
        }
        
        // Swap classes between elements
        function swapClasses(elem1, elem2) {
            const temp = elem1.className;
            elem1.className = elem2.className;
            elem2.className = temp;
        }
        
        // Conditional class replacement
        function setConditionalClass(element, condition, trueClass, falseClass) {
            element.className = condition ? trueClass : falseClass;
        }
        
        // Example: Reset to base classes
        function resetToBase(element, baseClasses) {
            element.className = baseClasses;
        }
        
        resetToBase(element, 'box');
        console.log(element.className); // "box"
        
        // WARNING: className overwrites all existing classes!
        // Prefer classList methods for incremental changes:
        // - classList.add() to add without removing
        // - classList.remove() to remove specific class
        // - classList.toggle() to switch class
        // Use className only for complete replacement
    </script>
</body>
</html>
```

**Why**: `className` provides string-based access to all classes at once. Use for complete class replacement, copying classes between elements, template-based class generation, or when you need the entire class attribute as a string. Prefer `classList` methods for incremental class manipulation.

## Summary

These 10 functions cover essential DOM navigation and class manipulation:

**Element Navigation (6 functions):**
- `firstElementChild` / `lastElementChild` - Navigate to first/last child elements
- `nextElementSibling` / `previousElementSibling` - Navigate between siblings
- `children` - Get all child elements as HTMLCollection
- `parentElement` - Navigate up the DOM tree

**Class Manipulation (4 functions):**
- `classList.toggle` - Switch class on/off
- `classList.contains` - Check class existence
- `classList.replace` - Swap one class for another
- `className` - Get/set all classes as string

All methods ignore text/comment nodes, making them more reliable than their non-Element counterparts for practical DOM manipulation.
