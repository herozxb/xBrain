## 1. Randomized Binary Search Tree

```python
# PROBLEM: Implement a binary search tree with randomized insertion to ensure balanced structure
# TIME: O(log n) average for insert/delete/search SPACE: O(n)
import random

class RandomizedBST:
    def __init__(self, val=None):
        self.val = val
        self.left = None
        self.right = None
        self.size = 0 if val is None else 1
    
    def update_size(self):
        """Update the size of the subtree rooted at this node"""
        left_size = self.left.size if self.left else 0
        right_size = self.right.size if self.right else 0
        self.size = left_size + right_size + (1 if self.val is not None else 0)
    
    def rotate_right(self):
        """Right rotation to maintain balance"""
        new_root = self.left
        self.left = new_root.right
        new_root.right = self
        self.update_size()
        new_root.update_size()
        return new_root
    
    def rotate_left(self):
        """Left rotation to maintain balance"""
        new_root = self.right
        self.right = new_root.left
        new_root.left = self
        self.update_size()
        new_root.update_size()
        return new_root
    
    def insert_at_root(self, val):
        """Insert value at root using rotations"""
        if self.val is None:
            self.val = val
            self.size = 1
            return self
        
        if val < self.val:
            if self.left is None:
                self.left = RandomizedBST(val)
            else:
                self.left = self.left.insert_at_root(val)
            return self.rotate_right()
        else:
            if self.right is None:
                self.right = RandomizedBST(val)
            else:
                self.right = self.right.insert_at_root(val)
            return self.rotate_left()
    
    def insert(self, val):
        """Insert with randomization: choose between root insertion and standard BST insertion"""
        if self.val is None:
            self.val = val
            self.size = 1
            return self
        
        # Probability 1/n to insert at root
        if random.random() < 1.0 / (self.size + 1):
            return self.insert_at_root(val)
        
        # Standard BST insertion
        if val < self.val:
            if self.left is None:
                self.left = RandomizedBST(val)
            else:
                self.left = self.left.insert(val)
        else:
            if self.right is None:
                self.right = RandomizedBST(val)
            else:
                self.right = self.right.insert(val)
        
        self.update_size()
        return self
    
    def search(self, val):
        """Search for a value in the tree"""
        if self.val is None:
            return False
        if val == self.val:
            return True
        if val < self.val:
            return self.left.search(val) if self.left else False
        return self.right.search(val) if self.right else False
```
**Why**: Randomization ensures expected O(log n) height without complex balancing logic, simpler than AVL or Red-Black trees

```python
# PROBLEM: Implement a Treap combining BST properties with heap priorities
# TIME: O(log n) expected for all operations SPACE: O(n)
import random

class TreapNode:
    def __init__(self, key, priority=None):
        self.key = key
        self.priority = priority if priority is not None else random.randint(1, 10000)
        self.left = None
        self.right = None

class Treap:
    def __init__(self):
        self.root = None
    
    def rotate_right(self, y):
        """Right rotation"""
        x = y.left
        y.left = x.right
        x.right = y
        return x
    
    def rotate_left(self, x):
        """Left rotation"""
        y = x.right
        x.right = y.left
        y.left = x
        return y
    
    def insert(self, root, key):
        """Insert key maintaining BST property by key and heap property by priority"""
        if root is None:
            return TreapNode(key)
        
        # Standard BST insertion
        if key < root.key:
            root.left = self.insert(root.left, key)
            # Maintain heap property: if child has higher priority, rotate
            if root.left and root.left.priority > root.priority:
                root = self.rotate_right(root)
        else:
            root.right = self.insert(root.right, key)
            # Maintain heap property
            if root.right and root.right.priority > root.priority:
                root = self.rotate_left(root)
        
        return root
    
    def delete(self, root, key):
        """Delete key while maintaining Treap properties"""
        if root is None:
            return None
        
        if key < root.key:
            root.left = self.delete(root.left, key)
        elif key > root.key:
            root.right = self.delete(root.right, key)
        else:
            # Node to delete found
            if root.left is None:
                return root.right
            if root.right is None:
                return root.left
            
            # Both children exist: rotate with higher priority child
            if root.left.priority > root.right.priority:
                root = self.rotate_right(root)
                root.right = self.delete(root.right, key)
            else:
                root = self.rotate_left(root)
                root.left = self.delete(root.left, key)
        
        return root
    
    def search(self, root, key):
        """Standard BST search"""
        if root is None:
            return False
        if key == root.key:
            return True
        if key < root.key:
            return self.search(root.left, key)
        return self.search(root.right, key)
```
**Why**: Combines BST and heap properties; random priorities provide probabilistic balance guarantee without complex rotations

```python
# PROBLEM: Implement an API Gateway with routing, rate limiting, and load balancing
# TIME: O(1) for routing decisions SPACE: O(n) for route table
from collections import defaultdict
import time
from threading import Lock

class APIGateway:
    def __init__(self):
        self.routes = {}  # path -> service_url
        self.rate_limiters = {}  # client_id -> RateLimiter
        self.load_balancers = {}  # service -> LoadBalancer
        self.middleware_chain = []
        self.lock = Lock()
    
    def register_route(self, path, service_url, load_balancer=None):
        """Register a route mapping path to service"""
        self.routes[path] = service_url
        if load_balancer:
            self.load_balancers[service_url] = load_balancer
    
    def add_middleware(self, middleware):
        """Add middleware to the processing chain"""
        self.middleware_chain.append(middleware)
    
    def check_rate_limit(self, client_id, limit=100, window=60):
        """Token bucket rate limiting"""
        with self.lock:
            if client_id not in self.rate_limiters:
                self.rate_limiters[client_id] = TokenBucket(limit, window)
            return self.rate_limiters[client_id].allow_request()
    
    def route_request(self, request):
        """Route request to appropriate service"""
        path = request.get('path')
        client_id = request.get('client_id')
        
        # Check rate limit
        if not self.check_rate_limit(client_id):
            return {'status': 429, 'message': 'Rate limit exceeded'}
        
        # Apply middleware
        for middleware in self.middleware_chain:
            result = middleware(request)
            if result is not None:  # Middleware rejected request
                return result
        
        # Find service
        service_url = self.routes.get(path)
        if not service_url:
            return {'status': 404, 'message': 'Route not found'}
        
        # Load balancing
        if service_url in self.load_balancers:
            target_url = self.load_balancers[service_url].next_server()
        else:
            target_url = service_url
        
        return self.forward_request(target_url, request)
    
    def forward_request(self, target_url, request):
        """Forward request to target service"""
        # In production: use HTTP client to forward
        return {'status': 200, 'service': target_url, 'data': request}

class TokenBucket:
    def __init__(self, capacity, window):
        self.capacity = capacity
        self.tokens = capacity
        self.window = window
        self.last_update = time.time()
    
    def allow_request(self):
        """Check if request is allowed under rate limit"""
        now = time.time()
        elapsed = now - self.last_update
        
        # Refill tokens
        self.tokens = min(self.capacity, self.tokens + elapsed * (self.capacity / self.window))
        self.last_update = now
        
        if self.tokens >= 1:
            self.tokens -= 1
            return True
        return False

class RoundRobinLoadBalancer:
    def __init__(self, servers):
        self.servers = servers
        self.current = 0
    
    def next_server(self):
        """Get next server using round-robin"""
        server = self.servers[self.current]
        self.current = (self.current + 1) % len(self.servers)
        return server
```
**Why**: Centralized entry point simplifies client interactions, provides cross-cutting concerns like rate limiting, authentication, and load balancing

## 5. Service Discovery Pattern

```python
# PROBLEM: Implement service discovery for dynamic service registration and lookup
# TIME: O(1) for registration, O(k) for discovery (k = instances) SPACE: O(n)
import time
from threading import Lock
from collections import defaultdict

class ServiceInstance:
    def __init__(self, service_id, host, port, metadata=None):
        self.service_id = service_id
        self.host = host
        self.port = port
        self.metadata = metadata or {}
        self.heartbeat = time.time()
        self.healthy = True

class ServiceRegistry:
    def __init__(self, heartbeat_timeout=30):
        self.services = defaultdict(dict)  # service_name -> {instance_id: ServiceInstance}
        self.heartbeat_timeout = heartbeat_timeout
        self.lock = Lock()
    
    def register(self, service_name, instance_id, host, port, metadata=None):
        """Register a service instance"""
        with self.lock:
            instance = ServiceInstance(instance_id, host, port, metadata)
            self.services[service_name][instance_id] = instance
        return True
    
    def deregister(self, service_name, instance_id):
        """Deregister a service instance"""
        with self.lock:
            if service_name in self.services:
                self.services[service_name].pop(instance_id, None)
        return True
    
    def send_heartbeat(self, service_name, instance_id):
        """Update heartbeat for service instance"""
        with self.lock:
            if service_name in self.services and instance_id in self.services[service_name]:
                self.services[service_name][instance_id].heartbeat = time.time()
                self.services[service_name][instance_id].healthy = True
                return True
        return False
    
    def discover(self, service_name):
        """Discover all healthy instances of a service"""
        self.check_health()
        with self.lock:
            instances = self.services.get(service_name, {})
            return [
                {
                    'id': inst.service_id,
                    'host': inst.host,
                    'port': inst.port,
                    'metadata': inst.metadata
                }
                for inst in instances.values() if inst.healthy
            ]
    
    def check_health(self):
        """Check health of all instances based on heartbeat"""
        with self.lock:
            current_time = time.time()
            for service_name in self.services:
                for instance in self.services[service_name].values():
                    if current_time - instance.heartbeat > self.heartbeat_timeout:
                        instance.healthy = False

class ServiceDiscoveryClient:
    def __init__(self, registry, service_name, load_balancer=None):
        self.registry = registry
        self.service_name = service_name
        self.load_balancer = load_balancer or RoundRobinBalancer()
        self.cached_instances = []
        self.cache_ttl = 10
        self.last_cache_update = 0
    
    def get_instance(self):
        """Get a service instance using load balancer"""
        # Refresh cache if expired
        if time.time() - self.last_cache_update > self.cache_ttl:
            self.cached_instances = self.registry.discover(self.service_name)
            self.last_cache_update = time.time()
        
        if not self.cached_instances:
            return None
        
        return self.load_balancer.select(self.cached_instances)

class RoundRobinBalancer:
    def __init__(self):
        self.index = 0
    
    def select(self, instances):
        """Select next instance in round-robin fashion"""
        if not instances:
            return None
        instance = instances[self.index % len(instances)]
        self.index += 1
        return instance
```
**Why**: Enables dynamic service location in microservices; automatic health checking and load balancing improve reliability

## 6. Saga Pattern for Distributed Transactions

```python
# PROBLEM: Implement Saga pattern for managing distributed transactions
# TIME: O(n) where n = number of steps SPACE: O(n) for storing compensation
from enum import Enum
from typing import List, Callable, Any

class SagaStep:
    def __init__(self, action: Callable, compensation: Callable, name: str):
        self.action = action  # Forward action
        self.compensation = compensation  # Rollback action
        self.name = name
        self.result = None
        self.completed = False

class SagaStatus(Enum):
    PENDING = "pending"
    IN_PROGRESS = "in_progress"
    COMPLETED = "completed"
    COMPENSATING = "compensating"
    FAILED = "failed"

class SagaOrchestrator:
    def __init__(self):
        self.steps: List[SagaStep] = []
        self.completed_steps: List[SagaStep] = []
        self.status = SagaStatus.PENDING
    
    def add_step(self, action: Callable, compensation: Callable, name: str):
        """Add a step to the saga"""
        self.steps.append(SagaStep(action, compensation, name))
        return self
    
    def execute(self) -> tuple[bool, Any]:
        """Execute the saga with automatic compensation on failure"""
        self.status = SagaStatus.IN_PROGRESS
        
        try:
            # Execute steps in order
            for step in self.steps:
                print(f"Executing step: {step.name}")
                step.result = step.action()
                step.completed = True
                self.completed_steps.append(step)
            
            self.status = SagaStatus.COMPLETED
            return True, "Saga completed successfully"
        
        except Exception as e:
            print(f"Saga failed at step, starting compensation: {e}")
            self.status = SagaStatus.COMPENSATING
            
            # Compensate in reverse order
            for step in reversed(self.completed_steps):
                if step.completed:
                    try:
                        print(f"Compensating step: {step.name}")
                        step.compensation()
                    except Exception as comp_error:
                        print(f"Compensation failed for {step.name}: {comp_error}")
            
            self.status = SagaStatus.FAILED
            return False, f"Saga failed: {str(e)}"

# Example usage: Booking a trip (flight + hotel + car)
class TripBookingSaga:
    @staticmethod
    def book_flight():
        # Simulate flight booking
        print("Booking flight...")
        if False:  # Simulate failure condition
            raise Exception("Flight booking failed")
        return {"flight_id": "FL123"}
    
    @staticmethod
    def cancel_flight():
        print("Cancelling flight...")
    
    @staticmethod
    def book_hotel():
        print("Booking hotel...")
        return {"hotel_id": "HT456"}
    
    @staticmethod
    def cancel_hotel():
        print("Cancelling hotel...")
    
    @staticmethod
    def book_car():
        print("Booking car...")
        return {"car_id": "CR789"}
    
    @staticmethod
    def cancel_car():
        print("Cancelling car...")
    
    @staticmethod
    def create_saga():
        saga = SagaOrchestrator()
        saga.add_step(
            TripBookingSaga.book_flight,
            TripBookingSaga.cancel_flight,
            "Book Flight"
        )
        saga.add_step(
            TripBookingSaga.book_hotel,
            TripBookingSaga.cancel_hotel,
            "Book Hotel"
        )
        saga.add_step(
            TripBookingSaga.book_car,
            TripBookingSaga.cancel_car,
            "Book Car"
        )
        return saga
```
**Why**: Maintains data consistency across microservices without distributed transactions; compensation logic handles failures gracefully

```python
# PROBLEM: Implement various iterator patterns for traversing collections
# TIME: O(1) for next/has_next operations SPACE: O(h) for tree iterators
from collections import deque
from typing import Any, Iterator

class TreeNode:
    def __init__(self, val=0):
        self.val = val
        self.left = None
        self.right = None

# Iterator protocol implementation
class InOrderIterator:
    """In-order traversal iterator for binary tree"""
    def __init__(self, root):
        self.stack = []
        self._push_left(root)
    
    def _push_left(self, node):
        """Push all left children onto stack"""
        while node:
            self.stack.append(node)
            node = node.left
    
    def __iter__(self):
        return self
    
    def __next__(self):
        if not self.stack:
            raise StopIteration
        
        node = self.stack.pop()
        self._push_left(node.right)
        return node.val

class BFSIterator:
    """Breadth-first (level-order) iterator"""
    def __init__(self, root):
        self.queue = deque([root]) if root else deque()
    
    def __iter__(self):
        return self
    
    def __next__(self):
        if not self.queue:
            raise StopIteration
        
        node = self.queue.popleft()
        if node.left:
            self.queue.append(node.left)
        if node.right:
            self.queue.append(node.right)
        return node.val

class CustomCollection:
    """Custom collection with multiple iteration strategies"""
    def __init__(self):
        self.items = []
    
    def add(self, item):
        self.items.append(item)
    
    def __iter__(self):
        """Default forward iterator"""
        return iter(self.items)
    
    def reverse_iterator(self):
        """Reverse iterator using generator"""
        for i in range(len(self.items) - 1, -1, -1):
            yield self.items[i]
    
    def filtered_iterator(self, predicate):
        """Filter iterator using generator"""
        for item in self.items:
            if predicate(item):
                yield item
    
    def batch_iterator(self, batch_size):
        """Batch iterator yielding chunks of items"""
        for i in range(0, len(self.items), batch_size):
            yield self.items[i:i + batch_size]

# Generator-based iterators
def fibonacci_iterator(n):
    """Infinite Fibonacci sequence generator"""
    a, b = 0, 1
    count = 0
    while count < n:
        yield a
        a, b = b, a + b
        count += 1

def tree_levels_iterator(root):
    """Yield tree nodes level by level"""
    if not root:
        return
    
    current_level = [root]
    while current_level:
        yield [node.val for node in current_level]
        next_level = []
        for node in current_level:
            if node.left:
                next_level.append(node.left)
            if node.right:
                next_level.append(node.right)
        current_level = next_level
```
**Why**: Separates traversal logic from collection implementation; generators provide memory-efficient lazy iteration

```python
def timing_decorator(func):
    """Measure execution time of function"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} took {end - start:.4f} seconds")
        return result

def retry_decorator(max_retries=3, delay=1):
    """Retry decorator with exponential backoff"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            retries = 0
            while retries < max_retries:
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    retries += 1
                    if retries == max_retries:
                        raise e
                    time.sleep(delay * (2 ** retries))
        return wrapper
    return decorator

def cache_decorator(func):
    """Memoization decorator"""
    cache = {}
    
    @wraps(func)
    def wrapper(*args):
        if args not in cache:
            cache[args] = func(*args)
        return cache[args]
    
    wrapper.cache = cache
    return wrapper

def log_decorator(func):
    """Logging decorator"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        logging.info(f"Calling {func.__name__} with args={args}, kwargs={kwargs}")
        result = func(*args, **kwargs)
        logging.info(f"{func.__name__} returned {result}")
        return result
    return wrapper

# Decorator with arguments
def validate_types(**types):
    """Type validation decorator"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            # Validate arguments
            for i, (arg_name, arg_type) in enumerate(types.items()):
                if arg_name in kwargs:
                    if not isinstance(kwargs[arg_name], arg_type):
                        raise TypeError(f"{arg_name} must be {arg_type}")
                elif i < len(args):
                    if not isinstance(args[i], arg_type):
                        raise TypeError(f"{arg_name} must be {arg_type}")
            return func(*args, **kwargs)
        return wrapper
    return decorator
```

```python
from dataclasses import dataclass
from datetime import datetime
from enum import Enum
from typing import List, Optional

        for item in self.items:
```

## 10. Kirkpatrick-Seidel Algorithm (Convex Hull)

```python
# PROBLEM: Find convex hull using Kirkpatrick-Seidel's ultimate convex hull algorithm
# TIME: O(n log h) where h = hull vertices SPACE: O(n)
from typing import List, Tuple

Point = Tuple[float, float]

def cross_product(o: Point, a: Point, b: Point) -> float:
    """Calculate cross product of vectors OA and OB"""
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

def bridge(points: List[Point], upper: bool = True) -> Tuple[Point, Point]:
    """Find upper or lower bridge (supporting line)"""
    if len(points) == 2:
        return (points[0], points[1]) if points[0][0] < points[1][0] else (points[1], points[0])
    
    # Pair up points and find bridges recursively
    pairs = []
    for i in range(0, len(points), 2):
        if i + 1 < len(points):
            pairs.append((points[i], points[i + 1]))
        else:
            pairs.append((points[i], points[i]))
    
    # Find median slope
    slopes = []
    for p1, p2 in pairs:
        if p1 != p2:
            slope = (p2[1] - p1[1]) / (p2[0] - p1[0])
            slopes.append(slope)
    
    if not slopes:
        return points[0], points[-1]
    
    slopes.sort()
    median_slope = slopes[len(slopes) // 2]
    
    # Find supporting line with median slope
    intercepts = []
    for p in points:
        intercept = p[1] - median_slope * p[0]
        intercepts.append((intercept, p))
    
    intercepts.sort()
    max_intercept = intercepts[-1][0] if upper else intercepts[0][0]
    
    # Find bridge endpoints
    bridge_points = [p for inter, p in intercepts if abs(inter - max_intercept) < 1e-9]
    bridge_points.sort(key=lambda p: p[0])
    
    return (bridge_points[0], bridge_points[-1])

def upper_hull(points: List[Point]) -> List[Point]:
    """Compute upper hull using divide and conquer"""
    if len(points) <= 2:
        return sorted(points, key=lambda p: p[0])
    
    # Find upper bridge
    b = bridge(points, upper=True)
    
    # Partition points
    left_points = [p for p in points if p[0] <= b[0][0]]
    right_points = [p for p in points if p[0] >= b[1][0]]
    
    # Recursively find upper hulls
    left_hull = upper_hull(left_points)
    right_hull = upper_hull(right_points)
    
    return left_hull + right_hull

def lower_hull(points: List[Point]) -> List[Point]:
    """Compute lower hull using divide and conquer"""
    if len(points) <= 2:
        return sorted(points, key=lambda p: p[0])
    
    # Find lower bridge
    b = bridge(points, upper=False)
    
    # Partition points
    left_points = [p for p in points if p[0] <= b[0][0]]
    right_points = [p for p in points if p[0] >= b[1][0]]
    
    # Recursively find lower hulls
    left_hull = lower_hull(left_points)
    right_hull = lower_hull(right_points)
    
    return left_hull + right_hull

def kirkpatrick_seidel(points: List[Point]) -> List[Point]:
    """
    Kirkpatrick-Seidel convex hull algorithm
    Returns convex hull vertices in counterclockwise order
    """
    if len(points) <= 3:
        return sorted(set(points))
    
    # Remove duplicates
    points = list(set(points))
    points.sort(key=lambda p: p[0])
    
    # Find leftmost and rightmost points
    leftmost = min(points, key=lambda p: p[0])
    rightmost = max(points, key=lambda p: p[0])
    
    # Compute upper and lower hulls
    upper = upper_hull(points)
    lower = lower_hull(points)
    
    # Merge hulls
    hull = []
    
    # Add upper hull (left to right)
    for p in upper:
        if p not in hull:
            hull.append(p)
    
    # Add lower hull (right to left)
    for p in reversed(lower):
        if p not in hull:
            hull.append(p)
    
    return hull

# Simplified version using Graham scan for comparison
def graham_scan(points: List[Point]) -> List[Point]:
    """Graham scan convex hull - O(n log n)"""
    if len(points) <= 3:
        return points
    
    # Sort by x-coordinate
    points = sorted(set(points))
    
    # Build lower hull
    lower = []
    for p in points:
        while len(lower) >= 2 and cross_product(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    
    # Build upper hull
    upper = []
    for p in reversed(points):
        while len(upper) >= 2 and cross_product(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    
    return lower[:-1] + upper[:-1]
```
**Why**: Optimal O(n log h) complexity for convex hull; efficient when hull has few vertices relative to total points
