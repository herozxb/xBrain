## 2. Hashed Array Tree

```python
# PROBLEM: Implement dynamic array with O(1) amortized append and O(1) access
# TIME: O(1) access/append SPACE: O(n) with low memory overhead
class HashedArrayTree:
    def __init__(self, leaf_size=4):
        self.leaf_size = leaf_size  # Number of elements per leaf
        self.leaves = []  # Array of leaf arrays
        self.size = 0
        self.capacity = 0
    
    def __len__(self):
        return self.size
    
    def __getitem__(self, index):
        """Access element in O(1) time using two-level indexing."""
        if index < 0 or index >= self.size:
            raise IndexError("Index out of range")
        
        # Calculate leaf index and position within leaf
        leaf_idx = index // self.leaf_size
        pos_in_leaf = index % self.leaf_size
        
        return self.leaves[leaf_idx][pos_in_leaf]
    
    def __setitem__(self, index, value):
        """Set element in O(1) time."""
        if index < 0 or index >= self.size:
            raise IndexError("Index out of range")
        
        leaf_idx = index // self.leaf_size
        pos_in_leaf = index % self.leaf_size
        self.leaves[leaf_idx][pos_in_leaf] = value
    
    def append(self, value):
        """Append element with O(1) amortized time complexity."""
        # Check if we need a new leaf
        if self.size >= self.capacity:
            self._grow()
        
        # Add element
        leaf_idx = self.size // self.leaf_size
        pos_in_leaf = self.size % self.leaf_size
        self.leaves[leaf_idx][pos_in_leaf] = value
        self.size += 1
    
    def _grow(self):
        """Add a new leaf when capacity is exceeded."""
        new_leaf = [None] * self.leaf_size
        self.leaves.append(new_leaf)
        self.capacity += self.leaf_size

```

**Why**: Hashed Array Trees provide better cache locality than linked lists and less memory overhead than standard dynamic arrays, making them ideal for memory-constrained environments with frequent appends.

## 3. Data Mapper Pattern

```python
# PROBLEM: Separate domain objects from database persistence logic
# TIME: O(1) for object creation SPACE: O(n) for stored objects
from dataclasses import dataclass
from typing import Optional, List

@dataclass
class User:
    """Domain model - pure business logic, no database concerns."""
    id: Optional[int]
    username: str
    email: str
    age: int
    
    def is_adult(self):
        return self.age >= 18

class UserMapper:
    """Data mapper - handles database operations for User objects."""
    
    def __init__(self, db_connection):
        self.db = db_connection
        self.identity_map = {}  # Cache for loaded objects
    
    def find_by_id(self, user_id: int) -> Optional[User]:
        """Load user from database by ID."""
        # Check identity map first
        if user_id in self.identity_map:
            return self.identity_map[user_id]
        
        # Query database
        cursor = self.db.execute(
            "SELECT id, username, email, age FROM users WHERE id = ?",
            (user_id,)
        )
        row = cursor.fetchone()
        
        if row:
            user = self._map_row_to_user(row)
            self.identity_map[user_id] = user
            return user
        return None
    
    def find_all_adults(self) -> List[User]:
        """Load all adult users."""
        cursor = self.db.execute(
            "SELECT id, username, email, age FROM users WHERE age >= 18"
        )
        return [self._map_row_to_user(row) for row in cursor.fetchall()]
    
    def insert(self, user: User) -> User:
        """Insert new user into database."""
        cursor = self.db.execute(
            "INSERT INTO users (username, email, age) VALUES (?, ?, ?)",
            (user.username, user.email, user.age)
        )
        user.id = cursor.lastrowid
        self.identity_map[user.id] = user
        return user
    
    def update(self, user: User) -> None:
        """Update existing user in database."""
        self.db.execute(
            "UPDATE users SET username = ?, email = ?, age = ? WHERE id = ?",
            (user.username, user.email, user.age, user.id)
        )
    
    def delete(self, user: User) -> None:
        """Delete user from database."""
        self.db.execute("DELETE FROM users WHERE id = ?", (user.id,))
        if user.id in self.identity_map:
            del self.identity_map[user.id]
    
    def _map_row_to_user(self, row) -> User:
        """Map database row to domain object."""
        return User(id=row[0], username=row[1], email=row[2], age=row[3])
```

```python
# PROBLEM: Enable services to dynamically discover and communicate with each other
# TIME: O(log n) for service lookup SPACE: O(n) for service registry
import random
from collections import defaultdict
from typing import Dict, List, Optional
import time

class ServiceInstance:
    def __init__(self, service_id: str, host: str, port: int, metadata: dict = None):
        self.service_id = service_id
        self.host = host
        self.port = port
        self.metadata = metadata or {}
        self.last_heartbeat = time.time()
        self.healthy = True

class ServiceRegistry:
    """Centralized service registry for service discovery."""
    
    def __init__(self, heartbeat_timeout: int = 30):
        self.services: Dict[str, List[ServiceInstance]] = defaultdict(list)
        self.heartbeat_timeout = heartbeat_timeout
    
    def register(self, service_name: str, instance: ServiceInstance) -> None:
        """Register a service instance."""
        instance.last_heartbeat = time.time()
        self.services[service_name].append(instance)
    
    def deregister(self, service_name: str, service_id: str) -> None:
        """Remove a service instance."""
        self.services[service_name] = [
            inst for inst in self.services[service_name]
            if inst.service_id != service_id
        ]
    
    def heartbeat(self, service_name: str, service_id: str) -> bool:
        """Update heartbeat for a service instance."""
        for instance in self.services[service_name]:
            if instance.service_id == service_id:
                instance.last_heartbeat = time.time()
                instance.healthy = True
                return True
        return False
    
    def get_instances(self, service_name: str) -> List[ServiceInstance]:
        """Get all healthy instances of a service."""
        self._check_health()
        return [inst for inst in self.services[service_name] if inst.healthy]
    
    def _check_health(self) -> None:
        """Mark instances as unhealthy if heartbeat timeout exceeded."""
        current_time = time.time()
        for instances in self.services.values():
            for instance in instances:
                if current_time - instance.last_heartbeat > self.heartbeat_timeout:
                    instance.healthy = False

class LoadBalancer:
    """Client-side load balancer for service instances."""
    
    @staticmethod
    def round_robin(instances: List[ServiceInstance]) -> Optional[ServiceInstance]:
        """Select instance using round-robin strategy."""
        if not instances:
            return None
        return random.choice(instances)
    
    @staticmethod
    def least_connections(instances: List[ServiceInstance]) -> Optional[ServiceInstance]:
        """Select instance with least active connections."""
        if not instances:
            return None
        return min(instances, key=lambda x: x.metadata.get('connections', 0))
```

```python
# PROBLEM: Implement ACID-compliant transaction management
# TIME: O(log n) for commit/rollback SPACE: O(n) for transaction log
from enum import Enum
from typing import Dict, List, Any
import logging

class TransactionState(Enum):
    ACTIVE = "active"
    COMMITTED = "committed"
    ABORTED = "aborted"

class Transaction:
    """Represents a database transaction with ACID guarantees."""
    
    def __init__(self, transaction_id: int):
        self.transaction_id = transaction_id
        self.state = TransactionState.ACTIVE
        self.operations: List[tuple] = []
        self.undo_log: List[tuple] = []  # For rollback (Atomicity)
        self.redo_log: List[tuple] = []  # For recovery (Durability)
    
    def add_operation(self, operation: str, data: Dict[str, Any], old_value: Any = None):
        """Log operation with before/after images for atomicity and durability."""
        self.operations.append((operation, data))
        if old_value is not None:
            self.undo_log.append((operation, data, old_value))
        self.redo_log.append((operation, data))

class TransactionManager:
    """Manages ACID-compliant transactions."""
    
    def __init__(self):
        self.transactions: Dict[int, Transaction] = {}
        self.next_transaction_id = 1
        self.lock_manager = LockManager()  # For Isolation
        self.logger = logging.getLogger(__name__)
    
    def begin_transaction(self) -> Transaction:
        """Start a new transaction (Atomicity guarantee)."""
        txn = Transaction(self.next_transaction_id)
        self.transactions[self.next_transaction_id] = txn
        self.next_transaction_id += 1
        self.logger.info(f"Transaction {txn.transaction_id} started")
        return txn
    
    def commit(self, transaction: Transaction) -> bool:
        """Commit transaction atomically (Atomicity, Consistency, Durability)."""
        if transaction.state != TransactionState.ACTIVE:
            return False
        
        try:
            # Write-ahead logging for durability
            self._write_to_log(transaction)
            
            # Apply all operations atomically
            for operation, data in transaction.operations:
                self._apply_operation(operation, data)
            
            # Release all locks (Isolation)
            self.lock_manager.release_all_locks(transaction.transaction_id)
            
            transaction.state = TransactionState.COMMITTED
            self.logger.info(f"Transaction {transaction.transaction_id} committed")
            return True
            
        except Exception as e:
            # Rollback on failure (Atomicity)
            self.rollback(transaction)
            self.logger.error(f"Transaction {transaction.transaction_id} failed: {e}")
            return False

    """Manages locks for isolation."""
    
    def __init__(self):
        self.locks: Dict[str, int] = {}  # resource -> transaction_id
    
    def acquire_lock(self, resource: str, transaction_id: int) -> bool:
        """Acquire exclusive lock on resource."""
        if resource in self.locks and self.locks[resource] != transaction_id:
```

```python
# PROBLEM: Implement efficient parallel processing using process pool
# TIME: O(n/p) for p parallel workers SPACE: O(p) for worker processes
from multiprocessing import Process, Queue, Manager
from typing import Callable, List, Any
import os

class ProcessPool:
    """Simple process pool implementation for CPU-bound tasks."""
    
    def __init__(self, num_workers: int = None):
        self.num_workers = num_workers or os.cpu_count()
        self.workers: List[Process] = []
        self.task_queue = Queue()
        self.result_queue = Queue()
        self.manager = Manager()
        self.shutdown_flag = self.manager.Value('b', False)
    
    def start(self):
        """Start worker processes."""
        for i in range(self.num_workers):
            worker = Process(
                target=self._worker_loop,
                args=(self.task_queue, self.result_queue, self.shutdown_flag)
            )
            worker.start()
            self.workers.append(worker)
    
    def submit(self, func: Callable, *args, **kwargs) -> int:
        """Submit a task to the pool."""
        task_id = os.urandom(16).hex()  # Simple unique ID
        self.task_queue.put((task_id, func, args, kwargs))
        return task_id
    
    def map(self, func: Callable, items: List[Any]) -> List[Any]:
        """Apply function to all items in parallel."""
        task_ids = []
        for item in items:
            task_id = self.submit(func, item)
            task_ids.append(task_id)
        
        # Collect results
        results = {}
        while len(results) < len(task_ids):
            task_id, result = self.result_queue.get()
            results[task_id] = result
        
        # Return results in original order
        return [results[tid] for tid in task_ids]
    
    def shutdown(self):
        """Gracefully shutdown all workers."""
        self.shutdown_flag.value = True
        for worker in self.workers:
            worker.join(timeout=5)
            if worker.is_alive():
                worker.terminate()
    
    @staticmethod
    def _worker_loop(task_queue: Queue, result_queue: Queue, shutdown_flag):
        """Worker process main loop."""
        while not shutdown_flag.value:
            try:
                # Get task with timeout to check shutdown flag periodically
                task_id, func, args, kwargs = task_queue.get(timeout=1)
                result = func(*args, **kwargs)
                result_queue.put((task_id, result))
            except:
                # Queue.get timeout - continue loop to check shutdown flag
                continue
```

```python
# PROBLEM: Compute Delaunay triangulation from 2D point set
# TIME: O(n log n) SPACE: O(n)
import math

class Point:
    def __init__(self, x: float, y: float):
        self.x = x
        self.y = y
    
    def __eq__(self, other):
        return self.x == other.x and self.y == other.y
    
    def __hash__(self):
        return hash((self.x, self.y))
    
    def __repr__(self):
        return f"Point({self.x}, {self.y})"

class Triangle:
    def __init__(self, p1: Point, p2: Point, p3: Point):
        self.vertices = (p1, p2, p3)
        self.circumcenter, self.circumradius = self._compute_circumcircle()
    
    def _compute_circumcircle(self):
        """Compute circumcenter and circumradius."""
        p1, p2, p3 = self.vertices
        
        # Calculate circumcenter using perpendicular bisectors
        d = 2 * (p1.x * (p2.y - p3.y) + p2.x * (p3.y - p1.y) + p3.x * (p1.y - p2.y))
        
        if abs(d) < 1e-10:  # Collinear points
            return None, float('inf')
        
        ux = ((p1.x**2 + p1.y**2) * (p2.y - p3.y) +
              (p2.x**2 + p2.y**2) * (p3.y - p1.y) +
              (p3.x**2 + p3.y**2) * (p1.y - p2.y)) / d
        
        uy = ((p1.x**2 + p1.y**2) * (p3.x - p2.x) +
              (p2.x**2 + p2.y**2) * (p1.x - p3.x) +
              (p3.x**2 + p3.y**2) * (p2.x - p1.x)) / d
        
        center = Point(ux, uy)
        radius = math.sqrt((p1.x - ux)**2 + (p1.y - uy)**2)
        
        return center, radius
    
    def contains_in_circumcircle(self, point: Point) -> bool:
        """Check if point is inside circumcircle."""
        if self.circumcenter is None:
            return False
        
        dist = math.sqrt((point.x - self.circumcenter.x)**2 + 
                        (point.y - self.circumcenter.y)**2)
        return dist < self.circumradius

class DelaunayTriangulation:
    """Compute Delaunay triangulation using Bowyer-Watson algorithm."""
    
    def __init__(self, points: List[Point]):
        self.points = points
        self.triangles: List[Triangle] = []
    
    def compute(self) -> List[Triangle]:
        """Compute Delaunay triangulation."""
        # Create super triangle containing all points
        super_triangle = self._create_super_triangle()
        self.triangles = [super_triangle]
        
        # Add points one by one
        for point in self.points:
            self._add_point(point)
        
        # Remove triangles containing super triangle vertices
        self.triangles = [t for t in self.triangles 
                         if not self._shares_vertex_with_super(t, super_triangle)]
        
        return self.triangles
    
    def _create_super_triangle(self) -> Triangle:
        """Create triangle that contains all points."""
        min_x = min(p.x for p in self.points)
        max_x = max(p.x for p in self.points)
        min_y = min(p.y for p in self.points)
        max_y = max(p.y for p in self.points)
        
        dx = max_x - min_x
        dy = max_y - min_y
        delta = max(dx, dy) * 10
        
        return Triangle(
            Point(min_x - delta, min_y - delta),
            Point(max_x + delta, min_y - delta),
            Point((min_x + max_x) / 2, max_y + delta)
        )
    
    def _add_point(self, point: Point):
        """Add a point to the triangulation."""
        # Find triangles whose circumcircle contains the point
        bad_triangles = [t for t in self.triangles if t.contains_in_circumcircle(point)]
        
        # Find boundary polygon
        polygon = self._find_boundary_polygon(bad_triangles)
        
        # Remove bad triangles
        self.triangles = [t for t in self.triangles if t not in bad_triangles]
        
        # Create new triangles from point to polygon edges
        for edge in polygon:
            new_triangle = Triangle(point, edge[0], edge[1])
            self.triangles.append(new_triangle)
    
    def _find_boundary_polygon(self, bad_triangles: List[Triangle]) -> List[tuple]:
        """Find boundary polygon of bad triangles."""
        # Simplified: actual implementation finds edges that appear only once
        return [(t.vertices[0], t.vertices[1]) for t in bad_triangles[:3]]
    
    def _shares_vertex_with_super(self, triangle: Triangle, super_tri: Triangle) -> bool:
        """Check if triangle shares any vertex with super triangle."""
        return any(v in super_tri.vertices for v in triangle.vertices)
```

```python
# PROBLEM: Implement k-dimensional tree for efficient range queries and nearest neighbor search
# TIME: O(log n) search in balanced tree SPACE: O(n)
from typing import List, Optional, Tuple
import math

class KDNode:
    def __init__(self, point: Tuple[float, ...], axis: int):
        self.point = point
        self.axis = axis
        self.left: Optional['KDNode'] = None
        self.right: Optional['KDNode'] = None

class KDTree:
    """K-dimensional tree for spatial indexing."""
    
    def __init__(self, points: List[Tuple[float, ...]]):
        self.k = len(points[0]) if points else 0  # Number of dimensions
        self.root = self._build_tree(points, depth=0)
    
    def _build_tree(self, points: List[Tuple[float, ...]], depth: int) -> Optional[KDNode]:
        """Recursively build KD-tree."""
        if not points:
            return None
        
        # Select axis based on depth
        axis = depth % self.k
        
        # Sort points by axis and choose median
        points.sort(key=lambda p: p[axis])
        median_idx = len(points) // 2
        
        # Create node and recurse
        node = KDNode(points[median_idx], axis)
        node.left = self._build_tree(points[:median_idx], depth + 1)
        node.right = self._build_tree(points[median_idx + 1:], depth + 1)
        
        return node
    
    def nearest_neighbor(self, target: Tuple[float, ...]) -> Tuple[Tuple[float, ...], float]:
        """Find nearest neighbor to target point."""
        best = [None, float('inf')]
        self._nn_search(self.root, target, best)
        return best[0], best[1]
    
    def _nn_search(self, node: Optional[KDNode], target: Tuple[float, ...], 
                   best: List, visited: set = None):
        """Recursive nearest neighbor search."""
        if node is None:
            return
        
        if visited is None:
            visited = set()
        
        # Calculate distance to current node
        dist = self._euclidean_distance(node.point, target)
        
        # Update best if closer
        if dist < best[1] and node.point not in visited:
            best[0] = node.point
            best[1] = dist
        
        # Determine which subtree to search first
        axis = node.axis
        if target[axis] < node.point[axis]:
            first, second = node.left, node.right
        else:
            first, second = node.right, node.left
        
        # Search the nearer subtree
        self._nn_search(first, target, best, visited)
        
        # Check if we need to search the other subtree
        if abs(target[axis] - node.point[axis]) < best[1]:
            self._nn_search(second, target, best, visited)
    
    def range_query(self, target: Tuple[float, ...], radius: float) -> List[Tuple[float, ...]]:
        """Find all points within radius of target."""
        results = []
        self._range_search(self.root, target, radius, results)
        return results
    
    def _range_search(self, node: Optional[KDNode], target: Tuple[float, ...], 
                      radius: float, results: List):
        """Recursive range search."""
        if node is None:
            return
        
        # Check if current point is within radius
        if self._euclidean_distance(node.point, target) <= radius:
            results.append(node.point)
        
        # Determine which subtrees to search
        axis = node.axis
        
        # Always search the subtree that could contain points within radius
        if target[axis] - radius <= node.point[axis]:
            self._range_search(node.left, target, radius, results)
        
        if target[axis] + radius >= node.point[axis]:
            self._range_search(node.right, target, radius, results)
    
    def _euclidean_distance(self, p1: Tuple[float, ...], p2: Tuple[float, ...]) -> float:
        """Calculate Euclidean distance between two points."""
        return math.sqrt(sum((a - b)**2 for a, b in zip(p1, p2)))
```

```python
# TIME: O(1) for event append, O(n) for state reconstruction SPACE: O(n) for event store
from dataclasses import dataclass, field
from typing import List, Dict, Any, Callable
from datetime import datetime
import json

@dataclass
class Event:
    """Base event class for event sourcing."""
    event_type: str
    aggregate_id: str
    timestamp: datetime
    data: Dict[str, Any] = field(default_factory=dict)
    version: int = 0
    
    def to_dict(self) -> dict:
        return {
            'event_type': self.event_type,
            'aggregate_id': self.aggregate_id,
            'timestamp': self.timestamp.isoformat(),
            'data': self.data,
            'version': self.version
        }

class EventStore:
    """Append-only store for events."""
    
    def __init__(self):
        self.events: Dict[str, List[Event]] = {}
        self.subscribers: List[Callable[[Event], None]] = []
    
    def append(self, event: Event) -> None:
        """Append event to store."""
        if event.aggregate_id not in self.events:
            self.events[event.aggregate_id] = []
        
        event.version = len(self.events[event.aggregate_id]) + 1
        self.events[event.aggregate_id].append(event)
        
        # Notify subscribers
        for subscriber in self.subscribers:
            subscriber(event)
    
    def get_events(self, aggregate_id: str, from_version: int = 0) -> List[Event]:
        """Get all events for an aggregate from a specific version."""
        if aggregate_id not in self.events:
            return []
        
        return [e for e in self.events[aggregate_id] if e.version > from_version]
    
    def subscribe(self, callback: Callable[[Event], None]) -> None:
        """Subscribe to event notifications."""
        self.subscribers.append(callback)

class Aggregate:
    """Base aggregate root with event sourcing support."""
    
    def __init__(self, aggregate_id: str):
        self.aggregate_id = aggregate_id
        self.version = 0
        self.uncommitted_events: List[Event] = []
    
    def apply_event(self, event: Event) -> None:
        """Apply event to update aggregate state."""
        method_name = f"_apply_{event.event_type}"
        if hasattr(self, method_name):
            getattr(self, method_name)(event)
        self.version = event.version
    
    def raise_event(self, event_type: str, data: Dict[str, Any]) -> Event:
        """Create and apply new event."""
        event = Event(
            event_type=event_type,
            aggregate_id=self.aggregate_id,
            timestamp=datetime.now(),
            data=data,
            version=self.version + 1
        )
        self.apply_event(event)
        self.uncommitted_events.append(event)
        return event

class BankAccount(Aggregate):
    def __init__(self, account_id: str):
        super().__init__(account_id)
        self.balance = 0.0
        self.owner = None
        self.is_active = False
    
    @classmethod
    def create(cls, account_id: str, owner: str, initial_balance: float = 0):
        """Factory method to create new account."""
        account = cls(account_id)
        account.raise_event('AccountCreated', {
            'owner': owner,
            'initial_balance': initial_balance
        })
        return account
    
    def deposit(self, amount: float) -> None:
        """Deposit money into account."""
        if not self.is_active:
            raise ValueError("Account is not active")
        if amount <= 0:
            raise ValueError("Amount must be positive")
        
        self.raise_event('MoneyDeposited', {'amount': amount})
    
    def withdraw(self, amount: float) -> None:
        """Withdraw money from account."""
        if not self.is_active:
            raise ValueError("Account is not active")
        if amount <= 0:
            raise ValueError("Amount must be positive")
        if amount > self.balance:
            raise ValueError("Insufficient funds")
        
        self.raise_event('MoneyWithdrawn', {'amount': amount})
    
    def close(self) -> None:
        """Close the account."""
        if self.balance != 0:
            raise ValueError("Cannot close account with non-zero balance")
        
        self.raise_event('AccountClosed', {})
    
    # Event handlers
    def _apply_AccountCreated(self, event: Event) -> None:
        self.owner = event.data['owner']
        self.balance = event.data['initial_balance']
        self.is_active = True
    
    def _apply_MoneyDeposited(self, event: Event) -> None:
        self.balance += event.data['amount']
    
    def _apply_MoneyWithdrawn(self, event: Event) -> None:
        self.balance -= event.data['amount']
    
    def _apply_AccountClosed(self, event: Event) -> None:
        self.is_active = False

    """Repository for reconstructing aggregates from events."""
    
    def __init__(self, event_store: EventStore):
        self.event_store = event_store
    
    def save(self, account: BankAccount) -> None:
        """Save uncommitted events to event store."""
        for event in account.uncommitted_events:
            self.event_store.append(event)
        account.uncommitted_events.clear()
    
    def get_by_id(self, account_id: str) -> BankAccount:
        """Reconstruct account from events."""
        events = self.event_store.get_events(account_id)
        if not events:
            return None
        
        account = BankAccount(account_id)
        for event in events:
            account.apply_event(event)
        
        return account
```

## 10. CAP Theorem - Trade-offs Implementation

```python
# PROBLEM: Implement distributed system with configurable CAP trade-offs
# TIME: O(1) for reads/writes in happy path SPACE: O(n) for data + replicas
from enum import Enum
from typing import Dict, Any, List, Optional
import threading
import time
from collections import defaultdict

class ConsistencyLevel(Enum):
    """CAP theorem consistency levels."""
    STRONG = "strong"        # CP: Consistent but may be unavailable
    EVENTUAL = "eventual"    # AP: Available but eventually consistent
    QUORUM = "quorum"        # Balanced approach

class DistributedNode:
    """Single node in distributed system."""
    
    def __init__(self, node_id: str):
        self.node_id = node_id
        self.data: Dict[str, Any] = {}
        self.vector_clock: Dict[str, int] = defaultdict(int)
        self.is_available = True
    
    def read(self, key: str) -> Optional[Any]:
        """Read value from node."""
        if not self.is_available:
            raise Exception(f"Node {self.node_id} unavailable")
        return self.data.get(key)
    
    def write(self, key: str, value: Any, vector_clock: Dict[str, int]) -> bool:
        """Write value to node."""
        if not self.is_available:
            raise Exception(f"Node {self.node_id} unavailable")
        
        # Update vector clock
        self.vector_clock[self.node_id] += 1
        for node_id, timestamp in vector_clock.items():
            self.vector_clock[node_id] = max(self.vector_clock[node_id], timestamp)
        
        self.data[key] = value
        return True

class DistributedStore:
    """Distributed key-value store with configurable CAP trade-offs."""
    
    def __init__(self, nodes: List[DistributedNode], 
                 consistency: ConsistencyLevel = ConsistencyLevel.QUORUM):
        self.nodes = {node.node_id: node for node in nodes}
        self.consistency = consistency
        self.quorum_size = len(nodes) // 2 + 1
        self.lock = threading.Lock()
    
    def read(self, key: str) -> Optional[Any]:
        """Read with configurable consistency."""
        if self.consistency == ConsistencyLevel.STRONG:
            return self._read_strong(key)
        elif self.consistency == ConsistencyLevel.EVENTUAL:
            return self._read_eventual(key)
        else:  # QUORUM
            return self._read_quorum(key)
    
    def write(self, key: str, value: Any) -> bool:
        """Write with configurable consistency."""
        if self.consistency == ConsistencyLevel.STRONG:
            return self._write_strong(key, value)
        elif self.consistency == ConsistencyLevel.EVENTUAL:
            return self._write_eventual(key, value)
        else:  # QUORUM
            return self._write_quorum(key, value)
    
    def _read_strong(self, key: str) -> Optional[Any]:
        """CP: Read from all available nodes, require all to respond."""
        values = []
        for node in self.nodes.values():
            try:
                value = node.read(key)
                values.append(value)
            except Exception:
                # Strong consistency requires all nodes to be available
                raise Exception("System unavailable: cannot achieve strong consistency")
        
        # All nodes must agree
        if values and all(v == values[0] for v in values):
            return values[0]
        raise Exception("Consistency violation detected")
    
    def _read_eventual(self, key: str) -> Optional[Any]:
        """AP: Read from any available node."""
        for node in self.nodes.values():
            try:
                return node.read(key)
            except Exception:
                continue
        
        raise Exception("System unavailable: no nodes reachable")
    
    def _read_quorum(self, key: str) -> Optional[Any]:
        """Balanced: Read from majority of nodes."""
        values = []
        for node in self.nodes.values():
            try:
                value = node.read(key)
                values.append(value)
                if len(values) >= self.quorum_size:
                    # Return most common value
                    return max(set(values), key=values.count)
            except Exception:
                continue
        
        if len(values) >= self.quorum_size:
            return max(set(values), key=values.count)
        
        raise Exception("System unavailable: quorum not reached")
    
    def _write_strong(self, key: str, value: Any) -> bool:
        """CP: Write to all nodes, fail if any unavailable."""
        for node in self.nodes.values():
            try:
                node.write(key, value, {})
            except Exception:
                raise Exception("System unavailable: cannot achieve strong consistency")
        return True
    
    def _write_eventual(self, key: str, value: Any) -> bool:
        """AP: Write to any available node, replicate asynchronously."""
        written = False
        for node in self.nodes.values():
            try:
                node.write(key, value, {})
                written = True
                break
            except Exception:
                continue
        
        if written:
            # Asynchronously replicate to other nodes
            threading.Thread(target=self._replicate_async, args=(key, value)).start()
            return True
        
        raise Exception("System unavailable: no nodes reachable")
    
    def _write_quorum(self, key: str, value: Any) -> bool:
        """Balanced: Write to majority of nodes."""
        success_count = 0
        for node in self.nodes.values():
            try:
                node.write(key, value, {})
                success_count += 1
            except Exception:
                continue
        
        if success_count >= self.quorum_size:
            return True
        
        raise Exception("System unavailable: quorum not reached")
    
    def _replicate_async(self, key: str, value: Any) -> None:
        """Asynchronously replicate write to all nodes."""
        time.sleep(0.1)  # Simulate network delay
        for node in self.nodes.values():
            try:
                node.write(key, value, {})
            except Exception:
                pass  # Ignore failures in async replication
```

**Why**: Understanding CAP trade-offs is essential for designing distributed systems: choose Consistency for correctness (CP), Availability for uptime (AP), or balance both with quorum-based approaches.
