## [Maximum Sum Subarray of Size K]

```python
# PROBLEM: Find the maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)
def max_sum_subarray_k(arr, k):
    """
    Sliding window technique to find maximum sum subarray of fixed size k.
    
    Args:
        arr: List of integers (can be negative)
        k: Size of the window (must be <= len(arr))
    
    Returns:
        Maximum sum of any contiguous subarray of size k
    """
    n = len(arr)
    
    # Edge case: window larger than array
    if k > n:
        return None
    
    # Calculate sum of first window
    window_sum = sum(arr[:k])
    max_sum = window_sum
    
    # Slide the window across the array
    # Remove leftmost element, add new rightmost element
    for i in range(n - k):
        # Subtract element leaving window, add element entering window
        window_sum = window_sum - arr[i] + arr[i + k]
        max_sum = max(max_sum, window_sum)
    
    return max_sum
```
**Why:** Classic sliding window problem. Demonstrates O(n) optimization over brute force O(nk) by reusing previous calculations. Window moves one step at a time, maintaining running sum efficiently.

## [Two Sum II - Sorted Array]

```python
# PROBLEM: Find two numbers in sorted array that sum to target (1-indexed result)
# TIME: O(n) SPACE: O(1)
def two_sum_sorted(numbers, target):
    """
    Two pointers technique on sorted array.
    
    Args:
        numbers: Sorted list of integers (ascending order)
        target: Target sum to find
    
    Returns:
        Tuple of 1-indexed positions, or None if not found
    """
    left, right = 0, len(numbers) - 1
    
    # Two pointers converge from both ends
    while left < right:
        current_sum = numbers[left] + numbers[right]
        
        if current_sum == target:
            # Return 1-indexed positions
            return (left + 1, right + 1)
        elif current_sum < target:
            # Sum too small, move left pointer right (increase sum)
            left += 1
        else:
            # Sum too large, move right pointer left (decrease sum)
            right -= 1
    
    return None  # No solution found
```
**Why:** Two pointers eliminates the need for binary search or hash map. Exploits sorted property to narrow search space. Each comparison rules out one element permanently.

```python
# PROBLEM: Generate all permutations of a list of distinct integers
# TIME: O(n! * n) SPACE: O(n) for recursion depth
def permute(nums):
    """
    Backtracking to generate all permutations.
    
    Args:
        nums: List of distinct integers
    
    Returns:
        List of all permutations
    """
    result = []
    n = len(nums)
    
    def backtrack(start):
        """
        Generate permutations by swapping elements.
        
        Args:
            start: Current position being fixed
        """
        # Base case: all positions fixed, record permutation
        if start == n:
            result.append(nums[:])  # Copy current state
            return
        
        # Try each element at position 'start'
        for i in range(start, n):
            # Swap: place element i at position start
            nums[start], nums[i] = nums[i], nums[start]
            
            # Recurse: fix next position
            backtrack(start + 1)
            
            # Backtrack: restore original order
            nums[start], nums[i] = nums[i], nums[start]
    
    backtrack(0)
    return result
```
**Why:** Backtracking builds solutions incrementally. Swap-based approach avoids extra memory. Time complexity is factorial because we generate n! permutations.

```python
# PROBLEM: Count connected components in an undirected graph
# TIME: O(n * α(n)) ≈ O(n) SPACE: O(n)
class UnionFind:
    """Union-Find (Disjoint Set Union) data structure."""
    
    def __init__(self, n):
        # Each node starts as its own parent (separate component)
        self.parent = list(range(n))
        # Rank for union by rank optimization
        self.rank = [0] * n
        # Track number of components
        self.components = n
    
    def find(self, x):
        """Find root with path compression."""
        if self.parent[x] != x:
            # Path compression: point directly to root
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        """Union two sets by rank. Returns True if merged."""
        root_x, root_y = self.find(x), self.find(y)
        
        # Already in same component
        if root_x == root_y:
            return False
        
        # Union by rank: attach shorter tree to taller
        if self.rank[root_x] < self.rank[root_y]:
            self.parent[root_x] = root_y
        elif self.rank[root_x] > self.rank[root_y]:
            self.parent[root_y] = root_x
        else:
            self.parent[root_y] = root_x
            self.rank[root_x] += 1
        
        self.components -= 1
        return True


def count_components(n, edges):
    """
    Count connected components using Union-Find.
    
    Args:
        n: Number of nodes (labeled 0 to n-1)
        edges: List of undirected edges as [u, v] pairs
    
    Returns:
        Number of connected components
    """
    uf = UnionFind(n)
    
    # Union all edges
    for u, v in edges:
        uf.union(u, v)
    
    return uf.components
```
**Why:** Union-Find excels at dynamic connectivity. Path compression and union by rank give near-constant amortized time. Ideal for graph component problems.

## [Longest Common Subsequence]

```python
# PROBLEM: Find length of longest common subsequence between two strings
# TIME: O(m * n) SPACE: O(min(m, n))
def longest_common_subsequence(text1, text2):
    """
    Classic dynamic programming problem.
    
    Args:
        text1: First string
        text2: Second string
    
    Returns:
        Length of longest common subsequence
    """
    # Optimize space: use shorter string for DP array
    if len(text1) < len(text2):
        text1, text2 = text2, text1
    
    m, n = len(text1), len(text2)
    
    # Only need previous row for DP
    prev = [0] * (n + 1)
    
    for i in range(1, m + 1):
        curr = [0] * (n + 1)
        
        for j in range(1, n + 1):
            if text1[i - 1] == text2[j - 1]:
                # Characters match: extend LCS
                curr[j] = prev[j - 1] + 1
            else:
                # Take maximum of excluding either character
                curr[j] = max(prev[j], curr[j - 1])
        
        prev = curr
    
    return prev[n]
```
**Why:** DP breaks problem into overlapping subproblems. Space optimization from O(mn) to O(min(m,n)) by keeping only previous row. LCS is fundamental to diff algorithms and bioinformatics.

## [Dijkstra's Shortest Path]

```python
# PROBLEM: Find shortest path from source to all nodes in weighted graph
# TIME: O((V + E) log V) SPACE: O(V)
import heapq

def dijkstra(n, edges, source):
    """
    Dijkstra's algorithm using min-heap.
    
    Args:
        n: Number of nodes (labeled 0 to n-1)
        edges: List of directed edges as [u, v, weight]
        source: Starting node
    
    Returns:
        List of shortest distances (infinity if unreachable)
    """
    # Build adjacency list
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u].append((v, w))
    
    # Initialize distances
    INF = float('inf')
    dist = [INF] * n
    dist[source] = 0
    
    # Min-heap: (distance, node)
    heap = [(0, source)]
    visited = set()
    
    while heap:
        d, u = heapq.heappop(heap)
        
        # Skip if already processed (better path found)
        if u in visited:
            continue
        visited.add(u)
        
        # Relax all neighbors
        for v, w in graph[u]:
            if v not in visited and dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(heap, (dist[v], v))
    
    return dist
```
**Why:** Dijkstra is optimal for single-source shortest path with non-negative weights. Min-heap provides efficient extraction of minimum. Foundation for GPS navigation and network routing.

## [Lowest Common Ancestor in BST]

```python
# PROBLEM: Find lowest common ancestor of two nodes in a binary search tree
# TIME: O(h) where h is height SPACE: O(1)
class TreeNode:
    """Binary tree node."""
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


def lowest_common_ancestor_bst(root, p, q):
    """
    Find LCA in BST using BST property.
    
    Args:
        root: Root of BST
        p: First node value
        q: Second node value
    
    Returns:
        Value of lowest common ancestor
    """
    # Ensure p < q for simpler logic
    if p > q:
        p, q = q, p
    
    current = root
    
    while current:
        # Both nodes in left subtree
        if q < current.val:
            current = current.left
        # Both nodes in right subtree
        elif p > current.val:
            current = current.right
        # Split point found: this is LCA
        # (one on each side, or current is one of them)
        else:
            return current.val
    
    return None  # Should never reach here if p, q exist
```
**Why:** BST property enables O(h) solution without recursion or extra storage. LCA crucial for tree path queries and computational biology applications.

## [Merge K Sorted Lists]

```python
# PROBLEM: Merge k sorted linked lists into one sorted list
# TIME: O(N log k) where N = total nodes, k = number of lists
# SPACE: O(k) for heap
import heapq

class ListNode:
    """Singly linked list node."""
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next
    
    # For heap comparison
    def __lt__(self, other):
        return self.val < other.val


def merge_k_sorted_lists(lists):
    """
    Merge k sorted lists using min-heap.
    
    Args:
        lists: List of ListNode heads
    
    Returns:
        Head of merged sorted list
    """
    # Remove empty lists and push first nodes to heap
    heap = []
    for head in lists:
        if head:
            heapq.heappush(heap, head)
    
    # Dummy head for easier list construction
    dummy = ListNode(0)
    current = dummy
    
    while heap:
        # Get smallest node
        smallest = heapq.heappop(heap)
        
        # Add to result list
        current.next = smallest
        current = current.next
        
        # Push next node from same list if exists
        if smallest.next:
            heapq.heappush(heap, smallest.next)
    
    return dummy.next
```
**Why:** Min-heap efficiently tracks smallest element across k lists. Each pop/push is O(log k), total N operations. Better than sequential merge O(kN) or divide-and-conquer O(N log k).

```python
# PROBLEM: Implement a trie (prefix tree) with insert, search, and startsWith
# TIME: O(m) per operation where m = word/prefix length
# SPACE: O(alphabet_size * total_characters)
class TrieNode:
    """Trie node with children map and end marker."""
    def __init__(self):
        self.children = {}  # char -> TrieNode
        self.is_end = False


class Trie:
    """Prefix tree for efficient string operations."""
    
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word):
        """
        Insert word into trie.
        
        Args:
            word: String to insert
        """
        node = self.root
        for char in word:
            # Create node if doesn't exist
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
    
    def search(self, word):
        """
        Check if complete word exists in trie.
        
        Args:
            word: String to search
        
        Returns:
            True if word exists, False otherwise
        """
        node = self._traverse(word)
        return node is not None and node.is_end
    
    def starts_with(self, prefix):
        """
        Check if any word starts with prefix.
        
        Args:
            prefix: Prefix to check
        
        Returns:
            True if prefix exists, False otherwise
        """
        return self._traverse(prefix) is not None
    
    def _traverse(self, prefix):
        """Helper: traverse to end of prefix, return node or None."""
        node = self.root
        for char in prefix:
            if char not in node.children:
                return None
            node = node.children[char]
        return node
```
**Why:** Trie provides O(m) prefix operations independent of total words stored. Essential for autocomplete, spell checkers, and IP routing tables. Space-efficient for shared prefixes.

## [LRU Cache]

```python
# PROBLEM: Design a Least Recently Used (LRU) cache
# TIME: O(1) for both get and put
# SPACE: O(capacity)
class DoublyNode:
    """Doubly linked list node."""
    def __init__(self, key=0, val=0):
        self.key = key
        self.val = val
        self.prev = None
        self.next = None


class LRUCache:
    """
    LRU Cache using hashmap + doubly linked list.
    
    Hashmap provides O(1) access.
    Doubly linked list maintains order with O(1) updates.
    """
    
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}  # key -> Node
        
        # Dummy head and tail for easier list manipulation
        self.head = DoublyNode()  # Most recent
        self.tail = DoublyNode()  # Least recent
        self.head.next = self.tail
        self.tail.prev = self.head
    
    def get(self, key):
        """
        Get value if exists. Moves to most recent.
        
        Returns:
            Value or -1 if not found
        """
        if key not in self.cache:
            return -1
        
        node = self.cache[key]
        # Move to front (most recently used)
        self._remove(node)
        self._add_to_front(node)
        
        return node.val
    
    def put(self, key, value):
        """
        Insert/update key-value pair.
        Evicts LRU item if at capacity.
        """
        if key in self.cache:
            # Update existing: remove old, add new
            self._remove(self.cache[key])
        
        node = DoublyNode(key, value)
        self.cache[key] = node
        self._add_to_front(node)
        
        # Evict LRU if over capacity
        if len(self.cache) > self.capacity:
            lru = self.tail.prev  # Least recently used
            self._remove(lru)
            del self.cache[lru.key]
    
    def _remove(self, node):
        """Remove node from linked list."""
        node.prev.next = node.next
        node.next.prev = node.prev
    
    def _add_to_front(self, node):
        """Add node right after head (most recent)."""
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node
```
**Why:** Hashmap + doubly linked list achieves O(1) for both operations. Directly manipulating pointers avoids list traversal. Pattern used in OS page replacement and database buffer management.
