## Suffix Tree

```python
# PROBLEM: Build a suffix tree for efficient pattern matching in strings
# TIME: O(n²) construction, O(m) search SPACE: O(n²)
class SuffixTreeNode:
    """Node in a suffix tree structure."""
    def __init__(self):
        self.children = {}  # Dictionary mapping characters to child nodes
        self.suffix_link = None  # Pointer for Ukkonen's algorithm optimization
        self.start = None  # Start index of the edge label
        self.end = None  # End index (can be a reference for leaves)
        self.suffix_index = -1  # Index of suffix starting at this node

class SuffixTree:
    """
    Suffix Tree implementation for string pattern matching.
    Uses Ukkonen's algorithm for O(n) construction in practice.
    """
    def __init__(self, text: str):
        self.text = text + "$"  # Add unique terminator
        self.root = SuffixTreeNode()
        self._build_tree()
    
    def _build_tree(self):
        """Build suffix tree using naive O(n²) insertion."""
        n = len(self.text)
        for i in range(n):
            self._insert_suffix(i)
    
    def _insert_suffix(self, start_idx: int):
        """Insert a single suffix starting at start_idx into the tree."""
        current = self.root
        i = start_idx
        
        while i < len(self.text):
            char = self.text[i]
            if char not in current.children:
                # Create new leaf node
                leaf = SuffixTreeNode()
                leaf.start = i
                leaf.end = len(self.text)
                leaf.suffix_index = start_idx
                current.children[char] = leaf
                return
            else:
                # Traverse existing edge
                child = current.children[char]
                edge_start = child.start
                edge_end = child.end if isinstance(child.end, int) else len(self.text)
                
                # Match along the edge
                j = 0
                while (j < edge_end - edge_start and 
                       i + j < len(self.text) and 
                       self.text[edge_start + j] == self.text[i + j]):
                    j += 1
                
                if j == edge_end - edge_start:
                    # Full edge match, continue to child
                    current = child
                    i += j
                else:
                    # Split edge - create internal node
                    split = SuffixTreeNode()
                    split.start = edge_start
                    split.end = edge_start + j
                    
                    # Update child's start
                    child.start = edge_start + j
                    split.children[self.text[edge_start + j]] = child
                    
                    # Create new leaf
                    leaf = SuffixTreeNode()
                    leaf.start = i + j
                    leaf.end = len(self.text)
                    leaf.suffix_index = start_idx
                    split.children[self.text[i + j]] = leaf
                    
                    current.children[char] = split
                    return
    
    def search(self, pattern: str) -> list[int]:
        """
        Search for all occurrences of pattern in the text.
        Returns list of starting indices where pattern appears.
        """
        current = self.root
        i = 0
        
        while i < len(pattern):
            char = pattern[i]
            if char not in current.children:
                return []  # Pattern not found
            child = current.children[char]
            edge_start = child.start
            edge_end = child.end if isinstance(child.end, int) else len(self.text)
            
            # Match along the edge
            j = 0
            while (j < edge_end - edge_start and 
                   i < len(pattern) and 
                   self.text[edge_start + j] == pattern[i]):
                j += 1
                i += 1
            
            if i < len(pattern) and j == edge_end - edge_start:
                current = child
            elif i == len(pattern):
                return self._get_all_leaves(child)
            else:
                return []
        
        return self._get_all_leaves(current)
    
    def _get_all_leaves(self, node: SuffixTreeNode) -> list[int]:
        """Collect all suffix indices from leaves under this node."""
        result = []
        if node.suffix_index >= 0:
            result.append(node.suffix_index)
        for child in node.children.values():
            result.extend(self._get_all_leaves(child))
        return sorted(result)
```

**Why**: Suffix trees enable O(m) substring search after O(n) construction, making them essential for bioinformatics, text indexing, and efficient pattern matching in large strings.

## Rat Maze (Backtracking)

```python
# PROBLEM: Find a path from start to end in a maze with obstacles
# TIME: O(4^(n*m)) worst case SPACE: O(n*m)
def solve_rat_maze(maze: list[list[int]]) -> list[list[int]] | None:
    """
    Solve the rat maze problem using backtracking.
    
    Args:
        maze: 2D grid where 1 = path, 0 = wall/blocked
    
    Returns:
        Solution grid with path marked as 1, or None if no path exists
    
    The rat starts at (0, 0) and needs to reach (n-1, m-1).
    Can only move right and down.
    """
    if not maze or not maze[0]:
        return None
    
    n, m = len(maze), len(maze[0])
    
    # Check if start or end is blocked
    if maze[0][0] == 0 or maze[n-1][m-1] == 0:
        return None
    
    # Initialize solution grid
    solution = [[0] * m for _ in range(n)]
    
    def is_safe(x: int, y: int) -> bool:
        """Check if position (x, y) is valid and not blocked."""
        return 0 <= x < n and 0 <= y < m and maze[x][y] == 1 and solution[x][y] == 0
    
    def backtrack(x: int, y: int) -> bool:
        """
        Recursively find path using backtracking.
        Returns True if path to destination is found.
        """
        # Check if we reached destination
        if x == n - 1 and y == m - 1:
            solution[x][y] = 1
            return True
        
        if is_safe(x, y):
            # Mark current cell as part of path
            solution[x][y] = 1
            
            # Move right
            if backtrack(x, y + 1):
                return True
            
            # Move down
            if backtrack(x + 1, y):
                return True
            
            # Move left (if allowed)
            if backtrack(x, y - 1):
                return True
            
            # Move up (if allowed)
            if backtrack(x - 1, y):
                return True
            
            # Backtrack: unmark current cell
            solution[x][y] = 0
        
        return False
    
    if backtrack(0, 0):
        return solution
    return None


def solve_rat_maze_all_paths(maze: list[list[int]]) -> list[list[list[int]]]:
    """
    Find all possible paths from start to end in the maze.
    Returns a list of all valid solutions.
    """
    if not maze or not maze[0]:
        return []
    
    n, m = len(maze), len(maze[0])
    if maze[0][0] == 0 or maze[n-1][m-1] == 0:
        return []
    
    all_solutions = []
    solution = [[0] * m for _ in range(n)]
    
    def is_safe(x: int, y: int) -> bool:
        return 0 <= x < n and 0 <= y < m and maze[x][y] == 1 and solution[x][y] == 0
    
    def backtrack(x: int, y: int):
        if x == n - 1 and y == m - 1:
            solution[x][y] = 1
            # Deep copy the solution
            all_solutions.append([row[:] for row in solution])
            solution[x][y] = 0
            return
        
        if is_safe(x, y):
            solution[x][y] = 1
            
            # Try all four directions
            for dx, dy in [(0, 1), (1, 0), (0, -1), (-1, 0)]:
                backtrack(x + dx, y + dy)
            
            solution[x][y] = 0
    
    backtrack(0, 0)
    return all_solutions
```

**Why**: Classic backtracking problem that demonstrates systematic exploration with pruning. Essential for understanding constraint satisfaction and path-finding algorithms.

## Rod Cutting

```python
# PROBLEM: Maximize profit by cutting a rod into pieces with given prices
# TIME: O(n²) SPACE: O(n)
def rod_cutting(prices: list[int], n: int) -> int:
    """
    Solve the rod cutting problem using dynamic programming.
    
    Args:
        prices: prices[i] = price for rod of length i+1
        n: length of the rod
    
    Returns:
        Maximum profit achievable
    """
    if n <= 0:
        return 0
    
    # dp[i] = maximum profit for rod of length i
    dp = [0] * (n + 1)
    
    for i in range(1, n + 1):
        max_val = float('-inf')
        for j in range(min(i, len(prices))):
            # Try cutting at length j+1
            max_val = max(max_val, prices[j] + dp[i - j - 1])
        dp[i] = max_val
    
    return dp[n]


def rod_cutting_with_cuts(prices: list[int], n: int) -> tuple[int, list[int]]:
    """
    Return both maximum profit and the optimal cuts.
    
    Returns:
        Tuple of (max_profit, list of cut lengths)
    """
    if n <= 0:
        return 0, []
    
    dp = [0] * (n + 1)
    cut = [0] * (n + 1)  # cut[i] = optimal first cut for length i
    
    for i in range(1, n + 1):
        max_val = float('-inf')
        for j in range(min(i, len(prices))):
            if prices[j] + dp[i - j - 1] > max_val:
                max_val = prices[j] + dp[i - j - 1]
                cut[i] = j + 1  # Length of first piece
        dp[i] = max_val
    
    # Reconstruct the cuts
    cuts = []
    length = n
    while length > 0:
        cuts.append(cut[length])
        length -= cut[length]
    
    return dp[n], cuts


def rod_cutting_memo(prices: list[int], n: int) -> int:
    """Top-down memoized version of rod cutting."""
    memo = {}
    
    def helper(length: int) -> int:
        if length == 0:
            return 0
        if length in memo:
            return memo[length]
        
        max_val = float('-inf')
        for j in range(min(length, len(prices))):
            max_val = max(max_val, prices[j] + helper(length - j - 1))
        
        memo[length] = max_val
        return max_val
    
    return helper(n)
```

**Why**: Fundamental DP problem demonstrating optimal substructure. Applications include resource allocation, stock trading, and any problem involving optimal partitioning.

## Maximal Square

```python
# PROBLEM: Find the largest square of 1s in a binary matrix
# TIME: O(m*n) SPACE: O(n) optimized
def maximal_square(matrix: list[list[str]]) -> int:
    """
    Find the area of the largest square containing only 1s.
    
    Args:
        matrix: 2D binary matrix with '0' and '1' characters
    
    Returns:
        Area of the largest square of 1s
    """
    if not matrix or not matrix[0]:
        return 0
    
    m, n = len(matrix), len(matrix[0])
    # dp[j] = side length of largest square ending at current row, column j
    dp = [0] * (n + 1)
    max_side = 0
    prev = 0  # dp[j-1] from previous iteration (diagonal)
    
    for i in range(m):
        for j in range(1, n + 1):
            temp = dp[j]  # Save before overwriting
            if matrix[i][j-1] == '1':
                # dp[j] = min(left, top, diagonal) + 1
                dp[j] = min(dp[j], dp[j-1], prev) + 1
                max_side = max(max_side, dp[j])
            else:
                dp[j] = 0
            prev = temp
        prev = 0
    
    return max_side * max_side


def maximal_square_with_position(matrix: list[list[str]]) -> tuple[int, tuple[int, int]]:
    """
    Find the largest square and its bottom-right position.
    
    Returns:
        Tuple of (area, (row, col)) of bottom-right corner
    """
    if not matrix or not matrix[0]:
        return 0, (-1, -1)
    
    m, n = len(matrix), len(matrix[0])
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    max_side = 0
    max_pos = (0, 0)
    
    for i in range(m):
        for j in range(n):
            if matrix[i][j] == '1':
                dp[i+1][j+1] = min(dp[i][j+1], dp[i+1][j], dp[i][j]) + 1
                if dp[i+1][j+1] > max_side:
                    max_side = dp[i+1][j+1]
                    max_pos = (i, j)
    
    return max_side * max_side, max_pos


def maximal_rectangle(matrix: list[list[str]]) -> int:
    """
    Find the largest rectangle containing only 1s.
    Uses histogram approach - harder variant of maximal square.
    """
    if not matrix or not matrix[0]:
        return 0
    
    m, n = len(matrix), len(matrix[0])
    heights = [0] * n
    max_area = 0
    
    def largest_rectangle_area(h: list[int]) -> int:
        """Helper: largest rectangle in histogram."""
        stack = []
        max_area = 0
        h = h + [0]  # Append sentinel
        
        for i, height in enumerate(h):
            while stack and h[stack[-1]] > height:
                top = stack.pop()
                width = i if not stack else i - stack[-1] - 1
                max_area = max(max_area, h[top] * width)
            stack.append(i)
        
        return max_area
    
    for row in matrix:
        for j in range(n):
            heights[j] = heights[j] + 1 if row[j] == '1' else 0
        max_area = max(max_area, largest_rectangle_area(heights))
    
    return max_area
```

**Why**: Demonstrates 2D DP optimization with space reduction. The histogram variant extends to maximum rectangle problems, useful in image processing and spatial analysis.

## KD-Tree

```python
# PROBLEM: Efficiently search for nearest neighbors in k-dimensional space
# TIME: O(n log n) build, O(log n) average search SPACE: O(n)
from typing import TypeVar, Generic
import math

T = TypeVar('T', int, float)

class KDNode(Generic[T]):
    """Node in a k-dimensional tree."""
    def __init__(self, point: tuple[T, ...], axis: int):
        self.point = point
        self.axis = axis
        self.left: KDNode[T] | None = None
        self.right: KDNode[T] | None = None

class KDTree(Generic[T]):
    """
    K-dimensional tree for efficient nearest neighbor search.
    Alternates between dimensions at each level.
    """
    def __init__(self, points: list[tuple[T, ...]]):
        self.k = len(points[0]) if points else 0
        self.root = self._build(points, 0)
    
    def _build(self, points: list[tuple[T, ...]], depth: int) -> KDNode[T] | None:
        """Build KD-tree by recursively partitioning points."""
        if not points:
            return None
        
        # Select axis based on depth
        axis = depth % self.k
        
        # Sort points by axis and choose median
        points.sort(key=lambda p: p[axis])
        median = len(points) // 2
        
        # Create node and construct subtrees
        node = KDNode(points[median], axis)
        node.left = self._build(points[:median], depth + 1)
        node.right = self._build(points[median + 1:], depth + 1)
        
        return node
    
    def nearest_neighbor(self, target: tuple[T, ...]) -> tuple[tuple[T, ...], float] | None:
        """
        Find the nearest neighbor to target point.
        
        Returns:
            Tuple of (nearest_point, distance) or None if tree is empty
        """
        if not self.root:
            return None
        
        self._best = None
        self._best_dist = float('inf')
        self._search(self.root, target, 0)
        return self._best, self._best_dist
    
    def _search(self, node: KDNode[T] | None, target: tuple[T, ...], depth: int):
        """Recursively search for nearest neighbor."""
        if node is None:
            return
        
        # Calculate distance to current node
        dist = self._distance(node.point, target)
        if dist < self._best_dist:
            self._best = node.point
            self._best_dist = dist
        
        # Determine which subtree to search first
        axis = node.axis
        if target[axis] < node.point[axis]:
            near, far = node.left, node.right
        else:
            near, far = node.right, node.left
        
        # Search near subtree first
        self._search(near, target, depth + 1)
        
        # Check if far subtree could contain closer points
        if (target[axis] - node.point[axis]) ** 2 < self._best_dist:
            self._search(far, target, depth + 1)
    
    def _distance(self, p1: tuple[T, ...], p2: tuple[T, ...]) -> float:
        """Calculate Euclidean distance between two points."""
        return sum((a - b) ** 2 for a, b in zip(p1, p2)) ** 0.5
    
    def range_search(self, low: tuple[T, ...], high: tuple[T, ...]) -> list[tuple[T, ...]]:
        """
        Find all points within the hyperrectangle defined by low and high.
        """
        result = []
        self._range_search_helper(self.root, low, high, result)
        return result
    
    def _range_search_helper(self, node: KDNode[T] | None, 
                             low: tuple[T, ...], high: tuple[T, ...],
                             result: list[tuple[T, ...]]):
        """Recursively find points in range."""
        if node is None:
            return
        
        # Check if current point is in range
        if all(low[i] <= node.point[i] <= high[i] for i in range(self.k)):
            result.append(node.point)
        
        axis = node.axis
        
        # Check left subtree
        if low[axis] <= node.point[axis]:
            self._range_search_helper(node.left, low, high, result)
        
        # Check right subtree
        if high[axis] >= node.point[axis]:
            self._range_search_helper(node.right, low, high, result)
```

**Why**: Essential for spatial databases, computer graphics, and machine learning. Enables efficient nearest neighbor queries in multi-dimensional spaces without checking all points.

## Bloom Filter Variants

```python
# PROBLEM: Space-efficient probabilistic membership test with tunable error rate
# TIME: O(k) where k = number of hash functions SPACE: O(m) bits
import hashlib
import math
from bitarray import bitarray

class BloomFilter:
    """
    Standard Bloom Filter for set membership testing.
    False positives possible, no false negatives.
    """
    def __init__(self, n: int, fp_rate: float):
        """
        Initialize bloom filter.
        
        Args:
            n: Expected number of elements
            fp_rate: Desired false positive rate
        """
        # Calculate optimal size and number of hash functions
        self.size = int(-n * math.log(fp_rate) / (math.log(2) ** 2))
        self.hash_count = int(self.size / n * math.log(2))
        self.bit_array = bitarray(self.size)
        self.bit_array.setall(0)
    
    def _hashes(self, item: bytes) -> list[int]:
        """Generate k hash values for an item using double hashing."""
        h1 = int(hashlib.md5(item).hexdigest(), 16)
        h2 = int(hashlib.sha1(item).hexdigest(), 16)
        return [(h1 + i * h2) % self.size for i in range(self.hash_count)]
    
    def add(self, item: str | bytes):
        """Add an item to the filter."""
        if isinstance(item, str):
            item = item.encode()
        for pos in self._hashes(item):
            self.bit_array[pos] = 1
    
    def contains(self, item: str | bytes) -> bool:
        """Check if item might be in the set (may have false positives)."""
        if isinstance(item, str):
            item = item.encode()
        return all(self.bit_array[pos] for pos in self._hashes(item))


class CountingBloomFilter:
    """
    Bloom filter variant that supports deletion using counters.
    Uses more space but allows element removal.
    """
    def __init__(self, n: int, fp_rate: float):
        self.size = int(-n * math.log(fp_rate) / (math.log(2) ** 2))
        self.hash_count = int(self.size / n * math.log(2))
        self.counters = [0] * self.size
    
    def _hashes(self, item: bytes) -> list[int]:
        h1 = int(hashlib.md5(item).hexdigest(), 16)
        h2 = int(hashlib.sha1(item).hexdigest(), 16)
        return [(h1 + i * h2) % self.size for i in range(self.hash_count)]
    
    def add(self, item: str | bytes):
        if isinstance(item, str):
            item = item.encode()
        for pos in self._hashes(item):
            self.counters[pos] += 1
    
    def remove(self, item: str | bytes) -> bool:
        """Remove an item. Returns False if item wasn't present."""
        if isinstance(item, str):
            item = item.encode()
        positions = self._hashes(item)
        
        # Check if item might be present
        if not all(self.counters[pos] > 0 for pos in positions):
            return False
        
        # Decrement counters
        for pos in positions:
            self.counters[pos] -= 1
        return True
    
    def contains(self, item: str | bytes) -> bool:
        if isinstance(item, str):
            item = item.encode()
        return all(self.counters[pos] > 0 for pos in self._hashes(item))


class ScalableBloomFilter:
    """
    Bloom filter that grows automatically as more items are added.
    Creates new larger filters when current one reaches capacity.
    """
    def __init__(self, initial_size: int = 1000, fp_rate: float = 0.01):
        self.filters = []
        self.initial_size = initial_size
        self.fp_rate = fp_rate
        self._add_filter()
    
    def _add_filter(self):
        """Add a new larger filter to the chain."""
        size = self.initial_size * (2 ** len(self.filters))
        self.filters.append(BloomFilter(size, self.fp_rate * (0.9 ** len(self.filters))))
    
    def add(self, item: str | bytes):
        """Add item to the most recent filter."""
        self.filters[-1].add(item)
        
        # Add new filter if current is getting full
        # This is a heuristic - actual check would track count
        if len(self.filters[-1].bit_array) / 2 < sum(self.filters[-1].bit_array):
            self._add_filter()
    
    def contains(self, item: str | bytes) -> bool:
        """Check all filters for the item."""
        return any(f.contains(item) for f in self.filters)
```

**Why**: Bloom filters are crucial for databases, caching, and network applications where space efficiency matters more than perfect accuracy. Counting variant enables deletion for dynamic sets.

## Lazy Initialization

```python
# PROBLEM: Defer expensive object creation until actually needed
# TIME: O(1) access after initialization SPACE: O(1) overhead
from typing import Callable, TypeVar, Generic
import threading

T = TypeVar('T')

class Lazy(Generic[T]):
    """
    Thread-safe lazy initialization wrapper.
    Creates value only when first accessed.
    """
    def __init__(self, factory: Callable[[], T]):
        """
        Args:
            factory: Function to create the value when needed
        """
        self._factory = factory
        self._value: T | None = None
        self._lock = threading.Lock()
        self._initialized = False
    
    @property
    def value(self) -> T:
        """Get the value, initializing if necessary."""
        if not self._initialized:
            with self._lock:
                # Double-check locking pattern
                if not self._initialized:
                    self._value = self._factory()
                    self._initialized = True
        return self._value
    
    def is_initialized(self) -> bool:
        """Check if value has been created."""
        return self._initialized
    
    def reset(self):
        """Reset to uninitialized state (value will be recreated on next access)."""
        with self._lock:
            self._value = None
            self._initialized = False


class LazyDict(dict):
    """
    Dictionary that creates values on-demand using factory functions.
    Useful for caching expensive computations.
    """
    def __init__(self, default_factory: Callable[[any], any] | None = None):
        super().__init__()
        self._factory = default_factory
    
    def __missing__(self, key):
        """Create value when key is missing."""
        if self._factory is None:
            raise KeyError(key)
        value = self._factory(key)
        self[key] = value
        return value


class VirtualProxy:
    """
    Virtual Proxy pattern for lazy loading of heavy objects.
    Acts as a placeholder until the real object is needed.
    """
    def __init__(self, subject_factory: Callable[[], any]):
        self._factory = subject_factory
        self._subject: any = None
        self._lock = threading.Lock()
    
    def _get_subject(self):
        """Get the real subject, creating if necessary."""
        if self._subject is None:
            with self._lock:
                if self._subject is None:
                    self._subject = self._factory()
        return self._subject
    
    def __getattr__(self, name):
        """Delegate all attribute access to the real subject."""
        return getattr(self._get_subject(), name)
    
    def __repr__(self):
        return f"VirtualProxy({self._subject.__class__.__name__ if self._subject else 'Not Loaded'})"
```

## Retry Pattern

```python
# PROBLEM: Handle transient failures by automatically retrying operations
# TIME: Variable based on retry count and delays SPACE: O(1)
import time
import random
from typing import Callable, TypeVar, Type
from functools import wraps
from dataclasses import dataclass

T = TypeVar('T')

@dataclass
class RetryConfig:
    """Configuration for retry behavior."""
    max_attempts: int = 3
    base_delay: float = 1.0  # Base delay in seconds
    max_delay: float = 60.0
    exponential_base: float = 2.0
    jitter: bool = True  # Add random jitter to prevent thundering herd


class RetryExhausted(Exception):
    """Raised when all retry attempts are exhausted."""
    def __init__(self, attempts: int, last_exception: Exception):
        self.attempts = attempts
        self.last_exception = last_exception
        super().__init__(f"All {attempts} retry attempts exhausted. Last error: {last_exception}")


def retry(config: RetryConfig | None = None, 
          exceptions: tuple[Type[Exception], ...] = (Exception,)):
    """
    Decorator for automatic retry on failure.
    
    Args:
        config: Retry configuration
        exceptions: Tuple of exception types to retry on
    """
    if config is None:
        config = RetryConfig()
    
    def decorator(func: Callable[..., T]) -> Callable[..., T]:
        @wraps(func)
        def wrapper(*args, **kwargs) -> T:
            last_exception = None
            
            for attempt in range(config.max_attempts):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    last_exception = e
                    
                    if attempt == config.max_attempts - 1:
                        raise RetryExhausted(config.max_attempts, e)
                    
                    # Calculate delay with exponential backoff
                    delay = min(
                        config.base_delay * (config.exponential_base ** attempt),
                        config.max_delay
                    )
                    
                    # Add jitter if enabled
                    if config.jitter:
                        delay = delay * (0.5 + random.random())
                    
                    time.sleep(delay)
            
            raise RetryExhausted(config.max_attempts, last_exception)
        
        return wrapper
    return decorator


class RetryableOperation:
    """
    Class-based retry for more complex scenarios.
    Supports callbacks and custom retry logic.
    """
    def __init__(self, config: RetryConfig | None = None):
        self.config = config or RetryConfig()
        self._on_retry_callback: Callable[[int, Exception, float], None] | None = None
        self._on_failure_callback: Callable[[Exception], None] | None = None
    
    def on_retry(self, callback: Callable[[int, Exception, float], None]):
        """Register callback called before each retry (attempt, exception, delay)."""
        self._on_retry_callback = callback
        return self
    
    def on_failure(self, callback: Callable[[Exception], None]):
        """Register callback called when all retries exhausted."""
        self._on_failure_callback = callback
        return self
    
    def execute(self, operation: Callable[[], T], 
                exceptions: tuple[Type[Exception], ...] = (Exception,)) -> T:
        """Execute operation with retry logic."""
        for attempt in range(self.config.max_attempts):
            try:
                return operation()
            except exceptions as e:
                if attempt == self.config.max_attempts - 1:
                    if self._on_failure_callback:
                        self._on_failure_callback(e)
                    raise RetryExhausted(self.config.max_attempts, e)
                
                delay = min(
                    self.config.base_delay * (self.config.exponential_base ** attempt),
                    self.config.max_delay
                )
                
                if self.config.jitter:
                    delay = delay * (0.5 + random.random())
                
                if self._on_retry_callback:
                    self._on_retry_callback(attempt + 1, e, delay)
                
                time.sleep(delay)

def circuit_breaker_retry(max_attempts: int = 3, 
                          failure_threshold: int = 5,
                          recovery_timeout: float = 30.0):
    """
    Combined retry with circuit breaker pattern.
    Opens circuit after threshold failures to prevent cascading failures.
    """
    def decorator(func: Callable[..., T]) -> Callable[..., T]:
        state = {'failures': 0, 'state': 'closed', 'last_failure': 0}
        lock = threading.Lock()
        
        @wraps(func)
        def wrapper(*args, **kwargs) -> T:
            with lock:
                if state['state'] == 'open':
                    if time.time() - state['last_failure'] > recovery_timeout:
                        state['state'] = 'half-open'
                    else:
                        raise Exception("Circuit breaker is open")
            
            try:
                result = func(*args, **kwargs)
                with lock:
                    if state['state'] == 'half-open':
                        state['state'] = 'closed'
                    state['failures'] = 0
                return result
            except Exception as e:
                with lock:
                    state['failures'] += 1
                    state['last_failure'] = time.time()
                    if state['failures'] >= failure_threshold:
                        state['state'] = 'open'
                raise
        
        return wrapper
    return decorator
```

**Why**: Essential for distributed systems where transient failures are common. Exponential backoff with jitter prevents overwhelming recovering services. Circuit breaker prevents cascading failures.

```python
import hashlib
import threading
import time
from typing import Any, Callable
from collections import OrderedDict

class LRUCache:
    """Thread-safe LRU cache implementation."""
    
    def __init__(self, max_size: int = 1000, ttl: float | None = None):
        """
        Args:
            max_size: Maximum number of items in cache
            ttl: Time-to-live in seconds (None = no expiration)
        """
        self.max_size = max_size
        self.ttl = ttl
        self._cache: OrderedDict[str, tuple[Any, float]] = OrderedDict()
        self._lock = threading.RLock()
    
    def get(self, key: str) -> Any | None:
        """Get value from cache, returns None if not found or expired."""
        with self._lock:
            if key not in self._cache:
                return None
            
            value, timestamp = self._cache[key]
            
            # Check TTL expiration
            if self.ttl and time.time() - timestamp > self.ttl:
                del self._cache[key]
                return None
            
            # Move to end (most recently used)
            self._cache.move_to_end(key)
            return value
    
    def set(self, key: str, value: Any):
        """Set value in cache."""
        with self._lock:
            if key in self._cache:
                del self._cache[key]
            elif len(self._cache) >= self.max_size:
                self._cache.popitem(last=False)  # Remove oldest
            
            self._cache[key] = (value, time.time())
    
    def delete(self, key: str) -> bool:
        """Delete key from cache. Returns True if key existed."""
        with self._lock:
            if key in self._cache:
                del self._cache[key]
                return True
            return False
    
    def clear(self):
        """Clear all items from cache."""
        with self._lock:
            self._cache.clear()


class ConsistentHashing:
    """
    Consistent hashing for distributing keys across cache nodes.
    Minimizes key redistribution when nodes are added/removed.
    """
    def __init__(self, virtual_nodes: int = 150):
        """
        Args:
            virtual_nodes: Number of virtual nodes per physical node
        """
        self.virtual_nodes = virtual_nodes
        self.ring: dict[int, str] = {}
        self.sorted_keys: list[int] = []
        self._lock = threading.Lock()
    
    def _hash(self, key: str) -> int:
        """Generate hash for a key using MD5."""
        return int(hashlib.md5(key.encode()).hexdigest(), 16)
    
    def add_node(self, node: str):
        """Add a cache node to the ring."""
        with self._lock:
            for i in range(self.virtual_nodes):
                virtual_key = f"{node}#{i}"
                hash_key = self._hash(virtual_key)
                self.ring[hash_key] = node
            self.sorted_keys = sorted(self.ring.keys())
    
    def remove_node(self, node: str):
        """Remove a cache node from the ring."""
        with self._lock:
            for i in range(self.virtual_nodes):
                virtual_key = f"{node}#{i}"
                hash_key = self._hash(virtual_key)
                if hash_key in self.ring:
                    del self.ring[hash_key]
            self.sorted_keys = sorted(self.ring.keys())
    
    def get_node(self, key: str) -> str | None:
        """Get the node responsible for a given key."""
        if not self.ring:
            return None
        
        hash_key = self._hash(key)
        
        # Binary search for first node >= hash_key
        for sorted_key in self.sorted_keys:
            if hash_key <= sorted_key:
                return self.ring[sorted_key]
        
        # Wrap around to first node
        return self.ring[self.sorted_keys[0]]


class DistributedCache:
    """
    Distributed cache with consistent hashing and write-through support.
    """
    def __init__(self, nodes: list[str] | None = None, 
                 local_cache_size: int = 1000,
                 ttl: float | None = None):
        self.hash_ring = ConsistentHashing()
        self.local_caches: dict[str, LRUCache] = {}
        self.ttl = ttl
        
        if nodes:
            for node in nodes:
                self.add_node(node)

        self.hash_ring.add_node(node)

            del self.local_caches[node]

        node = self.hash_ring.get_node(key)

    def get_or_compute(self, key: str, compute: Callable[[], Any]) -> Any:

        return value
```
