## js_classes_constructor

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>ES6 Class Constructor</title>
</head>
<body>
    <script>
        // PROBLEM: Initialize object state with constructor
        // TIME: O(1) SPACE: O(n) where n = number of instances
        
        class BankAccount {
            // Constructor initializes instance properties
            constructor(accountNumber, ownerName, initialBalance = 0) {
                // Validate inputs
                if (!accountNumber || !ownerName) {
                    throw new Error('Account number and owner name are required');
                }
                
                // Initialize instance properties
                this.accountNumber = accountNumber;
                this.ownerName = ownerName;
                this.balance = initialBalance;
                this.transactions = [];
                this.createdAt = new Date();
                
                // Log account creation
                this._logTransaction('ACCOUNT_CREATED', initialBalance);
            }
            
            // Instance method
            deposit(amount) {
                if (amount <= 0) {
                    throw new Error('Deposit amount must be positive');
                }
                this.balance += amount;
                this._logTransaction('DEPOSIT', amount);
                return this.balance;
            }
            
            // Instance method
            withdraw(amount) {
                if (amount <= 0) {
                    throw new Error('Withdrawal amount must be positive');
                }
                if (amount > this.balance) {
                    throw new Error('Insufficient funds');
                }
                this.balance -= amount;
                this._logTransaction('WITHDRAWAL', amount);
                return this.balance;
            }
            
            // Private method (convention with underscore)
            _logTransaction(type, amount) {
                this.transactions.push({
                    type,
                    amount,
                    timestamp: new Date(),
                    balanceAfter: this.balance
                });
            }
            
            // Getter method
            getAccountSummary() {
                return {
                    account: this.accountNumber,
                    owner: this.ownerName,
                    balance: this.balance,
                    transactionCount: this.transactions.length
                };
            }
        }
        
        // Usage
        const account = new BankAccount('123456789', 'John Doe', 1000);
        
        account.deposit(500);
        account.withdraw(200);
        
        const summary = account.getAccountSummary();
        
        document.body.innerHTML = `
            <h1>Bank Account Demo</h1>
            <div>
                <p><strong>Account:</strong> ${summary.account}</p>
                <p><strong>Owner:</strong> ${summary.owner}</p>
                <p><strong>Balance:</strong> $${summary.balance}</p>
                <p><strong>Transactions:</strong> ${summary.transactionCount}</p>
            </div>
            <h2>Transaction History</h2>
            <ul>
                ${account.transactions.map(t => 
                    `<li>${t.type}: $${t.amount} (Balance: $${t.balanceAfter})</li>`
                ).join('')}
            </ul>
        `;
    </script>
</body>
</html>
```

**Why:** Constructors provide structured initialization of objects, ensuring instances start in a valid state with proper validation and setup, fundamental to object-oriented programming in JavaScript.

```html
        // PROBLEM: Extract portions of strings without modifying original

        class TextProcessor {
            constructor(text) {
                this.text = text;
            }
            
            // Extract first n characters
            getFirstCharacters(n) {
                return this.text.slice(0, n);
            }
            
            // Extract last n characters
            getLastCharacters(n) {
                return this.text.slice(-n);
            }
            
            // Extract middle portion
            getMiddle(start, end) {
                return this.text.slice(start, end);
            }
            
            // Remove first n characters
            removeFirst(n) {
                return this.text.slice(n);
            }
            
            // Remove last n characters
            removeLast(n) {
                return this.text.slice(0, -n);
            }
            
            // Get file extension
            getFileExtension() {
                const lastDot = this.text.lastIndexOf('.');
                return lastDot !== -1 ? this.text.slice(lastDot + 1) : '';
            }
            
            // Truncate with ellipsis
            truncate(maxLength) {
                if (this.text.length <= maxLength) return this.text;
                return this.text.slice(0, maxLength - 3) + '...';
            }
            
            // Capitalize first letter
            capitalizeFirst() {
                return this.text.charAt(0).toUpperCase() + this.text.slice(1);
```

```html
    <script>
        // PROBLEM: Break strings into arrays using delimiters
        // TIME: O(n) where n = string length SPACE: O(n)
        
        class StringSplitter {
            constructor(str) {
                this.str = str;
            }
            
            // Split by single character
            splitByChar(char) {
                return this.str.split(char);
            }
            
            // Split by multiple delimiters using regex
            splitByMultiple(delimiters) {
                const regex = new RegExp(`[${delimiters}]`);
                return this.str.split(regex);
            }
            
            // Split into words
            splitIntoWords() {
                return this.str.trim().split(/\s+/);
            }
            
            // Split into lines
            splitIntoLines() {
                return this.str.split(/\r?\n/);
            }
            
            // Split with limit
            splitWithLimit(separator, limit) {
                return this.str.split(separator, limit);
            }
            
            // Split CSV line
            splitCSV() {
                return this.str.split(',').map(item => item.trim());
            }
            
            // Split into character array
            splitIntoChars() {
                return this.str.split('');
            }
            
            // Parse URL query string
            parseQueryString() {
                return this.str
                    .split('&')
                    .reduce((params, pair) => {
                        const [key, value] = pair.split('=');
                        params[decodeURIComponent(key)] = decodeURIComponent(value || '');
                        return params;
                    }, {});
            }
        }
```

## js_element_createElement

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>createElement() Method</title>
</head>
<body>
    <div id="app"></div>
    
    <script>
        // PROBLEM: Create DOM elements dynamically
        // TIME: O(1) for creation, O(n) for appending SPACE: O(n)
        
        class DOMFactory {
            constructor(containerId) {
                this.container = document.getElementById(containerId);
            }
            
            // Create element with attributes
            createElement(tag, attributes = {}, textContent = '') {
                const element = document.createElement(tag);
                
                // Set attributes
                Object.entries(attributes).forEach(([key, value]) => {
                    if (key === 'className') {
                        element.className = value;
                    } else if (key === 'style' && typeof value === 'object') {
                        Object.assign(element.style, value);
                    } else if (key.startsWith('on') && typeof value === 'function') {
                        element.addEventListener(key.substring(2).toLowerCase(), value);
                    } else if (key === 'dataset' && typeof value === 'object') {
                        Object.entries(value).forEach(([dataKey, dataValue]) => {
                            element.dataset[dataKey] = dataValue;
                        });
                    } else {
                        element.setAttribute(key, value);
                    }
                });
                
                // Set text content
                if (textContent) {
                    element.textContent = textContent;
                }
                
                return element;
            }
            
            // Create element with children
            createElementWithChildren(tag, attributes, children) {
                const parent = this.createElement(tag, attributes);
                children.forEach(child => {
                    if (typeof child === 'string') {
                        parent.appendChild(document.createTextNode(child));
                    } else {
                        parent.appendChild(child);
                    }
                });
                return parent;
            }
            
            // Append to container
            append(element) {
                this.container.appendChild(element);
            }
            
            // Clear container
            clear() {
                this.container.innerHTML = '';
            }
        }
        
        // Usage demo
        const factory = new DOMFactory('app');
        
        // Create header
        const header = factory.createElement(
            'h1',
            { 
                className: 'main-header',
                style: { color: '#333', textAlign: 'center' }
            },
            'Dynamic Element Creation'
        );
        
        // Create card component
        const card = factory.createElementWithChildren(
            'div',
            {
                className: 'card',
                style: {
                    border: '1px solid #ddd',
                    borderRadius: '8px',
                    padding: '20px',
                    margin: '20px 0',
                    maxWidth: '600px'
                }
            },
            [
                factory.createElement('h2', {}, 'User Profile'),
                factory.createElement('p', {}, 'Name: John Doe'),
                factory.createElement('p', {}, 'Email: john@example.com'),
                factory.createElement(
                    'button',
                    {
                        className: 'btn',
                        style: {
                            background: '#007bff',
                            color: 'white',
                            border: 'none',
                            padding: '10px 20px',
                            borderRadius: '4px',
                            cursor: 'pointer'
                        },
                        onClick: () => alert('Button clicked!')
                    },
                    'Click Me'
                )
            ]
        );
        
        // Create list
        const list = factory.createElementWithChildren(
            'ul',
            { className: 'item-list' },
            ['Item 1', 'Item 2', 'Item 3'].map(item =>
                factory.createElement('li', {}, item)
            )
        );
        
        // Append all elements
        factory.append(header);
        factory.append(card);
        factory.append(list);
        
        // Display creation info
        const info = factory.createElement(
            'div',
            {
                style: {
                    background: '#e7f3ff',
                    padding: '15px',
                    marginTop: '20px',
                    borderRadius: '4px'
                }
            }
        );
        info.innerHTML = '<strong>Note:</strong> All elements above were created dynamically using <code>document.createElement()</code>';
        factory.append(info);
    </script>
</body>
```

**Why:** createElement() is the foundation of dynamic DOM manipulation, enabling programmatic creation of elements with full control over attributes, styles, and event handlers, safer than innerHTML for user-generated content.
