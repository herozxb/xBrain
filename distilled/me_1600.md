## 1. Maximum Sum Subarray of Size K
```python
# PROBLEM: Find maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)
def max_sum_subarray(arr, k):
    # Handle edge case: array smaller than window
    if len(arr) < k:
        return None
    
    # Calculate sum of first window
    window_sum = sum(arr[:k])
    max_sum = window_sum
    
    # Slide window: subtract leftmost, add new rightmost
    for i in range(len(arr) - k):
        window_sum = window_sum - arr[i] + arr[i + k]
        max_sum = max(max_sum, window_sum)
    
    return max_sum
```
**Why**: Classic sliding window - avoid recalculating entire window sum each time by reusing previous sum.

## 2. Container With Most Water
```python
# PROBLEM: Find two lines that form container holding most water
# TIME: O(n) SPACE: O(1)
def max_area(height):
    left, right = 0, len(height) - 1
    max_water = 0
    
    while left < right:
        # Width is distance between lines
        width = right - left
        # Height limited by shorter line
        h = min(height[left], height[right])
        max_water = max(max_water, width * h)
        
        # Move pointer at shorter line (greedy optimization)
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1
    
    return max_water
```
**Why**: Two pointers from ends inward - moving shorter line's pointer is only way to potentially increase area.

## 3. Generate All Subsets (Power Set)
```python
# PROBLEM: Generate all possible subsets of a set
# TIME: O(n * 2^n) SPACE: O(n) for recursion stack
def subsets(nums):
    result = []
    
    def backtrack(start, current):
        # Add copy of current subset to results
        result.append(current[:])
        
        # Try adding each remaining element
        for i in range(start, len(nums)):
            current.append(nums[i])  # choose
            backtrack(i + 1, current)  # explore
            current.pop()  # unchoose (backtrack)
    
    backtrack(0, [])
    return result
```
**Why**: Backtracking explores all combinations - at each step, decide include/exclude element, then undo choice.

## 4. Number of Connected Components
```python
# PROBLEM: Count connected components in undirected graph
# TIME: O(n * α(n)) ≈ O(n) SPACE: O(n)
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.components = n  # track number of components
    
    def find(self, x):
        # Path compression: point directly to root
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        px, py = self.find(x), self.find(y)
        if px == py:
            return  # already connected
        # Union by rank: attach smaller tree to larger
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        self.rank[px] += self.rank[px] == self.rank[py]
        self.components -= 1

def count_components(n, edges):
    uf = UnionFind(n)
    for u, v in edges:
        uf.union(u, v)
    return uf.components
```
**Why**: Union-Find efficiently tracks connectivity - path compression and union by rank keep operations nearly O(1).

## 5. Longest Common Subsequence
```python
# PROBLEM: Find length of longest common subsequence between two strings
# TIME: O(m * n) SPACE: O(m * n)
def lcs(text1, text2):
    m, n = len(text1), len(text2)
    # dp[i][j] = LCS length of text1[:i] and text2[:j]
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if text1[i-1] == text2[j-1]:
                # Characters match: extend LCS by 1
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                # Take max of excluding one character
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])
    
    return dp[m][n]
```
**Why**: Classic DP - build solution from smaller subproblems, each cell depends on left, top, and diagonal.

## 6. BFS Shortest Path in Grid
```python
# PROBLEM: Find shortest path in grid from start to target (avoiding obstacles)
# TIME: O(m * n) SPACE: O(m * n)
from collections import deque

def shortest_path(grid, start, target):
    if not grid or not grid[0]:
        return -1
    
    rows, cols = len(grid), len(grid[0])
    sr, sc = start
    tr, tc = target
    
    # BFS queue: (row, col, distance)
    queue = deque([(sr, sc, 0)])
    visited = {(sr, sc)}
    
    # Four directional moves
    directions = [(0, 1), (1, 0), (0, -1), (-1, 0)]
    
    while queue:
        r, c, dist = queue.popleft()
        
        # Found target
        if (r, c) == (tr, tc):
            return dist
        
        # Explore neighbors
        for dr, dc in directions:
            nr, nc = r + dr, c + dc
            # Check bounds, obstacle, and visited
            if (0 <= nr < rows and 0 <= nc < cols and 
                grid[nr][nc] == 0 and (nr, nc) not in visited):
                visited.add((nr, nc))
                queue.append((nr, nc, dist + 1))
    
    return -1  # No path found
```
**Why**: BFS guarantees shortest path in unweighted graphs - explores level by level.

## 7. Lowest Common Ancestor in BST
```python
# PROBLEM: Find lowest common ancestor of two nodes in BST
# TIME: O(h) where h = height SPACE: O(1)
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def lowest_common_ancestor(root, p, q):
    # Start from root
    current = root
    
    while current:
        # Both nodes in left subtree
        if p.val < current.val and q.val < current.val:
            current = current.left
        # Both nodes in right subtree
        elif p.val > current.val and q.val > current.val:
            current = current.right
        # Split point found: this is LCA
        else:
            return current
    
    return None
```
**Why**: BST property enables efficient search - values guide us directly to the split point.

## 8. Merge K Sorted Lists
```python
# PROBLEM: Merge k sorted linked lists into one sorted list
# TIME: O(n log k) where n = total nodes, k = lists SPACE: O(k)
import heapq

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next
    
    # For heap comparison
    def __lt__(self, other):
        return self.val < other.val

def merge_k_lists(lists):
    # Min-heap to always get smallest element
    heap = []
    
    # Add head of each list to heap
    for head in lists:
        if head:
            heapq.heappush(heap, head)
    
    dummy = ListNode(0)
    current = dummy
    
    while heap:
        # Get smallest node
        smallest = heapq.heappop(heap)
        current.next = smallest
        current = current.next
        
        # Add next node from same list if exists
        if smallest.next:
            heapq.heappush(heap, smallest.next)
    
    return dummy.next
```
**Why**: Heap efficiently tracks k minimums - each extraction gives next smallest element overall.

## 9. Word Search with Trie
```python
# PROBLEM: Find all words from dictionary in 2D board
# TIME: O(m * n * 4^L) where L = max word length SPACE: O(total chars in words)
class TrieNode:
    def __init__(self):
        self.children = {}
        self.word = None  # Stores complete word at end node

def find_words(board, words):
    # Build trie from word list
    root = TrieNode()
    for word in words:
        node = root
        for c in word:
            if c not in node.children:
                node.children[c] = TrieNode()
            node = node.children[c]
        node.word = word  # Mark end of word
    
    result = []
    rows, cols = len(board), len(board[0])
    
    def dfs(r, c, node):
        char = board[r][c]
        
        # No word starts with this character
        if char not in node.children:
            return
        
        node = node.children[char]
        
        # Found a complete word
        if node.word:
            result.append(node.word)
            node.word = None  # Avoid duplicates
        
        # Mark visited
        board[r][c] = '#'
        
        # Explore 4 directions
        for dr, dc in [(0, 1), (1, 0), (0, -1), (-1, 0)]:
            nr, nc = r + dr, c + dc
            if 0 <= nr < rows and 0 <= nc < cols:
                dfs(nr, nc, node)
        
        # Restore cell
        board[r][c] = char
    
    # Start DFS from each cell
    for r in range(rows):
        for c in range(cols):
            dfs(r, c, root)
    
    return result
```
**Why**: Trie prunes search early - only continue if prefix exists, avoiding unnecessary exploration.

## 10. LRU Cache Implementation
```python
# PROBLEM: Design cache with O(1) get and put, evicts least recently used
# TIME: O(1) for both operations SPACE: O(capacity)
class Node:
    def __init__(self, key=0, val=0):
        self.key = key
        self.val = val
        self.prev = None
        self.next = None

class LRUCache:
    def __init__(self, capacity):
        self.cap = capacity
        self.cache = {}  # key -> Node
        
        # Dummy head and tail for doubly linked list
        self.head = Node()
        self.tail = Node()
        self.head.next = self.tail
        self.tail.prev = self.head
    
    def _remove(self, node):
        # Remove node from list
        node.prev.next = node.next
        node.next.prev = node.prev
    
    def _add_to_front(self, node):
        # Add node right after head (most recent)
        node.prev = self.head
        node.next = self.head.next
        self.head.next.prev = node
        self.head.next = node
    
    def get(self, key):
        if key not in self.cache:
            return -1
        
        node = self.cache[key]
        # Move to front (mark as recently used)
        self._remove(node)
        self._add_to_front(node)
        return node.val
    
    def put(self, key, value):
        if key in self.cache:
            # Update existing and move to front
            node = self.cache[key]
            node.val = value
            self._remove(node)
            self._add_to_front(node)
        else:
            # Create new node
            node = Node(key, value)
            self.cache[key] = node
            self._add_to_front(node)
            
            # Evict LRU if over capacity
            if len(self.cache) > self.cap:
                lru = self.tail.prev
                self._remove(lru)
                del self.cache[lru.key]
```
**Why**: Hash map + doubly linked list gives O(1) access and ordering - map for lookup, list for recency tracking.
