### Spy Object

```python
# PROBLEM: Verify interactions without affecting behavior
# TIME: O(1) SPACE: O(n)
class SpyObject:
    """Tracks method calls while delegating to real object."""
    
    def __init__(self, target):
        self._target = target
        self._calls = []
    
    def __getattr__(self, name):
        attr = getattr(self._target, name)
        if callable(attr):
            def wrapper(*args, **kwargs):
                self._calls.append({
                    'method': name,
                    'args': args,
                    'kwargs': kwargs
                })
                return attr(*args, **kwargs)
            return wrapper
        return attr
    
    def was_called(self, method_name):
        """Check if method was called."""
        return any(c['method'] == method_name for c in self._calls)
    
    def get_call_count(self, method_name):
        """Get number of times method was called."""
        return sum(1 for c in self._calls if c['method'] == method_name)
    
    def get_calls(self):
        """Return all recorded calls."""
        return self._calls.copy()

# Usage
class EmailService:
    def send(self, to, subject, body):
        return f"Sent to {to}"

email = SpyObject(EmailService())
email.send("user@example.com", "Hello", "World")
assert email.was_called('send')
assert email.get_call_count('send') == 1
```

**Why**: Test behavior without side effects. Records all interactions while allowing real object to function normally. Unlike mocks, spies don't replace behavior - they just observe.

### Table Data Gateway

```python
# PROBLEM: Centralize all database access for a table
# TIME: O(1) for queries SPACE: O(n) for results
from typing import List, Dict, Any, Optional
import sqlite3

class TableDataGateway:
    """Gateway pattern: single entry point for table operations."""
    
    def __init__(self, connection):
        self.conn = connection
    
    def find_by_id(self, id: int) -> Optional[Dict[str, Any]]:
        """Find single record by ID."""
        cursor = self.conn.execute(
            "SELECT * FROM users WHERE id = ?",
            (id,)
        )
        row = cursor.fetchone()
        if row:
            columns = [desc[0] for desc in cursor.description]
            return dict(zip(columns, row))
        return None
    
    def find_all(self) -> List[Dict[str, Any]]:
        """Retrieve all records."""
        cursor = self.conn.execute("SELECT * FROM users")
        columns = [desc[0] for desc in cursor.description]
        return [dict(zip(columns, row)) for row in cursor.fetchall()]
    
    def find_by_criteria(self, **criteria) -> List[Dict[str, Any]]:
        """Find records matching criteria."""
        where_clauses = [f"{k} = ?" for k in criteria.keys()]
        sql = f"SELECT * FROM users WHERE {' AND '.join(where_clauses)}"
        cursor = self.conn.execute(sql, tuple(criteria.values()))
        columns = [desc[0] for desc in cursor.description]
        return [dict(zip(columns, row)) for row in cursor.fetchall()]
    
    def insert(self, **data) -> int:
        """Insert new record, return ID."""
        columns = ', '.join(data.keys())
        placeholders = ', '.join(['?'] * len(data))
        sql = f"INSERT INTO users ({columns}) VALUES ({placeholders})"
        cursor = self.conn.execute(sql, tuple(data.values()))
        self.conn.commit()
        return cursor.lastrowid
    
    def update(self, id: int, **data) -> bool:
        """Update record by ID."""
        set_clauses = [f"{k} = ?" for k in data.keys()]
        sql = f"UPDATE users SET {', '.join(set_clauses)} WHERE id = ?"
        cursor = self.conn.execute(sql, tuple(data.values()) + (id,))
        self.conn.commit()
        return cursor.rowcount > 0
    
    def delete(self, id: int) -> bool:
        """Delete record by ID."""
        cursor = self.conn.execute(
            "DELETE FROM users WHERE id = ?",
            (id,)
        )
        self.conn.commit()
        return cursor.rowcount > 0

# Usage
conn = sqlite3.connect(':memory:')
gateway = TableDataGateway(conn)
user_id = gateway.insert(name='Alice', email='alice@example.com')
user = gateway.find_by_id(user_id)
```

**Why**: Single point for all SQL. Simplifies business logic by hiding database details. Easy to optimize queries in one place. Works well with simple CRUD operations.

### Row Data Gateway

```python
# PROBLEM: One object per database row with data and access
# TIME: O(1) SPACE: O(n)
from typing import Optional
import sqlite3

class RowDataGateway:
    """Each row is an object with data + database operations."""
    
    def __init__(self, connection, id: Optional[int] = None, 
                 name: str = '', email: str = ''):
        self.conn = connection
        self.id = id
        self.name = name
        self.email = email
    
    def save(self) -> bool:
        """Insert or update the record."""
        if self.id is None:
            # Insert new record
            cursor = self.conn.execute(
                "INSERT INTO users (name, email) VALUES (?, ?)",
                (self.name, self.email)
            )
            self.id = cursor.lastrowid
        else:
            # Update existing
            cursor = self.conn.execute(
                "UPDATE users SET name = ?, email = ? WHERE id = ?",
                (self.name, self.email, self.id)
            )
        self.conn.commit()
        return True
    
    def delete(self) -> bool:
        """Delete this record from database."""
        if self.id is None:
            return False
        cursor = self.conn.execute(
            "DELETE FROM users WHERE id = ?",
            (self.id,)
        )
        self.conn.commit()
        return cursor.rowcount > 0
    
    @classmethod
    def find_by_id(cls, connection, id: int) -> Optional['RowDataGateway']:
        """Load a row by ID."""
        cursor = connection.execute(
            "SELECT id, name, email FROM users WHERE id = ?",
            (id,)
        )
        row = cursor.fetchone()
        if row:
            return cls(connection, id=row[0], name=row[1], email=row[2])
        return None
    
    @classmethod
    def find_all(cls, connection) -> list:
        """Load all rows."""
        cursor = connection.execute("SELECT id, name, email FROM users")
        return [
            cls(connection, id=row[0], name=row[1], email=row[2])
            for row in cursor.fetchall()
        ]
    
    def __repr__(self):
        return f"User(id={self.id}, name={self.name}, email={self.email})"

# Usage
conn = sqlite3.connect(':memory:')
user = RowDataGateway(conn, name='Bob', email='bob@example.com')
user.save()  # Inserts into DB
user.name = 'Robert'
user.save()  # Updates existing record
```

**Why**: Row-level operations feel natural. Object IS the row. Good for simple domain logic. Difference from Table Gateway: each object knows how to save itself.

```python
# TIME: O(1) SPACE: O(1)
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import parse_qs, urlparse
import json

class PageController(BaseHTTPRequestHandler):
    """One controller per page/view."""
    
    def do_GET(self):
        """Route GET requests to page handlers."""
        path = urlparse(self.path).path
        
        if path == '/':
            self.home_page()
        elif path == '/about':
            self.about_page()
        elif path == '/contact':
            self.contact_page()
        else:
            self.not_found_page()
    
    def do_POST(self):
        """Route POST requests."""
        path = urlparse(self.path).path
        
        if path == '/contact':
            self.handle_contact_form()
        else:
            self.send_error(404)
    
    def home_page(self):
        """Render home page."""
        self.send_response(200)
        self.send_header('Content-type', 'text/html')
        self.end_headers()
        html = """
        <html>
        <body>
            <h1>Welcome Home</h1>
            <nav>
                <a href="/">Home</a> |
                <a href="/about">About</a> |
                <a href="/contact">Contact</a>
            </nav>
        </body>
        </html>
        """
        self.wfile.write(html.encode())
    
    def about_page(self):
        """Render about page."""
        self.send_response(200)
        self.send_header('Content-type', 'text/html')
        self.end_headers()
        html = "<html><body><h1>About Us</h1><p>This is our story.</p></body></html>"
        self.wfile.write(html.encode())
    
    def contact_page(self):
        """Render contact form."""
        self.send_response(200)
        self.send_header('Content-type', 'text/html')
        self.end_headers()
        html = """
        <html><body>
            <h1>Contact Us</h1>
            <form method="POST">
                <input name="email" type="email" placeholder="Your email">
                <textarea name="message" placeholder="Message"></textarea>
                <button type="submit">Send</button>
            </form>
        </body></html>
        """
        self.wfile.write(html.encode())
    
    def handle_contact_form(self):
        """Process contact form submission."""
        content_length = int(self.headers['Content-Length'])
        post_data = self.rfile.read(content_length)
        data = parse_qs(post_data.decode())
        
        email = data.get('email', [''])[0]
        message = data.get('message', [''])[0]
        
        # Process form (send email, save to DB, etc.)
        print(f"Contact from {email}: {message}")
        
        # Redirect or show success
        self.send_response(302)
        self.send_header('Location', '/contact?success=1')
        self.end_headers()
    
    def not_found_page(self):
        """404 page."""
        self.send_response(404)
        self.send_header('Content-type', 'text/html')
        self.end_headers()
        html = "<html><body><h1>404 Not Found</h1></body></html>"
        self.wfile.write(html.encode())

# Usage
# server = HTTPServer(('localhost', 8080), PageController)
# server.serve_forever()
```

**Why**: Simple routing. Each page has dedicated handler. Easy to understand and maintain. Best for small applications with straightforward navigation.

### Application Controller

```python
# PROBLEM: Centralize request routing and coordination
# TIME: O(1) SPACE: O(n) for routes
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import urlparse
from typing import Callable, Dict

class ApplicationController:
    """Central router: coordinates all request handling."""
    
    def __init__(self):
        self.routes: Dict[str, Dict[str, Callable]] = {
            'GET': {},
            'POST': {},
            'PUT': {},
            'DELETE': {}
        }
        self.before_filters = []
        self.after_filters = []
    
    def route(self, method: str, path: str):
        """Decorator to register a route handler."""
        def decorator(handler):
            self.routes[method][path] = handler
            return handler
        return decorator
    
    def before_request(self, filter_func):
        """Register pre-processing filter."""
        self.before_filters.append(filter_func)
        return filter_func
    
    def after_request(self, filter_func):
        """Register post-processing filter."""
        self.after_filters.append(filter_func)
        return filter_func
    
    def handle_request(self, method: str, path: str, request, response):
        """Process request through filters and handler."""
        parsed_path = urlparse(path).path
        
        # Run before filters
        for filter_func in self.before_filters:
            result = filter_func(request, response)
            if result is False:
                return None
        
        # Find and execute handler
        handler = self.routes.get(method, {}).get(parsed_path)
        if not handler:
            response['status'] = 404
            response['body'] = 'Not Found'
            return response
        
        result = handler(request, response)
        
        # Run after filters
        for filter_func in self.after_filters:
            filter_func(request, response)
        
        return result

# Usage
app = ApplicationController()

@app.before_request
def authenticate(request, response):
    """Check authentication before all requests."""
    token = request.get('headers', {}).get('Authorization')
    if not token:
        response['status'] = 401
        response['body'] = 'Unauthorized'
        return False
    request['user'] = {'id': 1, 'name': 'Alice'}

@app.after_request
def log_request(request, response):
    """Log all requests."""
    print(f"{request['method']} {request['path']} -> {response['status']}")

@app.route('GET', '/api/users')
def get_users(request, response):
    response['status'] = 200
    response['body'] = [{'id': 1, 'name': 'Alice'}]
    return response

@app.route('POST', '/api/users')
def create_user(request, response):
    response['status'] = 201
    response['body'] = {'id': 2, 'name': 'Bob'}
    return response

# Handle request
request = {
    'method': 'GET',
    'path': '/api/users',
    'headers': {'Authorization': 'token123'}
}
response = {}
result = app.handle_request('GET', '/api/users', request, response)
```

**Why**: Single point for routing, auth, logging. Centralized control flow. Easy to add cross-cutting concerns. Scales better than Page Controller for complex apps.

### View Helper

```python
# PROBLEM: Simplify complex view logic without cluttering templates
# TIME: O(1) or O(n) SPACE: O(n)
from datetime import datetime
from typing import List, Any
import html

class ViewHelper:
    """Utility methods for view rendering."""
    
    @staticmethod
    def format_currency(amount: float, currency: str = 'USD') -> str:
        """Format number as currency."""
        symbols = {'USD': '$', 'EUR': '€', 'GBP': '£'}
        symbol = symbols.get(currency, currency)
        return f"{symbol}{amount:,.2f}"
    
    @staticmethod
    def format_date(date: datetime, format_str: str = '%Y-%m-%d') -> str:
        """Format date for display."""
        return date.strftime(format_str)
    
    @staticmethod
    def truncate(text: str, length: int = 100, suffix: str = '...') -> str:
        """Truncate text to specified length."""
        if len(text) <= length:
            return text
        return text[:length - len(suffix)] + suffix
    
    @staticmethod
    def escape_html(text: str) -> str:
        """Escape HTML special characters."""
        return html.escape(text)
    
    @staticmethod
    def pluralize(count: int, singular: str, plural: str = None) -> str:
        """Return singular or plural form based on count."""
        if count == 1:
            return singular
        return plural or singular + 's'
    
    @staticmethod
    def join_list(items: List[Any], separator: str = ', ', 
                  final_separator: str = ' and ') -> str:
        """Join list with natural language separators."""
        if not items:
            return ''
        if len(items) == 1:
            return str(items[0])
        return separator.join(str(i) for i in items[:-1]) + final_separator + str(items[-1])
    
    @staticmethod
    def time_ago(date: datetime) -> str:
        """Human-readable time difference."""
        now = datetime.now()
        diff = now - date
        
        seconds = int(diff.total_seconds())
        if seconds < 60:
            return f"{seconds} {ViewHelper.pluralize(seconds, 'second')} ago"
        
        minutes = seconds // 60
        if minutes < 60:
            return f"{minutes} {ViewHelper.pluralize(minutes, 'minute')} ago"
        
        hours = minutes // 60
        if hours < 24:
            return f"{hours} {ViewHelper.pluralize(hours, 'hour')} ago"
        
        days = hours // 24
        return f"{days} {ViewHelper.pluralize(days, 'day')} ago"
    
    @staticmethod
    def highlight_text(text: str, search: str, tag: str = 'mark') -> str:
        """Highlight search terms in text."""
        if not search:
            return ViewHelper.escape_html(text)
        
        escaped_text = ViewHelper.escape_html(text)
        escaped_search = ViewHelper.escape_html(search)
        
        return escaped_text.replace(
            escaped_search,
            f'<{tag}>{escaped_search}</{tag}>'
        )

# Usage in template
helper = ViewHelper()

# Format currency
price = 1234.56
print(helper.format_currency(price))  # $1,234.56

# Truncate long text
long_text = "This is a very long text that needs to be truncated for display"
print(helper.truncate(long_text, 30))  # This is a very long text t...

# Time ago
posted = datetime(2026, 2, 19, 10, 0)
print(helper.time_ago(posted))  # 1 day ago

# Natural join
tags = ['python', 'django', 'flask']
print(helper.join_list(tags))  # python, django and flask

# Highlight search
result = helper.highlight_text("Python is awesome", "Python")
# <mark>Python</mark> is awesome
```

**Why**: Keep templates clean. Reusable formatting logic. Test helper methods separately. Common in Rails/Django templates.

```python
# PROBLEM: Build pages from reusable, composable view components
# TIME: O(n) SPACE: O(n)
from typing import List, Dict

class ViewComponent:
    """Base component for composable views."""
    
    def __init__(self, template: str = '', **context):
        self.template = template
        self.context = context
        self.children: List['ViewComponent'] = []
    
    def add_child(self, component: 'ViewComponent'):
        """Add child component."""
        self.children.append(component)
        return self
    
    def render(self) -> str:
        """Render this component and all children."""
        output = self._render_self()
        
        # Render children in designated slots
        for i, child in enumerate(self.children):
            slot_marker = f"{{{{slot_{i}}}}}"
            output = output.replace(slot_marker, child.render())
        
        return output
    
    def _render_self(self) -> str:
        """Render this component's template with context."""
        output = self.template
        for key, value in self.context.items():
            output = output.replace(f'{{{{{key}}}}}', str(value))
        return output

class LayoutComponent(ViewComponent):
```

### Intercepting Filter

```python
# PROBLEM: Pre/post-process requests through chain of filters
# TIME: O(n) for n filters SPACE: O(n)
from typing import List, Callable, Dict, Any

class Filter:
    """Base filter interface."""
    
    def execute(self, request: Dict[str, Any], response: Dict[str, Any]) -> bool:
        """
        Process request/response.
        Return True to continue chain, False to stop.
        """
        return True

class InterceptingFilter:
    """Manages chain of filters for request processing."""
    
    def __init__(self):
        self.pre_filters: List[Filter] = []
        self.post_filters: List[Filter] = []
    
    def add_pre_filter(self, filter: Filter):
        """Add filter to execute before handler."""
        self.pre_filters.append(filter)
        return self
    
    def add_post_filter(self, filter: Filter):
        """Add filter to execute after handler."""
        self.post_filters.append(filter)
        return self
    
    def process(self, request: Dict[str, Any], handler: Callable) -> Dict[str, Any]:
        """Process request through filter chain."""
        response = {'status': 200, 'body': ''}
        
        # Execute pre-filters
        for filter in self.pre_filters:
            if not filter.execute(request, response):
                return response
        
        # Execute main handler
        handler(request, response)
        
        # Execute post-filters (in reverse order)
        for filter in reversed(self.post_filters):
            if not filter.execute(request, response):
                break
        
        return response

# Concrete filters
class AuthenticationFilter(Filter):
    """Check user authentication."""
    
    def execute(self, request, response):
        token = request.get('headers', {}).get('Authorization')
        if not token:
            response['status'] = 401
            response['body'] = 'Unauthorized'
            return False
        request['user'] = {'id': 1, 'name': 'Alice'}
        return True

class LoggingFilter(Filter):
    """Log request details."""
    
    def execute(self, request, response):
        print(f"[LOG] {request['method']} {request['path']}")
        return True

class RateLimitFilter(Filter):
    """Rate limit requests."""
    
    def __init__(self, max_requests: int = 100):
        self.max_requests = max_requests
        self.request_counts: Dict[str, int] = {}
    
    def execute(self, request, response):
        ip = request.get('ip', 'unknown')
        count = self.request_counts.get(ip, 0)
        
        if count >= self.max_requests:
            response['status'] = 429
            response['body'] = 'Too Many Requests'
            return False
        
        self.request_counts[ip] = count + 1
        return True

class CompressionFilter(Filter):
    """Compress response (post-filter)."""
    
    def execute(self, request, response):
        # Simulated compression
        if 'gzip' in request.get('headers', {}).get('Accept-Encoding', ''):
            response['compressed'] = True
            print("[FILTER] Response compressed")
        return True

class CORFilter(Filter):
    """Add CORS headers (post-filter)."""
    
    def execute(self, request, response):
        headers = response.setdefault('headers', {})
        headers['Access-Control-Allow-Origin'] = '*'
        headers['Access-Control-Allow-Methods'] = 'GET, POST, PUT, DELETE'
        return True

# Usage
filter_chain = InterceptingFilter()
filter_chain.add_pre_filter(LoggingFilter())
filter_chain.add_pre_filter(RateLimitFilter(max_requests=5))
filter_chain.add_pre_filter(AuthenticationFilter())
filter_chain.add_post_filter(CompressionFilter())
filter_chain.add_post_filter(CORFilter())

def api_handler(request, response):
    """Main request handler."""
    response['status'] = 200
    response['body'] = f"Hello, {request['user']['name']}!"

# Process request
request = {
    'method': 'GET',
    'path': '/api/data',
    'ip': '192.168.1.1',
    'headers': {
        'Authorization': 'token123',
        'Accept-Encoding': 'gzip'
    }
}

result = filter_chain.process(request, api_handler)
print(result)
# {'status': 200, 'body': 'Hello, Alice!', 'compressed': True, 
#  'headers': {'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': '...'}}
```

**Why**: Cross-cutting concerns in reusable filters. Chain of responsibility. Easy to add/remove filters. Common in Java EE, Express.js middleware.

### Chain of Processors

```python
# PROBLEM: Process data through series of independent processors
# TIME: O(n) for n processors SPACE: O(1) or O(n)
from typing import Any, List, Optional
from abc import ABC, abstractmethod

class Processor(ABC):
    """Abstract processor in the chain."""
    
    def __init__(self):
        self.next_processor: Optional[Processor] = None
    
    def set_next(self, processor: 'Processor') -> 'Processor':
        """Set next processor in chain."""
        self.next_processor = processor
        return processor
    
    def process(self, data: Any) -> Any:
        """Process data and pass to next processor."""
        result = self._do_process(data)
        
        if self.next_processor:
            return self.next_processor.process(result)
        
        return result
    
    @abstractmethod
    def _do_process(self, data: Any) -> Any:
        """Implement actual processing logic."""
        pass

# Concrete processors for data pipeline
class ValidationProcessor(Processor):
    """Validate input data."""
    
    def _do_process(self, data):
        if not isinstance(data, dict):
            raise ValueError("Data must be a dictionary")
        
        required_fields = ['name', 'email']
        for field in required_fields:
            if field not in data:

        print(f"[Validation] Validated: {data}")

class CachingProcessor(Processor):
    """Cache processed results."""
    
    def __init__(self):
        super().__init__()
        self.cache = {}
    
    def _do_process(self, data):
        # Simple cache key based on data
        cache_key = hash(str(data))
        
        if cache_key in self.cache:
            print(f"[Caching] Cache hit!")
            return self.cache[cache_key]
        
        # Process and cache
        result = data
        if self.next_processor:
            result = self.next_processor.process(data)
        
        self.cache[cache_key] = result
        print(f"[Caching] Cached result")
        return None  # Return None to stop chain (we handled it)

class ProcessorChain:
    """Builder for processor chains."""
    
    def __init__(self):
        self.processors: List[Processor] = []
    
    def add(self, processor: Processor) -> 'ProcessorChain':
        """Add processor to chain."""
        if self.processors:
            self.processors[-1].set_next(processor)
        self.processors.append(processor)
        return self
    
    def process(self, data: Any) -> Any:
        """Start processing from first processor."""
        if not self.processors:
            return data
        return self.processors[0].process(data)
```

```python
from typing import Any, Optional, Callable
from datetime import datetime

class MemcachedClient:
    """Simplified Memcached client with common patterns."""
    
    def __init__(self):
        # In production, use: pymemcache.client.base.Client
        self.cache = {}
        self.stats = {'hits': 0, 'misses': 0}
    
    def get(self, key: str) -> Optional[Any]:
        """Get value from cache."""
        if key in self.cache:
            value, expires_at = self.cache[key]
            if expires_at and datetime.now() > expires_at:
                del self.cache[key]
                self.stats['misses'] += 1
                return None
            self.stats['hits'] += 1
            return value
        self.stats['misses'] += 1
        return None
    
    def set(self, key: str, value: Any, ttl: int = 3600):
        """Set value in cache with TTL."""
        from datetime import timedelta
        expires_at = datetime.now() + timedelta(seconds=ttl)
        self.cache[key] = (value, expires_at)
    
    def delete(self, key: str):
        """Delete key from cache."""
        if key in self.cache:
            del self.cache[key]
    
    def get_or_set(self, key: str, callback: Callable, ttl: int = 3600) -> Any:
        """Cache-aside pattern: get or compute and cache."""
        value = self.get(key)
        if value is not None:
            return value
        
        value = callback()
        self.set(key, value, ttl)
        return value

        if cached:
            value, expires_at = cached
            time_until_expiry = (expires_at - datetime.now()).total_seconds()
```
