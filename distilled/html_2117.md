## 1. html_track_kind_metadata

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Track Kind Metadata</title>
</head>
<body>
  <!-- 
    PROBLEM: Add metadata tracks to video elements for chapters, subtitles, and descriptions
    TIME: O(1) SPACE: O(1)
  -->
  <!-- 
    Why: The <track> element with kind="metadata" provides chapter markers and 
    supplementary data without visible text tracks, useful for navigation and analytics.
  -->
  
  <video id="myVideo" controls width="640" height="360" crossorigin="anonymous">
    <source src="video.mp4" type="video/mp4">
    
    <!-- Visible tracks -->
    <track kind="subtitles" src="subs_en.vtt" srclang="en" label="English">
    <track kind="captions" src="caps_en.vtt" srclang="en" label="English Captions">
    
    <!-- Metadata track - invisible but accessible via JavaScript -->
    <track kind="metadata" src="chapters.vtt" srclang="en" label="Chapters" default>
  </video>
  
  <div id="chapterList"></div>

  <script>
    // Access metadata track and extract chapter information
    const video = document.getElementById('myVideo');
    const chapterList = document.getElementById('chapterList');
    
    video.addEventListener('loadedmetadata', () => {
      // Find the metadata track
      const tracks = video.textTracks;
      let metadataTrack = null;
      
      for (let i = 0; i < tracks.length; i++) {
        if (tracks[i].kind === 'metadata') {
          metadataTrack = tracks[i];
          break;
        }
      }
      
      if (metadataTrack) {
        // Set mode to hidden to load cues (metadata is hidden by default)
        metadataTrack.mode = 'hidden';
        
        metadataTrack.addEventListener('cuechange', () => {
          const activeCues = metadataTrack.activeCues;
          if (activeCues.length > 0) {
            console.log('Current chapter:', activeCues[0].text);
          }
        });
        
        // Build chapter navigation
        metadataTrack.addEventListener('load', () => {
          const cues = metadataTrack.cues;
          const ul = document.createElement('ul');
          
          for (let i = 0; i < cues.length; i++) {
            const cue = cues[i];
            const li = document.createElement('li');
            const a = document.createElement('a');
            a.href = '#';
            a.textContent = cue.text;
            a.onclick = (e) => {
              e.preventDefault();
              video.currentTime = cue.startTime;
            };
            li.appendChild(a);
            ul.appendChild(li);
          }
          chapterList.appendChild(ul);
        });
      }
    });
  </script>
</body>
</html>
```

```html
<!DOCTYPE html>
<html lang="en">

  <!-- 
    PROBLEM: Load different image sources based on viewport conditions
    TIME: O(1) SPACE: O(1)
  -->
  <!-- 
    Why: Using media attributes on <source> enables art direction and performance 
    optimization by loading appropriate images for each device context.
  -->
  
  <h1>Responsive Image with Media Queries</h1>
  
  <picture>
    <!-- Large screens: landscape hero image -->
    <source 
      media="(min-width: 1200px)"
      srcset="hero-large.jpg 1x, hero-large-2x.jpg 2x"
      type="image/jpeg">
    
    <!-- Medium screens: cropped version -->
    <source 
      media="(min-width: 768px) and (max-width: 1199px)"
      srcset="hero-medium.jpg 1x, hero-medium-2x.jpg 2x"
      type="image/jpeg">
    
    <!-- Small screens: portrait focused -->
    <source 
      media="(max-width: 767px)"
      srcset="hero-small.jpg 1x, hero-small-2x.jpg 2x"
      type="image/jpeg">
    
    <!-- Fallback for older browsers -->
    <img src="hero-fallback.jpg" alt="Hero image" loading="lazy">
  </picture>

  <script>
    // Detect which source was loaded
    const picture = document.querySelector('picture');
    const img = picture.querySelector('img');
    
    function getCurrentBreakpoint() {
      const width = window.innerWidth;
      if (width >= 1200) return 'large';
      if (width >= 768) return 'medium';
      return 'small';
    }
    
    // Log current image source on load
    img.addEventListener('load', () => {
      console.log(`Loaded image for ${getCurrentBreakpoint()} viewport`);
      console.log('Current src:', img.currentSrc);
    });
    
    // Track responsive changes
    let currentBreakpoint = getCurrentBreakpoint();
    
    window.addEventListener('resize', () => {
      const newBreakpoint = getCurrentBreakpoint();
      if (newBreakpoint !== currentBreakpoint) {
        currentBreakpoint = newBreakpoint;
        console.log(`Breakpoint changed to: ${newBreakpoint}`);
      }
    });
  </script>
</body>
</html>
```

```html
    const landscapeQuery = window.matchMedia('(orientation: landscape)');
```

```html
  <table class="layout-auto" id="autoTable">

    </thead>
    <tbody>
      <tr>
        <td>1</td>
        <td>This is a very long description that will expand the column</td>
        <td>Active</td>
      </tr>

      </tr>

  <script>
    // Performance comparison utility
    function measureTableRender(tableEl, rowCount) {
      const start = performance.now();
      
      const tbody = tableEl.querySelector('tbody');
      tbody.innerHTML = '';
      
      for (let i = 0; i < rowCount; i++) {
        const row = document.createElement('tr');
        row.innerHTML = `
          <td>${i + 1}</td>
          <td>Item description ${i + 1}</td>
          <td>${i % 2 === 0 ? 'Active' : 'Pending'}</td>
        `;
        tbody.appendChild(row);
      }
      
      // Force reflow
      void tableEl.offsetHeight;
      
      const end = performance.now();
      return end - start;
    }
    
    // Compare rendering times
    const fixedTime = measureTableRender(document.getElementById('fixedTable'), 1000);
    const autoTime = measureTableRender(document.getElementById('autoTable'), 1000);
    
    console.log(`Fixed layout: ${fixedTime.toFixed(2)}ms`);
    console.log(`Auto layout: ${autoTime.toFixed(2)}ms`);
    console.log(`Fixed is ${(autoTime / fixedTime).toFixed(1)}x faster`);
    
    // Toggle between layouts
    function toggleTableLayout(tableEl) {
      const current = getComputedStyle(tableEl).tableLayout;
      tableEl.style.tableLayout = current === 'fixed' ? 'auto' : 'fixed';
      console.log(`Changed to: ${tableEl.style.tableLayout}`);
    }
  </script>
</body>
</html>
```

```html
<body>
  <!-- 
    PROBLEM: Create accessible horizontal navigation menus with keyboard support
    TIME: O(n) SPACE: O(1)
  -->
  <!-- 
    Why: role="menubar" indicates a horizontal menu bar to screen readers, 
    enabling proper keyboard navigation and ARIA patterns for desktop-like menus.
  -->
  
  <nav aria-label="Main navigation">
    <ul role="menubar" id="mainMenubar">
      <li role="none">
        <span role="menuitem" tabindex="0" aria-haspopup="true" aria-expanded="false">
          File
        </span>
        <ul role="menu" aria-label="File menu">
          <li role="none">
            <span role="menuitem" tabindex="-1">New</span>
          </li>
          <li role="none">
            <span role="menuitem" tabindex="-1">Open</span>
          </li>

          </li>
          <li role="none">

          </li>
        </ul>
      </li>
      <li role="none">
        <span role="menuitem" tabindex="0" aria-haspopup="true" aria-expanded="false">
          Edit
        </span>
        <ul role="menu" aria-label="Edit menu">
          <li role="none">

          </li>

            <span role="menuitem" tabindex="-1">Cut</span>
          </li>
          <li role="none">
            <span role="menuitem" tabindex="-1">Copy</span>

        </ul>
      </li>
      <li role="none">
        <span role="menuitem" tabindex="0">Help</span>
      </li>
    </ul>
  </nav>

  <script>
    const menubar = document.getElementById('mainMenubar');
    const topLevelItems = menubar.querySelectorAll(':scope > [role="none"] > [role="menuitem"]');
    let currentIndex = 0;
    let openMenu = null;
    
    // Open submenu
    function openSubMenu(menuItem) {
      const submenu = menuItem.nextElementSibling;
      if (submenu && submenu.getAttribute('role') === 'menu') {
        submenu.classList.add('visible');
        menuItem.setAttribute('aria-expanded', 'true');
        openMenu = { parent: menuItem, menu: submenu };
        
        // Focus first item
        const firstItem = submenu.querySelector('[role="menuitem"]');
        if (firstItem) firstItem.focus();
      }
    }
    
    // Close submenu
    function closeSubMenu() {
      if (openMenu) {
        openMenu.menu.classList.remove('visible');
        openMenu.parent.setAttribute('aria-expanded', 'false');
        openMenu.parent.focus();
        openMenu = null;
      }
    }
    
    // Top-level keyboard navigation
    topLevelItems.forEach((item, index) => {
      item.addEventListener('keydown', (e) => {
        switch (e.key) {
          case 'ArrowRight':
            e.preventDefault();
            currentIndex = (index + 1) % topLevelItems.length;
            topLevelItems[currentIndex].focus();
            closeSubMenu();
            break;
          case 'ArrowLeft':
            e.preventDefault();
            currentIndex = (index - 1 + topLevelItems.length) % topLevelItems.length;
            topLevelItems[currentIndex].focus();
            closeSubMenu();
            break;
          case 'ArrowDown':
          case 'Enter':
          case ' ':
            e.preventDefault();
            if (item.getAttribute('aria-haspopup') === 'true') {
              openSubMenu(item);
            }
            break;
          case 'Escape':
            closeSubMenu();
            break;
        }
      });
      
      item.addEventListener('click', () => {
        if (item.getAttribute('aria-haspopup') === 'true') {
          if (openMenu && openMenu.parent === item) {
            closeSubMenu();
          } else {
            closeSubMenu();
            openSubMenu(item);
          }
        }
      });
    });
    
    // Submenu keyboard navigation
    menubar.addEventListener('keydown', (e) => {
      if (!openMenu) return;
      
      const menuItems = openMenu.menu.querySelectorAll('[role="menuitem"]');
      const focusedIndex = Array.from(menuItems).indexOf(document.activeElement);
      
      switch (e.key) {
        case 'ArrowDown':
          e.preventDefault();
          const nextIndex = (focusedIndex + 1) % menuItems.length;
          menuItems[nextIndex].focus();
          break;
        case 'ArrowUp':
          e.preventDefault();
          const prevIndex = (focusedIndex - 1 + menuItems.length) % menuItems.length;
          menuItems[prevIndex].focus();
          break;
        case 'Escape':
          closeSubMenu();
          break;
      }
    });
    
    // Close on outside click
    document.addEventListener('click', (e) => {
      if (!menubar.contains(e.target)) {
        closeSubMenu();
      }
    });
  </script>
</body>
</html>
```

```html
body>
  <!-- 
    PROBLEM: Indicate selected state in selectable widgets for assistive technologies
    TIME: O(n) SPACE: O(1)
  -->
  <!-- 
    Why: aria-selected communicates selection state to screen readers, essential 
    for listboxes, tabs, grids, and tree views where selection matters.
  -->
  
  <h2>Select Fruits (Multi-select)</h2>
  <p>Hold Ctrl/Cmd for multiple selections</p>
  
  <ul role="listbox" id="fruitListbox" aria-label="Fruits" aria-multiselectable="true" tabindex="0">
    <li role="option" id="opt-apple" aria-selected="false">Apple</li>
    <li role="option" id="opt-banana" aria-selected="false">Banana</li>
    <li role="option" id="opt-cherry" aria-selected="false">Cherry</li>
    <li role="option" id="opt-date" aria-selected="false">Date</li>
    <li role="option" id="opt-elderberry" aria-selected="false">Elderberry</li>
    <li role="option" id="opt-fig" aria-selected="false">Fig</li>
    <li role="option" id="opt-grape" aria-selected="false">Grape</li>
  </ul>
  
  <div class="selected-count" id="selectedCount">Selected: 0 items</div>
  <div id="selectedItems"></div>

  <script>
    const listbox = document.getElementById('fruitListbox');
    const options = listbox.querySelectorAll('[role="option"]');
    let focusedIndex = 0;
    
    // Update selection display
    function updateSelectionDisplay() {
      const selected = Array.from(options).filter(opt => 
        opt.getAttribute('aria-selected') === 'true'
      );
      
      document.getElementById('selectedCount').textContent = 
        `Selected: ${selected.length} item${selected.length !== 1 ? 's' : ''}`;
      
      document.getElementById('selectedItems').textContent = 
        selected.length > 0 
          ? `Items: ${selected.map(opt => opt.textContent).join(', ')}` 
          : '';
    }
    
    // Toggle selection
    function toggleOption(option, forceState = null) {
      const currentState = option.getAttribute('aria-selected') === 'true';
      const newState = forceState !== null ? forceState : !currentState;
      
      option.setAttribute('aria-selected', String(newState));
      updateSelectionDisplay();
    }
    
    // Select range (shift+click)
    function selectRange(startIndex, endIndex) {
      const start = Math.min(startIndex, endIndex);
      const end = Math.max(startIndex, endIndex);
      
      for (let i = start; i <= end; i++) {
        options[i].setAttribute('aria-selected', 'true');
      }
      updateSelectionDisplay();
    }
    
    let lastClickedIndex = 0;
    
    // Click handling
    options.forEach((option, index) => {
      option.addEventListener('click', (e) => {
        if (e.shiftKey) {
          // Range selection
          selectRange(lastClickedIndex, index);
        } else if (e.ctrlKey || e.metaKey) {
          // Toggle individual
          toggleOption(option);
        } else {
          // Single selection
          options.forEach(opt => opt.setAttribute('aria-selected', 'false'));
          toggleOption(option, true);
        }
        
        lastClickedIndex = index;
        option.focus();
      });
    });
    
    // Keyboard navigation
    listbox.addEventListener('keydown', (e) => {
      const currentOption = options[focusedIndex];
      
      switch (e.key) {
        case 'ArrowDown':
          e.preventDefault();
          focusedIndex = Math.min(focusedIndex + 1, options.length - 1);
          options[focusedIndex].focus();
          break;
          
        case 'ArrowUp':
          e.preventDefault();
          focusedIndex = Math.max(focusedIndex - 1, 0);
          options[focusedIndex].focus();
          break;
          
        case ' ':
        case 'Enter':
          e.preventDefault();
          if (e.ctrlKey || e.metaKey) {
            toggleOption(currentOption);
          } else {
            options.forEach(opt => opt.setAttribute('aria-selected', 'false'));
            toggleOption(currentOption, true);
          }
          break;
          
        case 'a':
          if (e.ctrlKey || e.metaKey) {
            e.preventDefault();
            // Select all
            options.forEach(opt => opt.setAttribute('aria-selected', 'true'));
            updateSelectionDisplay();
          }
          break;
          
        case 'Escape':
          // Clear selection
          options.forEach(opt => opt.setAttribute('aria-selected', 'false'));
          updateSelectionDisplay();
          break;
      }
    });
    
    // Initial focus
    options[0].setAttribute('tabindex', '0');
  </script>
</body>
</html>
```

```html
    <div class="menu-item" role="menuitem" tabindex="0">New Document</div>

    // Handle aria-disabled clicks
    document.querySelectorAll('[aria-disabled]').forEach(el => {
      el.addEventListener('click', (e) => {
        if (el.getAttribute('aria-disabled') === 'true') {
          e.preventDefault();
          e.stopPropagation();

          const status = document.getElementById('status');
          status.textContent = `"${el.textContent.trim()}" is currently unavailable`;
```

```html
    const liveAnnouncer = document.getElementById('liveAnnouncer');
    const progressFill = document.getElementById('progressFill');
    const progressStatus = document.getElementById('progressStatus');
    const notificationArea = document.getElementById('notificationArea');
    
    // Announce message to screen readers
    function announce(message, priority = 'polite') {
      liveAnnouncer.setAttribute('aria-live', priority);
      liveAnnouncer.textContent = '';
      
      // Small delay ensures the change is detected
      setTimeout(() => {
        liveAnnouncer.textContent = message;
      }, 100);
    }
    
    // Simulated upload progress
    let progress = 0;
    let uploadInterval = null;
    
    document.getElementById('startUpload').addEventListener('click', function() {
      if (uploadInterval) return;
      
      progress = 0;
      announce('Upload started');
      progressStatus.textContent = 'Uploading...';
      
      uploadInterval = setInterval(() => {
        progress += Math.random() * 15;
        
        if (progress >= 100) {
          progress = 100;
          clearInterval(uploadInterval);
          uploadInterval = null;
          
          progressFill.style.width = '100%';
          progressStatus.textContent = 'Upload complete!';
          announce('Upload complete', 'assertive');
        } else {
          progressFill.style.width = `${progress}%`;
          
          // Announce at 25%, 50%, 75%
          const percent = Math.floor(progress / 25) * 25;
          if (progress % 25 < 15) {
            announce(`${percent}% uploaded`);
          }
        }
      }, 500);
    });
    
    // Add notifications
    let notificationCount = 0;
    const notificationTypes = [
      { type: 'success', message: 'File saved successfully' },
      { type: 'error', message: 'Connection lost, retrying...' },

      notification.textContent = `${notificationCount}. ${template.message}`;
      
      notificationArea.appendChild(notification);
      
      // Keep only last 5 notifications
      while (notificationArea.children.length > 5) {
        notificationArea.removeChild(notificationArea.firstChild);
      }
      
      // Screen readers will announce thanks to aria-live on notificationArea
    });
    
    // Example: Form validation feedback

      const errors = [];

      if (!name) {
        errors.push('Name is required');

      } else {
        announce('Form is valid, ready to submit');
      }
    }
```
