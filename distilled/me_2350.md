# PROBLEM: Find all occurrences of patterns in text using Trie
# TIME: O(n + m * k) where n=text length, m=number of patterns, k=avg pattern length
# SPACE: O(alphabet_size * total_pattern_length)
class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_end = False
        self.pattern_index = -1

def build_trie(patterns):
    """Build a trie from a list of patterns."""
    root = TrieNode()
    for i, pattern in enumerate(patterns):
        node = root
        for char in pattern:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
        node.pattern_index = i
    return root

def search_patterns(text, patterns):
    """Search for all patterns in text using trie."""
    root = build_trie(patterns)
    matches = {i: [] for i in range(len(patterns))}
    
    for i in range(len(text)):
        node = root
        j = i
        while j < len(text) and text[j] in node.children:
            node = node.children[text[j]]
            if node.is_end:
                matches[node.pattern_index].append(i)
            j += 1
    
    return matches

# Example usage
patterns = ["he", "she", "his", "hers"]
text = "ushers"
result = search_patterns(text, patterns)
# Output: {0: [2], 1: [1], 2: [], 3: [2]}
```

**Why**: Trie-based matching is efficient for searching multiple patterns simultaneously, better than running each pattern search separately.

# PROBLEM: Compress string using run-length encoding
# TIME: O(n) SPACE: O(n)
def compress_string(s):
    """Compress string using run-length encoding."""
    if not s:
        return ""
    
    result = []
    count = 1
    
    for i in range(1, len(s)):
        if s[i] == s[i-1]:
            count += 1
        else:
            result.append(s[i-1] + str(count))
            count = 1
    
    result.append(s[-1] + str(count))
    compressed = ''.join(result)
    
    return compressed if len(compressed) < len(s) else s

# Example: "aaabbbccc" -> "a3b3c3"
```

**Why**: Run-length encoding is simple yet effective for strings with many consecutive repeated characters, commonly used in image compression.

# PROBLEM: Encode data using arithmetic coding for better compression
# TIME: O(n) SPACE: O(1)
def arithmetic_encode(symbols, probabilities, message):
    """Encode message using arithmetic coding."""
    # Build cumulative probability ranges
    ranges = {}
    low = 0.0
    for symbol, prob in zip(symbols, probabilities):
        ranges[symbol] = (low, low + prob)
        low += prob
    
    # Encode
    low, high = 0.0, 1.0
    for symbol in message:
        range_width = high - low
        high = low + range_width * ranges[symbol][1]
        low = low + range_width * ranges[symbol][0]
    
    return (low + high) / 2

def arithmetic_decode(symbols, probabilities, code, length):
    """Decode arithmetic coded message."""
    # Build cumulative probability ranges
    ranges = {}
    low = 0.0
    for symbol, prob in zip(symbols, probabilities):
        ranges[symbol] = (low, low + prob)
        low += prob
    
    # Decode
    message = []
    for _ in range(length):
        for symbol, (symbol_low, symbol_high) in ranges.items():
            if symbol_low <= code < symbol_high:
                message.append(symbol)
                range_width = symbol_high - symbol_low
                code = (code - symbol_low) / range_width
                break
    
    return message

# Example: symbols=['A', 'B'], probabilities=[0.7, 0.3]
# message=['A', 'B', 'A'] -> encoded_value

**Why**: Arithmetic coding achieves near-optimal compression by representing entire messages as single floating-point numbers, outperforming Huffman for skewed distributions.

def mod_exp(base, exp, mod):
    """Fast modular exponentiation."""
    result = 1
    base = base % mod
    
    while exp > 0:
        if exp % 2 == 1:
            result = (result * base) % mod
        exp = exp >> 1
        base = (base * base) % mod

    """
    Fermat's Little Theorem: a^(p-1) ≡ 1 (mod p) for prime p

# TIME: O(√n) SPACE: O(1)
def euler_totient(n):
    """
    Euler's totient function: count integers 1 to n-1 coprime to n.
    φ(n) = n * ∏(1 - 1/p) for all prime factors p of n
    """
    result = n
    p = 2
    
    while p * p <= n:
        if n % p == 0:
            # Remove all occurrences of p from n
            while n % p == 0:
                n = n // p
            # Apply formula
            result -= result // p
        p += 1
    
    # If n is still greater than 1, it's a prime factor
    if n > 1:
        result -= result // n
    
    return result

def euler_totient_sieve(limit):
    """Compute totient for all numbers up to limit using sieve."""
    phi = list(range(limit + 1))
    
    for i in range(2, limit + 1):
        if phi[i] == i:  # i is prime
            for j in range(i, limit + 1, i):
                phi[j] -= phi[j] // i
    
    return phi

    Baby-step Giant-step algorithm for discrete logarithm.
    Find x such that g^x ≡ h (mod p)
    """
    n = int((p - 1) ** 0.5) + 1
    
    # Baby steps: compute g^j mod p for j in [0, n)
    baby_steps = {}
    power = 1
    for j in range(n):
        if power not in baby_steps:
            baby_steps[power] = j
        power = (power * g) % p

    gn_inverse = pow(g, n * (p - 2), p)  # Using Fermat's little theorem

    for i in range(n):
        if gamma in baby_steps:
            return i * n + baby_steps[gamma]
        gamma = (gamma * gn_inverse) % p

# PROBLEM: Solve system of linear equations using Gaussian elimination
# TIME: O(n^3) SPACE: O(n^2)
def gaussian_elimination(matrix):
    """
    Solve system of linear equations using Gaussian elimination.
    matrix: augmented matrix [A|b] where Ax = b
    Returns solution vector or None if no unique solution.
    """
    n = len(matrix)
    
    # Forward elimination
    for i in range(n):
        # Find pivot
        max_row = i
        for k in range(i + 1, n):
            if abs(matrix[k][i]) > abs(matrix[max_row][i]):
                max_row = k
        
        # Swap rows
        matrix[i], matrix[max_row] = matrix[max_row], matrix[i]
        
        # Check for singular matrix
        if abs(matrix[i][i]) < 1e-10:
            return None
        
        # Eliminate column
        for k in range(i + 1, n):
            factor = matrix[k][i] / matrix[i][i]
            for j in range(i, n + 1):
                matrix[k][j] -= factor * matrix[i][j]
    
    # Back substitution
    solution = [0] * n
    for i in range(n - 1, -1, -1):
        solution[i] = matrix[i][n]
        for j in range(i + 1, n):
            solution[i] -= matrix[i][j] * solution[j]
        solution[i] /= matrix[i][i]
    
    return solution

#              -2x + y + 2z = -3
# matrix = [[2, 1, -1, 8], [-3, -1, 2, -11], [-2, 1, 2, -3]]
# solution = [2, 3, -1]
```

# PROBLEM: Compute determinant of a matrix
# TIME: O(n!) naive, O(n^3) with LU decomposition SPACE: O(n^2)
def determinant_recursive(matrix):
    """Compute determinant using recursive expansion (for small matrices)."""
    n = len(matrix)
    
    # Base case
    if n == 1:
        return matrix[0][0]
    if n == 2:
        return matrix[0][0] * matrix[1][1] - matrix[0][1] * matrix[1][0]
    
    det = 0
    for j in range(n):
        # Create minor matrix
        minor = []
        for i in range(1, n):
            row = []
            for k in range(n):
                if k != j:
                    row.append(matrix[i][k])
            minor.append(row)
        
        # Recursive call
        cofactor = ((-1) ** j) * matrix[0][j] * determinant_recursive(minor)
        det += cofactor
    
    return det

def determinant_lu(matrix):
    """Compute determinant using LU decomposition (more efficient)."""
    n = len(matrix)
    # Make a copy
    mat = [row[:] for row in matrix]
    
    det = 1
    for i in range(n):
        # Find pivot
        max_row = i
        for k in range(i + 1, n):
            if abs(mat[k][i]) > abs(mat[max_row][i]):
                max_row = k
        
        # Swap rows (changes sign of determinant)
        if max_row != i:
            mat[i], mat[max_row] = mat[max_row], mat[i]
            det *= -1
        
        det *= mat[i][i]
        
        # Eliminate
        for k in range(i + 1, n):
            if mat[i][i] != 0:
                factor = mat[k][i] / mat[i][i]
                for j in range(i, n):
                    mat[k][j] -= factor * mat[i][j]
    
    return det

```

# PROBLEM: Find path from start to end in maze using backtracking
# TIME: O(2^(n*m)) worst case SPACE: O(n*m)
def solve_maze(maze, start, end):
    """
    Solve maze using backtracking.
    maze: 2D array where 0 = path, 1 = wall
    Returns path if exists, None otherwise.
    """
    rows, cols = len(maze), len(maze[0])
    path = []
    visited = [[False] * cols for _ in range(rows)]
    
    def is_safe(x, y):
        return (0 <= x < rows and 0 <= y < cols and 
                maze[x][y] == 0 and not visited[x][y])
    
    def backtrack(x, y):
        # Check if reached destination
        if (x, y) == end:
            path.append((x, y))
            return True
        
        # Check if current position is valid
        if not is_safe(x, y):
            return False
        
        # Mark current cell as visited
        visited[x][y] = True
        path.append((x, y))
        
        # Try all 4 directions: right, down, left, up
        directions = [(0, 1), (1, 0), (0, -1), (-1, 0)]
        for dx, dy in directions:
            if backtrack(x + dx, y + dy):
                return True
        
        # Backtrack
        path.pop()
        visited[x][y] = False
        return False
    
    if backtrack(start[0], start[1]):
        return path
    return None

# Example:
# maze = [[0, 1, 0, 0],
#         [0, 0, 0, 1],
#         [1, 0, 1, 0],
#         [0, 0, 0, 0]]
# start = (0, 0), end = (3, 3)
```

**Why**: Rat maze is a classic backtracking problem that demonstrates recursive problem-solving with state exploration and pruning.

# PROBLEM: Maximize value in knapsack with fractional items allowed
# TIME: O(n log n) SPACE: O(n)
def fractional_knapsack(capacity, items):
    """
    Solve fractional knapsack problem using greedy approach.
    items: list of (weight, value) tuples
    Returns maximum value achievable.
    """
    # Calculate value per unit weight
    items_with_ratio = [(value/weight, weight, value) for weight, value in items]
    
    # Sort by value per unit weight in descending order
    items_with_ratio.sort(reverse=True)
    
    total_value = 0
    remaining_capacity = capacity
    
    for ratio, weight, value in items_with_ratio:
        if remaining_capacity >= weight:
            # Take whole item
            total_value += value
            remaining_capacity -= weight
        else:
            # Take fraction of item
            fraction = remaining_capacity / weight
            total_value += value * fraction
            break
    
    return total_value

# Example: capacity = 50
# items = [(10, 60), (20, 100), (30, 120)]
# items with ratio: [(6, 10, 60), (5, 20, 100), (4, 30, 120)]
# Maximum value = 240.0
```

**Why**: Fractional knapsack demonstrates the greedy paradigm perfectly - always choosing the item with highest value density leads to optimal solution.

# TIME: O(m*n) SPACE: O(m*n)
def is_match_regex(s, p):
    """
    DP solution for regex matching.
    s: input string
    p: pattern with '.' (any char) and '*' (zero or more of preceding)
    """
    m, n = len(s), len(p)
    dp = [[False] * (n + 1) for _ in range(m + 1)]
    
    # Empty pattern matches empty string
    dp[0][0] = True
    
    # Handle patterns like a*, a*b*, a*b*c* which can match empty string
    for j in range(2, n + 1):
        if p[j-1] == '*':
            dp[0][j] = dp[0][j-2]
    
    # Fill DP table
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if p[j-1] == '*':
                # Zero occurrences of preceding element
                dp[i][j] = dp[i][j-2]
                # One or more occurrences if preceding element matches
                if p[j-2] == s[i-1] or p[j-2] == '.':
                    dp[i][j] = dp[i][j] or dp[i-1][j]
            elif p[j-1] == s[i-1] or p[j-1] == '.':
                dp[i][j] = dp[i-1][j-1]
    
    return dp[m][n]

# Example: is_match_regex("aab", "c*a*b") = True
```

# PROBLEM: Predict winner in game where players pick from ends of array
# TIME: O(n^2) SPACE: O(n^2)
def predict_winner(nums):
    """
    Given array of scores, two players take turns picking from ends.
    Predict if player 1 can win (or tie).
    """
    n = len(nums)
    
    # dp[i][j] = maximum score difference player can achieve
    # on subarray from i to j
    dp = [[0] * n for _ in range(n)]
    
    # Base case: single element
    for i in range(n):
        dp[i][i] = nums[i]
    
    # Fill for subarrays of increasing length
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            # Player can pick nums[i] or nums[j]
            # If pick nums[i], opponent will then optimize on (i+1, j)
            pick_left = nums[i] - dp[i+1][j]
            pick_right = nums[j] - dp[i][j-1]
            dp[i][j] = max(pick_left, pick_right)
    
    return dp[0][n-1] >= 0

# Example: predict_winner([1, 5, 2]) = False
# Player 1 picks 1 or 2, player 2 picks 5, player 2 wins
```

**Why**: This game theory problem demonstrates optimal play strategies and minimax thinking in competitive scenarios.

def solve_sudoku(board):

    Solve Sudoku using backtracking.
    board: 9x9 grid, 0 represents empty cell
    Returns True if solution exists, modifies board in-place.
    """
    def is_valid(board, row, col, num):
        # Check row
        if num in board[row]:
            return False
        
        # Check column
        if num in [board[i][col] for i in range(9)]:
            return False
        
        # Check 3x3 box
        box_row, box_col = 3 * (row // 3), 3 * (col // 3)
        for i in range(box_row, box_row + 3):
            for j in range(box_col, box_col + 3):
                if board[i][j] == num:
                    return False
        
        return True
    
    def solve(board):
        for i in range(9):
            for j in range(9):
                if board[i][j] == 0:
                    for num in range(1, 10):
                        if is_valid(board, i, j, num):
                            board[i][j] = num
                            if solve(board):
                                return True
                            board[i][j] = 0
                    return False
        return True
    
    return solve(board)

# PROBLEM: Add operators (+, -, *) to form target value
# TIME: O(4^n) SPACE: O(n) for recursion
def add_operators(num, target):
    """
    Insert +, -, or * between digits to reach target.
    Returns all valid expressions.
    """
    result = []
    n = len(num)
    
    def dfs(idx, path, value, prev):
        """
        idx: current position in num
        path: current expression string
        value: current evaluated value
        prev: previous operand (for multiplication precedence)
        """
        if idx == n:
            if value == target:
                result.append(path)
            return
        
        for j in range(idx + 1, n + 1):
            # Get current number
            curr_str = num[idx:j]
            curr_num = int(curr_str)
            
            # Skip numbers with leading zeros
            if len(curr_str) > 1 and curr_str[0] == '0':
                break
            
            if idx == 0:
                # First number, no operator
                dfs(j, curr_str, curr_num, curr_num)
            else:
                # Add
                dfs(j, path + '+' + curr_str, value + curr_num, curr_num)
                # Subtract
                dfs(j, path + '-' + curr_str, value - curr_num, -curr_num)
                # Multiply (need to handle precedence)
                dfs(j, path + '*' + curr_str, 
                    value - prev + prev * curr_num, 
                    prev * curr_num)
    
    dfs(0, "", 0, 0)
    return result

# Example: add_operators("123", 6) = ["1+2+3", "1*2*3"]
# add_operators("105", 5) = ["1*0+5", "10-5"]
```

import random

class RandomizedBSTNode:
    def __init__(self, key, value=None):
        self.key = key
        self.value = value
        self.left = None
        self.right = None
        self.size = 1

class RandomizedBST:
    def __init__(self):
        self.root = None
    
    def size(self, node):
        return node.size if node else 0
    
    def update_size(self, node):
        if node:
            node.size = 1 + self.size(node.left) + self.size(node.right)
    
    def rotate_right(self, node):
        left = node.left
        node.left = left.right
        left.right = node
        self.update_size(node)
        self.update_size(left)
        return left
    
    def rotate_left(self, node):
        right = node.right
        node.right = right.left
        right.left = node
        self.update_size(node)
        self.update_size(right)
        return right
    
    def insert(self, key, value=None):
        self.root = self._insert(self.root, key, value)
    
    def _insert(self, node, key, value):
        if not node:
            return RandomizedBSTNode(key, value)
        
        # Random decision: insert at root or recurse
        if random.random() < 1.0 / (self.size(node) + 1):
            # Insert at root
            new_node = RandomizedBSTNode(key, value)
            if key < node.key:
                new_node.right = node
                new_node.left = node.left
                node.left = None
            else:
                new_node.left = node
                new_node.right = node.right
                node.right = None
            self.update_size(node)
            self.update_size(new_node)
            return new_node
        
        # Normal BST insertion
        if key < node.key:
            node.left = self._insert(node.left, key, value)
        elif key > node.key:
            node.right = self._insert(node.right, key, value)
        else:
            node.value = value  # Update existing
        
        self.update_size(node)
        return node
    
    def search(self, key):
        node = self.root
        while node:
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                return node.value
        return None
```

**Why**: Randomized BSTs achieve probabilistic balance without complex rebalancing logic, providing expected O(log n) performance with simpler implementation than AVL or Red-Black trees.

            node.left = left.right

# PROBLEM: Implement weight-balanced tree where subtree sizes are balanced
# TIME: O(log n) for all operations SPACE: O(n)
class WBTNode:
    def __init__(self, key, value=None):
        self.key = key
        self.value = value
        self.left = None
        self.right = None
        self.weight = 1  # Number of nodes in subtree

class WeightBalancedTree:
    def __init__(self, alpha=0.3):
        self.root = None
        self.alpha = alpha  # Balance parameter
    
    def weight(self, node):
        return node.weight if node else 0
    
    def update_weight(self, node):
        if node:
            node.weight = 1 + self.weight(node.left) + self.weight(node.right)
    
    def is_balanced(self, node):
        if not node:
            return True
        left_weight = self.weight(node.left)
        right_weight = self.weight(node.right)
        total = left_weight + right_weight + 1
        return (left_weight <= self.alpha * total and 
                right_weight <= self.alpha * total)
    
    def rotate_right(self, node):
        left = node.left
        node.left = left.right
        left.right = node
        self.update_weight(node)
        self.update_weight(left)
        return left
    
    def rotate_left(self, node):
        right = node.right
        node.right = right.left
        right.left = node
        self.update_weight(node)
        self.update_weight(right)
        return right
    
    def rebalance(self, node):
        if not node:
            return node
        
        self.update_weight(node)
        
        if self.is_balanced(node):
            return node
        
        # Need to rebalance
        if self.weight(node.left) > self.weight(node.right):
            if self.weight(node.left.left) < self.weight(node.left.right):
                node.left = self.rotate_left(node.left)
            node = self.rotate_right(node)
        else:
            if self.weight(node.right.right) < self.weight(node.right.left):
                node.right = self.rotate_right(node.right)
            node = self.rotate_left(node)
        
        return node
    
    def insert(self, key, value=None):
        self.root = self._insert(self.root, key, value)
    
    def _insert(self, node, key, value):
        if not node:
            return WBTNode(key, value)
        
        if key < node.key:
            node.left = self._insert(node.left, key, value)
        elif key > node.key:
            node.right = self._insert(node.right, key, value)
        else:
            node.value = value
            return node
        
        return self.rebalance(node)
```

**Why**: Weight-balanced trees maintain balance based on subtree sizes, providing guaranteed O(log n) performance with different balance criteria than height-balanced trees.

# PROBLEM: Data structure for efficient orthogonal range queries
# TIME: O(log^d n) for d-dimensional query SPACE: O(n log^(d-1) n)
class RangeTreeNode:
    def __init__(self, point=None):
        self.point = point
        self.left = None
        self.right = None
        self.subtree = None  # Tree for next dimension

class RangeTree:
    def __init__(self, points):
        # points: list of (x, y, ...) tuples
        self.dimensions = len(points[0]) if points else 2
        self.root = self._build(points, 0) if points else None
    
    def _build(self, points, dimension):
        if not points:
            return None
        
        # Sort by current dimension
        points = sorted(points, key=lambda p: p[dimension])
        mid = len(points) // 2
        
        node = RangeTreeNode(points[mid])
        node.left = self._build(points[:mid], dimension)
        node.right = self._build(points[mid+1:], dimension)
        
        # Build tree for next dimension
        if dimension < self.dimensions - 1:
            node.subtree = RangeTree2D(points)
        
        return node
    
    def query_range(self, ranges):
        """
        Query points in ranges [(min1, max1), (min2, max2), ...]
        """
        return self._query(self.root, ranges, 0)
    
    def _query(self, node, ranges, dimension):
        if not node:
            return []
        
        min_val, max_val = ranges[dimension]
        point = node.point
        
        # If node point is in range for current dimension
        if point[dimension] < min_val:
            return self._query(node.right, ranges, dimension)
        elif point[dimension] > max_val:
            return self._query(node.left, ranges, dimension)
        else:
            # Check next dimensions
            if dimension == self.dimensions - 1:
                # Last dimension, collect all points in range
                result = []
                if self._point_in_range(point, ranges):
                    result.append(point)
                result.extend(self._query(node.left, ranges, dimension))
                result.extend(self._query(node.right, ranges, dimension))
                return result
            else:
                # Query subtree for remaining dimensions
                # Simplified: collect all in current dimension range
                result = []
                if self._point_in_range(point, ranges):
                    result.append(point)
                result.extend(self._query(node.left, ranges, dimension))
                result.extend(self._query(node.right, ranges, dimension))
                return result
    
    def _point_in_range(self, point, ranges):
        return all(ranges[i][0] <= point[i] <= ranges[i][1] 
                   for i in range(len(ranges)))

**Why**: Range trees enable efficient multidimensional range queries, crucial in computational geometry, databases, and spatial applications.

# PROBLEM: Space-partitioning data structure for 3D points
# TIME: O(log n) insertion, O(n) worst case SPACE: O(n)
class OctreeNode:
    def __init__(self, center, size):
        self.center = center  # (x, y, z)
        self.size = size
        self.children = [None] * 8
        self.points = []
        self.is_leaf = True

class Octree:
    def __init__(self, center, size, capacity=4):
        self.root = OctreeNode(center, size)
        self.capacity = capacity
    
    def insert(self, point):
        self._insert(self.root, point)
    
    def _get_octant(self, center, point):
        """Determine which octant the point belongs to."""
        octant = 0
        if point[0] >= center[0]:
            octant |= 1
        if point[1] >= center[1]:
            octant |= 2
        if point[2] >= center[2]:
            octant |= 4
        return octant
    
    def _insert(self, node, point):
        if node.is_leaf:
            if len(node.points) < self.capacity:
                node.points.append(point)
                return
            else:
                # Split node
                self._split(node)
        
        # Insert into appropriate child
        octant = self._get_octant(node.center, point)
        if node.children[octant] is None:
            # Create child node
            new_center = self._calculate_child_center(node.center, node.size, octant)
            node.children[octant] = OctreeNode(new_center, node.size / 2)
        
        self._insert(node.children[octant], point)
    
    def _split(self, node):
        node.is_leaf = False
        points = node.points
        node.points = []
        
        for point in points:
            self._insert(node, point)
    
    def _calculate_child_center(self, center, size, octant):
        offset = size / 4
        new_center = list(center)
        
        if octant & 1:
            new_center[0] += offset
        else:
            new_center[0] -= offset
        
        if octant & 2:
            new_center[1] += offset
        else:
            new_center[1] -= offset
        
        if octant & 4:
            new_center[2] += offset
        else:
            new_center[2] -= offset
        
        return tuple(new_center)
    
    def query_range(self, center, size):
        """Find all points within a cubic range."""
        return self._query_range(self.root, center, size)
    
    def _query_range(self, node, center, size):
        if not node:
            return []
        
        result = []
        
        # Check if node intersects with query range
        if not self._intersects(node, center, size):
            return []
        
        # If leaf, check points
        if node.is_leaf:
            for point in node.points:
                if self._point_in_range(point, center, size):
                    result.append(point)
            return result
        
        # Query children
        for child in node.children:
            if child:
                result.extend(self._query_range(child, center, size))
        
        return result
    
    def _intersects(self, node, center, size):
        # Simplified intersection check
        return True
    
    def _point_in_range(self, point, center, size):
        half_size = size / 2
        return (abs(point[0] - center[0]) <= half_size and
                abs(point[1] - center[1]) <= half_size and
                abs(point[2] - center[2]) <= half_size)
```

**Why**: Octrees efficiently organize 3D spatial data, widely used in computer graphics, game engines, and spatial databases for fast nearest neighbor and range queries.
