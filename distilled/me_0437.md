```python
# PROBLEM: Implement compressed trie (radix tree) for space-efficient string storage
# TIME: O(L) for insert/search/delete where L is key length
# SPACE: O(N * L_avg) where N is number of keys
class CompressedTrieNode:
    """Node for compressed trie with edge labels."""
    
    def __init__(self):
        self.children = {}  # char -> (label, node)
        self.is_end = False

class CompressedTrie:
    """
    Space-optimized trie that compresses chains of single-child nodes.
    Useful for dictionaries with common prefixes.
    """
    
    def __init__(self):
        self.root = CompressedTrieNode()
    
    def insert(self, word: str) -> None:
        """Insert word into compressed trie."""
        node = self.root
        i = 0
        
        while i < len(word):
            char = word[i]
            
            if char not in node.children:
                # Create new edge with remaining suffix
                node.children[char] = (word[i:], CompressedTrieNode())
                node.children[char][1].is_end = (i == len(word) - 1)
                return
            
            label, child = node.children[char]
            
            # Find common prefix between label and remaining word
            j = 0
            while j < len(label) and i + j < len(word) and label[j] == word[i + j]:
                j += 1
            
            if j == len(label):
                # Full label matches, continue to child
                node = child
                i += j
            else:
                # Partial match, need to split
                # Create intermediate node
                mid = CompressedTrieNode()
                
                # Update current edge to point to mid with shorter label
                node.children[char] = (label[:j], mid)
                
                # Connect mid to old child with remaining label
                mid.children[label[j]] = (label[j:], child)
                
                # Continue from mid
                node = mid
                i += j
        
        node.is_end = True
    
    def search(self, word: str) -> bool:
        """Check if word exists in trie."""
        node = self.root
        i = 0
        
        while i < len(word):
            char = word[i]
            
            if char not in node.children:
                return False
            
            label, child = node.children[char]
            
            # Check if label matches remaining word
            if word[i:i+len(label)] != label:
                return False
            
            node = child
            i += len(label)
        
        return node.is_end
```
**Why**: Compressed tries save significant memory when storing strings with common prefixes. Used in IP routing tables, autocomplete systems, and spell checkers.

```python
class RadixTrie:

    Each edge can store multiple characters.
    """

        self.root = {}

        """Insert word using radix tree compression."""
        node = self.root
        i = 0
        
        while i < len(word):
            char = word[i]
            
            if char not in node:
                # Create leaf with remaining string
                node[char] = {'_label': word[i:], '_end': True}
                return
            
            current = node[char]
            label = current.get('_label', '')
            
            # Find matching prefix
            match_len = 0
            while (match_len < len(label) and 
                   i + match_len < len(word) and 
                   label[match_len] == word[i + match_len]):
                match_len += 1
            
            if match_len == len(label):
                # Move to next node
                i += match_len
                if i == len(word):
                    current['_end'] = True
                node = current
            elif match_len > 0:
                # Split node at match point
                # This is a simplified version; full implementation
                # would handle all edge cases
                i += match_len
                node = current
            else:
                # No match, create new branch
                node[char] = {'_label': word[i:], '_end': True}
                return

            if char not in node:
                return []

            label = current.get('_label', '')
            path += label
            i += len(label)

        for key, child in node.items():
            if key.startswith('_'):
                continue
            self._collect(child, path + child.get('_label', ''), results)
```

```python
import heapq
from typing import List, Generic, TypeVar, Optional

T = TypeVar('T')

class MinHeap(Generic[T]):
    """Min-heap priority queue implementation."""
    
    def __init__(self):
        self._heap: List[T] = []
    
    def push(self, item: T) -> None:
        """Add item to heap."""
        heapq.heappush(self._heap, item)
    
    def pop(self) -> Optional[T]:
        """Remove and return minimum item."""
        return heapq.heappop(self._heap) if self._heap else None
    
    def peek(self) -> Optional[T]:
        """Return minimum item without removing."""
        return self._heap[0] if self._heap else None
    
    def __len__(self) -> int:
        return len(self._heap)


class MaxHeap(Generic[T]):
    """Max-heap using negation trick."""
    
    def __init__(self):
        self._heap: List[T] = []
    
    def push(self, item: T) -> None:
        """Add item to max-heap."""
        heapq.heappush(self._heap, (-item, item) if isinstance(item, (int, float)) else item)
    
    def pop(self) -> Optional[T]:
        """Remove and return maximum item."""
        if not self._heap:
            return None
        result = heapq.heappop(self._heap)
        return -result[0] if isinstance(result, tuple) else result


class IndexedPriorityQueue:
    """
    Priority queue supporting decrease-key operation.
    Useful for Dijkstra's algorithm.
    """
    
    def __init__(self, capacity: int):
        self._heap: List[int] = []
        self._position = {}  # item -> index in heap
        self._priorities = {}  # item -> priority
        self.capacity = capacity
    
    def push(self, item: int, priority: float) -> None:
        """Insert item with priority."""
        if item in self._position:
            self.decrease_key(item, priority)
            return
        
        self._priorities[item] = priority
        self._heap.append(item)
        idx = len(self._heap) - 1
        self._position[item] = idx
        self._swim(idx)
    
    def pop(self) -> Optional[int]:
        """Remove and return item with minimum priority."""
        if not self._heap:
            return None
        
        min_item = self._heap[0]
        del self._position[min_item]
        del self._priorities[min_item]
        
        last = self._heap.pop()
        if self._heap:
            self._heap[0] = last
            self._position[last] = 0
            self._sink(0)
        
        return min_item
    
    def decrease_key(self, item: int, new_priority: float) -> None:
        """Decrease priority of existing item."""
        if new_priority >= self._priorities.get(item, float('inf')):
            return
        
        self._priorities[item] = new_priority
        idx = self._position[item]
        self._swim(idx)
    
    def _swim(self, idx: int) -> None:
        """Bubble up element at index."""
        while idx > 0:
            parent = (idx - 1) // 2
            if self._priorities[self._heap[parent]] <= self._priorities[self._heap[idx]]:
                break
            self._swap(idx, parent)
            idx = parent
    
    def _sink(self, idx: int) -> None:
        """Sink down element at index."""
        n = len(self._heap)
        while 2 * idx + 1 < n:
            child = 2 * idx + 1
            if child + 1 < n and self._priorities[self._heap[child + 1]] < self._priorities[self._heap[child]]:
                child += 1
            if self._priorities[self._heap[idx]] <= self._priorities[self._heap[child]]:
                break
            self._swap(idx, child)
            idx = child
    
    def _swap(self, i: int, j: int) -> None:
        """Swap elements and update positions."""
        self._heap[i], self._heap[j] = self._heap[j], self._heap[i]
        self._position[self._heap[i]] = i
        self._position[self._heap[j]] = j
```
**Why**: Priority queues are fundamental for scheduling, graph algorithms, and sorting. Indexed variant enables efficient priority updates in algorithms like Dijkstra's.

```python
# PROBLEM: Implement union-find with path compression and union by rank
# TIME: O(α(n)) amortized where α is inverse Ackermann function
# SPACE: O(n)
class UnionFind:
    """
    Basic union-find with path compression and union by rank.
    """
    
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.components = n
    
    def find(self, x: int) -> int:
        """Find root with path compression."""
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x: int, y: int) -> bool:
        """Union by rank, returns True if merged."""
        px, py = self.find(x), self.find(y)
        
        if px == py:
            return False
        
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        self.components -= 1
        return True
    
    def connected(self, x: int, y: int) -> bool:
        """Check if x and y are in same component."""
        return self.find(x) == self.find(y)


class UnionFindSize:
    """
    Union-find tracking component sizes.
    """
    
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.size = [1] * n
        self.components = n
    
    def find(self, x: int) -> int:
        """Find with path halving."""
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]
            x = self.parent[x]
        return x
    
    def union(self, x: int, y: int) -> bool:
        """Union by size."""
        px, py = self.find(x), self.find(y)
        
        if px == py:
            return False
        
        # Attach smaller tree to larger
        if self.size[px] < self.size[py]:
            px, py = py, px
        
        self.parent[py] = px
        self.size[px] += self.size[py]
        self.components -= 1
        return True
    
    def get_size(self, x: int) -> int:
        """Get size of component containing x."""
        return self.size[self.find(x)]


class PersistentUnionFind:
    """
    Union-find supporting undo operations.
    """
    
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.history = []
    
    def find(self, x: int) -> int:
        """Find without path compression (to support undo)."""
        while self.parent[x] != x:
            x = self.parent[x]
        return x
    
    def union(self, x: int, y: int) -> bool:
        """Union with history tracking."""
        px, py = self.find(x), self.find(y)
        
        if px == py:
            self.history.append(None)
            return False
        
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        
        # Save state for undo
        self.history.append((py, self.parent[py], px, self.rank[px]))
        
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        return True
    
    def undo(self) -> bool:
        """Undo last union operation."""
        if not self.history:
            return False
        
        record = self.history.pop()
        if record is None:
            return False
        
        child, old_parent, parent, old_rank = record
        self.parent[child] = old_parent
        self.rank[parent] = old_rank
        return True
```
**Why**: Union-find variants enable efficient connectivity queries in dynamic graphs. Persistence allows rollback for backtracking algorithms.

```python
# PROBLEM: Implement hopscotch hashing for cache-friendly hash table
# TIME: O(1) average for insert/lookup/delete
# SPACE: O(n) with H-hop neighborhood
from typing import Optional, TypeVar, Generic

T = TypeVar('T')

class HopscotchHashTable(Generic[T]):
    """
    Hopscotch hashing with H-hop neighborhood.
    Provides cache-friendly access with bounded probe lengths.
    """
    
    def __init__(self, capacity: int = 16, hop_range: int = 32):
        self.capacity = capacity
        self.hop_range = hop_range
        self.table: List[Optional[T]] = [None] * capacity
        self.hops = [0] * capacity  # Bitmask for neighborhood
        self.size = 0
    
    def _hash(self, key) -> int:
        """Compute hash value."""
        return hash(key) % self.capacity
    
    def insert(self, key) -> bool:
        """Insert key using hopscotch probing."""
        if self.size >= self.capacity:
            self._resize()
        
        h = self._hash(key)
        
        # Check if already in neighborhood
        if self._find_in_neighborhood(key, h) is not None:
            return False
        
        # Find empty slot
        empty = self._find_empty_slot(h)
        if empty is None:
            self._resize()
            return self.insert(key)
        
        # Move empty slot into neighborhood if needed
        while (empty - h) % self.capacity >= self.hop_range:
            empty = self._move_closer(empty)
            if empty is None:
                self._resize()
                return self.insert(key)
        
        # Insert at empty slot
        self.table[empty] = key
        self.hops[h] |= 1 << ((empty - h) % self.capacity)
        self.size += 1
        return True
    
    def _find_in_neighborhood(self, key, h: int) -> Optional[int]:
        """Find key in hop neighborhood of h."""
        hops = self.hops[h]
        offset = 0
        while hops:
            if hops & 1:
                idx = (h + offset) % self.capacity
                if self.table[idx] == key:
                    return idx
            hops >>= 1
            offset += 1
        return None
    
    def _find_empty_slot(self, start: int) -> Optional[int]:
        """Find first empty slot from start."""
        for i in range(self.capacity):
            idx = (start + i) % self.capacity
            if self.table[idx] is None:
                return idx
        return None
    
    def _move_closer(self, empty: int) -> Optional[int]:
        """Move an element to bring empty slot closer."""
        for offset in range(1, self.hop_range):
            candidate = (empty - offset) % self.capacity
            hops = self.hops[candidate]
            
            for i in range(self.hop_range):
                if hops & (1 << i):
                    move_from = (candidate + i) % self.capacity
                    if (empty - candidate) % self.capacity < self.hop_range:
                        # Swap
                        self.table[empty] = self.table[move_from]
                        self.table[move_from] = None
                        
                        # Update hop bits
                        self.hops[candidate] &= ~(1 << i)
                        self.hops[candidate] |= 1 << ((empty - candidate) % self.capacity)
                        
                        return move_from
        return None
    
    def _resize(self) -> None:
        """Double the table size."""
        old_table = self.table
        old_capacity = self.capacity
        self.capacity *= 2
        self.table = [None] * self.capacity
        self.hops = [0] * self.capacity
        self.size = 0
        
        for item in old_table:
            if item is not None:
                self.insert(item)
```
**Why**: Hopscotch hashing provides excellent cache performance by keeping related elements in nearby memory locations, bounded by H-hop distance.

```python
# PROBLEM: Implement hash table with linear probing collision resolution
# TIME: O(1) average, O(n) worst case
# SPACE: O(n)
from typing import Optional, TypeVar, Generic, Hashable

K = TypeVar('K', bound=Hashable)
V = TypeVar('V')

class LinearProbingHashTable(Generic[K, V]):
    """
    Hash table using linear probing for collision resolution.
    Simple but can suffer from primary clustering.
    """
    
    DELETED = object()  # Marker for deleted entries
    
    def __init__(self, capacity: int = 16, load_factor: float = 0.75):
        self.capacity = capacity
        self.load_factor = load_factor
        self.keys: List[Optional[K]] = [None] * capacity
        self.values: List[Optional[V]] = [None] * capacity
        self.size = 0
        self.deleted_count = 0
    
    def _hash(self, key: K) -> int:
        """Compute hash index."""
        return hash(key) % self.capacity
    
    def _probe(self, key: K) -> int:
        """Find index for key using linear probing."""
        idx = self._hash(key)
        first_deleted = -1
        
        while self.keys[idx] is not None:
            if self.keys[idx] is self.DELETED:
                if first_deleted == -1:
                    first_deleted = idx
            elif self.keys[idx] == key:
                return idx
            
            idx = (idx + 1) % self.capacity
        
        # Return deleted slot if found, otherwise empty slot
        return first_deleted if first_deleted != -1 else idx
    
    def insert(self, key: K, value: V) -> None:
        """Insert or update key-value pair."""
        if self._should_resize():
            self._resize()
        
        idx = self._probe(key)
        
        if self.keys[idx] is None or self.keys[idx] is self.DELETED:
            self.size += 1
            if self.keys[idx] is self.DELETED:
                self.deleted_count -= 1
        
        self.keys[idx] = key
        self.values[idx] = value
    
    def get(self, key: K) -> Optional[V]:
        """Get value for key."""
        idx = self._hash(key)
        start_idx = idx
        
        while self.keys[idx] is not None:
            if self.keys[idx] == key:
                return self.values[idx]
            
            idx = (idx + 1) % self.capacity
            if idx == start_idx:
                break
        
        return None
    
    def delete(self, key: K) -> bool:
        """Delete key using lazy deletion."""
        idx = self._hash(key)
        start_idx = idx
        
        while self.keys[idx] is not None:
            if self.keys[idx] == key:
                self.keys[idx] = self.DELETED
                self.values[idx] = None
                self.size -= 1
                self.deleted_count += 1
                return True
            
            idx = (idx + 1) % self.capacity
            if idx == start_idx:
                break
        
        return False
    
    def _should_resize(self) -> bool:
        """Check if resize needed."""
        return (self.size + self.deleted_count) > self.capacity * self.load_factor
    
    def _resize(self) -> None:
        """Rehash to larger table."""
        old_keys = self.keys
        old_values = self.values
        
        self.capacity *= 2
        self.keys = [None] * self.capacity
        self.values = [None] * self.capacity
        self.size = 0
        self.deleted_count = 0
        
        for k, v in zip(old_keys, old_values):
            if k is not None and k is not self.DELETED:
                self.insert(k, v)
```
**Why**: Linear probing is simple and cache-friendly but suffers from primary clustering. Good for small load factors and when cache performance matters.

```python
# PROBLEM: Implement hash table with quadratic probing
# TIME: O(1) average
# SPACE: O(n)
from typing import Optional, Generic, TypeVar, Hashable

K = TypeVar('K', bound=Hashable)
V = TypeVar('V')

class QuadraticProbingHashTable(Generic[K, V]):
    """
    Hash table using quadratic probing: (h + c1*i + c2*i^2)
    Reduces primary clustering compared to linear probing.
    """
    
    def __init__(self, capacity: int = 16, c1: int = 0, c2: int = 1):
        self.capacity = capacity
        self.c1 = c1
        self.c2 = c2
        self.keys: List[Optional[K]] = [None] * capacity
        self.values: List[Optional[V]] = [None] * capacity
        self.size = 0
    
    def _hash(self, key: K) -> int:
        """Base hash function."""
        return hash(key) % self.capacity
    
    def _probe(self, key: K) -> int:
        """Quadratic probe sequence."""
        h = self._hash(key)
        
        for i in range(self.capacity):
            idx = (h + self.c1 * i + self.c2 * i * i) % self.capacity
            
            if self.keys[idx] is None or self.keys[idx] == key:
                return idx
        
        raise RuntimeError("Hash table is full")
    
    def insert(self, key: K, value: V) -> None:
        """Insert or update key-value pair."""
        if self.size > self.capacity * 0.7:
            self._resize()
        
        idx = self._probe(key)
        
        if self.keys[idx] is None:
            self.size += 1
        
        self.keys[idx] = key
        self.values[idx] = value
    
    def get(self, key: K) -> Optional[V]:
        """Retrieve value by key."""
        h = self._hash(key)
        
        for i in range(self.capacity):
            idx = (h + self.c1 * i + self.c2 * i * i) % self.capacity
            
            if self.keys[idx] is None:
                return None
            
            if self.keys[idx] == key:
                return self.values[idx]
        
        return None
    
    def _resize(self) -> None:
        """Resize and rehash."""
        old_keys = self.keys
        old_values = self.values
        
        self.capacity *= 2
        self.keys = [None] * self.capacity
        self.values = [None] * self.capacity
        self.size = 0
        
        for k, v in zip(old_keys, old_values):
            if k is not None:
                self.insert(k, v)
```
**Why**: Quadratic probing reduces primary clustering by using non-linear probe sequence. Still cache-friendly with better distribution than linear probing.

```python
# PROBLEM: Implement hash table with double hashing collision resolution
# TIME: O(1) average with good hash functions
# SPACE: O(n)
from typing import Optional, Generic, TypeVar, Hashable

K = TypeVar('K', bound=Hashable)
V = TypeVar('V')

class DoubleHashTable(Generic[K, V]):
    """
    Hash table using double hashing: (h1 + i * h2) % size
    Provides uniform probe sequences, eliminates clustering.
    """
    
    def __init__(self, capacity: int = 16):
        # Ensure capacity is prime for better distribution
        self.capacity = self._next_prime(capacity)
        self.keys: List[Optional[K]] = [None] * self.capacity
        self.values: List[Optional[V]] = [None] * self.capacity
        self.size = 0
    
    def _hash1(self, key: K) -> int:
        """Primary hash function."""
        return hash(key) % self.capacity
    
    def _hash2(self, key: K) -> int:
        """Secondary hash function (must be coprime with capacity)."""
        h = hash(key) % (self.capacity - 1)
        return h + 1  # Ensure step size is at least 1
    
    def _probe(self, key: K) -> int:
        """Double hash probe sequence."""
        h1 = self._hash1(key)
        h2 = self._hash2(key)
        
        for i in range(self.capacity):
            idx = (h1 + i * h2) % self.capacity
            
            if self.keys[idx] is None or self.keys[idx] == key:
                return idx
        
        raise RuntimeError("Hash table full")
    
    def insert(self, key: K, value: V) -> None:
        """Insert key-value pair."""
        if self.size > self.capacity * 0.7:
            self._resize()
        
        idx = self._probe(key)
        
        if self.keys[idx] is None:
            self.size += 1
        
        self.keys[idx] = key
        self.values[idx] = value
    
    def get(self, key: K) -> Optional[V]:
        """Get value for key."""
        h1 = self._hash1(key)
        h2 = self._hash2(key)
        
        for i in range(self.capacity):
            idx = (h1 + i * h2) % self.capacity
            
            if self.keys[idx] is None:
                return None
            
            if self.keys[idx] == key:
                return self.values[idx]
        
        return None
    
    def _next_prime(self, n: int) -> int:
        """Find next prime >= n."""
        def is_prime(x: int) -> bool:
            if x < 2:
                return False
            for i in range(2, int(x ** 0.5) + 1):
                if x % i == 0:
                    return False
            return True
        
        while not is_prime(n):
            n += 1
        return n
    
    def _resize(self) -> None:
        """Resize to next prime capacity."""
        old_keys = self.keys
        old_values = self.values
        
        self.capacity = self._next_prime(self.capacity * 2)
        self.keys = [None] * self.capacity
        self.values = [None] * self.capacity
        self.size = 0
        
        for k, v in zip(old_keys, old_values):
            if k is not None:
                self.insert(k, v)
```
**Why**: Double hashing provides excellent key distribution by using two independent hash functions. Eliminates both primary and secondary clustering.

### Perfect Hashing

```python
# PROBLEM: Implement perfect hashing for static key sets
# TIME: O(1) worst case for lookup
# SPACE: O(n) expected
from typing import List, Optional, Generic, TypeVar

T = TypeVar('T')

class PerfectHashTable(Generic[T]):
    """
    Two-level perfect hashing for static sets.
    Guarantees O(1) worst-case lookup.
    """
    
    def __init__(self, keys: List[T]):
        self.n = len(keys)
        if self.n == 0:
            return
        
        # First level hash
        self._build_first_level(keys)
    
    def _build_first_level(self, keys: List[T]) -> None:
        """Build first level with collision resolution."""
        import random
        
        # Try random hash parameters until no bucket has > sqrt(n) items
        while True:
            self.a1 = random.randint(1, 10**9)
            self.b1 = random.randint(0, 10**9)
            self.p1 = self._next_prime(self.n * 2)
            
            buckets = [[] for _ in range(self.n)]
            
            for key in keys:
                idx = self._hash1(key)
                buckets[idx].append(key)
            
            # Check if bucket sizes are reasonable
            max_bucket = max(len(b) for b in buckets) if buckets else 0
            if max_bucket * max_bucket <= self.n * 4:
                break
        
        # Build second level for each bucket
        self.second_level = []
        for bucket in buckets:
            if len(bucket) == 0:
                self.second_level.append(None)
            else:
                self.second_level.append(self._build_second_level(bucket))
    
    def _build_second_level(self, keys: List[T]) -> dict:
        """Build collision-free second level for bucket."""
        import random
        
        m = len(keys) * len(keys)  # Square the size
        
        while True:
            a2 = random.randint(1, 10**9)
            b2 = random.randint(0, 10**9)
            p2 = self._next_prime(m)
            
            table = [None] * (m + 1)
            table[0] = {'a': a2, 'b': b2, 'p': p2}  # Store params
            
            collision = False
            for key in keys:
                idx = 1 + ((a2 * hash(key) + b2) % p2) % len(keys)
                if table[idx] is not None:
                    collision = True
                    break
                table[idx] = key
            
            if not collision:
                return table
    
    def _hash1(self, key: T) -> int:
        """First level hash."""
        return (self.a1 * hash(key) + self.b1) % self.p1 % self.n
    
    def contains(self, key: T) -> bool:
        """Check if key exists - O(1) worst case."""
        if self.n == 0:
            return False
        
        idx1 = self._hash1(key)
        bucket = self.second_level[idx1]
        
        if bucket is None:
            return False
        
        params = bucket[0]
        a2, b2, p2 = params['a'], params['b'], params['p']
        m = len(bucket) - 1
        
        idx2 = 1 + ((a2 * hash(key) + b2) % p2) % m
        
        return bucket[idx2] == key
    
    def _next_prime(self, n: int) -> int:
        """Find next prime >= n."""
        def is_prime(x: int) -> bool:
            if x < 2:
                return False
            for i in range(2, int(x ** 0.5) + 1):
                if x % i == 0:
                    return False
            return True
        
        while not is_prime(n):
            n += 1
        return n
```
**Why**: Perfect hashing provides O(1) worst-case lookup for static sets. Used in compilers, databases, and any application with known key sets.
