## [Memento Design Pattern - Undo/Redo System]

```python
# PROBLEM: Implement an undo/redo system using the Memento design pattern
# TIME: O(n) SPACE: O(n)

class Memento:
    """Stores the internal state of the Originator object"""
    
    def __init__(self, state: str):
        # Save the state at time of creation
        self._state = state
    
    def get_state(self) -> str:
        # Retrieve the saved state
        return self._state


class Originator:
    """The object whose state we want to save and restore"""
    
    def __init__(self):
        # Initial state
        self._state = ""
    
    def set_state(self, state: str) -> None:
        # Update current state
        self._state = state
        print(f"State set to: {self._state}")
    
    def save_to_memento(self) -> Memento:
        # Create a memento containing a snapshot of current state
        return Memento(self._state)
    
    def restore_from_memento(self, memento: Memento) -> None:
        # Restore state from a memento object
        self._state = memento.get_state()
        print(f"State restored to: {self._state}")


class Caretaker:
    """Manages the history of mementos (undo/redo stack)"""
    
    def __init__(self):
        # Stack for undo history
        self._undo_stack = []
        # Stack for redo history
        self._redo_stack = []
    
    def add_memento(self, memento: Memento) -> None:
        # Push current state to undo stack
        self._undo_stack.append(memento)
        # Clear redo stack when new state is added
        self._redo_stack.clear()
    
    def get_memento(self, current_memento: Memento) -> Memento:
        """Undo: move current state to redo stack and return previous state"""
        if len(self._undo_stack) <= 1:
            print("No more states to undo")
            return current_memento
        
        # Save current state to redo stack
        self._redo_stack.append(current_memento)
        # Pop and return previous state
        return self._undo_stack.pop()
    
    def redo(self) -> Memento:
        """Redo: restore next state from redo stack"""
        if not self._redo_stack:
            print("No more states to redo")
            return None
        
        # Pop next state from redo stack
        memento = self._redo_stack.pop()
        # Push it back to undo stack
        self._undo_stack.append(memento)
        return memento


def f():
    """
    Demonstrate undo/redo functionality using Memento pattern
    """
    # Create originator (the text editor)
    editor = Originator()
    # Create caretaker (manages history)
    history = Caretaker()
    
    # Initial state
    editor.set_state("Hello")
    history.add_memento(editor.save_to_memento())
    
    # Make changes
    editor.set_state("Hello World")
    history.add_memento(editor.save_to_memento())
    
    editor.set_state("Hello World!")
    history.add_memento(editor.save_to_memento())
    
    # Undo operation
    print("\n--- Undoing ---")
    current = editor.save_to_memento()
    previous = history.get_memento(current)
    editor.restore_from_memento(previous)
    
    # Redo operation
    print("\n--- Redoing ---")
    next_state = history.redo()
    if next_state:
        editor.restore_from_memento(next_state)
    
    return


if __name__ == "__main__":
    f()
```

**Why**: The Memento pattern captures and externalizes an object's internal state without violating encapsulation, enabling undo/redo functionality critical for editors, games, and transaction systems. It separates state management (Caretaker) from business logic (Originator), ensuring clean architecture and single responsibility.
