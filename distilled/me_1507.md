## [Tim Sort]

```python
# PROBLEM: Hybrid stable sort combining merge sort and insertion sort
# TIME: O(n log n) average/worst, O(n) best SPACE: O(n)
def tim_sort(arr, min_run=32):
    """
    Tim Sort - Python's built-in sorting algorithm.
    Combines merge sort for large runs with insertion sort for small runs.
    Uses galloping mode to optimize merging of already-sorted sequences.
    """
    n = len(arr)
    if n < 2:
        return arr
    
    # Calculate minimum run size (between 32 and 64)
    # Aim for runs that are powers of 2 for efficient merging
    def calc_min_run(n):
        r = 0
        while n >= min_run:
            r |= n & 1  # Track if any bit is set
            n >>= 1
        return n + r
    
    min_run = calc_min_run(n)
    
    # Insertion sort for small runs - efficient for nearly sorted data
    def insertion_sort(arr, left, right):
        for i in range(left + 1, right + 1):
            key = arr[i]
            j = i - 1
            # Shift elements greater than key to the right
            while j >= left and arr[j] > key:
                arr[j + 1] = arr[j]
                j -= 1
            arr[j + 1] = key
    
    # Merge function using temporary storage
    def merge(arr, l, m, r):
        # Copy left and right halves
        left = arr[l:m + 1]
        right = arr[m + 1:r + 1]
        
        i = j = 0
        k = l
        
        # Merge while both halves have elements
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                arr[k] = left[i]
                i += 1
            else:
                arr[k] = right[j]
                j += 1
            k += 1
        
        # Copy remaining elements
        while i < len(left):
            arr[k] = left[i]
            i += 1
            k += 1
        while j < len(right):
            arr[k] = right[j]
            j += 1
            k += 1
    
    # Sort individual runs with insertion sort
    for start in range(0, n, min_run):
        end = min(start + min_run - 1, n - 1)
        insertion_sort(arr, start, end)
    
    # Merge runs of increasing size
    size = min_run
    while size < n:
        # Pick start of left run and merge pairs
        for left in range(0, n, 2 * size):
            mid = min(n - 1, left + size - 1)
            right = min((left + 2 * size - 1), (n - 1))
            
            if mid < right:
                merge(arr, left, mid, right)
        size *= 2
    
    return arr
```
**Why**: Python's default sort - demonstrates hybrid algorithm design, adaptive performance on real-world data patterns.

## [Boyer-Moore]

```python
# PROBLEM: Efficient string matching using bad character and good suffix heuristics
# TIME: O(n/m) best, O(n+m) average, O(nm) worst SPACE: O(k) where k=alphabet size
def boyer_moore(text, pattern):
    """
    Boyer-Moore string matching algorithm.
    Scans pattern right-to-left, uses two heuristics to skip sections:
    1. Bad Character: mismatched char in text - align pattern to next occurrence
    2. Good Suffix: matched suffix - find where it appears elsewhere in pattern
    Achieves sublinear time O(n/m) on typical text.
    """
    n, m = len(text), len(pattern)
    if m == 0:
        return [0] if n >= 0 else []
    if m > n:
        return []
    
    # Build bad character table: rightmost position of each char in pattern
    # If char not in pattern, shift by full pattern length
    def build_bad_char_table(pattern):
        # Store rightmost occurrence of each character
        bad_char = {}
        for i, char in enumerate(pattern):
            bad_char[char] = i
        return bad_char
    
    # Build good suffix table: shift when suffix matches but we have mismatch
    def build_good_suffix_table(pattern):
        m = len(pattern)
        good_suffix = [0] * (m + 1)
        
        # Case 1: suffix appears elsewhere in pattern with different preceding char
        suffix = [0] * (m + 1)
        
        # Compute suffix array using border concept
        f = [0] * (m + 1)  # f[i] = starting position of widest border of suffix i..m-1
        f[m - 1] = m
        g = m - 1
        
        for i in range(m - 2, -1, -1):
            if i > g and f[i + m - 1 - f[m - 1]] < i - g:
                f[i] = f[i + m - 1 - f[m - 1]]
            else:
                g = min(g, i)
                while g >= 0 and pattern[g] == pattern[g + m - 1 - i]:
                    g -= 1
                f[i] = i - g
        
        # Case 2: suffix of suffix matches prefix of pattern
        j = 0
        for i in range(m, -1, -1):
            if f[m - 1] == i:  # suffix i..m-1 matches prefix
                j = i
            good_suffix[m - i] = j + m - i
        
        # Case 3: only a piece of the suffix matches a prefix
        for i in range(m - 1):
            good_suffix[m - 1 - f[i]] = m - 1 - i + f[i]
        
        return good_suffix
    
    bad_char = build_bad_char_table(pattern)
    good_suffix = build_good_suffix_table(pattern)
    
    matches = []
    s = 0  # Shift of pattern with respect to text
    
    while s <= n - m:
        j = m - 1  # Start comparing from rightmost character
        
        # Keep reducing j while characters match
        while j >= 0 and pattern[j] == text[s + j]:
            j -= 1
        
        if j < 0:
            # Pattern found!
            matches.append(s)
            # Shift using good suffix rule for complete match
            s += good_suffix[0] if good_suffix[0] > 0 else 1
        else:
            # Mismatch at position j
            # Bad character shift: align pattern's rightmost occurrence of text[s+j]
            bc_shift = j - bad_char.get(text[s + j], -1)
            bc_shift = max(1, bc_shift)  # Must shift at least 1
            
            # Good suffix shift
            gs_shift = good_suffix[j + 1] if j + 1 < len(good_suffix) else 1
            
            # Take maximum shift from both heuristics
            s += max(bc_shift, gs_shift)
    
    return matches
```
**Why**: Sublinear string search - skips sections without examining every character, practical for text editors and search engines.

## [Dinic's Algorithm]

```python
# PROBLEM: Maximum flow in a network using level graphs and blocking flows
# TIME: O(V²E) general, O(VE log V) with scaling, O(min(V^(2/3), E^(1/2))E) for unit caps
# SPACE: O(V + E)
from collections import deque

class DinicMaxFlow:
    """
    Dinic's algorithm for maximum flow.
    Uses BFS to build level graph (distance from source),
    then DFS to find blocking flows in the level graph.
    Repeats until no augmenting path exists.
    """
    
    def __init__(self, n):
        self.n = n
        self.adj = [[] for _ in range(n)]
        self.capacity = {}  # (u, v) -> capacity
    
    def add_edge(self, u, v, cap):
        """Add directed edge with given capacity."""
        self.adj[u].append(v)
        self.adj[v].append(u)  # Residual edge
        self.capacity[(u, v)] = cap
        self.capacity[(v, u)] = 0  # Residual capacity starts at 0
    
    def bfs_level_graph(self, source, sink):
        """
        Build level graph using BFS.
        level[v] = distance from source in terms of edges.
        Returns True if sink is reachable.
        """
        self.level = [-1] * self.n
        queue = deque([source])
        self.level[source] = 0
        
        while queue:
            u = queue.popleft()
            for v in self.adj[u]:
                # Only traverse edges with positive residual capacity
                if self.level[v] < 0 and self.capacity[(u, v)] > 0:
                    self.level[v] = self.level[u] + 1
                    queue.append(v)
        
        return self.level[sink] >= 0  # Sink reachable?
    
    def dfs_blocking_flow(self, u, sink, flow, ptr):
        """
        DFS to send flow through level graph.
        ptr[u] tracks which edges we've tried from vertex u.
        Returns amount of flow actually sent.
        """
        if u == sink:
            return flow
        
        # Try each edge from u, starting from ptr[u]
        for i in range(ptr[u], len(self.adj[u])):
            v = self.adj[u][i]
            cap = self.capacity[(u, v)]
            
            # Valid edge: has capacity and goes to next level
            if self.level[v] == self.level[u] + 1 and cap > 0:
                # Recursively try to send flow
                pushed = self.dfs_blocking_flow(
                    v, sink, min(flow, cap), ptr
                )
                
                if pushed > 0:
                    # Update residual capacities
                    self.capacity[(u, v)] -= pushed
                    self.capacity[(v, u)] += pushed
                    return pushed
            
            ptr[u] += 1
        
        return 0
    
    def max_flow(self, source, sink):
        """
        Compute maximum flow from source to sink.
        Repeatedly builds level graph and finds blocking flows.
        """
        total_flow = 0
        
        while self.bfs_level_graph(source, sink):
            # Reset pointers for DFS
            ptr = [0] * self.n
            
            # Find all blocking flows in current level graph
            while True:
                flow = self.dfs_blocking_flow(source, sink, float('inf'), ptr)
                if flow == 0:
                    break
                total_flow += flow
        
        return total_flow
    
    def get_flow_edges(self):
        """Return edges with their final flow values."""
        flows = []
        for (u, v), cap in self.capacity.items():
            if cap >= 0:  # Original edge (not residual)
                original_cap = cap + self.capacity[(v, u)]
                if original_cap > 0 and self.capacity[(v, u)] > 0:
                    flows.append((u, v, self.capacity[(v, u)], original_cap))
        return flows
```
**Why**: Faster than Ford-Fulkerson for dense graphs, used in bipartite matching, image segmentation, and network routing.

## [Treap]

```python
# PROBLEM: Self-balancing BST using random priorities for expected O(log n) operations
# TIME: O(log n) expected for all operations SPACE: O(n)
import random

class TreapNode:
    """Node in a Treap - combines BST key with heap priority."""
    def __init__(self, key, priority=None):
        self.key = key
        self.priority = priority if priority else random.random()
        self.left = None
        self.right = None
        self.size = 1  # Subtree size for order statistics
    
    def update_size(self):
        """Recalculate subtree size after modifications."""
        self.size = 1
        if self.left:
            self.size += self.left.size
        if self.right:
            self.size += self.right.size


class Treap:
    """
    Treap (Tree + Heap): BST by key, max-heap by priority.
    Random priorities ensure expected O(log n) height.
    No complex rotation rules like AVL or Red-Black trees.
    """
    
    def __init__(self):
        self.root = None
    
    def _rotate_right(self, y):
        """Right rotation - promotes left child."""
        x = y.left
        y.left = x.right
        x.right = y
        
        # Update sizes bottom-up
        y.update_size()
        x.update_size()
        return x
    
    def _rotate_left(self, x):
        """Left rotation - promotes right child."""
        y = x.right
        x.right = y.left
        y.left = x
        
        # Update sizes bottom-up
        x.update_size()
        y.update_size()
        return y
    
    def insert(self, key):
        """Insert key into treap."""
        self.root = self._insert(self.root, key)
    
    def _insert(self, node, key):
        """Recursive insert with heap property maintenance."""
        if node is None:
            return TreapNode(key)
        
        if key < node.key:
            node.left = self._insert(node.left, key)
            # Heap property: parent should have higher priority
            if node.left.priority > node.priority:
                node = self._rotate_right(node)
        elif key > node.key:
            node.right = self._insert(node.right, key)
            if node.right.priority > node.priority:
                node = self._rotate_left(node)
        # Equal keys: no duplicate insertion
        
        node.update_size()
        return node
    
    def delete(self, key):
        """Delete key from treap."""
        self.root = self._delete(self.root, key)
    
    def _delete(self, node, key):
        """Recursive delete - rotate down to leaf then remove."""
        if node is None:
            return None
        
        if key < node.key:
            node.left = self._delete(node.left, key)
        elif key > node.key:
            node.right = self._delete(node.right, key)
        else:
            # Found node to delete - rotate it to a leaf
            if node.left is None:
                return node.right
            elif node.right is None:
                return node.left
            else:
                # Has two children - rotate with higher priority child
                if node.left.priority > node.right.priority:
                    node = self._rotate_right(node)
                    node.right = self._delete(node.right, key)
                else:
                    node = self._rotate_left(node)
                    node.left = self._delete(node.left, key)
        
        if node:
            node.update_size()
        return node
    
    def search(self, key):
        """Search for key - returns True if found."""
        return self._search(self.root, key)
    
    def _search(self, node, key):
        if node is None:
            return False
        if key == node.key:
            return True
        elif key < node.key:
            return self._search(node.left, key)
        else:
            return self._search(node.right, key)
    
    def kth_smallest(self, k):
        """Find kth smallest element (1-indexed)."""
        return self._kth_smallest(self.root, k)
    
    def _kth_smallest(self, node, k):
        if node is None:
            return None
        
        left_size = node.left.size if node.left else 0
        
        if k <= left_size:
            return self._kth_smallest(node.left, k)
        elif k == left_size + 1:
            return node.key
        else:
            return self._kth_smallest(node.right, k - left_size - 1)
    
    def inorder(self):
        """Return sorted list of all keys."""
        result = []
        self._inorder(self.root, result)
        return result
    
    def _inorder(self, node, result):
        if node:
            self._inorder(node.left, result)
            result.append(node.key)
            self._inorder(node.right, result)
```
**Why**: Simpler than AVL/RB trees, expected O(log n) without complex cases, excellent for randomized algorithms and implicit treaps.

## [Flyweight Pattern]

```python
# PROBLEM: Minimize memory usage by sharing common state between many similar objects
# TIME: O(1) for accessing shared state SPACE: O(n) shared + O(m) unique state
class Flyweight:
    """
    Flyweight pattern: Share intrinsic (shared) state across objects,
    keep extrinsic (unique) state separate.
    Useful when creating thousands/millions of similar objects.
    """
    def __init__(self, shared_state):
        self._shared_state = shared_state  # Intrinsic state - shared
    
    def operation(self, unique_state):
        """
        Operation uses both shared and unique state.
        Shared state is internal, unique state is passed by client.
        """
        return f"Shared: {self._shared_state}, Unique: {unique_state}"


class FlyweightFactory:
    """
    Factory manages flyweight objects.
    Returns existing flyweight if shared_state already used,
    creates new one otherwise.
    """
    _flyweights = {}  # shared_state -> Flyweight
    
    @classmethod
    def get_flyweight(cls, shared_state):
        """Get or create flyweight with given shared state."""
        key = cls._make_key(shared_state)
        
        if key not in cls._flyweights:
            print(f"Creating new flyweight for: {shared_state}")
            cls._flyweights[key] = Flyweight(shared_state)
        else:
            print(f"Reusing existing flyweight for: {shared_state}")
        
        return cls._flyweights[key]
    
    @classmethod
    def _make_key(cls, shared_state):
        """Convert shared state to hashable key."""
        if isinstance(shared_state, dict):
            return tuple(sorted(shared_state.items()))
        return shared_state
    
    @classmethod
    def count(cls):
        """Return number of flyweights created."""
        return len(cls._flyweights)


# Example: Text editor with character formatting
class CharacterFlyweight:
    """Shared character formatting - font, size, color."""
    def __init__(self, font, size, color):
        self.font = font
        self.size = size
        self.color = color
    
    def render(self, char, position):
        """Render character at position using this formatting."""
        return f"'{char}' at ({position}) using {self.font} {self.size}pt {self.color}"


class CharacterFactory:
    """Factory for character formatting flyweights."""
    _formats = {}
    
    @classmethod
    def get_format(cls, font="Arial", size=12, color="black"):
        """Get or create formatting flyweight."""
        key = (font, size, color)
        if key not in cls._formats:
            cls._formats[key] = CharacterFlyweight(font, size, color)
        return cls._formats[key]
    
    @classmethod
    def format_count(cls):
        return len(cls._formats)


class TextEditor:
    """
    Text editor using flyweights.
    Each character stores only: char value + position + reference to format.
    Format objects are shared across all characters with same formatting.
    """
    def __init__(self):
        self.characters = []  # List of (char, position, format_flyweight)
    
    def insert(self, char, position, font="Arial", size=12, color="black"):
        """Insert character with formatting."""
        format_fw = CharacterFactory.get_format(font, size, color)
        self.characters.append((char, position, format_fw))
    
    def render(self):
        """Render all characters."""
        return [fmt.render(char, pos) for char, pos, fmt in self.characters]
    
    def stats(self):
        """Show memory savings."""
        total_chars = len(self.characters)
        unique_formats = CharacterFactory.format_count()
        # Without flyweight: each char would store font, size, color
        # With flyweight: each char stores reference to shared format
        memory_saved = total_chars * 3 - unique_formats * 3  # Rough estimate
        return f"Characters: {total_chars}, Unique formats: {unique_formats}"


# Usage example
if __name__ == "__main__":
    editor = TextEditor()
    
    # Insert many characters - only 2 format objects created
    for i, char in enumerate("Hello World"):
        if i < 5:
            editor.insert(char, i, "Arial", 12, "black")
        else:
            editor.insert(char, i, "Times", 14, "red")
    
    print(editor.stats())
    # Output: Characters: 11, Unique formats: 2
```
**Why**: Dramatically reduces memory for millions of similar objects - used in game engines (trees, particles), text editors, and UI frameworks.

```python
# PROBLEM: Coordinate access to shared resources across distributed nodes
# TIME: O(1) for lock operations, O(n) for quorum checks SPACE: O(k) for k locks
import time
import uuid
import threading
from contextlib import contextmanager

class DistributedLock:
    """
    Distributed lock implementation concepts.
    Key challenges: network partitions, node failures, clock skew.
    Key solutions: leases, fencing tokens, quorum.
    """
    pass


class RedLock:
    """
    Redlock algorithm (Redis-based distributed lock).
    Uses multiple Redis instances for fault tolerance.
    Acquires lock on majority of nodes within TTL.
    """
    
    def __init__(self, redis_clients, lock_name, ttl_ms=10000):
        """
        redis_clients: list of Redis client connections to different nodes
        lock_name: name of the lock resource
        ttl_ms: time-to-live in milliseconds (auto-release timeout)
        """
        self.redis_clients = redis_clients
        self.lock_name = lock_name
        self.ttl_ms = ttl_ms
        self.lock_value = None  # Unique identifier for this lock holder
        self.acquired = False
    
    def acquire(self, retry_times=3, retry_delay_ms=200):
        """
        Try to acquire lock on majority of Redis nodes.
        Returns True if lock acquired, False otherwise.
        """
        self.lock_value = str(uuid.uuid4())  # Unique per lock attempt
        quorum = len(self.redis_clients) // 2 + 1  # Majority
        
        for attempt in range(retry_times):
            start_time = time.time()
            acquired_count = 0
            
            # Try to acquire lock on each node
            for client in self.redis_clients:
                try:
                    # SET with NX (only if not exists) and PX (expiry)
                    acquired = client.set(
                        self.lock_name,
                        self.lock_value,
                        nx=True,
                        px=self.ttl_ms
                    )
                    if acquired:
                        acquired_count += 1
                except Exception as e:
                    print(f"Error acquiring from node: {e}")
            
            # Calculate time elapsed
            elapsed_ms = (time.time() - start_time) * 1000
            validity = self.ttl_ms - elapsed_ms
            
            # Lock acquired if:
            # 1. Acquired on majority of nodes
            # 2. Remaining validity time is positive
            if acquired_count >= quorum and validity > 0:
                self.acquired = True
                return True
            
            # Failed - release any locks we did acquire
            self._release_internal()
            
            # Wait before retry
            time.sleep(retry_delay_ms / 1000)
        
        return False
    
    def release(self):
        """Release lock only if we still hold it."""
        if self.acquired:
            self._release_internal()
            self.acquired = False
    
    def _release_internal(self):
        """
        Release lock using Lua script for atomicity.
        Only releases if lock_value matches (prevents releasing others' locks).
        """
        lua_script = """
        if redis.call("get", KEYS[1]) == ARGV[1] then
            return redis.call("del", KEYS[1])
        else
            return 0
        end
        """
        
        for client in self.redis_clients:
            try:
                client.eval(lua_script, 1, self.lock_name, self.lock_value)
            except Exception as e:
                print(f"Error releasing from node: {e}")


class FencedLock:
    """
    Fencing token approach for distributed locking.
    Generates monotonically increasing tokens.
    Services only accept requests with highest token seen.
    Protects against delayed/stale lock holders.
    """
    
    def __init__(self, lock_service):
        self.lock_service = lock_service
        self.fencing_token = 0
        self.lock_held = False
    
    def acquire(self):
        """Acquire lock and get fencing token."""
        self.fencing_token = self.lock_service.get_next_token()
        self.lock_held = True
        return self.fencing_token
    
    def release(self):
        """Release lock."""
        self.lock_held = False
    
    def get_token(self):
        """Get current fencing token for requests."""
        if not self.lock_held:
            raise RuntimeError("Lock not held")
        return self.fencing_token


class LockService:
    """Simulated lock service providing monotonically increasing tokens."""
    
    def __init__(self):
        self.counter = 0
        self.lock = threading.Lock()
    
    def get_next_token(self):
        """Get next fencing token atomically."""
        with self.lock:
            self.counter += 1
            return self.counter


class ZookeeperLock:
    """
    ZooKeeper-based distributed lock using ephemeral nodes.
    Uses sequential ephemeral nodes for fair ordering.
    Watches predecessor node for notification.
    """
    
    def __init__(self, zk_client, lock_path):
        self.zk = zk_client
        self.lock_path = lock_path
        self.lock_node = None
    
    def acquire(self, timeout=30):
        """
        Acquire lock by creating sequential ephemeral node.
        If we have lowest sequence number, we hold the lock.
        Otherwise, wait for predecessor to be deleted.
        """
        # Create sequential ephemeral node
        self.lock_node = self.zk.create(
            f"{self.lock_path}/lock-",
            ephemeral=True,
            sequence=True
        )
        
        start_time = time.time()
        
        while True:
            # Get all lock candidates
            children = self.zk.get_children(self.lock_path)
            children.sort()
            
            # Extract our sequence number
            my_seq = self.lock_node.split('-')[-1]
            
            # Find our position
            for i, child in enumerate(children):
                if child.endswith(my_seq):
                    if i == 0:
                        # We're first - lock acquired!
                        return True
                    
                    # Watch predecessor
                    predecessor = f"{self.lock_path}/{children[i-1]}"
                    if self.zk.exists(predecessor, watch=True):
                        # Wait for watch event or timeout
                        elapsed = time.time() - start_time
                        if elapsed >= timeout:
                            self._cleanup()
                            return False
                        time.sleep(0.1)
                    break
        
        return False
    
    def release(self):
        """Release lock by deleting our ephemeral node."""
        if self.lock_node:
            self.zk.delete(self.lock_node)
            self.lock_node = None
    
    def _cleanup(self):
        """Clean up lock node on failure."""
        if self.lock_node:
            try:
                self.zk.delete(self.lock_node)
            except:
                pass
            self.lock_node = None


@contextmanager
def distributed_lock(lock):
    """Context manager for distributed lock."""
    acquired = lock.acquire()
    try:
        if not acquired:
            raise RuntimeError("Failed to acquire distributed lock")
        yield lock
    finally:
        lock.release()
```
**Why**: Essential for distributed systems - prevents race conditions across nodes, handles failures gracefully with TTLs and fencing tokens.
