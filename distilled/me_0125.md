```python
# PROBLEM: Implement template method pattern with multiple hook points and variant behaviors
# TIME: O(1) SPACE: O(1)
from abc import ABC, abstractmethod
from typing import Optional, List, Any

class DataProcessor(ABC):
    """Abstract base class defining the template method algorithm skeleton."""
    
    # Template method - defines the algorithm structure
    def process(self, data: List[Any]) -> Optional[List[Any]]:
        """Main template method that orchestrates the processing pipeline."""
        # Step 1: Validate input data
        if not self.validate(data):
            return None
        
        # Step 2: Pre-processing hook (optional)
        data = self.before_process(data)
        
        # Step 3: Core processing (required - abstract)
        result = self.do_process(data)
        
        # Step 4: Post-processing hook (optional)
        result = self.after_process(result)
        
        # Step 5: Finalize and return
        self.finalize(result)
        return result
    
    # Abstract method - must be implemented by subclasses
    @abstractmethod
    def do_process(self, data: List[Any]) -> List[Any]:
        """Core processing logic - must be implemented by concrete classes."""
        pass
    
    # Hook methods - optional customization points
    def validate(self, data: List[Any]) -> bool:
        """Hook: Validate input data. Default accepts all."""
        return data is not None and len(data) > 0
    
    def before_process(self, data: List[Any]) -> List[Any]:
        """Hook: Pre-processing transformation. Default returns unchanged."""
        return data
    
    def after_process(self, result: List[Any]) -> List[Any]:
        """Hook: Post-processing transformation. Default returns unchanged."""
        return result
    
    def finalize(self, result: List[Any]) -> None:
        """Hook: Cleanup or logging after processing. Default does nothing."""
        pass


class JsonProcessor(DataProcessor):
    """Concrete implementation for JSON data processing."""
    
    def do_process(self, data: List[Any]) -> List[Any]:
        # Core processing logic for JSON data
        return [{"processed": item} for item in data]
    
    def validate(self, data: List[Any]) -> bool:
        # Override validation for JSON-specific requirements
        return super().validate(data) and all(isinstance(item, dict) for item in data)
    
    def before_process(self, data: List[Any]) -> List[Any]:
        # Clean JSON data before processing
        return [{k: v for k, v in item.items() if v is not None} for item in data]


# Usage
processor = JsonProcessor()
result = processor.process([{"name": "Alice", "age": None}, {"name": "Bob", "age": 25}])
```
**Why**: Template method pattern provides a flexible algorithm skeleton with customizable hooks, enabling code reuse while allowing subclasses to override specific steps without changing the overall structure.

```python
# PROBLEM: Implement a simple IoC container for dependency injection and service management
# TIME: O(1) for registration, O(n) for resolution where n is dependency depth SPACE: O(m) where m is number of services
from typing import Dict, Type, Any, Callable, Optional
from dataclasses import dataclass
from enum import Enum

class ServiceLifetime(Enum):
    SINGLETON = "singleton"
    TRANSIENT = "transient"

@dataclass
class ServiceDescriptor:
    """Describes how a service should be instantiated and managed."""
    service_type: Type
    implementation: Callable
    lifetime: ServiceLifetime
    instance: Optional[Any] = None

class IoCContainer:
    """Simple Inversion of Control container for dependency management."""
    
    def __init__(self):
        self._services: Dict[Type, ServiceDescriptor] = {}
        self._singletons: Dict[Type, Any] = {}
    
    def register_singleton(self, service_type: Type, implementation: Callable) -> 'IoCContainer':
        """Register a service with singleton lifetime."""
        self._services[service_type] = ServiceDescriptor(
            service_type=service_type,
            implementation=implementation,
            lifetime=ServiceLifetime.SINGLETON
        )
        return self  # Enable fluent registration
    
    def register_transient(self, service_type: Type, implementation: Callable) -> 'IoCContainer':
        """Register a service with transient lifetime (new instance each time)."""
        self._services[service_type] = ServiceDescriptor(
            service_type=service_type,
            implementation=implementation,
            lifetime=ServiceLifetime.TRANSIENT
        )
        return self
    
    def resolve(self, service_type: Type) -> Any:
        """Resolve and return an instance of the requested service."""
        if service_type not in self._services:
            raise ValueError(f"Service {service_type.__name__} not registered")
        
        descriptor = self._services[service_type]
        
        # Return existing singleton if available
        if descriptor.lifetime == ServiceLifetime.SINGLETON:
            if service_type in self._singletons:
                return self._singletons[service_type]
            
            # Create and cache singleton instance
            instance = descriptor.implementation(self)
            self._singletons[service_type] = instance
            return instance
        
        # Create new transient instance
        return descriptor.implementation(self)
    
    def get_instance(self, service_type: Type) -> Any:
        """Alias for resolve() - more intuitive API."""
        return self.resolve(service_type)


# Example usage
class DatabaseService:
    def __init__(self, connection_string: str):
        self.connection_string = connection_string
    
    def query(self, sql: str) -> list:
        return [{"data": "example"}]
```

```python
# PROBLEM: Implement interceptors for cross-cutting concerns like logging, authentication, and caching
# TIME: O(n) where n is number of interceptors SPACE: O(n)
from typing import Callable, Any, List, Optional
from dataclasses import dataclass
from functools import wraps
import time

@dataclass
class RequestContext:
    """Context object passed through the interceptor chain."""
    method: str
    params: dict
    metadata: dict
    response: Any = None
    error: Optional[Exception] = None

class Interceptor(ABC):
    """Abstract base class for all interceptors."""
    
    @abstractmethod
    def intercept(self, context: RequestContext, next_interceptor: Callable) -> Any:
        """Process request and delegate to next interceptor."""
        pass

from abc import ABC

class LoggingInterceptor(Interceptor):
    """Interceptor for request/response logging."""
    
    def intercept(self, context: RequestContext, next_interceptor: Callable) -> Any:
        print(f"[LOG] {context.method} called with params: {context.params}")
        start_time = time.time()
        
        try:
            result = next_interceptor(context)
            context.response = result
            elapsed = time.time() - start_time
            print(f"[LOG] {context.method} completed in {elapsed:.3f}s")
            return result
        except Exception as e:
            context.error = e
            print(f"[LOG] {context.method} failed with error: {e}")
            raise

class CachingInterceptor(Interceptor):
    """Interceptor for response caching."""
    
    def __init__(self):
        self._cache: dict = {}
    
    def intercept(self, context: RequestContext, next_interceptor: Callable) -> Any:
        # Create cache key from method and params
        cache_key = f"{context.method}:{str(sorted(context.params.items()))}"
        
        # Check cache
        if cache_key in self._cache:
            print(f"[CACHE] Hit for {context.method}")
            return self._cache[cache_key]
        
        # Execute and cache result
        result = next_interceptor(context)
        self._cache[cache_key] = result
        print(f"[CACHE] Stored for {context.method}")
        return result

class InterceptorChain:
    """Manages a chain of interceptors."""
    
    def __init__(self):
        self._interceptors: List[Interceptor] = []
    
    def add(self, interceptor: Interceptor) -> 'InterceptorChain':
        """Add interceptor to the chain."""
        self._interceptors.append(interceptor)
        return self
    
    def execute(self, context: RequestContext, target: Callable) -> Any:
        """Execute the interceptor chain and finally the target."""
        def create_chain(index: int) -> Callable:
            if index >= len(self._interceptors):
                return lambda ctx: target(ctx)
            
            interceptor = self._interceptors[index]
            return lambda ctx: interceptor.intercept(ctx, create_chain(index + 1))
        
        chain = create_chain(0)
        return chain(context)

# Usage
def my_service_method(context: RequestContext) -> dict:
    return {"result": "success", "data": [1, 2, 3]}

chain = InterceptorChain()
chain.add(LoggingInterceptor()).add(CachingInterceptor())

ctx = RequestContext(method="get_data", params={"id": 123}, metadata={})
result = chain.execute(ctx, my_service_method)
```
**Why**: Interceptor pattern separates cross-cutting concerns from business logic, enabling clean, reusable implementations of logging, caching, authentication, and other aspects.

```python
# PROBLEM: Implement an open host service pattern for exposing domain operations via well-defined protocols
# TIME: O(1) for routing, O(n) for handler execution SPACE: O(m) where m is number of services
from typing import Dict, Callable, Any, Optional
from dataclasses import dataclass
from enum import Enum
import json

class ProtocolType(Enum):
    REST = "rest"
    GRPC = "grpc"
    GRAPHQL = "graphql"

@dataclass
class ServiceRequest:
    """Encapsulates a service request."""
    service_name: str
    operation: str
    payload: dict
    protocol: ProtocolType
    headers: dict

@dataclass
class ServiceResponse:
    """Encapsulates a service response."""
    success: bool
    data: Any
    error: Optional[str] = None
    status_code: int = 200

class OpenHostService:
    """
    Open Host Service pattern implementation.
    Exposes domain services through multiple protocols with a unified interface.
    """
    
    def __init__(self, service_name: str):
        self.service_name = service_name
        self._operations: Dict[str, Callable] = {}
        self._protocols: Dict[ProtocolType, Callable] = {}
    
    def register_operation(self, name: str, handler: Callable) -> 'OpenHostService':
        """Register a domain operation with its handler."""
        self._operations[name] = handler
        return self
    
    def register_protocol(self, protocol: ProtocolType, adapter: Callable) -> 'OpenHostService':
        """Register a protocol adapter for handling requests."""
        self._protocols[protocol] = adapter
        return self
    
    def handle_request(self, request: ServiceRequest) -> ServiceResponse:
        """
        Main entry point for all service requests.
        Routes to appropriate operation and returns response.
        """
        # Validate service
        if request.service_name != self.service_name:
            return ServiceResponse(
                success=False,
                data=None,
                error=f"Unknown service: {request.service_name}",
                status_code=404
            )
        
        # Validate operation
        if request.operation not in self._operations:
            return ServiceResponse(
                success=False,
                data=None,
                error=f"Unknown operation: {request.operation}",
                status_code=404
            )
        
        # Get protocol adapter
        protocol_adapter = self._protocols.get(request.protocol)
        if not protocol_adapter:
            return ServiceResponse(
                success=False,
                data=None,
                error=f"Unsupported protocol: {request.protocol}",
                status_code=400
            )
        
        try:
            # Adapt request for the operation
            adapted_request = protocol_adapter(request)
            
            # Execute operation
            handler = self._operations[request.operation]
            result = handler(adapted_request)
            
            return ServiceResponse(success=True, data=result, status_code=200)
        
        except Exception as e:
            return ServiceResponse(
                success=False,
                data=None,
                error=str(e),
                status_code=500
            )


# Example domain service
class OrderService(OpenHostService):
    """Order domain service exposed as Open Host Service."""
    
    def __init__(self):
        super().__init__("order-service")
        
        # Register domain operations
        self.register_operation("create_order", self._create_order)
        self.register_operation("get_order", self._get_order)
        
        # Register protocol adapters
        self.register_protocol(ProtocolType.REST, self._adapt_rest)
        self.register_protocol(ProtocolType.GRAPHQL, self._adapt_graphql)
    
    def _create_order(self, request: dict) -> dict:
        """Create a new order."""
        return {"order_id": "ORD-12345", "status": "created"}
    
    def _get_order(self, request: dict) -> dict:
        """Retrieve an existing order."""
        return {"order_id": request.get("order_id"), "status": "shipped"}
    
    def _adapt_rest(self, request: ServiceRequest) -> dict:
        """Adapt REST request to domain format."""
        return request.payload
    
    def _adapt_graphql(self, request: ServiceRequest) -> dict:
        """Adapt GraphQL request to domain format."""
        return request.payload.get("variables", {})


# Usage
order_service = OrderService()
rest_request = ServiceRequest(
    service_name="order-service",
    operation="create_order",
    payload={"customer_id": "CUST-001", "items": ["item1", "item2"]},
    protocol=ProtocolType.REST,
    headers={}
)
response = order_service.handle_request(rest_request)
```
**Why**: Open Host Service provides a well-defined, protocol-agnostic interface for exposing domain services, enabling multiple client types to consume services while maintaining a clean separation between protocol and domain logic.

```python
# PROBLEM: Implement a shared language/model for communication between bounded contexts
# TIME: O(1) for serialization, O(n) for validation SPACE: O(n)
from typing import Dict, Any, Type, Optional
from dataclasses import dataclass, asdict
from datetime import datetime
import json

@dataclass
class PublishedEvent:
    """Base class for all published language events."""
    event_type: str
    timestamp: str
    version: str = "1.0"
    
    def to_dict(self) -> Dict[str, Any]:
        """Serialize event to dictionary."""
        return asdict(self)
    
    def to_json(self) -> str:
        """Serialize event to JSON string."""
        return json.dumps(self.to_dict())

class PublishedLanguageRegistry:
    """
    Registry for managing published language schemas and transformations.
    Acts as the shared vocabulary between bounded contexts.
    """
    
    def __init__(self):
        self._schemas: Dict[str, Type] = {}
        self._transformers: Dict[str, Callable] = {}
    
    def register_schema(self, event_type: str, schema_class: Type) -> None:
        """Register a schema class for an event type."""
        self._schemas[event_type] = schema_class
    
    def register_transformer(
        self,
        source_context: str,
        target_context: str,
        transformer: Callable
    ) -> None:
        """Register a transformer for converting between contexts."""
        key = f"{source_context}:{target_context}"
        self._transformers[key] = transformer
    
    def parse_event(self, data: Dict[str, Any]) -> Optional[PublishedEvent]:
        """Parse raw data into a typed event object."""
        event_type = data.get("event_type")
        if event_type not in self._schemas:
            return None
        
        schema_class = self._schemas[event_type]
        return schema_class(**data)
    
    def transform(
        self,
        event: PublishedEvent,
        target_context: str,
        source_context: str
    ) -> Dict[str, Any]:
        """Transform event from source context to target context format."""
        key = f"{source_context}:{target_context}"
        transformer = self._transformers.get(key)
        
        if transformer:
            return transformer(event.to_dict())
        
        # Default: return as-is if no transformer
        return event.to_dict()
```

## BRIN Index Implementation
```python
# PROBLEM: Implement Block Range Index (BRIN) for summarizing large ranges of data
# TIME: O(log n) for lookup, O(1) for insertion SPACE: O(n/r) where r is pages_per_range
from typing import List, Tuple, Optional, Any
from dataclasses import dataclass
import struct

@dataclass
class BRINTuple:
    """Represents a single BRIN index tuple with min/max values."""
    block_start: int
    block_end: int
    min_value: Any
    max_value: Any
    has_nulls: bool = False

class BRINIndex:
    """
    Block Range INdex implementation for columnar data.
    Stores summary information about blocks of data.
    """
    
    def __init__(self, pages_per_range: int = 128):
        self.pages_per_range = pages_per_range
        self.tuples: List[BRINTuple] = []
        self._current_block_data: List[Any] = []
        self._current_block_start: int = 0
    
    def insert(self, value: Any) -> None:
        """Insert a value into the current block range."""
        self._current_block_data.append(value)
        
        # Check if block range is full
        if len(self._current_block_data) >= self.pages_per_range:
            self._finalize_block()
    
    def _finalize_block(self) -> None:
        """Finalize current block and create BRIN tuple."""
        if not self._current_block_data:
            return
        
        # Calculate min/max for the block
        non_null_values = [v for v in self._current_block_data if v is not None]
        has_nulls = len(non_null_values) < len(self._current_block_data)
        
        if non_null_values:
            min_val = min(non_null_values)
            max_val = max(non_null_values)
        else:
            min_val = max_val = None
        
        # Create BRIN tuple
        block_end = self._current_block_start + len(self._current_block_data) - 1
        brin_tuple = BRINTuple(
            block_start=self._current_block_start,
            block_end=block_end,
            min_value=min_val,
            max_value=max_val,
            has_nulls=has_nulls
        )
        
        self.tuples.append(brin_tuple)
        
        # Reset for next block
        self._current_block_start = block_end + 1
        self._current_block_data = []
    
    def scan(self, value: Any) -> List[Tuple[int, int]]:
        """
        Scan index for blocks that might contain the value.
        Returns list of (block_start, block_end) tuples.
        """
        matching_blocks = []
        
        for brin_tuple in self.tuples:
            # Check if value falls within min/max range
            if brin_tuple.min_value is not None and brin_tuple.max_value is not None:
                if brin_tuple.min_value <= value <= brin_tuple.max_value:
                    matching_blocks.append(
                        (brin_tuple.block_start, brin_tuple.block_end)
                    )
        
        return matching_blocks
    
    def range_scan(self, min_val: Any, max_val: Any) -> List[Tuple[int, int]]:
        """Scan for blocks overlapping with value range."""
        matching_blocks = []
        
        for brin_tuple in self.tuples:
            if brin_tuple.max_value < min_val or brin_tuple.min_value > max_val:
                continue
            matching_blocks.append(
                (brin_tuple.block_start, brin_tuple.block_end)
            )
        
        return matching_blocks
    
    def get_size_estimate(self) -> int:
        """Estimate index size in bytes."""
        # Each tuple: 4 ints (32 bytes) + 2 values + 1 bool
        return len(self.tuples) * (32 + 16 + 1)
    
    def vacuum(self) -> None:
        """Compact index by merging adjacent compatible blocks."""
        if len(self.tuples) < 2:
            return
        
        merged = [self.tuples[0]]
        
        for current in self.tuples[1:]:
            previous = merged[-1]
            
            # Check if blocks are adjacent and can be merged
            if (current.block_start == previous.block_end + 1 and
                current.min_value >= previous.min_value and
                current.max_value <= previous.max_value):
                # Extend previous block
                merged[-1] = BRINTuple(
                    block_start=previous.block_start,
                    block_end=current.block_end,
                    min_value=previous.min_value,
                    max_value=previous.max_value,
                    has_nulls=previous.has_nulls or current.has_nulls
                )
            else:
                merged.append(current)
        
        self.tuples = merged


# Usage
brin_index = BRINIndex(pages_per_range=100)

# Insert values
for i in range(500):
    brin_index.insert(i * 2)  # Even numbers 0 to 998

brin_index._finalize_block()  # Finalize last block

# Query for blocks containing value 250
blocks = brin_index.scan(250)
print(f"Value 250 might be in blocks: {blocks}")

# Range query
range_blocks = brin_index.range_scan(100, 200)
print(f"Range [100, 200] overlaps with blocks: {range_blocks}")
```
**Why**: BRIN indexes provide extremely compact indexing for large, naturally-ordered datasets by storing only summary information (min/max) per block range, making them ideal for time-series and append-only data.

## SP-GiST Index Implementation
```python
# PROBLEM: Implement Space-Partitioned GiST index for efficient space partitioning
# TIME: O(log n) for search/insert SPACE: O(n)
from typing import List, Optional, Any, Tuple, Callable
from dataclasses import dataclass, field
from enum import Enum

class PartitionType(Enum):
    """Types of space partitioning strategies."""
    QUADTREE = "quadtree"
    KD_TREE = "kdtree"
    RADIX = "radix"

@dataclass
class SPGistNode:
    """Node in SP-GiST tree."""
    level: int
    partition_value: Any = None
    children: dict = field(default_factory=dict)
    leaf_tuples: List[Tuple[Any, Any]] = field(default_factory=list)

class SPGistIndex:
    """
    Space-Partitioned GiST index implementation.
    Supports various partitioning strategies for different data types.
    """
    
    def __init__(self, partition_type: PartitionType = PartitionType.QUADTREE):
        self.partition_type = partition_type
        self.root: Optional[SPGistNode] = None
        self._partition_func: Callable = self._get_partition_func()
    
    def _get_partition_func(self) -> Callable:
        """Get partition function based on type."""
        if self.partition_type == PartitionType.QUADTREE:
            return self._quadtree_partition
        elif self.partition_type == PartitionType.KD_TREE:
            return self._kdtree_partition
        else:
            return self._radix_partition
    
    def _quadtree_partition(
        self,
        point: Tuple[float, float],
        level: int,
        bounds: Tuple[float, float, float, float]
    ) -> int:
        """Partition 2D point into quadrant (0-3)."""
        x, y = point
        min_x, min_y, max_x, max_y = bounds
        mid_x = (min_x + max_x) / 2
        mid_y = (min_y + max_y) / 2
        
        quadrant = 0
        if x >= mid_x:
            quadrant |= 1
        if y >= mid_y:
            quadrant |= 2
        return quadrant
    
    def _kdtree_partition(
        self,
        point: Tuple[float, float],
        level: int,
        bounds: Tuple[float, float, float, float]
    ) -> int:
        """Partition point based on alternating dimension split."""
        dim = level % 2  # Alternate between x and y
        mid = (bounds[dim] + bounds[dim + 2]) / 2
        
        return 0 if point[dim] < mid else 1
    
    def _radix_partition(
        self,
        value: int,
        level: int,
        bounds: Tuple[int, int]
    ) -> int:
        """Partition integer by radix bits."""
        shift = 28 - (level * 4)  # Process 4 bits at a time
        return (value >> shift) & 0xF if shift >= 0 else 0
    
    def insert(
        self,
        key: Any,
        value: Any,
        bounds: Tuple = (0, 0, 100, 100)
    ) -> None:
        """Insert key-value pair into the index."""
        if self.root is None:
            self.root = SPGistNode(level=0)
        
        self._insert_recursive(self.root, key, value, 0, bounds)
    
    def _insert_recursive(
        self,
        node: SPGistNode,
        key: Any,
        value: Any,
        level: int,
        bounds: Tuple
    ) -> None:
        """Recursively insert into the tree."""
        partition = self._partition_func(key, level, bounds)
        
        if partition not in node.children:
            # Create new child node
            node.children[partition] = SPGistNode(level=level + 1)
        
        child = node.children[partition]
        
        # Check if should be leaf or internal node
        if level >= 8:  # Max depth reached, store as leaf
            child.leaf_tuples.append((key, value))
        else:
            new_bounds = self._get_child_bounds(bounds, partition)
            self._insert_recursive(child, key, value, level + 1, new_bounds)
    
    def _get_child_bounds(
        self,
        bounds: Tuple,
        partition: int
    ) -> Tuple:
        """Calculate bounds for child partition."""
        min_x, min_y, max_x, max_y = bounds
        mid_x = (min_x + max_x) / 2
        mid_y = (min_y + max_y) / 2
        
        if partition == 0:
            return (min_x, min_y, mid_x, mid_y)
        elif partition == 1:
            return (mid_x, min_y, max_x, mid_y)
        elif partition == 2:
            return (min_x, mid_y, mid_x, max_y)
        else:
            return (mid_x, mid_y, max_x, max_y)
    
    def search(self, key: Any, bounds: Tuple = (0, 0, 100, 100)) -> List[Any]:
        """Search for values matching the key."""
        if self.root is None:
            return []
        
        results = []
        self._search_recursive(self.root, key, 0, bounds, results)
        return results
    
    def _search_recursive(
        self,
        node: SPGistNode,
        key: Any,
        level: int,
        bounds: Tuple,
        results: List
    ) -> None:
        """Recursively search for matching keys."""
        # Check leaf tuples
        for stored_key, value in node.leaf_tuples:
            if stored_key == key:
                results.append(value)
        
        # Navigate to appropriate partition
        partition = self._partition_func(key, level, bounds)
        
        if partition in node.children:
            new_bounds = self._get_child_bounds(bounds, partition)
            self._search_recursive(
                node.children[partition],
                key,
                level + 1,
                new_bounds,
                results
            )
    
    def range_query(
        self,
        query_bounds: Tuple[float, float, float, float]
    ) -> List[Tuple[Any, Any]]:
        """Find all points within query bounds."""
        if self.root is None:
            return []
        
        results = []
        self._range_query_recursive(
            self.root,
            query_bounds,
            (0, 0, 100, 100),
            results
        )
        return results
    
    def _range_query_recursive(
        self,
        node: SPGistNode,
        query_bounds: Tuple[float, float, float, float],
        node_bounds: Tuple[float, float, float, float],
        results: List
    ) -> None:
        """Recursively find points in range."""
        # Check leaf tuples
        for key, value in node.leaf_tuples:
            x, y = key
            qx1, qy1, qx2, qy2 = query_bounds
            if qx1 <= x <= qx2 and qy1 <= y <= qy2:
                results.append((key, value))
        
        # Recurse into overlapping children
        for partition, child in node.children.items():
            child_bounds = self._get_child_bounds(node_bounds, partition)
            if self._bounds_overlap(query_bounds, child_bounds):
                self._range_query_recursive(
                    child,
                    query_bounds,
                    child_bounds,
                    results
                )
    
    def _bounds_overlap(
        self,
        b1: Tuple[float, float, float, float],
        b2: Tuple[float, float, float, float]
    ) -> bool:
        """Check if two bounding boxes overlap."""
        return not (b1[2] < b2[0] or b1[0] > b2[2] or
                    b1[3] < b2[1] or b1[1] > b2[3])


# Usage
spgist = SPGistIndex(partition_type=PartitionType.QUADTREE)

# Insert 2D points
points = [(25, 25), (75, 75), (25, 75), (75, 25), (50, 50)]
for i, point in enumerate(points):
    spgist.insert(point, f"point_{i}")

# Search for specific point
results = spgist.search((25, 25))
print(f"Found at (25, 25): {results}")

# Range query
range_results = spgist.range_query((20, 20, 60, 60))
print(f"Points in range [20,60]x[20,60]: {range_results}")
```
**Why**: SP-GiST provides a flexible framework for implementing space-partitioned indexes like quadtrees and k-d trees, enabling efficient search for spatial data and custom partitioning strategies.

## Await Implementation Pattern
```python
# PROBLEM: Implement custom awaitable objects and async patterns
# TIME: O(1) for await, O(n) for coroutine chain SPACE: O(n)
from typing import Any, Coroutine, Generator, TypeVar, Optional
from dataclasses import dataclass
import asyncio
from collections import deque

T = TypeVar('T')
T_co = TypeVar('T_co', covariant=True)

# 1. Custom Awaitable Object
class DelayedValue:
    """Custom awaitable that delays returning a value."""
    
    def __init__(self, value: T, delay: float):
        self._value = value
        self._delay = delay
    
    def __await__(self) -> Generator[Any, None, T]:
        """Make this object awaitable by implementing __await__."""
        return self._async_get().__await__()
    
    async def _async_get(self) -> T:
        """Internal async method that performs the delay."""
        await asyncio.sleep(self._delay)
        return self._value

# 2. Custom Async Context Manager
class AsyncTimer:
    """Async context manager for timing code blocks."""
    
    async def __aenter__(self) -> 'AsyncTimer':
        import time
        self._start = time.time()
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb) -> None:
        import time
        elapsed = time.time() - self._start
        print(f"Elapsed time: {elapsed:.3f}s")
    
    @property
    def elapsed(self) -> float:
        import time
        return time.time() - self._start

# 3. Custom Async Iterator
class AsyncRange:
    """Async iterator that yields values with delay."""
    
    def __init__(self, start: int, stop: int, delay: float = 0.1):
        self._current = start
        self._stop = stop
        self._delay = delay
    
    def __aiter__(self) -> 'AsyncRange':
        return self
    
    async def __anext__(self) -> int:
        if self._current >= self._stop:
            raise StopAsyncIteration
        
        await asyncio.sleep(self._delay)
        value = self._current
        self._current += 1
        return value

# 4. Async Queue with Priority
@dataclass(order=True)
class PrioritizedItem:
    """Item with priority for async priority queue."""
    priority: int
    data: Any = None

class AsyncPriorityQueue:
    """Async priority queue implementation."""
    
    def __init__(self):
        self._queue: deque[PrioritizedItem] = deque()
        self._event = asyncio.Event()
    
    async def put(self, priority: int, data: Any) -> None:
        """Add item with priority to queue."""
        item = PrioritizedItem(priority=priority, data=data)
        self._queue.append(item)
        self._queue = deque(sorted(self._queue))
        self._event.set()
    
    async def get(self) -> Any:
        """Get highest priority item from queue."""
        while not self._queue:
            self._event.clear()
            await self._event.wait()
        
        return self._queue.popleft().data
    
    def qsize(self) -> int:
        """Return queue size."""
        return len(self._queue)

# 5. Async Rate Limiter
class AsyncRateLimiter:
    """Rate limiter for controlling async operation frequency."""
    
    def __init__(self, rate: float, capacity: int = 1):
        self._rate = rate  # Operations per second
        self._capacity = capacity  # Max burst
        self._tokens = capacity
        self._last_update = asyncio.get_event_loop().time()
        self._lock = asyncio.Lock()
    
    async def acquire(self) -> None:
        """Acquire permission to proceed, waiting if necessary."""
        async with self._lock:
            now = asyncio.get_event_loop().time()
            elapsed = now - self._last_update
            
            # Replenish tokens
            self._tokens = min(
                self._capacity,
                self._tokens + elapsed * self._rate
            )
            self._last_update = now
            
            if self._tokens < 1:
                # Wait for token
                wait_time = (1 - self._tokens) / self._rate
                await asyncio.sleep(wait_time)
                self._tokens = 0
            else:
                self._tokens -= 1

# 6. Async Pipeline Pattern
class AsyncPipeline:
    """Pipeline for chaining async operations."""
    
    def __init__(self):
        self._stages: list = []
    
    def add_stage(self, processor: Callable[[Any], Coroutine[Any, None, Any]]) -> 'AsyncPipeline':
        """Add a processing stage to the pipeline."""
        self._stages.append(processor)
        return self
    
    async def execute(self, initial_value: Any) -> Any:
        """Execute all pipeline stages in sequence."""
        result = initial_value
        for stage in self._stages:
            result = await stage(result)
        return result

# Usage examples
async def demonstrate_awaitables():

    delayed = DelayedValue("Hello after delay", 0.5)
    result = await delayed
    print(f"Delayed result: {result}")
    
    # Async context manager

        await asyncio.sleep(0.3)
        print("Work done")
    
    # Async iterator
    print("Async range:")
    async for num in AsyncRange(0, 5, 0.1):
        print(f"  Got: {num}")
    
    # Async priority queue
    queue = AsyncPriorityQueue()
    await queue.put(3, "low priority")
    await queue.put(1, "high priority")
    await queue.put(2, "medium priority")
    
    while queue.qsize() > 0:
        item = await queue.get()
        print(f"Dequeued: {item}")
    
    # Async rate limiter
    limiter = AsyncRateLimiter(rate=2.0)  # 2 ops/sec

        await limiter.acquire()
        print(f"Rate-limited operation {i}")

        await asyncio.sleep(0.1)

        await asyncio.sleep(0.1)

    async def to_string(x: int) -> str:
        return f"Result: {x}"
    
    pipeline = AsyncPipeline()
    pipeline.add_stage(double).add_stage(add_ten).add_stage(to_string)
```
