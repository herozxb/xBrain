## 1. Publisher Pattern

```python
# PROBLEM: Implement a publisher that broadcasts messages to multiple subscribers
# TIME: O(n) for publish where n = subscriber count SPACE: O(n) for subscribers
class Publisher:
    """Publisher class implementing the publish-subscribe pattern."""
    
    def __init__(self):
        # Dictionary to store topic -> list of subscribers
        self._subscribers: dict[str, list[callable]] = {}
    
    def subscribe(self, topic: str, callback: callable) -> None:
        """Register a callback for a specific topic."""
        if topic not in self._subscribers:
            self._subscribers[topic] = []
        self._subscribers[topic].append(callback)
    
    def unsubscribe(self, topic: str, callback: callable) -> bool:
        """Remove a callback from a topic. Returns True if found."""
        if topic in self._subscribers:
            try:
                self._subscribers[topic].remove(callback)
                return True
            except ValueError:
                return False
        return False
    
    def publish(self, topic: str, message: any) -> int:
        """Broadcast message to all subscribers of topic. Returns count notified."""
        if topic not in self._subscribers:
            return 0
        
        count = 0
        for callback in self._subscribers[topic]:
            try:
                callback(message)
                count += 1
            except Exception as e:
                # Log error but continue notifying other subscribers
                print(f"Subscriber error: {e}")
        
        return count
    
    def get_subscriber_count(self, topic: str) -> int:
        """Return the number of subscribers for a topic."""
        return len(self._subscribers.get(topic, []))
```
**Why**: Decouples message producers from consumers, enabling scalable event-driven architectures with loose coupling.

## 2. Observer Pattern Variants (Push Model)

```python

# TIME: O(n) for notify where n = observer count SPACE: O(n)
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import List, Any

@dataclass
class NewsItem:
    """Data class representing a news update."""
    title: str
    content: str
    timestamp: float

class Observer(ABC):
    """Abstract base class for observers."""
    
    @abstractmethod
    def update(self, data: Any) -> None:
        """Receive update from subject with pushed data."""
        pass

class Subject:
    """Subject that maintains observers and pushes updates."""
    
    def __init__(self):
        self._observers: List[Observer] = []
        self._state: Any = None
    
    def attach(self, observer: Observer) -> None:
        """Add an observer to receive notifications."""
        if observer not in self._observers:
            self._observers.append(observer)
    
    def detach(self, observer: Observer) -> None:
        """Remove an observer from notifications."""
        if observer in self._observers:
            self._observers.remove(observer)
    
    def notify(self, data: Any) -> None:
        """Push data to all attached observers."""
        self._state = data
        for observer in self._observers:
            observer.update(data)

    """Concrete observer that sends email notifications."""
    
    def __init__(self, email: str):

```

## 3. Mediator Pattern Variants (Event-Driven)

```python
# PROBLEM: Coordinate complex interactions between components without direct coupling
# TIME: O(1) for registration, O(n) for event broadcast SPACE: O(n)
from collections import defaultdict
from typing import Callable, Dict, List, Any
from enum import Enum

class EventType(Enum):
    """Enumeration of possible system events."""
    USER_LOGIN = "user_login"
    USER_LOGOUT = "user_logout"
    MESSAGE_SENT = "message_sent"
    FILE_UPLOADED = "file_uploaded"

class EventMediator:
    """Central mediator for event-driven component communication."""
    
    def __init__(self):
        # Map event types to list of handlers
        self._handlers: Dict[EventType, List[Callable]] = defaultdict(list)
        # Track component references
        self._components: Dict[str, Any] = {}
    
    def register_component(self, name: str, component: Any) -> None:
        """Register a component with the mediator."""
        self._components[name] = component
    
    def subscribe(self, event_type: EventType, handler: Callable) -> None:
        """Subscribe a handler to an event type."""
        self._handlers[event_type].append(handler)
    
    def unsubscribe(self, event_type: EventType, handler: Callable) -> bool:
        """Remove a handler from an event type."""
        try:
            self._handlers[event_type].remove(handler)
            return True
        except ValueError:
            return False
    
    def emit(self, event_type: EventType, payload: Any = None) -> None:
        """Emit an event to all subscribed handlers."""
        for handler in self._handlers[event_type]:
            try:
                handler(event_type, payload)
            except Exception as e:
                print(f"Handler error for {event_type}: {e}")
    
    def get_component(self, name: str) -> Any:
        """Retrieve a registered component by name."""
        return self._components.get(name)

class LoggerComponent:
    """Component that logs all events."""
    
    def __init__(self, mediator: EventMediator):
        self.mediator = mediator
        # Subscribe to all events we want to log
        for event in EventType:
            mediator.subscribe(event, self.log_event)
    
    def log_event(self, event_type: EventType, payload: Any) -> None:
        """Log received events."""
        print(f"[LOG] {event_type.value}: {payload}")
```
**Why**: Event-driven mediator reduces component coupling and enables flexible, reactive system architecture.

## 4. Command Pattern Variants (Undo/Redo)

```python
# PROBLEM: Implement command pattern with undo/redo support and macro commands
# TIME: O(1) for execute/undo SPACE: O(n) for command history
from abc import ABC, abstractmethod
from typing import List, Optional
from dataclasses import dataclass, field

class Command(ABC):
    """Abstract command with execute and undo operations."""
    
    @abstractmethod
    def execute(self) -> None:
        """Execute the command."""
        pass
    
    @abstractmethod
    def undo(self) -> None:
        """Undo the command."""
        pass
    
    def redo(self) -> None:
        """Redo by re-executing (can be overridden for optimization)."""
        self.execute()

@dataclass
class TextEditor:
    """Simple text editor as command receiver."""
    content: str = ""
    cursor_pos: int = 0
    
    def insert(self, text: str) -> None:
        """Insert text at cursor position."""
        self.content = (self.content[:self.cursor_pos] + 
                       text + 
                       self.content[self.cursor_pos:])
        self.cursor_pos += len(text)
    
    def delete(self, length: int) -> str:
        """Delete text from cursor position, returns deleted text."""
        deleted = self.content[self.cursor_pos:self.cursor_pos + length]
        self.content = (self.content[:self.cursor_pos] + 
                       self.content[self.cursor_pos + length:])
        return deleted

class InsertCommand(Command):
    """Command to insert text."""
    
    def __init__(self, editor: TextEditor, text: str):
        self.editor = editor
        self.text = text
    
    def execute(self) -> None:
        self.editor.insert(self.text)
    
    def undo(self) -> None:
        # Move cursor back and delete the inserted text
        self.editor.cursor_pos -= len(self.text)
        self.editor.delete(len(self.text))

class MacroCommand(Command):
    """Composite command that executes multiple commands as one."""
    
    def __init__(self, commands: Optional[List[Command]] = None):
        self.commands: List[Command] = commands or []
    
    def add(self, command: Command) -> None:
        """Add a command to the macro."""
        self.commands.append(command)
    
    def execute(self) -> None:
        """Execute all commands in order."""
        for cmd in self.commands:
            cmd.execute()
    
    def undo(self) -> None:
        """Undo all commands in reverse order."""
        for cmd in reversed(self.commands):
            cmd.undo()

class CommandHistory:
    """Manages command history for undo/redo operations."""
    
    def __init__(self, max_size: int = 100):
        self._history: List[Command] = []
        self._position: int = 0
        self._max_size: int = max_size
    
    def push(self, command: Command) -> None:
        """Add executed command and truncate future history."""
        # Remove any commands after current position (lost redo history)
        self._history = self._history[:self._position]
        self._history.append(command)
        
        # Limit history size
        if len(self._history) > self._max_size:
            self._history.pop(0)
        else:
            self._position += 1
    
    def undo(self) -> Optional[Command]:
        """Undo last command, returns the undone command."""
        if self._position > 0:
            self._position -= 1
            command = self._history[self._position]
            command.undo()
            return command
        return None
    
    def redo(self) -> Optional[Command]:
        """Redo next command, returns the redone command."""
        if self._position < len(self._history):
            command = self._history[self._position]
            command.redo()
            self._position += 1
            return command
        return None
```
**Why**: Encapsulates operations as objects enabling undo/redo, macro recording, and operation queuing.

## 6. Read Committed Isolation

```python
# PROBLEM: Implement read committed isolation level preventing dirty reads
# TIME: O(1) for lock acquisition SPACE: O(n) for active transactions
import threading
import time
from typing import Dict, Any, Optional, Set
from dataclasses import dataclass, field
from enum import Enum

class LockType(Enum):
    """Lock types for row-level locking."""
    SHARED = "shared"    # Read lock
    EXCLUSIVE = "exclusive"  # Write lock

@dataclass
class Transaction:
    """Represents an active database transaction."""
    id: int
    start_time: float
    locks_held: Dict[str, LockType] = field(default_factory=dict)
    changes: Dict[str, Any] = field(default_factory=dict)
    
class RowLock:
    """Manages locks on a single row."""
    
    def __init__(self):
        self._holders: Dict[int, LockType] = {}  # tx_id -> lock_type
        self._lock = threading.Lock()
    
    def acquire(self, tx_id: int, lock_type: LockType, timeout: float = 5.0) -> bool:
        """Attempt to acquire a lock with timeout."""
        start = time.time()
        
        while time.time() - start < timeout:
            with self._lock:
                # If no holders, acquire immediately
                if not self._holders:
                    self._holders[tx_id] = lock_type
                    return True
                
                # If we already hold the lock
                if tx_id in self._holders:
                    # Upgrade from shared to exclusive if needed
                    if lock_type == LockType.EXCLUSIVE and self._holders[tx_id] == LockType.SHARED:
                        if len(self._holders) == 1:
                            self._holders[tx_id] = lock_type
                            return True
                    return True
                
                # Shared locks can coexist with other shared locks
                if lock_type == LockType.SHARED:
                    if all(lt == LockType.SHARED for lt in self._holders.values()):
                        self._holders[tx_id] = lock_type
                        return True
                
                # Check for compatible lock: exclusive requires no other holders
                if lock_type == LockType.EXCLUSIVE:
                    if not self._holders:
                        self._holders[tx_id] = lock_type
                        return True
            
            time.sleep(0.01)  # Brief sleep before retry
        
        return False  # Timeout
    
    def release(self, tx_id: int) -> None:
        """Release lock held by transaction."""
        with self._lock:
            self._holders.pop(tx_id, None)

class ReadCommittedStore:
    """Key-value store implementing READ COMMITTED isolation."""
    
    def __init__(self):
        self._data: Dict[str, Any] = {}  # row_id -> value
        self._committed_versions: Dict[str, int] = {}  # row_id -> commit_ts
        self._locks: Dict[str, RowLock] = {}  # row_id -> lock
        self._transactions: Dict[int, Transaction] = {}
        self._next_tx_id = 1
        self._global_lock = threading.Lock()
        self._commit_counter = 0
    
    def begin_transaction(self) -> int:
        """Start a new transaction."""
        with self._global_lock:
            tx_id = self._next_tx_id
            self._next_tx_id += 1
            self._transactions[tx_id] = Transaction(
                id=tx_id,
                start_time=time.time()
            )
            return tx_id
    
    def read(self, tx_id: int, row_id: str) -> Optional[Any]:
        """Read a row - only sees committed data."""
        if row_id not in self._locks:
            self._locks[row_id] = RowLock()
        
        # Acquire shared lock for reading
        if not self._locks[row_id].acquire(tx_id, LockType.SHARED):
            raise RuntimeError(f"Lock timeout on read for row {row_id}")
        
        try:
            # READ COMMITTED: only see committed data
            return self._data.get(row_id)
        finally:
            # Release read lock immediately (READ COMMITTED behavior)
            self._locks[row_id].release(tx_id)
    
    def write(self, tx_id: int, row_id: str, value: Any) -> bool:
        """Write to a row - requires exclusive lock."""
        tx = self._transactions.get(tx_id)
        if not tx:
            raise ValueError(f"Transaction {tx_id} not found")
        
        if row_id not in self._locks:
            self._locks[row_id] = RowLock()
        
        # Acquire exclusive lock for writing
        if not self._locks[row_id].acquire(tx_id, LockType.EXCLUSIVE):
            raise RuntimeError(f"Lock timeout on write for row {row_id}")
        
        # Stage the change (not visible to others until commit)
        tx.changes[row_id] = value
        tx.locks_held[row_id] = LockType.EXCLUSIVE
        return True
    
    def commit(self, tx_id: int) -> bool:
        """Commit transaction - makes changes visible atomically."""
        tx = self._transactions.get(tx_id)
        if not tx:
            return False
        
        try:
            # Apply all changes
            for row_id, value in tx.changes.items():
                self._data[row_id] = value
                self._committed_versions[row_id] = self._commit_counter
                self._commit_counter += 1
            
            return True
        finally:
            # Release all locks
            for row_id in tx.locks_held:
                if row_id in self._locks:
                    self._locks[row_id].release(tx_id)
            
            del self._transactions[tx_id]
```
**Why**: Prevents dirty reads while allowing non-repeatable reads, balancing consistency with concurrency.

## 7. Repeatable Read Isolation

```python
# PROBLEM: Implement repeatable read isolation preventing non-repeatable reads
# TIME: O(n) for snapshot creation SPACE: O(n) for version storage
import threading
import time
from typing import Dict, Any, Optional, List
from dataclasses import dataclass, field
from copy import deepcopy

@dataclass
class Version:
    """Represents a version of a row with timestamp."""
    value: Any
    commit_ts: int
    tx_id: int

@dataclass
class Transaction:
    """Transaction with snapshot isolation support."""
    id: int
    start_ts: int
    snapshot: Dict[str, Any] = field(default_factory=dict)
    changes: Dict[str, Any] = field(default_factory=dict)
    read_set: set = field(default_factory=set)
    write_set: set = field(default_factory=set)

class RepeatableReadStore:
    """MVCC-based store implementing REPEATABLE READ isolation."""
    
    def __init__(self):
        # Each row stores list of versions (newest first)
        self._versions: Dict[str, List[Version]] = {}
        self._transactions: Dict[int, Transaction] = {}
        self._lock = threading.Lock()
        self._next_tx_id = 1
        self._commit_ts = 0
    
    def begin_transaction(self) -> int:
        """Start transaction with a snapshot of committed data."""
        with self._lock:
            tx_id = self._next_tx_id
            self._next_tx_id += 1
            
            # Create snapshot at current commit timestamp
            snapshot = {}
            for row_id, version_list in self._versions.items():
                if version_list:
                    # Find the latest committed version
                    for v in version_list:
                        if v.commit_ts <= self._commit_ts:
                            snapshot[row_id] = deepcopy(v.value)
                            break
            
            tx = Transaction(
                id=tx_id,
                start_ts=self._commit_ts,
                snapshot=snapshot
            )
            self._transactions[tx_id] = tx
            return tx_id
    
    def read(self, tx_id: int, row_id: str) -> Optional[Any]:
        """Read from transaction's snapshot - repeatable reads guaranteed."""
        tx = self._transactions.get(tx_id)
        if not tx:
            raise ValueError(f"Transaction {tx_id} not found")
        
        tx.read_set.add(row_id)
        
        # Check if we have uncommitted changes for this row
        if row_id in tx.changes:
            return tx.changes[row_id]
        
        # Read from snapshot taken at transaction start
        return tx.snapshot.get(row_id)
    
    def write(self, tx_id: int, row_id: str, value: Any) -> bool:
        """Stage a write in transaction's private changes."""
        tx = self._transactions.get(tx_id)
        if not tx:
            raise ValueError(f"Transaction {tx_id} not found")
        
        tx.write_set.add(row_id)
        tx.changes[row_id] = value
        return True
    
    def _check_conflict(self, tx: Transaction) -> bool:
        """Check for write-write conflicts."""
        for row_id in tx.write_set:
            if row_id in self._versions:
                version_list = self._versions[row_id]
                if version_list:
                    latest = version_list[0]
                    # Check if someone else committed after our start
                    if latest.commit_ts > tx.start_ts:
                        return True  # Conflict detected
        return False
    
    def commit(self, tx_id: int) -> bool:
        """Commit with conflict detection."""
        with self._lock:
            tx = self._transactions.get(tx_id)
            if not tx:
                return False
            
            # Check for conflicts
            if self._check_conflict(tx):
                self._transactions.pop(tx_id, None)
                raise RuntimeError("Serialization failure: concurrent modification")
            
            # Get new commit timestamp
            new_commit_ts = self._commit_ts + 1
            
            # Apply all changes as new versions
            for row_id, value in tx.changes.items():
                if row_id not in self._versions:
                    self._versions[row_id] = []
                
                new_version = Version(
                    value=deepcopy(value),
                    commit_ts=new_commit_ts,
                    tx_id=tx_id
                )
                self._versions[row_id].insert(0, new_version)
            
            self._commit_ts = new_commit_ts
            del self._transactions[tx_id]
            return True
    
    def rollback(self, tx_id: int) -> None:
        """Rollback transaction, discarding all changes."""
        with self._lock:
            self._transactions.pop(tx_id, None)
```
**Why**: MVCC snapshot isolation guarantees repeatable reads without blocking readers, improving concurrency.

## 8. Heavy Light Decomposition Variants (Path Queries)

```python
# PROBLEM: Answer path queries on trees efficiently using HLD with segment tree
# TIME: O(log²n) for path queries, O(log n) for updates SPACE: O(n)
from typing import List, Optional, Tuple
from dataclasses import dataclass

@dataclass
class SegmentTree:
    """Segment tree for range queries and point updates."""
    
    def __init__(self, n: int, neutral: int = 0, combine=lambda a, b: a + b):
        self.n = n
        self.neutral = neutral
        self.combine = combine
        self.size = 1
        while self.size < n:
            self.size *= 2
        self.tree = [neutral] * (2 * self.size)
    
    def update(self, idx: int, value: int) -> None:
        """Point update: set position idx to value."""
        idx += self.size
        self.tree[idx] = value
        idx //= 2
        while idx:
            self.tree[idx] = self.combine(self.tree[2 * idx], self.tree[2 * idx + 1])
            idx //= 2
    
    def query(self, l: int, r: int) -> int:
        """Range query on [l, r)."""
        l += self.size
        r += self.size
        res = self.neutral
        while l < r:
            if l % 2:
                res = self.combine(res, self.tree[l])
                l += 1
            if r % 2:
                r -= 1
                res = self.combine(res, self.tree[r])
            l //= 2
            r //= 2
        return res

class HeavyLightDecomposition:
    """HLD supporting path queries with segment tree backend."""
    
    def __init__(self, n: int, adj: List[List[int]], values: List[int]):
        self.n = n
        self.adj = adj
        self.values = values
        
        # HLD arrays
        self.parent = [-1] * n
        self.depth = [0] * n
        self.size = [0] * n      # Subtree sizes
        self.heavy = [-1] * n     # Heavy child of each node
        self.head = [0] * n       # Head of chain containing node
        self.pos = [0] * n        # Position in segment tree
        self.pos_to_node = [0] * n  # Reverse mapping
        
        # Build HLD
        self._dfs_size(0)  # Compute sizes and find heavy edges
        self._dfs_hld(0, 0)  # Decompose and assign positions
        
        # Initialize segment tree with values in HLD order
        self.seg = SegmentTree(n, 0, lambda a, b: a + b)
        for i in range(n):
            self.seg.update(self.pos[i], self.values[i])
    
    def _dfs_size(self, u: int) -> int:
        """First DFS: compute subtree sizes and identify heavy children."""
        self.size[u] = 1
        max_subtree = 0
        
        for v in self.adj[u]:
            if v != self.parent[u]:
                self.parent[v] = u
                self.depth[v] = self.depth[u] + 1
                subtree_size = self._dfs_size(v)
                self.size[u] += subtree_size
                
                # Update heavy child if this subtree is largest
                if subtree_size > max_subtree:
                    max_subtree = subtree_size
                    self.heavy[u] = v
        
        return self.size[u]
    
    def _dfs_hld(self, u: int, h: int, cur_pos: List[int] = None) -> None:
        """Second DFS: decompose tree into chains."""
        if cur_pos is None:
            cur_pos = [0]
        
        self.head[u] = h
        self.pos[u] = cur_pos[0]
        self.pos_to_node[cur_pos[0]] = u
        cur_pos[0] += 1
        
        # Process heavy child first (same chain)
        if self.heavy[u] != -1:
            self._dfs_hld(self.heavy[u], h, cur_pos)
        
        # Process light children (start new chains)
        for v in self.adj[u]:
            if v != self.parent[u] and v != self.heavy[u]:
                self._dfs_hld(v, v, cur_pos)
    
    def path_query(self, u: int, v: int) -> int:
        """Query sum on path from u to v."""
        res = 0
        while self.head[u] != self.head[v]:
            # Ensure u is deeper
            if self.depth[self.head[u]] < self.depth[self.head[v]]:
                u, v = v, u
            
            # Query current chain segment
            cur_head = self.head[u]
            res += self.seg.query(self.pos[cur_head], self.pos[u] + 1)
            u = self.parent[cur_head]
        
        # Now u and v are on same chain
        if self.depth[u] > self.depth[v]:
            u, v = v, u
        res += self.seg.query(self.pos[u], self.pos[v] + 1)
        return res
    
    def point_update(self, u: int, value: int) -> None:
        """Update value at node u."""
        self.values[u] = value
        self.seg.update(self.pos[u], value)
    
    def lca(self, u: int, v: int) -> int:
        """Find LCA by climbing chains."""
        while self.head[u] != self.head[v]:
            if self.depth[self.head[u]] < self.depth[self.head[v]]:
                u, v = v, u
            u = self.parent[self.head[u]]
        
        return u if self.depth[u] < self.depth[v] else v
```
**Why**: Enables O(log²n) path queries on trees by decomposing into heavy paths indexed by segment trees.

## 9. Metaclass Patterns (Singleton Metaclass)

```python
# PROBLEM: Implement singleton and registry patterns using metaclasses
# TIME: O(1) for instance access SPACE: O(n) for registry
from typing import Dict, Any, Type, Optional
from abc import ABCMeta

class SingletonMeta(type):
    """Metaclass implementing the Singleton pattern."""
    
    _instances: Dict[type, Any] = {}
    _lock = None  # Will be initialized when needed
    
    def __call__(cls, *args, **kwargs):
        """Override instance creation to ensure single instance."""
        if cls not in cls._instances:
            # Lazy import to avoid circular dependency
            import threading
            if cls._lock is None:
                cls._lock = threading.Lock()
            
            with cls._lock:
                # Double-check locking pattern
                if cls not in cls._instances:
                    instance = super().__call__(*args, **kwargs)
                    cls._instances[cls] = instance
        
        return cls._instances[cls]
    
    @classmethod
    def get_instance(cls, class_type: type) -> Optional[Any]:
        """Get existing instance without creating new one."""
        return cls._instances.get(class_type)
    
    @classmethod
    def clear_instance(cls, class_type: type) -> bool:
        """Remove instance (useful for testing)."""
        if class_type in cls._instances:
            del cls._instances[class_type]
            return True
        return False

class RegistryMeta(ABCMeta):
    """Metaclass for automatic class registration."""
    
    _registry: Dict[str, type] = {}
    
    def __new__(mcs, name: str, bases: tuple, namespace: dict):
        """Create new class and register it."""
        cls = super().__new__(mcs, name, bases, namespace)
        
        # Register class if it has a registry_key
        if hasattr(cls, 'registry_key'):
            key = cls.registry_key
            mcs._registry[key] = cls
        
        return cls
    
    @classmethod
    def get_class(mcs, key: str) -> Optional[type]:
        """Retrieve a class by its registry key."""
        return mcs._registry.get(key)
    
    @classmethod
    def create_instance(mcs, key: str, *args, **kwargs) -> Optional[Any]:
        """Create instance of registered class."""
        cls = mcs.get_class(key)
        if cls:
            return cls(*args, **kwargs)
        return None
    
    @classmethod
    def list_registered(mcs) -> Dict[str, type]:
        """Return copy of registry."""
        return dict(mcs._registry)

# Example usage with SingletonMeta
class DatabaseConnection(metaclass=SingletonMeta):
    """Database connection that should only have one instance."""
    
    def __init__(self, connection_string: str = "default"):
        self.connection_string = connection_string
        self.connected = False
    
    def connect(self) -> bool:
        """Establish connection."""
        if not self.connected:
            print(f"Connecting to {self.connection_string}")
            self.connected = True
        return self.connected

# Example usage with RegistryMeta
class Handler(metaclass=RegistryMeta):
    """Base handler class with automatic registration."""
    
    @classmethod
    def handle(cls, data: Any) -> Any:
        """Process data - to be implemented by subclasses."""
        raise NotImplementedError

class JSONHandler(Handler):
    """Handler for JSON data."""
    registry_key = "json"
    
    @classmethod
    def handle(cls, data: Any) -> Any:
        import json
        return json.loads(data) if isinstance(data, str) else data

class XMLHandler(Handler):
    """Handler for XML data."""
    registry_key = "xml"
    
    @classmethod
    def handle(cls, data: Any) -> Any:
        import xml.etree.ElementTree as ET
        return ET.fromstring(data) if isinstance(data, str) else data

class ValidatingMeta(type):
    """Metaclass that adds attribute validation."""
    
    def __new__(mcs, name: str, bases: tuple, namespace: dict):
        # Collect validators from base classes
        validators = {}
        for base in bases:
            if hasattr(base, '_validators'):
                validators.update(base._validators)
        
        # Add validators from current class
        if '_validators' in namespace:
            validators.update(namespace['_validators'])
        
        namespace['_validators'] = validators
        
        cls = super().__new__(mcs, name, bases, namespace)
        
        # Wrap __setattr__ to perform validation
        original_setattr = cls.__setattr__ if hasattr(cls, '__setattr__') else object.__setattr__
        
        def validated_setattr(obj, name, value):
            if name in cls._validators:
                validator = cls._validators[name]
                if not validator(value):
                    raise ValueError(f"Invalid value for {name}: {value}")
            original_setattr(obj, name, value)
        
        cls.__setattr__ = validated_setattr
        return cls
```
**Why**: Metaclasses enable class-level behavior modification, useful for patterns like Singleton, Registry, and attribute validation.

## 10. Decomposition Pattern (Strangler Fig)

```python
# PROBLEM: Gradually migrate monolith to microservices without big-bang rewrite
# TIME: O(1) for routing decision SPACE: O(n) for route configuration
from abc import ABC, abstractmethod
from typing import Dict, Any, Optional, Callable
from dataclasses import dataclass
from enum import Enum
import re

class RouteStatus(Enum):
    """Status of a migration route."""
    MONOLITH = "monolith"      # Still using legacy
    MIGRATING = "migrating"    # Dual-write phase
    MIGRATED = "migrated"      # Fully on new service

@dataclass
class Route:
    """Configuration for routing a request."""
    pattern: str              # URL pattern to match
    service_name: str         # Target microservice
    status: RouteStatus       # Migration status
    feature_flag: str = None  # Optional feature flag for gradual rollout
    percentage: int = 100     # Traffic percentage to new service

class RequestHandler(ABC):
    """Abstract handler for requests."""
    
    @abstractmethod
    def handle(self, request: Dict[str, Any]) -> Dict[str, Any]:
        """Process request and return response."""
        pass

class MonolithHandler(RequestHandler):
    """Handler for legacy monolith."""
    
    def handle(self, request: Dict[str, Any]) -> Dict[str, Any]:
        # Simulate monolith processing
        print(f"[Monolith] Processing: {request.get('path')}")
        return {
            "source": "monolith",
            "status": 200,
            "data": request.get("data")
        }

class MicroserviceHandler(RequestHandler):
    """Handler for new microservice."""
    
    def __init__(self, service_name: str, endpoint: str):
        self.service_name = service_name
        self.endpoint = endpoint
    
    def handle(self, request: Dict[str, Any]) -> Dict[str, Any]:
        # Simulate microservice call
        print(f"[{self.service_name}] Processing: {request.get('path')}")
        return {
            "source": self.service_name,
            "status": 200,
            "data": request.get("data"),
            "endpoint": self.endpoint
        }

class StranglerRouter:
    """Router implementing Strangler Fig pattern for gradual migration."""
    
    def __init__(self, monolith: MonolithHandler):
        self.monolith = monolith
        self._services: Dict[str, MicroserviceHandler] = {}
        self._routes: list[Route] = []
        self._feature_flags: Dict[str, bool] = {}
        self._traffic_counters: Dict[str, int] = {}
    
    def register_service(self, name: str, handler: MicroserviceHandler) -> None:
        """Register a new microservice handler."""
        self._services[name] = handler
        self._traffic_counters[name] = 0
    
    def add_route(self, route: Route) -> None:
        """Add a routing rule."""
        self._routes.append(route)
        # Sort by specificity (longer patterns first)
        self._routes.sort(key=lambda r: len(r.pattern), reverse=True)
    
    def set_feature_flag(self, flag: str, enabled: bool) -> None:
        """Set feature flag state."""
        self._feature_flags[flag] = enabled
    
    def _match_route(self, path: str) -> Optional[Route]:
        """Find matching route for path."""
        for route in self._routes:
            if re.match(route.pattern, path):
                return route
        return None
    
    def _should_use_new_service(self, route: Route) -> bool:
        """Determine if request should go to new service."""
        # Check feature flag if set
        if route.feature_flag:
            if not self._feature_flags.get(route.feature_flag, False):
                return False
        
        # Check percentage-based rollout
        if route.percentage < 100:
            self._traffic_counters[route.service_name] += 1
            counter = self._traffic_counters[route.service_name]
            return (counter % 100) < route.percentage
        
        return True
    
    def route_request(self, request: Dict[str, Any]) -> Dict[str, Any]:
        """Route request based on migration status."""
        path = request.get("path", "")
        route = self._match_route(path)
        
        if route is None:
            # No route found, use monolith
            return self.monolith.handle(request)
        
        if route.status == RouteStatus.MONOLITH:
            return self.monolith.handle(request)
        
        if route.status == RouteStatus.MIGRATED:
            # Fully migrated to new service
            handler = self._services.get(route.service_name)
            if handler:
                return handler.handle(request)
            return self.monolith.handle(request)
        
        if route.status == RouteStatus.MIGRATING:
            # Dual-write / shadow mode phase
            if self._should_use_new_service(route):
                handler = self._services.get(route.service_name)
                if handler:
                    new_response = handler.handle(request)
                    # Optionally also call monolith for comparison (shadow traffic)
                    # self.monolith.handle(request)
                    return new_response
            return self.monolith.handle(request)
        
        return self.monolith.handle(request)
    
    def get_migration_status(self) -> Dict[str, Any]:
        """Return current migration status for all routes."""
        return {
            route.service_name: {
                "pattern": route.pattern,
                "status": route.status.value,
                "percentage": route.percentage
            }
            for route in self._routes
        }
```
**Why**: Enables incremental, low-risk migration from monolith to microservices with traffic shifting and rollback capabilities.
