## [Maximum Sum Subarray of Size K]
```python
# PROBLEM: Find the maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)

def max_sum_subarray_k(arr, k):
    """
    Sliding window technique to find maximum sum of subarray of size k.
    Instead of recalculating sum for each window, we slide and update.
    """
    if len(arr) < k:
        return None
    
    # Calculate sum of first window
    window_sum = sum(arr[:k])
    max_sum = window_sum
    
    # Slide the window: subtract leftmost, add new rightmost
    for i in range(k, len(arr)):
        window_sum = window_sum - arr[i - k] + arr[i]
        max_sum = max(max_sum, window_sum)
    
    return max_sum

# Example
print(max_sum_subarray_k([2, 1, 5, 1, 3, 2], 3))  # Output: 9 (5+1+3)
```
**Why**: Classic sliding window that teaches how to avoid redundant calculations. O(n²) → O(n) optimization.

## [Container With Most Water]
```python
# PROBLEM: Find two lines that form container holding maximum water
# TIME: O(n) SPACE: O(1)

def max_area(height):
    """
    Two pointers approach. Start from widest container.
    Move the shorter line inward because only that can increase area.
    """
    left, right = 0, len(height) - 1
    max_water = 0
    
    while left < right:
        # Area = width * min height
        width = right - left
        current_area = width * min(height[left], height[right])
        max_water = max(max_water, current_area)
        
        # Move the shorter line (only way to potentially increase area)
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1
    
    return max_water

# Example
print(max_area([1, 8, 6, 2, 5, 4, 8, 3, 7]))  # Output: 49
```
**Why**: Elegant two-pointer technique. Shows greedy choice: always move the limiting factor.

## [N-Queens Problem]
```python
# PROBLEM: Place N queens on NxN board so none attack each other
# TIME: O(N!) SPACE: O(N²)

def solve_n_queens(n):
    """
    Backtracking with pruning. Place queens row by row,
    checking if position is safe before recursing.
    """
    def is_safe(board, row, col):
        # Check column
        for i in range(row):
            if board[i][col] == 'Q':
                return False
        
        # Check upper-left diagonal
        i, j = row - 1, col - 1
        while i >= 0 and j >= 0:
            if board[i][j] == 'Q':
                return False
            i -= 1
            j -= 1
        
        # Check upper-right diagonal
        i, j = row - 1, col + 1
        while i >= 0 and j < n:
            if board[i][j] == 'Q':
                return False
            i -= 1
            j += 1
        
        return True
    
    def solve(board, row):
        if row == n:
            solutions.append([''.join(row) for row in board])
            return
        
        for col in range(n):
            if is_safe(board, row, col):
                board[row][col] = 'Q'
                solve(board, row + 1)
                board[row][col] = '.'  # Backtrack
    
    solutions = []
    board = [['.' for _ in range(n)] for _ in range(n)]
    solve(board, 0)
    return solutions

# Example
print(len(solve_n_queens(4)))  # Output: 2 solutions
```
**Why**: Classic backtracking problem. Teaches constraint satisfaction and pruning.

## [Number of Connected Components]
```python
# PROBLEM: Count connected components in undirected graph
# TIME: O(n * α(n)) ≈ O(n) SPACE: O(n)

class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.components = n
    
    def find(self, x):
        # Path compression: point to root directly
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        # Union by rank: attach smaller tree to larger
        px, py = self.find(x), self.find(y)
        if px == py:
            return
        
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        self.components -= 1

def count_components(n, edges):
    """
    Union-Find approach. Each union reduces component count.
    """
    uf = UnionFind(n)
    for u, v in edges:
        uf.union(u, v)
    return uf.components

# Example
print(count_components(5, [[0,1], [1,2], [3,4]]))  # Output: 2
```
**Why**: Union-Find with path compression and union by rank. Essential for dynamic connectivity.

## [Longest Palindromic Subsequence]
```python
# PROBLEM: Find length of longest subsequence that is a palindrome
# TIME: O(n²) SPACE: O(n²)

def longest_palindrome_subseq(s):
    """
    DP approach. dp[i][j] = longest palindromic subsequence in s[i:j+1]
    If ends match, add 2 to inner result. Otherwise, take max of skip strategies.
    """
    n = len(s)
    dp = [[0] * n for _ in range(n)]
    
    # Every single character is palindrome of length 1
    for i in range(n):
        dp[i][i] = 1
    
    # Build up from length 2 to n
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            
            if s[i] == s[j]:
                # Ends match: add 2 to inner palindrome
                if length == 2:
                    dp[i][j] = 2
                else:
                    dp[i][j] = dp[i + 1][j - 1] + 2
            else:
                # Ends don't match: take best of skip strategies
                dp[i][j] = max(dp[i + 1][j], dp[i][j - 1])
    
    return dp[0][n - 1]

# Example
print(longest_palindrome_subseq("bbbab"))  # Output: 4 ("bbbb")
```
**Why**: 2D DP with clever state definition. Shows how palindrome problems often have symmetrical structure.

## [Dijkstra's Shortest Path]
```python
# PROBLEM: Find shortest path from source to all nodes in weighted graph
# TIME: O((V + E) log V) SPACE: O(V)

import heapq

def dijkstra(n, edges, source):
    """
    Greedy + heap. Always process node with smallest known distance.
    Works only with non-negative weights.
    """
    # Build adjacency list
    graph = {i: [] for i in range(n)}
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))  # Undirected
    
    # Distance array, initialized to infinity
    dist = [float('inf')] * n
    dist[source] = 0
    
    # Priority queue: (distance, node)
    pq = [(0, source)]
    
    while pq:
        d, u = heapq.heappop(pq)
        
        # Skip if we've found better path already
        if d > dist[u]:
            continue
        
        # Relax all neighbors
        for v, weight in graph[u]:
            if dist[u] + weight < dist[v]:
                dist[v] = dist[u] + weight
                heapq.heappush(pq, (dist[v], v))
    
    return dist

# Example
edges = [(0, 1, 4), (0, 2, 1), (2, 1, 2), (1, 3, 1)]
print(dijkstra(4, edges, 0))  # Output: [0, 3, 1, 4]
```
**Why**: Foundation of shortest path algorithms. Heap optimization makes it efficient for sparse graphs.

## [Lowest Common Ancestor in BST]
```python
# PROBLEM: Find lowest common ancestor of two nodes in BST
# TIME: O(h) where h is height SPACE: O(1)

def lowest_common_ancestor(root, p, q):
    """
    Use BST property: LCA is first node where p and q diverge.
    If both less, go left. If both greater, go right. Otherwise, found.
    """
    while root:
        if p.val < root.val and q.val < root.val:
            root = root.left
        elif p.val > root.val and q.val > root.val:
            root = root.right
        else:
            # Split point or match - this is LCA
            return root
    return None

# Tree node definition
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

# Example usage
#        6
#       / \
#      2   8
#     / \
#    0   4
root = TreeNode(6)
root.left = TreeNode(2, TreeNode(0), TreeNode(4))
root.right = TreeNode(8)
lca = lowest_common_ancestor(root, root.left.left, root.left.right)
print(lca.val)  # Output: 2
```
**Why**: Utilizes BST ordering property for elegant solution. Extends to general trees with preprocessing.

## [Merge K Sorted Lists]

```python
# PROBLEM: Merge k sorted linked lists into one sorted list
# TIME: O(N log k) where N = total nodes, k = number of lists
# SPACE: O(k) for heap

import heapq

def merge_k_lists(lists):
    """
    Min-heap approach. Keep smallest element from each list in heap.
    Pop minimum, add its next node to heap.
    """
    # Custom node wrapper for comparison
    heap = []
    for i, node in enumerate(lists):
        if node:
            heapq.heappush(heap, (node.val, i, node))
    
    dummy = ListNode(0)
    current = dummy
    
    while heap:
        val, i, node = heapq.heappop(heap)
        current.next = node
        current = current.next
        
        if node.next:
            heapq.heappush(heap, (node.next.val, i, node.next))
    
    return dummy.next

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

# Example
def create_list(arr):
    dummy = ListNode()
    curr = dummy
    for x in arr:
        curr.next = ListNode(x)
        curr = curr.next
    return dummy.next

lists = [create_list([1, 4, 5]), create_list([1, 3, 4]), create_list([2, 6])]
merged = merge_k_lists(lists)
# Result: 1 -> 1 -> 2 -> 3 -> 4 -> 4 -> 5 -> 6
```
**Why**: Classic heap application. Efficient merging without loading all elements into memory.

## [Implement Trie (Prefix Tree)]
```python
# PROBLEM: Implement trie with insert, search, and startsWith
# TIME: O(m) per operation where m = key length
# SPACE: O(alphabet_size * total_nodes)

class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_end = False

class Trie:
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word):
        """Insert word into trie"""
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
    
    def search(self, word):
        """Check if exact word exists"""
        node = self._find_node(word)
        return node is not None and node.is_end
    
    def starts_with(self, prefix):
        """Check if any word starts with prefix"""
        return self._find_node(prefix) is not None
    
    def _find_node(self, prefix):
        """Helper: find node at end of prefix"""
        node = self.root
        for char in prefix:
            if char not in node.children:
                return None
            node = node.children[char]
        return node

# Example
trie = Trie()
trie.insert("apple")
print(trie.search("apple"))      # True
print(trie.search("app"))        # False
print(trie.starts_with("app"))   # True
```
**Why**: Essential for autocomplete, spell checkers. Space-efficient for common prefixes.

## [Design LRU Cache]
```python
# PROBLEM: Design cache with O(1) get and put, evicts least recently used
# TIME: O(1) for both operations SPACE: O(capacity)

from collections import OrderedDict

class LRUCache:
    """
    OrderedDict maintains insertion order. Move to end on access.
    First item is always LRU when capacity exceeded.
    """
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = OrderedDict()
    
    def get(self, key):
        """Get value and mark as recently used"""
        if key not in self.cache:
            return -1
        
        # Move to end (most recently used)
        self.cache.move_to_end(key)
        return self.cache[key]
    
    def put(self, key, value):
        """Insert/update and evict LRU if needed"""
        if key in self.cache:
            # Update existing and mark as recent
            self.cache.move_to_end(key)
        else:
            # Check capacity before insert
            if len(self.cache) >= self.capacity:
                # Remove first item (LRU)
                self.cache.popitem(last=False)
        
        self.cache[key] = value

# Alternative: Manual doubly linked list + hashmap
class Node:
    def __init__(self, key=0, val=0):
        self.key = key
        self.val = val
        self.prev = None
        self.next = None

class LRUCacheManual:
    def __init__(self, capacity):
        self.cap = capacity
        self.cache = {}
        # Dummy head and tail for easier manipulation
        self.head = Node()
        self.tail = Node()
        self.head.next = self.tail
        self.tail.prev = self.head
    
    def _remove(self, node):
        """Remove node from list"""
        node.prev.next = node.next
        node.next.prev = node.prev
    
    def _add_to_front(self, node):
        """Add node right after head"""
        node.prev = self.head
        node.next = self.head.next
        self.head.next.prev = node
        self.head.next = node
    
    def get(self, key):
        if key not in self.cache:
            return -1
        node = self.cache[key]
        self._remove(node)
        self._add_to_front(node)
        return node.val
    
    def put(self, key, value):
        if key in self.cache:
            self._remove(self.cache[key])
        
        node = Node(key, value)
        self.cache[key] = node
        self._add_to_front(node)
        
        if len(self.cache) > self.cap:
            # Remove LRU (node before tail)
            lru = self.tail.prev
            self._remove(lru)
            del self.cache[lru.key]

# Example
cache = LRUCache(2)
cache.put(1, 1)
cache.put(2, 2)
print(cache.get(1))    # Returns 1, makes 1 most recent
cache.put(3, 3)        # Evicts key 2
print(cache.get(2))    # Returns -1 (not found)
```
**Why**: Combines hashmap + doubly linked list pattern. Ubiquitous in real systems (databases, browsers, OS).

**Key Patterns:**
- **Sliding Window**: Maintain window, update incrementally
- **Two Pointers**: Start from ends, move strategically
- **Backtracking**: Try, recurse, backtrack
- **Union Find**: Path compression + union by rank
- **DP**: Define state, find recurrence, build bottom-up
- **Graphs**: BFS/DFS/Dijkstra based on requirements
- **Trees**: Leverage structure (BST property, recursion)
- **Heap**: When you need "always min/max"
- **Trie**: Prefix-based operations
- **LRU**: Hashmap + doubly linked list
