## 1. Math.asin() - Arcsine Calculation

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Math.asin() - Arcsine Calculator</title>
</head>
<body>
    <h2>Arcsine Calculator</h2>
    <input type="number" id="sinInput" placeholder="Enter value (-1 to 1)" step="0.01" />
    <button onclick="calculateArcsin()">Calculate asin</button>
    <p id="result"></p>

    <script>
        /**
         * Calculate the arcsine (inverse sine) of a value
         * # PROBLEM: Find angle whose sine equals the input value
         * # TIME: O(1) SPACE: O(1)
         * 
         * Why: Arcsine is essential for trigonometric calculations,
         * game physics, and angle computations from sine ratios.
         * 
         * @param {number} value - Input value between -1 and 1
         * @returns {number} Angle in radians between -π/2 and π/2
         */
        function arcsin(value) {
            // Validate input range: asin only accepts [-1, 1]
            if (value < -1 || value > 1) {
                return NaN; // Return NaN for invalid range
            }
            
            // Math.asin returns angle in radians
            const angleRadians = Math.asin(value);
            
            return angleRadians;
        }

        /**
         * Convert radians to degrees for human-readable output
         * @param {number} radians - Angle in radians
         * @returns {number} Angle in degrees
         */
        function radiansToDegrees(radians) {
            return radians * (180 / Math.PI);
        }

        // Event handler for button click
        function calculateArcsin() {
            const input = parseFloat(document.getElementById('sinInput').value);
            const resultElement = document.getElementById('result');
            
            if (isNaN(input)) {
                resultElement.textContent = 'Please enter a valid number';
                return;
            }
            
            const radians = arcsin(input);
            
            if (isNaN(radians)) {
                resultElement.textContent = 'Input must be between -1 and 1';
            } else {
                const degrees = radiansToDegrees(radians);
                resultElement.innerHTML = `
                    asin(${input}) = ${radians.toFixed(4)} radians<br>
                    = ${degrees.toFixed(2)} degrees
                `;
            }
        }
    </script>
</body>
</html>
```

```html
        /**
         * Format JSON with customizable spacing for pretty printing
         * # PROBLEM: Convert object to formatted JSON string with controlled indentation
         * # TIME: O(n) where n is object size SPACE: O(n) for output string
         * 
         * Why: Readable JSON output is crucial for debugging, logging,
         * API responses, and configuration files.
         * 
         * @param {Object|Array} data - Data to serialize
         * @param {number|string} spacing - Number of spaces or string for indentation
         * @returns {string} Formatted JSON string
         */
        function formatJSONWithSpacing(data, spacing = 2) {
            // spacing can be: number (spaces) or string (e.g., '\t')
            // null or undefined = compact (no whitespace)
            
            if (typeof spacing === 'number') {
                // Limit spacing to reasonable range
                spacing = Math.max(0, Math.min(10, spacing));
            }
            
            // JSON.stringify(data, replacer, spacing)
            // - data: the value to convert
            // - replacer: null (no filtering)
            // - spacing: indentation control
            return JSON.stringify(data, null, spacing);
        }

        /**
         * Compact JSON output (no extra whitespace)
         * @param {Object|Array} data - Data to serialize
         * @returns {string} Compact JSON string
         */
        function compactJSON(data) {
            // Passing null/undefined as third parameter removes whitespace
            return JSON.stringify(data);
        }

        // UI handler

            const input = document.getElementById('jsonInput').value;
            const spacing = parseInt(document.getElementById('spacing').value, 10);
            const output = document.getElementById('output');

                const data = JSON.parse(input);
                const formatted = formatJSONWithSpacing(data, spacing);
                output.textContent = formatted;
            } catch (e) {
                output.textContent = 'Error: Invalid JSON - ' + e.message;
```

```html
         * @param {Map} map - The Map to extract values from
         * @returns {Iterator} Iterator yielding map values
         */
        function getMapValues(map) {
            // map.values() returns a new Iterator object
            // It yields each value in insertion order
            return map.values();
        }

        /**
         * Collect all values from a Map into an array
         * @param {Map} map - The Map to extract values from
         * @returns {Array} Array of all values
         */
        function mapValuesToArray(map) {
            // Spread operator works with iterators
            return [...map.values()];
        }

         * Calculate sum of numeric values in a Map
         * @param {Map} map - Map with numeric values

        function sumMapValues(map) {
            let total = 0;
            
            // Direct iteration without creating array
            for (const value of map.values()) {

                    total += value;
                }
            }
```

## 4. Proxy.has() - Property Check Trap

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Proxy.has() - 'in' Operator Trap</title>
</head>
<body>
    <h2>Proxy.has() - Intercept Property Checks</h2>
    <div id="output"></div>

    <script>
        /**
         * Create a proxy with custom 'in' operator behavior
         * # PROBLEM: Intercept and customize property existence checks
         * # TIME: O(1) for check SPACE: O(1) per property rule
         * 
         * Why: Proxy.has enables virtual property checking, access control,
         * hiding private properties, and creating dynamic property namespaces.
         * 
         * @param {Object} target - The object to proxy
         * @param {Array} hiddenProps - Properties to hide from 'in' checks
         * @returns {Proxy} Proxied object with custom has behavior
         */
        function createSecureProxy(target, hiddenProps = []) {
            const hiddenSet = new Set(hiddenProps);
            
            const handler = {
                /**
                 * Trap for the 'in' operator
                 * @param {Object} target - The target object
                 * @param {string} prop - Property name to check
                 * @returns {boolean} Whether property should appear to exist
                 */
                has(target, prop) {
                    // Hide specified properties from 'in' checks
                    if (hiddenSet.has(prop)) {
                        console.log(`[Proxy] Hiding property: ${prop}`);
                        return false;
                    }
                    
                    // Check actual property existence
                    return prop in target;
                }
            };
            
            return new Proxy(target, handler);
        }

        /**
         * Create a proxy that allows prefix-based property matching
         * @param {Object} target - Base object
         * @param {string} prefix - Required prefix for properties
         * @returns {Proxy} Proxy with prefix-based has check
         */
        function createPrefixProxy(target, prefix) {
            return new Proxy(target, {
                has(target, prop) {
                    // Property exists if it starts with prefix
                    // OR actually exists on target
                    return prop.startsWith(prefix) || (prop in target);
                }
            });
        }

        // Demo: Secure proxy hiding sensitive data
        const user = {
            name: 'Alice',
            email: 'alice@example.com',
            _password: 'secret123',  // Private convention
            _apiKey: 'key-abc-xyz'   // Sensitive data
        };

        const secureUser = createSecureProxy(user, ['_password', '_apiKey']);

        // Testing
        const output = document.getElementById('output');
        const results = [];
        
        results.push(`'name' in user: ${'name' in secureUser}`);      // true
        results.push(`'email' in user: ${'email' in secureUser}`);    // true
        results.push(`'_password' in user: ${'_password' in secureUser}`); // false (hidden!)
        results.push(`'_apiKey' in user: ${'_apiKey' in secureUser}`);     // false (hidden!)
        results.push(`'unknown' in user: ${'unknown' in secureUser}`);     // false
        
        output.innerHTML = results.map(r => `<p><code>${r}</code></p>`).join('');

        // Note: Direct access still works - has only affects 'in' operator
        console.log('Direct access to _password:', secureUser._password);
        // But 'in' check returns false:
        console.log('"_password" in secureUser:', '_password' in secureUser);
```

## 5. Yield Keyword - Generator Pausing

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">

<body>
    <h2>Yield Keyword - Pause and Resume Execution</h2>
    <div id="output"></div>

    <script>
        /**
         * Generator function demonstrating yield keyword
         * # PROBLEM: Create iterable sequence with lazy evaluation
         * # TIME: O(1) per yield SPACE: O(1) per suspended state
         * 
         * Why: yield enables memory-efficient sequences, infinite streams,
         * cooperative multitasking, and step-by-step algorithm visualization.
         */

        /**
         * Generate sequence of squares up to max
         * @param {number} max - Maximum number to square
         * @yields {number} Next square in sequence
         */
        function* squareGenerator(max) {
            for (let i = 1; i <= max; i++) {
                // yield pauses execution and returns value
                // Execution resumes here on next .next() call
                yield i * i;
            }
            // Function completes after all yields
        }

        /**
         * Generate Fibonacci sequence infinitely
         * @yields {number} Next Fibonacci number
         */
        function* fibonacciGenerator() {
            let prev = 0, curr = 1;
            
            while (true) {
                yield curr;
                [prev, curr] = [curr, prev + curr];
            }
        }

        /**
         * Yield as two-way communication channel
         * @yields {number} Current value
         * @returns {number} Final sum of all received values
         */
        function* twoWayGenerator() {
            let total = 0;
            
            while (true) {
                // yield returns value AND can receive value via .next(val)
                const received = yield total;
                
                if (received === 'done') {
                    return total; // Ends generator
                }
                
                if (typeof received === 'number') {
                    total += received;
                }
            }
        }

        // Demo output
        const output = document.getElementById('output');
        let html = '';

        // 1. Square generator
        html += '<h3>Squares Generator:</h3><ul>';
        for (const square of squareGenerator(5)) {
            html += `<li>${square}</li>`;
        }
        html += '</ul>';

        // 2. Fibonacci - first 10 numbers
        html += '<h3>First 10 Fibonacci Numbers:</h3><p>';
        const fib = fibonacciGenerator();
        const fibNumbers = [];
        for (let i = 0; i < 10; i++) {
            fibNumbers.push(fib.next().value);
        }
        html += fibNumbers.join(', ') + '</p>';

        // 3. Two-way communication
        html += '<h3>Two-way Yield Communication:</h3><p>';
        const twoWay = twoWayGenerator();
        twoWay.next(); // Start generator
        twoWay.next(10); // Send 10
        twoWay.next(20); // Send 20
        twoWay.next(30); // Send 30
        const final = twoWay.next('done'); // End and get sum
        html += `Sum of 10 + 20 + 30 = ${final.value}</p>`;

        output.innerHTML = html;
    </script>
```

## 6. Prototype Inheritance - Chain Demonstration

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Prototype Inheritance - JavaScript OOP</title>
</head>
<body>
    <h2>Prototype Inheritance Chain</h2>
    <div id="output"></div>

    <script>
        /**
         * Demonstrate JavaScript prototype inheritance
         * # PROBLEM: Create object hierarchy with shared behavior
         * # TIME: O(1) property access SPACE: O(n) per constructor
         * 
         * Why: Prototype inheritance is JavaScript's native OOP mechanism,
         * enabling memory-efficient method sharing and object hierarchies.
         */

        // Base constructor
        function Animal(name) {
            this.name = name;
        }

        // Method on Animal's prototype (shared by all instances)
        Animal.prototype.speak = function() {
            return `${this.name} makes a sound`;
        };

        Animal.prototype.sleep = function() {
            return `${this.name} is sleeping`;
        };

        // Derived constructor
        function Dog(name, breed) {
            // Call parent constructor
            Animal.call(this, name);
            this.breed = breed;
        }

        // Set up prototype chain
        // Dog.prototype inherits from Animal.prototype
        Dog.prototype = Object.create(Animal.prototype);

        // Fix constructor reference
        Dog.prototype.constructor = Dog;

        // Override parent method
        Dog.prototype.speak = function() {
            return `${this.name} barks: Woof!`;
        };

        // Add new method
        Dog.prototype.fetch = function() {
            return `${this.name} fetches the ball`;
        };

        // Further derived constructor
        function Puppy(name, breed, age) {
            Dog.call(this, name, breed);
            this.age = age;
        }

        Puppy.prototype = Object.create(Dog.prototype);
        Puppy.prototype.constructor = Puppy;

        Puppy.prototype.speak = function() {
            return `${this.name} (age ${age}) yips: Yap yap!`;
        };

        /**
         * Visualize prototype chain
         * @param {Object} obj - Object to analyze
         * @returns {Array} List of prototypes
         */
        function getPrototypeChain(obj) {
            const chain = [];
            let current = obj;
            
            while (current) {
                chain.push(current.constructor.name || 'Object');
                current = Object.getPrototypeOf(current);
            }
            
            return chain;
        }

        // Create instances
        const genericAnimal = new Animal('Creature');
        const buddy = new Dog('Buddy', 'Golden Retriever');

        // Display results

        html += `<p>${getPrototypeChain(buddy).join(' → ')}</p>`;
```

## 7. Lexical Scope - Closure Foundation

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Lexical Scope - Closures Foundation</title>
</head>
<body>
    <h2>Lexical Scope in Closures</h2>
    <div id="output"></div>

    <script>
        /**
         * Demonstrate lexical scoping in closures
         * # PROBLEM: Understand variable resolution in nested functions
         * # TIME: O(1) variable lookup SPACE: O(depth) scope chain
         * 
         * Why: Lexical scope determines how closures capture variables,
         * critical for understanding callback behavior and module patterns.
         */

        /**
         * Outer function creates a scope
         * Inner functions "close over" variables from outer scope
         */
        function createCounter() {
            // 'count' is in lexical scope of inner functions
            // Each call to createCounter creates NEW lexical environment
            let count = 0;
            
            return {
                increment: function() {
                    // Accesses 'count' from enclosing lexical scope
                    count++;
                    return count;
                },
                decrement: function() {
                    count--;
                    return count;
                },
                getCount: function() {
                    // Read-only access
                    return count;
                }
            };
        }

        /**
         * Demonstrate scope chain resolution
         */
        function outerScope() {
            const outer = 'I am outer';
            
            function middleScope() {
                const middle = 'I am middle';
                
                function innerScope() {
                    const inner = 'I am inner';
                    
                    // Lexical scope chain: inner → middle → outer → global
                    return `${inner}, ${middle}, ${outer}`;
                }
                
                return innerScope();
            }
            
            return middleScope();
        }

        /**
         * Scope shadowing demonstration
         */
        function shadowingDemo() {
            let x = 'outer x';
            
            function inner() {
                let x = 'inner x'; // Shadows outer x
                let y = 'inner y';
                return `inner sees: x=${x}, y=${y}`;
            }
            
            return {
                innerResult: inner(),
                outerX: x // Still 'outer x' - inner's x doesn't affect it
            };
        }

        /**
         * Loop closure gotcha (and fix)
         */
        function loopClosureDemo() {
            const broken = [];
            const fixed = [];
            
            // Problem: All closures share same 'i'
            for (var i = 0; i < 3; i++) {
                broken.push(() => i);
            }
            
            // Fix 1: Use let (block scope)
            for (let j = 0; j < 3; j++) {
                fixed.push(() => j);
            }
            
            return { broken: broken.map(f => f()), fixed: fixed.map(f => f()) };
        }

        // Display results
        const output = document.getElementById('output');
        let html = '';

        // Counter demo
        const counter1 = createCounter();
        const counter2 = createCounter(); // Separate lexical scope!
        
        html += '<h3>Independent Lexical Scopes:</h3>';
        html += `<p>Counter1: ${counter1.increment()}, ${counter1.increment()}, ${counter1.getCount()}</p>`;
        html += `<p>Counter2: ${counter2.increment()}, ${counter2.getCount()}</p>`;

        // Scope chain
        html += '<h3>Scope Chain:</h3>';
        html += `<p>${outerScope()}</p>`;

        // Shadowing
        html += '<h3>Variable Shadowing:</h3>';
        const shadow = shadowingDemo();
        html += `<p>${shadow.innerResult}</p>`;
        html += `<p>Outer still sees: x=${shadow.outerX}</p>`;

        // Loop closure
        html += '<h3>Loop Closure (var vs let):</h3>';
        const loops = loopClosureDemo();
        html += `<p>var (broken): [${loops.broken.join(', ')}]</p>`;
        html += `<p>let (fixed): [${loops.fixed.join(', ')}]</p>`;
```

```html
        /**
         * Pattern attribute validation handler
         * # PROBLEM: Validate input against regex pattern without JavaScript
         * # TIME: O(n) regex match SPACE: O(1)
         * 
         * Why: HTML5 pattern attribute provides declarative validation,
         * reducing JavaScript code and improving accessibility.
         */

        /**
         * Programmatically check pattern validity
         * @param {HTMLInputElement} input - Input element to validate
         * @returns {boolean} Whether input matches pattern
         */
        function validatePattern(input) {
            // Check validity state
            const isValid = input.checkValidity();
            
            // Access the pattern attribute
            const pattern = input.pattern;
            const value = input.value;
            
            // Pattern is automatically anchored (wrapped in ^...$)
            // unless it contains | (OR)
            const regex = new RegExp(`^(?:${pattern})$`);
            
            return regex.test(value);
        }

        /**
         * Extract pattern info for display
         * @param {HTMLInputElement} input - Input element
         * @returns {Object} Pattern information
         */
        function getPatternInfo(input) {
            return {
                pattern: input.pattern || 'none',
                value: input.value,
                valid: input.validity.valid,
                patternMismatch: input.validity.patternMismatch
            };

            inputs.forEach((input, index) => {
                const info = getPatternInfo(input);
                results.push(`
                    <li>
                        <strong>${input.previousElementSibling?.textContent || `Input ${index + 1}`}</strong>:
                        "${info.value}" - ${info.valid ? '✓ Valid' : '✗ Invalid'}
```

```html
    <script>
        /**
         * Password input handling and validation
         * # PROBLEM: Securely collect and validate password input
         * # TIME: O(1) for input, O(n) for strength check SPACE: O(1)
         * 
         * Why: Password inputs mask characters, but need visibility toggle,
         * strength feedback, and confirmation matching for good UX.
         */

        /**
         * Toggle password visibility
         * @param {string} inputId - ID of password input
         */
        function togglePassword(inputId) {
            const input = document.getElementById(inputId);
            const toggleBtn = input.nextElementSibling;
            
            if (input.type === 'password') {
                input.type = 'text';
                toggleBtn.textContent = '🔒';
            } else {
                input.type = 'password';
                toggleBtn.textContent = '👁️';
            }
        }

        /**
         * Calculate password strength score
         * @param {string} password - Password to evaluate
         * @returns {Object} Strength score and details
         */
        function checkPasswordStrength(password) {
            let score = 0;
            const checks = {
                length: password.length >= 8,
                uppercase: /[A-Z]/.test(password),
                lowercase: /[a-z]/.test(password),
                numbers: /[0-9]/.test(password),
                special: /[^A-Za-z0-9]/.test(password),
                longEnough: password.length >= 12
            };
            
            // Calculate score
            Object.values(checks).forEach(passed => {
                if (passed) score++;
            });
            
            // Determine strength level
            let strength = 'weak';
            if (score >= 4) strength = 'medium';
            if (score >= 5) strength = 'strong';
            
            return { score, strength, checks };
        }

        /**
         * Update strength meter UI
         * @param {string} password - Current password value
         */
        function updateStrengthMeter(password) {
            const meter = document.getElementById('strengthMeter');
            const text = document.getElementById('strengthText');
            
            if (!password) {
                meter.className = 'strength-meter';
                text.textContent = '';
                return;
            }
            
            const { strength, checks } = checkPasswordStrength(password);
            
            meter.className = `strength-meter ${strength}`;
            text.innerHTML = `
                Strength: <strong>${strength.toUpperCase()}</strong><br>
                <small>
                    ${checks.length ? '✓' : '✗'} 8+ chars
                    ${checks.uppercase ? '✓' : '✗'} Uppercase
                    ${checks.lowercase ? '✓' : '✗'} Lowercase
                    ${checks.numbers ? '✓' : '✗'} Number
                    ${checks.special ? '✓' : '✗'} Special
                </small>
            `;
        }

        /**
         * Check password confirmation match
         */
        function checkPasswordMatch() {
            const password = document.getElementById('password').value;
            const confirm = document.getElementById('confirmPassword').value;
            const indicator = document.getElementById('matchIndicator');
            
            if (!confirm) {
                indicator.textContent = '';
                return;
            }
            
            indicator.textContent = password === confirm ? '✓ Match' : '✗ No match';
            indicator.style.color = password === confirm ? 'green' : 'red';
        }

        // Event listeners
        document.getElementById('password').addEventListener('input', (e) => {
            updateStrengthMeter(e.target.value);
            checkPasswordMatch();
        });

        document.getElementById('confirmPassword').addEventListener('input', checkPasswordMatch);

        // Form submission
        document.getElementById('passwordForm').addEventListener('submit', (e) => {
            e.preventDefault();
            
            const password = document.getElementById('password').value;
            const confirm = document.getElementById('confirmPassword').value;
            
            if (password !== confirm) {
                alert('Passwords do not match!');
                return;
            }
            
            const { strength } = checkPasswordStrength(password);
            
            document.getElementById('output').innerHTML = `
                <h3>Form Submitted</h3>
                <p>Password length: ${password.length}</p>
                <p>Strength: ${strength}</p>
                <p><em>(In production, send to server over HTTPS)</em></p>
            `;
        });
    </script>
```
