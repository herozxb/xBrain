        /**
         * Custom Range class implementing the iterable protocol
         * The protocol requires [Symbol.iterator]() method returning an iterator
         */
        class Range {
            constructor(start, end, step = 1) {
                this.start = start;
                this.end = end;
                this.step = step;
            }
            
            // Implement the iterable protocol
            [Symbol.iterator]() {
                let current = this.start;
                const end = this.end;
                const step = this.step;
                
                // Return an iterator object with next() method
                return {
                    next() {
                        if ((step > 0 && current < end) || (step < 0 && current > end)) {
                            const value = current;
                            current += step;
                            return { done: false, value };
                        }
                        return { done: true, value: undefined };
                    }
                };
            }
        }

        }
        results.push(`Values: ${values.join(', ')}`);
        
        // Works with spread operator
        results.push('<br><strong>With spread operator:</strong>');
        const range2 = new Range(0, 5);
        results.push(`[${...range2}]`);

        // PROBLEM: Use closures to maintain state in event handlers
        // TIME: O(1) per event SPACE: O(n) for closure scopes
        
        const logEl = document.getElementById('log');
        function log(message) {
            const entry = document.createElement('div');
            entry.textContent = `${new Date().toLocaleTimeString()}: ${message}`;
            logEl.prepend(entry);
            if (logEl.children.length > 10) {
                logEl.removeChild(logEl.lastChild);
            }

        // 1. Classic closure problem: Loop with var
        // PROBLEM: All buttons would log "Button 4" with var
        // SOLUTION: Use let or create closure with IIFE
        for (let i = 1; i <= 3; i++) {
            const btn = document.getElementById(`btn${i}`);
            btn.addEventListener('click', function() {
                // Closure captures 'i' correctly because of 'let'
                log(`Button ${i} clicked (closure captured i=${i})`);
            });
        }

        function createCounter() {
            let count = 0; // Private variable
            
            return {
                increment() {

                    return count;

                decrement() {
                    count--;
                    return count;

                getCount() {
                    return count;
                }

        const CounterModule = (function() {
            let count = 0;
            const listeners = [];
            
            return {
                add(fn) {
                    listeners.push(fn);
                },
                increment() {
                    count++;
                    listeners.forEach(fn => fn(count));
                    return count;
                },
                reset() {
                    count = 0;
                    listeners.forEach(fn => fn(count));
                }

            log(`Module counter updated: ${newCount}`);

            CounterModule.increment();

        function createHandler(action, multiplier) {
