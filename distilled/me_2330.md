```python
# PROBLEM: Compute all prime factors of a number with their exponents
# TIME: O(√n) SPACE: O(log n)
def prime_factorization(n: int) -> dict[int, int]:
    """
    Compute prime factorization of a positive integer.
    
    Args:
        n: Positive integer to factorize
        
    Returns:
        Dictionary mapping prime factors to their exponents
        
    Example:
        >>> prime_factorization(360)
        {2: 3, 3: 2, 5: 1}  # 360 = 2^3 * 3^2 * 5^1
    """
    if n < 1:
        raise ValueError("Input must be positive")
    if n == 1:
        return {}
    
    factors = {}
    divisor = 2
    
    # Check for factor of 2 separately for efficiency
    while n % 2 == 0:
        factors[2] = factors.get(2, 0) + 1
        n //= 2
    
    # Check odd divisors from 3 to √n
    divisor = 3
    while divisor * divisor <= n:
        while n % divisor == 0:
            factors[divisor] = factors.get(divisor, 0) + 1
            n //= divisor
        divisor += 2
    
    # If n is still greater than 1, it's a prime factor
    if n > 1:
        factors[n] = 1
    
    return factors
```

**Why**: Fundamental algorithm in number theory with applications in cryptography, number theory problems, and mathematical computations.

## Rod Cutting

```python
# PROBLEM: Maximize revenue by cutting a rod into pieces with given prices
# TIME: O(n²) SPACE: O(n)
def rod_cutting(prices: list[int], length: int) -> tuple[int, list[int]]:
    """
    Solve the rod cutting problem using dynamic programming.
    
    Args:
        prices: prices[i] is the price for a rod of length i+1
        length: Total length of the rod to cut
        
    Returns:
        Tuple of (maximum revenue, list of optimal cut lengths)
        
    Example:
        >>> prices = [1, 5, 8, 9, 10, 17, 17, 20]
        >>> rod_cutting(prices, 8)
        (22, [2, 6])  # Cut into lengths 2 and 6
    """
    if length <= 0:
        return 0, []
    
    # dp[i] stores maximum revenue for rod of length i
    dp = [0] * (length + 1)
    # cuts[i] stores the first cut length for optimal solution
    cuts = [0] * (length + 1)
    
    # Build solution bottom-up
    for i in range(1, length + 1):
        max_val = float('-inf')
        # Try all possible first cuts
        for j in range(1, min(i, len(prices)) + 1):
            # Price for piece of length j + optimal solution for remainder
            if j <= len(prices):
                val = prices[j - 1] + dp[i - j]
                if val > max_val:
                    max_val = val
                    cuts[i] = j
        dp[i] = max_val
    
    # Reconstruct the optimal cuts
    result_cuts = []
    remaining = length
    while remaining > 0:
        cut = cuts[remaining]
        result_cuts.append(cut)
        remaining -= cut
    
    return dp[length], result_cuts
```

**Why**: Classic dynamic programming problem demonstrating optimal substructure and memoization, applicable to resource allocation and pricing optimization.

```python
# PROBLEM: Implement a k-d tree for efficient range searches in k-dimensional space
# TIME: O(n log n) build, O(log n) search avg SPACE: O(n)
from typing import Generic, TypeVar, List, Tuple, Optional

T = TypeVar('T', int, float)

class KDNode(Generic[T]):
    """Node in a k-d tree."""
    def __init__(self, point: Tuple[T, ...], left: Optional['KDNode'] = None, 
                 right: Optional['KDNode'] = None):
        self.point = point
        self.left = left
        self.right = right

class KDTree(Generic[T]):
    """
    K-dimensional tree for efficient range searches.
    
    Supports operations like nearest neighbor search and range queries
    in k-dimensional space.
    
    Example:
        >>> points = [(2, 3), (5, 4), (9, 6), (4, 7), (8, 1), (7, 2)]
        >>> tree = KDTree(points, k=2)
        >>> tree.nearest_neighbor((9, 2))
        (8, 1)
    """
    
    def __init__(self, points: List[Tuple[T, ...]], k: int = 2):
        """
        Build k-d tree from list of points.
        
        Args:
            points: List of k-dimensional points
            k: Number of dimensions
        """
        self.k = k
        self.root = self._build(points, depth=0) if points else None
    
    def _build(self, points: List[Tuple[T, ...]], depth: int) -> Optional[KDNode]:
        """Recursively build k-d tree."""
        if not points:
            return None
        
        # Select axis based on depth
        axis = depth % self.k
        
        # Sort points by axis and choose median
        points.sort(key=lambda x: x[axis])
        median = len(points) // 2
        
        # Create node and construct subtrees
        return KDNode(
            point=points[median],
            left=self._build(points[:median], depth + 1),
            right=self._build(points[median + 1:], depth + 1)
        )
    
    def nearest_neighbor(self, target: Tuple[T, ...]) -> Optional[Tuple[T, ...]]:
        """
        Find the nearest neighbor to target point.
        
        Args:
            target: Query point
            
        Returns:
            Nearest point in the tree, or None if tree is empty
        """
        if not self.root:
            return None
        
        best = [self.root.point]
        best_dist = [self._distance(self.root.point, target)]
        
        def search(node: Optional[KDNode], depth: int):
            if not node:
                return
            
            # Update best if current node is closer
            dist = self._distance(node.point, target)
            if dist < best_dist[0]:
                best[0] = node.point
                best_dist[0] = dist
            
            # Determine which subtree to search first
            axis = depth % self.k
            if target[axis] < node.point[axis]:
                near, far = node.left, node.right
            else:
                near, far = node.right, node.left
            
            # Search nearer subtree first
            search(near, depth + 1)
            
            # Check if we need to search far subtree
            if (target[axis] - node.point[axis]) ** 2 < best_dist[0]:
                search(far, depth + 1)
        
        search(self.root, 0)
        return best[0]
    
    def _distance(self, p1: Tuple[T, ...], p2: Tuple[T, ...]) -> float:
        """Calculate squared Euclidean distance between two points."""
        return sum((a - b) ** 2 for a, b in zip(p1, p2))
```

**Why**: Essential data structure for spatial indexing, nearest neighbor searches, and range queries used in computer graphics, machine learning, and geographic information systems.

```python
# PROBLEM: Implement a pairing heap with efficient merge operation
# TIME: O(1) insert/merge, O(log n) amortized extract_min SPACE: O(n)
from typing import Generic, TypeVar, Optional, List

T = TypeVar('T')

class PairingHeapNode(Generic[T]):
    """Node in a pairing heap."""
    def __init__(self, value: T):
        self.value = value
        self.child: Optional['PairingHeapNode[T]'] = None
        self.sibling: Optional['PairingHeapNode[T]'] = None

class PairingHeap(Generic[T]):
    """
    Pairing heap - a heap with excellent amortized performance.
    
    Particularly efficient for merge operations, making it ideal for
    applications requiring frequent heap merges.
    
    Example:
        >>> heap = PairingHeap[int]()
        >>> heap.insert(5)
        >>> heap.insert(3)
        >>> heap.insert(7)
        >>> heap.extract_min()
        3
    """
    
    def __init__(self):
        self.root: Optional[PairingHeapNode[T]] = None
        self._size = 0
    
    def is_empty(self) -> bool:
        """Check if heap is empty."""
        return self.root is None
    
    def insert(self, value: T) -> None:
        """Insert a value into the heap."""
        new_node = PairingHeapNode(value)
        self.root = self._merge(self.root, new_node)
        self._size += 1
    
    def find_min(self) -> Optional[T]:
        """Return minimum value without removing it."""
        return self.root.value if self.root else None
    
    def extract_min(self) -> Optional[T]:
        """Remove and return the minimum value."""
        if not self.root:
            return None
        
        min_val = self.root.value
        self.root = self._merge_pairs(self.root.child)
        self._size -= 1
        return min_val
    
    def merge(self, other: 'PairingHeap[T]') -> None:
        """Merge another heap into this one."""
        if other.root:
            self.root = self._merge(self.root, other.root)
            self._size += other._size
    
    def _merge(self, h1: Optional[PairingHeapNode[T]], 
               h2: Optional[PairingHeapNode[T]]) -> Optional[PairingHeapNode[T]]:
        """Merge two heaps."""
        if not h1:
            return h2
        if not h2:
            return h1
        
        # Ensure h1 has smaller root
        if h1.value > h2.value:
            h1, h2 = h2, h1
        
        # Make h2 a child of h1
        h2.sibling = h1.child
        h1.child = h2
        return h1
    
    def _merge_pairs(self, node: Optional[PairingHeapNode[T]]) -> Optional[PairingHeapNode[T]]:
        """Merge pairs of trees from left to right, then right to left."""
        if not node or not node.sibling:
            return node
        
        # Collect all siblings
        trees: List[PairingHeapNode[T]] = []
        current = node
        while current:
            next_node = current.sibling
            current.sibling = None  # Clear sibling link
            trees.append(current)
            current = next_node
        
        # Merge pairs from left to right
        merged: List[PairingHeapNode[T]] = []
        i = 0
        while i < len(trees) - 1:
            merged.append(self._merge(trees[i], trees[i + 1]))
            i += 2
        if i == len(trees) - 1:
            merged.append(trees[i])
        
        # Merge from right to left
        result = merged[-1]
        for i in range(len(merged) - 2, -1, -1):
            result = self._merge(result, merged[i])
        
        return result
```

**Why**: Simple to implement with excellent practical performance, especially good for applications requiring frequent merge operations like graph algorithms.

## Lazy Initialization Pattern

```python
# PROBLEM: Defer object creation until first use to optimize resource usage
# TIME: O(1) after initialization SPACE: O(1)
from typing import Generic, TypeVar, Callable, Optional

T = TypeVar('T')

class Lazy(Generic[T]):
    """
    Lazy initialization wrapper for deferred object creation.
    
    The object is created only when first accessed, saving resources
    if the object is never used.
    
    Example:
        >>> lazy_list = Lazy(lambda: [1, 2, 3])
        >>> lazy_list.is_created()
        False
        >>> lazy_list.get()
        [1, 2, 3]
        >>> lazy_list.is_created()
        True
    """
    
    def __init__(self, factory: Callable[[], T]):
        """
        Initialize lazy wrapper.
        
        Args:
            factory: Function that creates the object
        """
        self._factory = factory
        self._value: Optional[T] = None
        self._created = False
    
    def get(self) -> T:
        """
        Get the value, creating it if necessary.
        
        Returns:
            The lazily created value
        """
        if not self._created:
            self._value = self._factory()
            self._created = True
        return self._value
    
    def is_created(self) -> bool:
        """Check if value has been created."""
        return self._created
    
    def reset(self) -> None:
        """Reset the lazy value, allowing it to be recreated."""
        self._value = None
        self._created = False

class LazyProperty:
    """
    Decorator for lazy evaluation of class properties.
    
    The property value is computed once on first access and cached.
    
    Example:
        >>> class DataProcessor:
        ...     @LazyProperty
        ...     def expensive_data(self):
        ...         print("Computing...")
        ...         return [1, 2, 3]
        >>> processor = DataProcessor()
        >>> processor.expensive_data  # Only computes once
        Computing...
        [1, 2, 3]
        >>> processor.expensive_data  # Returns cached value
        [1, 2, 3]
    """
    
    def __init__(self, method: Callable):
        self.method = method
        self.cache_name = f"_lazy_{method.__name__}"
    
    def __get__(self, instance, owner):
        if instance is None:
            return self
        
        if not hasattr(instance, self.cache_name):
            # Compute and cache the value
            value = self.method(instance)
            setattr(instance, self.cache_name, value)
        
        return getattr(instance, self.cache_name)
```

**Why**: Optimizes resource usage by deferring expensive initialization until actually needed, improving application startup time and memory efficiency.

## Event-Driven Architecture

```python
# PROBLEM: Implement event-driven system with loose coupling between components
# TIME: O(1) emit, O(n) for n subscribers SPACE: O(n)
from typing import Callable, Dict, List, Any
from dataclasses import dataclass
from datetime import datetime
from collections import defaultdict

@dataclass
class Event:
    """Base event class."""
    event_type: str
    timestamp: datetime
    data: Dict[str, Any]

class EventBus:
    """
    Central event bus for publish-subscribe pattern.
    
    Enables loose coupling between components by allowing them to
    communicate through events without direct dependencies.
    
    Example:
        >>> bus = EventBus()
        >>> def handler(event: Event):
        ...     print(f"Received: {event.event_type}")
        >>> bus.subscribe("user.created", handler)
        >>> bus.emit("user.created", {"user_id": 123})
        Received: user.created
    """
    
    def __init__(self):
        self._subscribers: Dict[str, List[Callable]] = defaultdict(list)
        self._event_log: List[Event] = []
    
    def subscribe(self, event_type: str, handler: Callable[[Event], None]) -> None:
        """
        Subscribe to an event type.
        
        Args:
            event_type: Type of event to subscribe to
            handler: Function to call when event occurs
        """
        self._subscribers[event_type].append(handler)
    
    def unsubscribe(self, event_type: str, handler: Callable[[Event], None]) -> bool:
        """
        Unsubscribe from an event type.
        
        Returns:
            True if handler was removed, False if not found
        """
        try:
            self._subscribers[event_type].remove(handler)
            return True
        except ValueError:
            return False
    
    def emit(self, event_type: str, data: Dict[str, Any] = None) -> Event:
        """
        Emit an event to all subscribers.
        
        Args:
            event_type: Type of event to emit
            data: Event data payload
            
        Returns:
            The emitted event
        """
        event = Event(
            event_type=event_type,
            timestamp=datetime.now(),
            data=data or {}
        )
        
        # Log the event
        self._event_log.append(event)
        
        # Notify all subscribers
        for handler in self._subscribers[event_type]:
            try:
                handler(event)
            except Exception as e:
                print(f"Error in event handler: {e}")
        
        return event
    
    def get_event_log(self) -> List[Event]:
        """Get all emitted events."""
        return self._event_log.copy()

# Example usage with microservices pattern
class OrderService:
    """Example service that emits events."""
    
    def __init__(self, event_bus: EventBus):
        self.event_bus = event_bus
    
    def create_order(self, order_id: str, items: list) -> dict:
        """Create an order and emit event."""
        order = {
            "order_id": order_id,
            "items": items,
            "status": "created"
        }
        
        # Emit event for other services to handle
        self.event_bus.emit("order.created", order)
        
        return order

class InventoryService:

```

**Why**: Enables scalable, maintainable systems with loose coupling between components, essential for microservices and distributed systems.

## Generic Repository Pattern

```python
# PROBLEM: Implement type-safe repository pattern with generics
# TIME: O(1) for in-memory operations SPACE: O(n)
from typing import Generic, TypeVar, Protocol, Optional, List, Dict
from dataclasses import dataclass
from abc import ABC, abstractmethod

T = TypeVar('T', bound='Entity')
ID = TypeVar('ID')

@dataclass
class Entity:
    """Base entity with ID."""
    id: str

class Repository(Protocol[T]):
    """Repository protocol defining CRUD operations."""
    
    def find_by_id(self, id: str) -> Optional[T]:
        """Find entity by ID."""
        ...
    
    def find_all(self) -> List[T]:
        """Find all entities."""
        ...
    
    def save(self, entity: T) -> T:
        """Save an entity."""
        ...
    
    def delete(self, id: str) -> bool:
        """Delete entity by ID."""
        ...

class InMemoryRepository(Generic[T], Repository[T]):
    """
    In-memory implementation of repository pattern.
    
    Provides type-safe CRUD operations for any entity type.
    
    Example:
        >>> @dataclass
        ... class User(Entity):
        ...     name: str
        >>> repo = InMemoryRepository[User]()
        >>> user = repo.save(User(id="1", name="Alice"))
        >>> repo.find_by_id("1")
        User(id='1', name='Alice')
    """
    
    def __init__(self):
        self._storage: Dict[str, T] = {}
    
    def find_by_id(self, id: str) -> Optional[T]:
        """Find entity by ID."""
        return self._storage.get(id)
    
    def find_all(self) -> List[T]:
        """Find all entities."""
        return list(self._storage.values())
    
    def save(self, entity: T) -> T:
        """Save an entity."""
        self._storage[entity.id] = entity
        return entity
    
    def delete(self, id: str) -> bool:
        """Delete entity by ID."""
        if id in self._storage:
            del self._storage[id]
            return True
        return False
    
    def count(self) -> int:
        """Count entities."""
        return len(self._storage)

# Example entity and specialized repository
@dataclass
class Product(Entity):
    """Product entity."""
    name: str
    price: float
    category: str

class ProductRepository(InMemoryRepository[Product]):
    """Specialized repository for products."""
    
    def find_by_category(self, category: str) -> List[Product]:
        """Find products by category."""
        return [p for p in self.find_all() if p.category == category]
    
    def find_by_price_range(self, min_price: float, max_price: float) -> List[Product]:
        """Find products within price range."""
        return [
            p for p in self.find_all() 
            if min_price <= p.price <= max_price
        ]
```

**Why**: Provides abstraction over data access, enabling easy testing, separation of concerns, and ability to swap data sources without changing business logic.

```python
# PROBLEM: Manage async resources with automatic cleanup
# TIME: O(1) for context switch SPACE: O(1)
from typing import AsyncIterator, TypeVar, Generic
from contextlib import asynccontextmanager
import asyncio

T = TypeVar('T')

class AsyncResource:
    """
    Example async resource requiring cleanup.
    
    Demonstrates proper async context manager implementation.
    """
    
    async def __aenter__(self):
        """Acquire resource asynchronously."""
        print("Acquiring async resource...")
        await asyncio.sleep(0.1)  # Simulate async setup
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        """Release resource asynchronously."""
        print("Releasing async resource...")
        await asyncio.sleep(0.1)  # Simulate async cleanup
        return False  # Don't suppress exceptions
    
    async def do_work(self):
        """Perform some async work."""
        await asyncio.sleep(0.05)
        return "Work completed"

class AsyncPool(Generic[T]):
    """
    Async resource pool with context management.
    
    Example:
        >>> async def main():
        ...     async with AsyncPool(max_size=3) as pool:
        ...         resource = await pool.acquire()
        ...         await resource.do_work()
    """
    
    def __init__(self, max_size: int = 10):
        self.max_size = max_size
        self._resources: asyncio.Queue = asyncio.Queue(maxsize=max_size)
        self._lock = asyncio.Lock()
    
    async def __aenter__(self) -> 'AsyncPool[T]':
        """Initialize pool."""
        await self._initialize_pool()
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        """Cleanup pool."""
        await self._cleanup_pool()
        return False
    
    async def _initialize_pool(self):
        """Initialize pool with resources."""
        for _ in range(self.max_size):
            await self._resources.put(await self._create_resource())
    
    async def _cleanup_pool(self):
        """Cleanup all resources."""
        while not self._resources.empty():
            resource = await self._resources.get()
            await self._close_resource(resource)
    
    async def _create_resource(self) -> T:
        """Create a new resource (override in subclass)."""
        return AsyncResource()
    
    async def _close_resource(self, resource: T):
        """Close a resource (override in subclass)."""
        pass
    
    @asynccontextmanager
    async def acquire(self) -> AsyncIterator[T]:
        """
        Acquire a resource from the pool.
        
        Automatically returns resource to pool when done.
        """
        resource = await self._resources.get()
        try:
            yield resource
        finally:
            await self._resources.put(resource)
```

```python
# PROBLEM: Implement B+ tree for database indexing with efficient range queries
# TIME: O(log n) search/insert, O(k) for k results in range SPACE: O(n)
from typing import Generic, TypeVar, List, Optional, Tuple

K = TypeVar('K')  # Key type
V = TypeVar('V')  # Value type

class BPlusTreeNode(Generic[K, V]):
    """Node in B+ tree."""
    
    def __init__(self, is_leaf: bool = True):
        self.is_leaf = is_leaf
        self.keys: List[K] = []
        self.children: List['BPlusTreeNode[K, V]'] = [] if not is_leaf else []
        self.values: List[V] = [] if is_leaf else []
        self.next: Optional['BPlusTreeNode[K, V]'] = None  # For leaf nodes

class BPlusTree(Generic[K, V]):
    """
    B+ tree implementation for database indexing.
    
    All data stored in leaf nodes, internal nodes only contain keys.
    Leaf nodes linked for efficient range queries.
    
    Example:
        >>> tree = BPlusTree[int, str](order=4)
        >>> tree.insert(10, "A")
        >>> tree.insert(20, "B")
        >>> tree.search(10)
        'A'
        >>> tree.range_query(5, 15)
        [('A', 10)]
    """
    
    def __init__(self, order: int = 4):
        """
        Initialize B+ tree.
        
        Args:
            order: Maximum number of children per node
        """
        self.order = order
        self.root = BPlusTreeNode(is_leaf=True)
        self._size = 0
    
    def search(self, key: K) -> Optional[V]:
        """
        Search for a key in the tree.
        
        Args:
            key: Key to search for
            
        Returns:
            Associated value or None if not found
        """
        return self._search_node(self.root, key)
    
    def _search_node(self, node: BPlusTreeNode[K, V], key: K) -> Optional[V]:
        """Recursively search in node."""
        if node.is_leaf:
            # Linear search in leaf
            for i, k in enumerate(node.keys):
                if k == key:
                    return node.values[i]
            return None
        else:
            # Find child to descend into
            i = 0
            while i < len(node.keys) and key >= node.keys[i]:
                i += 1
            return self._search_node(node.children[i], key)
    
    def insert(self, key: K, value: V) -> None:
        """
        Insert key-value pair into tree.
        
        Args:
            key: Key to insert
            value: Associated value
        """
        result = self._insert_node(self.root, key, value)
        
        if result:  # Root was split
            new_root = BPlusTreeNode(is_leaf=False)
            new_root.keys = [result[0]]
            new_root.children = [self.root, result[1]]
            self.root = new_root
        
        self._size += 1
    
    def _insert_node(self, node: BPlusTreeNode[K, V], key: K, value: V) -> Optional[Tuple[K, BPlusTreeNode[K, V]]]:
        """Insert into node and handle splits."""
        if node.is_leaf:
            # Insert in sorted order
            i = 0
            while i < len(node.keys) and node.keys[i] < key:
                i += 1
            
            node.keys.insert(i, key)
            node.values.insert(i, value)
            
            # Split if necessary
            if len(node.keys) > self.order - 1:
                return self._split_leaf(node)
            return None
        else:
            # Find child to insert into
            i = 0
            while i < len(node.keys) and key >= node.keys[i]:
                i += 1
            
            result = self._insert_node(node.children[i], key, value)
            
            if result:  # Child was split
                mid_key, new_child = result
                # Insert middle key and new child
                node.keys.insert(i, mid_key)
                node.children.insert(i + 1, new_child)
                
                # Split if necessary
                if len(node.keys) > self.order - 1:
                    return self._split_internal(node)
            
            return None
    
    def _split_leaf(self, node: BPlusTreeNode[K, V]) -> Tuple[K, BPlusTreeNode[K, V]]:
        """Split leaf node and return (middle_key, new_node)."""
        mid = len(node.keys) // 2
        
        new_node = BPlusTreeNode(is_leaf=True)
        new_node.keys = node.keys[mid:]
        new_node.values = node.values[mid:]
        
        node.keys = node.keys[:mid]
        node.values = node.values[:mid]
        
        # Maintain linked list
        new_node.next = node.next
        node.next = new_node
        
        return (new_node.keys[0], new_node)
    
    def _split_internal(self, node: BPlusTreeNode[K, V]) -> Tuple[K, BPlusTreeNode[K, V]]:
        """Split internal node and return (middle_key, new_node)."""
        mid = len(node.keys) // 2
        
        new_node = BPlusTreeNode(is_leaf=False)
        new_node.keys = node.keys[mid + 1:]
        new_node.children = node.children[mid + 1:]
        
        mid_key = node.keys[mid]
        
        node.keys = node.keys[:mid]
        node.children = node.children[:mid + 1]
        
        return (mid_key, new_node)
    
    def range_query(self, start: K, end: K) -> List[Tuple[V, K]]:
        """
        Query all key-value pairs in range [start, end].
        
        Args:
            start: Start of range (inclusive)
            end: End of range (inclusive)
            
        Returns:
            List of (value, key) tuples in range
        """
        results = []
        
        # Find starting leaf
        node = self._find_leaf(self.root, start)
        
        # Iterate through leaves
        while node:
            for i, k in enumerate(node.keys):
                if start <= k <= end:
                    results.append((node.values[i], k))
                elif k > end:
                    return results
            node = node.next
        
        return results
    
    def _find_leaf(self, node: BPlusTreeNode[K, V], key: K) -> BPlusTreeNode[K, V]:
        """Find leaf node that should contain key."""
        if node.is_leaf:
            return node
        
        i = 0
        while i < len(node.keys) and key >= node.keys[i]:
            i += 1
        return self._find_leaf(node.children[i], key)
```
