```html
            cursor: pointer;
            font-size: 1rem;
        }
        
        button:hover {
            background: #2980b9;
        }
    </style>
</head>
<body>
    <h1>Email Input Demo</h1>
    
    <form id="emailForm" novalidate>
        <div class="form-group">
            <label for="email">Email Address *</label>
            <!-- type="email" provides built-in validation -->
            <input 
                type="email" 
                id="email" 
                name="email"
                placeholder="you@example.com"
                required
                autocomplete="email"
                multiple
            >
            <p class="error-message">Please enter a valid email address</p>
            <p class="hint">Multiple emails allowed (comma-separated)</p>
        </div>
        
        <button type="submit">Subscribe</button>
    </form>
    
    <script>
        const form = document.getElementById('emailForm');
        const emailInput = document.getElementById('email');
        
        // Real-time validation feedback
        emailInput.addEventListener('input', function() {
            // Check validity using constraint validation API
            if (this.validity.typeMismatch) {
                this.setCustomValidity('Please enter a valid email address');
            } else if (this.validity.valueMissing) {
                this.setCustomValidity('Email is required');
            } else {
                this.setCustomValidity('');
            }
        });
        
        // Form submission with validation
        form.addEventListener('submit', function(e) {
            e.preventDefault();
            
            if (!emailInput.checkValidity()) {
                emailInput.reportValidity();
                return;
            }
            
            // Parse multiple emails if present
            const emails = emailInput.value
                .split(',')
                .map(e => e.trim())
                .filter(e => e.length > 0);
            
            console.log('Submitted emails:', emails);
            alert(`Subscribed: ${emails.join(', ')}`);
        });
        
        // Advanced email regex (optional enhancement)
        const validateEmailStrict = (email) => {
            const regex = /^[a-zA-Z0-9.!#$%&'*+/=?^_`{|}~-]+@[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?(?:\.[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?)*$/;
            return regex.test(email);
        };
    </script>
</body>
</html>
```

**Why**: `type="email"` provides built-in validation, mobile keyboard optimization (@ symbol), and works with `multiple` attribute for comma-separated emails. Use `autocomplete="email"` for better UX. Always combine with server-side validation.

## Summary

These 10 HTML elements cover essential semantic markup:

| Element | Purpose | Key Features |
|---------|---------|--------------|
| `<header>` | Page/section header | Logo, navigation, intro content |
| `<nav>` | Navigation block | Links, aria-label for multiple navs |
| `<main>` | Primary content | One per page, skip link target |
| `<article>` | Self-contained content | Syndicatable, blog posts, comments |
| `<section>` | Thematic grouping | Requires heading, document outline |
| `<input type="email">` | Email input | Built-in validation, mobile keyboard |
| `<input type="number">` | Numeric input | Min/max/step, spinner controls |
| `<input type="date">` | Date picker | Native UI, YYYY-MM-DD format |
| `<td>` | Table data cell | colspan, rowspan, numeric alignment |
| `<dl>` | Definition list | Term-description pairs, metadata |

**Best Practices:**
- Always use semantic elements over `<div>` when meaning exists
- Combine with ARIA attributes for enhanced accessibility
- Validate on both client and server side
- Test with screen readers for accessibility compliance
