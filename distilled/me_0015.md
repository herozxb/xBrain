## Inbox Pattern

```python
# PROBLEM: Implement inbox pattern for reliable message delivery with deduplication
# TIME: O(1) for insert/lookup, O(n) for batch processing SPACE: O(n)
class InboxPattern:
    def __init__(self):
        self.inbox = {}  # message_id -> message
        self.processed = set()  # processed message IDs
    
    def receive_message(self, message_id, message):
        """Store incoming message if not already processed."""
        if message_id in self.processed:
            return False  # Already processed, skip
        
        if message_id not in self.inbox:
            self.inbox[message_id] = {
                'message': message,
                'received_at': time.time(),
                'attempts': 0
            }
        return True
    
    def process_batch(self, processor_func, batch_size=10):
        """Process messages in batches with idempotency."""
        processed_count = 0
        for msg_id in list(self.inbox.keys())[:batch_size]:
            if msg_id not in self.processed:
                try:
                    processor_func(self.inbox[msg_id]['message'])
                    self.processed.add(msg_id)
                    del self.inbox[msg_id]
                    processed_count += 1
                except Exception as e:
                    self.inbox[msg_id]['attempts'] += 1
        return processed_count
```
**Why**: Ensures exactly-once processing by tracking processed IDs, preventing duplicate message handling.

## Strangler Fig Pattern

```python
# PROBLEM: Incrementally migrate legacy system by intercepting and routing requests
# TIME: O(1) for routing SPACE: O(n) for feature flags
class StranglerFigRouter:
    def __init__(self, legacy_system, new_system):
        self.legacy = legacy_system
        self.new_system = new_system
        self.feature_flags = {}  # endpoint -> migration_percentage
    
    def route_request(self, endpoint, request_data):
        """Route requests between legacy and new system based on migration status."""
        migration_pct = self.feature_flags.get(endpoint, 0)
        
        # Deterministic routing based on request ID for consistency
        if self._should_use_new_system(request_data['id'], migration_pct):
            try:
                return self.new_system.handle(endpoint, request_data)
            except Exception as e:
                # Fallback to legacy on failure
                return self.legacy.handle(endpoint, request_data)
        else:
            return self.legacy.handle(endpoint, request_data)
    
    def _should_use_new_system(self, request_id, percentage):
        """Deterministically route based on request ID hash."""
        return (hash(request_id) % 100) < percentage
    
    def update_migration(self, endpoint, percentage):
        """Update migration percentage for an endpoint."""
        self.feature_flags[endpoint] = percentage
```
**Why**: Allows gradual system migration without big-bang deployments, reducing risk and enabling rollback.

## Distributed Logging

```python
# PROBLEM: Aggregate logs from multiple services with correlation IDs
# TIME: O(1) for append, O(log n) for query SPACE: O(n)
import threading
from collections import defaultdict
from datetime import datetime

class DistributedLogger:
    def __init__(self):
        self.logs = defaultdict(list)  # correlation_id -> logs
        self.lock = threading.Lock()
    
    def log(self, correlation_id, service_name, level, message, metadata=None):
        """Append log entry with correlation tracking."""
        entry = {
            'timestamp': datetime.utcnow().isoformat(),
            'correlation_id': correlation_id,
            'service': service_name,
            'level': level,
            'message': message,
            'metadata': metadata or {}
        }
        
        with self.lock:
            self.logs[correlation_id].append(entry)
    
    def get_correlated_logs(self, correlation_id):
        """Retrieve all logs for a request flow across services."""
        return sorted(
            self.logs.get(correlation_id, []),
            key=lambda x: x['timestamp']
        )
    
    def search_by_service(self, service_name, start_time, end_time):
        """Search logs by service within time range."""
        results = []
        for entries in self.logs.values():
            for entry in entries:
                if (entry['service'] == service_name and 
                    start_time <= entry['timestamp'] <= end_time):
                    results.append(entry)
        return results
```
**Why**: Enables tracing request flows across microservices, critical for debugging distributed systems.

## CQRS with Event Sourcing

```python
# PROBLEM: Separate read and write models with event-based state reconstruction
# TIME: O(1) write, O(n) replay SPACE: O(n) for events
from dataclasses import dataclass
from typing import List, Dict
from datetime import datetime

@dataclass
class Event:
    event_type: str
    aggregate_id: str
    data: dict
    timestamp: str

class CQRSEventStore:
    def __init__(self):
        self.events = []  # Write model: event log
        self.read_models = {}  # Read model: projected state
    
    def append_event(self, event_type, aggregate_id, data):
        """Write-only: append event to store."""
        event = Event(
            event_type=event_type,
            aggregate_id=aggregate_id,
            data=data,
            timestamp=datetime.utcnow().isoformat()
        )
        self.events.append(event)
        self._update_read_model(event)
        return event
    
    def _update_read_model(self, event):
        """Project events to read model for fast queries."""
        if event.aggregate_id not in self.read_models:
            self.read_models[event.aggregate_id] = {}
        
        model = self.read_models[event.aggregate_id]
        model['last_updated'] = event.timestamp
        
        if event.event_type == 'created':
            model.update(event.data)
        elif event.event_type == 'updated':
            model.update(event.data)
        elif event.event_type == 'deleted':
            model['deleted'] = True
    
    def get_state(self, aggregate_id):
        """Read-only: get current projected state."""
        return self.read_models.get(aggregate_id)
    
    def replay_events(self, aggregate_id):
        """Rebuild state from event history."""
        return [e for e in self.events if e.aggregate_id == aggregate_id]
```
**Why**: Enables audit trails, temporal queries, and optimized read/write paths for complex domains.

## Columnar Storage

```python
# PROBLEM: Store data in column-oriented format for analytical queries
# TIME: O(1) column append, O(n) row reconstruction SPACE: O(n)
import numpy as np
from typing import Dict, List

class ColumnarStore:
    def __init__(self):
        self.columns: Dict[str, List] = {}
        self.row_count = 0
    
    def insert_row(self, row_data: dict):
        """Append values to each column."""
        for col_name, value in row_data.items():
            if col_name not in self.columns:
                self.columns[col_name] = []
            self.columns[col_name].append(value)
        self.row_count += 1
    
    def get_column(self, col_name):
        """Retrieve entire column for aggregation."""
        return self.columns.get(col_name, [])
    
    def aggregate_column(self, col_name, operation='sum'):
        """Perform aggregation on single column efficiently."""
        col_data = self.get_column(col_name)
        if not col_data:
            return None
        
        if operation == 'sum':
            return sum(col_data)
        elif operation == 'avg':
            return sum(col_data) / len(col_data)
        elif operation == 'min':
            return min(col_data)
        elif operation == 'max':
            return max(col_data)
    
    def get_row(self, row_idx):
        """Reconstruct row from columns (slower)."""
        return {col: self.columns[col][row_idx] for col in self.columns}
    
    def compress_column(self, col_name):
        """RLE compression for repetitive data."""
        col_data = self.get_column(col_name)
        if not col_data:
            return []
        
        compressed = []
        current_val = col_data[0]
        count = 1
        
        for val in col_data[1:]:
            if val == current_val:
                count += 1
            else:
                compressed.append((current_val, count))
                current_val = val
                count = 1
        compressed.append((current_val, count))
        return compressed
```
**Why**: Optimizes analytical workloads by reading only needed columns, enabling better compression and vectorized operations.

## Time Series Optimization

```python
# PROBLEM: Efficiently store and query time-series data with downsampling
# TIME: O(1) write, O(log n) query SPACE: O(n) with compression
from collections import deque
from datetime import datetime, timedelta

class TimeSeriesDB:
    def __init__(self, retention_days=30):
        self.raw_data = deque()  # Raw high-resolution data
        self.downsampled = {  # Aggregated buckets
            '1h': [],
            '1d': []
        }
        self.retention = timedelta(days=retention_days)
    
    def insert(self, timestamp, value, tags=None):
        """Insert data point with automatic downsampling."""
        self.raw_data.append({
            'ts': timestamp,
            'val': value,
            'tags': tags or {}
        })
        self._downsample()
        self._enforce_retention()
    
    def _downsample(self):
        """Aggregate data into time buckets."""
        # Hour aggregation
        hour_buckets = {}
        for point in self.raw_data:
            hour_key = point['ts'].replace(minute=0, second=0, microsecond=0)
            if hour_key not in hour_buckets:
                hour_buckets[hour_key] = {'sum': 0, 'count': 0}
            hour_buckets[hour_key]['sum'] += point['val']
            hour_buckets[hour_key]['count'] += 1
        
        self.downsampled['1h'] = [
            {'ts': k, 'avg': v['sum'] / v['count']}
            for k, v in sorted(hour_buckets.items())
        ]
    
    def query_range(self, start, end, resolution='raw'):
        """Query data within time range at specified resolution."""
        if resolution == 'raw':
            return [p for p in self.raw_data if start <= p['ts'] <= end]
        else:
            return [p for p in self.downsampled.get(resolution, [])
                    if start <= p['ts'] <= end]
    
    def _enforce_retention(self):
        """Remove old data based on retention policy."""
        cutoff = datetime.utcnow() - self.retention
        while self.raw_data and self.raw_data[0]['ts'] < cutoff:
            self.raw_data.popleft()
```
**Why**: Reduces storage costs and query latency through automatic downsampling and retention policies.

## Sharding Strategies

```python
# PROBLEM: Distribute data across multiple shards with consistent hashing
# TIME: O(log n) for shard lookup SPACE: O(n)
import hashlib
from bisect import bisect

class ConsistentHashSharding:
    def __init__(self, num_shards, replicas=150):
        self.shards = {}
        self.ring = []  # Sorted hash positions
        self.replicas = replicas
        self._initialize_ring(num_shards)
    
    def _initialize_ring(self, num_shards):
        """Create virtual nodes for each shard."""
        for i in range(num_shards):
            shard_name = f"shard_{i}"
            for replica in range(self.replicas):
                virtual_key = f"{shard_name}:{replica}"
                hash_pos = self._hash(virtual_key)
                self.ring.append(hash_pos)
                self.shards[hash_pos] = shard_name
        self.ring.sort()
    
    def _hash(self, key):
        """Generate consistent hash position."""
        return int(hashlib.md5(key.encode()).hexdigest(), 16)
    
    def get_shard(self, key):
        """Find shard responsible for given key."""
        hash_pos = self._hash(key)
        idx = bisect(self.ring, hash_pos)
        
        # Wrap around to first shard if past end
        if idx >= len(self.ring):
            idx = 0
        
        return self.shards[self.ring[idx]]
    
    def get_shards_for_range(self, start_key, end_key):
        """Get all shards that might contain range data."""
        start_shard = self.get_shard(start_key)
        end_shard = self.get_shard(end_key)
        # In practice, need more sophisticated range logic
        return list(set([start_shard, end_shard]))
```
**Why**: Enables horizontal scaling with minimal data redistribution when adding/removing shards.

## Read Replica Routing

```python
# PROBLEM: Route read queries to replicas for load distribution
# TIME: O(1) for routing SPACE: O(n)
import random
from enum import Enum

class ReadWrite(Enum):
    READ = "read"
    WRITE = "write"

class ReplicaRouter:
    def __init__(self, primary, replicas):
        self.primary = primary
        self.replicas = replicas
        self.replica_health = {r: True for r in replicas}
        self.read_counter = 0
    
    def route(self, query_type, query):
        """Route queries based on type and replica health."""
        if query_type == ReadWrite.WRITE:
            return self._execute_write(query)
        else:
            return self._execute_read(query)
    
    def _execute_write(self, query):
        """All writes go to primary."""
        return self.primary.execute(query)
    
    def _execute_read(self, query):
        """Distribute reads across healthy replicas."""
        healthy = [r for r, healthy in self.replica_health.items() if healthy]
        
        if not healthy:
            # Fallback to primary if all replicas down
            return self.primary.execute(query)
        
        # Round-robin load balancing
        replica = healthy[self.read_counter % len(healthy)]
        self.read_counter += 1
        
        try:
            return replica.execute(query)
        except Exception:
            # Mark unhealthy and retry with next replica
            self.replica_health[replica] = False
            return self._execute_read(query)
    
    def health_check(self):
        """Periodically check and restore replica health."""
        for replica in self.replicas:
            try:
                replica.ping()
                self.replica_health[replica] = True
            except:
                self.replica_health[replica] = False
```
**Why**: Improves read throughput and provides fallback options during replica failures.

## Flyweight Memory Optimization

```python
# PROBLEM: Share common parts of objects to reduce memory usage
# TIME: O(1) for flyweight lookup SPACE: O(k) where k is unique shared states
class FlyweightFactory:
    def __init__(self):
        self.flyweights = {}  # Shared intrinsic state
    
    def get_flyweight(self, shared_state):
        """Get or create flyweight for shared state."""
        key = self._make_key(shared_state)
        
        if key not in self.flyweights:
            self.flyweights[key] = {
                'state': shared_state,
                'ref_count': 0
            }
        
        self.flyweights[key]['ref_count'] += 1
        return key
    
    def _make_key(self, state):
        """Create hashable key from state."""
        if isinstance(state, dict):
            return tuple(sorted(state.items()))
        return str(state)
    
    def release_flyweight(self, key):
        """Decrement reference count and cleanup if zero."""
        if key in self.flyweights:
            self.flyweights[key]['ref_count'] -= 1
            if self.flyweights[key]['ref_count'] <= 0:
                del self.flyweights[key]

class Tree:
    """Example: trees in a forest share texture/color flyweights."""
    def __init__(self, x, y, flyweight_factory, texture, color):
        self.x = x  # Extrinsic state (unique to each tree)
        self.y = y
        # Intrinsic state (shared via flyweight)
        self.texture_key = flyweight_factory.get_flyweight(
            {'texture': texture, 'color': color}
        )
```
**Why**: Dramatically reduces memory when many objects share common immutable state.

## Cache-Oblivious Algorithms

```python
# PROBLEM: Algorithms that perform well across all cache hierarchy levels
# TIME: O(n log n) with optimal cache efficiency SPACE: O(n)
def cache_oblivious_matrix_multiply(A, B, result=None):
    """
    Recursive matrix multiplication with optimal cache behavior.
    Divides matrices until they fit in cache automatically.
    """
    n = len(A)
    
    if result is None:
        result = [[0] * n for _ in range(n)]
    
    # Base case: small enough for cache
    if n <= 16:  # Threshold tuned experimentally
        for i in range(n):
            for k in range(n):
                for j in range(n):
                    result[i][j] += A[i][k] * B[k][j]
        return result
    
    # Divide into submatrices
    mid = n // 2
    
    # Extract submatrices
    A11 = [row[:mid] for row in A[:mid]]
    A12 = [row[mid:] for row in A[:mid]]
    A21 = [row[:mid] for row in A[mid:]]
    A22 = [row[mid:] for row in A[mid:]]
    
    B11 = [row[:mid] for row in B[:mid]]
    B12 = [row[mid:] for row in B[:mid]]
    B21 = [row[:mid] for row in B[mid:]]
    B22 = [row[mid:] for row in B[mid:]]
    
    # Recursive multiplications
    cache_oblivious_matrix_multiply(A11, B11, result)
    cache_oblivious_matrix_multiply(A12, B21, result)
    
    return result

def cache_oblivious_scan(arr):
    """
    Funnel-shaped scan pattern for optimal cache usage.
    """
    def scan_recursive(arr, start, end):
        if end - start <= 1:
            return arr[start] if start < len(arr) else 0
        
        mid = (start + end) // 2
        left_sum = scan_recursive(arr, start, mid)
        right_sum = scan_recursive(arr, mid, end)
        
        return left_sum + right_sum
    
    return scan_recursive(arr, 0, len(arr))
```
**Why**: Automatically adapts to any cache size without tuning, optimal for modern multi-level cache hierarchies.
