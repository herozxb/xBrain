## js_array_every
```html

    /**
     * Array.every(callback(element, index, array))
     * PROBLEM: Check if ALL elements pass a test condition
     * TIME: O(n) SPACE: O(1)
     * 
     * Returns true if callback returns truthy for ALL elements.
     * Short-circuits on first false (stops iterating).
     * Returns true for empty arrays (vacuous truth).
     */
    
    // Basic usage: check if all numbers are positive
    const positiveNumbers = [1, 2, 3, 4, 5];
    const allPositive = positiveNumbers.every(n => n > 0);
    console.log('All positive:', allPositive); // true
    
    const mixedNumbers = [1, -2, 3, 4, 5];
    const allPositiveMixed = mixedNumbers.every(n => n > 0);
    console.log('All positive (mixed):', allPositiveMixed); // false
    
    // Check if all users are adults
    const users = [
      { name: 'Alice', age: 25 },
      { name: 'Bob', age: 30 },
      { name: 'Charlie', age: 22 }
    ];
    
    const allAdults = users.every(user => user.age >= 18);
    console.log('All adults:', allAdults); // true
    
    // Form validation: check all fields are filled
    const formData = {
      username: 'john_doe',
      email: 'john@example.com',
      password: 'secure123'
    };
    
    const allFieldsFilled = Object.values(formData).every(
      value => value && value.trim() !== ''
    );
    console.log('All fields filled:', allFieldsFilled); // true
    
    // Check if all items are in stock
    const cartItems = [
      { product: 'Laptop', inStock: true },
      { product: 'Mouse', inStock: true },
      { product: 'Keyboard', inStock: true }
    ];
    
    const allInStock = cartItems.every(item => item.inStock);
    console.log('All in stock:', allInStock); // true
    
    // Check if array is sorted ascending
    const sortedArray = [1, 2, 3, 4, 5];
    const isSorted = sortedArray.every((val, i, arr) => 
      i === 0 || arr[i - 1] <= val
    );
    console.log('Is sorted:', isSorted); // true
    
    // Password strength checker
    function isStrongPassword(password) {
      const checks = {
        'length >= 8': password.length >= 8,
        'has uppercase': /[A-Z]/.test(password),
        'has lowercase': /[a-z]/.test(password),
        'has number': /[0-9]/.test(password),
        'has special': /[^A-Za-z0-9]/.test(password)
      };
      
      const allPassed = Object.values(checks).every(Boolean);
      return { allPassed, checks };
    }
    
    console.log(isStrongPassword('Pass123!'));
    // { allPassed: true, checks: {...} }
    
    // Check all checkboxes are checked
    const checkboxes = document.querySelectorAll('.checkbox');
    // const allChecked = Array.from(checkboxes).every(cb => cb.checked);
    
    // Empty array always returns true (vacuous truth)
    const emptyArray = [];
    const emptyResult = emptyArray.every(x => x > 1000);
    console.log('Empty array every:', emptyResult); // true
    
    // Short-circuit demonstration
    let iterations = 0;
    const shortCircuit = [1, 2, -3, 4, 5].every(n => {
      iterations++;
      console.log(`Checking ${n}`);
      return n > 0;
    });
    console.log('Short-circuit at iteration:', iterations); // 3
    
    // Practical: Validate permissions
    const requiredPermissions = ['read', 'write', 'delete'];
    const userPermissions = ['read', 'write', 'delete', 'admin'];
    
    const hasAllPermissions = requiredPermissions.every(
      perm => userPermissions.includes(perm)
    );

  </script>
</body>
</html>
```
**Why**: Tests if every element satisfies a condition. Returns immediately on first failure (short-circuit). Perfect for form validation, permission checks, and verifying data integrity. Remember: empty arrays return true.

```html
     * Array.sort(compareFunction(a, b))
     * PROBLEM: Sort array elements in place
     * TIME: O(n log n) average SPACE: O(log n) for Quicksort/Timsort
     * 
     * WITHOUT compare function: converts to strings, sorts by UTF-16!
     * WITH compare function: returns negative/zero/positive
     *   - negative: a comes before b
     *   - positive: a comes after b  
     *   - zero: order unchanged (stable sort preserves original)
     */
    
    // WARNING: Default string sorting!
    const numbers = [10, 2, 30, 4, 100];
    numbers.sort(); // WRONG: [10, 100, 2, 30, 4] (string order!)
    console.log('String sort:', numbers);
    
    // Correct numeric sorting (ascending)
    const nums = [10, 2, 30, 4, 100];
    nums.sort((a, b) => a - b);
    console.log('Numeric ascending:', nums); // [2, 4, 10, 30, 100]
    
    // Numeric descending
    const numsDesc = [10, 2, 30, 4, 100];
    numsDesc.sort((a, b) => b - a);
    console.log('Numeric descending:', numsDesc); // [100, 30, 10, 4, 2]
    
    // Sort objects by property
    const products = [
      { name: 'Laptop', price: 999 },

      { name: 'Keyboard', price: 79 }

    console.log('By price:', products.map(p => p.name)); // Mouse, Keyboard, Laptop
    
    // Sort by name alphabetically
    products.sort((a, b) => a.name.localeCompare(b.name));
    console.log('By name:', products.map(p => p.name)); // Keyboard, Laptop, Mouse
    
    // Sort by multiple criteria (price then name)

    items.sort((a, b) => {
      if (a.price !== b.price) return a.price - b.price;
      return a.name.localeCompare(b.name);
    });
    console.log('Multi-sort:', items);
    // [{name: 'Apple', price: 1}, {name: 'Apricot', price: 1}, ...]

    ];
    
    events.sort((a, b) => a.date - b.date);

    // Random shuffle (Fisher-Yates via sort is not truly uniform)
    const shuffleArray = arr => {
      const shuffled = [...arr];
      for (let i = shuffled.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];

      return shuffled;
    };
    
    // Case-insensitive string sort
    const names = ['bob', 'Alice', 'charlie', 'Bob'];
    names.sort((a, b) => a.toLowerCase().localeCompare(b.toLowerCase()));
    console.log('Case-insensitive:', names);
    
    // Sort with null/undefined handling (push to end)
    const mixed = [3, null, 1, undefined, 2];
    mixed.sort((a, b) => {
      if (a == null) return 1;
      if (b == null) return -1;
      return a - b;
    });
    console.log('Null-safe sort:', mixed); // [1, 2, 3, null, undefined]
    
    // Immutable sort (non-mutating)
    const original = [3, 1, 2];
    const sorted = [...original].sort((a, b) => a - b);
    console.log('Original preserved:', original); // [3, 1, 2]
    console.log('Sorted copy:', sorted); // [1, 2, 3]
    
    // Top K elements (partial sort alternative)
    const scores = [95, 87, 92, 78, 88, 91, 85];
    const top3 = [...scores].sort((a, b) => b - a).slice(0, 3);
    console.log('Top 3:', top3); // [95, 92, 91]
  </script>
</body>
</html>
```
**Why**: Sorts arrays in-place with customizable comparison. WARNING: default converts to strings! Use (a,b)=>a-b for numbers. Supports multi-criteria sorting, date sorting, and custom object ordering. Use [...arr].sort() for immutable version.

## js_sessionStorage_setItem
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>sessionStorage.setItem() Method</title>
</head>
<body>
  <script>
    /**
     * sessionStorage.setItem(key, value)
     * PROBLEM: Store data temporarily for the current browser session
     * TIME: O(1) SPACE: Limited to ~5MB per origin
     * 
     * - Data persists only until tab/window closes
     * - Values are always stored as STRINGS
     * - Separate storage per origin (domain+protocol+port)
     * - Not shared between tabs (unlike localStorage)
     */
    
    // Basic storage
    sessionStorage.setItem('username', 'john_doe');
    sessionStorage.setItem('theme', 'dark');
    
    // Numbers are converted to strings!
    sessionStorage.setItem('count', 42);
    console.log(typeof sessionStorage.getItem('count')); // "string"
    
    // Storing objects (must stringify)
    const user = {
      id: 123,
      name: 'Alice',
      role: 'admin'
    };
    sessionStorage.setItem('user', JSON.stringify(user));
    
    // Retrieving and parsing
    const storedUser = JSON.parse(sessionStorage.getItem('user'));
    console.log(storedUser.name); // Alice
    
    // Form data persistence (page refresh survival)
    function saveFormData(formData) {
      sessionStorage.setItem('formData', JSON.stringify(formData));
    }
    
    function loadFormData() {
      const data = sessionStorage.getItem('formData');
      return data ? JSON.parse(data) : null;
    }
    
    // Wizard/multi-step form tracking
    function saveWizardStep(step, data) {
      sessionStorage.setItem('wizardStep', step.toString());
      sessionStorage.setItem(`step${step}Data`, JSON.stringify(data));
    }
    
    // Temporary authentication token
    function setSessionToken(token) {
      sessionStorage.setItem('authToken', token);
      sessionStorage.setItem('tokenExpiry', Date.now().toString());
    }
    
    function getSessionToken() {
      const token = sessionStorage.getItem('authToken');
      const expiry = parseInt(sessionStorage.getItem('tokenExpiry') || '0');
      
      if (Date.now() - expiry > 3600000) { // 1 hour
        clearSessionToken();
        return null;
      }
      return token;
    }
    
    function clearSessionToken() {
      sessionStorage.removeItem('authToken');
      sessionStorage.removeItem('tokenExpiry');
    }
    
    // Shopping cart for session
    const cartManager = {
      getCart: () => JSON.parse(sessionStorage.getItem('cart') || '[]'),
      
      addToCart: (item) => {
        const cart = cartManager.getCart();
        cart.push(item);
        sessionStorage.setItem('cart', JSON.stringify(cart));
      },
      
      clearCart: () => sessionStorage.removeItem('cart'),
      
      getCartTotal: () => {
        const cart = cartManager.getCart();
        return cart.reduce((sum, item) => sum + item.price, 0);
      }
    };
    
    // Tab-specific state (not shared between tabs)
    sessionStorage.setItem('tabId', Math.random().toString(36));
    
    // Handling storage quota exceeded
    function safeSetItem(key, value) {
      try {
        sessionStorage.setItem(key, value);
        return true;
      } catch (e) {
        if (e.name === 'QuotaExceededError') {
          console.error('Session storage is full');
          // Clear old data or notify user
          return false;
        }
        throw e;
      }
    }
    
    // Check if session storage is available
    function isSessionStorageAvailable() {
      try {
        const test = '__test__';
        sessionStorage.setItem(test, test);
        sessionStorage.removeItem(test);
        return true;
      } catch (e) {
        return false;
      }
    }
    
    // Storing arrays directly (gets stringified)
    const items = ['apple', 'banana', 'orange'];
    sessionStorage.setItem('items', JSON.stringify(items));
    const retrieved = JSON.parse(sessionStorage.getItem('items'));
    console.log(Array.isArray(retrieved)); // true
    
    // Clear all session data
    // sessionStorage.clear(); // Removes everything for this origin
    
    // Listen for changes (only fires in OTHER tabs/windows)
    window.addEventListener('storage', (e) => {
      console.log('Storage changed:', e.key, e.newValue);
    });
  </script>
  
  <h3>Session Storage Demo</h3>

    Save Click
  </button>

    Check Click
  </button>
  <button onclick="sessionStorage.clear(); alert('Cleared!')">
    Clear All
  </button>
</body>
</html>
```
**Why**: Stores session-scoped data that survives page refreshes but clears when tab closes. Perfect for temporary form data, wizard state, session tokens, and tab-specific settings. Values are always strings—use JSON.stringify/parse for objects.
