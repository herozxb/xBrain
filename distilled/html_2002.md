## HTML Input Tel
```html
<!-- Telephone input with validation -->
<label for="phone">Phone Number:</label>
<input type="tel" id="phone" name="phone" 
       pattern="[0-9]{3}-[0-9]{3}-[0-9]{4}" 
       placeholder="123-456-7890"
       required>

<script>
// # PROBLEM: Validate phone number format before submission
// # TIME: O(n) SPACE: O(1) where n = input length
function validatePhone(input) {
  const phonePattern = /^[0-9]{3}-[0-9]{3}-[0-9]{4}$/;
  const isValid = phonePattern.test(input.value);
  
  // Visual feedback
  input.style.borderColor = isValid ? 'green' : 'red';
  input.setCustomValidity(isValid ? '' : 'Please enter format: 123-456-7890');
  
  return isValid;
}

// Attach real-time validation
document.getElementById('phone').addEventListener('input', function() {
  validatePhone(this);
});
</script>
```
**Why**: Telephone inputs provide mobile keyboards optimized for phone entry and pattern validation ensures consistent formatting.

```html
<script>
// # PROBLEM: Dynamically adjust column widths
// # TIME: O(n) SPACE: O(1) where n = number of columns
function setColumnWidths(tableId, widths) {
  const table = document.getElementById(tableId);
  let colgroup = table.querySelector('colgroup');
  
  // Create colgroup if not exists
  if (!colgroup) {
    colgroup = document.createElement('colgroup');
    table.insertBefore(colgroup, table.firstChild);
  }
  
  // Clear existing cols
  colgroup.innerHTML = '';
  
  // Create col elements with specified widths
  widths.forEach(width => {
    const col = document.createElement('col');
    col.style.width = width;
    colgroup.appendChild(col);
  });
}

// Usage: setColumnWidths('myTable', ['25%', '50%', '25%']);
</script>
```
**Why**: Col elements efficiently style entire columns without repeating classes on every cell, improving maintainability.

## HTML Image Picture
```html
<!-- Responsive images with picture element -->
<picture>
  <!-- WebP for modern browsers -->
  <source srcset="image-large.webp" 
          type="image/webp"
          media="(min-width: 800px)">
  <source srcset="image-small.webp" 
          type="image/webp">
  
  <!-- Fallback formats -->
  <source srcset="image-large.jpg" 
          media="(min-width: 800px)">
  <img src="image-small.jpg" 
       alt="Responsive image"
       loading="lazy"
       width="800" height="600">
</picture>

<script>
// # PROBLEM: Detect optimal image format and size for current viewport
// # TIME: O(1) SPACE: O(1)
function getOptimalImageSrc(pictureElement) {
  const sources = pictureElement.querySelectorAll('source');
  const img = pictureElement.querySelector('img');
  
  // Check each source against media query and format support
  for (const source of sources) {
    const media = source.getAttribute('media');
    const type = source.getAttribute('type');
    
    // Check media query match
    const mediaMatches = !media || window.matchMedia(media).matches;
    
    // Check format support (WebP, AVIF, etc.)
    const formatSupported = !type || canDisplayImageType(type);
    
    if (mediaMatches && formatSupported) {
      return source.srcset;
    }
  }
  
  // Fallback to img src
  return img.src;
}

// Helper: Check if browser supports image format
function canDisplayImageType(mimeType) {
  const canvas = document.createElement('canvas');
  canvas.width = 1;
  canvas.height = 1;
  return canvas.toDataURL(mimeType).startsWith('data:' + mimeType);
}
</script>
```
**Why**: Picture element enables art direction (different images per breakpoint) and format selection (WebP for modern browsers, JPEG fallback).

```html
<script>
// # PROBLEM: Calculate total element size including padding
// # TIME: O(1) SPACE: O(1)
function getFullElementSize(element) {
  const styles = window.getComputedStyle(element);
  
  // Parse padding values (convert '20px' to 20)
  const parsePadding = (value) => parseInt(value, 10) || 0;
  
  const paddingTop = parsePadding(styles.paddingTop);
  const paddingBottom = parsePadding(styles.paddingBottom);
  const paddingLeft = parsePadding(styles.paddingLeft);
  const paddingRight = parsePadding(styles.paddingRight);
  
  // Get content dimensions
  const rect = element.getBoundingClientRect();
  
  return {
    contentWidth: rect.width - paddingLeft - paddingRight,
    contentHeight: rect.height - paddingTop - paddingBottom,
    totalWidth: rect.width,
    totalHeight: rect.height,
    horizontalPadding: paddingLeft + paddingRight,
    verticalPadding: paddingTop + paddingBottom
  };
}
</script>
```
**Why**: Padding creates internal spacing within elements, separating content from borders. Understanding it is crucial for accurate layout calculations.

## JavaScript This Keyword
```html
<button id="btn1">Click Me (Object)</button>
<button id="btn2">Click Me (Bound)</button>
<button id="btn3">Click Me (Arrow)</button>

<script>
// # PROBLEM: Understand 'this' binding in different contexts
// # TIME: O(1) SPACE: O(1)

// 1. Object method - 'this' refers to the object
const handler = {
  name: 'Handler Object',
  count: 0,
  handleClick: function() {
    this.count++;
    console.log(`${this.name}: clicked ${this.count} times`);
  }
};

document.getElementById('btn1').addEventListener('click', handler.handleClick);
// Problem: 'this' becomes the button, not handler!

// 2. Solution A: Bind the context
const boundHandler = handler.handleClick.bind(handler);
document.getElementById('btn2').addEventListener('click', boundHandler);

// 3. Solution B: Arrow function (lexically binds 'this')
const arrowHandler = {
  name: 'Arrow Handler',
  count: 0,
  handleClick: function() {
    // Arrow function inherits 'this' from enclosing scope
    const inner = () => {
      this.count++;
      console.log(`${this.name}: clicked ${this.count} times`);
    };
    return inner;
  }
};

document.getElementById('btn3').addEventListener('click', arrowHandler.handleClick());

// 4. Explicit binding with call/apply/bind
function greet(greeting, punctuation) {
  console.log(`${greeting}, ${this.name}${punctuation}`);
}

const person = { name: 'Alice' };

greet.call(person, 'Hello', '!');       // Invoke immediately with this
greet.apply(person, ['Hi', '?']);       // Same, but args as array
const boundGreet = greet.bind(person);  // Returns new function
boundGreet('Hey', '.');
</script>
```
**Why**: Understanding 'this' is fundamental to JavaScript. Its value depends on call-site context, not definition location. Mastering bind/call/apply is essential for event handlers and callbacks.

## DOM Insert Before
```html
<ul id="list">
  <li>First item</li>
  <li>Last item</li>
</ul>

<button id="addBtn">Add Item Before Second</button>

<script>
// # PROBLEM: Insert element at specific position in DOM
// # TIME: O(1) SPACE: O(1) - DOM operations are constant time

function insertBefore(parent, newElement, referenceElement) {
  // insertBefore(newNode, referenceNode)
  // If referenceNode is null, inserts at end (like appendChild)
  return parent.insertBefore(newElement, referenceElement);
}

// Create and insert a new list item
document.getElementById('addBtn').addEventListener('click', function() {
  const list = document.getElementById('list');
  const items = list.getElementsByTagName('li');
  
  // Create new element
  const newItem = document.createElement('li');
  newItem.textContent = `Inserted item ${Date.now()}`;
  newItem.style.backgroundColor = '#e0f7fa';
  
  // Insert before the second item (index 1)
  if (items.length >= 2) {
    insertBefore(list, newItem, items[1]);
  } else {
    // If less than 2 items, append to end
    list.appendChild(newItem);
  }
});

// Utility: Insert at specific index
function insertAtIndex(parent, element, index) {
  const children = Array.from(parent.children);
  
  if (index >= children.length) {
    parent.appendChild(element);
  } else {
    parent.insertBefore(element, children[index]);
  }
}
</script>
```
**Why**: insertBefore is the fundamental DOM insertion method. It's the only way to insert at a specific position; appendChild only adds to the end.

## Event Remove Event Listener
```html
<button id="toggleBtn">Enable Click Counter</button>
<span id="counter">0</span>
<button id="clickBtn">Click to Count</button>

<script>
// # PROBLEM: Properly remove event listeners to prevent memory leaks
// # TIME: O(1) SPACE: O(1)

// WRONG: Anonymous functions cannot be removed!
// element.addEventListener('click', () => console.log('click'));
// Cannot remove this later!

// RIGHT: Named function reference for removal
let clickCount = 0;
const counterSpan = document.getElementById('counter');
const clickBtn = document.getElementById('clickBtn');
const toggleBtn = document.getElementById('toggleBtn');

// Named function (required for removal)
function handleClick() {
  clickCount++;
  counterSpan.textContent = clickCount;
}

// Track listener state
let isListening = false;

toggleBtn.addEventListener('click', function() {
  if (isListening) {
    // Remove: must pass same function reference
    clickBtn.removeEventListener('click', handleClick);
    toggleBtn.textContent = 'Enable Click Counter';
    clickBtn.style.opacity = '0.5';
  } else {
    // Add listener
    clickBtn.addEventListener('click', handleClick);
    toggleBtn.textContent = 'Disable Click Counter';
    clickBtn.style.opacity = '1';
  }
  isListening = !isListening;
});

// Alternative: One-time listener (auto-removes after first call)
function once(element, event, callback) {
  element.addEventListener(event, function handler(e) {
    element.removeEventListener(event, handler);
    callback(e);
  });
}

// Usage: once(document.getElementById('onceBtn'), 'click', () => alert('Only once!'));
</script>
```
**Why**: Properly removing event listeners prevents memory leaks and unwanted behavior. Always use named functions when you need to remove listeners later.

## HTML Input URL
```html
<form id="urlForm">
  <label for="website">Website URL:</label>
  <input type="url" 
         id="website" 
         name="website"
         placeholder="https://example.com"
         pattern="https?://.+"
         required>
  <button type="submit">Submit</button>
  <span id="urlError" style="color: red;"></span>
</form>

<script>
// # PROBLEM: Validate and normalize URL input
// # TIME: O(n) SPACE: O(n) where n = URL length

const urlInput = document.getElementById('website');
const urlError = document.getElementById('urlError');

function validateURL(value) {
  try {
    // Browser's built-in URL parsing
    const url = new URL(value);
    
    // Additional validation
    const validProtocols = ['http:', 'https:'];
    if (!validProtocols.includes(url.protocol)) {
      return { valid: false, error: 'Only http and https protocols allowed' };
    }
    
    return { 
      valid: true, 
      url: url,
      normalized: url.href 
    };
  } catch (e) {
    return { valid: false, error: 'Invalid URL format' };
  }
}

// Real-time validation
urlInput.addEventListener('input', function() {
  const result = validateURL(this.value);
  
  if (this.value === '') {
    urlError.textContent = '';
    this.setCustomValidity('');
  } else if (!result.valid) {
    urlError.textContent = result.error;
    this.setCustomValidity(result.error);
  } else {
    urlError.textContent = '✓ Valid: ' + result.url.hostname;
    this.setCustomValidity('');
  }
});

// Auto-add protocol if missing
urlInput.addEventListener('blur', function() {
  let value = this.value.trim();
  if (value && !value.match(/^https?:\/\//)) {
    this.value = 'https://' + value;
    this.dispatchEvent(new Event('input'));
  }
});

document.getElementById('urlForm').addEventListener('submit', function(e) {
  e.preventDefault();
  const result = validateURL(urlInput.value);
  if (result.valid) {
    console.log('Submitting:', result.normalized);
    alert('Valid URL: ' + result.normalized);
  }
});
</script>
```
**Why**: URL inputs trigger mobile-optimized keyboards and provide built-in validation. The URL API offers robust parsing and manipulation capabilities.

```html
<script>
// # PROBLEM: Toggle element visibility while preserving layout space
// # TIME: O(1) SPACE: O(1)

function toggleDisplay(element, show) {
  if (show) {
    element.style.display = '';  // Revert to CSS default
  } else {
    element.style.display = 'none';  // Remove from document flow
  }
}

function toggleVisibility(element, visible) {
  // visibility: hidden hides but preserves space
  // visibility: visible shows element
  element.style.visibility = visible ? 'visible' : 'hidden';
}

function toggleOpacity(element, visible) {
  // opacity: 0 is invisible but still interactive
  // opacity: 1 is fully visible
  element.style.opacity = visible ? '1' : '0';
}

// Detect computed display type
function getDisplayType(element) {
  return window.getComputedStyle(element).display;
}

// Toggle between display values
function cycleDisplay(element, displays = ['block', 'inline', 'none']) {
  const current = getDisplayType(element);
  const idx = displays.indexOf(current);
  const nextIdx = (idx + 1) % displays.length;
  element.style.display = displays[nextIdx];
  return displays[nextIdx];
}
</script>
```
**Why**: The display property controls element rendering behavior. Understanding block/inline/inline-block/flex/grid is fundamental to CSS layout.

## JavaScript Prototypes
```html
<div id="output"></div>

<script>
// # PROBLEM: Implement inheritance and shared methods via prototype chain
// # TIME: O(1) for property lookup SPACE: O(n) for prototype chain depth

const output = document.getElementById('output');
function log(msg) {
  output.innerHTML += msg + '<br>';
}

// Constructor function (ES5 style)
function Animal(name, age) {
  this.name = name;
  this.age = age;
}

// Add method to prototype (shared by all instances)
Animal.prototype.speak = function() {
  return `${this.name} makes a sound`;
};

Animal.prototype.info = function() {
  return `${this.name} is ${this.age} years old`;
};

// Inheritance: Dog extends Animal
function Dog(name, age, breed) {
  Animal.call(this, name, age);  // Call parent constructor
  this.breed = breed;
}

// Set up prototype chain
Dog.prototype = Object.create(Animal.prototype);
Dog.prototype.constructor = Dog;  // Fix constructor reference

// Override parent method
Dog.prototype.speak = function() {
  return `${this.name} barks!`;
};

// Add new method
Dog.prototype.fetch = function() {
  return `${this.name} fetches the ball`;
};

// Usage
const genericAnimal = new Animal('Creature', 5);
const myDog = new Dog('Buddy', 3, 'Golden Retriever');

log(genericAnimal.speak());  // "Creature makes a sound"
log(myDog.speak());          // "Buddy barks!"
log(myDog.info());           // "Buddy is 3 years old"
log(myDog.fetch());          // "Buddy fetches the ball"

// ES6 Class syntax (syntactic sugar over prototypes)
class Cat extends Animal {
  constructor(name, age, color) {
    super(name, age);
    this.color = color;
  }
  
  speak() {
    return `${this.name} meows!`;
  }
}

const myCat = new Cat('Whiskers', 2, 'orange');
log(myCat.speak());  // "Whiskers meows!"
log(Object.getPrototypeOf(myCat) === Cat.prototype);  // true

// Prototype inspection
log(`Dog prototype has speak: ${Dog.prototype.hasOwnProperty('speak')}`);
log(`myDog's prototype chain includes Animal: ${myDog instanceof Animal}`);
</script>
```
**Why**: Prototypes are JavaScript's inheritance mechanism. Every object has a prototype, creating a chain for property lookup. Classes are syntactic sugar over this system.
