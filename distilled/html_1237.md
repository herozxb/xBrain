            if (/[a-z]/.test(password)) strength++;
            if (/[A-Z]/.test(password)) strength++;

            if (/[^a-zA-Z0-9]/.test(password)) strength++;

                <td>John Doe</td>

            </tr>
            <tr>
                <td>Jane Smith</td>
                <td>34</td>
                <td>$65,000</td>
            </tr>
            <tr>
                <td>Bob Johnson</td>
                <td>25</td>
                <td>$45,000</td>
            </tr>
        </tbody>
    </table>
    
    <script>
        // Parse currency string to number
        function parseCurrency(value) {
            return parseFloat(value.replace(/[^0-9.-]+/g, ''));
        }
        
        // Sort table by column
        function sortTable(tableId, column, type = 'string') {
            const table = document.getElementById(tableId);
            const tbody = table.querySelector('tbody');
            const rows = Array.from(tbody.querySelectorAll('tr'));
            const header = table.querySelector(`th[data-column="${column}"]`);
            
            // Toggle sort direction
            const isAscending = header.classList.contains('sorted-asc');
            
            // Clear all sort indicators
            table.querySelectorAll('th').forEach(th => {
                th.classList.remove('sorted-asc', 'sorted-desc');
            });
            
            // Set new sort indicator
            header.classList.add(isAscending ? 'sorted-desc' : 'sorted-asc');
            
            // Sort rows
            rows.sort((a, b) => {
                const aCell = a.querySelector(`td:nth-child(${getColumnIndex(table, column)})`);
                const bCell = b.querySelector(`td:nth-child(${getColumnIndex(table, column)})`);
                
                let aValue = aCell.textContent.trim();
                let bValue = bCell.textContent.trim();
                
                // Type conversion
                if (type === 'number') {
                    aValue = parseFloat(aValue);
                    bValue = parseFloat(bValue);
                } else if (type === 'currency') {
                    aValue = parseCurrency(aValue);
                    bValue = parseCurrency(bValue);
                }
                
                if (aValue < bValue) return isAscending ? 1 : -1;
                if (aValue > bValue) return isAscending ? -1 : 1;
                return 0;
            });
            
            // Re-append sorted rows
            rows.forEach(row => tbody.appendChild(row));
        }
        
        // Get column index from column name
        function getColumnIndex(table, column) {
            const headers = table.querySelectorAll('th');
            for (let i = 0; i < headers.length; i++) {
                if (headers[i].dataset.column === column) {
                    return i + 1;
                }
            }
            return 1;
        }
        
        // Add click handlers to headers
        document.querySelectorAll('#sortableTable th').forEach(th => {
            th.addEventListener('click', function() {
                const column = this.dataset.column;
                const type = column === 'age' ? 'number' : 
                             column === 'salary' ? 'currency' : 'string';
                sortTable('sortableTable', column, type);
            });
        });
    </script>
</body>
</html>
```

**Why**: Client-side table sorting enhances data exploration without server round-trips. Efficient for small-to-medium datasets. Provides instant feedback and reduces server load.

        // Automatically add noopener to target="_blank" links
        function secureExternalLinks() {
            const links = document.querySelectorAll('a[target="_blank"]');
            
            links.forEach(link => {
                const rel = link.getAttribute('rel') || '';
                
                if (!rel.includes('noopener')) {
                    link.setAttribute('rel', rel + ' noopener').trim();
                }
                
                // Log for demonstration
                console.log('Secured link:', link.href, '| rel:', link.rel);

        function validateRelAttributes() {
            const externalLinks = document.querySelectorAll('a[target="_blank"]');
            const issues = [];
            
            externalLinks.forEach((link, index) => {

                if (!rel.includes('noopener')) {
                    issues.push(`Link ${index + 1}: Missing noopener (security risk)`);
                }
            });
            
            return issues.length ? issues : ['✅ All external links properly secured'];
        }
        
        // Run security enhancement
        secureExternalLinks();
        console.log('Validation results:', validateRelAttributes());
    </script>

                            retries++;
                            if (retries < maxRetries) {
                                log(`Retry ${retries}/${maxRetries}...`);
                                attempt();
                            } else {
                                reject(error);
                            }
                        });
                }
                
                attempt();
            });

        // BAD APPROACH: Direct DOM manipulation (causes multiple reflows)
        function addItemsWithoutFragment(count) {
            clearList();
            const list = document.getElementById('itemList');
            const startTime = performance.now();
            
            // Each appendChild triggers a reflow!
            for (let i = 0; i < count; i++) {
                const div = document.createElement('div');
                div.textContent = `Item ${i + 1}`;
                list.appendChild(div); // Triggers reflow
            }
            
            const endTime = performance.now();
            displayPerformance('Without Fragment', endTime - startTime, count);
        }
        
        // GOOD APPROACH: Using DocumentFragment (single reflow)
        function addItemsWithFragment(count) {
            clearList();
            const list = document.getElementById('itemList');
            const startTime = performance.now();
            
            // Create fragment to hold all elements
            const fragment = document.createDocumentFragment();
            
            for (let i = 0; i < count; i++) {
                const div = document.createElement('div');
                div.textContent = `Item ${i + 1}`;
                fragment.appendChild(div); // No reflow!
            }
            
            // Single DOM update
            list.appendChild(fragment); // Only ONE reflow
            
            const endTime = performance.now();
            displayPerformance('With Fragment', endTime - startTime, count);
        }

                Time: ${timeMs.toFixed(2)}ms<br>
                Items: ${count}<br>
                Speed: ${opsPerMs} items/ms
            `;
            
            console.log(`${method}: ${timeMs.toFixed(2)}ms for ${count} items`);

            document.getElementById('itemList').innerHTML = '';

        function createFragmentFromHTML(html) {
            const template = document.createElement('template');
            template.innerHTML = html.trim();
            return template.content;

        // Batch update with fragment
        function batchUpdate(elementId, items, createItemFn) {
            const container = document.getElementById(elementId);
            const fragment = document.createDocumentFragment();
            
            items.forEach((item, index) => {
                const element = createItemFn(item, index);
                fragment.appendChild(element);
            });
            
            container.innerHTML = '';
            container.appendChild(fragment);
        }
        
        // Example: Build table with fragment
        function buildTableWithFragment(data) {
            const fragment = document.createDocumentFragment();
            
            data.forEach(row => {
                const tr = document.createElement('tr');
                
                Object.values(row).forEach(value => {
                    const td = document.createElement('td');
                    td.textContent = value;
                    tr.appendChild(td);
                });
                
                fragment.appendChild(tr);
            });
            
            return fragment;
        }
        
        console.log('DocumentFragment reduces reflows from O(n) to O(1)');
    </script>

</html>
```

**Why**: DocumentFragment batches DOM updates into a single operation, reducing reflows from O(n) to O(1). Critical for performance when adding many elements. Browser can optimize rendering without intermediate layouts.

        function throttle(func, limit) {
            let inThrottle;
            let lastFunc;
            let lastRan;
            
            return function(...args) {
                const context = this;
                
                if (!inThrottle) {
                    func.apply(context, args);
                    lastRan = Date.now();
                    inThrottle = true;
                } else {
                    clearTimeout(lastFunc);
                    lastFunc = setTimeout(() => {
                        if ((Date.now() - lastRan) >= limit) {
                            func.apply(context, args);
                            lastRan = Date.now();
                        }
                    }, limit - (Date.now() - lastRan));
                }
            };
