## Shell Sort
```python
# PROBLEM: Sort an array using Shell Sort with gap sequence optimization
# TIME: O(n log²n) average, O(n²) worst SPACE: O(1)
def shell_sort(arr):
    """
    Shell Sort - an optimized insertion sort using gap sequences.
    Uses Knuth's sequence: h = 3h + 1, starting from 1.
    Elements are h-sorted, progressively reducing gap until 1.
    """
    n = len(arr)
    
    # Generate Knuth's gap sequence (find largest gap < n)
    gap = 1
    while gap < n // 3:
        gap = 3 * gap + 1  # 1, 4, 13, 40, 121, ...
    
    # Reduce gap progressively
    while gap >= 1:
        # Perform insertion sort for this gap size
        for i in range(gap, n):
            temp = arr[i]
            j = i
            # Shift earlier gap-sorted elements until correct position
            while j >= gap and arr[j - gap] > temp:
                arr[j] = arr[j - gap]
                j -= gap
            arr[j] = temp
        gap //= 3  # Move to next smaller gap
    
    return arr
```
**Why**: Bridges insertion sort and merge sort efficiency; in-place, adaptive, excellent for medium-sized arrays and partially sorted data.

## Counting Sort
```python
# PROBLEM: Sort integers in a known range using frequency counting
# TIME: O(n + k) where k is range SPACE: O(n + k)
def counting_sort(arr, min_val=None, max_val=None):
    """
    Counting Sort - non-comparative sorting using element frequencies.
    Best for integers in a limited range; stable sort.
    """
    if not arr:
        return []
    
    # Determine range if not provided
    if min_val is None:
        min_val = min(arr)
    if max_val is None:
        max_val = max(arr)
    
    range_size = max_val - min_val + 1
    
    # Count occurrences of each value
    count = [0] * range_size
    for num in arr:
        count[num - min_val] += 1
    
    # Convert counts to cumulative positions (for stability)
    for i in range(1, range_size):
        count[i] += count[i - 1]
    
    # Build output array (iterate backwards for stability)
    output = [0] * len(arr)
    for num in reversed(arr):
        count[num - min_val] -= 1
        output[count[num - min_val]] = num
    
    return output
```
**Why**: Linear time for bounded integers; foundation for radix sort; stable sort essential for satellite data.

## Bucket Sort
```python
# PROBLEM: Sort floating-point numbers uniformly distributed in [0, 1)
# TIME: O(n + k) average, O(n²) worst SPACE: O(n + k)
def bucket_sort(arr, bucket_count=None):
    """
    Bucket Sort - distribute elements into buckets, sort each bucket.
    Optimal for uniformly distributed data (e.g., floating points).
    """
    if not arr:
        return []
    
    n = len(arr)
    if bucket_count is None:
        bucket_count = n
    
    # Create empty buckets
    buckets = [[] for _ in range(bucket_count)]
    
    # Distribute elements into buckets based on value
    for num in arr:
        # Map value to bucket index (assumes range [0, 1) or normalized)
        index = int(num * bucket_count)
        # Handle edge case where num == 1.0
        index = min(index, bucket_count - 1)
        buckets[index].append(num)
    
    # Sort individual buckets (insertion sort for small buckets)
    for bucket in buckets:
        bucket.sort()  # Or use insertion_sort for O(n²/2) optimization
    
    # Concatenate all buckets
    result = []
    for bucket in buckets:
        result.extend(bucket)
    
    return result
```
**Why**: Achieves linear time for uniform distributions; easily parallelizable; used in histogram-based algorithms.

## Prim's Algorithm
```python
# PROBLEM: Find Minimum Spanning Tree (MST) of a weighted undirected graph
# TIME: O(E log V) with binary heap SPACE: O(V + E)
import heapq

def prim_mst(graph, start=0):
    """
    Prim's Algorithm - grow MST by adding minimum edge to visited set.
    Uses min-heap for efficient edge selection.
    
    Args:
        graph: adjacency list {node: [(neighbor, weight), ...]}
        start: starting node
    
    Returns:
        mst_edges: list of (u, v, weight) in MST
        total_weight: sum of MST edge weights
    """
    visited = set()
    mst_edges = []
    total_weight = 0
    
    # Min-heap: (weight, parent, child)
    heap = [(0, -1, start)]  # -1 indicates no parent for start
    
    while heap and len(visited) < len(graph):
        weight, parent, node = heapq.heappop(heap)
        
        if node in visited:
            continue
        
        visited.add(node)
        
        # Add edge to MST (skip for start node with no parent)
        if parent != -1:
            mst_edges.append((parent, node, weight))
            total_weight += weight
        
        # Add all edges from current node to unvisited neighbors
        for neighbor, edge_weight in graph[node]:
            if neighbor not in visited:
                heapq.heappush(heap, (edge_weight, node, neighbor))
    
    return mst_edges, total_weight
```
**Why**: Greedy MST construction; efficient for dense graphs; foundation for network design algorithms.

## Tarjan's SCC Algorithm
```python
# PROBLEM: Find all strongly connected components in a directed graph
# TIME: O(V + E) SPACE: O(V)
def tarjan_scc(graph):
    """
    Tarjan's Algorithm - find SCCs using DFS and low-link values.
    A strongly connected component has paths between all pairs of nodes.
    Single pass algorithm using index tracking and stack.
    """
    index_counter = [0]  # Mutable counter for closure
    stack = []
    on_stack = set()
    indices = {}  # Discovery time for each node
    low_link = {}  # Lowest index reachable from subtree
    sccs = []  # List of SCCs
    
    def strongconnect(node):
        # Set depth index and low-link value
        indices[node] = index_counter[0]
        low_link[node] = index_counter[0]
        index_counter[0] += 1
        
        stack.append(node)
        on_stack.add(node)
        
        # Consider successors
        for successor in graph.get(node, []):
            if successor not in indices:
                # Successor not visited; recurse
                strongconnect(successor)
                low_link[node] = min(low_link[node], low_link[successor])
            elif successor in on_stack:
                # Successor is on stack, hence in current SCC
                low_link[node] = min(low_link[node], indices[successor])
        
        # If node is root of SCC, pop stack until node
        if low_link[node] == indices[node]:
            scc = []
            while True:
                w = stack.pop()
                on_stack.remove(w)
                scc.append(w)
                if w == node:
                    break
            sccs.append(scc)
    
    # Process all nodes
    for node in graph:
        if node not in indices:
            strongconnect(node)
    
    return sccs
```
**Why**: Single-pass SCC detection; key for dependency analysis, cycle detection, and graph simplification.

## Convex Hull - Graham Scan
```python
# PROBLEM: Find convex hull of 2D points (smallest convex polygon containing all points)
# TIME: O(n log n) SPACE: O(n)
def convex_hull_graham(points):
    """
    Graham Scan - compute convex hull using angular sorting and stack.
    Returns vertices of convex hull in counter-clockwise order.
    """
    if len(points) < 3:
        return points
    
    # Find lowest point (and leftmost if tie)
    start = min(points, key=lambda p: (p[1], p[0]))
    
    def polar_angle(p):
        """Calculate polar angle from start point."""
        dx = p[0] - start[0]
        dy = p[1] - start[1]
        return (dx, dy)  # Compare by angle using cross product
    
    def cross(o, a, b):
        """Cross product of vectors OA and OB."""
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
    
    # Sort by polar angle (and distance for collinear points)
    def sort_key(p):
        dx, dy = polar_angle(p)
        dist_sq = dx * dx + dy * dy
        # Sort by angle (using atan2 approximation), then distance
        import math
        return (math.atan2(dy, dx), dist_sq)
    
    sorted_pts = sorted([p for p in points if p != start], key=sort_key)
    sorted_pts = [start] + sorted_pts
    
    # Build hull using stack
    hull = []
    for p in sorted_pts:
        # Pop while turn is not counter-clockwise
        while len(hull) >= 2 and cross(hull[-2], hull[-1], p) <= 0:
            hull.pop()
        hull.append(p)
    
    return hull
```
**Why**: O(n log n) convex hull; fundamental in computational geometry for collision detection, shape analysis.

## Rabin-Karp Algorithm
```python
# PROBLEM: Find pattern occurrences in text using rolling hash
# TIME: O(n + m) average, O(nm) worst SPACE: O(1)
def rabin_karp(text, pattern, base=256, prime=101):
    """
    Rabin-Karp - pattern matching using polynomial rolling hash.
    Hash slides through text, comparing only when hash matches.
    """
    n, m = len(text), len(pattern)
    if m > n:
        return []
    
    # Precompute base^(m-1) % prime for rolling hash
    h = pow(base, m - 1, prime)
    
    # Compute initial hash for pattern and first window
    pattern_hash = 0
    window_hash = 0
    
    for i in range(m):
        pattern_hash = (base * pattern_hash + ord(pattern[i])) % prime
        window_hash = (base * window_hash + ord(text[i])) % prime
    
    matches = []
    
    # Slide pattern over text
    for i in range(n - m + 1):
        # Check hash match and verify with actual comparison
        if pattern_hash == window_hash:
            if text[i:i + m] == pattern:
                matches.append(i)
        
        # Compute hash for next window (rolling)
        if i < n - m:
            window_hash = (base * (window_hash - ord(text[i]) * h) + 
                          ord(text[i + m])) % prime
            
            # Handle negative hash
            if window_hash < 0:
                window_hash += prime
    
    return matches
```
**Why**: Efficient for multiple pattern matching; foundation for plagiarism detection and DNA sequence analysis.

## Miller-Rabin Primality Test
```python
# PROBLEM: Probabilistically test if a number is prime
# TIME: O(k log³n) for k rounds SPACE: O(1)
import random

def miller_rabin(n, k=5):
    """
    Miller-Rabin - probabilistic primality test.
    Returns True if n is probably prime, False if definitely composite.
    Error probability < 4^(-k).
    """
    # Handle small cases
    if n < 2:
        return False
    if n == 2 or n == 3:
        return True
    if n % 2 == 0:
        return False
    
    # Write n-1 as 2^r * d where d is odd
    r, d = 0, n - 1
    while d % 2 == 0:
        r += 1
        d //= 2
    
    # Witness loop
    def check_witness(a):
        """Check if 'a' is a witness to compositeness."""
        x = pow(a, d, n)
        
        if x == 1 or x == n - 1:
            return False  # Probably prime
        
        # Square r-1 times
        for _ in range(r - 1):
            x = pow(x, 2, n)
            if x == n - 1:
                return False  # Probably prime
        
        return True  # Composite
    
    # Test k random witnesses
    for _ in range(k):
        a = random.randrange(2, n - 1)
        if check_witness(a):
            return False
    
    return True
```
**Why**: Fast probabilistic primality; essential for RSA key generation and cryptographic applications.

## Matrix Exponentiation

```python
# PROBLEM: Compute matrix power efficiently using binary exponentiation
# TIME: O(log n * k³) for k×k matrix SPACE: O(k²)
def matrix_multiply(A, B, mod=None):
    """Multiply two matrices (optionally with modulo)."""
    n = len(A)
    m = len(B[0])
    k = len(B)
    result = [[0] * m for _ in range(n)]
    
    for i in range(n):
        for j in range(m):
            for p in range(k):
                result[i][j] += A[i][p] * B[p][j]
                if mod:
                    result[i][j] %= mod
    
    return result

def matrix_power(M, power, mod=None):
    """
    Matrix Exponentiation - compute M^power using binary exponentiation.
    Used for linear recurrence relations (e.g., Fibonacci in O(log n)).
    """
    n = len(M)
    
    # Start with identity matrix
    result = [[1 if i == j else 0 for j in range(n)] for i in range(n)]
    
    base = [row[:] for row in M]  # Deep copy
    
    while power > 0:
        if power % 2 == 1:
            result = matrix_multiply(result, base, mod)
        base = matrix_multiply(base, base, mod)
        power //= 2
    
    return result

def fibonacci_fast(n, mod=10**9 + 7):
    """
    Compute nth Fibonacci number in O(log n) using matrix exponentiation.
    [F(n+1), F(n)] = [[1,1],[1,0]]^n * [F(1), F(0)]
    """
    if n <= 1:
        return n
    
    # Transformation matrix for Fibonacci
    F = [[1, 1], [1, 0]]
    result = matrix_power(F, n, mod)
    return result[0][1]
```
**Why**: O(log n) for linear recurrences; powers dynamic programming problems from O(n) to O(log n).

## Egg Dropping Problem
```python
# PROBLEM: Find minimum trials to find critical floor with k eggs and n floors
# TIME: O(k * n²) or O(k * n * log n) with optimization SPACE: O(k * n)
def egg_drop_dp(k, n):
    """
    Classic DP: minimum egg drops to guarantee finding critical floor.
    dp[eggs][floors] = min trials for eggs and floors.
    """
    # dp[i][j] = min trials with i eggs and j floors
    dp = [[0] * (n + 1) for _ in range(k + 1)]
    
    # Base cases
    for i in range(1, k + 1):
        dp[i][1] = 1  # 1 floor needs 1 trial
        dp[i][0] = 0  # 0 floors need 0 trials
    
    for j in range(1, n + 1):
        dp[1][j] = j  # 1 egg needs j trials (linear search)
    
    # Fill DP table
    for i in range(2, k + 1):  # 2+ eggs
        for j in range(2, n + 1):  # 2+ floors
            dp[i][j] = float('inf')
            for x in range(1, j + 1):  # Try dropping from each floor
                # Max of: egg breaks (i-1 eggs, x-1 floors below)
                #         egg survives (i eggs, j-x floors above)
                # Plus 1 for current drop
                worst_case = max(dp[i - 1][x - 1], dp[i][j - x]) + 1
                dp[i][j] = min(dp[i][j], worst_case)
    
    return dp[k][n]

def egg_drop_optimized(k, n):
    """
    Optimized with binary search for floor selection.
    TIME: O(k * n * log n)
    """
    dp = [[0] * (n + 1) for _ in range(k + 1)]
    
    for i in range(1, k + 1):
        dp[i][1] = 1
    
    for j in range(1, n + 1):
        dp[1][j] = j
    
    for i in range(2, k + 1):
        for j in range(2, n + 1):
            dp[i][j] = float('inf')
            # Binary search for optimal floor x
            low, high = 1, j
            while low <= high:
                mid = (low + high) // 2
                break_case = dp[i - 1][mid - 1]  # Egg breaks
                survive_case = dp[i][j - mid]     # Egg survives
                
                if break_case > survive_case:
                    high = mid - 1
                    worst_case = break_case
                else:
                    low = mid + 1
                    worst_case = survive_case
                
                dp[i][j] = min(dp[i][j], worst_case + 1)
    
    return dp[k][n]
```
**Why**: Classic optimization DP; teaches state space design and optimization with binary search.

## Burst Balloons
```python
# PROBLEM: Maximize coins by bursting balloons optimally
# TIME: O(n³) SPACE: O(n²)
def max_coins_burst_balloons(nums):
    """
    Burst Balloons - maximize coins when bursting balloon i gets
    nums[i-1] * nums[i] * nums[i+1] coins.
    Key insight: think in reverse - which balloon is burst LAST.
    """
    # Add virtual balloons with value 1 at boundaries
    balloons = [1] + nums + [1]
    n = len(balloons)
    
    # dp[i][j] = max coins from bursting balloons in (i, j) exclusive
    dp = [[0] * n for _ in range(n)]
    
    # Length of range to consider
    for length in range(2, n):  # Minimum gap of 2 (at least 1 balloon)
        for i in range(n - length):
            j = i + length
            
            # Try each balloon as the LAST one to burst in (i, j)
            for k in range(i + 1, j):
                # When k is burst last, it gets coins from boundaries i and j
                coins = balloons[i] * balloons[k] * balloons[j]
                # Add coins from left and right subproblems
                coins += dp[i][k] + dp[k][j]
                dp[i][j] = max(dp[i][j], coins)
    
    return dp[0][n - 1]
```
**Why**: Tricky interval DP; demonstrates reverse thinking (last instead of first) in dynamic programming.

## Sliding Window Maximum
```python
# PROBLEM: Find maximum in each sliding window of size k
# TIME: O(n) SPACE: O(k)
from collections import deque

def sliding_window_maximum(nums, k):
    """
    Use deque to maintain candidates in decreasing order.
    Front of deque always holds maximum for current window.
    """
    if not nums or k == 0:
        return []
    
    dq = deque()  # Stores indices, values in decreasing order
    result = []
    
    for i in range(len(nums)):
        # Remove indices outside current window
        while dq and dq[0] < i - k + 1:
            dq.popleft()
        
        # Remove indices of smaller elements (they'll never be max)
        while dq and nums[dq[-1]] < nums[i]:
            dq.pop()
        
        dq.append(i)
        
        # Start recording once first window is complete
        if i >= k - 1:
            result.append(nums[dq[0]])
    
    return result
```
**Why**: Monotonic deque technique; O(n) solution for window aggregate problems.

## Daily Temperatures
```python
# PROBLEM: For each day, find days until a warmer temperature
# TIME: O(n) SPACE: O(n)
def daily_temperatures(temperatures):
    """
    Monotonic stack - maintain decreasing temperature indices.
    When warmer day found, resolve all previous cooler days.
    """
    n = len(temperatures)
    answer = [0] * n
    stack = []  # Stack of indices with decreasing temperatures
    
    for i, temp in enumerate(temperatures):
        # Current temp resolves previous cooler days
        while stack and temperatures[stack[-1]] < temp:
            prev_idx = stack.pop()
            answer[prev_idx] = i - prev_idx
        
        stack.append(i)
    
    return answer
```
**Why**: Classic monotonic stack application; pattern applies to next greater/smaller element problems.

## Trap Rain Water
```python
# PROBLEM: Calculate how much rainwater can be trapped between bars
# TIME: O(n) SPACE: O(1) with two pointers
def trap_rain_water(height):
    """
    Two-pointer approach: water at position i depends on
    min(max_left, max_right) - height[i].
    Process from both ends, always moving from shorter side.
    """
    if not height:
        return 0
    
    left, right = 0, len(height) - 1
    left_max, right_max = 0, 0
    water = 0
    
    while left < right:
        # Process from shorter side
        if height[left] < height[right]:
            if height[left] >= left_max:
                left_max = height[left]
            else:
                water += left_max - height[left]
            left += 1
        else:
            if height[right] >= right_max:
                right_max = height[right]
            else:
                water += right_max - height[right]
            right -= 1
    
    return water
```
**Why**: Two-pointer technique at its finest; transforms O(n) space to O(1) while maintaining correctness.

## Combination Sum
```python
# PROBLEM: Find all unique combinations that sum to target (with repetition)
# TIME: O(N^(T/M)) where N=candidates, T=target, M=min candidate SPACE: O(T/M)
def combination_sum(candidates, target):
    """
    Backtracking with sorted candidates.
    Each number can be used unlimited times.
    Avoid duplicates by only moving forward in candidates.
    """
    result = []
    candidates.sort()  # Sort for early termination and dedup
    
    def backtrack(start, current_combination, remaining):
        if remaining == 0:
            result.append(current_combination[:])
            return
        
        if remaining < 0:
            return
        
        for i in range(start, len(candidates)):
            # Early termination if candidate exceeds remaining
            if candidates[i] > remaining:
                break
            
            # Include candidates[i] and continue from i (allows repetition)
            current_combination.append(candidates[i])
            backtrack(i, current_combination, remaining - candidates[i])
            current_combination.pop()  # Backtrack
    
    backtrack(0, [], target)
    return result

def combination_sum_ii(candidates, target):
    """
    Variation: each candidate can only be used once.
    Handle duplicates in input array.
    """
    result = []
    candidates.sort()
    
    def backtrack(start, current_combination, remaining):
        if remaining == 0:
            result.append(current_combination[:])
            return
        
        for i in range(start, len(candidates)):
            # Skip duplicates at same level
            if i > start and candidates[i] == candidates[i - 1]:
                continue
            
            if candidates[i] > remaining:
                break
            
            current_combination.append(candidates[i])
            # Move to i+1 since each element used once
            backtrack(i + 1, current_combination, remaining - candidates[i])
            current_combination.pop()
    
    backtrack(0, [], target)
    return result
```
**Why**: Classic backtracking template; demonstrates state management, pruning, and duplicate handling.
