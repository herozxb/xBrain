## Hopscotch Hashing

```python
# PROBLEM: Implement hopscotch hashing - an open addressing scheme that guarantees
# elements are found within H-1 positions of their ideal bucket
# TIME: O(1) average for insert/lookup/delete, O(n) worst case
# SPACE: O(n) for table + O(H) metadata per bucket
class HopscotchHashTable:
    def __init__(self, capacity=16, hop_range=4):
        self.capacity = capacity
        self.hop_range = hop_range  # H - maximum displacement
        self.table = [None] * capacity
        self.hop_info = [0] * capacity  # Bitmap for elements within hop_range
        self.size = 0
    
    def _hash(self, key):
        return hash(key) % self.capacity
    
    def _find_slot(self, start):
        # Find next empty slot starting from 'start'
        for i in range(self.capacity):
            idx = (start + i) % self.capacity
            if self.table[idx] is None:
                return idx
        return -1
    
    def _try_move(self, empty_idx):
        # Try to move elements to create space within hop_range
        for dist in range(self.hop_range - 1, 0, -1):
            check_idx = (empty_idx - dist + self.capacity) % self.capacity
            if self.table[check_idx] is not None:
                # Check if we can swap
                key, value = self.table[check_idx]
                ideal = self._hash(key)
                
                # Verify the element at check_idx can move to empty_idx
                if (empty_idx - ideal + self.capacity) % self.capacity < self.hop_range:
                    # Move it
                    self.table[empty_idx] = self.table[check_idx]
                    self.table[check_idx] = None
                    
                    # Update hop_info bitmaps
                    self.hop_info[ideal] |= (1 << ((check_idx - ideal + self.capacity) % self.capacity))
                    self.hop_info[ideal] &= ~(1 << ((empty_idx - ideal + self.capacity) % self.capacity))
                    
                    return check_idx
        return -1
    
    def insert(self, key, value):
        if self.size >= self.capacity * 0.75:
            self._resize()
        
        ideal = self._hash(key)
        
        # Check if key already exists
        if self._lookup_internal(key, ideal) is not None:
            return False
        
        # Find empty slot
        empty_idx = self._find_slot(ideal)
        if empty_idx == -1:
            return False
        
        # If within hop_range, place directly
        while (empty_idx - ideal + self.capacity) % self.capacity >= self.hop_range:
            empty_idx = self._try_move(empty_idx)
            if empty_idx == -1:
                return False
        
        # Place the element
        self.table[empty_idx] = (key, value)
        self.hop_info[ideal] |= (1 << ((empty_idx - ideal + self.capacity) % self.capacity))
        self.size += 1
        return True
    
    def _lookup_internal(self, key, ideal):
        hop_bitmap = self.hop_info[ideal]
        for i in range(self.hop_range):
            if hop_bitmap & (1 << i):
                idx = (ideal + i) % self.capacity
                if self.table[idx] is not None and self.table[idx][0] == key:
                    return idx
        return None
    
    def lookup(self, key):
        ideal = self._hash(key)
        idx = self._lookup_internal(key, ideal)
        return self.table[idx][1] if idx is not None else None
    
    def _resize(self):
        old_table = self.table
        old_capacity = self.capacity
        self.capacity *= 2
        self.table = [None] * self.capacity
        self.hop_info = [0] * self.capacity
        self.size = 0
        
        for item in old_table:
            if item is not None:
                self.insert(item[0], item[1])
```

**Why**: Hopscotch hashing improves cache performance by guaranteeing elements stay within a small range of their ideal bucket, reducing cache misses compared to standard open addressing.

## Quadratic Probing

```python
# PROBLEM: Implement hash table with quadratic probing for collision resolution
# TIME: O(1) average for operations, O(n) worst case
# SPACE: O(n) for table storage
class QuadraticProbingHashTable:
    def __init__(self, capacity=16):
        self.capacity = self._next_power_of_2(capacity)
        self.table = [None] * self.capacity
        self.deleted = [False] * self.capacity  # Track deleted slots
        self.size = 0
    
    def _next_power_of_2(self, n):
        power = 1
        while power < n:
            power *= 2
        return power
    
    def _hash(self, key):
        return hash(key) % self.capacity
    
    def _probe(self, key, i):
        # Quadratic probing: (h(k) + c1*i + c2*i^2) mod m
        # Using c1=0.5, c2=0.5 for better distribution
        return (self._hash(key) + i * (i + 1) // 2) % self.capacity
    
    def _find_slot(self, key):
        # Returns (index, found) where found=True if key exists
        first_deleted = -1
        
        for i in range(self.capacity):
            idx = self._probe(key, i)
            
            if self.table[idx] is None:
                # Empty slot - key not found
                return (first_deleted if first_deleted != -1 else idx, False)
            
            if self.deleted[idx]:
                # Remember first deleted slot
                if first_deleted == -1:
                    first_deleted = idx
            elif self.table[idx][0] == key:
                # Found the key
                return (idx, True)
        
        # Table is full
        return (first_deleted if first_deleted != -1 else -1, False)
    
    def insert(self, key, value):
        if self.size >= self.capacity * 0.7:
            self._resize()
        
        idx, found = self._find_slot(key)
        
        if idx == -1:
            raise Exception("Hash table is full")
        
        if found:
            # Update existing
            self.table[idx] = (key, value)
        else:
            # Insert new
            self.table[idx] = (key, value)
            self.deleted[idx] = False
            self.size += 1
        
        return not found
    
    def lookup(self, key):
        idx, found = self._find_slot(key)
        return self.table[idx][1] if found else None
    
    def delete(self, key):
        idx, found = self._find_slot(key)
        
        if found:
            self.table[idx] = None
            self.deleted[idx] = True
            self.size -= 1
            return True
        return False
    
    def _resize(self):
        old_table = self.table
        self.capacity *= 2
        self.table = [None] * self.capacity
        self.deleted = [False] * self.capacity
        self.size = 0
        
        for item in old_table:
            if item is not None:
                self.insert(item[0], item[1])
```

**Why**: Quadratic probing reduces clustering compared to linear probing by using a quadratic function to determine probe sequence, providing better distribution while maintaining cache locality.

## Sidecar Pattern

```python
# PROBLEM: Implement the sidecar pattern - a helper container running alongside
# the main application to handle cross-cutting concerns (logging, config, etc.)
# TIME: O(1) for message passing
# SPACE: O(n) for message queue
import threading
import queue
import time
from abc import ABC, abstractmethod

class Sidecar(ABC):
    """Abstract base class for sidecar implementations"""
    
    def __init__(self, name):
        self.name = name
        self.message_queue = queue.Queue()
        self.running = False
        self.thread = None
    
    @abstractmethod
    def process_message(self, message):
        """Process incoming message from main application"""
        pass
    
    def start(self):
        """Start the sidecar thread"""
        self.running = True
        self.thread = threading.Thread(target=self._run, daemon=True)
        self.thread.start()
    
    def stop(self):
        """Stop the sidecar"""
        self.running = False
        if self.thread:
            self.thread.join(timeout=5)
    
    def send(self, message):
        """Send message to sidecar"""
        self.message_queue.put(message)
    
    def _run(self):
        """Main loop for processing messages"""
        while self.running:
            try:
                message = self.message_queue.get(timeout=0.1)
                self.process_message(message)
            except queue.Empty:
                continue


class LoggingSidecar(Sidecar):
    """Sidecar for centralized logging"""
    
    def __init__(self, log_file="app.log"):
        super().__init__("logging-sidecar")
        self.log_file = log_file
        self.buffer = []
        self.buffer_size = 100
    
    def process_message(self, message):
        # Buffer logs for batch write
        self.buffer.append({
            'timestamp': time.time(),
            'level': message.get('level', 'INFO'),
            'message': message.get('message', '')
        })
        
        if len(self.buffer) >= self.buffer_size:
            self._flush()
    
    def _flush(self):
        # Write buffered logs to file
        with open(self.log_file, 'a') as f:
            for log in self.buffer:
                f.write(f"[{log['timestamp']}] {log['level']}: {log['message']}\n")
        self.buffer.clear()
    
    def stop(self):
        self._flush()  # Flush remaining logs
        super().stop()


class ConfigurationSidecar(Sidecar):
    """Sidecar for dynamic configuration management"""
    
    def __init__(self, config_file="config.json"):
        super().__init__("config-sidecar")
        self.config_file = config_file
        self.config = self._load_config()
        self.subscribers = {}
    
    def _load_config(self):
        import json
        try:
            with open(self.config_file) as f:
                return json.load(f)
        except FileNotFoundError:
            return {}
    
    def process_message(self, message):
        action = message.get('action')
        
        if action == 'get':
            key = message.get('key')
            return self.config.get(key)
        
        elif action == 'set':
            key = message.get('key')
            value = message.get('value')
            self.config[key] = value
            self._notify_subscribers(key, value)
        
        elif action == 'reload':
            self.config = self._load_config()
    
    def subscribe(self, key, callback):
        if key not in self.subscribers:
            self.subscribers[key] = []
        self.subscribers[key].append(callback)
    
    def _notify_subscribers(self, key, value):
        if key in self.subscribers:
            for callback in self.subscribers[key]:
                callback(key, value)
```

**Why**: Sidecar pattern separates cross-cutting concerns from main application logic, enabling independent deployment, scaling, and updates of auxiliary services while maintaining loose coupling.

## Ambassador Pattern

```python
# PROBLEM: Implement ambassador pattern - a proxy service that handles connection
# management, monitoring, and retry logic for remote service connections
# TIME: O(1) for routing, O(n) for retries
# SPACE: O(n) for connection pool
import socket
import time
from dataclasses import dataclass
from typing import Optional, Callable
from enum import Enum

class ConnectionState(Enum):
    HEALTHY = "healthy"
    UNHEALTHY = "unhealthy"
    UNKNOWN = "unknown"

@dataclass
class ServiceEndpoint:
    host: str
    port: int
    state: ConnectionState = ConnectionState.UNKNOWN
    last_check: float = 0
    failure_count: int = 0

class Ambassador:
    """Ambassador service for managing remote connections"""
    
    def __init__(self, endpoints: list, 
                 health_check_interval: int = 30,
                 max_failures: int = 3,
                 retry_attempts: int = 3):
        self.endpoints = [
            ServiceEndpoint(host=ep['host'], port=ep['port']) 
            for ep in endpoints
        ]
        self.health_check_interval = health_check_interval
        self.max_failures = max_failures
        self.retry_attempts = retry_attempts
        self.request_hooks = []
    
    def add_hook(self, hook: Callable):
        """Add a hook for request/response processing"""
        self.request_hooks.append(hook)
    
    def _select_endpoint(self) -> Optional[ServiceEndpoint]:
        """Select a healthy endpoint using round-robin"""
        healthy = [ep for ep in self.endpoints 
                   if ep.state == ConnectionState.HEALTHY]
        
        if not healthy:
            # Fallback to all endpoints if none are healthy
            healthy = self.endpoints
        
        # Simple round-robin selection
        return healthy[0] if healthy else None
    
    def _check_health(self, endpoint: ServiceEndpoint) -> bool:
        """Check if endpoint is healthy"""
        current_time = time.time()
        
        # Skip if checked recently
        if current_time - endpoint.last_check < self.health_check_interval:
            return endpoint.state == ConnectionState.HEALTHY
        
        try:
            # Simple TCP health check
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(2)
            sock.connect((endpoint.host, endpoint.port))
            sock.close()
            
            endpoint.state = ConnectionState.HEALTHY
            endpoint.failure_count = 0
            endpoint.last_check = current_time
            return True
        
        except Exception:
            endpoint.failure_count += 1
            endpoint.last_check = current_time
            
            if endpoint.failure_count >= self.max_failures:
                endpoint.state = ConnectionState.UNHEALTHY
            
            return False
    
    def execute_request(self, request_func: Callable, *args, **kwargs):
        """Execute request with retry logic and circuit breaker"""
        
        for attempt in range(self.retry_attempts):
            endpoint = self._select_endpoint()
            
            if not endpoint:
                raise Exception("No available endpoints")
            
            # Health check before request
            if not self._check_health(endpoint):
                continue
            
            try:
                # Execute hooks before request
                for hook in self.request_hooks:
                    args, kwargs = hook.before_request(args, kwargs)
                
                # Make the request
                response = request_func(endpoint, *args, **kwargs)
                
                # Execute hooks after request
                for hook in self.request_hooks:
                    response = hook.after_request(response)
                
                return response
            
            except Exception as e:
                endpoint.failure_count += 1
                if endpoint.failure_count >= self.max_failures:
                    endpoint.state = ConnectionState.UNHEALTHY
                
                if attempt == self.retry_attempts - 1:
                    raise e
                
                time.sleep(2 ** attempt)  # Exponential backoff


class RequestHook:
    """Base class for request hooks"""
    
    def before_request(self, args, kwargs):
        return args, kwargs
    
    def after_request(self, response):
        return response


class MetricsHook(RequestHook):
    """Hook for collecting metrics"""
    
    def __init__(self):
        self.metrics = {
            'request_count': 0,
            'error_count': 0,
            'total_latency': 0
        }
    
    def before_request(self, args, kwargs):
        self.start_time = time.time()
        return args, kwargs
    
    def after_request(self, response):
        latency = time.time() - self.start_time
        self.metrics['request_count'] += 1
        self.metrics['total_latency'] += latency
        return response
    
    def record_error(self):
        self.metrics['error_count'] += 1
```

**Why**: Ambassador pattern offloads connection management, retry logic, and health checking from the main application, providing a unified interface to remote services with built-in resilience.

## Service Mesh

```python
# PROBLEM: Implement a lightweight service mesh for managing service-to-service
# communication with load balancing, circuit breaking, and observability
# TIME: O(1) for routing, O(log n) for weighted load balancing
# SPACE: O(n) for service registry
import time
import random
from collections import defaultdict
from dataclasses import dataclass
from typing import Dict, List, Optional

@dataclass
class ServiceInstance:
    id: str
    host: str
    port: int
    weight: int = 1
    healthy: bool = True
    
    # Metrics
    request_count: int = 0
    error_count: int = 0
    total_latency: float = 0


class ServiceRegistry:
    """Service registry for service discovery"""
    
    def __init__(self):
        self.services: Dict[str, List[ServiceInstance]] = defaultdict(list)
    
    def register(self, service_name: str, instance: ServiceInstance):
        """Register a service instance"""
        self.services[service_name].append(instance)
    
    def deregister(self, service_name: str, instance_id: str):
        """Deregister a service instance"""
        self.services[service_name] = [
            inst for inst in self.services[service_name] 
            if inst.id != instance_id
        ]
    
    def get_instances(self, service_name: str) -> List[ServiceInstance]:
        """Get all instances of a service"""
        return [
            inst for inst in self.services.get(service_name, [])
            if inst.healthy
        ]


class LoadBalancer:
    """Load balancer with multiple strategies"""
    
    @staticmethod
    def round_robin(instances: List[ServiceInstance], index: int) -> ServiceInstance:
        """Simple round-robin selection"""
        return instances[index % len(instances)]
    
    @staticmethod
    def weighted_random(instances: List[ServiceInstance]) -> ServiceInstance:
        """Weighted random selection"""
        total_weight = sum(inst.weight for inst in instances)
        rand = random.uniform(0, total_weight)
        
        cumulative = 0
        for instance in instances:
            cumulative += instance.weight
            if rand <= cumulative:
                return instance
        
        return instances[-1]
    
    @staticmethod
    def least_connections(instances: List[ServiceInstance]) -> ServiceInstance:
        """Select instance with least active connections"""
        return min(instances, key=lambda x: x.request_count - x.error_count)


class CircuitBreaker:
    """Circuit breaker for fault tolerance"""
    
    def __init__(self, failure_threshold: int = 5, 
                 recovery_timeout: int = 60):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.failures = defaultdict(int)
        self.last_failure = defaultdict(float)
        self.state = defaultdict(lambda: "closed")
    
    def is_allowed(self, instance_id: str) -> bool:
        """Check if requests are allowed to instance"""
        if self.state[instance_id] == "closed":
            return True
        
        if self.state[instance_id] == "open":
            # Check if recovery timeout has passed
            if time.time() - self.last_failure[instance_id] > self.recovery_timeout:
                self.state[instance_id] = "half-open"
                return True
            return False
        
        # Half-open state - allow one request
        return True
    
    def record_success(self, instance_id: str):
        """Record successful request"""
        self.failures[instance_id] = 0
        self.state[instance_id] = "closed"
    
    def record_failure(self, instance_id: str):
        """Record failed request"""
        self.failures[instance_id] += 1
        self.last_failure[instance_id] = time.time()
        
        if self.failures[instance_id] >= self.failure_threshold:
            self.state[instance_id] = "open"


class ServiceMesh:
    """Service mesh for managing inter-service communication"""
    
    def __init__(self):
        self.registry = ServiceRegistry()
        self.circuit_breaker = CircuitBreaker()
        self.round_robin_index = 0
        self.metrics = defaultdict(lambda: {
            'requests': 0,
            'errors': 0,
            'latency_sum': 0
        })
    
    def register_service(self, service_name: str, 
                        instance_id: str, host: str, port: int, weight: int = 1):
        """Register a service instance"""
        instance = ServiceInstance(
            id=instance_id, host=host, port=port, weight=weight
        )
        self.registry.register(service_name, instance)
    
    def get_instance(self, service_name: str, 
                    strategy: str = "round_robin") -> Optional[ServiceInstance]:
        """Get a service instance using specified load balancing strategy"""
        instances = self.registry.get_instances(service_name)
        
        if not instances:
            return None
        
        # Filter by circuit breaker
        available = [
            inst for inst in instances 
            if self.circuit_breaker.is_allowed(inst.id)
        ]
        
        if not available:
            return None
        
        # Apply load balancing strategy
        if strategy == "round_robin":
            instance = LoadBalancer.round_robin(available, self.round_robin_index)
            self.round_robin_index += 1
        elif strategy == "weighted":
            instance = LoadBalancer.weighted_random(available)
        elif strategy == "least_conn":
            instance = LoadBalancer.least_connections(available)
        else:
            instance = available[0]
        
        return instance
    
    def record_request(self, service_name: str, instance_id: str, 
                      latency: float, success: bool):
        """Record request metrics"""
        self.metrics[service_name]['requests'] += 1
        self.metrics[service_name]['latency_sum'] += latency
        
        if success:
            self.circuit_breaker.record_success(instance_id)
        else:
            self.metrics[service_name]['errors'] += 1
            self.circuit_breaker.record_failure(instance_id)
    
    def get_metrics(self, service_name: str) -> dict:
        """Get metrics for a service"""
        return dict(self.metrics[service_name])
```

**Why**: Service mesh provides a dedicated infrastructure layer for handling service-to-service communication, offering load balancing, circuit breaking, and observability without application code changes.

## API Composition

```python
# PROBLEM: Implement API composition pattern for aggregating data from
# multiple microservices into a single response
# TIME: O(max(services)) for parallel, O(sum(services)) for sequential
# SPACE: O(n) for response aggregation
import asyncio
import aiohttp
from dataclasses import dataclass
from typing import Dict, List, Any, Optional
from enum import Enum

class ServiceStatus(Enum):
    SUCCESS = "success"
    FAILED = "failed"
    TIMEOUT = "timeout"

@dataclass
class ServiceResponse:
    service_name: str
    status: ServiceStatus
    data: Any = None
    error: Optional[str] = None
    latency: float = 0


class APIComposer:
    """Compose responses from multiple microservices"""
    
    def __init__(self, timeout: float = 5.0):
        self.timeout = timeout
        self.service_registry = {}
        self.fallback_handlers = {}
    
    def register_service(self, name: str, base_url: str, 
                        required: bool = True):
        """Register a microservice"""
        self.service_registry[name] = {
            'base_url': base_url,
            'required': required
        }
    
    def register_fallback(self, service_name: str, handler):
        """Register fallback handler for a service"""
        self.fallback_handlers[service_name] = handler
    
    async def _call_service(self, session: aiohttp.ClientSession,
                           service_name: str, endpoint: str,
                           params: dict = None) -> ServiceResponse:
        """Call a single microservice"""
        import time
        start_time = time.time()
        
        try:
            service = self.service_registry[service_name]
            url = f"{service['base_url']}/{endpoint}"
            
            async with session.get(url, params=params, 
                                  timeout=self.timeout) as response:
                latency = time.time() - start_time
                
                if response.status == 200:
                    data = await response.json()
                    return ServiceResponse(
                        service_name=service_name,
                        status=ServiceStatus.SUCCESS,
                        data=data,
                        latency=latency
                    )
                else:
                    return ServiceResponse(
                        service_name=service_name,
                        status=ServiceStatus.FAILED,
                        error=f"HTTP {response.status}",
                        latency=latency
                    )
        
        except asyncio.TimeoutError:
            latency = time.time() - start_time
            return ServiceResponse(
                service_name=service_name,
                status=ServiceStatus.TIMEOUT,
                error="Request timeout",
                latency=latency
            )
        
        except Exception as e:
            latency = time.time() - start_time
            return ServiceResponse(
                service_name=service_name,
                status=ServiceStatus.FAILED,
                error=str(e),
                latency=latency
            )
    
    async def compose(self, composition_plan: Dict[str, dict],
                     parallel: bool = True) -> Dict[str, Any]:
        """
        Compose data from multiple services
        
        Args:
            composition_plan: Dict mapping service names to their endpoints
            parallel: Whether to call services in parallel
        """
        async with aiohttp.ClientSession() as session:
            if parallel:
                # Execute all calls in parallel
                tasks = [
                    self._call_service(
                        session, service_name, 
                        config['endpoint'], 
                        config.get('params')
                    )
                    for service_name, config in composition_plan.items()
                ]
                responses = await asyncio.gather(*tasks)
            
            else:
                # Execute sequentially
                responses = []
                for service_name, config in composition_plan.items():
                    response = await self._call_service(
                        session, service_name,
                        config['endpoint'],
                        config.get('params')
                    )
                    responses.append(response)
        
        # Process responses
        result = {}
        errors = []
        
        for response in responses:
            if response.status == ServiceStatus.SUCCESS:
                result[response.service_name] = response.data
            
            else:
                # Try fallback
                if response.service_name in self.fallback_handlers:
                    fallback_data = self.fallback_handlers[response.service_name]()
                    result[response.service_name] = fallback_data
                
                elif self.service_registry[response.service_name]['required']:
                    errors.append({
                        'service': response.service_name,
                        'error': response.error
                    })
                
                else:
                    result[response.service_name] = None
        
        if errors:
            result['_errors'] = errors
        
        return result


class CompositionBuilder:
    """Builder for creating composition plans"""
    
    def __init__(self):
        self.plan = {}
    
    def add_service(self, name: str, endpoint: str, 
                   params: dict = None) -> 'CompositionBuilder':
        """Add a service to the composition plan"""
        self.plan[name] = {
            'endpoint': endpoint,
            'params': params
        }
        return self
    
    def build(self) -> Dict[str, dict]:
        """Build the composition plan"""
        return self.plan

# composer = APIComposer(timeout=3.0)
# composer.register_service('users', 'http://users-service/api', required=True)
# composer.register_service('orders', 'http://orders-service/api', required=False)
# composer.register_fallback('orders', lambda: [])
#
```

```python
                ConsistencyLevel.EVENTUAL):

        self.connections: List[DatabaseConnection] = []
        self.primary: Optional[DatabaseConnection] = None
        self.read_index = 0
        self.session_cache = {}  # For read-your-writes consistency
        self.lock = threading.Lock()
    
    def add_connection(self, connection: DatabaseConnection):
        """Add a database connection"""
        self.connections.append(connection)

            self.primary = connection

                      session_id: Optional[str] = None) -> DatabaseConnection:

            # Writes always go to primary
            return self.primary

            return self.primary

                last_write_time = self.session_cache[session_id]

            # Round-robin selection

            self.read_index += 1
```

## CAP Theorem

```python
# PROBLEM: Implement different consistency models based on CAP theorem trade-offs
# - CP (Consistency + Partition Tolerance): Strong consistency, may be unavailable
# - AP (Availability + Partition Tolerance): Always available, eventual consistency
# - CA (Consistency + Availability): No partition tolerance (single node)
# TIME: O(1) for AP, O(n) for CP (quorum)
# SPACE: O(n) for node states
import time
import threading
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Any
from enum import Enum
from collections import defaultdict

class CAPMode(Enum):
    CP = "consistency_partition"  # Strong consistency
    AP = "availability_partition"  # High availability
    CA = "consistency_availability"  # Single node

@dataclass
class VectorClock:
    """Vector clock for tracking causality"""
    clocks: Dict[str, int] = field(default_factory=dict)
    
    def increment(self, node_id: str):
        self.clocks[node_id] = self.clocks.get(node_id, 0) + 1
    
    def merge(self, other: 'VectorClock'):
        for node_id, timestamp in other.clocks.items():
            self.clocks[node_id] = max(
                self.clocks.get(node_id, 0), timestamp
            )
    
    def happens_before(self, other: 'VectorClock') -> bool:
        """Check if this clock happens before other"""
        at_least_one_less = False
        for node_id in set(self.clocks.keys()) | set(other.clocks.keys()):
            self_ts = self.clocks.get(node_id, 0)
            other_ts = other.clocks.get(node_id, 0)
            
            if self_ts > other_ts:
                return False
            if self_ts < other_ts:
                at_least_one_less = True
        
        return at_least_one_less

@dataclass
class DataItem:
    key: str
    value: Any
    version: VectorClock
    timestamp: float
    node_id: str

class CAPStore:
    """Distributed store implementing CAP trade-offs"""
    
    def __init__(self, node_id: str, mode: CAPMode, 
                 quorum_size: int = 2):
        self.node_id = node_id
        self.mode = mode
        self.quorum_size = quorum_size
        self.data: Dict[str, List[DataItem]] = defaultdict(list)
        self.lock = threading.RLock()
        self.peers: List[str] = []
        self.partitioned = False
    
    def add_peer(self, peer_id: str):
        """Add a peer node"""
        self.peers.append(peer_id)
    
    def set_partition(self, partitioned: bool):
        """Simulate network partition"""
        self.partitioned = partitioned
    
    def write(self, key: str, value: Any) -> bool:
        """Write value based on CAP mode"""
        
        if self.mode == CAPMode.CP:
            return self._write_cp(key, value)
        
        elif self.mode == CAPMode.AP:
            return self._write_ap(key, value)
        
        else:  # CA mode
            return self._write_ca(key, value)
    
    def _write_cp(self, key: str, value: Any) -> bool:
        """CP: Strong consistency with quorum"""
        if self.partitioned:
            # Cannot maintain consistency during partition
            return False
        
        # Attempt quorum write
        version = VectorClock()
        version.increment(self.node_id)
        
        item = DataItem(
            key=key,
            value=value,
            version=version,
            timestamp=time.time(),
            node_id=self.node_id
        )
        
        # In production, would coordinate with peers for quorum
        # Simplified: assume quorum if not partitioned
        with self.lock:
            self.data[key].append(item)
        
        return True
    
    def _write_ap(self, key: str, value: Any) -> bool:
        """AP: Always available, eventual consistency"""
        version = VectorClock()
        version.increment(self.node_id)
        
        item = DataItem(
            key=key,
            value=value,
            version=version,
            timestamp=time.time(),
            node_id=self.node_id
        )
        
        with self.lock:
            # Always accept write, resolve conflicts later
            self.data[key].append(item)
        
        # In production, would asynchronously replicate to peers
        return True
    
    def _write_ca(self, key: str, value: Any) -> bool:
        """CA: Single node, no partition tolerance"""
        version = VectorClock()
        version.increment(self.node_id)
        
        item = DataItem(
            key=key,
            value=value,
            version=version,
            timestamp=time.time(),
            node_id=self.node_id
        )
        
        with self.lock:
            self.data[key] = [item]
        
        return True
    
    def read(self, key: str) -> Optional[Any]:
        """Read value based on CAP mode"""
        
        if self.mode == CAPMode.CP:
            return self._read_cp(key)
        
        elif self.mode == CAPMode.AP:
            return self._read_ap(key)
        
        else:
            return self._read_ca(key)
    
    def _read_cp(self, key: str) -> Optional[Any]:
        """CP: Strong consistency read"""
        if self.partitioned:
            # Cannot guarantee consistency during partition
            return None
        
        # In production, would read from quorum
        with self.lock:
            items = self.data.get(key, [])
            return items[-1].value if items else None
    
    def _read_ap(self, key: str) -> Optional[Any]:
        """AP: Read any available version"""
        with self.lock:
            items = self.data.get(key, [])
            if not items:
                return None
            
            # Return most recent local version
            # May be stale due to eventual consistency
            return max(items, key=lambda x: x.timestamp).value
    
    def _read_ca(self, key: str) -> Optional[Any]:
        """CA: Simple read from single node"""
        with self.lock:
            items = self.data.get(key, [])
            return items[-1].value if items else None
    
    def resolve_conflicts(self):
        """Resolve conflicts using last-write-wins (for AP mode)"""
        with self.lock:
            for key in list(self.data.keys()):
                items = self.data[key]
                
                if len(items) > 1:
                    # Last-write-wins based on timestamp
                    winner = max(items, key=lambda x: x.timestamp)
                    self.data[key] = [winner]


class ConflictResolver:
    """Advanced conflict resolution strategies"""
    
    @staticmethod
    def last_write_wins(items: List[DataItem]) -> DataItem:
        """Resolve by timestamp"""
        return max(items, key=lambda x: x.timestamp)
    
    @staticmethod
    def vector_clock_wins(items: List[DataItem]) -> Optional[DataItem]:
        """Resolve using vector clocks"""
        # Find item that is not dominated by any other
        for item1 in items:
            dominated = False
            for item2 in items:
                if item1 != item2:
                    if item1.version.happens_before(item2.version):
                        dominated = True
                        break
            
            if not dominated:
                return item1
        
        # Concurrent writes - use timestamp as fallback
        return ConflictResolver.last_write_wins(items)
    
    @staticmethod
    def application_merge(items: List[DataItem], merge_func) -> DataItem:
        """Application-specific merge (e.g., for counters, sets)"""
        merged_value = merge_func([item.value for item in items])
        
        return DataItem(
            key=items[0].key,
            value=merged_value,
            version=items[0].version,
            timestamp=time.time(),
            node_id="merged"
        )
```

**Why**: CAP theorem implementations allow systems to make explicit trade-offs between consistency, availability, and partition tolerance based on application requirements, with different strategies for conflict resolution.

```python
        """Return list of supported processor types"""
        pass


class DefaultProcessorFactory(DataProcessorFactory):
    """Default factory implementation"""
    
    @classmethod
    def supported_types(cls):
        return PluginMeta.list_plugins()
    
    def create_processor(self, processor_type: str) -> DataProcessor:
        processor_class = PluginMeta.get_plugin(processor_type)
        
        if processor_class is None:
            raise ValueError(f"Unknown processor type: {processor_type}")
        
        return processor_class()


# Composite pattern with ABC
class CompositeProcessor(DataProcessor):
    """Composite processor that chains multiple processors"""
    
    plugin_name = "composite"
    
    def __init__(self, processors: list = None):
        self.processors = processors or []
    
    def add_processor(self, processor: DataProcessor):
        self.processors.append(processor)
    
    def validate(self, data: Any) -> bool:
        return all(p.validate(data) for p in self.processors)
    
    def process(self, data: Any) -> Any:
        result = data
        for processor in self.processors:
            result = processor.process(result)
        return result


# Registry pattern
class ProcessorRegistry:
    """Registry for data processors"""
    
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._processors = {}
        return cls._instance
    
    def register(self, name: str, processor: DataProcessor):
        self._processors[name] = processor
    
    def get(self, name: str) -> DataProcessor:
        return self._processors.get(name)
    
    def list_all(self) -> list:
        return list(self._processors.keys())
```
