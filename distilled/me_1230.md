## Maximum Sum Subarray of Size K

```python
# PROBLEM: Find the maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)
def max_sum_subarray(arr, k):
    """
    Sliding window technique: maintain a window of size k,
    slide it through the array while tracking the maximum sum.
    """
    if len(arr) < k:
        return None
    
    # Calculate sum of first window
    window_sum = sum(arr[:k])
    max_sum = window_sum
    
    # Slide the window: subtract leftmost, add new rightmost
    for i in range(k, len(arr)):
        window_sum = window_sum - arr[i - k] + arr[i]
        max_sum = max(max_sum, window_sum)
    
    return max_sum
```
**Why**: Classic sliding window pattern. Avoids O(n*k) brute force by reusing previous calculations.

## 3Sum

```python
# PROBLEM: Find all unique triplets that sum to zero
# TIME: O(n²) SPACE: O(1) excluding output
def three_sum(nums):
    """
    Two pointers technique on sorted array.
    Fix one element, use two pointers to find pairs.
    """
    nums.sort()
    result = []
    n = len(nums)
    
    for i in range(n - 2):
        # Skip duplicates for first element
        if i > 0 and nums[i] == nums[i - 1]:
            continue
        
        left, right = i + 1, n - 1
        
        while left < right:
            total = nums[i] + nums[left] + nums[right]
            
            if total < 0:
                left += 1
            elif total > 0:
                right -= 1
            else:
                result.append([nums[i], nums[left], nums[right]])
                # Skip duplicates
                while left < right and nums[left] == nums[left + 1]:
                    left += 1
                while left < right and nums[right] == nums[right - 1]:
                    right -= 1
                left += 1
                right -= 1
    
    return result
```
**Why**: Two pointers reduces O(n³) brute force to O(n²). Sorting enables the pointer strategy.

## Permutations

```python
# PROBLEM: Generate all permutations of a list of numbers
# TIME: O(n! * n) SPACE: O(n) for recursion depth
def permute(nums):
    """
    Backtracking: build permutations by trying each unused element,
    backtrack when reaching a complete permutation.
    """
    result = []
    used = [False] * len(nums)
    
    def backtrack(current):
        # Base case: complete permutation
        if len(current) == len(nums):
            result.append(current[:])
            return
        
        for i in range(len(nums)):
            if used[i]:
                continue
            
            # Choose
            used[i] = True
            current.append(nums[i])
            
            # Explore
            backtrack(current)
            
            # Unchoose (backtrack)
            current.pop()
            used[i] = False
    
    backtrack([])
    return result
```
**Why**: Backtracking explores all possibilities systematically. The choose-explore-unchoose pattern is fundamental.

## Number of Connected Components

```python
# PROBLEM: Count connected components in an undirected graph
# TIME: O(n * α(n)) ≈ O(n) SPACE: O(n)
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.components = n
    
    def find(self, x):
        # Path compression
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        px, py = self.find(x), self.find(y)
        if px == py:
            return False
        
        # Union by rank
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        self.components -= 1
        return True

def count_components(n, edges):
    uf = UnionFind(n)
    for u, v in edges:
        uf.union(u, v)
    return uf.components
```
**Why**: Union-Find (Disjoint Set Union) is optimal for connectivity queries. Path compression and union by rank ensure near-constant time operations.

## Longest Increasing Subsequence

```python
# PROBLEM: Find the length of the longest increasing subsequence
# TIME: O(n log n) SPACE: O(n)
def length_of_lis(nums):
    """
    Binary search approach: maintain an array where tails[i] is the
    smallest tail of all increasing subsequences of length i+1.
    """
    if not nums:
        return 0
    
    tails = []
    
    for num in nums:
        # Binary search for the position to replace
        left, right = 0, len(tails)
        while left < right:
            mid = (left + right) // 2
            if tails[mid] < num:
                left = mid + 1
            else:
                right = mid
        
        if left == len(tails):
            tails.append(num)
        else:
            tails[left] = num
    
    return len(tails)
```
**Why**: Binary search optimization transforms O(n²) DP to O(n log n). Key insight: smaller tails allow more extensions.

## Dijkstra's Shortest Path

```python
# PROBLEM: Find shortest path from source to all nodes in weighted graph
# TIME: O((V + E) log V) SPACE: O(V)
import heapq

def dijkstra(n, edges, source):
    """
    Greedy BFS with min-heap: always process closest unvisited node.
    Works because shortest path to processed nodes is finalized.
    """
    # Build adjacency list
    graph = {i: [] for i in range(n)}
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))  # Undirected
    
    # Distance array, heap: (distance, node)
    dist = [float('inf')] * n
    dist[source] = 0
    heap = [(0, source)]
    
    while heap:
        d, u = heapq.heappop(heap)
        
        # Skip if we've found a better path already
        if d > dist[u]:
            continue
        
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(heap, (dist[v], v))
    
    return dist
```
**Why**: Min-heap ensures we always process the closest node first. Cannot use on graphs with negative edges.

## Lowest Common Ancestor in Binary Tree

```python
# PROBLEM: Find LCA of two nodes in a binary tree
# TIME: O(n) SPACE: O(h) where h is height
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def lowest_common_ancestor(root, p, q):
    """
    Recursively search for p and q. LCA is where both subtrees
    contain one of the target nodes, or the current node is a target.
    """
    if not root:
        return None
    
    # If current node is p or q, it could be LCA
    if root == p or root == q:
        return root
    
    # Search in both subtrees
    left = lowest_common_ancestor(root.left, p, q)
    right = lowest_common_ancestor(root.right, p, q)
    
    # If both sides found something, this is the LCA
    if left and right:
        return root
    
    # Otherwise return the non-null result
    return left if left else right
```
**Why**: Post-order traversal naturally finds LCA. If targets are in different subtrees, current node is the split point.

## Merge K Sorted Lists

```python
# PROBLEM: Merge k sorted linked lists into one sorted list
# TIME: O(n log k) where n = total nodes, k = number of lists
# SPACE: O(k) for the heap
import heapq

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next
    
    # For heap comparison
    def __lt__(self, other):
        return self.val < other.val

def merge_k_lists(lists):
    """
    Min-heap approach: always pick the smallest available node.
    Each extraction is O(log k), done n times total.
    """
    # Initialize heap with first node from each list
    heap = [lst for lst in lists if lst]
    heapq.heapify(heap)
    
    dummy = ListNode(0)
    current = dummy
    
    while heap:
        # Get smallest node
        node = heapq.heappop(heap)
        current.next = node
        current = current.next
        
        # Add next node from same list if exists
        if node.next:
            heapq.heappush(heap, node.next)
    
    return dummy.next
```
**Why**: Heap efficiently tracks k minimums. Avoids O(nk) naive merge by leveraging sorted property.

## Trie - Word Search

```python
# PROBLEM: Implement insert, search, and startsWith for a trie
# TIME: O(m) per operation where m = word length
# SPACE: O(alphabet_size * total_chars)
class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_word = False

class Trie:
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word):
        """Add word to trie, creating nodes as needed."""
        node = self.root
        for ch in word:
            if ch not in node.children:
                node.children[ch] = TrieNode()
            node = node.children[ch]
        node.is_word = True
    
    def search(self, word):
        """Check if complete word exists in trie."""
        node = self.root
        for ch in word:
            if ch not in node.children:
                return False
            node = node.children[ch]
        return node.is_word
    
    def starts_with(self, prefix):
        """Check if any word starts with given prefix."""
        node = self.root
        for ch in prefix:
            if ch not in node.children:
                return False
            node = node.children[ch]
        return True
```
**Why**: Trie provides O(m) prefix operations regardless of dictionary size. Essential for autocomplete and spell checkers.

## LRU Cache

```python
# PROBLEM: Design a cache with O(1) get and put, evicting least recently used
# TIME: O(1) for both operations
# SPACE: O(capacity)
class Node:
    """Doubly linked list node for O(1) removal."""
    def __init__(self, key=0, val=0):
        self.key = key
        self.val = val
        self.prev = None
        self.next = None

class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}  # key -> Node
        
        # Dummy head and tail for easier manipulation
        self.head = Node()
        self.tail = Node()
        self.head.next = self.tail
        self.tail.prev = self.head
    
    def _remove(self, node):
        """Remove node from doubly linked list."""
        node.prev.next = node.next
        node.next.prev = node.prev
    
    def _add_to_front(self, node):
        """Add node right after head (most recent)."""
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node
    
    def get(self, key):
        if key not in self.cache:
            return -1
        
        node = self.cache[key]
        # Move to front (most recently used)
        self._remove(node)
        self._add_to_front(node)
        return node.val
    
    def put(self, key, value):
        if key in self.cache:
            # Update existing and move to front
            node = self.cache[key]
            node.val = value
            self._remove(node)
            self._add_to_front(node)
        else:
            # Add new node
            if len(self.cache) >= self.capacity:
                # Evict LRU (node before tail)
                lru = self.tail.prev
                self._remove(lru)
                del self.cache[lru.key]
            
            new_node = Node(key, value)
            self.cache[key] = new_node
            self._add_to_front(new_node)
```
**Why**: HashMap + Doubly linked list achieves O(1) for both operations. HashMap for fast lookup, DLL for O(1) reordering.
