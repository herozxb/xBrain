## [html_link_rel_next]

```html
<!-- PROBLEM: How to indicate the next page in a paginated sequence -->
<!-- TIME: O(1) SPACE: O(1) -->

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Pagination - Next Link</title>
    <!-- 
        rel="next" tells search engines and browsers about pagination.
        Helps SEO by indicating document sequence relationships.
        Use with rel="prev" for complete pagination support.
    -->
    <link rel="next" href="page-2.html">
</head>
<body>
    <nav aria-label="Pagination">
        <ul class="pagination">
            <li><span class="current">Page 1</span></li>
            <li><a href="page-2.html" rel="next">Page 2 →</a></li>
            <li><a href="page-3.html">Page 3</a></li>
        </ul>
    </nav>
    
    <script>
        // Programmatic rel="next" link creation
        function createNextLink(href) {
            const link = document.createElement('link');
            link.rel = 'next';
            link.href = href;
            document.head.appendChild(link);
            return link;
        }
        
        // Usage
        createNextLink('page-2.html');
    </script>
</body>
</html>
```

**Why**: `rel="next"` improves SEO by helping search engines understand pagination sequences, preventing duplicate content issues and consolidating link equity across paginated content.

## [html_link_rel_prev]

```html
<!-- PROBLEM: How to indicate the previous page in a paginated sequence -->
<!-- TIME: O(1) SPACE: O(1) -->

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Pagination - Previous Link</title>
    <!-- 
        rel="prev" indicates the previous document in a series.
        Always pair with rel="next" for proper pagination.
        Essential for search engine pagination handling.
    -->
    <link rel="prev" href="page-1.html">
    <link rel="next" href="page-3.html">
</head>
<body>
    <nav aria-label="Pagination">
        <ul class="pagination">
            <li><a href="page-1.html" rel="prev">← Page 1</a></li>
            <li><span class="current">Page 2</span></li>
            <li><a href="page-3.html" rel="next">Page 3 →</a></li>
        </ul>
    </nav>
    
    <script>
        // Auto-generate pagination links based on current page
        function setupPaginationLinks(currentPage, totalPages, baseUrl) {
            // Remove existing prev/next links
            document.querySelectorAll('link[rel="prev"], link[rel="next"]')
                .forEach(link => link.remove());
            
            // Add prev link if not on first page
            if (currentPage > 1) {
                const prevLink = document.createElement('link');
                prevLink.rel = 'prev';
                prevLink.href = `${baseUrl}?page=${currentPage - 1}`;
                document.head.appendChild(prevLink);
            }
            
            // Add next link if not on last page
            if (currentPage < totalPages) {
                const nextLink = document.createElement('link');
                nextLink.rel = 'next';
                nextLink.href = `${baseUrl}?page=${currentPage + 1}`;
                document.head.appendChild(nextLink);
            }
        }
        
        // Usage: Page 2 of 10
        setupPaginationLinks(2, 10, '/articles');
    </script>
</body>
</html>
```

**Why**: `rel="prev"` completes the pagination chain, enabling search engines to properly index sequential content and users to navigate predictably through paginated results.

## [css_grid_auto_columns]

```html
<!-- PROBLEM: How to automatically size grid columns based on content -->
<!-- TIME: O(1) SPACE: O(1) -->

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Grid Auto Columns</title>
    <style>
        /* 
            grid-auto-columns controls the size of implicitly created columns.
            Used when grid items are placed outside defined columns.
            Values: auto, min-content, max-content, fr units, lengths.
        */
        
        .grid-auto-example {
            display: grid;
            grid-template-columns: 100px; /* Only one explicit column */
            grid-auto-columns: 150px; /* Implicit columns are 150px */
            gap: 10px;
        }
        
        .item { background: lightblue; padding: 10px; }
        
        /* More examples */
        .grid-auto-minmax {
            display: grid;
            grid-template-columns: 200px;
            /* Implicit columns: min 100px, max 1fr */
            grid-auto-columns: minmax(100px, 1fr);
            gap: 10px;
        }
        
        .grid-auto-content {
            display: grid;
            grid-template-columns: repeat(2, 1fr);
            /* Columns sized to their max-content */
            grid-auto-columns: max-content;
            gap: 10px;
        }
        
        /* Dynamic column sizing with auto */
        .grid-dynamic {
            display: grid;
            grid-template-columns: 1fr;
            grid-auto-columns: auto; /* Size based on content */
            gap: 15px;
        }
    </style>
</head>
<body>
    <h3>Grid Auto Columns - Implicit Column Sizing</h3>
    
    <div class="grid-auto-example">
        <div class="item">Column 1 (explicit)</div>
        <div class="item" style="grid-column: 3;">Column 3 (implicit, 150px)</div>
    </div>
    
    <h3>Grid with minmax()</h3>
    <div class="grid-auto-minmax">
        <div class="item">Explicit</div>
        <div class="item" style="grid-column: 3;">Implicit minmax column</div>
    </div>
    
    <script>
        // JavaScript to dynamically set grid-auto-columns
        function setAutoColumns(element, value) {
            element.style.gridAutoColumns = value;
        }
        
        // Get computed grid-auto-columns
        function getAutoColumns(element) {
            return getComputedStyle(element).gridAutoColumns;
        }
        
        // Usage
        const grid = document.querySelector('.grid-auto-example');
        setAutoColumns(grid, 'minmax(50px, auto)');
        console.log(getAutoColumns(grid)); // "minmax(50px, auto)"
    </script>
</body>
</html>
```

**Why**: `grid-auto-columns` provides control over implicit grid tracks created when items overflow defined columns, ensuring consistent layouts even with dynamic content placement.

## [css_selector_child]

```html
<!-- PROBLEM: How to select only direct children of an element -->
<!-- TIME: O(1) SPACE: O(1) -->

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Child Selector (>)</title>
    <style>
        /* 
            Child combinator (>) selects only direct children.
            More specific than descendant selector (space).
            Prevents styles from cascading to nested elements.
        */
        
        /* Only direct <li> children of .menu */
        .menu > li {
            background: #f0f0f0;
            padding: 10px;
            margin: 5px 0;
            list-style: none;
        }
        
        /* Nested lists are NOT affected */
        .menu > li ul {
            padding-left: 20px;
        }
        
        /* Direct paragraph children of article */
        article > p {
            line-height: 1.6;
            margin-bottom: 1em;
        }
        
        /* Multiple child selectors */
        .container > .header > h1 {
            color: navy;
            border-bottom: 2px solid navy;
        }
        
        /* Child selector with classes */
        .card > .title {
            font-size: 1.5em;
            font-weight: bold;
        }
        
        .card > .content {
            color: #666;
        }
    </style>
</head>
<body>
    <h3>Child Selector Demo</h3>
    
    <ul class="menu">
        <li>Menu Item 1 (direct child - styled)</li>
        <li>
            Menu Item 2 (direct child - styled)
            <ul>
                <li>Sub-item (NOT direct child - unstyled)</li>
                <li>Another sub-item</li>
            </ul>
        </li>
        <li>Menu Item 3 (direct child - styled)</li>
    </ul>
    
    <article>
        <p>Direct child paragraph - styled with line-height</p>
        <div>
            <p>Nested paragraph - NOT affected by article > p</p>
        </div>
        <p>Another direct child paragraph</p>
    </article>
    
    <script>
        // JavaScript: querySelector with child combinator
        const directChildren = document.querySelectorAll('.menu > li');
        console.log(`Direct menu items: ${directChildren.length}`);
        
        // Get only direct element children
        function getDirectChildren(parent, selector) {
            return Array.from(parent.children).filter(
                child => child.matches(selector)
            );
        }
        
        const menu = document.querySelector('.menu');
        const liChildren = getDirectChildren(menu, 'li');
        console.log(`LI children: ${liChildren.length}`); // 3
        
        // Alternative: use :scope > selector
        const scopedChildren = menu.querySelectorAll(':scope > li');
        console.log(`Scoped children: ${scopedChildren.length}`); // 3
    </script>
</body>
</html>
```

**Why**: The child combinator (`>`) provides precise control over styling by targeting only direct children, preventing unintended style inheritance in nested structures.

## [css_pseudo_class_focus]

```html

<!-- TIME: O(1) SPACE: O(1) -->
```


## [js_callback_functions]

```html
<!-- PROBLEM: How to pass functions as arguments to other functions -->
<!-- TIME: O(1) per call SPACE: O(1) -->

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>JavaScript Callback Functions</title>
</head>
<body>
    <h3>Callback Functions Demo</h3>
    <div id="output"></div>
    
    <script>
        const output = document.getElementById('output');
        
        function log(msg) {
            output.innerHTML += `<p>${msg}</p>`;
            console.log(msg);
        }
        
        // Basic callback example
        function greet(name, callback) {
            const message = `Hello, ${name}!`;
            callback(message); // Execute the callback
        }
        
        // Named function as callback
        function displayMessage(msg) {
            log(`[Display] ${msg}`);
        }
        
        greet('Alice', displayMessage);
        
        // Anonymous function as callback
        greet('Bob', function(msg) {
            log(`[Anonymous] ${msg}`);
        });
        
        // Arrow function as callback
        greet('Charlie', (msg) => {
            log(`[Arrow] ${msg}`);
        });
        
        // Callback with multiple arguments
        function calculate(a, b, operation, callback) {
            const result = operation(a, b);
            callback(a, b, result);
        }
        
        calculate(5, 3, 
            (x, y) => x + y,  // Operation callback
            (a, b, result) => log(`${a} + ${b} = ${result}`)  // Result callback
        );
        
        calculate(10, 4,
            (x, y) => x * y,
            (a, b, result) => log(`${a} × ${b} = ${result}`)
        );
        
        // Async callback (setTimeout)
        function delayedGreeting(name, callback) {
            setTimeout(() => {
                callback(`Delayed hello to ${name}!`);
            }, 1000);
        }
        
        delayedGreeting('Async User', (msg) => {
            log(`[After 1s] ${msg}`);
        });
        
        // Array method callbacks
        const numbers = [1, 2, 3, 4, 5];
        
        // forEach with callback
        log('forEach:');
        numbers.forEach((num, index) => {
            log(`  Index ${index}: ${num}`);
        });
        
        // map with callback - transform each element
        const doubled = numbers.map((num) => num * 2);
        log(`map (×2): [${doubled}]`);
        
        // filter with callback - keep matching elements
        const evens = numbers.filter((num) => num % 2 === 0);
        log(`filter (evens): [${evens}]`);
        
        // reduce with callback - accumulate values
        const sum = numbers.reduce((acc, num) => acc + num, 0);
        log(`reduce (sum): ${sum}`);
        
        // find with callback - find first match
        const firstGreaterThan3 = numbers.find((num) => num > 3);
        log(`find (>3): ${firstGreaterThan3}`);
        
        // Error-first callback pattern (Node.js style)
        function readFileMock(filename, callback) {
            setTimeout(() => {
                if (filename === 'error.txt') {
                    callback(new Error('File not found'), null);
                } else {
                    callback(null, `Content of ${filename}`);
                }
            }, 500);
        }
        
        readFileMock('data.txt', (err, data) => {
            if (err) {
                log(`Error: ${err.message}`);
            } else {
                log(`Success: ${data}`);
            }
        });
        
        // Callback with context (thisArg)
        const processor = {
            prefix: 'Processed:',
            process: function(items, callback) {
                return items.map(callback, this);
            }
        };
        
        const processed = processor.process([1, 2, 3], function(item) {
            return `${this.prefix} ${item}`;
        });
        log(`With context: [${processed}]`);
    </script>
</body>
</html>
```

**Why**: Callbacks enable asynchronous programming, event handling, and functional patterns, forming the foundation of JavaScript's non-blocking architecture.

```html
        log(`Original: [${fruits}]`);
        
        const newLength = fruits.unshift('apple');
        log(`After unshift('apple'): [${fruits}]`);
        log(`Returns new length: ${newLength}`);
        
        // Add multiple elements at once
        fruits.unshift('mango', 'grape');
        log(`After unshift('mango', 'grape'): [${fruits}]`);
        
        // unshift returns the new length
        let numbers = [3, 4, 5];
        const len = numbers.unshift(1, 2);
        log(`Numbers: [${numbers}], Length: ${len}`);
        
        // Building a stack-like structure (unshift + pop = stack)
        const stack = [];
        stack.unshift('first');
        stack.unshift('second');
        stack.unshift('third');
        log(`Stack: [${stack}]`);
        const popped = stack.pop();
        log(`Popped: ${popped}, Stack: [${stack}]`);

        const queue = [];
        queue.push('task1');
        queue.push('task2');
        queue.unshift('urgent');
        log(`Queue: [${queue}]`);
        
        // Real-world example: adding priority items
        const tasks = [
            { id: 2, task: 'Write docs' },
            { id: 3, task: 'Review code' }
        ];
        
        // Add urgent task at beginning
        tasks.unshift({ id: 1, task: 'Fix critical bug', urgent: true });
        log(`Tasks: ${JSON.stringify(tasks)}`);
        
        // Prepending with spread operator
        const existing = [2, 3, 4];
        const combined = [1, ...existing];
        log(`Spread prepend: [${combined}]`);
        
        // Performance note: unshift is O(n)
        function benchmarkUnshift(size) {
            const arr = [];
            const start = performance.now();
            
            for (let i = 0; i < size; i++) {
                arr.unshift(i);
            }
            
            const end = performance.now();
            return (end - start).toFixed(2);
        }
        
        // Faster alternative for large arrays: push + reverse
        function benchmarkPushReverse(size) {
            const arr = [];
            const start = performance.now();
            
            for (let i = 0; i < size; i++) {
                arr.push(i);
            }
            arr.reverse();
            
            const end = performance.now();
            return (end - start).toFixed(2);
        }
        
        log(`unshift(1000): ${benchmarkUnshift(1000)}ms`);
        log(`push+reverse(1000): ${benchmarkPushReverse(1000)}ms`);
        
        // unshift with array-like objects
        function logArguments() {
            // Arguments is array-like, not a real array
            // Convert and prepend
            const args = ['prefix', ...arguments];

        // Mutating vs non-mutating alternatives
        const original = [2, 3, 4];

        original.unshift(1);
        log(`After unshift: [${original}]`);
        
        // Non-mutating: spread (creates new array)
        const original2 = [2, 3, 4];
        const newArray = [1, ...original2];
        log(`Original: [${original2}], New: [${newArray}]`);
        
        // Using unshift in a loop to build reversed array
        const source = [1, 2, 3, 4, 5];
        const reversed = [];
        source.forEach(item => reversed.unshift(item));
        log(`Reversed using unshift: [${reversed}]`);
```

**Why**: `unshift()` adds elements to the start of an array, useful for prepending items, building stacks, or managing priority queues where new items take precedence.

```html
<!-- PROBLEM: How to convert an object into an array of key-value pairs -->
<!-- TIME: O(n) SPACE: O(n) -->

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>JavaScript Object.entries()</title>
</head>
<body>
    <h3>Object.entries() Demo</h3>
    <div id="output"></div>
    
    <script>
        const output = document.getElementById('output');
        
        function log(msg) {
            output.innerHTML += `<p>${msg}</p>`;
            console.log(msg);
        }
        
        // Basic Object.entries()
        const user = {
            name: 'Alice',
            age: 30,
            city: 'New York'
        };
        
        const entries = Object.entries(user);
        log(`Object: ${JSON.stringify(user)}`);
        log(`Entries: ${JSON.stringify(entries)}`);
        
        // Iterate over entries
        log('Iterating with for...of:');
        for (const [key, value] of Object.entries(user)) {
            log(`  ${key}: ${value}`);
        }
        
        // Iterate with forEach
        log('Iterating with forEach:');
        Object.entries(user).forEach(([key, value]) => {
            log(`  ${key} => ${value}`);
        });
        
        // Transform object values using entries
        const prices = {
            apple: 1.5,
            banana: 0.75,
            orange: 2.0
        };
        
        // Apply 10% discount
        const discounted = Object.fromEntries(
            Object.entries(prices).map(([item, price]) => [
                item, 
                (price * 0.9).toFixed(2)
            ])
        );
        log(`Original prices: ${JSON.stringify(prices)}`);
        log(`Discounted: ${JSON.stringify(discounted)}`);
        
        // Filter object properties
        const scores = { alice: 85, bob: 62, charlie: 90, david: 45 };
        const passing = Object.fromEntries(
            Object.entries(scores).filter(([name, score]) => score >= 70)
        );
        log(`Passing scores: ${JSON.stringify(passing)}`);
        
        // Sort object by values
        const sortedScores = Object.fromEntries(
            Object.entries(scores).sort((a, b) => b[1] - a[1])
        );
        log(`Sorted by score: ${JSON.stringify(sortedScores)}`);
        
        // Convert to Map
        const userMap = new Map(Object.entries(user));
        log(`Map from entries: ${JSON.stringify([...userMap])}`);
        
        // Create object from Map
        const map = new Map([['a', 1], ['b', 2], ['c', 3]]);
        const objFromMap = Object.fromEntries(map);
        log(`Object from Map: ${JSON.stringify(objFromMap)}`);
        
        // Query string builder
        function buildQueryString(params) {
            return Object.entries(params)
                .map(([key, value]) => 
                    `${encodeURIComponent(key)}=${encodeURIComponent(value)}`
                )
                .join('&');
        }
        
        const queryParams = { search: 'js', page: 1, limit: 10 };
        log(`Query string: ${buildQueryString(queryParams)}`);
        
        // Deep cloning with entries
        const original = { a: 1, b: { c: 2 } };
        const shallowClone = Object.fromEntries(Object.entries(original));
        log(`Shallow clone: ${JSON.stringify(shallowClone)}`);
        
        // Check if object is empty
        function isEmpty(obj) {
            return Object.entries(obj).length === 0;
        }
        
        log(`{} is empty: ${isEmpty({})}`);
        log(`{a:1} is empty: ${isEmpty({a: 1})}`);
        
        // Find key by value
        function findKey(obj, targetValue) {
            const entry = Object.entries(obj).find(([k, v]) => v === targetValue);
            return entry ? entry[0] : undefined;
        }
        
        const colors = { red: '#f00', green: '#0f0', blue: '#00f' };
        log(`Key for #0f0: ${findKey(colors, '#0f0')}`);
        
        // Merge objects with entries
        const obj1 = { a: 1, b: 2 };
        const obj2 = { b: 3, c: 4 };
        
        const merged = Object.fromEntries([
            ...Object.entries(obj1),
            ...Object.entries(obj2)
        ]);
        log(`Merged: ${JSON.stringify(merged)}`);
        
        // Count occurrences
        const words = ['apple', 'banana', 'apple', 'orange', 'banana', 'apple'];
        const wordCount = words.reduce((acc, word) => {
            acc[word] = (acc[word] || 0) + 1;
            return acc;
        }, {});
        
        log(`Word count: ${JSON.stringify(wordCount)}`);
        log(`As entries: ${JSON.stringify(Object.entries(wordCount))}`);
        
        // Stringify object with custom formatting
        function formatObject(obj, indent = 2) {
            return Object.entries(obj)
                .map(([k, v]) => ' '.repeat(indent) + `${k}: ${JSON.stringify(v)}`)
                .join('\n');
        }
        
        log(`Formatted:\n${formatObject(user)}`);
    </script>
</body>
</html>
```

**Why**: `Object.entries()` bridges objects and arrays, enabling functional transformations, iterations, and conversions essential for data manipulation and API interactions.
