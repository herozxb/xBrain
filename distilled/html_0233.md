## html_meta_twitter_site

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <!-- Twitter Card Meta Tags -->
    <!-- PROBLEM: Configure Twitter Card with site attribution -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <!-- Twitter site username - must start with @ -->
    <meta name="twitter:site" content="@yourusername">
    
    <!-- Required Twitter Card type -->
    <meta name="twitter:card" content="summary_large_image">
    
    <title>Twitter Site Meta Tag Example</title>
</head>
<body>
    <h1>Twitter Site Configuration</h1>
    <p>This page demonstrates the twitter:site meta tag.</p>
    
    <script>
        // Verify twitter:site meta tag is present
        // TIME: O(1) SPACE: O(1)
        
        function checkTwitterSite() {
            const metaTag = document.querySelector('meta[name="twitter:site"]');
            if (metaTag) {
                const siteValue = metaTag.getAttribute('content');
                console.log(`Twitter site configured: ${siteValue}`);
                
                // Validate format (should start with @)
                if (!siteValue.startsWith('@')) {
                    console.warn('Warning: twitter:site should start with @');
                }
                
                return siteValue;
            }
            console.error('twitter:site meta tag not found');
            return null;
        }
        
        // Usage
        checkTwitterSite();
    </script>
</body>
</html>
```

**Why**: The `twitter:site` meta tag identifies the Twitter account associated with the website, enabling proper attribution when content is shared on Twitter. It's essential for Twitter Card integration and helps users discover the official account.

## html_meta_twitter_creator

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <!-- Twitter Creator Meta Tags -->
    <!-- PROBLEM: Attribute content to specific author on Twitter -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <meta name="twitter:card" content="summary">
    <meta name="twitter:site" content="@website">
    
    <!-- Individual content creator's Twitter username -->
    <meta name="twitter:creator" content="@authorname">

</head>
<body>
    <article>
        <h1>Article Title</h1>
        <p>Written by a specific author with Twitter attribution.</p>
    </article>
    
    <script>
        // Extract and validate Twitter creator information
        // TIME: O(1) SPACE: O(1)
        
        function getTwitterCreator() {
            const creatorMeta = document.querySelector('meta[name="twitter:creator"]');
            
            if (creatorMeta) {
                const creator = creatorMeta.getAttribute('content');
                
                // Validate format
                const isValid = /^@[\w]{1,15}$/.test(creator);
                
                if (isValid) {
                    console.log(`Content creator: ${creator}`);
                    return {
                        username: creator,
                        profileUrl: `https://twitter.com/${creator.substring(1)}`
                    };
                } else {
                    console.warn('Invalid Twitter username format');
                }
            }
            
            return null;
        }
        
        // Usage
        const creatorInfo = getTwitterCreator();
        if (creatorInfo) {
            console.log('Creator Profile:', creatorInfo.profileUrl);
        }
    </script>
</body>
</html>
```

**Why**: The `twitter:creator` meta tag attributes individual pieces of content to their specific author's Twitter account, distinct from the overall website account. This is crucial for multi-author blogs and publications to give proper credit to individual creators.

## html_meta_twitter_image

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <!-- Twitter Image Meta Tags -->
    <!-- PROBLEM: Specify image for Twitter Card preview -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <meta name="twitter:card" content="summary_large_image">
    <meta name="twitter:site" content="@yourhandle">
    <meta name="twitter:title" content="Article Title">
    <meta name="twitter:description" content="Article description">
    
    <!-- Twitter image URL (must be absolute path) -->
    <meta name="twitter:image" content="https://example.com/images/preview.jpg">
    
    <!-- Optional: Image alt text for accessibility -->
    <meta name="twitter:image:alt" content="Descriptive text about the image">
    
    <title>Twitter Image Meta Tag Example</title>
</head>
<body>
    <h1>Twitter Image Configuration</h1>
    
    <script>
        // Validate Twitter image configuration
        // TIME: O(1) SPACE: O(1)
        
        function validateTwitterImage() {
            const imageMeta = document.querySelector('meta[name="twitter:image"]');
            const altMeta = document.querySelector('meta[name="twitter:image:alt"]');
            
            const config = {
                image: null,
                alt: null,
                valid: false,
                issues: []
            };
            
            // Check image meta tag
            if (!imageMeta) {
                config.issues.push('Missing twitter:image meta tag');
            } else {
                const imageUrl = imageMeta.getAttribute('content');
                config.image = imageUrl;
                
                // Validate URL format
                try {
                    const url = new URL(imageUrl);
                    
                    // Check protocol
                    if (url.protocol !== 'https:' && url.protocol !== 'http:') {
                        config.issues.push('Image URL must use http or https');
                    }
                    
                    // Check file extension
                    const validExtensions = ['.jpg', '.jpeg', '.png', '.gif', '.webp'];
                    const hasValidExt = validExtensions.some(ext => 
                        imageUrl.toLowerCase().endsWith(ext)
                    );
                    
                    if (!hasValidExt) {
                        config.issues.push('Image should be JPG, PNG, GIF, or WebP');
                    }
                    
                    console.log('Image URL validated:', url.href);
                    
                } catch (e) {
                    config.issues.push('Invalid image URL format');
                }
            }
            
            // Check alt text
            if (!altMeta) {
                config.issues.push('Missing twitter:image:alt (recommended for accessibility)');
            } else {
                config.alt = altMeta.getAttribute('content');
                
                if (config.alt.length < 10) {
                    config.issues.push('Alt text should be more descriptive (min 10 chars)');
                }
            }
            
            // Recommended image sizes
            const recommendations = {
                summary: '1200x628px',
                summary_large_image: '1200x675px (16:9 ratio)',
                maxFileSize: '5MB'
            };
            
            config.recommendations = recommendations;
            config.valid = config.issues.length === 0;
            
            console.log('Twitter Image Config:', config);
            
            return config;
        }
        
        // Preload image to check if it loads
        function preloadTwitterImage(url) {
            return new Promise((resolve, reject) => {
                const img = new Image();
                img.onload = () => resolve({ 
                    loaded: true, 
                    width: img.width, 
                    height: img.height 
                });
                img.onerror = () => reject(new Error('Image failed to load'));
                img.src = url;
            });
        }
        
        // Usage
        const imageConfig = validateTwitterImage();
        
        if (imageConfig.image) {
            preloadTwitterImage(imageConfig.image)
                .then(result => {
                    console.log('Image dimensions:', result.width, 'x', result.height);
                    
                    // Check aspect ratio for summary_large_image (should be 16:9)
                    const ratio = result.width / result.height;
                    const expectedRatio = 16 / 9;
                    
                    if (Math.abs(ratio - expectedRatio) < 0.1) {
                        console.log('✓ Good aspect ratio for summary_large_image');
                    }
                })
                .catch(err => console.error(err));
        }
    </script>
</body>
</html>
```

**Why**: The `twitter:image` meta tag specifies the preview image shown in Twitter Cards, which significantly impacts engagement rates. Images should be high-quality, properly sized (1200x675px for large cards), and use absolute URLs to ensure they display correctly when shared.

## html_meta_canonical

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <!-- Canonical URL Meta Tags -->
    <!-- PROBLEM: Specify preferred URL to avoid duplicate content issues -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <!-- Canonical URL - the preferred version of this page -->
    <link rel="canonical" href="https://example.com/blog/best-practices">
    
    <!-- Related Open Graph URL (should match canonical) -->
    <meta property="og:url" content="https://example.com/blog/best-practices">
    
    <title>Best Practices Guide</title>
</head>
<body>
    <article>
        <h1>SEO Best Practices</h1>
        <p>Understanding canonical URLs is crucial for SEO.</p>
    </article>
    
    <script>
        // Validate canonical URL configuration
        // TIME: O(1) SPACE: O(1)
        
        function validateCanonicalUrl() {
            const canonicalLink = document.querySelector('link[rel="canonical"]');
            const ogUrl = document.querySelector('meta[property="og:url"]');
            
            const analysis = {
                canonical: null,
                ogUrl: null,
                matches: false,
                issues: []
            };
            
            // Check canonical link
            if (!canonicalLink) {
                analysis.issues.push('Missing canonical URL - SEO risk: duplicate content');
            } else {
                analysis.canonical = canonicalLink.getAttribute('href');
                
                // Validate URL
                try {
                    const url = new URL(analysis.canonical);
                    
                    // Should be absolute URL
                    if (!url.protocol || !url.host) {
                        analysis.issues.push('Canonical URL should be absolute');
                    }
                    
                    // Should match current page domain (usually)
                    if (url.hostname !== window.location.hostname) {
                        console.warn('Canonical URL points to different domain');
                    }
                    
                    // Check for common mistakes
                    if (analysis.canonical.includes('?') || analysis.canonical.includes('&')) {
                        analysis.issues.push('Canonical URL should not include query parameters');
                    }
                    
                } catch (e) {
                    analysis.issues.push('Invalid canonical URL format');
                }
            }
            
            // Check Open Graph URL
            if (ogUrl) {
                analysis.ogUrl = ogUrl.getAttribute('content');
                analysis.matches = analysis.canonical === analysis.ogUrl;
                
                if (!analysis.matches) {
                    analysis.issues.push('Canonical URL and og:url should match');
                }
            }
            
            // Compare with current page URL
            if (analysis.canonical) {
                const currentUrl = window.location.href.split('?')[0].split('#')[0];
                const canonicalClean = analysis.canonical.split('?')[0].split('#')[0];
                
                if (currentUrl !== canonicalClean) {
                    console.log('Canonical differs from current URL (this may be intentional)');
                    console.log('Current:', currentUrl);
                    console.log('Canonical:', canonicalClean);
                }
            }
            
            console.log('Canonical Analysis:', analysis);
            
            return analysis;
        }
        
        // Function to get all URL variants that should canonicalize to same page
        function getUrlVariants() {
            const variants = [
                window.location.href,
                window.location.origin + window.location.pathname,
                window.location.href.replace('http://', 'https://'),
                window.location.href.replace('www.', ''),
            ];
            
            console.log('URL variants that should have same canonical:');
            variants.forEach((url, i) => {
                console.log(`${i + 1}. ${url}`);
            });
            
            return variants;
        }
        
        // Usage
        validateCanonicalUrl();
        getUrlVariants();
    </script>
</body>
</html>
```

**Why**: The canonical URL tells search engines which version of a page is the "master" copy, preventing duplicate content penalties when the same content is accessible through multiple URLs. This is essential for proper SEO and ensures link equity is consolidated to one URL.

## html_meta_refresh_redirect

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <!-- Meta Refresh Redirect -->
    <!-- PROBLEM: Redirect users after a delay or immediately -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <!-- Redirect after 5 seconds -->
    <meta http-equiv="refresh" content="5; url=https://example.com/new-page">
    
    <!-- Alternative: Immediate redirect (0 seconds) -->
    <!-- <meta http-equiv="refresh" content="0; url=https://example.com/new-page"> -->
    
    <!-- Also set canonical to destination for SEO -->
    <link rel="canonical" href="https://example.com/new-page">
    
    <title>Page Moving...</title>
    
    <style>
        body {
            font-family: Arial, sans-serif;
            text-align: center;
            padding: 50px;
        }
        .countdown {
            font-size: 48px;
            color: #007bff;
            font-weight: bold;
        }
        .progress-bar {
            width: 100%;
            max-width: 400px;
            height: 4px;
            background: #e0e0e0;
            margin: 20px auto;
            border-radius: 2px;
            overflow: hidden;
        }
        .progress-fill {
            height: 100%;
            background: #007bff;
            width: 0%;
            transition: width 1s linear;
        }
        .manual-link {
            margin-top: 20px;
            display: block;
        }
    </style>
</head>
<body>
    <h1>This page has moved</h1>
    <p>You will be redirected automatically in <span class="countdown" id="countdown">5</span> seconds.</p>
    
    <div class="progress-bar">
        <div class="progress-fill" id="progress"></div>
    </div>
    
    <a class="manual-link" href="https://example.com/new-page">Click here if you are not redirected</a>
    
    <script>
        // Enhanced redirect with countdown and progress
        // TIME: O(1) SPACE: O(1)
        
        (function() {
            const redirectUrl = 'https://example.com/new-page';
            const delaySeconds = 5;
            let remaining = delaySeconds;
            
            const countdownEl = document.getElementById('countdown');
            const progressEl = document.getElementById('progress');
            
            // Update countdown every second
            // TIME: O(1) SPACE: O(1) per interval
            
            const interval = setInterval(() => {
                remaining--;
                countdownEl.textContent = remaining;
                
                // Update progress bar
                const progress = ((delaySeconds - remaining) / delaySeconds) * 100;
                progressEl.style.width = progress + '%';
                
                if (remaining <= 0) {
                    clearInterval(interval);
                }
            }, 1000);
            
            // Animate progress bar smoothly
            progressEl.style.transition = `width ${delaySeconds}s linear`;
            progressEl.style.width = '100%';
            
            // Fallback: Ensure redirect happens even if meta refresh fails
            setTimeout(() => {
                window.location.href = redirectUrl;
            }, delaySeconds * 1000);
            
            // Log redirect information
            console.log('Redirect scheduled:', {
                url: redirectUrl,
                delay: delaySeconds + 's',
                timestamp: new Date().toISOString()
            });
        })();
        
        // Function to parse meta refresh content
        function parseMetaRefresh() {
            const metaRefresh = document.querySelector('meta[http-equiv="refresh"]');
            
            if (!metaRefresh) {
                return null;
            }
            
            const content = metaRefresh.getAttribute('content');
            
            // Parse format: "seconds; url=URL"
            const match = content.match(/^(\d+);\s*url=(.+)$/i);
            
            if (match) {
                return {
                    delay: parseInt(match[1], 10),
                    url: match[2].trim()
                };
            }
            
            // Just a delay without URL (refresh same page)
            const delayOnly = content.match(/^(\d+)$/);
            if (delayOnly) {
                return {
                    delay: parseInt(delayOnly[1], 10),
                    url: window.location.href
                };
            }
            
            return null;
        }
        
        // Usage
        const redirectInfo = parseMetaRefresh();
        console.log('Meta refresh config:', redirectInfo);
        
        // Warning about SEO implications
        console.warn('Note: Meta refresh redirects are not recommended for SEO.');
        console.warn('Use server-side 301/302 redirects instead when possible.');
    </script>
</body>
</html>
```

**Why**: The meta refresh redirect provides a client-side redirect mechanism, useful for temporary redirects or when server-side redirects aren't available. However, it should be used sparingly as search engines prefer server-side 301/302 redirects for SEO purposes. Always provide a manual link as a fallback.

## html_meta_theme_color

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <!-- Theme Color Meta Tags -->
    <!-- PROBLEM: Customize browser UI color to match brand -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <!-- Default theme color -->
    <meta name="theme-color" content="#4285f4">
    
    <!-- Theme color for dark mode -->
    <meta name="theme-color" content="#1a1a2e" media="(prefers-color-scheme: dark)">
    
    <!-- Theme color for light mode -->
    <meta name="theme-color" content="#4285f4" media="(prefers-color-scheme: light)">
    
    <!-- Apple Safari pinned tab color -->
    <meta name="apple-mobile-web-app-status-bar-style" content="black-translucent">
    
    <!-- Windows Tiles -->
    <meta name="msapplication-TileColor" content="#4285f4">
    
    <title>Theme Color Example</title>
    
    <style>
        :root {
            --theme-primary: #4285f4;
            --theme-dark: #1a1a2e;
        }
        
        body {
            margin: 0;
            padding: 20px;
            font-family: system-ui, -apple-system, sans-serif;
            transition: background-color 0.3s ease;
        }
        
        @media (prefers-color-scheme: dark) {
            body {
                background-color: var(--theme-dark);
                color: white;
            }
        }
        
        @media (prefers-color-scheme: light) {
            body {
                background-color: white;
                color: black;
            }
        }
        
        .color-preview {
            width: 100px;
            height: 100px;
            border-radius: 8px;
            margin: 10px;
            display: inline-block;
            box-shadow: 0 2px 8px rgba(0,0,0,0.1);
        }
    </style>
</head>
<body>
    <h1>Theme Color Configuration</h1>
    <p>The browser UI color adapts to match your brand.</p>
    
    <div>
        <div class="color-preview" id="lightColor"></div>
        <div class="color-preview" id="darkColor"></div>
    </div>
    
    <button id="toggleTheme">Toggle Theme Color</button>
    
    <script>
        // Dynamic theme color management
        // TIME: O(1) SPACE: O(1)
        
        class ThemeColorManager {
            constructor() {
                this.themeMeta = document.querySelector('meta[name="theme-color"]');
                this.isDark = window.matchMedia('(prefers-color-scheme: dark)').matches;
                
                this.colors = {
                    light: '#4285f4',
                    dark: '#1a1a2e'
                };
                
                this.init();
            }
            
            init() {
                // Set initial theme color
                this.updateThemeColor();
                
                // Listen for system theme changes
                const darkModeQuery = window.matchMedia('(prefers-color-scheme: dark)');
                darkModeQuery.addEventListener('change', (e) => {
                    this.isDark = e.matches;
                    this.updateThemeColor();
                });
                
                // Update color previews
                this.updateColorPreviews();
            }
            
            // Update the theme-color meta tag
            updateThemeColor() {
                const color = this.isDark ? this.colors.dark : this.colors.light;
                this.themeMeta.setAttribute('content', color);
                console.log(`Theme color updated: ${color} (${this.isDark ? 'dark' : 'light'} mode)`);
            }
            
            // Set custom theme color
            setThemeColor(color, mode = null) {
                if (mode === 'dark') {
                    this.colors.dark = color;
                } else if (mode === 'light') {
                    this.colors.light = color;
                } else {
                    // Set for current mode
                    if (this.isDark) {
                        this.colors.dark = color;
                    } else {
                        this.colors.light = color;
                    }
                }
                this.updateThemeColor();
            }
            
            // Toggle between light and dark
            toggle() {
                this.isDark = !this.isDark;
                this.updateThemeColor();
                this.updateColorPreviews();
            }
            
            // Update visual previews
            updateColorPreviews() {
                document.getElementById('lightColor').style.backgroundColor = this.colors.light;
                document.getElementById('darkColor').style.backgroundColor = this.colors.dark;
            }
            
            // Validate color format
            static validateColor(color) {
                // Check hex format
                if (/^#([0-9A-F]{3}){1,2}$/i.test(color)) {
                    return { valid: true, format: 'hex' };
                }
                
                // Check rgb/rgba format
                if (/^rgba?\(\s*\d+\s*,\s*\d+\s*,\s*\d+\s*(,\s*[\d.]+\s*)?\)$/i.test(color)) {
                    return { valid: true, format: 'rgb' };
                }
                
                return { valid: false, format: null };
            }
        }
        
        // Initialize theme manager
        const themeManager = new ThemeColorManager();
        
        // Toggle button
        document.getElementById('toggleTheme').addEventListener('click', () => {
            themeManager.toggle();
        });
        
        // Example: Set custom colors
        // themeManager.setThemeColor('#ff6b6b', 'light');
        // themeManager.setThemeColor('#2d1b69', 'dark');
        
        // Validate a color
        console.log('Color validation:', ThemeColorManager.validateColor('#4285f4'));
    </script>
</body>
</html>
```

**Why**: The `theme-color` meta tag customizes the browser's UI elements (address bar, status bar) to match your brand colors, creating a more immersive app-like experience. Supporting both light and dark modes ensures your brand remains consistent regardless of user preference.

        // Icon configuration analyzer and validator
        // TIME: O(n) SPACE: O(n) where n = number of icon links
        
        class IconManager {
            constructor() {
                this.icons = this.collectIcons();
            }
            
            // Collect all icon link elements
            collectIcons() {
                const icons = [];
                
                // Regular icons
                document.querySelectorAll('link[rel="icon"]').forEach(link => {
                    icons.push({
                        type: 'icon',
                        href: link.href,
                        sizes: link.getAttribute('sizes') || 'any',
                        mimeType: link.type
                    });
                });
                
                // Apple touch icons
                document.querySelectorAll('link[rel="apple-touch-icon"]').forEach(link => {
                    icons.push({
                        type: 'apple-touch-icon',
                        href: link.href,
                        sizes: link.getAttribute('sizes') || 'unknown',
                        mimeType: link.type
                    });
                });
                
                // Check for manifest
                const manifest = document.querySelector('link[rel="manifest"]');
                if (manifest) {
                    icons.push({
                        type: 'manifest',
                        href: manifest.href,
                        sizes: 'various',
                        mimeType: 'application/json'
                    });
                }
                
                return icons;
            }
            
            // Validate icon configuration
            validate() {
                const issues = [];
                const recommendations = [];
                
                // Check for essential icons
                const hasFavicon = this.icons.some(i => i.type === 'icon' && i.sizes.includes('32'));
                const hasAppleIcon = this.icons.some(i => i.type === 'apple-touch-icon');
                const hasManifest = this.icons.some(i => i.type === 'manifest');
                const hasSvg = this.icons.some(i => i.mimeType === 'image/svg+xml');
                
                if (!hasFavicon) {
                    issues.push('Missing 32x32 favicon');
                }
                
                if (!hasAppleIcon) {
                    recommendations.push('Add apple-touch-icon for iOS devices');
                }
                
                if (!hasManifest) {
                    recommendations.push('Add manifest.json for Android/PWA support');
                }
                
                if (!hasSvg) {
                    recommendations.push('Consider adding SVG favicon for scalability');
                }
                
                // Check for ico fallback
                const hasIco = this.icons.some(i => i.mimeType === 'image/x-icon');
                if (!hasIco) {
                    recommendations.push('Add favicon.ico for legacy browser support');
                }
                
                return {
                    valid: issues.length === 0,
                    issues,
                    recommendations,
                    iconCount: this.icons.length
                };
            }
            
            // Display icon previews
            displayPreviews(containerId) {
                const container = document.getElementById(containerId);
                
                this.icons.forEach(icon => {
                    if (icon.type === 'manifest') {
                        return; // Skip manifest in preview
                    }
                    
                    const preview = document.createElement('div');
                    preview.className = 'icon-preview';
                    
                    const img = document.createElement('img');
                    img.src = icon.href;
                    img.width = 32;
                    img.height = 32;
                    img.onerror = () => {
                        img.style.display = 'none';
                    };
                    
                    const info = document.createElement('div');
                    info.className = 'icon-info';
                    info.innerHTML = `
                        <strong>${icon.type}</strong><br>
                        Size: ${icon.sizes}<br>
                        Type: ${icon.mimeType || 'unknown'}
                    `;
                    
                    preview.appendChild(img);
                    preview.appendChild(info);
                    container.appendChild(preview);
                });
            }
            
            // Generate icon HTML for given sizes
            static generateIconHtml(sizes = [16, 32, 180]) {
                return sizes.map(size => {
                    if (size === 180) {
                        return `<link rel="apple-touch-icon" sizes="${size}x${size}" href="/icons/apple-touch-icon.png">`;
                    }
                    return `<link rel="icon" type="image/png" sizes="${size}x${size}" href="/icons/favicon-${size}x${size}.png">`;
                }).join('\n');
            }
        }
        
        // Initialize icon manager
        const iconManager = new IconManager();
        
        // Validate configuration
        const validation = iconManager.validate();
        console.log('Icon Configuration:', validation);
        
        // Display previews
        iconManager.displayPreviews('iconPreviews');
        
        // Generate recommended HTML
        console.log('Recommended icon HTML:');
        console.log(IconManager.generateIconHtml([16, 32, 180, 192, 512]));
        
        // Preload icons to verify they exist
        function preloadIcon(url) {
            return new Promise((resolve, reject) => {
                const img = new Image();
                img.onload = () => resolve({ url, loaded: true });
                img.onerror = () => reject({ url, loaded: false });
                img.src = url;
            });
        }
        
        // Check all icons
        Promise.allSettled(
            iconManager.icons
                .filter(i => i.type !== 'manifest')
                .map(i => preloadIcon(i.href))
        ).then(results => {
            results.forEach(result => {
                if (result.status === 'fulfilled') {
                    console.log('✓ Icon loaded:', result.value.url);
                } else {
                    console.error('✗ Icon failed:', result.reason.url);
                }
            });
        });
    </script>

<body>
    <h1>Web App Manifest Configuration</h1>
    
    <div id="installPrompt" class="install-prompt">
        <p>Install this app on your device for a better experience!</p>
        <button id="installBtn">Install</button>
        <button id="dismissBtn">Not now</button>
    </div>
    
    <div id="manifestInfo"></div>
    
    <script>
        // Web App Manifest manager
        // TIME: O(1) SPACE: O(1) for most operations
        
        class ManifestManager {
            constructor() {
                this.manifestUrl = this.getManifestUrl();
                this.manifest = null;
                this.deferredPrompt = null;
                
                this.init();
            }
            
            async init() {
                // Load and parse manifest
                await this.loadManifest();
                
                // Setup install prompt
                this.setupInstallPrompt();
                
                // Display manifest info
                this.displayManifestInfo();
            }
            
            // Get manifest URL from link tag
            getManifestUrl() {
                const manifestLink = document.querySelector('link[rel="manifest"]');
                return manifestLink ? manifestLink.href : null;
            }
            
            // Load and parse manifest file
            async loadManifest() {
                if (!this.manifestUrl) {
                    console.error('No manifest link found');
                    return null;
                }
                
                try {
                    const response = await fetch(this.manifestUrl);
                    this.manifest = await response.json();
                    console.log('Manifest loaded:', this.manifest);
                    return this.manifest;
                } catch (error) {
                    console.error('Failed to load manifest:', error);
                    return null;
                }
            }
            
            // Validate manifest structure
            validate() {
                if (!this.manifest) {
                    return { valid: false, errors: ['Manifest not loaded'] };
                }
                
                const errors = [];
                const warnings = [];
                
                // Required fields
                if (!this.manifest.name) {
                    errors.push('Missing required field: name');
                }
                
                if (!this.manifest.short_name) {
                    warnings.push('Recommended field missing: short_name');
                }
                
                if (!this.manifest.start_url) {
                    errors.push('Missing required field: start_url');
                }
                
                if (!this.manifest.display) {
                    warnings.push('Recommended field missing: display');
                }
                
                // Check icons
                if (!this.manifest.icons || this.manifest.icons.length === 0) {
                    errors.push('No icons defined');
                } else {
                    // Check for required icon sizes
                    const sizes = this.manifest.icons.map(i => i.sizes);
                    const has192 = sizes.some(s => s.includes('192'));
                    const has512 = sizes.some(s => s.includes('512'));
                    
                    if (!has192) {
                        warnings.push('Missing 192x192 icon (required for Android)');
                    }
                    if (!has512) {
                        warnings.push('Missing 512x512 icon (required for splash screen)');
                    }
                }
                
                return {
                    valid: errors.length === 0,
                    errors,
                    warnings
                };
            }
            
            // Setup PWA install prompt
            setupInstallPrompt() {
                // Listen for beforeinstallprompt event
                window.addEventListener('beforeinstallprompt', (e) => {
                    // Prevent Chrome 67+ from automatically showing the prompt
                    e.preventDefault();
                    
                    // Stash the event so it can be triggered later
                    this.deferredPrompt = e;
                    
                    // Show install prompt
                    document.getElementById('installPrompt').classList.add('show');
                });
                
                // Install button click handler
                document.getElementById('installBtn').addEventListener('click', async () => {
                    if (!this.deferredPrompt) {
                        console.log('Install prompt not available');
                        return;
                    }
                    
                    // Show the prompt
                    this.deferredPrompt.prompt();
                    
                    // Wait for user response
                    const { outcome } = await this.deferredPrompt.userChoice;
                    console.log(`User response: ${outcome}`);
                    
                    // Clear the deferred prompt
                    this.deferredPrompt = null;
                    
                    // Hide the install prompt
                    document.getElementById('installPrompt').classList.remove('show');
                });
                
                // Dismiss button
                document.getElementById('dismissBtn').addEventListener('click', () => {
                    document.getElementById('installPrompt').classList.remove('show');
                });
                
                // Listen for successful installation
                window.addEventListener('appinstalled', () => {
                    console.log('App installed successfully!');
                    this.deferredPrompt = null;
                });
            }
            
            // Display manifest information
            displayManifestInfo() {
                const container = document.getElementById('manifestInfo');
                
                if (!this.manifest) {
                    container.innerHTML = '<p>Manifest not loaded</p>';
                    return;
                }
                
                const validation = this.validate();
                
                let html = '<h2>Manifest Details</h2>';
                html += '<ul>';
                html += `<li><strong>Name:</strong> ${this.manifest.name || 'Not set'}</li>`;
                html += `<li><strong>Short Name:</strong> ${this.manifest.short_name || 'Not set'}</li>`;
                html += `<li><strong>Start URL:</strong> ${this.manifest.start_url || 'Not set'}</li>`;
                html += `<li><strong>Display:</strong> ${this.manifest.display || 'Not set'}</li>`;
                html += `<li><strong>Orientation:</strong> ${this.manifest.orientation || 'Not set'}</li>`;
                html += `<li><strong>Theme Color:</strong> ${this.manifest.theme_color || 'Not set'}</li>`;
                html += `<li><strong>Background Color:</strong> ${this.manifest.background_color || 'Not set'}</li>`;
                html += '</ul>';
                
                // Icons
                if (this.manifest.icons) {
                    html += '<h3>Icons</h3><ul>';
                    this.manifest.icons.forEach(icon => {
                        html += `<li>${icon.sizes} - ${icon.src} (${icon.type || 'unknown'})</li>`;
                    });
                    html += '</ul>';
                }
                
                // Validation results
                html += '<h3>Validation</h3>';
                html += `<p>Status: ${validation.valid ? '✓ Valid' : '✗ Invalid'}</p>`;
                
                if (validation.errors.length > 0) {
                    html += '<p style="color: red;">Errors:</p><ul>';
                    validation.errors.forEach(e => {
                        html += `<li>${e}</li>`;
                    });
                    html += '</ul>';
                }
                
                if (validation.warnings.length > 0) {
                    html += '<p style="color: orange;">Warnings:</p><ul>';
                    validation.warnings.forEach(w => {
                        html += `<li>${w}</li>`;
                    });
                    html += '</ul>';
                }
                
                container.innerHTML = html;
            }
            
            // Generate manifest.json content
            static generateManifest(options = {}) {
                const defaults = {
                    name: 'My App',
                    short_name: 'App',
                    description: 'A Progressive Web App',
                    start_url: '/',
                    display: 'standalone',
                    background_color: '#ffffff',
                    theme_color: '#4285f4',
                    icons: [
                        { src: '/icons/icon-192.png', sizes: '192x192', type: 'image/png' },
                        { src: '/icons/icon-512.png', sizes: '512x512', type: 'image/png' }
                    ]
                };
                
                return { ...defaults, ...options };
            }
        }
        
        // Initialize manifest manager
        const manifestManager = new ManifestManager();
        
        // Generate sample manifest for reference
        console.log('Sample manifest.json:');
        console.log(JSON.stringify(ManifestManager.generateManifest(), null, 2));
        
        // Check if app is running as PWA
        function isPwa() {
            return window.matchMedia('(display-mode: standalone)').matches ||
                   window.navigator.standalone === true;
        }
        
        console.log('Running as PWA:', isPwa());
    </script>
</body>
</html>

**Why**: The web app manifest enables Progressive Web App features like installation to home screen, app-like fullscreen display, and splash screens. It's essential for creating a native app-like experience on mobile devices and improves user engagement through easy access.
