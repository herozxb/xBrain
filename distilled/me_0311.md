## [Multiton Pattern]
```python
# PROBLEM: Ensure a class has only named instances, providing global access
# TIME: O(1) SPACE: O(n) where n is number of named instances
class Multiton:
    """Multiton pattern: one instance per key."""
    _instances = {}
    
    def __new__(cls, key: str):
        # Return existing instance if key exists
        if key not in cls._instances:
            cls._instances[key] = super().__new__(cls)
        return cls._instances[key]
    
    def __init__(self, key: str):
        # Guard against re-initialization
        if not hasattr(self, '_initialized'):
            self.key = key
            self._initialized = True

# Usage: Multiton("db1") is Multiton("db1")  # True
```
**Why**: Useful when you need multiple singleton-like instances identified by keys (e.g., multiple database connections, logger channels).

## [Retry Pattern]

```python
# PROBLEM: Automatically retry failed operations with exponential backoff
# TIME: O(n) where n is max retries SPACE: O(1)
import time
import random
from functools import wraps

def retry(max_attempts: int = 3, base_delay: float = 1.0, 
          backoff_factor: float = 2.0, jitter: bool = True):
    """Decorator that retries a function on failure with exponential backoff."""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    last_exception = e
                    if attempt < max_attempts - 1:
                        # Calculate delay with exponential backoff
                        delay = base_delay * (backoff_factor ** attempt)
                        # Add jitter to prevent thundering herd
                        if jitter:
                            delay *= random.uniform(0.5, 1.5)
                        time.sleep(delay)
            raise last_exception  # Re-raise last exception
        return wrapper
    return decorator

@retry(max_attempts=3, base_delay=0.5)
def fetch_data(url: str):
    # Network operation that may fail
    pass
```
**Why**: Essential for distributed systems where transient failures are common. Exponential backoff prevents overwhelming struggling services.

## [Cache-Aside Pattern]
```python
# PROBLEM: Maintain consistency between cache and database in read-heavy systems
# TIME: O(1) for cache, O(query) for db SPACE: O(cache_size)
from typing import Any, Optional, Callable

class CacheAside:
    """Cache-Aside: Application manages cache and database explicitly."""
    
    def __init__(self, cache_get: Callable, cache_set: Callable, 
                 cache_delete: Callable, db_read: Callable, db_write: Callable):
        self.cache_get = cache_get
        self.cache_set = cache_set
        self.cache_delete = cache_delete
        self.db_read = db_read
        self.db_write = db_write
    
    def get(self, key: str) -> Optional[Any]:
        """Read: Check cache first, fallback to database."""
        # 1. Try cache first
        value = self.cache_get(key)
        if value is not None:
            return value
        
        # 2. Cache miss - read from database
        value = self.db_read(key)
        if value is not None:
            # 3. Populate cache for future reads
            self.cache_set(key, value)
        return value
    
    def set(self, key: str, value: Any) -> None:
        """Write: Update database, then invalidate cache."""
        # 1. Write to database first
        self.db_write(key, value)
        # 2. Invalidate cache (or update it)
        self.cache_delete(key)
```
**Why**: Most common caching pattern. Application controls cache explicitly, ensuring consistency by invalidating cache on writes.

## [Weak References]

```python
# PROBLEM: Allow objects to be garbage collected while still referenced
# TIME: O(1) SPACE: O(1) per weak reference
import weakref
from typing import Optional

class ExpensiveObject:
    """Large object we want to cache but allow GC."""
    def __init__(self, data: str):
        self.data = data

class WeakRefCache:
    """Cache that doesn't prevent garbage collection."""
    
    def __init__(self):
        # WeakValueDictionary allows values to be GC'd
        self._cache = weakref.WeakValueDictionary()
    
    def get(self, key: str) -> Optional[ExpensiveObject]:
        """Get object from cache if still alive."""
        return self._cache.get(key)
    
    def set(self, key: str, obj: ExpensiveObject) -> None:
        """Store object; will be removed if no strong refs exist."""
        self._cache[key] = obj

# Usage: When all strong references are gone, object is GC'd automatically
cache = WeakRefCache()
obj = ExpensiveObject("data")
cache.set("key1", obj)
# obj can still be GC'd if we delete 'obj' variable
```
**Why**: Perfect for caches, observer patterns, and parent-child relationships where you don't want to prevent garbage collection.

## [Concurrent Futures - ThreadPool]
```python
# PROBLEM: Execute multiple I/O-bound tasks concurrently with thread pool
# TIME: O(max_task_time) for parallel execution SPACE: O(pool_size)
from concurrent.futures import ThreadPoolExecutor, as_completed
from typing import List, Any, Callable

def parallel_execute(tasks: List[Callable], max_workers: int = 4) -> List[Any]:
    """Execute tasks in parallel using thread pool."""
    results = []
    
    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        # Submit all tasks
        future_to_task = {executor.submit(task): task for task in tasks}
        
        # Process results as they complete
        for future in as_completed(future_to_task):
            task = future_to_task[future]
            try:
                result = future.result()
                results.append((task.__name__, result, None))
            except Exception as e:
                results.append((task.__name__, None, e))
    
    return results

# Example: Fetch multiple URLs concurrently
def fetch_url(url: str) -> str:
    import urllib.request
    return urllib.request.urlopen(url).read()

urls = ["http://example1.com", "http://example2.com"]
tasks = [lambda u=url: fetch_url(u) for url in urls]
results = parallel_execute(tasks, max_workers=4)
```
**Why**: Simple parallelism for I/O-bound tasks. Thread pool reuses threads, avoiding thread creation overhead.

```python
# PROBLEM: Demonstrate ACID properties in a simple transaction system
# TIME: O(n) for n operations SPACE: O(state_size)
from typing import Dict, Any, List
from copy import deepcopy

class ACIDTransaction:
    """Simulates ACID transaction properties."""
    
    def __init__(self, storage: Dict[str, Any]):
        self.storage = storage
        self._backup: Dict[str, Any] = {}
        self._operations: List[tuple] = []
        self._committed = False
    
    def begin(self):
        """Isolation: Create snapshot of current state."""
        self._backup = deepcopy(self.storage)
    
    def write(self, key: str, value: Any):
        """Atomic: Stage operation for commit."""
        self._operations.append(('write', key, value))
    
    def commit(self):
        """Durability: Apply all operations atomically."""
        if self._committed:
            raise RuntimeError("Transaction already committed")
        
        # Atomicity: Apply all or nothing
        try:
            for op_type, key, value in self._operations:
                self.storage[key] = value
            self._committed = True
        except Exception:
            # Rollback on failure
            self.rollback()
            raise
    
    def rollback(self):
        """Undo all changes to maintain consistency."""
        self.storage.clear()
        self.storage.update(self._backup)
        self._operations.clear()

# Usage
db = {"balance": 100}
txn = ACIDTransaction(db)
txn.begin()
txn.write("balance", 150)
txn.commit()  # db["balance"] == 150
```
**Why**: Understanding ACID is crucial for database design. This demonstrates Atomicity, Consistency, Isolation, and Durability principles.

## [Microservices Circuit Breaker]
```python
# PROBLEM: Prevent cascade failures in microservices by failing fast
# TIME: O(1) per request SPACE: O(1)
import time
from enum import Enum
from typing import Optional, Callable

class CircuitState(Enum):
    CLOSED = "closed"      # Normal operation
    OPEN = "open"          # Failing, reject all requests
    HALF_OPEN = "half_open"  # Testing if recovered

class CircuitBreaker:
    """Circuit breaker for microservice fault tolerance."""
    
    def __init__(self, failure_threshold: int = 5, 
                 recovery_timeout: float = 30.0,
                 fallback: Optional[Callable] = None):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.fallback = fallback
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time: Optional[float] = None
    
    def call(self, func: Callable, *args, **kwargs):
        """Execute function through circuit breaker."""
        if self.state == CircuitState.OPEN:
            # Check if recovery timeout elapsed
            if time.time() - self.last_failure_time > self.recovery_timeout:
                self.state = CircuitState.HALF_OPEN
            else:
                if self.fallback:
                    return self.fallback(*args, **kwargs)
                raise Exception("Circuit breaker is OPEN")
        
        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise
    
    def _on_success(self):
        """Reset on successful call."""
        self.failure_count = 0
        self.state = CircuitState.CLOSED
    
    def _on_failure(self):
        """Track failures, open circuit if threshold exceeded."""
        self.failure_count += 1
        self.last_failure_time = time.time()
        
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
```
**Why**: Essential pattern for microservices resilience. Prevents cascading failures by failing fast when downstream service is unhealthy.

## [Write-Behind Cache]
```python
# PROBLEM: Improve write performance by batching writes asynchronously
# TIME: O(1) for writes, O(batch) for flush SPACE: O(queue_size)
import threading
import queue
import time
from typing import Dict, Any, Callable

class WriteBehindCache:
    """Asynchronously batch writes to database."""
    
    def __init__(self, db_write: Callable[[Dict[str, Any]], None],
                 flush_interval: float = 1.0, batch_size: int = 100):
        self.cache: Dict[str, Any] = {}
        self.write_queue: queue.Queue = queue.Queue()
        self.db_write = db_write
        self.flush_interval = flush_interval
        self.batch_size = batch_size
        self._running = True
        self._worker = threading.Thread(target=self._flush_worker, daemon=True)
        self._worker.start()
    
    def get(self, key: str) -> Any:
        """Read from cache (cache-aside for reads)."""
        return self.cache.get(key)
    
    def set(self, key: str, value: Any) -> None:
        """Write to cache immediately, queue for async DB write."""
        self.cache[key] = value
        self.write_queue.put((key, value))
    
    def _flush_worker(self):
        """Background thread that batches writes to database."""
        batch: Dict[str, Any] = {}
        last_flush = time.time()
        
        while self._running:
            try:
                # Non-blocking get with timeout
                key, value = self.write_queue.get(timeout=0.1)
                batch[key] = value
                
                # Flush if batch full or interval elapsed
                should_flush = (
                    len(batch) >= self.batch_size or
                    time.time() - last_flush >= self.flush_interval
                )
                
                if should_flush:
                    self.db_write(batch)
                    batch.clear()
                    last_flush = time.time()
            except queue.Empty:
                # Flush remaining on timeout
                if batch:
                    self.db_write(batch)
                    batch.clear()
                    last_flush = time.time()
    
    def shutdown(self):
        """Gracefully stop worker and flush remaining."""
        self._running = False
        self._worker.join()
```
**Why**: Best for write-heavy workloads where eventual consistency is acceptable. Reduces database load by batching writes.
