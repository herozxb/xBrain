## 1. Bipartite Graph Check

```python
# PROBLEM: Determine if a graph is bipartite (can be colored with 2 colors such that
#          no adjacent nodes have the same color)
# TIME: O(V + E) SPACE: O(V)
from collections import deque
from typing import List

def is_bipartite(graph: List[List[int]]) -> bool:
    """
    Check if an undirected graph is bipartite using BFS.
    
    Args:
        graph: Adjacency list where graph[i] contains neighbors of node i
    
    Returns:
        True if the graph is bipartite, False otherwise
    """
    n = len(graph)
    # -1: uncolored, 0: color A, 1: color B
    color = [-1] * n
    
    # Handle disconnected components
    for start in range(n):
        if color[start] != -1:
            continue
        
        # BFS from this unvisited node
        queue = deque([start])
        color[start] = 0  # Assign first color
        
        while queue:
            node = queue.popleft()
            
            for neighbor in graph[node]:
                if color[neighbor] == -1:
                    # Color with opposite color
                    color[neighbor] = 1 - color[node]
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:
                    # Same color on adjacent nodes - not bipartite
                    return False
    
    return True
```
**Why**: Bipartite graphs are fundamental in matching problems, scheduling, and resource allocation. BFS coloring is the standard approach with linear time complexity.

## 2. Longest Palindromic Subsequence

```python
# PROBLEM: Find the length of the longest subsequence that is a palindrome
# TIME: O(n²) SPACE: O(n²)
from typing import List

def longest_palindromic_subsequence(s: str) -> int:
    """
    Find the length of the longest palindromic subsequence using DP.
    
    dp[i][j] = length of longest palindromic subsequence in s[i:j+1]
    
    Args:
        s: Input string
    
    Returns:
        Length of the longest palindromic subsequence
    """
    n = len(s)
    if n == 0:
        return 0
    
    # dp[i][j] represents LPS length for substring s[i..j]
    dp = [[0] * n for _ in range(n)]
    
    # Every single character is a palindrome of length 1
    for i in range(n):
        dp[i][i] = 1
    
    # Fill the table for substrings of length 2 to n
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            
            if s[i] == s[j]:
                if length == 2:
                    dp[i][j] = 2
                else:
                    dp[i][j] = dp[i + 1][j - 1] + 2
            else:
                dp[i][j] = max(dp[i + 1][j], dp[i][j - 1])
    
    return dp[0][n - 1]


def longest_palindromic_subsequence_optimized(s: str) -> int:
    """
    Space-optimized version using only O(n) space.
    """
    n = len(s)
    if n == 0:
        return 0
    
    # Only need previous row
    prev = [0] * n
    
    for i in range(n - 1, -1, -1):
        curr = [0] * n
        curr[i] = 1  # Single character
        
        for j in range(i + 1, n):
            if s[i] == s[j]:
                curr[j] = prev[j - 1] + 2 if j - 1 >= i + 1 else 2
            else:
                curr[j] = max(prev[j], curr[j - 1])
        
        prev = curr
    
    return prev[n - 1]
```
**Why**: LPS is a classic DP problem with applications in computational biology (DNA sequence analysis). The space-optimized version reduces memory from O(n²) to O(n).

## 3. Minimum Window Substring

```python
# PROBLEM: Find the minimum window in string s that contains all characters of string t
# TIME: O(|s| + |t|) SPACE: O(|s| + |t|)
from collections import Counter
from typing import Tuple

def min_window(s: str, t: str) -> str:
    """
    Find the smallest substring of s that contains all characters of t
    (including duplicates) using the sliding window technique.
    
    Args:
        s: Source string to search in
        t: Target string whose characters must all be present
    
    Returns:
        The minimum window substring, or empty string if not found
    """
    if not s or not t or len(s) < len(t):
        return ""
    
    # Frequency count of characters needed
    need = Counter(t)
    # Current frequency count in window
    have = Counter()
    
    # Number of unique characters that satisfy the requirement
    formed = 0
    required = len(need)  # Number of unique characters needed
    
    # Result: (window_length, left, right)
    result: Tuple[int, int, int] = (float('inf'), 0, 0)
    
    left = 0
    
    for right, char in enumerate(s):
        # Add current character to window
        have[char] += 1
        
        # Check if this character's count is now satisfied
        if char in need and have[char] == need[char]:
            formed += 1
        
        # Try to shrink the window while maintaining validity
        while left <= right and formed == required:
            char_left = s[left]
            
            # Update result if this window is smaller
            if right - left + 1 < result[0]:
                result = (right - left + 1, left, right)
            
            # Remove leftmost character from window
            have[char_left] -= 1
            if char_left in need and have[char_left] < need[char_left]:
                formed -= 1
            
            left += 1
    
    return s[result[1]:result[2] + 1] if result[0] != float('inf') else ""
```
**Why**: This is the classic sliding window pattern for substring problems. The technique of tracking "formed vs required" characters is reusable for many similar problems.

## 4. Next Smaller Element

```python
# PROBLEM: For each element in array, find the next smaller element to its right
# TIME: O(n) SPACE: O(n)
from typing import List

def next_smaller_elements(arr: List[int]) -> List[int]:
    """
    For each element, find the first smaller element to its right.
    If no such element exists, return -1 for that position.
    
    Uses monotonic stack (increasing order).
    
    Args:
        arr: Input array of integers
    
    Returns:
        List where result[i] is the next smaller element of arr[i], or -1
    """
    n = len(arr)
    result = [-1] * n
    stack = []  # Stack stores indices
    
    for i in range(n):
        # Pop elements that are greater than current element
        # For those popped elements, current is their next smaller
        while stack and arr[stack[-1]] > arr[i]:
            idx = stack.pop()
            result[idx] = arr[i]
        
        stack.append(i)
    
    return result


def next_smaller_element_indices(arr: List[int]) -> List[int]:
    """
    Returns indices instead of values.
    
    Args:
        arr: Input array of integers
    
    Returns:
        List where result[i] is the index of next smaller element, or -1
    """
    n = len(arr)
    result = [-1] * n
    stack = []  # Stack stores indices
    
    for i in range(n):
        while stack and arr[stack[-1]] > arr[i]:
            idx = stack.pop()
            result[idx] = i
        
        stack.append(i)
    
    return result


def previous_smaller_elements(arr: List[int]) -> List[int]:
    """
    For each element, find the previous smaller element to its left.
    """
    n = len(arr)
    result = [-1] * n
    stack = []
    
    for i in range(n):
        while stack and arr[stack[-1]] >= arr[i]:
            stack.pop()
        
        if stack:
            result[i] = arr[stack[-1]]
        
        stack.append(i)
    
    return result
```
**Why**: The monotonic stack pattern is essential for range queries and appears in problems like Largest Rectangle in Histogram. Understanding next/previous smaller/larger variants is crucial.

## 5. Extended GCD

```python
# PROBLEM: Find gcd(a, b) and integers x, y such that ax + by = gcd(a, b)
# TIME: O(log(min(a, b))) SPACE: O(log(min(a, b))) for recursion
from typing import Tuple

def extended_gcd(a: int, b: int) -> Tuple[int, int, int]:
    """
    Extended Euclidean Algorithm.
    
    Returns gcd(a, b) and coefficients x, y such that:
        a*x + b*y = gcd(a, b)
    
    Args:
        a, b: Two integers
    
    Returns:
        Tuple (gcd, x, y) where a*x + b*y = gcd
    """
    if b == 0:
        return a, 1, 0
    
    gcd, x1, y1 = extended_gcd(b, a % b)
    
    # Back-substitute to find x and y
    x = y1
    y = x1 - (a // b) * y1
    
    return gcd, x, y


def extended_gcd_iterative(a: int, b: int) -> Tuple[int, int, int]:
    """
    Iterative version of extended GCD to avoid stack overflow.
    """
    old_r, r = a, b
    old_s, s = 1, 0
    old_t, t = 0, 1
    
    while r != 0:
        quotient = old_r // r
        old_r, r = r, old_r - quotient * r
        old_s, s = s, old_s - quotient * s
        old_t, t = t, old_t - quotient * t
    
    return old_r, old_s, old_t


def mod_inverse(a: int, m: int) -> int:
    """
    Find modular multiplicative inverse of a modulo m.
    
    Returns x such that (a * x) % m == 1
    Only exists if gcd(a, m) == 1.
    """
    gcd, x, _ = extended_gcd(a % m, m)
    
    if gcd != 1:
        raise ValueError(f"Modular inverse doesn't exist for {a} mod {m}")
    
    return (x % m + m) % m
```
**Why**: Extended GCD is fundamental for modular arithmetic, cryptography (RSA), and solving linear Diophantine equations. The modular inverse is essential for division in modular arithmetic.

## 6. Modular Exponentiation

```python
# PROBLEM: Compute (base^exp) % mod efficiently
# TIME: O(log exp) SPACE: O(1)
from typing import Tuple

def mod_power(base: int, exp: int, mod: int) -> int:
    """
    Fast modular exponentiation using binary exponentiation.
    
    Computes (base^exp) % mod efficiently by breaking down
    the exponent into powers of 2.
    
    Args:
        base: The base number
        exp: The exponent (non-negative integer)
        mod: The modulus
    
    Returns:
        (base^exp) % mod
    """
    if mod == 1:
        return 0
    
    result = 1
    base = base % mod
    
    while exp > 0:
        # If exp is odd, multiply result with base
        if exp & 1:
            result = (result * base) % mod
        
        # Square the base
        base = (base * base) % mod
        
        # Divide exp by 2
        exp >>= 1
    
    return result


def mod_power_recursive(base: int, exp: int, mod: int) -> int:
    """
    Recursive version of modular exponentiation.
    """
    if exp == 0:
        return 1
    if exp == 1:
        return base % mod
    
    half = mod_power_recursive(base, exp // 2, mod)
    half = (half * half) % mod
    
    if exp % 2 == 0:
        return half
    else:
        return (half * (base % mod)) % mod


def matrix_mod_power(matrix: Tuple[Tuple[int, ...], ...], 
                     exp: int, 
                     mod: int) -> Tuple[Tuple[int, ...], ...]:
    """
    Matrix exponentiation for computing Fibonacci-like sequences.
    
    Example: [[1,1],[1,0]]^n gives Fibonacci numbers.
    """
    n = len(matrix)
    
    def multiply(A: Tuple[Tuple[int, ...], ...], 
                 B: Tuple[Tuple[int, ...], ...]) -> Tuple[Tuple[int, ...], ...]:
        result = [[0] * n for _ in range(n)]
        for i in range(n):
            for j in range(n):
                for k in range(n):
                    result[i][j] = (result[i][j] + A[i][k] * B[k][j]) % mod
        return tuple(tuple(row) for row in result)
    
    # Identity matrix
    result = tuple(tuple(1 if i == j else 0 for j in range(n)) for i in range(n))
    base = matrix
    
    while exp > 0:
        if exp & 1:
            result = multiply(result, base)
        base = multiply(base, base)
        exp >>= 1
    
    return result
```
**Why**: Binary exponentiation reduces exponentiation from O(n) to O(log n). This is critical for cryptographic operations and competitive programming. Matrix exponentiation enables computing linear recurrence relations in O(log n).

## 7. Wildcard Matching

```python
# PROBLEM: Implement wildcard pattern matching with '?' and '*'
# TIME: O(m*n) SPACE: O(m*n) or O(min(m,n)) optimized
from typing import List

def is_match_wildcard(s: str, p: str) -> bool:
    """
    Match string s against pattern p where:
    - '?' matches any single character
    - '*' matches any sequence of characters (including empty)
    
    Args:
        s: Input string
        p: Pattern with wildcards
    
    Returns:
        True if s matches pattern p
    """
    m, n = len(s), len(p)
    
    # dp[i][j] = True if s[0:i] matches p[0:j]
    dp = [[False] * (n + 1) for _ in range(m + 1)]
    
    # Empty string matches empty pattern
    dp[0][0] = True
    
    # Handle patterns that start with '*' - can match empty string
    for j in range(1, n + 1):
        if p[j - 1] == '*':
            dp[0][j] = dp[0][j - 1]
    
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if p[j - 1] == '*':
                # '*' matches empty (dp[i][j-1]) or extends previous match (dp[i-1][j])
                dp[i][j] = dp[i][j - 1] or dp[i - 1][j]
            elif p[j - 1] == '?' or s[i - 1] == p[j - 1]:
                dp[i][j] = dp[i - 1][j - 1]
    
    return dp[m][n]


def is_match_wildcard_optimized(s: str, p: str) -> bool:
    """
    Space-optimized version using O(min(m,n)) space.
    Also includes early termination for better average case.
    """
    m, n = len(s), len(p)
    
    # Optimize: remove consecutive '*' (they're equivalent to single '*')
    pattern: List[str] = []
    for char in p:
        if not (pattern and pattern[-1] == '*' and char == '*'):
            pattern.append(char)
    p = ''.join(pattern)
    n = len(p)
    
    # Use shorter dimension for DP array
    if m < n:
        s, p = p, s
        m, n = n, m
    
    prev = [False] * (n + 1)
    prev[0] = True
    
    for j in range(1, n + 1):
        if p[j - 1] == '*':
            prev[j] = prev[j - 1]
    
    for i in range(1, m + 1):
        curr = [False] * (n + 1)
        
        for j in range(1, n + 1):
            if p[j - 1] == '*':
                curr[j] = curr[j - 1] or prev[j]
            elif p[j - 1] == '?' or s[i - 1] == p[j - 1]:
                curr[j] = prev[j - 1]
        
        prev = curr
    
    return prev[n]
```
**Why**: Wildcard matching appears in file system operations, search functionality, and URL routing. The two-pointer greedy approach with backtracking is also common but DP is more intuitive.

## 8. Longest Substring with K Distinct Characters

```python
# PROBLEM: Find the longest substring with at most k distinct characters
# TIME: O(n) SPACE: O(k)
from collections import defaultdict

def longest_substring_k_distinct(s: str, k: int) -> int:
    """
    Find the length of the longest substring that contains at most k
    distinct characters using sliding window.
    
    Args:
        s: Input string
        k: Maximum number of distinct characters allowed
    
    Returns:
        Length of the longest valid substring
    """
    if k == 0 or not s:
        return 0
    
    n = len(s)
    char_count = defaultdict(int)
    left = 0
    max_length = 0
    
    for right in range(n):
        # Add current character to window
        char_count[s[right]] += 1
        
        # Shrink window while we have more than k distinct characters
        while len(char_count) > k:
            char_count[s[left]] -= 1
            if char_count[s[left]] == 0:
                del char_count[s[left]]
            left += 1
        
        # Update maximum length
        max_length = max(max_length, right - left + 1)
    
    return max_length


def longest_substring_k_distinct_with_result(s: str, k: int) -> tuple:
    """
    Returns both length and the actual substring.
    
    Returns:
        Tuple of (length, substring)
    """
    if k == 0 or not s:
        return 0, ""
    
    n = len(s)
    char_count = defaultdict(int)
    left = 0
    max_length = 0
    best_start = 0
    
    for right in range(n):
        char_count[s[right]] += 1
        
        while len(char_count) > k:
            char_count[s[left]] -= 1
            if char_count[s[left]] == 0:
                del char_count[s[left]]
            left += 1
        
        if right - left + 1 > max_length:
            max_length = right - left + 1
            best_start = left
    
    return max_length, s[best_start:best_start + max_length]


def longest_substring_exactly_k_distinct(s: str, k: int) -> int:
    """
    Find longest substring with exactly k distinct characters.
    
    Uses the formula: exactly(k) = atMost(k) - atMost(k-1)
    """
    def at_most_k(k: int) -> int:
        if k < 0:
            return 0
        char_count = defaultdict(int)
        left = 0
        max_len = 0
        
        for right in range(len(s)):
            char_count[s[right]] += 1
            
            while len(char_count) > k:
                char_count[s[left]] -= 1
                if char_count[s[left]] == 0:
                    del char_count[s[left]]
                left += 1
            
            max_len = max(max_len, right - left + 1)
        
        return max_len
    
    return at_most_k(k) - at_most_k(k - 1)
```
**Why**: This is a fundamental sliding window problem. The "at most K" pattern is reusable for many substring problems, and the "exactly K = atMost(K) - atMost(K-1)" trick is elegant.

## 9. Palindrome Partitioning

```python
# PROBLEM: Partition string such that every substring is a palindrome
# TIME: O(n * 2^n) SPACE: O(n²) for palindrome table
from typing import List

def partition_palindrome(s: str) -> List[List[str]]:
    """
    Find all possible palindrome partitions of string s.
    
    Uses backtracking with precomputed palindrome table for efficiency.
    
    Args:
        s: Input string
    
    Returns:
        List of all valid palindrome partitions
    """
    n = len(s)
    
    # Precompute which substrings are palindromes
    # is_palindrome[i][j] = True if s[i:j+1] is palindrome
    is_palindrome = [[False] * n for _ in range(n)]
    
    # Every single character is a palindrome
    for i in range(n):
        is_palindrome[i][i] = True
    
    # Check for length 2
    for i in range(n - 1):
        is_palindrome[i][i + 1] = (s[i] == s[i + 1])
    
    # Check for lengths 3 and above
    for length in range(3, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            is_palindrome[i][j] = (s[i] == s[j]) and is_palindrome[i + 1][j - 1]
    
    result = []
    
    def backtrack(start: int, current_partition: List[str]):
        # Base case: reached end of string
        if start == n:
            result.append(current_partition[:])
            return
        
        # Try all possible end positions
        for end in range(start, n):
            if is_palindrome[start][end]:
                # Add this palindrome to partition
                current_partition.append(s[start:end + 1])
                backtrack(end + 1, current_partition)
                current_partition.pop()
    
    backtrack(0, [])
    return result


def min_cut_palindrome(s: str) -> int:
    """
    Find minimum cuts needed to partition s into palindromes.
    
    Uses DP: cuts[i] = minimum cuts for s[0:i+1]
    """
    n = len(s)
    
    # Precompute palindrome table
    is_palindrome = [[False] * n for _ in range(n)]
    
    for i in range(n):
        is_palindrome[i][i] = True
    
    for i in range(n - 1):
        is_palindrome[i][i + 1] = (s[i] == s[i + 1])
    
    for length in range(3, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            is_palindrome[i][j] = (s[i] == s[j]) and is_palindrome[i + 1][j - 1]
    
    # DP for minimum cuts
    cuts = [0] * n
    
    for i in range(n):
        # Worst case: cut at every character
        min_cut = i
        
        for j in range(i + 1):
            if is_palindrome[j][i]:
                if j == 0:
                    # Entire substring s[0:i+1] is palindrome
                    min_cut = 0
                else:
                    min_cut = min(min_cut, cuts[j - 1] + 1)
        
        cuts[i] = min_cut
    
    return cuts[n - 1]
```
**Why**: Combines DP (palindrome checking) with backtracking. The minimum cuts variant is a classic follow-up that uses a different DP approach. Precomputing palindromes is a useful optimization pattern.

## 10. Coin Change Variants

```python
# PROBLEM: Multiple coin change problems with different constraints
# TIME: O(amount * n_coins) SPACE: O(amount)
from typing import List
from functools import lru_cache

def coin_change_min_coins(coins: List[int], amount: int) -> int:
    """
    Find minimum number of coins to make up the amount.
    Classic unbounded knapsack / coin change problem.
    
    Args:
        coins: Available coin denominations (can use each unlimited times)
        amount: Target amount
    
    Returns:
        Minimum number of coins needed, or -1 if impossible
    """
    # dp[i] = minimum coins to make amount i
    dp = [float('inf')] * (amount + 1)
    dp[0] = 0
    
    for coin in coins:
        for i in range(coin, amount + 1):
            dp[i] = min(dp[i], dp[i - coin] + 1)
    
    return dp[amount] if dp[amount] != float('inf') else -1


def coin_change_ways(coins: List[int], amount: int) -> int:
    """
    Count number of ways to make amount (order doesn't matter).
    
    Args:
        coins: Available coin denominations
        amount: Target amount
    
    Returns:
        Number of distinct combinations
    """
    dp = [0] * (amount + 1)
    dp[0] = 1  # One way to make amount 0: use no coins
    
    for coin in coins:
        for i in range(coin, amount + 1):
            dp[i] += dp[i - coin]
    
    return dp[amount]


def coin_change_ways_ordered(coins: List[int], amount: int) -> int:
    """
    Count ways to make amount where order matters.
    (1+2 and 2+1 are counted as different)
    """
    dp = [0] * (amount + 1)
    dp[0] = 1
    
    for i in range(1, amount + 1):
        for coin in coins:
            if coin <= i:
                dp[i] += dp[i - coin]
    
    return dp[amount]


def coin_change_with_limit(coins: List[int], counts: List[int], amount: int) -> int:
    """
    Coin change where each coin type has a limited supply.
    
    Args:
        coins: Coin denominations
        counts: Maximum count for each coin type
        amount: Target amount
    
    Returns:
        Minimum number of coins needed
    """
    n = len(coins)
    dp = [float('inf')] * (amount + 1)
    dp[0] = 0
    
    for i in range(n):
        coin, count = coins[i], counts[i]
        
        # Process in reverse to avoid using same coin multiple times incorrectly
        for j in range(amount, -1, -1):
            if dp[j] != float('inf'):
                for k in range(1, count + 1):
                    if j + coin * k <= amount:
                        dp[j + coin * k] = min(dp[j + coin * k], dp[j] + k)
    
    return dp[amount] if dp[amount] != float('inf') else -1


def coin_change_all_combinations(coins: List[int], amount: int) -> List[List[int]]:
    """
    Find all combinations of coins that sum to amount.
    """
    result = []
    coins.sort(reverse=True)  # Process larger coins first for efficiency
    
    def backtrack(start: int, remaining: int, current: List[int]):
        if remaining == 0:
            result.append(current[:])
            return
        
        for i in range(start, len(coins)):
            coin = coins[i]
            if coin > remaining:
                continue
            
            current.append(coin)
            backtrack(i, remaining - coin, current)  # i not i+1 (unlimited)
            current.pop()
    
    backtrack(0, amount, [])
    return result
```
**Why**: Coin change is the canonical unbounded knapsack problem. The variants cover: minimum coins, counting ways (ordered/unordered), bounded supply, and enumeration. These patterns transfer to many optimization problems.
