# HTML & JavaScript Function Reference

## 1. String Methods

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>String Methods Demo</title>
</head>
<body>
    <!-- Input for string manipulation -->
    <input type="text" id="inputString" placeholder="Enter a string" />
    <div id="output"></div>

    <script>
        /**
         * Demonstrate common JavaScript string methods
         * Shows: charAt, includes, indexOf, slice, split, toUpperCase, toLowerCase, trim
         */
        function demonstrateStringMethods() {
            const input = document.getElementById('inputString').value;
            const output = document.getElementById('output');
            
            if (!input.trim()) {
                output.innerHTML = '<p>Please enter a string</p>';
                return;
            }

            // Display results of various string methods
            const results = {
                'Original': input,
                'Length': input.length,
                'First Char': input.charAt(0),
                'Last Char': input.charAt(input.length - 1),
                'Uppercase': input.toUpperCase(),
                'Lowercase': input.toLowerCase(),
                'Trimmed': input.trim(),
                'Includes "a"': input.includes('a'),
                'Index of "a"': input.indexOf('a'),
                'Slice (0,3)': input.slice(0, 3),
                'Split by space': input.split(' ')
            };

            // Format output as HTML
            let html = '<h3>String Method Results:</h3>';
            for (const [method, result] of Object.entries(results)) {
                html += `<p><strong>${method}:</strong> ${JSON.stringify(result)}</p>`;
            }
            output.innerHTML = html;
        }

        // Event listener for input changes
        document.getElementById('inputString').addEventListener('input', demonstrateStringMethods);
    </script>
</body>
</html>
```

**Why**: Demonstrates essential string manipulation techniques for text processing and validation in web applications. Strings are fundamental to handling user input, API responses, and text display.

## 2. Object Methods

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Object Methods Demo</title>
</head>
<body>
    <div id="output"></div>

    <script>
        /**
         * Demonstrate JavaScript object methods
         * Shows: Object.keys, Object.values, Object.entries, Object.assign, hasOwnProperty
         */
        function demonstrateObjectMethods() {
            const user = {
                name: 'John Doe',
                email: 'john@example.com',

            };

            html += `<p><strong>Entries:</strong> ${JSON.stringify(entries)}</p>`;
            html += `<p><strong>Cloned Object:</strong> ${JSON.stringify(userClone)}</p>`;
            html += `<p><strong>Has 'name' property:</strong> ${hasName}</p>`;

            output.innerHTML = html;
        }

        // Run on page load
        demonstrateObjectMethods();
    </script>
</body>
</html>
```

**Why**: Object methods are essential for data manipulation, iteration, and object introspection. Critical for working with JSON data, API responses, and complex data structures.

## 3. Virtual DOM

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Virtual DOM Demo</title>
</head>
<body>
    <button id="increment">Increment Counter</button>
    <div id="app"></div>

    <script>
        /**
         * Simple Virtual DOM implementation
         * Demonstrates: Creating virtual nodes, diffing algorithm, efficient DOM updates
         */
        class VirtualDOM {
            constructor() {
                this.currentVNode = null;
                this.rootElement = null;
            }

            // Create a virtual node representation
            createElement(tag, props = {}, children = []) {
                return { tag, props, children };
            }

            // Render virtual node to actual DOM element
            render(vNode) {
                if (typeof vNode === 'string') {
                    return document.createTextNode(vNode);
                }

                const element = document.createElement(vNode.tag);
                
                // Apply props
                for (const [key, value] of Object.entries(vNode.props)) {
                    element.setAttribute(key, value);
                }

                // Render children
                vNode.children.forEach(child => {
                    element.appendChild(this.render(child));
                });

                return element;
            }

            // Diff two virtual nodes and update only changed parts
            diff(oldVNode, newVNode) {
                // If nodes are different types, replace entirely
                if (!oldVNode || oldVNode.tag !== newVNode.tag) {
                    return this.render(newVNode);
                }

                // If text content changed
                if (typeof oldVNode === 'string' || typeof newVNode === 'string') {
                    if (oldVNode !== newVNode) {
                        return this.render(newVNode);
                    }
                    return null;
                }

                // Reuse existing element, only update props
                const element = this.render(newVNode);
                return element;
            }

            // Mount virtual DOM to real DOM
            mount(vNode, container) {
                this.rootElement = container;
                this.currentVNode = vNode;
                container.appendChild(this.render(vNode));
            }

            // Update with new virtual node
            update(newVNode) {
                const patches = this.diff(this.currentVNode, newVNode);
                if (patches) {
                    this.rootElement.replaceChild(patches, this.rootElement.firstChild);
                    this.currentVNode = newVNode;
                }
            }
        }

        // Usage example
        const vdom = new VirtualDOM();
        let count = 0;

        function createCounterNode(count) {
            return vdom.createElement('div', { id: 'counter' }, [
                `Count: ${count}`
            ]);
        }

        // Mount initial state
        vdom.mount(createCounterNode(count), document.getElementById('app'));

        // Update on button click
        document.getElementById('increment').addEventListener('click', () => {
            count++;
            vdom.update(createCounterNode(count));
        });
    </script>
</body>
</html>
```

**Why**: Virtual DOM enables efficient UI updates by minimizing direct DOM manipulation. This concept powers modern frameworks like React and Vue, providing better performance and declarative programming.

## 4. Context Menu Event

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Context Menu Event Demo</title>
    <style>
        .custom-context-menu {
            position: absolute;
            background: white;
            border: 1px solid #ccc;
            box-shadow: 2px 2px 5px rgba(0,0,0,0.2);
            display: none;
            z-index: 1000;
        }
        .menu-item {
            padding: 8px 16px;
            cursor: pointer;
        }
        .menu-item:hover {
            background: #f0f0f0;
        }
        .target-area {
            width: 300px;
            height: 200px;
            background: #e0e0e0;
            margin: 20px;
            padding: 20px;
        }
    </style>
</head>
<body>
    <div class="target-area" id="targetArea">
        Right-click here for custom context menu
    </div>

    <!-- Custom context menu -->
    <div id="contextMenu" class="custom-context-menu">
        <div class="menu-item" data-action="copy">Copy</div>
        <div class="menu-item" data-action="paste">Paste</div>
        <div class="menu-item" data-action="refresh">Refresh</div>
    </div>

    <script>
        /**
         * Handle context menu (right-click) events
         * Demonstrates: preventDefault, custom menu positioning, menu item actions
         */
        function setupContextMenu() {
            const targetArea = document.getElementById('targetArea');
            const contextMenu = document.getElementById('contextMenu');

            // Show custom menu on right-click
            targetArea.addEventListener('contextmenu', (event) => {
                // Prevent default browser context menu
                event.preventDefault();

                // Position custom menu at mouse location
                contextMenu.style.left = event.pageX + 'px';
                contextMenu.style.top = event.pageY + 'px';
                contextMenu.style.display = 'block';
            });

            // Handle menu item clicks
            contextMenu.querySelectorAll('.menu-item').forEach(item => {
                item.addEventListener('click', () => {
                    const action = item.getAttribute('data-action');
                    console.log(`Action: ${action}`);
                    contextMenu.style.display = 'none';
                });
            });

            // Hide menu when clicking elsewhere
            document.addEventListener('click', () => {
                contextMenu.style.display = 'none';
            });
        }

        setupContextMenu();
    </script>
</body>
</html>
```

**Why**: Custom context menus enhance user experience by providing context-specific actions. Essential for applications requiring specialized right-click functionality like editors, dashboards, and interactive tools.

## 5. Clipboard Event

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Clipboard Event Demo</title>
</head>
<body>
    <textarea id="textInput" rows="4" cols="50" placeholder="Try copy, cut, or paste here"></textarea>
    <div id="log"></div>

    <script>
        /**
         * Handle clipboard events (copy, cut, paste)
         * Demonstrates: event interception, clipboard data access, modification
         */
        function setupClipboardEvents() {
            const textInput = document.getElementById('textInput');
            const logDiv = document.getElementById('log');

            // Handle copy event
            textInput.addEventListener('copy', (event) => {
                const selection = document.getSelection();
                logDiv.innerHTML += '<p>Copied text: ' + selection.toString() + '</p>';
            });

            // Handle cut event
            textInput.addEventListener('cut', (event) => {
                const selection = document.getSelection();
                logDiv.innerHTML += '<p>Cut text: ' + selection.toString() + '</p>';
            });

            // Handle paste event
            textInput.addEventListener('paste', (event) => {
                // Prevent default paste behavior
                event.preventDefault();

                // Get clipboard data
                const pastedText = (event.clipboardData || window.clipboardData).getData('text');
                
                logDiv.innerHTML += '<p>Pasted text: ' + pastedText + '</p>';
                
                // Optionally transform pasted content
                const transformed = pastedText.toUpperCase();
                
                // Insert transformed content
                const cursorPos = textInput.selectionStart;
                const textBefore = textInput.value.substring(0, cursorPos);
                const textAfter = textInput.value.substring(textInput.selectionEnd);
                textInput.value = textBefore + transformed + textAfter;
            });
        }

        setupClipboardEvents();
    </script>
</body>
</html>
```

**Why**: Clipboard events enable rich text editing, content transformation, and security controls. Vital for applications needing to sanitize, transform, or track clipboard operations.

## 7. Form Validation API

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Form Validation API Demo</title>
    <style>
        .error {
            color: red;
            font-size: 0.9em;
        }
        input:invalid {
            border-color: red;
        }
        input:valid {
            border-color: green;
        }
    </style>
</head>
<body>
    <form id="userForm">
        <div>
            <label>Email:</label>
            <input type="email" id="email" name="email" required />
            <span id="emailError" class="error"></span>
        </div>
        
        <div>
            <label>Password (min 8 chars):</label>
            <input type="password" id="password" name="password" 
                   minlength="8" required />
            <span id="passwordError" class="error"></span>
        </div>
        
        <div>
            <label>Age (18-99):</label>
            <input type="number" id="age" name="age" min="18" max="99" required />
            <span id="ageError" class="error"></span>
        </div>
        
        <button type="submit">Submit</button>
    </form>

    <script>
        /**
         * Demonstrate HTML5 Form Validation API
         * Shows: checkValidity, validationMessage, setCustomValidity, validity states
         */
        function setupFormValidation() {
            const form = document.getElementById('userForm');
            const emailInput = document.getElementById('email');
            const passwordInput = document.getElementById('password');
            const ageInput = document.getElementById('age');

            // Real-time validation
            [emailInput, passwordInput, ageInput].forEach(input => {
                input.addEventListener('input', () => {
                    validateField(input);
                });
            });

            // Validate individual field
            function validateField(input) {
                const errorSpan = document.getElementById(input.id + 'Error');
                
                // Check validity using Validation API
                if (!input.checkValidity()) {
                    // Display browser's validation message
                    errorSpan.textContent = input.validationMessage;
                    return false;
                } else {
                    errorSpan.textContent = '';
                    return true;
                }
            }

            // Form submission
            form.addEventListener('submit', (event) => {
                event.preventDefault();
                
                // Check overall form validity
                if (!form.checkValidity()) {
                    console.log('Form is invalid');
                    return;
                }

                // Custom validation example
                if (passwordInput.value.length < 8) {
                    passwordInput.setCustomValidity('Password must be at least 8 characters');
                    passwordInput.reportValidity();
                    return;
                }

                // Form is valid, process submission
                console.log('Form submitted successfully');
                const formData = new FormData(form);
                console.log('Data:', Object.fromEntries(formData));
            });
        }

        setupFormValidation();
    </script>
</body>
</html>
```

**Why**: The Form Validation API provides built-in validation with JavaScript control. Essential for creating user-friendly forms with immediate feedback, reducing server-side validation load.

## 8. Video Poster Attribute

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Video Poster Attribute Demo</title>
</head>
<body>
    <!-- Video with poster shows image before playback starts -->
    <video width="640" height="360" controls poster="https://example.com/poster.jpg">
        <source src="video.mp4" type="video/mp4">
        <source src="video.webm" type="video/webm">
        Your browser does not support the video tag.
    </video>

    <script>
        /**
         * Demonstrate video poster attribute usage
         * Shows: poster attribute, dynamic poster changes, poster as thumbnail
         */
        function setupVideoPoster() {
            const video = document.querySelector('video');

            // Change poster dynamically
            function changePoster(newPosterUrl) {
                video.poster = newPosterUrl;
            }

            // Generate poster from video frame
            function captureFrameAsPoster(video, time = 0) {
                const canvas = document.createElement('canvas');
                const ctx = canvas.getContext('2d');
                
                // Seek to specific time
                video.currentTime = time;
                
                video.addEventListener('seeked', () => {
                    // Draw current frame to canvas
                    canvas.width = video.videoWidth;
                    canvas.height = video.videoHeight;
                    ctx.drawImage(video, 0, 0);
                    
                    // Convert to data URL and set as poster
                    video.poster = canvas.toDataURL();
                }, { once: true });
            }

            // Example: Capture frame at 5 seconds as poster
            video.addEventListener('loadeddata', () => {
                captureFrameAsPoster(video, 5);
            });

            // Event listeners for poster changes
            video.addEventListener('play', () => {
                console.log('Video started - poster hidden');
            });

            video.addEventListener('pause', () => {
                console.log('Video paused');
            });
        }

        setupVideoPoster();
    </script>
</body>
</html>
```

**Why**: The poster attribute improves UX by showing a preview image before video loads. Critical for video platforms, providing visual feedback and branding before playback begins.

```html

<body>

        <div class="flex-item">1</div>

    </div>

    <h3>flex-direction: row-reverse</h3>
    <div class="flex-container row-reverse">
        <div class="flex-item">1</div>
        <div class="flex-item">2</div>
        <div class="flex-item">3</div>
    </div>

    <h3>flex-direction: column</h3>
    <div class="flex-container column">
        <div class="flex-item">1</div>
        <div class="flex-item">2</div>
        <div class="flex-item">3</div>
    </div>

    <h3>flex-direction: column-reverse</h3>
    <div class="flex-container column-reverse">
        <div class="flex-item">1</div>
        <div class="flex-item">2</div>
        <div class="flex-item">3</div>
    </div>

    <script>
        /**
         * Demonstrate flexbox flex-direction property
         * Shows: row, row-reverse, column, column-reverse, responsive layouts
         */
        function demonstrateFlexDirection() {
            const containers = document.querySelectorAll('.flex-container');
            
            containers.forEach(container => {
                const direction = getComputedStyle(container).flexDirection;
                console.log(`Container direction: ${direction}`);
                
                // Count items
                const items = container.querySelectorAll('.flex-item');
                console.log(`  Items: ${items.length}`);
            });

            // Dynamic direction change based on screen size
            function updateFlexDirection() {
                const width = window.innerWidth;
                
                containers.forEach(container => {
                    if (width < 600) {
                        // Mobile: stack items vertically
                        container.style.flexDirection = 'column';
                    } else {
                        // Desktop: use original direction
                        container.style.flexDirection = '';
                    }
                });
            }

            // Update on resize
            window.addEventListener('resize', updateFlexDirection);
            updateFlexDirection();
        }

        demonstrateFlexDirection();
    </script>
</body>
</html>
```

**Why**: Flexbox direction controls layout flow, essential for responsive design. Enables easy switching between horizontal and vertical layouts for different screen sizes and use cases.

## 10. Universal Selector

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Universal Selector Demo</title>
    <style>
        /* Universal selector: applies to ALL elements */
        * {
            box-sizing: border-box;
        }

        /* Reset default margins/padding */
        * {
            margin: 0;
            padding: 0;
        }

        /* More specific selectors override universal selector */
        .container {
            max-width: 1200px;
            margin: 20px auto;
            padding: 20px;
            background: #f5f5f5;
        }

        .box {
            padding: 20px;
            margin: 10px;
            background: #007bff;
            color: white;
        }

        /* Universal selector with descendant */
        .container * {
            border: 1px solid #ddd;
        }
    </style>
</head>
<body>
    <div class="container">
        <h2>Universal Selector (*)</h2>
        <p>Selects ALL elements in the document</p>
        
        <div class="box">Box 1</div>
        <div class="box">Box 2</div>
        
        <ul>
            <li>List item 1</li>
            <li>List item 2</li>
        </ul>
    </div>

    <script>
        /**
         * Demonstrate universal selector (*) behavior
         * Shows: global styles, specificity, performance considerations
         */
        function demonstrateUniversalSelector() {
            // Select all elements using universal selector
            const allElements = document.querySelectorAll('*');
            
            console.log(`Total elements: ${allElements.length}`);
            
            // Common use case: box-sizing reset
            allElements.forEach(element => {
                const computed = getComputedStyle(element);
                console.log(`${element.tagName}: box-sizing = ${computed.boxSizing}`);
            });

            // Universal selector in CSS applies to everything
            // Useful for:
            // 1. CSS resets
            // 2. Box-sizing standardization
            // 3. Debugging (borders on all elements)
            
            // Warning: can impact performance on large documents
            // Use with caution in production

            // Example: Debug mode - highlight all elements
            function enableDebugMode() {
                const style = document.createElement('style');
                style.textContent = '* { outline: 1px solid red !important; }';
                document.head.appendChild(style);
            }

            // Uncomment to see all element boundaries:
            // enableDebugMode();
        }

        demonstrateUniversalSelector();
    </script>
</body>
</html>
```

**Why**: The universal selector enables CSS resets and global styles. Essential for normalizing browser defaults and implementing consistent box-sizing across all elements.
