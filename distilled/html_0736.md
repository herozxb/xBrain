```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Dynamic Table Generator</title>
</head>
<body>
  <!-- 
    # PROBLEM: Create and manipulate HTML tables dynamically
    # TIME: O(n*m) SPACE: O(n*m)
  -->
  <table id="dataTable" border="1">
    <thead>
      <tr id="headerRow"></tr>
    </thead>
    <tbody id="tableBody"></tbody>
  </table>

  <script>
    // Sample data
    const data = [
      { name: 'Alice', age: 25, city: 'NYC' },
      { name: 'Bob', age: 30, city: 'LA' }
    ];

    // Generate table headers
    function generateTableHead(table, data) {
      const thead = table.createTHead();
      const row = thead.insertRow();
      for (const key of Object.keys(data[0])) {
        const th = document.createElement('th');
        const text = document.createTextNode(key.toUpperCase());
        th.appendChild(text);
        row.appendChild(th);
      }
    }

    // Generate table rows
    function generateTable(table, data) {
      for (const element of data) {
        const row = table.insertRow();
        for (const key in element) {
          const cell = row.insertCell();
          const text = document.createTextNode(element[key]);
          cell.appendChild(text);
        }
      }
    }

    const table = document.getElementById('dataTable');
    generateTableHead(table, data);
    generateTable(table.querySelector('tbody'), data);
  </script>
</body>
</html>
```
**Why**: Dynamically creates tables from data arrays. Useful for rendering API responses or user-generated datasets.

## js_comparison_operators
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>JavaScript Comparison Operators</title>
</head>
<body>
  <script>
    /*
      # PROBLEM: Compare values using equality and relational operators
      # TIME: O(1) SPACE: O(1)
    */

    // == (loose equality - type coercion)
    console.log(5 == '5');      // true (string coerced to number)
    console.log(null == undefined); // true
    console.log(0 == false);    // true

    // === (strict equality - no coercion)
    console.log(5 === '5');     // false (different types)
    console.log(null === undefined); // false
    console.log(0 === false);   // false

    // != (loose inequality)
    console.log(5 != '5');      // false

    // !== (strict inequality)
    console.log(5 !== '5');     // true

    // Relational operators
    console.log(10 > 5);        // true
    console.log(10 >= 10);      // true
    console.log(3 < 7);         // true
    console.log(3 <= 3);        // true

    // String comparison (lexicographic)
    console.log('apple' < 'banana'); // true
    console.log('Apple' < 'apple');   // true (uppercase < lowercase)

    // Object reference comparison
    const obj1 = { a: 1 };
    const obj2 = { a: 1 };
    const obj3 = obj1;
    console.log(obj1 === obj2); // false (different references)
    console.log(obj1 === obj3); // true (same reference)

    // NaN comparisons
    console.log(NaN === NaN);   // false (NaN never equals itself)
    console.log(Number.isNaN(NaN)); // true (proper NaN check)

    // typeof comparisons
    console.log(typeof 'hello' === 'string'); // true
    console.log(typeof 42 === 'number');      // true
  </script>
</body>
</html>
```
**Why**: Understanding loose vs strict equality prevents bugs. Always prefer `===` and `!==` for predictable comparisons. Objects compare by reference, not value.

## js_proxy_object
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>JavaScript Proxy Object</title>
</head>
<body>
  <script>
    /*
      # PROBLEM: Intercept and customize object operations using Proxy
      # TIME: O(1) SPACE: O(n)
    */

    const target = {
      name: 'Alice',
      age: 25
    };

    const handler = {
      // Intercept property access
      get(obj, prop) {
        console.log(`Getting ${prop}`);
        return prop in obj ? obj[prop] : 'Not found';
      },

      // Intercept property assignment
      set(obj, prop, value) {
        console.log(`Setting ${prop} = ${value}`);
        if (prop === 'age' && typeof value !== 'number') {
          throw new TypeError('Age must be a number');
        }
        obj[prop] = value;
        return true; // Indicate success
      },

      // Intercept property check
      has(obj, prop) {
        console.log(`Checking ${prop}`);
        return prop in obj;
      },

      // Intercept deletion
      deleteProperty(obj, prop) {
        console.log(`Deleting ${prop}`);
        delete obj[prop];
        return true;
      }
    };

    const proxy = new Proxy(target, handler);

    // Usage
    console.log(proxy.name);   // Logs: Getting name, returns: Alice
    proxy.age = 30;            // Logs: Setting age = 30
    console.log('age' in proxy); // Logs: Checking age, returns: true

    // Use case: Validation proxy
    function createValidatedUser(initialData) {
      return new Proxy(initialData, {
        set(obj, prop, value) {
          const validators = {
            email: (v) => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v),
            age: (v) => v >= 0 && v <= 120
          };
          
          if (validators[prop] && !validators[prop](value)) {
            throw new Error(`Invalid ${prop}: ${value}`);
          }
          obj[prop] = value;
          return true;
        }
      });
    }

    const user = createValidatedUser({ email: 'test@example.com' });
    user.age = 25; // OK
    // user.age = 150; // Error: Invalid age
  </script>
</body>
</html>
```
**Why**: Proxies enable metaprogramming - intercept operations for validation, logging, or virtual properties. Foundation for reactive frameworks like Vue 3.

## event_keypress
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Keypress Event Handling</title>
  <style>
    #output {
      padding: 20px;
      background: #f5f5f5;
      margin: 10px 0;
      min-height: 50px;
    }
    .hotkey-hint {
      color: #666;
      font-size: 12px;
    }
  </style>
</head>
<body>
  <!-- 
    # PROBLEM: Handle keyboard input and shortcuts
    # TIME: O(1) SPACE: O(1)
  -->
  
  <input type="text" id="inputField" placeholder="Type here...">
  <div id="output">Last key: none</div>
  
  <p class="hotkey-hint">
    Press: Ctrl+S (save), Ctrl+Enter (submit), Escape (clear)
  </p>

  <script>
    const input = document.getElementById('inputField');
    const output = document.getElementById('output');

    // keypress (deprecated - use keydown instead)
    // Only fires for character keys, not modifiers
    
    // keydown - fires for all keys
    input.addEventListener('keydown', (e) => {
      console.log('Key:', e.key);
      console.log('Code:', e.code);
      console.log('KeyCode:', e.keyCode); // Deprecated
      
      // Display key info
      output.innerHTML = `
        Key: ${e.key}<br>
        Code: ${e.code}<br>
        Ctrl: ${e.ctrlKey}, Shift: ${e.shiftKey}, Alt: ${e.altKey}
      `;

      // Keyboard shortcuts
      if (e.ctrlKey && e.key === 's') {
        e.preventDefault(); // Prevent browser save dialog
        console.log('Save triggered');
        alert('Saved!');
      }
      
      if (e.ctrlKey && e.key === 'Enter') {
        e.preventDefault();
        console.log('Submit form');
        alert(`Submitted: ${input.value}`);
      }
      
      if (e.key === 'Escape') {
        input.value = '';
        output.textContent = 'Cleared';
      }
    });

    // keyup - fires when key is released
    input.addEventListener('keyup', (e) => {
      console.log('Key released:', e.key);
    });

    // Global keyboard shortcuts (document level)
    document.addEventListener('keydown', (e) => {
      // Focus input with '/'
      if (e.key === '/' && document.activeElement !== input) {
        e.preventDefault();
        input.focus();
      }
    });

    // Detecting special keys
    function isSpecialKey(event) {
      return [
        'Enter', 'Tab', 'Escape', 'Backspace', 'Delete',
        'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight',
        'Home', 'End', 'PageUp', 'PageDown'
      ].includes(event.key);
    }
  </script>
</body>
</html>
```
**Why**: `keydown` is preferred over deprecated `keypress`. Use `e.key` for character, `e.code` for physical key location. Check modifier keys (`ctrlKey`, `shiftKey`) for shortcuts.
