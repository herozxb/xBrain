## Filter Pattern

```python
# PROBLEM: Dynamically filter collection items based on multiple criteria
# TIME: O(n*m) where n = items, m = filters SPACE: O(n) for filtered results
from typing import List, Callable, Any, Protocol

class FilterCriteria(Protocol):
    """Protocol defining filter criteria interface"""
    def matches(self, item: Any) -> bool:
        """Check if item matches this criteria"""
        ...

class Filter:
    """Generic filter implementation supporting multiple criteria"""
    
    def __init__(self):
        self.criteria: List[FilterCriteria] = []
    
    def add_criteria(self, criteria: FilterCriteria) -> None:
        """Add a filtering criteria"""
        self.criteria.append(criteria)
    
    def apply(self, items: List[Any]) -> List[Any]:
        """Apply all criteria to filter items"""
        result = items
        for criteria in self.criteria:
            result = [item for item in result if criteria.matches(item)]
        return result

# Example: Filter products by price range
class PriceRangeCriteria:
    def __init__(self, min_price: float, max_price: float):
        self.min_price = min_price
        self.max_price = max_price
    
    def matches(self, product: dict) -> bool:
        return self.min_price <= product['price'] <= self.max_price

# Example usage
products = [
    {'name': 'Laptop', 'price': 999, 'category': 'electronics'},
    {'name': 'Mouse', 'price': 29, 'category': 'electronics'},
    {'name': 'Desk', 'price': 199, 'category': 'furniture'}
]

price_filter = Filter()
price_filter.add_criteria(PriceRangeCriteria(50, 500))
filtered = price_filter.apply(products)
# Returns: [{'name': 'Desk', 'price': 199, 'category': 'furniture'}]
```

**Why**: Enables flexible, composable filtering logic without modifying data classes. Supports runtime composition of multiple filter criteria.

## Front Controller Pattern

```python
# PROBLEM: Centralize request handling and routing in a web application
# TIME: O(1) for routing SPACE: O(n) for registered handlers
from typing import Dict, Callable, Any, Optional
from enum import Enum
from dataclasses import dataclass

class HttpMethod(Enum):
    GET = "GET"
    POST = "POST"
    PUT = "PUT"
    DELETE = "DELETE"

@dataclass
class Request:
    """Represents an HTTP request"""
    method: HttpMethod
    path: str
    body: Optional[dict] = None
    headers: Dict[str, str] = None
    
    def __post_init__(self):
        if self.headers is None:
            self.headers = {}

@dataclass
class Response:
    """Represents an HTTP response"""
    status_code: int
    body: Any
    headers: Dict[str, str] = None
    
    def __post_init__(self):
        if self.headers is None:
            self.headers = {}

class FrontController:
    """Central dispatcher for all incoming requests"""
    
    def __init__(self):
        self.routes: Dict[str, Dict[HttpMethod, Callable]] = {}
        self.pre_interceptors: list = []
        self.post_interceptors: list = []
    
    def register(self, path: str, method: HttpMethod, handler: Callable) -> None:
        """Register a handler for a specific path and method"""
        if path not in self.routes:
            self.routes[path] = {}
        self.routes[path][method] = handler
    
    def add_pre_interceptor(self, interceptor: Callable) -> None:
        """Add pre-processing interceptor"""
        self.pre_interceptors.append(interceptor)
    
    def add_post_interceptor(self, interceptor: Callable) -> None:
        """Add post-processing interceptor"""
        self.post_interceptors.append(interceptor)
    
    def handle(self, request: Request) -> Response:
        """Process request through interceptors and dispatch to handler"""
        # Pre-processing
        for interceptor in self.pre_interceptors:
            request = interceptor(request)
        
        # Route dispatch
        if request.path not in self.routes:
            return Response(404, {"error": "Not Found"})
        
        if request.method not in self.routes[request.path]:
            return Response(405, {"error": "Method Not Allowed"})
        
        handler = self.routes[request.path][request.method]
        response = handler(request)
        
        # Post-processing
        for interceptor in self.post_interceptors:
            response = interceptor(response)
        
        return response

# Example usage
controller = FrontController()

def get_users(request: Request) -> Response:
    return Response(200, {"users": ["Alice", "Bob"]})

def create_user(request: Request) -> Response:
    return Response(201, {"created": request.body})

controller.register("/users", HttpMethod.GET, get_users)
controller.register("/users", HttpMethod.POST, create_user)

req = Request(method=HttpMethod.GET, path="/users")
resp = controller.handle(req)
```

**Why**: Provides a single entry point for all requests, enabling centralized authentication, logging, error handling, and routing logic.

```python
from dataclasses import dataclass, asdict

        """Create DTO from dictionary"""
```

## Separate Ways Pattern (DDD)

```python
# PROBLEM: Coordinate independent teams working on non-overlapping domains
# TIME: O(1) for context isolation SPACE: O(n) per bounded context
from typing import Dict, List, Set
from dataclasses import dataclass, field

@dataclass
class BoundedContext:
    """Represents an isolated domain boundary"""
    name: str
    domain_models: Set[str] = field(default_factory=set)
    apis: Set[str] = field(default_factory=set)
    dependencies: Set[str] = field(default_factory=set)
    
    def add_model(self, model: str) -> None:
        """Add domain model to this context"""
        self.domain_models.add(model)
    
    def add_api(self, api: str) -> None:
        """Add API endpoint to this context"""
        self.apis.add(api)
    
    def is_isolated(self) -> bool:
        """Check if context has no external dependencies"""
        return len(self.dependencies) == 0

class SeparateWaysCoordinator:
    """Manages completely independent bounded contexts"""
    
    def __init__(self):
        self.contexts: Dict[str, BoundedContext] = {}
    
    def register_context(self, context: BoundedContext) -> None:
        """Register a bounded context with zero coupling"""
        if context.dependencies:
            raise ValueError("Separate Ways requires zero dependencies")
        self.contexts[context.name] = context
    
    def get_context(self, name: str) -> BoundedContext:
        """Retrieve a bounded context"""
        return self.contexts.get(name)
    
    def can_parallelize(self) -> bool:
        """Check if all contexts can be developed independently"""
        return all(ctx.is_isolated() for ctx in self.contexts.values())
    
    def generate_contracts(self) -> Dict[str, List[str]]:
        """Generate interface contracts for each context"""
        return {
            name: list(ctx.apis)
            for name, ctx in self.contexts.items()
        }

# Example: Two completely separate domains
billing_context = BoundedContext(name="billing")
billing_context.add_model("Invoice")
billing_context.add_model("Payment")
billing_context.add_api("/billing/process")

analytics_context = BoundedContext(name="analytics")
analytics_context.add_model("Report")
analytics_context.add_model("Metric")
analytics_context.add_api("/analytics/generate")

coordinator = SeparateWaysCoordinator()
coordinator.register_context(billing_context)
coordinator.register_context(analytics_context)

print(f"Can parallelize: {coordinator.can_parallelize()}")
# Output: Can parallelize: True
```

**Why**: Enables complete team autonomy when domains have no overlap. Ideal for legacy system replacement or acquiring distinct business units.

## Query Bus Pattern

```python
# PROBLEM: Decouple query execution from query definition
# TIME: O(1) for dispatch SPACE: O(n) for registered handlers
from typing import Dict, Callable, Any, Type, TypeVar
from dataclasses import dataclass
from abc import ABC, abstractmethod

T = TypeVar('T')

@dataclass
class Query(ABC):
    """Base class for all queries"""
    pass

@dataclass
class GetUserQuery(Query):
    """Query to fetch user by ID"""
    user_id: int

@dataclass
class SearchProductsQuery(Query):
    """Query to search products"""
    keyword: str
    limit: int = 10

class QueryHandler(ABC):
    """Abstract handler for queries"""
    
    @abstractmethod
    def handle(self, query: Query) -> Any:
        """Process the query and return result"""
        pass

class GetUserHandler(QueryHandler):
    """Handler for GetUserQuery"""
    
    def __init__(self, user_repository):
        self.user_repository = user_repository
    
    def handle(self, query: GetUserQuery) -> dict:
        user = self.user_repository.find(query.user_id)
        return user if user else {}

class QueryBus:
    """Dispatches queries to appropriate handlers"""
    
    def __init__(self):
        self.handlers: Dict[Type[Query], QueryHandler] = {}
    
    def register(self, query_type: Type[Query], handler: QueryHandler) -> None:
        """Register a handler for a query type"""
        self.handlers[query_type] = handler
    
    def dispatch(self, query: Query) -> Any:
        """Execute query and return result"""
        query_type = type(query)
        if query_type not in self.handlers:
            raise ValueError(f"No handler registered for {query_type}")
        
        handler = self.handlers[query_type]
        return handler.handle(query)

# Example usage
class MockUserRepo:
    def find(self, user_id: int) -> dict:
        return {'id': user_id, 'name': f'User {user_id}'}

bus = QueryBus()
bus.register(GetUserQuery, GetUserHandler(MockUserRepo()))

result = bus.dispatch(GetUserQuery(user_id=42))
print(result)  # Output: {'id': 42, 'name': 'User 42'}
```

**Why**: Enables CQRS by separating read models from write models. Allows caching, logging, and validation middleware to be applied uniformly to all queries.

## Handler Pattern

```python
# PROBLEM: Chain multiple processors for a single request
# TIME: O(n) for n handlers SPACE: O(1) per handler
from typing import Optional, Callable, Any, List
from abc import ABC, abstractmethod
from dataclasses import dataclass

@dataclass
class Context:
    """Request context passed through handlers"""
    data: dict
    metadata: dict
    processed_by: List[str]
    
    def mark_processed(self, handler_name: str) -> None:
        """Track which handlers have processed this context"""
        self.processed_by.append(handler_name)

class Handler(ABC):
    """Abstract handler in the chain"""
    
    def __init__(self, name: str):
        self.name = name
        self.next_handler: Optional[Handler] = None
    
    def set_next(self, handler: 'Handler') -> 'Handler':
        """Set next handler in chain"""
        self.next_handler = handler
        return handler
    
    @abstractmethod
    def process(self, context: Context) -> Context:
        """Process the context and optionally pass to next handler"""
        pass
    
    def handle(self, context: Context) -> Context:
        """Template method: process and delegate"""
        context.mark_processed(self.name)
        context = self.process(context)
        
        if self.next_handler:
            return self.next_handler.handle(context)
        return context

class ValidationHandler(Handler):
    """Validates input data"""
    
    def process(self, context: Context) -> Context:
        required_fields = ['email', 'name']
        for field in required_fields:
            if field not in context.data:
                raise ValueError(f"Missing required field: {field}")
        context.metadata['validated'] = True
        return context

class SanitizationHandler(Handler):
    """Sanitizes input data"""
    
    def process(self, context: Context) -> Context:
        context.data['email'] = context.data['email'].lower().strip()
        context.data['name'] = context.data['name'].strip()
        context.metadata['sanitized'] = True
        return context

class LoggingHandler(Handler):
    """Logs processing step"""
    
    def process(self, context: Context) -> Context:
        print(f"[LOG] Processed by: {self.name}")
        context.metadata['logged'] = True
        return context

# Build chain
validation = ValidationHandler("validator")
sanitization = SanitizationHandler("sanitizer")
logging = LoggingHandler("logger")

validation.set_next(sanitization).set_next(logging)

# Process request
ctx = Context(
    data={'email': '  ALICE@EXAMPLE.COM  ', 'name': '  Alice  '},
    metadata={},
    processed_by=[]
)

result = validation.handle(ctx)
print(result.data)  # Output: {'email': 'alice@example.com', 'name': 'Alice'}
```

**Why**: Promotes single responsibility principle. Each handler focuses on one task. Chain can be reconfigured at runtime for different workflows.

## Compression Algorithms

```python
# PROBLEM: Reduce data size for storage/transmission
# TIME: O(n) for compression SPACE: O(n) for compressed output
from typing import List, Tuple
from collections import defaultdict
import heapq

class RunLengthEncoding:
    """Simple compression for repeated consecutive elements"""
    
    @staticmethod
    def compress(data: str) -> List[Tuple[str, int]]:
        """Compress string to list of (char, count) pairs"""
        if not data:
            return []
        
        compressed = []
        current_char = data[0]
        count = 1
        
        for char in data[1:]:
            if char == current_char:
                count += 1
            else:
                compressed.append((current_char, count))
                current_char = char
                count = 1
        
        compressed.append((current_char, count))
        return compressed
    
    @staticmethod
    def decompress(compressed: List[Tuple[str, int]]) -> str:
        """Decompress back to original string"""
        return ''.join(char * count for char, count in compressed)

class HuffmanNode:
    """Node for Huffman tree"""
    def __init__(self, char: str, freq: int, left=None, right=None):
        self.char = char
        self.freq = freq
        self.left = left
        self.right = right
    
    def __lt__(self, other):
        return self.freq < other.freq

class HuffmanCompression:
    """Huffman coding compression algorithm"""
    
    def __init__(self):
        self.codes = {}
        self.reverse_codes = {}
    
    def _build_tree(self, data: str) -> HuffmanNode:
        """Build Huffman tree from character frequencies"""
        freq = defaultdict(int)
        for char in data:
            freq[char] += 1
        
        heap = [HuffmanNode(char, f) for char, f in freq.items()]
        heapq.heapify(heap)
        
        while len(heap) > 1:
            left = heapq.heappop(heap)
            right = heapq.heappop(heap)
            merged = HuffmanNode(None, left.freq + right.freq, left, right)
            heapq.heappush(heap, merged)
        
        return heap[0]
    
    def _generate_codes(self, node: HuffmanNode, code: str = "") -> None:
        """Generate binary codes from Huffman tree"""
        if node.char is not None:
            self.codes[node.char] = code
            self.reverse_codes[code] = node.char
            return
        
        if node.left:
            self._generate_codes(node.left, code + "0")
        if node.right:
            self._generate_codes(node.right, code + "1")
    
    def compress(self, data: str) -> str:
        """Compress string to binary representation"""
        if not data:
            return ""
        
        root = self._build_tree(data)
        self._generate_codes(root)
        
        compressed = ""
        for char in data:
            compressed += self.codes[char]
        
        return compressed
    
    def decompress(self, compressed: str) -> str:
        """Decompress binary back to original string"""
        if not compressed:
            return ""
        
        result = ""
        current_code = ""
        
        for bit in compressed:
            current_code += bit
            if current_code in self.reverse_codes:
                result += self.reverse_codes[current_code]
                current_code = ""
        
        return result

# Example usage
rle = RunLengthEncoding()
rle_compressed = rle.compress("AAAABBBCCD")
print(rle_compressed)  # Output: [('A', 4), ('B', 3), ('C', 2), ('D', 1)]

huffman = HuffmanCompression()
huff_compressed = huffman.compress("hello world")
print(f"Compressed: {huff_compressed}")
decompressed = huffman.decompress(huff_compressed)
print(f"Decompressed: {decompressed}")  # Output: hello world
```

**Why**: Reduces storage and bandwidth costs. RLE is optimal for repetitive data, Huffman for variable-frequency characters.

## Parallel Scan

```python
# PROBLEM: Compute prefix sums in parallel across multiple processors
# TIME: O(log n) with O(n) processors SPACE: O(n) for intermediate results
from typing import List
import concurrent.futures
import math

class ParallelScan:
    """Parallel prefix scan implementation"""
    
    def __init__(self, num_workers: int = 4):
        self.num_workers = num_workers
    
    def sequential_scan(self, data: List[int]) -> List[int]:
        """Traditional sequential prefix sum"""
        result = [0] * len(data)
        running_sum = 0
        for i, val in enumerate(data):
            running_sum += val
            result[i] = running_sum
        return result
    
    def _partial_scan(self, chunk: List[int], offset: int) -> List[int]:
        """Compute partial scan for a chunk"""
        result = []
        running_sum = offset
        for val in chunk:
            running_sum += val
            result.append(running_sum)
        return result
    
    def parallel_prefix_sum(self, data: List[int]) -> List[int]:
        """Parallel prefix sum using divide-and-conquer"""
        if len(data) == 0:
            return []
        
        # Chunk size per worker
        chunk_size = math.ceil(len(data) / self.num_workers)
        chunks = [
            data[i:i + chunk_size]
            for i in range(0, len(data), chunk_size)
        ]
        
        # Phase 1: Compute partial scans in parallel
        with concurrent.futures.ThreadPoolExecutor(max_workers=self.num_workers) as executor:
            partial_results = list(executor.map(
                self.sequential_scan,
                chunks
            ))
        
        # Phase 2: Compute offsets from last element of each partial result
        offsets = [0]
        for i, partial in enumerate(partial_results[:-1]):
            offsets.append(offsets[i] + partial[-1])
        
        # Phase 3: Apply offsets to get final results
        final_result = []
        for offset, partial in zip(offsets, partial_results):
            adjusted = [val + offset - partial[0] + chunks[offsets.index(offset)][0] 
                       if offsets.index(offset) > 0 else val 
                       for val in partial]
            final_result.extend(adjusted)
        
        return final_result
    
    def parallel_max_scan(self, data: List[int]) -> List[int]:
        """Parallel prefix maximum (running max)"""
        result = data.copy()
        n = len(result)
        
        # Up-sweep phase
        step = 1
        while step < n:
            for i in range(step, n, 2 * step):
                result[i] = max(result[i], result[i - step])
            step *= 2
        
        return result

# Example usage
scanner = ParallelScan(num_workers=4)
data = [1, 3, 5, 7, 9, 2, 4, 6, 8, 10]

seq_result = scanner.sequential_scan(data)
print(f"Sequential: {seq_result}")

par_result = scanner.parallel_prefix_sum(data)
print(f"Parallel:   {par_result}")

max_scan = scanner.parallel_max_scan([3, 1, 4, 1, 5, 9, 2, 6])
print(f"Max scan:   {max_scan}")  # Output: [3, 3, 4, 4, 5, 9, 9, 9]
```

**Why**: Achieves logarithmic time complexity with sufficient processors. Essential for GPU computing and distributed data processing pipelines.

## Dataclass Patterns

```python
# PROBLEM: Reduce boilerplate for data-holding classes
# TIME: O(1) for instantiation SPACE: O(k) for k fields
from dataclasses import dataclass, field, asdict, astuple
from typing import List, Optional, Dict, ClassVar
from datetime import datetime
import json

@dataclass
class Product:
    """Immutable product entity"""
    id: int
    name: str
    price: float
    category: str = "general"
    tags: List[str] = field(default_factory=list)
    metadata: Dict[str, str] = field(default_factory=dict)
    
    # Class variable (not included in __init__)
    tax_rate: ClassVar[float] = 0.08
    
    def __post_init__(self):
        """Validate and transform after initialization"""
        if self.price < 0:
            raise ValueError("Price cannot be negative")
        self.name = self.name.strip().title()
    
    def calculate_total(self, quantity: int = 1) -> float:
        """Calculate total price including tax"""
        return self.price * quantity * (1 + self.tax_rate)
    
    def to_json(self) -> str:
        """Serialize to JSON"""
        return json.dumps(asdict(self))

@dataclass(frozen=True)
class ImmutablePoint:
    """Immutable coordinate point"""
    x: float
    y: float
    
    def distance_to(self, other: 'ImmutablePoint') -> float:
        """Calculate Euclidean distance"""
        return ((self.x - other.x) ** 2 + (self.y - other.y) ** 2) ** 0.5

@dataclass
class Order:
    """Order with default_factory for mutable fields"""
    order_id: str
    customer_id: int
    items: List[Product] = field(default_factory=list)
    created_at: datetime = field(default_factory=datetime.now)
    _internal_id: int = field(default=0, repr=False, compare=False)
    
    def add_item(self, product: Product, quantity: int = 1) -> None:
        """Add product to order"""
        for _ in range(quantity):
            self.items.append(product)
    
    def total(self) -> float:
        """Calculate order total"""
        return sum(item.price for item in self.items)

# Advanced: Inheritance with dataclasses
@dataclass
class DiscountedProduct(Product):
    """Product with discount capability"""
    discount_percent: float = 0.0
    
    def calculate_total(self, quantity: int = 1) -> float:
        """Override to apply discount"""
        base_total = super().calculate_total(quantity)
        return base_total * (1 - self.discount_percent / 100)

# Example usage

print(product.calculate_total(2))  # Output: 2159.978...

order = Order(order_id="ORD-001", customer_id=42)
order.add_item(product, quantity=2)
print(f"Order total: ${order.total():.2f}")

discounted = DiscountedProduct(id=2, name="Mouse", price=49.99, discount_percent=20)
print(f"Discounted price: ${discounted.calculate_total():.2f}")
```

**Why**: Eliminates boilerplate code for __init__, __repr__, __eq__. Provides type safety, immutability options, and clean serialization patterns.

## Enum Patterns

```python
# PROBLEM: Represent fixed sets of constants with type safety
# TIME: O(1) for access and comparison SPACE: O(n) for n enum members
from enum import Enum, IntEnum, Flag, IntFlag, auto
from typing import List, Optional

class OrderStatus(str, Enum):
    """String enum for order statuses - serializes to JSON"""
    PENDING = "pending"
    PROCESSING = "processing"
    SHIPPED = "shipped"
    DELIVERED = "delivered"
    CANCELLED = "cancelled"
    
    def can_transition_to(self, new_status: 'OrderStatus') -> bool:
        """Define valid state transitions"""
        transitions = {
            OrderStatus.PENDING: [OrderStatus.PROCESSING, OrderStatus.CANCELLED],
            OrderStatus.PROCESSING: [OrderStatus.SHIPPED, OrderStatus.CANCELLED],
            OrderStatus.SHIPPED: [OrderStatus.DELIVERED],
            OrderStatus.DELIVERED: [],
            OrderStatus.CANCELLED: []
        }
        return new_status in transitions.get(self, [])

class Priority(IntEnum):
    """Integer enum for priority levels"""
    LOW = 1
    MEDIUM = 2
    HIGH = 3
    CRITICAL = 4
    
    @classmethod
    def from_string(cls, value: str) -> 'Priority':
        """Create from string representation"""
        mapping = {
            'low': cls.LOW,
            'medium': cls.MEDIUM,
            'high': cls.HIGH,
            'critical': cls.CRITICAL
        }
        return mapping.get(value.lower(), cls.LOW)
```
