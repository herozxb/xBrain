## Editable Table Cells
```html
<!-- PROBLEM: Create table with inline editable cells -->
<!-- TIME: O(1) SPACE: O(1) -->
<table id="editable-table">
  <thead>
    <tr>
      <th>Name</th>
      <th>Email</th>
      <th>Role</th>
      <th>Actions</th>
    </tr>
  </thead>
  <tbody>
    <tr data-id="1">
      <td contenteditable="true" data-field="name">John Doe</td>
      <td contenteditable="true" data-field="email">john@example.com</td>
      <td contenteditable="true" data-field="role">Admin</td>
      <td><button class="delete-btn">Delete</button></td>
    </tr>
    <tr data-id="2">
      <td contenteditable="true" data-field="name">Jane Smith</td>
      <td contenteditable="true" data-field="email">jane@example.com</td>
      <td contenteditable="true" data-field="role">User</td>
      <td><button class="delete-btn">Delete</button></td>
    </tr>
  </tbody>
</table>

<script>
const table = document.getElementById('editable-table');
const changes = new Map();

// Track edits
table.addEventListener('input', (e) => {
  if (e.target.hasAttribute('contenteditable')) {
    const row = e.target.closest('tr');
    const id = row.dataset.id;
    const field = e.target.dataset.field;
    const value = e.target.textContent.trim();
    
    if (!changes.has(id)) changes.set(id, {});
    changes.get(id)[field] = value;
    
    row.classList.add('modified');
  }
});

// Delete row
table.addEventListener('click', (e) => {
  if (e.target.classList.contains('delete-btn')) {
    const row = e.target.closest('tr');
    row.style.opacity = '0.5';
    setTimeout(() => row.remove(), 200);
  }
});

// Save changes
function saveChanges() {
  const modified = Array.from(changes.entries()).map(([id, fields]) => ({
    id,
    ...fields
  }));
  console.log('Saving:', modified);
  changes.clear();
  document.querySelectorAll('.modified').forEach(el => el.classList.remove('modified'));
}
</script>
```
**Why**: Inline editing provides intuitive content modification without separate forms, improving UX for data management tasks.

## ARIA Alert Dialog
```html
<!-- PROBLEM: Create accessible modal dialog with alert role -->
<!-- TIME: O(1) SPACE: O(1) -->
<button id="open-dialog">Delete Account</button>

<div id="alert-dialog" role="alertdialog" 
     aria-modal="true" 
     aria-labelledby="dialog-title"
     aria-describedby="dialog-desc"
     hidden>
  <div class="dialog-backdrop"></div>
  <div class="dialog-content">
    <h2 id="dialog-title">Confirm Deletion</h2>
    <p id="dialog-desc">This action cannot be undone. All your data will be permanently deleted.</p>
    <div class="dialog-actions">
      <button id="confirm-btn" class="danger">Delete</button>
      <button id="cancel-btn">Cancel</button>
    </div>
  </div>
</div>

<script>
const dialog = document.getElementById('alert-dialog');
const openBtn = document.getElementById('open-dialog');
const confirmBtn = document.getElementById('confirm-btn');
const cancelBtn = document.getElementById('cancel-btn');
let previousFocus = null;

function openDialog() {
  previousFocus = document.activeElement;
  dialog.hidden = false;
  confirmBtn.focus();
  document.body.style.overflow = 'hidden';
}

function closeDialog() {
  dialog.hidden = true;
  document.body.style.overflow = '';
  previousFocus?.focus();
}

openBtn.addEventListener('click', openDialog);
cancelBtn.addEventListener('click', closeDialog);
confirmBtn.addEventListener('click', () => {
  console.log('Account deleted');
  closeDialog();
});

dialog.querySelector('.dialog-backdrop').addEventListener('click', closeDialog);

document.addEventListener('keydown', (e) => {
  if (e.key === 'Escape' && !dialog.hidden) closeDialog();
});
</script>

<style>
#alert-dialog[hidden] { display: none; }
.dialog-backdrop {
  position: fixed; inset: 0;
  background: rgba(0,0,0,0.5);
}
.dialog-content {
  position: fixed; top: 50%; left: 50%;
  transform: translate(-50%, -50%);
  background: white; padding: 2rem;
  border-radius: 8px; max-width: 400px;
}
.dialog-actions { display: flex; gap: 1rem; margin-top: 1.5rem; }
.danger { background: #dc3545; color: white; }
</style>
```
**Why**: Alert dialogs with proper ARIA roles ensure screen readers announce critical confirmations, preventing accidental destructive actions.

## Array FlatMap
```html
<!-- PROBLEM: Flatten and map arrays in single operation -->
<!-- TIME: O(n*m) where n=items, m=avg nested depth SPACE: O(n) -->
<script>
// Example data: users with multiple roles
const users = [
  { name: 'Alice', roles: ['admin', 'editor'] },
  { name: 'Bob', roles: ['viewer'] },
  { name: 'Charlie', roles: ['editor', 'moderator', 'viewer'] }
];

// Get all unique roles
const allRoles = [...new Set(
  users.flatMap(user => user.roles)
)];
console.log(allRoles); // ['admin', 'editor', 'viewer', 'moderator']

// Count users per role
const roleCounts = users
  .flatMap(user => user.roles)
  .reduce((acc, role) => {
    acc[role] = (acc[role] || 0) + 1;
    return acc;
  }, {});
console.log(roleCounts); // { admin: 1, editor: 2, viewer: 2, moderator: 1 }

// Flatten nested arrays with transformation
const sentences = ['Hello World', 'How are you'];
const words = sentences.flatMap(s => s.split(' ').map(w => w.toLowerCase()));
console.log(words); // ['hello', 'world', 'how', 'are', 'you']

// Filter and flatten simultaneously
const data = [
  { items: [1, 2, 3], active: true },
  { items: [4, 5], active: false },
  { items: [6, 7, 8], active: true }
];

const activeItems = data.flatMap(d => d.active ? d.items : []);
console.log(activeItems); // [1, 2, 3, 6, 7, 8]
</script>
```
**Why**: `flatMap` combines map and flat operations efficiently, ideal for transforming nested structures into flat arrays in one pass.

## Proxy Get Trap
```html
<!-- PROBLEM: Intercept property access for validation/logging -->
<!-- TIME: O(1) for access, O(n) for validation SPACE: O(1) -->
<script>
// API response with default values
const createAPIResponse = (data) => {
  const defaults = {
    status: 'success',
    message: 'OK',
    data: null,
    timestamp: () => new Date().toISOString()
  };
  
  return new Proxy(data, {
    get(target, prop, receiver) {
      // Return default if property missing
      if (!(prop in target)) {
        console.log(`Property "${prop}" not found, using default`);
        const value = defaults[prop];
        return typeof value === 'function' ? value() : value;
      }
      return Reflect.get(target, prop, receiver);
    }
  });
};

const response = createAPIResponse({ data: { id: 1, name: 'Test' } });
console.log(response.data);    // { id: 1, name: 'Test' }
console.log(response.status);  // 'success' (default)
console.log(response.timestamp); // ISO string (computed)

// Computed/cached properties
const createCachedProxy = (obj) => {
  const cache = new Map();
  
  return new Proxy(obj, {
    get(target, prop) {
      if (prop.startsWith('compute')) {
        if (cache.has(prop)) {
          console.log(`Cache hit: ${prop}`);
          return cache.get(prop);
        }
        console.log(`Computing: ${prop}`);
        const result = target[prop]();
        cache.set(prop, result);
        return result;
      }
      return target[prop];
    }
  });
};

const expensiveOps = createCachedProxy({
  computeHeavy: () => {
    let result = 0;
    for (let i = 0; i < 1000000; i++) result += Math.sqrt(i);
    return result;
  }
});

expensiveOps.computeHeavy; // Computes
expensiveOps.computeHeavy; // Cached
</script>
```
**Why**: Proxy get traps enable dynamic property behavior, defaults, caching, and access logging without modifying source objects.

## Event Delegation
```html
<!-- PROBLEM: Handle events for dynamic elements efficiently -->
<!-- TIME: O(1) event setup, O(log n) for target search SPACE: O(1) -->
<ul id="todo-list">
  <li data-id="1">
    <span class="task-text">Learn HTML</span>
    <button class="complete-btn">✓</button>
    <button class="delete-btn">✕</button>
  </li>
  <li data-id="2">
    <span class="task-text">Learn CSS</span>
    <button class="complete-btn">✓</button>
    <button class="delete-btn">✕</button>
  </li>
  <li data-id="3">
    <span class="task-text">Learn JavaScript</span>
    <button class="complete-btn">✓</button>
    <button class="delete-btn">✕</button>
  </li>
</ul>

<button id="add-todo">Add Task</button>

<script>
const list = document.getElementById('todo-list');
const addBtn = document.getElementById('add-todo');
let nextId = 4;

// Single handler for all list interactions
list.addEventListener('click', (e) => {
  const target = e.target;
  const li = target.closest('li');
  
  if (!li) return;
  
  if (target.classList.contains('complete-btn')) {
    li.classList.toggle('completed');
    target.textContent = li.classList.contains('completed') ? '↩' : '✓';
  }
  
  if (target.classList.contains('delete-btn')) {
    li.style.opacity = '0';
    setTimeout(() => li.remove(), 200);
  }
});

// Add new items - no new event listeners needed!
addBtn.addEventListener('click', () => {
  const li = document.createElement('li');
  li.dataset.id = nextId++;
  li.innerHTML = `
    <span class="task-text">New Task ${nextId - 1}</span>
    <button class="complete-btn">✓</button>
    <button class="delete-btn">✕</button>
  `;
  list.appendChild(li);
});
</script>

<style>
#todo-list { list-style: none; padding: 0; }
#todo-list li { 
  padding: 0.5rem; margin: 0.25rem 0; 
  background: #f8f9fa; 
  display: flex; gap: 0.5rem; align-items: center;
  transition: opacity 0.2s;
}
.completed .task-text { 
  text-decoration: line-through; 
  opacity: 0.5; 
}
button { cursor: pointer; padding: 0.25rem 0.5rem; }
</style>
```
**Why**: Event delegation attaches one listener to parent instead of many to children, automatically handling dynamically added elements and reducing memory usage.
