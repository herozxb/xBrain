## LinkedList Variants

```python
# PROBLEM: Implement a doubly linked list with sentinel nodes and reverse capability
# TIME: O(1) for insert/delete at known position, O(n) for search/reverse SPACE: O(n)
class Node:
    """Node for doubly linked list with prev/next pointers."""
    __slots__ = ('val', 'prev', 'next')
    
    def __init__(self, val=None):
        self.val = val
        self.prev = None
        self.next = None


class DoublyLinkedList:
    """Doubly linked list using sentinel head/tail nodes for simplified edge handling."""
    
    def __init__(self):
        # Sentinel nodes eliminate null checks
        self.head = Node()  # Dummy head
        self.tail = Node()  # Dummy tail
        self.head.next = self.tail
        self.tail.prev = self.head
        self._size = 0
    
    def _insert_between(self, node, pred, succ):
        """Insert node between predecessor and successor."""
        node.prev = pred
        node.next = succ
        pred.next = node
        succ.prev = node
        self._size += 1
    
    def _remove_node(self, node):
        """Remove node and return its value."""
        pred, succ = node.prev, node.next
        pred.next = succ
        succ.prev = pred
        self._size -= 1
        return node.val
    
    def append_left(self, val):
        """Add element to front of list."""
        self._insert_between(Node(val), self.head, self.head.next)
    
    def append_right(self, val):
        """Add element to end of list."""
        self._insert_between(Node(val), self.tail.prev, self.tail)
    
    def pop_left(self):
        """Remove and return first element."""
        if self._size == 0:
            raise IndexError("pop from empty list")
        return self._remove_node(self.head.next)
    
    def pop_right(self):
        """Remove and return last element."""
        if self._size == 0:
            raise IndexError("pop from empty list")
        return self._remove_node(self.tail.prev)
    
    def reverse(self):
        """Reverse the list in-place by swapping prev/next pointers."""
        curr = self.head
        # Swap prev and next for all nodes including sentinels
        while curr:
            curr.prev, curr.next = curr.next, curr.prev
            curr = curr.prev  # Move to what was next (now prev)
        # Swap sentinel references
        self.head, self.tail = self.tail, self.head
    
    def __len__(self):
        return self._size
    
    def __iter__(self):
        curr = self.head.next
        while curr is not self.tail:
            yield curr.val
            curr = curr.next
```
**Why**: Sentinel nodes simplify edge cases; reverse uses pointer swapping for O(1) space. Common interview question variant.

## Stub Object Pattern

```python
# PROBLEM: Provide minimal, canned responses for testing without real implementation
# TIME: O(1) SPACE: O(1)
from typing import Any, Optional
from dataclasses import dataclass


@dataclass
class StubConfig:
    """Configuration for stub behavior."""
    return_value: Any = None
    raise_exception: Optional[Exception] = None
    call_count: int = 0


class StubPaymentGateway:
    """Stub for payment gateway that returns canned responses."""
    
    def __init__(self, config: Optional[StubConfig] = None):
        self.config = config or StubConfig()
        self._calls = []
    
    def process_payment(self, amount: float, card_token: str) -> dict:
        """Return stubbed response or raise configured exception."""
        self._calls.append({'method': 'process_payment', 'amount': amount})
        self.config.call_count += 1
        
        if self.config.raise_exception:
            raise self.config.raise_exception
        
        return {
            'success': True,
            'transaction_id': f'STUB-{self.config.call_count}',
            'amount': amount
        }
    
    def refund(self, transaction_id: str) -> dict:
        """Return stubbed refund response."""
        self._calls.append({'method': 'refund', 'transaction_id': transaction_id})
        return {'success': True, 'refunded_id': f'REFUND-{transaction_id}'}
    
    def get_calls(self):
        """Return list of all method calls for verification."""
        return self._calls.copy()
    
    def reset(self):
        """Reset call history."""
        self._calls.clear()
        self.config.call_count = 0


# Usage in tests
class TestOrderService:
    def test_successful_order(self):
        # Arrange: Create stub with specific return value
        stub = StubPaymentGateway(StubConfig(return_value={'success': True}))
        
        # Act: Use stub in place of real gateway
        result = stub.process_payment(99.99, 'card_123')
        
        # Assert: Verify behavior
        assert result['success'] is True
        assert len(stub.get_calls()) == 1
```
**Why**: Stubs provide controlled, predictable responses for unit testing without external dependencies. Lighter than mocks.

## Fake Object Pattern

```python
# PROBLEM: Implement working in-memory version of external dependency for testing
# TIME: O(1) average for operations SPACE: O(n)
from typing import Dict, List, Optional
from collections import defaultdict


class InMemoryUserRepository:
    """Fake repository with working implementation using in-memory storage."""
    
    def __init__(self):
        self._storage: Dict[int, dict] = {}
        self._email_index: Dict[str, int] = {}
        self._next_id = 1
    
    def save(self, user: dict) -> dict:
        """Save user and return with generated ID."""
        if 'id' not in user:
            user['id'] = self._next_id
            self._next_id += 1
        
        # Update email index
        if 'email' in user:
            self._email_index[user['email']] = user['id']
        
        # Deep copy to simulate persistence
        self._storage[user['id']] = user.copy()
        return user
    
    def find_by_id(self, user_id: int) -> Optional[dict]:
        """Find user by ID."""
        return self._storage.get(user_id)
    
    def find_by_email(self, email: str) -> Optional[dict]:
        """Find user by email using index."""
        user_id = self._email_index.get(email)
        return self._storage.get(user_id) if user_id else None
    
    def find_all(self) -> List[dict]:
        """Return all users."""
        return list(self._storage.values())
    
    def delete(self, user_id: int) -> bool:
        """Delete user by ID."""
        if user_id not in self._storage:
            return False
        
        user = self._storage[user_id]
        if 'email' in user:
            self._email_index.pop(user['email'], None)
        
        del self._storage[user_id]
        return True
    
    def clear(self):
        """Reset repository state."""
        self._storage.clear()
        self._email_index.clear()
        self._next_id = 1


class InMemoryEventBus:
    """Fake event bus with actual pub/sub behavior for integration testing."""
    
    def __init__(self):
        self._handlers: Dict[str, List[callable]] = defaultdict(list)
        self._published_events: List[tuple] = []
    
    def subscribe(self, event_type: str, handler: callable):
        """Register handler for event type."""
        self._handlers[event_type].append(handler)
    
    def publish(self, event_type: str, payload: dict):
        """Publish event to all subscribers."""
        self._published_events.append((event_type, payload))
        for handler in self._handlers[event_type]:
            handler(event_type, payload)
    
    def get_published_events(self) -> List[tuple]:
        """Return all published events for verification."""
        return self._published_events.copy()
```
**Why**: Fakes have working implementations suitable for integration tests. More realistic than stubs, faster than real dependencies.

## Front Controller Pattern

```python
# PROBLEM: Centralize request handling for consistent processing across all endpoints
# TIME: O(1) for routing SPACE: O(n) for route registry
from typing import Callable, Dict, Optional, Any
from dataclasses import dataclass
from functools import wraps


@dataclass
class Request:
    """HTTP request representation."""
    method: str
    path: str
    headers: Dict[str, str]
    body: Optional[dict] = None
    query_params: Optional[Dict[str, str]] = None


@dataclass
class Response:
    """HTTP response representation."""
    status: int
    body: Any
    headers: Dict[str, str] = None
    
    def __post_init__(self):
        if self.headers is None:
            self.headers = {}


class FrontController:
    """Central dispatcher handling all incoming requests."""
    
    def __init__(self):
        self._routes: Dict[str, Dict[str, Callable]] = {}
        self._pre_interceptors: list = []
        self._post_interceptors: list = []
        self._error_handlers: Dict[type, Callable] = {}
    
    def register(self, method: str, path: str, handler: Callable):
        """Register route handler."""
        if path not in self._routes:
            self._routes[path] = {}
        self._routes[path][method.upper()] = handler
    
    def add_pre_interceptor(self, interceptor: Callable):
        """Add middleware to run before handler."""
        self._pre_interceptors.append(interceptor)
    
    def add_post_interceptor(self, interceptor: Callable):
        """Add middleware to run after handler."""
        self._post_interceptors.append(interceptor)
    
    def add_error_handler(self, exception_type: type, handler: Callable):
        """Register custom error handler for exception type."""
        self._error_handlers[exception_type] = handler
    
    def handle(self, request: Request) -> Response:
        """Main entry point for all requests."""
        try:
            # Run pre-interceptors (auth, logging, etc.)
            for interceptor in self._pre_interceptors:
                result = interceptor(request)
                if result:  # Interceptor can short-circuit
                    return result
            
            # Route dispatch
            path_routes = self._routes.get(request.path, {})
            handler = path_routes.get(request.method)
            
            if not handler:
                return Response(404, {'error': 'Not Found'})
            
            # Execute handler
            response = handler(request)
            
            # Run post-interceptors
            for interceptor in self._post_interceptors:
                response = interceptor(request, response) or response
            
            return response
            
        except Exception as e:
            # Check for custom error handler
            for exc_type, handler in self._error_handlers.items():
                if isinstance(e, exc_type):
                    return handler(e)
            # Default error response
            return Response(500, {'error': str(e)})


# Helper decorator for route registration
def route(controller: FrontController, method: str, path: str):
    """Decorator to register handler with front controller."""
    def decorator(func):
        controller.register(method, path, func)
        return func
    return decorator
```
**Why**: Single entry point enables cross-cutting concerns (auth, logging, error handling) in one place. Common in web frameworks.

## Pipeline Pattern

```python
# PROBLEM: Chain multiple processing steps where each transforms input to output
# TIME: O(n) where n is number of stages SPACE: O(1) per stage
from typing import Callable, TypeVar, Generic, List, Optional
from dataclasses import dataclass

T = TypeVar('T')


@dataclass
class PipelineContext:
    """Shared context passed through pipeline stages."""
    data: dict
    metadata: dict
    errors: List[str]
    stopped: bool = False
    
    def add_error(self, error: str):
        self.errors.append(error)
    
    def stop(self):
        """Stop pipeline execution."""
        self.stopped = True


class PipelineStage(Generic[T]):
    """Base class for pipeline stages."""
    
    def __init__(self, name: str = None):
        self.name = name or self.__class__.__name__
    
    def process(self, context: PipelineContext) -> None:
        """Process data in context. Override in subclasses."""
        raise NotImplementedError


class Pipeline:
    """Orchestrates sequential execution of stages."""
    
    def __init__(self, name: str = "Pipeline"):
        self.name = name
        self._stages: List[PipelineStage] = []
        self._before_hooks: List[Callable] = []
        self._after_hooks: List[Callable] = []
    
    def add_stage(self, stage: PipelineStage) -> 'Pipeline':
        """Add stage to pipeline. Returns self for chaining."""
        self._stages.append(stage)
        return self
    
    def add_before_hook(self, hook: Callable) -> 'Pipeline':
        """Add hook to run before each stage."""
        self._before_hooks.append(hook)
        return self
    
    def execute(self, initial_data: dict) -> PipelineContext:
        """Run all stages and return final context."""
        context = PipelineContext(
            data=initial_data,
            metadata={},
            errors=[]
        )
        
        for stage in self._stages:
            if context.stopped:
                break
            
            # Run before hooks
            for hook in self._before_hooks:
                hook(stage.name, context)
            
            try:
                stage.process(context)
            except Exception as e:
                context.add_error(f"{stage.name}: {str(e)}")
            
            # Run after hooks
            for hook in self._after_hooks:
                hook(stage.name, context)
        
        return context


# Concrete stages
class ValidationStage(PipelineStage):
    """Validate input data."""
    
    def process(self, context: PipelineContext):
        data = context.data
        required = ['email', 'name']
        
        for field in required:
            if field not in data or not data[field]:
                context.add_error(f"Missing required field: {field}")
                context.stop()


class TransformStage(PipelineStage):
    """Transform data fields."""
    
    def process(self, context: PipelineContext):
        if 'email' in context.data:
            context.data['email'] = context.data['email'].lower().strip()
        if 'name' in context.data:
            context.data['name'] = context.data['name'].title()


class EnrichmentStage(PipelineStage):
    """Add computed fields."""
    
    def process(self, context: PipelineContext):
        context.metadata['processed_at'] = '2026-02-20T09:40:00'
        if 'email' in context.data:
            context.data['domain'] = context.data['email'].split('@')[-1]


# Usage
pipeline = Pipeline("UserRegistration")
pipeline.add_stage(ValidationStage())
pipeline.add_stage(TransformStage())
pipeline.add_stage(EnrichmentStage())

result = pipeline.execute({'email': '  TEST@EXAMPLE.COM  ', 'name': 'john doe'})
```
**Why**: Decomposes complex processing into testable, reusable stages. Easy to add/remove/reorder steps.

## Load Balancing Algorithms

```python
# PROBLEM: Distribute requests across multiple servers using various strategies
# TIME: O(1) for most algorithms, O(n) for least connections SPACE: O(n)
import random
import hashlib
from typing import List, Dict
from dataclasses import dataclass, field
from collections import defaultdict


@dataclass
class Server:
    """Backend server representation."""
    name: str
    address: str
    weight: int = 1
    alive: bool = True
    connections: int = 0


class RoundRobinBalancer:
    """Simple round-robin load balancer."""
    
    def __init__(self, servers: List[Server]):
        self.servers = servers
        self._current = 0
    
    def select(self, request: dict = None) -> Server:
        """Select next server in rotation."""
        alive_servers = [s for s in self.servers if s.alive]
        if not alive_servers:
            raise Exception("No available servers")
        
        server = alive_servers[self._current % len(alive_servers)]
        self._current += 1
        return server


class WeightedRoundRobinBalancer:
    """Round-robin with server weights."""
    
    def __init__(self, servers: List[Server]):
        self.servers = servers
        self._expanded = []
        self._build_expanded_list()
    
    def _build_expanded_list(self):
        """Build weighted list where each server appears weight times."""
        self._expanded = []
        for server in self.servers:
            if server.alive:
                self._expanded.extend([server] * server.weight)
    
    def select(self, request: dict = None) -> Server:
        """Select server based on weights."""
        if not self._expanded:
            self._build_expanded_list()
        
        if not self._expanded:
            raise Exception("No available servers")
        
        return random.choice(self._expanded)


class LeastConnectionsBalancer:
    """Select server with fewest active connections."""
    
    def __init__(self, servers: List[Server]):
        self.servers = servers
    
    def select(self, request: dict = None) -> Server:
        """Find server with minimum connections."""
        alive = [s for s in self.servers if s.alive]
        if not alive:
            raise Exception("No available servers")
        
        return min(alive, key=lambda s: s.connections)


class ConsistentHashBalancer:
    """Route requests based on hash of request key."""
    
    def __init__(self, servers: List[Server], replicas: int = 150):
        self.servers = servers
        self.replicas = replicas
        self._ring: Dict[int, Server] = {}
        self._build_ring()
    
    def _build_ring(self):
        """Build consistent hash ring with virtual nodes."""
        for server in self.servers:
            if server.alive:
                for i in range(self.replicas):
                    key = self._hash(f"{server.name}:{i}")
                    self._ring[key] = server
    
    def _hash(self, key: str) -> int:
        """Generate consistent hash for key."""
        return int(hashlib.md5(key.encode()).hexdigest(), 16)
    
    def select(self, request: dict) -> Server:
        """Select server based on request key hash."""
        if not self._ring:
            raise Exception("No available servers")
        
        # Use provided key or default
        req_key = request.get('key', str(random.random()))
        hash_val = self._hash(req_key)
        
        # Find first server with hash >= request hash
        sorted_keys = sorted(self._ring.keys())
        for key in sorted_keys:
            if key >= hash_val:
                return self._ring[key]
        
        # Wrap around to first server
        return self._ring[sorted_keys[0]]


class IPHashBalancer:
    """Route based on client IP address."""
    
    def __init__(self, servers: List[Server]):
        self.servers = servers
    
    def select(self, request: dict) -> Server:
        """Select server based on client IP hash."""
        alive = [s for s in self.servers if s.alive]
        if not alive:
            raise Exception("No available servers")
        
        client_ip = request.get('client_ip', '127.0.0.1')
        hash_val = int(hashlib.md5(client_ip.encode()).hexdigest(), 16)
        return alive[hash_val % len(alive)]
```
**Why**: Different algorithms suit different needs - RR for simplicity, least connections for uneven loads, consistent hashing for cache locality.

## RabbitMQ Patterns

```python
# PROBLEM: Implement common messaging patterns with RabbitMQ
# TIME: O(1) for publish, O(n) for batch consume SPACE: O(1)
import json
from typing import Callable, Optional, Dict, Any
from dataclasses import dataclass


@dataclass
class Message:
    """Message wrapper with metadata."""
    body: Any
    headers: Dict[str, str]
    routing_key: str
    exchange: str = ''


class RabbitMQPublisher:
    """Publisher with exchange and routing patterns."""
    
    def __init__(self, channel):
        self.channel = channel
    
    def publish_direct(self, queue: str, message: dict, correlation_id: str = None):
        """Publish to direct exchange (point-to-point)."""
        self.channel.basic_publish(
            exchange='',
            routing_key=queue,
            body=json.dumps(message),
            properties={'correlation_id': correlation_id}
        )
    
    def publish_fanout(self, exchange: str, message: dict):
        """Publish to fanout exchange (broadcast to all queues)."""
        self.channel.basic_publish(
            exchange=exchange,
            routing_key='',
            body=json.dumps(message)
        )
    
    def publish_topic(self, exchange: str, routing_key: str, message: dict):
        """Publish to topic exchange (pattern-based routing)."""
        # routing_key patterns: "user.created", "order.*.shipped", "#.error"
        self.channel.basic_publish(
            exchange=exchange,
            routing_key=routing_key,
            body=json.dumps(message)
        )
    
    def publish_delayed(self, queue: str, message: dict, delay_ms: int):
        """Publish with delay using dead-letter exchange pattern."""
        # Requires x-delayed-message plugin or DLX setup
        self.channel.basic_publish(
            exchange='',
            routing_key=f"{queue}.delayed",
            body=json.dumps(message),
            properties={'headers': {'x-delay': delay_ms}}
        )


class RabbitMQConsumer:
    """Consumer with acknowledgment patterns."""
    
    def __init__(self, channel):
        self.channel = channel
    
    def consume_simple(self, queue: str, handler: Callable[[dict], bool]):
        """Simple consumer with manual acknowledgment."""
        def callback(ch, method, properties, body):
            message = json.loads(body)
            try:
                success = handler(message)
                if success:
                    ch.basic_ack(delivery_tag=method.delivery_tag)
                else:
                    ch.basic_nack(delivery_tag=method.delivery_tag, requeue=True)
            except Exception as e:
                # Reject and don't requeue on error
                ch.basic_nack(delivery_tag=method.delivery_tag, requeue=False)
        
        self.channel.basic_consume(queue=queue, on_message_callback=callback)
        self.channel.start_consuming()
    
    def consume_with_prefetch(self, queue: str, handler: Callable, prefetch: int = 10):
        """Consumer with prefetch for fair dispatch."""
        self.channel.basic_qos(prefetch_count=prefetch)
        self.consume_simple(queue, handler)


class DeadLetterHandler:
    """Handle failed messages with dead-letter queue pattern."""
    
    def __init__(self, channel, main_queue: str, dlq: str):
        self.channel = channel
        self.main_queue = main_queue
        self.dlq = dlq
    
    def setup(self):
        """Create main queue with DLX configured."""
        # Dead letter exchange routes failed messages to DLQ
        self.channel.exchange_declare('dlx', exchange_type='fanout')
        self.channel.queue_declare(self.dlq)
        self.channel.queue_bind(self.dlq, 'dlx')
        
        # Main queue with DLX
        self.channel.queue_declare(
            self.main_queue,
            arguments={
                'x-dead-letter-exchange': 'dlx',
                'x-message-ttl': 30000  # Retry after 30s
            }
        )


class RequestReplyPattern:
    """RPC-style request-reply pattern."""
    
    def __init__(self, channel):
        self.channel = channel
        self._callbacks: Dict[str, Callable] = {}
    
    def send_request(self, queue: str, request: dict, reply_to: str, callback: Callable):
        """Send request and register callback for response."""
        corr_id = str(hash(str(request)))
        self._callbacks[corr_id] = callback
        
        self.channel.basic_publish(
            exchange='',
            routing_key=queue,
            body=json.dumps(request),
            properties={
                'reply_to': reply_to,
                'correlation_id': corr_id
            }
        )
    
    def handle_reply(self, ch, method, properties, body):
        """Route reply to registered callback."""
        corr_id = properties.correlation_id
        if corr_id in self._callbacks:
            response = json.loads(body)
            self._callbacks[corr_id](response)
            del self._callbacks[corr_id]
            ch.basic_ack(delivery_tag=method.delivery_tag)
```
**Why**: RabbitMQ patterns enable reliable async communication, decoupling services, and handling failures gracefully.

## Pessimistic Locking

```python
# PROBLEM: Prevent concurrent modifications by acquiring locks before reading
# TIME: O(1) for lock acquisition SPACE: O(n) for lock tracking
import threading
import time
from typing import Dict, Optional, Set
from dataclasses import dataclass, field
from enum import Enum
from contextlib import contextmanager


class LockType(Enum):
    SHARED = 'shared'      # Read lock - multiple readers allowed
    EXCLUSIVE = 'exclusive'  # Write lock - single writer only


@dataclass
class Lock:
    """Represents a lock on a resource."""
    resource_id: str
    holder: str
    lock_type: LockType
    acquired_at: float = field(default_factory=time.time)


class PessimisticLockManager:
    """Manages pessimistic locks on database records."""
    
    def __init__(self, lock_timeout: float = 30.0):
        self._locks: Dict[str, Lock] = {}
        self._lock_holders: Dict[str, Set[str]] = {}  # holder -> set of resources
        self._lock = threading.RLock()
        self._timeout = lock_timeout
    
    def acquire(self, resource_id: str, holder: str, 
                lock_type: LockType = LockType.EXCLUSIVE,
                timeout: float = None) -> bool:
        """Acquire lock on resource, blocking until available."""
        deadline = time.time() + (timeout or self._timeout)
        
        while time.time() < deadline:
            with self._lock:
                existing = self._locks.get(resource_id)
                
                # No lock exists
                if not existing:
                    self._grant_lock(resource_id, holder, lock_type)
                    return True
                
                # Shared lock requested and existing is shared
                if lock_type == LockType.SHARED and existing.lock_type == LockType.SHARED:
                    # Allow multiple shared locks
                    self._grant_lock(resource_id, holder, lock_type)
                    return True
                
                # Same holder already has lock
                if existing.holder == holder:
                    return True
            
            # Wait and retry
            time.sleep(0.01)
        
        return False  # Timeout
    
    def _grant_lock(self, resource_id: str, holder: str, lock_type: LockType):
        """Internal method to grant lock."""
        self._locks[resource_id] = Lock(resource_id, holder, lock_type)
        if holder not in self._lock_holders:
            self._lock_holders[holder] = set()
        self._lock_holders[holder].add(resource_id)
    
    def release(self, resource_id: str, holder: str) -> bool:
        """Release lock held by holder."""
        with self._lock:
            lock = self._locks.get(resource_id)
            if lock and lock.holder == holder:
                del self._locks[resource_id]
                self._lock_holders[holder].discard(resource_id)
                return True
            return False
    
    def release_all(self, holder: str):
        """Release all locks held by a holder."""
        with self._lock:
            resources = self._lock_holders.get(holder, set()).copy()
            for resource_id in resources:
                self._locks.pop(resource_id, None)
            self._lock_holders.pop(holder, None)
    
    @contextmanager
    def lock_context(self, resource_id: str, holder: str, 
                     lock_type: LockType = LockType.EXCLUSIVE):
        """Context manager for automatic lock release."""
        acquired = self.acquire(resource_id, holder, lock_type)
        if not acquired:
            raise TimeoutError(f"Could not acquire lock on {resource_id}")
        try:
            yield
        finally:
            self.release(resource_id, holder)


class PessimisticRepository:
    """Repository with pessimistic locking for concurrent access."""
    
    def __init__(self, lock_manager: PessimisticLockManager):
        self._data: Dict[int, dict] = {}
        self._lock_manager = lock_manager
    
    def get_for_update(self, record_id: int, holder: str) -> Optional[dict]:
        """Get record with exclusive lock (SELECT FOR UPDATE pattern)."""
        with self._lock_manager.lock_context(f"record:{record_id}", holder):
            return self._data.get(record_id)
    
    def update(self, record_id: int, holder: str, updates: dict) -> bool:
        """Update record with lock held during modification."""
        with self._lock_manager.lock_context(f"record:{record_id}", holder):
            if record_id not in self._data:
                return False
            self._data[record_id].update(updates)
            return True
    
    def transfer(self, from_id: int, to_id: int, holder: str, amount: float) -> bool:
        """Transfer between records with lock ordering to prevent deadlock."""
        # Always lock in consistent order (by ID)
        first_id, second_id = sorted([from_id, to_id])
        
        with self._lock_manager.lock_context(f"record:{first_id}", holder):
            with self._lock_manager.lock_context(f"record:{second_id}", holder):
                # Both locks held, safe to transfer
                if self._data[from_id]['balance'] < amount:
                    return False
                self._data[from_id]['balance'] -= amount
                self._data[to_id]['balance'] += amount
                return True
```
**Why**: Pessimistic locking prevents conflicts before they occur, ideal for high-contention scenarios where collisions are likely.

                      max_retries: int = 3,
                      delay: float = 1.0,
                      backoff: float = 2.0) -> Any:

    current_delay = delay

        try:
            return await coro_factory()
        except Exception as e:
            last_error = e
            if attempt < max_retries - 1:
                await asyncio.sleep(current_delay)
                current_delay *= backoff
    
    raise last_error

                            queue_size: int = 100,
                            num_consumers: int = 3):

    async def producer():
        async for item in producer_func():
            await queue.put(item)
        # Signal completion
        for _ in range(num_consumers):
            await queue.put(None)
    
    async def consumer(consumer_id: int):
        while True:
            item = await queue.get()
            if item is None:
                break
            await consumer_func(item, consumer_id)

    await asyncio.gather(
        producer(),
        *[consumer(i) for i in range(num_consumers)]

class EventLoopManager:
    """Manages event loop lifecycle and task orchestration."""
    
    def __init__(self):
        self._tasks: Set[asyncio.Task] = set()
        self._shutdown_handlers: list = []
    
    def create_tracked_task(self, coro, name: str = None) -> asyncio.Task:
        """Create task that's automatically tracked for cleanup."""
        task = asyncio.create_task(coro)
        task.set_name(name or f"task-{id(task)}")
        self._tasks.add(task)
        task.add_done_callback(self._tasks.discard)
        return task
    
    def add_shutdown_handler(self, handler: Callable):
        """Register handler to run on shutdown."""
        self._shutdown_handlers.append(handler)
    
    async def graceful_shutdown(self, timeout: float = 10.0):
        """Cancel all tasks and run shutdown handlers."""
        # Cancel all tracked tasks
        for task in self._tasks:
            task.cancel()
        
        # Wait for tasks to complete or timeout
        if self._tasks:
            try:
                await asyncio.wait_for(
                    asyncio.gather(*self._tasks, return_exceptions=True),
                    timeout=timeout
                )
            except asyncio.TimeoutError:
                pass
        
        # Run shutdown handlers
        for handler in self._shutdown_handlers:
            try:
                if asyncio.iscoroutinefunction(handler):
                    await handler()
                else:
                    handler()
            except Exception:
                pass  # Don't fail shutdown on handler error


class PeriodicTask:
    """Run coroutine periodically on event loop."""
    
    def __init__(self, coro_factory: Callable[[], asyncio.coroutine], 
                 interval: float,
                 loop: asyncio.AbstractEventLoop = None):
        self._coro_factory = coro_factory
        self._interval = interval
        self._loop = loop or asyncio.get_event_loop()
        self._task: Optional[asyncio.Task] = None
        self._running = False
    
    def start(self):
        """Start periodic execution."""
        if self._running:
            return
        self._running = True
        self._task = asyncio.create_task(self._run_periodically())
    
    async def _run_periodically(self):
        """Internal loop for periodic execution."""
        while self._running:
            try:
                await self._coro_factory()
            except asyncio.CancelledError:
                break
            except Exception:
                pass  # Log in production
            await asyncio.sleep(self._interval)
    
    def stop(self):
        """Stop periodic execution."""
        self._running = False
        if self._task:
            self._task.cancel()

async def throttle(coro_factory: Callable[[], asyncio.coroutine],
                   min_interval: float,
                   last_run: list = None) -> Any:
    """Throttle coroutine execution to minimum interval."""
    if last_run is None:
        last_run = [0]
    
    now = asyncio.get_event_loop().time()
    elapsed = now - last_run[0]
    
    if elapsed < min_interval:
        await asyncio.sleep(min_interval - elapsed)
    
    last_run[0] = asyncio.get_event_loop().time()
    return await coro_factory()

                   wait: float) -> Any:

        await asyncio.sleep(wait)
        return await coro_factory()

    if task_ref[0] and not task_ref[0].done():
        task_ref[0].cancel()
    
    task_ref[0] = asyncio.create_task(debounced())
