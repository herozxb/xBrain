## Reservoir Sampling

```python
# PROBLEM: Select k random elements from a stream of unknown size with uniform probability
# TIME: O(n) SPACE: O(k)
import random

def reservoir_sampling(stream, k):
    """
    Randomly select k elements from a stream where n is unknown.
    Each element has equal probability k/n of being selected.
    """
    reservoir = []
    
    # Fill reservoir with first k elements
    for i, item in enumerate(stream):
        if i < k:
            reservoir.append(item)
        else:
            # Generate random index from 0 to i (inclusive)
            j = random.randint(0, i)
            # Replace with probability k/(i+1)
            if j < k:
                reservoir[j] = item
    
    return reservoir
```
**Why**: Essential for random sampling from data streams, logs, or datasets too large for memory. Used in big data systems and analytics.

## Boyer-Moore Voting Algorithm

```python
# PROBLEM: Find the majority element (appears > n/2 times) in linear time and O(1) space
# TIME: O(n) SPACE: O(1)
def boyer_moore_majority(nums):
    """
    Find majority element using Boyer-Moore voting algorithm.
    Returns None if no majority element exists.
    """
    # Phase 1: Find candidate
    candidate = None
    count = 0
    
    for num in nums:
        if count == 0:
            candidate = num
            count = 1
        elif num == candidate:
            count += 1
        else:
            count -= 1
    
    # Phase 2: Verify candidate is actually majority
    if candidate is not None:
        count = sum(1 for num in nums if num == candidate)
        if count > len(nums) // 2:
            return candidate
    
    return None
```
**Why**: Optimal solution for majority element problems. No hash map needed. Extends to find elements appearing > n/k times.

## Floyd's Cycle Detection

```python
# PROBLEM: Detect if a linked list has a cycle and find the cycle start
# TIME: O(n) SPACE: O(1)
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def detect_cycle(head):
    """
    Floyd's Tortoise and Hare algorithm.
    Returns the node where cycle begins, or None if no cycle.
    """
    if not head or not head.next:
        return None
    
    # Phase 1: Detect if cycle exists
    slow = head
    fast = head
    
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        
        if slow == fast:
            # Cycle detected, find start of cycle
            # Phase 2: Reset slow to head, move both at same speed
            slow = head
            while slow != fast:
                slow = slow.next
                fast = fast.next
            return slow
    
    return None  # No cycle
```
**Why**: Elegant O(1) space cycle detection. Also finds duplicate numbers in arrays (treat array as linked list). Classic interview problem.

## Kadane's Algorithm

```python
# PROBLEM: Find the maximum sum contiguous subarray
# TIME: O(n) SPACE: O(1)
def kadane_max_subarray(nums):
    """
    Find maximum sum of any contiguous subarray.
    Returns (max_sum, start_index, end_index).
    """
    if not nums:
        return 0, -1, -1
    
    max_sum = nums[0]
    current_sum = nums[0]
    start = 0
    end = 0
    temp_start = 0
    
    for i in range(1, len(nums)):
        # Either extend current subarray or start new one
        if current_sum + nums[i] > nums[i]:
            current_sum = current_sum + nums[i]
        else:
            current_sum = nums[i]
            temp_start = i
        
        # Update max and track indices
        if current_sum > max_sum:
            max_sum = current_sum
            start = temp_start
            end = i
    
    return max_sum, start, end


def kadane_circular(nums):
    """
    Maximum subarray sum for circular array.
    """
    # Case 1: Max is within array (standard Kadane)
    max_kadane = kadane_max_subarray(nums)[0]
    
    # Case 2: Max wraps around (total - min subarray)
    total = sum(nums)
    # Invert and find max (which is min of original)
    inverted = [-x for x in nums]
    min_subarray = -kadane_max_subarray(inverted)[0]
    
    max_wrap = total - min_subarray
    
    # Handle all-negative case
    if max_wrap == 0:
        return max_kadane
    
    return max(max_kadane, max_wrap)
```
**Why**: Fundamental DP-like algorithm. Handles circular arrays too. Appears frequently in coding interviews and competitive programming.

## Dutch National Flag (3-Way Partition)

```python
# PROBLEM: Sort array with 3 distinct values (e.g., 0s, 1s, 2s) in one pass
# TIME: O(n) SPACE: O(1)
def dutch_national_flag(nums):
    """
    Sort array of 0s, 1s, and 2s in-place.
    Maintains three regions: [0, low-1], [low, mid-1], [high+1, end].
    """
    low = 0
    mid = 0
    high = len(nums) - 1
    
    while mid <= high:
        if nums[mid] == 0:
            # Swap to low region
            nums[low], nums[mid] = nums[mid], nums[low]
            low += 1
            mid += 1
        elif nums[mid] == 1:
            # Already in correct region
            mid += 1
        else:  # nums[mid] == 2
            # Swap to high region
            nums[mid], nums[high] = nums[high], nums[mid]
            high -= 1
            # Don't increment mid, need to check swapped element
    
    return nums
```
**Why**: Single-pass sort for arrays with limited distinct values. Extends to partition around pivot with equal elements together. Optimal for the "sort colors" problem.

## Ternary Search

```python
# PROBLEM: Find maximum/minimum in a unimodal function
# TIME: O(log n) SPACE: O(1)
def ternary_search_max(f, left, right, epsilon=1e-9):
    """
    Find maximum of unimodal function f in [left, right].
    Unimodal: increases then decreases (or just one of them).
    """
    while right - left > epsilon:
        mid1 = left + (right - left) / 3
        mid2 = right - (right - left) / 3
        
        if f(mid1) < f(mid2):
            # Maximum is in right third
            left = mid1
        else:
            # Maximum is in left third
            right = mid2
    
    return (left + right) / 2


def ternary_search_max_discrete(arr):
    """
    Find peak element in a bitonic array (strictly increasing then decreasing).
    """
    left, right = 0, len(arr) - 1
    
    while left < right:
        mid = (left + right) // 2
        
        if arr[mid] < arr[mid + 1]:
            # Peak is on the right
            left = mid + 1
        else:
            # Peak is on the left or at mid
            right = mid
    
    return arr[left]
```
**Why**: Efficient for finding extrema in unimodal functions. Used in optimization problems, computational geometry, and numerical methods.

## Jump Search

```python
# PROBLEM: Search in sorted array by jumping ahead by fixed steps
# TIME: O(√n) SPACE: O(1)
import math

def jump_search(arr, target):
    """
    Search by jumping √n elements, then linear search in block.
    Only works on sorted arrays.
    """
    n = len(arr)
    if n == 0:
        return -1
    
    # Optimal jump size is √n
    jump = int(math.sqrt(n))
    
    # Find the block where target might be
    prev = 0
    curr = jump
    
    while curr < n and arr[curr] < target:
        prev = curr
        curr += jump
    
    # Linear search in the identified block
    # Limit to prevent going out of bounds
    end = min(curr, n)
    for i in range(prev, end):
        if arr[i] == target:
            return i
    
    return -1
```
**Why**: Middle ground between linear and binary search. Useful when jumping back is expensive (e.g., linked lists, external memory). Simpler than binary search with reasonable performance.

## Interpolation Search

```python
# PROBLEM: Search in uniformly distributed sorted array using interpolation
# TIME: O(log log n) average, O(n) worst SPACE: O(1)
def interpolation_search(arr, target):
    """
    Estimate position using linear interpolation.
    Works best on uniformly distributed data.
    """
    low = 0
    high = len(arr) - 1
    
    while low <= high and arr[low] <= target <= arr[high]:
        if low == high:
            return low if arr[low] == target else -1
        
        # Estimate position using interpolation formula
        # pos = low + ((target - arr[low]) * (high - low)) / (arr[high] - arr[low])
        pos = low + int(
            (target - arr[low]) * (high - low) / (arr[high] - arr[low])
        )
        
        if arr[pos] == target:
            return pos
        elif arr[pos] < target:
            low = pos + 1
        else:
            high = pos - 1
    
    return -1
```
**Why**: Outperforms binary search on uniformly distributed data (O(log log n) vs O(log n)). Used in database indexing and searching phone books/dictionaries.

## Exponential Search

```python
# PROBLEM: Search in unbounded/infinite sorted array
# TIME: O(log n) SPACE: O(1)
def exponential_search(arr, target):
    """
    Find range by doubling index, then binary search within range.
    Useful for unbounded arrays or when length is unknown.
    """
    n = len(arr)
    if n == 0:
        return -1
    
    # If first element is target
    if arr[0] == target:
        return 0
    
    # Find range for binary search by doubling
    i = 1
    while i < n and arr[i] <= target:
        i *= 2
    
    # Binary search in range [i//2, min(i, n-1)]
    return binary_search_range(arr, target, i // 2, min(i, n - 1))


def binary_search_range(arr, target, left, right):
    """Standard binary search within a specific range."""
    while left <= right:
        mid = (left + right) // 2
        
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    
    return -1
```
**Why**: Best for searching unbounded arrays (like searching in infinite sorted stream). Also useful when element is near the beginning. Combines well with binary search.

## Fisher-Yates Shuffle

```python
# PROBLEM: Generate uniformly random permutation of an array
# TIME: O(n) SPACE: O(1) in-place
import random

def fisher_yates_shuffle(arr):
    """
    In-place uniform random shuffle.
    Each permutation has equal probability 1/n!.
    """
    # Iterate from last to first element
    for i in range(len(arr) - 1, 0, -1):
        # Pick random index from 0 to i (inclusive)
        j = random.randint(0, i)
        # Swap
        arr[i], arr[j] = arr[j], arr[i]
    
    return arr


def reservoir_shuffle-equivalent(stream):
    """
    Alternative: shuffle using reservoir sampling concept.
    Each position gets random element from remaining pool.
    """
    arr = list(stream)
    for i in range(len(arr)):
        # Pick random index from i to end
        j = random.randint(i, len(arr) - 1)
        arr[i], arr[j] = arr[j], arr[i]
    
    return arr
```
**Why**: Only correct way to shuffle uniformly. Naive approaches create bias. Used in card games, random sampling, and Monte Carlo simulations. Simple but frequently implemented incorrectly.
