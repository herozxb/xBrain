```html
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>JavaScript Classes: Methods</title>
</head>
<body>
  <!--
    PROBLEM: Define reusable behavior in classes
    TIME: O(1) per method call SPACE: O(n) for instance data
    
    Why: Methods are functions defined inside classes. They access
    instance data via 'this'. Prototype methods are shared across
    all instances, saving memory compared to per-instance functions.
  -->
  <div id="output"></div>

  <script>
    class BankAccount {
      // Constructor initializes instance properties
      constructor(owner, initialBalance = 0) {
        this.owner = owner;
        this.balance = initialBalance;
        this.transactions = [];
      }
      
      // Instance method - deposit money
      deposit(amount) {
        if (amount <= 0) {
          throw new Error('Deposit must be positive');
        }
        this.balance += amount;
        this.transactions.push({ type: 'deposit', amount, date: new Date() });
        return this.balance; // Enable method chaining
      }
      
      // Instance method - withdraw money
      withdraw(amount) {
        if (amount <= 0) {
          throw new Error('Withdrawal must be positive');
        }
        if (amount > this.balance) {
          throw new Error('Insufficient funds');
        }
        this.balance -= amount;
        this.transactions.push({ type: 'withdrawal', amount, date: new Date() });
        return this.balance;
      }
      
      // Getter method - computed property
      get accountInfo() {
        return `${this.owner}: $${this.balance.toFixed(2)}`;
      }
      
      // Method returning transaction history
      getHistory() {
        return this.transactions.map(t => 
          `${t.date.toLocaleDateString()}: ${t.type} $${t.amount}`
        );
      }
    }
    
    // Usage demonstration
    const account = new BankAccount('Alice', 1000);
    
    // Method chaining (methods return this or useful values)
    account.deposit(500).withdraw(200);
    
    const output = document.getElementById('output');
    output.innerHTML = `
      <h2>Bank Account Demo</h2>
      <p><strong>Account:</strong> ${account.accountInfo}</p>
      <p><strong>Balance:</strong> $${account.balance}</p>
      <h3>Transaction History:</h3>
      <ul>
        ${account.getHistory().map(t => `<li>${t}</li>`).join('')}
      </ul>
    `;
    
    // Verify method is on prototype (shared)
    console.log(account.hasOwnProperty('deposit')); // false
    console.log(BankAccount.prototype.hasOwnProperty('deposit')); // true
  </script>
</body>
```

```html
      // Common method for all animals
      describe() {
        return `${this.name} is ${this.age} years old`;
      }

        return `${this.name} is sleeping`;
      }
    }
    
    // Child class - inherits from Animal

        super(name, age); // Must call super() before using 'this'

      }
      
      // Override parent method
      speak() {
        return `${this.name} barks: Woof!`;
      }
      
      // Extend parent functionality
      sleep() {
        return `${super.sleep()} on the couch`; // Call parent method
      }
      
      // New method specific to Dog

        return `${this.name} fetches the ball`;
      }
    }
```

```html
  <div id="output"></div>

  <script>
    // Define some symbols
    const secretKey = Symbol('secret');
    const metaKey = Symbol('meta');
    const privateMethod = Symbol('privateMethod');
    
    // Create object with mixed property types
    const user = {
      name: 'Alice',           // String key
      age: 30,                 // String key
      [secretKey]: 'hidden',   // Symbol key
      [metaKey]: {             // Symbol key with object value
        created: new Date(),
        version: 1
      },
      [privateMethod]() {      // Symbol method
        return 'This is private';
      }
    };
    
    // Add another symbol property
    const anotherSymbol = Symbol('extra');
    user[anotherSymbol] = 'extra value';

    const symbols = Object.getOwnPropertySymbols(user);
```

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">

  <title>JavaScript: Object.getOwnPropertyDescriptors()</title>
</head>
<body>
  <!--
    PROBLEM: Get full property descriptors (writable, enumerable, configurable, value/get/set)
    TIME: O(n) SPACE: O(n)
    
    Why: Unlike Object.getOwnPropertyDescriptor() (single property), this returns
    ALL descriptors. Essential for: cloning objects with getters/setters,
    preserving property attributes, implementing mixins, deep copying.
  -->
  <div id="output"></div>

  <script>
    // Create object with various property types
    const source = {
      // Normal property
      name: 'Alice',
      
      // Non-enumerable property
      id: 123
    };
    
    // Make id non-enumerable
    Object.defineProperty(source, 'id', {
      enumerable: false,
      writable: false
    });
    
    // Add a getter/setter
    let _age = 25;
    Object.defineProperty(source, 'age', {
      get() { return _age; },
      set(value) { 
        if (value >= 0) _age = value; 
      },
      enumerable: true,
      configurable: true
    });
    
    // Add a Symbol property
    const secret = Symbol('secret');
    source[secret] = 'hidden value';
    
    // Get all property descriptors
    const descriptors = Object.getOwnPropertyDescriptors(source);
    
    // Clone object preserving all attributes
    const clone = Object.create(
      Object.getPrototypeOf(source),
      descriptors
    );
    
    // Utility function: true clone with getters/setters
    function deepCloneWithDescriptors(obj) {
      const descriptors = Object.getOwnPropertyDescriptors(obj);
      const clone = Object.create(Object.getPrototypeOf(obj), descriptors);
      
      // Handle nested objects (shallow for demo)
      for (const key in descriptors) {
        if (typeof descriptors[key].value === 'object' && descriptors[key].value !== null) {
          clone[key] = deepCloneWithDescriptors(descriptors[key].value);
        }
      }
      return clone;
    }
    
    const output = document.getElementById('output');
    
    // Format descriptor for display
    function formatDescriptor(desc) {
      const parts = [];
      if ('value' in desc) parts.push(`value: ${JSON.stringify(desc.value)}`);
      if ('get' in desc) parts.push('get: function');
      if ('set' in desc) parts.push('set: function');
      parts.push(`writable: ${desc.writable}`);
      parts.push(`enumerable: ${desc.enumerable}`);
      parts.push(`configurable: ${desc.configurable}`);
      return parts.join(', ');
    }
    
    let descriptorsHTML = '';
    for (const [key, desc] of Object.entries(descriptors)) {
      descriptorsHTML += `
        <tr>
          <td><strong>${String(key)}</strong></td>
          <td><code>${formatDescriptor(desc)}</code></td>
        </tr>
      `;
    }
    
    output.innerHTML = `
      <h2>Object.getOwnPropertyDescriptors() Demo</h2>
      
      <h3>Source Object Descriptors:</h3>
      <table border="1" cellpadding="8">
        <tr><th>Property</th><th>Descriptor</th></tr>
        ${descriptorsHTML}
      </table>
      
      <h3>Verification:</h3>
      <p>Clone age (getter): ${clone.age}</p>
      <p>Clone id enumerable: ${clone.propertyIsEnumerable('id')}</p>
      
      <h3>Key Differences:</h3>
      <ul>
```
