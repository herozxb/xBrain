## 1. HTML Input Reset Button

```html
<!-- PROBLEM: Create a form reset button that clears all inputs
     TIME: O(1) SPACE: O(1) -->
<form id="userForm">
  <label for="username">Username:</label>
  <input type="text" id="username" name="username" placeholder="Enter username">
  
  <label for="email">Email:</label>
  <input type="email" id="email" name="email" placeholder="Enter email">
  
  <!-- Reset button clears all form fields to initial values -->
  <input type="reset" value="Clear Form">
  
  <button type="submit">Submit</button>
</form>

<script>
  // Reset button behavior - fires when form is reset
  document.getElementById('userForm').addEventListener('reset', (e) => {
    console.log('Form has been reset to default values');
    // e.preventDefault(); // Uncomment to prevent actual reset
  });
</script>
```
**Why**: The `<input type="reset">` provides a native way to clear form data without JavaScript, resetting all fields to their initial values.

## 2. HTML Select Element

```html
<!-- PROBLEM: Create a dropdown selection with options
     TIME: O(1) SPACE: O(n) where n is number of options -->
<form id="selectionForm">
  <label for="country">Select Country:</label>
  <select id="country" name="country" required>
    <option value="" disabled selected>Choose a country...</option>
    <option value="us">United States</option>
    <option value="uk">United Kingdom</option>
    <option value="cn">China</option>
    <option value="jp">Japan</option>
    <option value="de">Germany</option>
  </select>
  
  <button type="submit">Submit</button>
</form>

<script>
  const select = document.getElementById('country');
  
  // Listen for selection changes
  select.addEventListener('change', (e) => {
    const selectedValue = e.target.value;
    const selectedText = e.target.options[e.target.selectedIndex].text;
    console.log(`Selected: ${selectedText} (${selectedValue})`);
  });
  
  // Programmatically set value
  function setCountry(code) {
    select.value = code;
    select.dispatchEvent(new Event('change'));
  }
</script>
```
**Why**: `<select>` provides accessible dropdown navigation with built-in keyboard support and form integration.

## 3. HTML Button Element

```html
<!-- PROBLEM: Create interactive buttons with different types
     TIME: O(1) SPACE: O(1) -->
<div class="button-group">
```

## 5. HTML Table Foot Element

```html
<!-- PROBLEM: Create table footer for summaries or totals
     TIME: O(n) SPACE: O(n) where n is number of columns -->
<table id="salesTable">
  <thead>
    <tr>
      <th>Product</th>
      <th>Quantity</th>
      <th>Price</th>
    </tr>
  </thead>
  <tbody>
    <tr>
      <td>Widget A</td>
      <td class="qty">10</td>
      <td class="price">$5.00</td>
    </tr>
    <tr>
      <td>Widget B</td>
      <td class="qty">5</td>
      <td class="price">$12.00</td>
    </tr>
    <tr>
      <td>Widget C</td>
      <td class="qty">8</td>
      <td class="price">$7.50</td>
    </tr>
  </tbody>
  <tfoot>
    <tr>
      <td><strong>Total</strong></td>
      <td id="totalQty"><strong>23</strong></td>
      <td id="totalPrice"><strong>$152.00</strong></td>
    </tr>
  </tfoot>
</table>

<script>
  // Calculate totals dynamically
  function updateTotals() {
    const quantities = document.querySelectorAll('#salesTable .qty');
    const prices = document.querySelectorAll('#salesTable .price');
    
    let totalQty = 0;
    let totalPrice = 0;
    
    quantities.forEach((td, i) => {
      const qty = parseInt(td.textContent) || 0;
      const price = parseFloat(prices[i].textContent.replace('$', '')) || 0;
      totalQty += qty;
      totalPrice += qty * price;
    });
    
    document.getElementById('totalQty').textContent = totalQty;
    document.getElementById('totalPrice').textContent = `$${totalPrice.toFixed(2)}`;
  }
  
  updateTotals();
</script>
```
**Why**: `<tfoot>` semantically groups footer content, enabling proper table structure and accessibility for screen readers.

```html
<script>
```

**Why**: `<caption>` provides accessible table descriptions, helping screen reader users understand table context before reading data.

## 7. HTML Video Poster Attribute

```html
<!-- PROBLEM: Show preview image before video plays
     TIME: O(1) SPACE: O(1) -->
<div class="video-container">
  <video 
    id="demoVideo"
    width="640" 
    height="360"
    controls
    poster="https://example.com/preview-image.jpg"
    preload="metadata"
  >
    <source src="video.mp4" type="video/mp4">
    <source src="video.webm" type="video/webm">
    Your browser does not support the video tag.
  </video>
  
  <div class="video-controls">
    <button id="playPauseBtn">Play</button>
    <button id="changePosterBtn">Change Poster</button>
  </div>
</div>

<script>
  const video = document.getElementById('demoVideo');
  const playPauseBtn = document.getElementById('playPauseBtn');
  
  // Play/Pause toggle
  playPauseBtn.addEventListener('click', () => {
    if (video.paused) {
      video.play();
      playPauseBtn.textContent = 'Pause';
    } else {
      video.pause();
      playPauseBtn.textContent = 'Play';
    }
  });
  
  // Change poster dynamically
  document.getElementById('changePosterBtn').addEventListener('click', () => {
    video.poster = 'https://example.com/another-preview.jpg';
  });
  
  // Remove poster when video starts
  video.addEventListener('play', () => {
    video.poster = ''; // Optional: clear poster when playing
  });
</script>
```
**Why**: The `poster` attribute shows a preview image before video loads, improving perceived performance and UX.

## 8. HTML Video Autoplay

```html
<!-- PROBLEM: Auto-play video with proper browser restrictions
     TIME: O(1) SPACE: O(1) -->
<div class="autoplay-demo">
  <!-- Autoplay requires muted for most browsers -->
  <video 
    id="bgVideo"
    autoplay
    muted
    loop
    playsinline
    class="background-video"
  >
    <source src="background.mp4" type="video/mp4">
  </video>
  
  <!-- Video with user interaction to enable sound -->
  <video id="contentVideo" muted>
    <source src="content.mp4" type="video/mp4">
  </video>
  
  <button id="unmuteBtn">Unmute & Play</button>
</div>

<script>
  const bgVideo = document.getElementById('bgVideo');
  const contentVideo = document.getElementById('contentVideo');
  const unmuteBtn = document.getElementById('unmuteBtn');
  
  // Check if autoplay is supported
  async function checkAutoplaySupport() {
    try {
      await contentVideo.play();
      console.log('Autoplay supported');
      return true;
    } catch (e) {
      console.log('Autoplay blocked, user interaction required');
      return false;
    }
  }
  
  // Enable sound with user interaction
  unmuteBtn.addEventListener('click', async () => {
    contentVideo.muted = false;
    await contentVideo.play();
    unmuteBtn.style.display = 'none';
  });
  
  // Background video: muted autoplay is widely supported
  bgVideo.play().catch(e => {
    console.log('Background autoplay blocked');
  });
</script>
```
**Why**: Modern browsers block autoplay with sound. Use `muted` + `autoplay` together, then enable sound via user interaction.

## 9. CSS Colors

```html
<!-- PROBLEM: Understand different CSS color value formats
     TIME: O(1) SPACE: O(1) -->
<style>
  .color-demo {
    display: grid;
    grid-template-columns: repeat(auto-fill, minmax(100px, 1fr));
    gap: 10px;
  }
  
  /* Named colors */
  .named-red { background-color: red; }
  .named-blue { background-color: steelblue; }
  
  /* Hex colors */
  .hex-3 { background-color: #f00; }      /* #RGB */
  .hex-6 { background-color: #ff0000; }   /* #RRGGBB */
  .hex-8 { background-color: #ff000080; } /* #RRGGBBAA (with alpha) */
  
  /* RGB and RGBA */
  .rgb { background-color: rgb(255, 0, 0); }
  .rgba { background-color: rgba(255, 0, 0, 0.5); }
  
  /* HSL and HSLA */
  .hsl { background-color: hsl(0, 100%, 50%); }
  .hsla { background-color: hsla(0, 100%, 50%, 0.5); }
  
  /* Modern color functions */
  .rgb-modern { background-color: rgb(255 0 0); }
  .rgba-modern { background-color: rgb(255 0 0 / 50%); }
  .hsl-modern { background-color: hsl(0deg 100% 50%); }
  
  /* Color keywords */
  .transparent { background-color: transparent; }
  .currentcolor { border: 2px solid currentColor; color: blue; }
  
  .color-box {
    height: 50px;
    display: flex;
    align-items: center;
    justify-content: center;
    color: white;
    font-size: 10px;
    text-shadow: 1px 1px 1px black;
  }
</style>

<div class="color-demo">
  <div class="color-box named-red">named</div>
  <div class="color-box hex-6">hex</div>
  <div class="color-box hex-8">hex+a</div>
  <div class="color-box rgba">rgba</div>
  <div class="color-box hsla">hsla</div>
  <div class="color-box rgb-modern">modern</div>
</div>

<script>
  // Manipulate colors with JavaScript
  function hexToRgb(hex) {
    const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
    return result ? {
      r: parseInt(result[1], 16),
      g: parseInt(result[2], 16),
      b: parseInt(result[3], 16)
    } : null;
  }
  
  function rgbToHex(r, g, b) {
    return '#' + [r, g, b].map(x => x.toString(16).padStart(2, '0')).join('');
  }
  
  console.log(hexToRgb('#ff0000')); // {r: 255, g: 0, b: 0}
  console.log(rgbToHex(255, 0, 0)); // '#ff0000'
</script>
```
**Why**: CSS supports multiple color formats. Hex is compact, HSL is intuitive for adjustments, RGBA enables transparency.

```html
<div class="bg-demo">

  <div class="bg-image demo-label">background-image</div>
  <div class="bg-gradient demo-label">linear-gradient</div>

  <div class="bg-multiple demo-label">multiple layers</div>

<script>
  // Manipulate backgrounds with JavaScript
  function setBackground(element, options) {
    const el = typeof element === 'string' 
      ? document.querySelector(element) 
      : element;
    
    if (options.color) el.style.backgroundColor = options.color;
    if (options.image) el.style.backgroundImage = `url('${options.image}')`;
    if (options.size) el.style.backgroundSize = options.size;
    if (options.position) el.style.backgroundPosition = options.position;
    if (options.repeat !== undefined) {
      el.style.backgroundRepeat = options.repeat ? 'repeat' : 'no-repeat';
    }
  }
  
  // Example: Change background on click
  document.querySelectorAll('.bg-demo > div').forEach(el => {
    el.style.cursor = 'pointer';
    el.addEventListener('click', () => {
      const hue = Math.random() * 360;
      el.style.filter = `hue-rotate(${hue}deg)`;
    });
  });
</script>
```
**Why**: CSS backgrounds support colors, images, gradients, and patterns. Layering enables complex visual effects with minimal markup.
