## Edmonds-Karp Algorithm

```python
# PROBLEM: Find maximum flow in a flow network using BFS-based Ford-Fulkerson
# TIME: O(VE²) SPACE: O(V²)
from collections import deque
from typing import List, Dict, Tuple

class EdmondsKarp:
    """Maximum flow algorithm using BFS to find augmenting paths.
    
    Edmonds-Karp guarantees O(VE²) complexity because each augmenting
    path is found via BFS (shortest path in terms of edges), ensuring
    the total number of augmentations is bounded.
    """
    
    def __init__(self, n: int):
        """
        Initialize flow network.
        
        Args:
            n: Number of vertices in the graph
        """
        self.n = n
        # Capacity matrix: capacity[u][v] = edge capacity from u to v
        self.capacity: List[List[int]] = [[0] * n for _ in range(n)]
        # Adjacency list for efficient neighbor iteration
        self.graph: List[List[int]] = [[] for _ in range(n)]
    
    def add_edge(self, u: int, v: int, cap: int) -> None:
        """
        Add a directed edge with given capacity.
        
        Args:
            u: Source vertex
            v: Target vertex
            cap: Edge capacity (non-negative)
        """
        # Add forward edge with capacity
        self.capacity[u][v] += cap
        # Build adjacency list (avoid duplicates)
        if v not in self.graph[u]:
            self.graph[u].append(v)
            self.graph[v].append(u)  # For residual graph traversal
    
    def _bfs(self, source: int, sink: int) -> Tuple[int, List[int]]:
        """
        Find shortest augmenting path using BFS.
        
        Returns the bottleneck capacity and parent array for path reconstruction.
        
        Args:
            source: Flow source vertex
            sink: Flow sink vertex
            
        Returns:
            Tuple of (bottleneck_capacity, parent_array)
            bottleneck_capacity is 0 if no path exists
        """
        parent = [-1] * self.n
        parent[source] = source
        # Queue stores (vertex, flow_so_far)
        queue = deque([(source, float('inf'))])
        
        while queue:
            u, flow = queue.popleft()
            
            # Explore all neighbors
            for v in self.graph[u]:
                # Check if edge has residual capacity and v is unvisited
                if parent[v] == -1 and self.capacity[u][v] > 0:
                    parent[v] = u
                    # Bottleneck is minimum of current flow and edge capacity
                    new_flow = min(flow, self.capacity[u][v])
                    
                    # Reached sink - return path info
                    if v == sink:
                        return new_flow, parent
                    
                    queue.append((v, new_flow))
        
        # No augmenting path found
        return 0, parent
    
    def max_flow(self, source: int, sink: int) -> int:
        """
        Compute maximum flow from source to sink.
        
        Repeatedly finds augmenting paths via BFS and pushes flow
        until no more augmenting paths exist.
        
        Args:
            source: Source vertex
            sink: Sink vertex
            
        Returns:
            Maximum flow value
        """
        total_flow = 0
        
        while True:
            # Find augmenting path
            flow, parent = self._bfs(source, sink)
            
            # No more augmenting paths - we're done
            if flow == 0:
                break
            
            # Augment flow along the path
            total_flow += flow
            v = sink
            
            while v != source:
                u = parent[v]
                # Update residual capacities
                self.capacity[u][v] -= flow  # Forward edge decreases
                self.capacity[v][u] += flow  # Backward edge increases
                v = u
        
        return total_flow
```

**Why**: Edmonds-Karp is the practical choice for max-flow problems when you need guaranteed polynomial time. It's simpler than Dinic but still efficient enough for most applications. Key insight: BFS ensures shortest augmenting paths, which bounds the number of iterations.

## Next Smaller Element

```python
# PROBLEM: For each element, find the next smaller element to its right
# TIME: O(n) SPACE: O(n)
from typing import List

def next_smaller_element(arr: List[int]) -> List[int]:
    """
    Find the next smaller element for each position.
    
    For each element arr[i], find the first element arr[j] where j > i
    and arr[j] < arr[i]. If no such element exists, return -1.
    
    Uses a monotonic stack to process elements in O(n) time.
    
    Args:
        arr: List of integers
        
    Returns:
        List where result[i] is the next smaller element after arr[i],
        or -1 if no smaller element exists to the right
        
    Example:
        >>> next_smaller_element([4, 8, 5, 2, 25])
        [2, 5, 2, -1, -1]
    """
    n = len(arr)
    result = [-1] * n
    # Stack stores indices of elements waiting for their next smaller
    stack: List[int] = []
    
    for i in range(n):
        # While stack is non-empty and current element is smaller
        # than the element at the top of the stack
        while stack and arr[i] < arr[stack[-1]]:
            # Current element is the next smaller for stack top
            idx = stack.pop()
            result[idx] = arr[i]
        
        # Push current index to wait for its next smaller
        stack.append(i)
    
    # Remaining indices in stack have no next smaller element
    # (result already initialized to -1)
    
    return result


def next_smaller_element_left(arr: List[int]) -> List[int]:
    """
    Find the nearest smaller element to the LEFT of each element.
    
    Args:
        arr: List of integers
        
    Returns:
        List where result[i] is the nearest smaller element to the left,
        or -1 if no such element exists
    """
    n = len(arr)
    result = [-1] * n
    stack: List[int] = []
    
    for i in range(n):
        # Pop elements that are >= current (they can't be "smaller")
        while stack and arr[stack[-1]] >= arr[i]:
            stack.pop()
        
        # If stack has elements, top is the nearest smaller on left
        if stack:
            result[i] = arr[stack[-1]]
        
        stack.append(i)
    
    return result


def previous_and_next_smaller(arr: List[int]) -> tuple[List[int], List[int]]:
    """
    Compute both previous and next smaller elements in one pass.
    
    Useful for problems like "sum of subarray minimums" where you need
    the range in which each element is the minimum.
    
    Args:
        arr: Input array
        
    Returns:
        Tuple of (prev_smaller, next_smaller) arrays
        prev_smaller[i] = index of previous smaller element (or -1)
        next_smaller[i] = index of next smaller element (or -1)
    """
    n = len(arr)
    prev = [-1] * n
    next_smaller = [n] * n
    stack: List[int] = []
    
    for i in range(n):
        # Maintain monotonic increasing stack
        while stack and arr[stack[-1]] >= arr[i]:
            stack.pop()
        
        if stack:
            prev[i] = stack[-1]
        
        stack.append(i)
    
    # Clear stack for second pass
    stack.clear()
    
    # Traverse from right to left for next smaller
    for i in range(n - 1, -1, -1):
        while stack and arr[stack[-1]] >= arr[i]:
            stack.pop()
        
        if stack:
            next_smaller[i] = stack[-1]
        
        stack.append(i)
    
    return prev, next_smaller
```

**Why**: Monotonic stack is a powerful technique for "nearest neighbor" queries in arrays. This pattern appears in many problems: largest rectangle in histogram, trapping rain water, stock span, etc. The key insight is maintaining elements in sorted order to answer queries efficiently.

## Minimum Window Substring

```python
# PROBLEM: Find minimum window in s that contains all characters of t
# TIME: O(|s| + |t|) SPACE: O(|alphabet|)
from collections import Counter, defaultdict
from typing import Dict

def min_window(s: str, t: str) -> str:
    """
    Find the smallest substring of s that contains all characters of t.
    
    Uses sliding window with two pointers and a character frequency counter.
    The window expands to find valid candidates, then contracts to find minimum.
    
    Args:
        s: Source string to search in
        t: Target string whose characters must all be included
        
    Returns:
        Minimum window substring, or empty string if no valid window exists
        
    Example:
        >>> min_window("ADOBECODEBANC", "ABC")
        "BANC"
        >>> min_window("a", "a")
        "a"
        >>> min_window("a", "aa")
        ""
    """
    if not s or not t:
        return ""
    
    # Count characters needed from t
    need: Dict[str, int] = Counter(t)
    # Track characters in current window
    window: Dict[str, int] = defaultdict(int)
    
    # How many unique characters have been satisfied
    # A character is satisfied when window[char] >= need[char]
    satisfied = 0
    required = len(need)  # Number of unique chars needed
    
    # Result tracking
    min_len = float('inf')
    min_start = 0
    
    left = 0
    
    for right, char in enumerate(s):
        # Expand window by including char at right
        if char in need:
            window[char] += 1
            # Check if this char is now satisfied
            if window[char] == need[char]:
                satisfied += 1
        
        # Try to contract window while it remains valid
        while satisfied == required:
            # Update minimum if current window is smaller
            window_len = right - left + 1
            if window_len < min_len:
                min_len = window_len
                min_start = left
            
            # Remove leftmost character from window
            left_char = s[left]
            if left_char in need:
                if window[left_char] == need[left_char]:
                    # This char will become unsatisfied
                    satisfied -= 1
                window[left_char] -= 1
            
            left += 1
    
    return s[min_start:min_start + min_len] if min_len != float('inf') else ""


def min_window_optimized(s: str, t: str) -> str:
    """
    Optimized version that skips irrelevant characters.
    
    Instead of iterating through all characters in s, we first
    build a filtered list of (index, char) pairs for chars in t.
    This is faster when s is long but contains few target characters.
    
    Args:
        s: Source string
        t: Target string
        
    Returns:
        Minimum window substring
    """
    if not s or not t:
        return ""
    
    need = Counter(t)
    required = len(need)
    
    # Filter s to only include relevant characters
    filtered_s = [(i, c) for i, c in enumerate(s) if c in need]
    
    window = defaultdict(int)
    satisfied = 0
    
    min_len = float('inf')
    min_start = 0
    
    left = 0
    
    for right, (idx, char) in enumerate(filtered_s):
        window[char] += 1
        if window[char] == need[char]:
            satisfied += 1
        
        while satisfied == required:
            # Get actual string indices
            start_idx = filtered_s[left][0]
            end_idx = idx
            
            if end_idx - start_idx + 1 < min_len:
                min_len = end_idx - start_idx + 1
                min_start = start_idx
            
            # Contract from left
            left_char = filtered_s[left][1]
            if window[left_char] == need[left_char]:
                satisfied -= 1
            window[left_char] -= 1
            left += 1
    
    return s[min_start:min_start + min_len] if min_len != float('inf') else ""
```

**Why**: The sliding window pattern is essential for substring/subarray problems with constraints. The "expand until valid, then contract" strategy guarantees we find the minimum. The optimization shows how to handle sparse data efficiently—skip irrelevant elements entirely.

## LFU Cache

```python
# PROBLEM: Design a Least Frequently Used cache with O(1) operations
# TIME: O(1) for get and put SPACE: O(capacity)
from collections import defaultdict
from typing import Dict

class Node:
    """Doubly linked list node for O(1) removal."""
    __slots__ = ['key', 'val', 'freq', 'prev', 'next']
    
    def __init__(self, key: int = 0, val: int = 0):
        self.key = key
        self.val = val
        self.freq = 1
        self.prev: 'Node' = None
        self.next: 'Node' = None


class DLinkedList:
    """Doubly linked list with sentinel nodes for simpler edge handling."""
    
    def __init__(self):
        self.head = Node()  # Sentinel head
        self.tail = Node()  # Sentinel tail
        self.head.next = self.tail
        self.tail.prev = self.head
        self.size = 0
    
    def append(self, node: Node) -> None:
        """Add node to the end (before tail sentinel)."""
        node.prev = self.tail.prev
        node.next = self.tail
        self.tail.prev.next = node
        self.tail.prev = node
        self.size += 1
    
    def remove(self, node: Node) -> None:
        """Remove node from list."""
        node.prev.next = node.next
        node.next.prev = node.prev
        self.size -= 1
    
    def pop_left(self) -> Node:
        """Remove and return the first node (after head sentinel)."""
        if self.size == 0:
            return None
        node = self.head.next
        self.remove(node)
        return node
    
    def is_empty(self) -> bool:
        return self.size == 0


class LFUCache:
    """
    Least Frequently Used (LFU) Cache.
    
    When the cache is full, evict the least frequently used item.
    If there's a tie, evict the least recently used among them.
    
    All operations are O(1) using a combination of:
    - Hash map for key -> node lookup
    - Hash map for frequency -> linked list of nodes
    - Each frequency list maintains LRU order
    
    Attributes:
        capacity: Maximum number of items in cache
        min_freq: Current minimum frequency among all items
        size: Current number of items
    """
    
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.size = 0
        self.min_freq = 0
        
        # key -> Node for O(1) lookup
        self.key_to_node: Dict[int, Node] = {}
        
        # freq -> DLinkedList of nodes with that frequency
        self.freq_to_list: Dict[int, DLinkedList] = defaultdict(DLinkedList)
    
    def _update_freq(self, node: Node) -> None:
        """
        Update node's frequency after access.
        
        Remove node from current frequency list, increment frequency,
        and add to new frequency list.
        """
        freq = node.freq
        
        # Remove from current frequency list
        self.freq_to_list[freq].remove(node)
        
        # Update min_freq if necessary
        if freq == self.min_freq and self.freq_to_list[freq].is_empty():
            self.min_freq += 1
        
        # Increment frequency and add to new list
        node.freq += 1
        self.freq_to_list[node.freq].append(node)
    
    def get(self, key: int) -> int:
        """
        Get value for key if it exists in cache.
        
        Args:
            key: Key to look up
            
        Returns:
            Value if key exists, -1 otherwise
        """
        if key not in self.key_to_node:
            return -1
        
        node = self.key_to_node[key]
        self._update_freq(node)
        return node.val
    
    def put(self, key: int, value: int) -> None:
        """
        Insert or update a key-value pair.
        
        If key exists, update value and frequency.
        If key doesn't exist and cache is full, evict LFU item first.
        
        Args:
            key: Key to insert/update
            value: Value to store
        """
        if self.capacity == 0:
            return
        
        if key in self.key_to_node:
            # Update existing key
            node = self.key_to_node[key]
            node.val = value
            self._update_freq(node)
            return
        
        # Insert new key
        if self.size == self.capacity:
            # Evict LFU item (first item in min_freq list)
            evicted = self.freq_to_list[self.min_freq].pop_left()
            del self.key_to_node[evicted.key]
            self.size -= 1
        
        # Create new node with frequency 1
        node = Node(key, value)
        self.key_to_node[key] = node
        self.freq_to_list[1].append(node)
        self.min_freq = 1  # New node has minimum frequency
        self.size += 1
```

**Why**: LFU cache demonstrates how to combine multiple data structures (hash maps + doubly linked lists) for efficient operations. The key insight is maintaining separate LRU lists for each frequency level, which lets us always find the eviction candidate in O(1) time.

## B-Tree

```python
# PROBLEM: Implement a B-tree for efficient disk-based storage
# TIME: O(log n) for search/insert/delete SPACE: O(n)
from typing import List, Optional, Tuple

class BTreeNode:
    """B-tree node containing keys and children."""
    
    def __init__(self, is_leaf: bool = False):
        self.keys: List[int] = []
        self.children: List['BTreeNode'] = []
        self.is_leaf = is_leaf
    
    def is_full(self, t: int) -> bool:
        """Check if node has maximum keys (2t-1)."""
        return len(self.keys) == 2 * t - 1


class BTree:
    """
    B-tree implementation with minimum degree t.
    
    Properties:
    - Every node has at most 2t-1 keys
    - Every non-leaf node (except root) has at least t-1 keys
    - Root has at least 1 key (unless tree is empty)
    - Non-leaf node with k keys has k+1 children
    - All leaves are at the same level
    
    B-trees are optimized for systems that read/write large blocks
    of data, making them ideal for databases and file systems.
    """
    
    def __init__(self, t: int):
        """
        Initialize empty B-tree.
        
        Args:
            t: Minimum degree (minimum children = t, maximum = 2t)
        """
        self.t = t  # Minimum degree
        self.root: Optional[BTreeNode] = None
    
    def search(self, key: int) -> Optional[Tuple[BTreeNode, int]]:
        """
        Search for key in B-tree.
        
        Args:
            key: Key to search for
            
        Returns:
            Tuple of (node, index) if found, None otherwise
        """
        if self.root is None:
            return None
        return self._search_node(self.root, key)
    
    def _search_node(self, node: BTreeNode, key: int) -> Optional[Tuple[BTreeNode, int]]:
        """Recursively search for key starting at node."""
        # Find first key >= target
        i = 0
        while i < len(node.keys) and key > node.keys[i]:
            i += 1
        
        # Found the key
        if i < len(node.keys) and key == node.keys[i]:
            return (node, i)
        
        # Key not found and this is a leaf
        if node.is_leaf:
            return None
        
        # Recurse to appropriate child
        return self._search_node(node.children[i], key)
    
    def insert(self, key: int) -> None:
        """
        Insert key into B-tree.
        
        If root is full, split it before insertion to maintain
        the invariant that we never insert into a full node.
        
        Args:
            key: Key to insert
        """
        if self.root is None:
            self.root = BTreeNode(is_leaf=True)
            self.root.keys.append(key)
            return
        
        if self.root.is_full(self.t):
            # Root is full - split it
            new_root = BTreeNode(is_leaf=False)
            new_root.children.append(self.root)
            self._split_child(new_root, 0)
            self.root = new_root
        
        self._insert_non_full(self.root, key)
    
    def _split_child(self, parent: BTreeNode, index: int) -> None:
        """
        Split the (full) child at parent.children[index].
        
        The middle key moves up to parent, and the child splits
        into two nodes with (t-1) keys each.
        """
        t = self.t
        full_child = parent.children[index]
        
        # Create new node for right half
        new_node = BTreeNode(is_leaf=full_child.is_leaf)
        
        # Move median key up to parent
        median_key = full_child.keys[t - 1]
        parent.keys.insert(index, median_key)
        parent.children.insert(index + 1, new_node)
        
        # Split keys: left keeps [0, t-1), right gets [t, 2t-1)
        new_node.keys = full_child.keys[t:]
        full_child.keys = full_child.keys[:t - 1]
        
        # Split children if not leaf
        if not full_child.is_leaf:
            new_node.children = full_child.children[t:]
            full_child.children = full_child.children[:t]
    
    def _insert_non_full(self, node: BTreeNode, key: int) -> None:
        """Insert key into a node that is not full."""
        i = len(node.keys) - 1
        
        if node.is_leaf:
            # Insert key in sorted position
            node.keys.append(0)  # Placeholder
            while i >= 0 and key < node.keys[i]:
                node.keys[i + 1] = node.keys[i]
                i -= 1
            node.keys[i + 1] = key
        else:
            # Find child to descend into
            while i >= 0 and key < node.keys[i]:
                i -= 1
            i += 1
            
            # Split child if full
            if node.children[i].is_full(self.t):
                self._split_child(node, i)
                if key > node.keys[i]:
                    i += 1
            
            self._insert_non_full(node.children[i], key)
    
    def traverse(self) -> List[int]:
        """Return all keys in sorted order."""
        result = []
        if self.root:
            self._traverse_node(self.root, result)
        return result
    
    def _traverse_node(self, node: BTreeNode, result: List[int]) -> None:
        """In-order traversal of node."""
        i = 0
        for i in range(len(node.keys)):
            if not node.is_leaf:
                self._traverse_node(node.children[i], result)
            result.append(node.keys[i])
        
        if not node.is_leaf:
            self._traverse_node(node.children[len(node.keys)], result)
    
    def delete(self, key: int) -> bool:
        """
        Delete key from B-tree.
        
        This is a simplified implementation that handles the basic cases.
        Production implementations also handle borrowing from siblings
        and merging nodes when they become too small.
        
        Args:
            key: Key to delete
            
        Returns:
            True if key was found and deleted, False otherwise
        """
        if not self.root:
            return False
        
        result = self._delete_from_node(self.root, key)
        
        # Root may have become empty after deletion
        if len(self.root.keys) == 0:
            if not self.root.is_leaf:
                self.root = self.root.children[0]
            else:
                self.root = None
        
        return result
    
    def _delete_from_node(self, node: BTreeNode, key: int) -> bool:
        """Delete key from subtree rooted at node."""
        t = self.t
        
        # Find key or child to descend into
        i = 0
        while i < len(node.keys) and key > node.keys[i]:
            i += 1
        
        # Case 1: Key is in this node
        if i < len(node.keys) and key == node.keys[i]:
            if node.is_leaf:
                # Simple case: remove from leaf
                node.keys.pop(i)
                return True
            else:
                # Replace with predecessor or successor
                if len(node.children[i].keys) >= t:
                    # Get predecessor (largest in left subtree)
                    pred = self._get_predecessor(node.children[i])
                    node.keys[i] = pred
                    return self._delete_from_node(node.children[i], pred)
                elif len(node.children[i + 1].keys) >= t:
                    # Get successor (smallest in right subtree)
                    succ = self._get_successor(node.children[i + 1])
                    node.keys[i] = succ
                    return self._delete_from_node(node.children[i + 1], succ)
        
        # Key not found and this is a leaf
        if node.is_leaf:
            return False
        
        # Recurse to appropriate child
        return self._delete_from_node(node.children[i], key)
    
    def _get_predecessor(self, node: BTreeNode) -> int:
        """Get the largest key in subtree rooted at node."""
        while not node.is_leaf:
            node = node.children[-1]
        return node.keys[-1]
    
    def _get_successor(self, node: BTreeNode) -> int:
        """Get the smallest key in subtree rooted at node."""
        while not node.is_leaf:
            node = node.children[0]
        return node.keys[0]
```

**Why**: B-trees are fundamental to database systems because they minimize disk I/O. Each node is designed to fit in a disk block, so reading one node = one disk access. The balanced structure guarantees O(log n) operations, and the high branching factor means shallow trees (fewer disk reads).

```python
# PROBLEM: Implement event sourcing for auditability and state reconstruction
# TIME: O(1) for append, O(n) for replay SPACE: O(n) for event store
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any, Dict, List, Optional, Callable, Type
from enum import Enum
import json
from abc import ABC, abstractmethod

class EventType(Enum):
    """Common event types for a typical application."""
    USER_CREATED = "user_created"
    USER_UPDATED = "user_updated"
    USER_DELETED = "user_deleted"
    MONEY_DEPOSITED = "money_deposited"

@dataclass
class Event:

    Base event class for all domain events.
    
    Events are immutable facts that happened in the system.
    They should be named in past tense (e.g., UserCreated, MoneyDeposited).
    """

            'timestamp': self.timestamp.isoformat(),
            'version': self.version,
            'metadata': self.metadata
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> 'Event':
        """Deserialize event from dictionary."""
        return cls(
            event_type=EventType(data['event_type']),
            aggregate_id=data['aggregate_id'],
            data=data['data'],
            timestamp=datetime.fromisoformat(data['timestamp']),
            version=data.get('version', 1),
            metadata=data.get('metadata', {})
        )


class EventStore:
    """
    Append-only store for events.
    
    The event store is the source of truth. Current state can always
    be reconstructed by replaying events from the store.
    """
    
    def __init__(self):
        # In production, this would be a database
        self._events: List[Event] = []
        # Index for quick lookup by aggregate
        self._by_aggregate: Dict[str, List[int]] = {}
    
    def append(self, event: Event) -> None:
        """
        Append event to store.
        
        Events are immutable - once appended, they cannot be changed.
        """
        idx = len(self._events)
        self._events.append(event)
        
        # Update index
        if event.aggregate_id not in self._by_aggregate:
            self._by_aggregate[event.aggregate_id] = []
        self._by_aggregate[event.aggregate_id].append(idx)
    
    def get_events(self, aggregate_id: Optional[str] = None) -> List[Event]:
        """
        Get events, optionally filtered by aggregate.
        
        Args:
            aggregate_id: If provided, only return events for this aggregate
            
        Returns:
            List of events in chronological order
        """
        if aggregate_id is None:
            return list(self._events)
        
        indices = self._by_aggregate.get(aggregate_id, [])
        return [self._events[i] for i in sorted(indices)]
    
    def get_events_since(self, timestamp: datetime) -> List[Event]:
        """Get all events after a given timestamp."""
        return [e for e in self._events if e.timestamp > timestamp]


class Aggregate(ABC):
    """
    Base class for event-sourced aggregates.
    
    An aggregate is a cluster of domain objects that can be treated
    as a single unit. All changes to an aggregate must go through
    events to maintain the audit trail.
    """
    
    def __init__(self, aggregate_id: str):
        self.id = aggregate_id
        self.version = 0
        self._pending_events: List[Event] = []
    
    def apply_event(self, event: Event) -> None:
        """
        Apply an event to update aggregate state.
        
        This method dispatches to the appropriate handler based on
        event type. Subclasses should implement handlers for their events.
        """
        handler_name = f"_apply_{event.event_type.value}"
        handler = getattr(self, handler_name, None)
        
        if handler:
            handler(event.data)
            self.version += 1
        else:
            raise ValueError(f"No handler for event type: {event.event_type}")
    
    def load_from_history(self, events: List[Event]) -> None:
        """
        Rebuild aggregate state from event history.
        
        This is called when loading an aggregate from the event store.
        """
        for event in events:
            self.apply_event(event)
        self._pending_events.clear()
    
    def add_event(self, event_type: EventType, data: Dict[str, Any]) -> None:
        """
        Create and apply a new event.
        
        The event is added to pending events for later persistence.
        """
        event = Event(
            event_type=event_type,
            aggregate_id=self.id,
            data=data,
            version=self.version + 1
        )
        self.apply_event(event)
        self._pending_events.append(event)
    
    def get_pending_events(self) -> List[Event]:
        """Get events that need to be persisted."""
        return list(self._pending_events)
    
    def clear_pending_events(self) -> None:
        """Clear pending events after persistence."""
        self._pending_events.clear()


class BankAccount(Aggregate):
    """
    Example aggregate: a bank account using event sourcing.
    
    All state changes (deposits, withdrawals) are captured as events.
    The current balance can always be recomputed from event history.
    """
    
    def __init__(self, aggregate_id: str):
        super().__init__(aggregate_id)
        self.owner: str = ""
        self.balance: float = 0.0
        self.is_closed: bool = False
    
    def create(self, owner: str, initial_deposit: float = 0) -> None:
        """Create a new bank account."""
        if self.version > 0:
            raise ValueError("Account already exists")
        
        self.add_event(EventType.USER_CREATED, {
            'owner': owner,
            'initial_deposit': initial_deposit
        })
    
    def deposit(self, amount: float) -> None:
        """Deposit money into account."""
        if self.is_closed:
            raise ValueError("Cannot deposit to closed account")
        if amount <= 0:
            raise ValueError("Deposit amount must be positive")
        
        self.add_event(EventType.MONEY_DEPOSITED, {
            'amount': amount,
            'balance_before': self.balance
        })
    
    def withdraw(self, amount: float) -> None:
        """Withdraw money from account."""
        if self.is_closed:
            raise ValueError("Cannot withdraw from closed account")
        if amount <= 0:
            raise ValueError("Withdrawal amount must be positive")
        if amount > self.balance:
            raise ValueError("Insufficient funds")
        
        self.add_event(EventType.MONEY_WITHDRAWN, {
            'amount': amount,
            'balance_before': self.balance
        })
    
    def close(self) -> None:
        """Close the account."""
        if self.balance != 0:
            raise ValueError("Cannot close account with non-zero balance")
        
        self.add_event(EventType.USER_DELETED, {
            'closed_at': datetime.now().isoformat()
        })
    
    # Event handlers - these update the actual state
    
    def _apply_user_created(self, data: Dict[str, Any]) -> None:
        self.owner = data['owner']
        self.balance = data.get('initial_deposit', 0)
    
    def _apply_money_deposited(self, data: Dict[str, Any]) -> None:
        self.balance += data['amount']
    
    def _apply_money_withdrawn(self, data: Dict[str, Any]) -> None:
        self.balance -= data['amount']
    
    def _apply_user_deleted(self, data: Dict[str, Any]) -> None:
        self.is_closed = True
    
    def __repr__(self) -> str:
        return f"BankAccount(id={self.id}, owner={self.owner}, balance={self.balance})"


class Repository:
    """
    Repository for saving and loading aggregates.
    
    Handles the interaction between aggregates and the event store.
    """
    
    def __init__(self, event_store: EventStore, aggregate_class: Type[Aggregate]):
        self.event_store = event_store
        self.aggregate_class = aggregate_class
    
    def save(self, aggregate: Aggregate) -> None:
        """
        Save aggregate by persisting its pending events.
        """
        events = aggregate.get_pending_events()
        for event in events:
            self.event_store.append(event)
        aggregate.clear_pending_events()
    
    def load(self, aggregate_id: str) -> Aggregate:
        """
        Load aggregate by replaying its event history.
        """
        aggregate = self.aggregate_class(aggregate_id)
        events = self.event_store.get_events(aggregate_id)
        if events:
            aggregate.load_from_history(events)
        return aggregate


# Usage example
def demo_event_sourcing():
    """Demonstrate event sourcing with bank accounts."""
    
    # Set up infrastructure
    event_store = EventStore()
    account_repo = Repository(event_store, BankAccount)
    
    # Create account
    account = BankAccount("acc-001")
    account.create("Alice", initial_deposit=100)
    account_repo.save(account)
    print(f"Created: {account}")
    
    # Load and modify
    account = account_repo.load("acc-001")
    account.deposit(50)
    account.withdraw(30)
    account_repo.save(account)
    print(f"After transactions: {account}")
    
    # Audit trail
    print("\nEvent history:")
    for event in event_store.get_events("acc-001"):
        print(f"  {event.event_type.value}: {event.data}")


if __name__ == "__main__":
    demo_event_sourcing()
```

**Why**: Event sourcing provides a complete audit trail and enables powerful features like time-travel debugging, event replay, and CQRS. Instead of storing current state, we store the sequence of events that led to that state. This pattern is essential for financial systems, collaboration tools, and any domain where understanding "what happened" is as important as "what is."
