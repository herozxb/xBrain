## Longest Path in DAG

```python
# PROBLEM: Find the longest path in a Directed Acyclic Graph (DAG)
# TIME: O(V + E) SPACE: O(V)
from collections import defaultdict, deque

def longest_path_dag(graph: dict[int, list[int]]) -> int:
    """
    Compute the length of the longest path in a DAG using topological sort.
    
    Args:
        graph: Adjacency list representation {node: [neighbors]}
    
    Returns:
        Length of the longest path (number of edges)
    """
    # Step 1: Compute in-degree for each node
    in_degree = defaultdict(int)
    all_nodes = set(graph.keys())
    for node in graph:
        for neighbor in graph[node]:
            in_degree[neighbor] += 1
            all_nodes.add(neighbor)
    
    # Step 2: Initialize distance array and queue with sources (in-degree = 0)
    dist = {node: 0 for node in all_nodes}
    queue = deque([node for node in all_nodes if in_degree[node] == 0])
    
    # Step 3: Process nodes in topological order
    topo_order = []
    while queue:
        node = queue.popleft()
        topo_order.append(node)
        for neighbor in graph.get(node, []):
            # Relax edge: update longest distance to neighbor
            dist[neighbor] = max(dist[neighbor], dist[node] + 1)
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    
    return max(dist.values()) if dist else 0
```
**Why**: DAG longest path is solved in linear time via topological sort—unlike general graphs where it's NP-hard. Key for scheduling and critical path problems.

## Closest Pair of Points

```python
# PROBLEM: Find the minimum distance between any two points in 2D plane
# TIME: O(n log n) SPACE: O(n)
import math
from dataclasses import dataclass
from typing import List, Tuple, Optional

@dataclass
class Point:
    x: float
    y: float
    
    def distance_to(self, other: 'Point') -> float:
        return math.sqrt((self.x - other.x) ** 2 + (self.y - other.y) ** 2)

def closest_pair(points: List[Point]) -> Tuple[Optional[Tuple[Point, Point]], float]:
    """
    Find the closest pair of points using divide and conquer.
    
    Returns:
        Tuple of (point pair, minimum distance), or (None, inf) if < 2 points
    """
    n = len(points)
    if n < 2:
        return None, float('inf')
    
    # Sort by x-coordinate once
    points_sorted = sorted(points, key=lambda p: p.x)
    
    def brute_force(pts: List[Point]) -> Tuple[Optional[Tuple[Point, Point]], float]:
        """O(n^2) comparison for small sets."""
        min_dist = float('inf')
        best_pair = None
        for i in range(len(pts)):
            for j in range(i + 1, len(pts)):
                d = pts[i].distance_to(pts[j])
                if d < min_dist:
                    min_dist = d
                    best_pair = (pts[i], pts[j])
        return best_pair, min_dist
    
    def closest_recursive(pts: List[Point]) -> Tuple[Optional[Tuple[Point, Point]], float]:
        """Recursive divide-and-conquer core."""
        n = len(pts)
        if n <= 3:
            return brute_force(pts)
        
        # Divide: split at median x
        mid = n // 2
        mid_x = pts[mid].x
        
        left_best, left_dist = closest_recursive(pts[:mid])
        right_best, right_dist = closest_recursive(pts[mid:])
        
        # Conquer: pick better of left/right
        if left_dist < right_dist:
            best_pair, min_dist = left_best, left_dist
        else:
            best_pair, min_dist = right_best, right_dist
        
        # Check strip: points within min_dist of vertical line
        strip = [p for p in pts if abs(p.x - mid_x) < min_dist]
        strip.sort(key=lambda p: p.y)  # Sort by y
        
        # Compare each point with next 7 in strip (proved bound)
        for i in range(len(strip)):
            for j in range(i + 1, min(i + 8, len(strip))):
                d = strip[i].distance_to(strip[j])
                if d < min_dist:
                    min_dist = d
                    best_pair = (strip[i], strip[j])
        
        return best_pair, min_dist
    
    return closest_recursive(points_sorted)
```
**Why**: Classic divide-and-conquer demonstrating the "strip optimization" where geometric bounds reduce comparisons. Foundation for spatial algorithms.

## Scapegoat Tree

```python
# PROBLEM: Self-balancing BST without per-node balance info
# TIME: O(log n) amortized for all operations SPACE: O(n)
from typing import Optional, List, Generic, TypeVar

T = TypeVar('T')

class ScapegoatTree(Generic[T]):
    """
    Self-balancing BST that rebuilds subtrees when they become unbalanced.
    No height/balance factor stored per node—uses alpha-weight balancing.
    """
    
    def __init__(self, alpha: float = 0.67):
        """
        Args:
            alpha: Balance threshold (0.5 < alpha < 1.0). 
                   Lower = more balanced, more rebuilds.
        """
        self.alpha = alpha
        self.root: Optional['Node[T]'] = None
        self.size = 0
        self.max_size = 0  # Track size for deletion triggering
    
    class Node(Generic[T]):
        def __init__(self, key: T):
            self.key = key
            self.left: Optional['ScapegoatTree.Node[T]'] = None
            self.right: Optional['ScapegoatTree.Node[T]'] = None
    
    def _size_of(self, node: Optional[Node[T]]) -> int:
        """Count nodes in subtree."""
        if not node:
            return 0
        return 1 + self._size_of(node.left) + self._size_of(node.right)
    
    def _is_alpha_balanced(self, node: Node[T]) -> bool:
        """Check if node satisfies alpha-weight balance."""
        left_size = self._size_of(node.left)
        right_size = self._size_of(node.right)
        total = 1 + left_size + right_size
        return left_size <= self.alpha * total and right_size <= self.alpha * total
    
    def _flatten(self, node: Optional[Node[T]], out: List[Node[T]]) -> None:
        """In-order traversal to collect nodes."""
        if node:
            self._flatten(node.left, out)
            out.append(node)
            self._flatten(node.right, out)
    
    def _build_balanced(self, nodes: List[Node[T]], start: int, end: int) -> Optional[Node[T]]:
        """Build balanced BST from sorted node list."""
        if start >= end:
            return None
        mid = (start + end) // 2
        root = nodes[mid]
        root.left = self._build_balanced(nodes, start, mid)
        root.right = self._build_balanced(nodes, mid + 1, end)
        return root
    
    def _rebuild(self, node: Node[T]) -> Node[T]:
        """Rebuild subtree rooted at node."""
        nodes: List['ScapegoatTree.Node[T]'] = []
        self._flatten(node, nodes)
        return self._build_balanced(nodes, 0, len(nodes))
    
    def insert(self, key: T) -> None:
        """Insert key, rebuilding if necessary."""
        path: List['ScapegoatTree.Node[T]'] = []
        node = self.root
        parent = None
        
        # Standard BST insert with path tracking
        while node:
            path.append(node)
            parent = node
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                return  # Duplicate, ignore
        
        new_node = self.Node(key)
        if not self.root:
            self.root = new_node
        elif key < parent.key:
            parent.left = new_node
        else:
            parent.right = new_node
        
        self.size += 1
        self.max_size = max(self.max_size, self.size)
        
        # Find scapegoat: first unbalanced ancestor
        path.append(new_node)
        scapegoat = None
        scapegoat_parent = None
        
        for i in range(len(path) - 1):
            node = path[i]
            if not self._is_alpha_balanced(node):
                scapegoat = node
                scapegoat_parent = path[i - 1] if i > 0 else None
        
        # Rebuild scapegoat subtree
        if scapegoat:
            rebuilt = self._rebuild(scapegoat)
            if not scapegoat_parent:
                self.root = rebuilt
            elif scapegoat_parent.left == scapegoat:
                scapegoat_parent.left = rebuilt
            else:
                scapegoat_parent.right = rebuilt
    
    def search(self, key: T) -> bool:
        """Standard BST search."""
        node = self.root
        while node:
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                return True
        return False
    
    def delete(self, key: T) -> None:
        """Delete key; rebuild entire tree if too sparse."""
        # Standard BST delete (simplified)
        self.root = self._delete_node(self.root, key)
        if self.root:
            self.size -= 1
            # Rebuild if tree became too sparse
            if self.size < self.alpha * self.max_size:
                self.root = self._rebuild(self.root)
                self.max_size = self.size
    
    def _delete_node(self, node: Optional[Node[T]], key: T) -> Optional[Node[T]]:
        """Recursive delete helper."""
        if not node:
            return None
        if key < node.key:
            node.left = self._delete_node(node.left, key)
        elif key > node.key:
            node.right = self._delete_node(node.right, key)
        else:
            if not node.left:
                return node.right
            if not node.right:
                return node.left
            # Two children: find inorder successor
            succ = node.right
            while succ.left:
                succ = succ.left
            node.key = succ.key
            node.right = self._delete_node(node.right, succ.key)
        return node
```
**Why**: Unique self-balancing tree that rebuilds subtrees rather than rotating. Simpler than AVL/Red-Black, memory-efficient (no balance metadata). Great for read-heavy workloads.

## Aho-Corasick Algorithm

```python
# PROBLEM: Find all occurrences of multiple patterns in text simultaneously
# TIME: O(n + m + k) SPACE: O(m * alphabet)
# where n = text length, m = total pattern length, k = total matches
from collections import deque, defaultdict
from typing import List, Dict, Set

class AhoCorasick:
    """
    Multi-pattern string matching using automaton with failure links.
    Finds all pattern occurrences in a single text pass.
    """
    
    class Node:
        def __init__(self):
            self.children: Dict[str, 'AhoCorasick.Node'] = {}
            self.fail: Optional['AhoCorasick.Node'] = None
            self.output: Set[int] = set()  # Indices of patterns ending here
    
    def __init__(self, patterns: List[str]):
        """
        Build automaton from patterns.
        
        Args:
            patterns: List of patterns to search for
        """
        self.patterns = patterns
        self.root = self.Node()
        self._build_trie()
        self._build_failure_links()
    
    def _build_trie(self) -> None:
        """Insert all patterns into trie."""
        for idx, pattern in enumerate(self.patterns):
            node = self.root
            for char in pattern:
                if char not in node.children:
                    node.children[char] = self.Node()
                node = node.children[char]
            node.output.add(idx)
    
    def _build_failure_links(self) -> None:
        """Build failure links using BFS (like KMP failure function)."""
        queue = deque()
        
        # Level 1: fail to root
        for child in self.root.children.values():
            child.fail = self.root
            queue.append(child)
        
        # BFS for remaining nodes
        while queue:
            current = queue.popleft()
            
            for char, child in current.children.items():
                queue.append(child)
                
                # Follow failure links until we find a match or reach root
                fail = current.fail
                while fail and char not in fail.children:
                    fail = fail.fail
                
                child.fail = fail.children.get(char, self.root) if fail else self.root
                
                # Merge output from fail node
                child.output |= child.fail.output
    
    def search(self, text: str) -> Dict[int, List[int]]:
        """
        Find all pattern occurrences in text.
        
        Returns:
            Dict mapping pattern index to list of start positions
        """
        results: Dict[int, List[int]] = {i: [] for i in range(len(self.patterns))}
        node = self.root
        
        for i, char in enumerate(text):
            # Follow failure links until we can advance or reach root
            while node and char not in node.children:
                node = node.fail
            
            if not node:
                node = self.root
                continue
            
            node = node.children[char]
            
            # Record all patterns ending at this position
            for pattern_idx in node.output:
                pattern_len = len(self.patterns[pattern_idx])
                start_pos = i - pattern_len + 1
                results[pattern_idx].append(start_pos)
        
        return results


def demo():
    patterns = ["he", "she", "his", "hers"]
    text = "ushers"
    
    ac = AhoCorasick(patterns)
    matches = ac.search(text)
    
    for idx, positions in matches.items():
        if positions:
            for pos in positions:
                print(f"Pattern '{patterns[idx]}' found at position {pos}")
```
**Why**: Optimal multi-pattern matching—build once, search many texts in O(n). Essential for virus scanning, profanity filters, and keyword extraction.

## Facade Pattern

```python
# PROBLEM: Provide simplified interface to complex subsystem
# TIME: O(1) for facade call SPACE: O(1) additional
from typing import Optional
from dataclasses import dataclass

# Complex subsystem classes (normally spread across modules)
@dataclass
class InventoryItem:
    product_id: str
    quantity: int
    price: float

class InventorySystem:
    """Subsystem: manages stock levels."""
    
    def __init__(self):
        self._stock: dict[str, InventoryItem] = {}
    
    def check_availability(self, product_id: str, qty: int) -> bool:
        item = self._stock.get(product_id)
        return item is not None and item.quantity >= qty
    
    def reserve(self, product_id: str, qty: int) -> bool:
        if not self.check_availability(product_id, qty):
            return False
        self._stock[product_id].quantity -= qty
        return True
    
    def restock(self, product_id: str, qty: int, price: float) -> None:
        if product_id in self._stock:
            self._stock[product_id].quantity += qty
        else:
            self._stock[product_id] = InventoryItem(product_id, qty, price)


class PaymentSystem:
    """Subsystem: handles payments."""
    
    def __init__(self):
        self._transactions: list[dict] = []
    
    def process_payment(self, card_token: str, amount: float) -> str:
        """Process payment and return transaction ID."""
        import uuid
        txn_id = str(uuid.uuid4())[:8]
        self._transactions.append({
            'id': txn_id,
            'card': card_token[-4:],
            'amount': amount,
            'status': 'completed'
        })
        return txn_id
    
    def refund(self, txn_id: str) -> bool:
        for txn in self._transactions:
            if txn['id'] == txn_id:
                txn['status'] = 'refunded'
                return True
        return False


class ShippingSystem:
    """Subsystem: handles shipping."""
    
    def __init__(self):
        self._shipments: dict[str, str] = {}
    
    def create_shipment(self, order_id: str, address: str) -> str:
        """Create shipment and return tracking number."""
        tracking = f"TRK-{order_id[:6]}"
        self._shipments[tracking] = address
        return tracking
    
    def get_status(self, tracking: str) -> str:
        return self._shipments.get(tracking, "Unknown")


class NotificationSystem:
    """Subsystem: sends notifications."""
    
    def send_email(self, to: str, subject: str, body: str) -> bool:
        print(f"[EMAIL] To: {to} | Subject: {subject}")
        return True
    
    def send_sms(self, phone: str, message: str) -> bool:
        print(f"[SMS] {phone}: {message}")
        return True


# Facade: Simplified ordering interface
class OrderFacade:
    """
    Facade providing simplified interface for complex ordering process.
    Client code only interacts with this class.
    """
    
    def __init__(self):
        self._inventory = InventorySystem()
        self._payment = PaymentSystem()
        self._shipping = ShippingSystem()
        self._notification = NotificationSystem()
    
    def place_order(
        self,
        product_id: str,
        quantity: int,
        card_token: str,
        email: str,
        phone: str,
        address: str
    ) -> dict:
        """
        Complete order flow: check inventory -> payment -> shipping -> notify.
        
        Returns:
            Order result with transaction and tracking info
        """
        import uuid
        order_id = str(uuid.uuid4())
        
        # Step 1: Check and reserve inventory
        if not self._inventory.check_availability(product_id, quantity):
            return {'success': False, 'error': 'Out of stock'}
        
        item = self._inventory._stock[product_id]
        total = item.price * quantity
        
        # Step 2: Process payment
        txn_id = self._payment.process_payment(card_token, total)
        
        # Step 3: Reserve inventory (after payment succeeds)
        self._inventory.reserve(product_id, quantity)
        
        # Step 4: Create shipment
        tracking = self._shipping.create_shipment(order_id, address)
        
        # Step 5: Notify customer
        self._notification.send_email(
            email,
            "Order Confirmed",
            f"Order {order_id} confirmed. Tracking: {tracking}"
        )
        self._notification.send_sms(phone, f"Order placed! Track: {tracking}")
        
        return {
            'success': True,
            'order_id': order_id,
            'transaction_id': txn_id,
            'tracking': tracking,
            'total': total
        }
    
    def cancel_order(self, order_id: str, txn_id: str, email: str) -> bool:
        """Simplified cancellation through facade."""
        refunded = self._payment.refund(txn_id)
        if refunded:
            self._notification.send_email(
                email,
                "Order Cancelled",
                f"Order {order_id} has been cancelled and refunded."
            )
        return refunded


# Usage example
if __name__ == "__main__":
    facade = OrderFacade()
    
    # Setup inventory
    facade._inventory.restock("PROD-001", 100, 29.99)
    
    # Place order (simple client call)
    result = facade.place_order(
        product_id="PROD-001",
        quantity=2,
        card_token="4242424242424242",
        email="customer@example.com",
        phone="+1234567890",
        address="123 Main St, City"
    )
    print(result)
```
**Why**: Decouples clients from subsystem complexity. Makes subsystems easier to use, test, and evolve independently. Classic GoF pattern for layered architecture.

## Sliding Window Rate Limiter

```python
# PROBLEM: Rate limit requests with precise sliding window
# TIME: O(log n) per request with sorted storage SPACE: O(n) for n requests in window
import time
from collections import deque
from threading import Lock
from typing import Optional

class SlidingWindowRateLimiter:
    """
    Sliding window rate limiter tracking exact request timestamps.
    More precise than fixed windows, avoids boundary bursts.
    """
    
    def __init__(self, max_requests: int, window_seconds: float):
        """
        Args:
            max_requests: Maximum allowed requests in window
            window_seconds: Window duration in seconds
        """
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self._requests: deque[float] = deque()
        self._lock = Lock()
    
    def is_allowed(self) -> bool:
        """
        Check if request is allowed under rate limit.
        
        Returns:
            True if request permitted, False if rate limited
        """
        with self._lock:
            now = time.time()
            cutoff = now - self.window_seconds
            
            # Remove expired timestamps (sliding window)
            while self._requests and self._requests[0] <= cutoff:
                self._requests.popleft()
            
            # Check if under limit
            if len(self._requests) < self.max_requests:
                self._requests.append(now)
                return True
            
            return False
    
    def get_current_count(self) -> int:
        """Get current request count in window."""
        with self._lock:
            now = time.time()
            cutoff = now - self.window_seconds
            
            # Clean and count
            while self._requests and self._requests[0] <= cutoff:
                self._requests.popleft()
            
            return len(self._requests)
    
    def time_until_allowed(self) -> Optional[float]:
        """Get seconds until next request would be allowed."""
        with self._lock:
            now = time.time()
            cutoff = now - self.window_seconds
            
            while self._requests and self._requests[0] <= cutoff:
                self._requests.popleft()
            
            if len(self._requests) < self.max_requests:
                return 0.0
            
            # Time until oldest request exits window
            return self._requests[0] - cutoff + 0.001


class DistributedSlidingWindow:
    """
    Approximated sliding window using two adjacent fixed windows.
    Suitable for distributed systems with Redis.
    
    Formula: count = (prev_count * overlap_ratio) + curr_count
    """
    
    def __init__(self, max_requests: int, window_seconds: float):
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self._prev_count = 0
        self._prev_window = 0
        self._curr_count = 0
        self._curr_window = 0
        self._lock = Lock()
    
    def _get_window(self) -> int:
        """Get current window number."""
        return int(time.time() // self.window_seconds)
    
    def is_allowed(self) -> bool:
        """Check if request allowed using weighted window calculation."""
        with self._lock:
            now_window = self._get_window()
            
            # Reset if moved to new window
            if now_window > self._curr_window:
                if now_window > self._curr_window + 1:
                    # Skipped window
                    self._prev_count = 0
                else:
                    self._prev_count = self._curr_count
                self._prev_window = self._curr_window
                self._curr_count = 0
                self._curr_window = now_window
            
            # Calculate weighted count
            now = time.time()
            window_start = now_window * self.window_seconds
            elapsed = now - window_start
            overlap = 1.0 - (elapsed / self.window_seconds)
            
            weighted_count = (self._prev_count * overlap) + self._curr_count
            
            if weighted_count < self.max_requests:
                self._curr_count += 1
                return True
            
            return False


# Usage
if __name__ == "__main__":
    # 5 requests per 2 seconds
    limiter = SlidingWindowRateLimiter(max_requests=5, window_seconds=2.0)
    
    for i in range(10):
        if limiter.is_allowed():
            print(f"Request {i+1}: ALLOWED")
        else:
            wait = limiter.time_until_allowed()
            print(f"Request {i+1}: BLOCKED (wait {wait:.3f}s)")
        time.sleep(0.3)
```
**Why**: Precise rate limiting without fixed-window boundary spikes. Critical for API quotas, DDoS protection, and fair resource allocation.

## Database Isolation Levels

```python
# PROBLEM: Understand and implement database isolation levels
# TIME: Varies by level SPACE: O(n) for version chains
"""
Database Isolation Levels (from weakest to strongest):

1. READ UNCOMMITTED: Can read uncommitted changes (dirty reads)
2. READ COMMITTED: Only see committed changes (no dirty reads)
3. REPEATABLE READ: Consistent reads within transaction (snapshot)
4. SERIALIZABLE: Complete isolation, transactions appear sequential

Each level prevents specific anomalies:
- Dirty Read: Reading uncommitted data
- Non-repeatable Read: Same query returns different rows
- Phantom Read: Same query returns different number of rows
- Write Skew: Concurrent transactions create inconsistent state
"""

from dataclasses import dataclass, field
from typing import Dict, List, Optional, Any
from enum import Enum
from threading import Lock
import time

class IsolationLevel(Enum):
    READ_UNCOMMITTED = 1
    READ_COMMITTED = 2
    REPEATABLE_READ = 3
    SERIALIZABLE = 4


@dataclass
class Version:
    """MVCC version representing a row state at a point in time."""
    data: Dict[str, Any]
    created_by: int  # Transaction ID
    deleted_by: Optional[int] = None
    timestamp: float = field(default_factory=time.time)


@dataclass
class Row:
    """Row with version chain for MVCC."""
    key: str
    versions: List[Version] = field(default_factory=list)
    
    def get_version_for_tx(self, tx_id: int, level: IsolationLevel) -> Optional[Version]:
        """Get appropriate version based on isolation level."""
        if level == IsolationLevel.READ_UNCOMMITTED:
            # Return latest version regardless of commit status
            return self.versions[-1] if self.versions else None
        
        # For committed levels, find latest committed version before tx
        for version in reversed(self.versions):
            if version.created_by <= tx_id and not self._is_active(version.created_by):
                if not version.deleted_by or self._is_active(version.deleted_by):
                    return version
        return None

class MVCCStore:
    """
    Simplified MVCC (Multi-Version Concurrency Control) implementation.
    Demonstrates how isolation levels work with version chains.
    """
    
    def __init__(self, isolation_level: IsolationLevel = IsolationLevel.READ_COMMITTED):
        self.isolation_level = isolation_level
        self._rows: Dict[str, Row] = {}
        self._active_tx: Dict[int, bool] = {}
        self._next_tx_id = 1
        self._lock = Lock()
    
    def begin_transaction(self) -> int:
        """Start new transaction and return its ID."""
        with self._lock:
            tx_id = self._next_tx_id
            self._next_tx_id += 1
            self._active_tx[tx_id] = True
            return tx_id
    
    def commit(self, tx_id: int) -> None:
        """Mark transaction as committed."""
        with self._lock:
            self._active_tx[tx_id] = False
    
    def rollback(self, tx_id: int) -> None:
        """Rollback: mark versions as deleted by this tx."""
        with self._lock:
            for row in self._rows.values():
                for version in row.versions:
                    if version.created_by == tx_id:
                        version.deleted_by = tx_id
            self._active_tx[tx_id] = False
    
    def read(self, tx_id: int, key: str) -> Optional[Dict[str, Any]]:
        """Read row respecting isolation level."""
        row = self._rows.get(key)
        if not row:
            return None
        
        version = row.get_version_for_tx(tx_id, self.isolation_level)
        return version.data if version else None
    
    def write(self, tx_id: int, key: str, data: Dict[str, Any]) -> None:
        """Write new version of row."""
        with self._lock:
            if key not in self._rows:
                self._rows[key] = Row(key=key)
            
            self._rows[key].versions.append(Version(
                data=data,
                created_by=tx_id
            ))
    
    def delete(self, tx_id: int, key: str) -> None:
        """Mark row as deleted by creating tombstone."""
        row = self._rows.get(key)
        if row and row.versions:
            row.versions[-1].deleted_by = tx_id
```

```python
# PROBLEM: Efficiently index 2D spatial data for range queries
# TIME: O(log n) average search/insert, O(n) worst case SPACE: O(n)
from dataclasses import dataclass
from typing import List, Optional, Callable
import math

@dataclass
class Point:
    x: float
    y: float
    data: Optional[object] = None
    
    def distance_to(self, other: 'Point') -> float:
        return math.sqrt((self.x - other.x) ** 2 + (self.y - other.y) ** 2)


@dataclass
class Rectangle:
    """Axis-aligned bounding box."""
    x: float
    y: float
    width: float
    height: float
    
    def contains(self, point: Point) -> bool:
        """Check if point is within rectangle."""
        return (
            self.x <= point.x < self.x + self.width and
            self.y <= point.y < self.y + self.height
        )
    
    def intersects(self, other: 'Rectangle') -> bool:
        """Check if two rectangles overlap."""
        return not (
            other.x >= self.x + self.width or
            other.x + other.width <= self.x or
            other.y >= self.y + self.height or
            other.y + other.height <= self.y
        )
    
    @property
    def center(self) -> tuple[float, float]:
        return (self.x + self.width / 2, self.y + self.height / 2)


class QuadTree:
    """
    Quad tree for 2D spatial indexing.
    Recursively divides space into quadrants when capacity exceeded.
    """
    
    def __init__(
        self,
        boundary: Rectangle,
        capacity: int = 4,
        max_depth: int = 8,
        depth: int = 0
    ):
        """
        Args:
            boundary: Rectangular region this node covers
            capacity: Max points before subdivision
            max_depth: Maximum tree depth
            depth: Current depth (internal use)
        """
        self.boundary = boundary
        self.capacity = capacity
        self.max_depth = max_depth
        self.depth = depth
        
        self.points: List[Point] = []
        self.divided = False
        
        # Child quadrants
        self.northeast: Optional[QuadTree] = None
        self.northwest: Optional[QuadTree] = None
        self.southeast: Optional[QuadTree] = None
        self.southwest: Optional[QuadTree] = None
    
    def _subdivide(self) -> None:
        """Create four child quadrants."""
        x, y = self.boundary.x, self.boundary.y
        hw, hh = self.boundary.width / 2, self.boundary.height / 2
        
        self.northeast = QuadTree(
            Rectangle(x + hw, y, hw, hh),
            self.capacity, self.max_depth, self.depth + 1
        )
        self.northwest = QuadTree(
            Rectangle(x, y, hw, hh),
            self.capacity, self.max_depth, self.depth + 1
        )
        self.southeast = QuadTree(
            Rectangle(x + hw, y + hh, hw, hh),
            self.capacity, self.max_depth, self.depth + 1
        )
        self.southwest = QuadTree(
            Rectangle(x, y + hh, hw, hh),
            self.capacity, self.max_depth, self.depth + 1
        )
        
        self.divided = True
        
        # Redistribute existing points to children
        for point in self.points:
            for child in [self.northeast, self.northwest, self.southeast, self.southwest]:
                if child.boundary.contains(point):
                    child.insert(point)
                    break
        
        self.points = []
    
    def insert(self, point: Point) -> bool:
        """
        Insert point into tree.
        
        Returns:
            True if inserted, False if outside boundary
        """
        if not self.boundary.contains(point):
            return False
        
        # If we have capacity and haven't divided, store here
        if not self.divided and len(self.points) < self.capacity:
            self.points.append(point)
            return True
        
        # Subdivide if at capacity and not too deep
        if not self.divided and self.depth < self.max_depth:
            self._subdivide()
        
        # Insert into appropriate child
        if self.divided:
            for child in [self.northeast, self.northwest, self.southeast, self.southwest]:
                if child.boundary.contains(point):
                    return child.insert(point)
        
        # Fallback: store at this level (max depth reached)
        self.points.append(point)
        return True
    
    def query(self, range_rect: Rectangle) -> List[Point]:
        """
        Find all points within rectangle.
        
        Args:
            range_rect: Query rectangle
            
        Returns:
            List of points within range
        """
        found: List[Point] = []
        
        # No intersection with this node's boundary
        if not self.boundary.intersects(range_rect):
            return found
        
        # Check points at this level
        for point in self.points:
            if range_rect.contains(point):
                found.append(point)
        
        # Recursively check children
        if self.divided:
            found.extend(self.northeast.query(range_rect))
            found.extend(self.northwest.query(range_rect))
            found.extend(self.southeast.query(range_rect))
            found.extend(self.southwest.query(range_rect))
        
        return found
    
    def nearest_neighbors(
        self,
        point: Point,
        k: int = 1,
        max_distance: float = float('inf')
    ) -> List[tuple[Point, float]]:
        """
        Find k nearest neighbors to point.
        
        Returns:
            List of (point, distance) tuples, sorted by distance
        """
        candidates: List[tuple[Point, float]] = []
        
        def search(node: QuadTree):
            if not node.boundary.intersects(Rectangle(
                point.x - max_distance,
                point.y - max_distance,
                max_distance * 2,
                max_distance * 2
            )):
                return
            
            # Check points at this level
            for p in node.points:
                d = point.distance_to(p)
                if d <= max_distance and d > 0:  # Exclude self
                    candidates.append((p, d))
            
            # Search children
            if node.divided:
                for child in [node.northeast, node.northwest, node.southeast, node.southwest]:
                    search(child)
        
        search(self)
        
        # Sort by distance and return top k
        candidates.sort(key=lambda x: x[1])
        return candidates[:k]
    
    def __len__(self) -> int:
        """Count total points in tree."""
        count = len(self.points)
        if self.divided:
            count += len(self.northeast) + len(self.northwest)
            count += len(self.southeast) + len(self.southwest)
        return count


# Usage example
if __name__ == "__main__":
    # Create quad tree covering 100x100 area
    boundary = Rectangle(0, 0, 100, 100)
    qt = QuadTree(boundary, capacity=4)
    
    # Insert points
    points = [
        Point(10, 10, "A"),
        Point(20, 20, "B"),
        Point(30, 30, "C"),
        Point(50, 50, "D"),
        Point(60, 60, "E"),
        Point(70, 70, "F"),
    ]
    
    for p in points:
        qt.insert(p)
    
    # Range query
    query_rect = Rectangle(15, 15, 20, 20)
    results = qt.query(query_rect)
    print(f"Points in {query_rect}: {[p.data for p in results]}")
    
    # Nearest neighbors
    target = Point(55, 55)
    neighbors = qt.nearest_neighbors(target, k=2)
    print(f"Nearest to (55,55): {[(p.data, f'{d:.1f}') for p, d in neighbors]}")
```

**Why**: Optimal for 2D spatial queries (games, GIS, collision detection). O(log n) queries when points are well-distributed. Foundation for many spatial databases.

## Async Iterators

```python
# PROBLEM: Create and consume asynchronous iterators for streaming data
# TIME: O(1) per item (amortized) SPACE: O(buffer_size)
import asyncio
from typing import AsyncIterator, List, TypeVar, Generic
from dataclasses import dataclass
from collections import deque

T = TypeVar('T')


class AsyncRange(Generic[T]):
    """Simple async iterator yielding a range of numbers."""
    
    def __init__(self, start: int, end: int, delay: float = 0.1):
        self.start = start
        self.end = end
        self.delay = delay
    
    def __aiter__(self) -> 'AsyncRange[T]':
        return self
    
    async def __anext__(self) -> int:
        if self.start >= self.end:
            raise StopAsyncIteration
        await asyncio.sleep(self.delay)  # Simulate async work
        value = self.start
        self.start += 1
        return value


@dataclass
class Message:
    id: int
    content: str


class AsyncMessageStream:
    """
    Async iterator for message streams with backpressure.
    Demonstrates producer-consumer pattern with bounded buffer.
    """
    
    def __init__(self, max_buffer: int = 10):
        self._buffer: deque[Message] = deque()
        self._max_buffer = max_buffer
        self._closed = False
        self._not_empty = asyncio.Event()
        self._not_full = asyncio.Event()
        self._not_full.set()  # Initially not full
    
    async def publish(self, message: Message) -> None:
        """Publish message (blocks when buffer full)."""
        await self._not_full.wait()
        self._buffer.append(message)
        self._not_empty.set()
        if len(self._buffer) >= self._max_buffer:
            self._not_full.clear()
    
    def close(self) -> None:
        """Signal end of stream."""
        self._closed = True
        self._not_empty.set()  # Wake up consumer
    
    def __aiter__(self) -> 'AsyncMessageStream':
        return self
    
    async def __anext__(self) -> Message:
        await self._not_empty.wait()
        
        if not self._buffer and self._closed:
            raise StopAsyncIteration
        
        if self._buffer:
            message = self._buffer.popleft()
            self._not_full.set()
            return message
        
        raise StopAsyncIteration


async def async_enumerate(
    async_iter: AsyncIterator[T],
    start: int = 0
) -> AsyncIterator[tuple[int, T]]:
    """Async version of enumerate."""
    i = start
    async for item in async_iter:
        yield i, item
        i += 1


async def async_batch(
    async_iter: AsyncIterator[T],
    batch_size: int
) -> AsyncIterator[List[T]]:
    """Batch async iterator items into lists."""
    batch: List[T] = []
    async for item in async_iter:
        batch.append(item)
        if len(batch) >= batch_size:
            yield batch
            batch = []
    if batch:
        yield batch


async def async_merge(
    *iters: AsyncIterator[T]
) -> AsyncIterator[T]:
    """Merge multiple async iterators (interleaved)."""
    async def wrap_with_index(idx: int, it: AsyncIterator[T]) -> AsyncIterator[tuple[int, T]]:
        async for item in it:
            yield idx, item
    
    wrapped = [wrap_with_index(i, it) for i, it in enumerate(iters)]
    nexts = [ait.__anext__() for ait in wrapped]
    
    while nexts:
        done, pending = await asyncio.wait(
            nexts,
            return_when=asyncio.FIRST_COMPLETED
        )
        
        for task in done:
            try:
                idx, item = task.result()
                yield item
                nexts.append(wrapped[idx].__anext__())
            except StopAsyncIteration:
                pass
        
        nexts = list(pending)

class AsyncFileLineReader:
    """Async iterator over file lines with chunked reading."""
    
    def __init__(self, filepath: str, chunk_size: int = 8192):
        self.filepath = filepath
        self.chunk_size = chunk_size
        self._buffer = ""
        self._file = None
    
    async def __aenter__(self) -> 'AsyncFileLineReader':
        # In real code, use aiofiles
        self._file = open(self.filepath, 'r')
        return self
    
    async def __aexit__(self, *args) -> None:
        if self._file:
            self._file.close()
    
    def __aiter__(self) -> 'AsyncFileLineReader':
        return self
    
    async def __anext__(self) -> str:
        while True:
            # Check buffer for complete line
            if '\n' in self._buffer:
                line, self._buffer = self._buffer.split('\n', 1)
                return line
            
            # Read more data
            if self._file:
                chunk = self._file.read(self.chunk_size)
                await asyncio.sleep(0)  # Yield control
                if not chunk:
                    if self._buffer:
                        line = self._buffer
                        self._buffer = ""
                        return line
                    raise StopAsyncIteration
                self._buffer += chunk
            else:
                raise StopAsyncIteration
```

## ARC Cache (Adaptive Replacement Cache)

```python
# PROBLEM: Self-tuning cache combining LRU and LFU strategies
# TIME: O(1) for get/put SPACE: O(capacity)
from collections import OrderedDict
from typing import Optional, Generic, TypeVar, Hashable

K = TypeVar('K', bound=Hashable)
V = TypeVar('V')


class ARCCache(Generic[K, V]):
    """
    Adaptive Replacement Cache (ARC).
    
    Maintains two LRU lists:
    - T1: Recently used items (favors recency)
    - T2: Frequently used items (favors frequency)
    
    Ghost lists (B1, B2) track evicted items to adapt the balance.
    Automatically adjusts between LRU and LFU behavior.
    """
    
    def __init__(self, capacity: int):
        """
        Initialize ARC cache.
        
        Args:
            capacity: Maximum number of items to cache
        """
        self.capacity = capacity
        self.p = 0  # Target size for T1
        
        # Main cache lists (ordered by recency)
        self.t1: OrderedDict[K, V] = OrderedDict()  # Recent
        self.t2: OrderedDict[K, V] = OrderedDict()  # Frequent
        
        # Ghost lists (track evicted keys)
        self.b1: OrderedDict[K, None] = OrderedDict()  # Recent evicted
        self.b2: OrderedDict[K, None] = OrderedDict()  # Frequent evicted
    
    def get(self, key: K) -> Optional[V]:
        """
        Retrieve item from cache.
        
        Returns:
            Value if found, None otherwise
        """
        # Hit in T1: promote to T2 (most recent)
        if key in self.t1:
            value = self.t1.pop(key)
            self.t2[key] = value
            self.t2.move_to_end(key)
            return value
        
        # Hit in T2: just update recency
        if key in self.t2:
            self.t2.move_to_end(key)
            return self.t2[key]
        
        # Miss (including ghost lists)
        return None
    
    def put(self, key: K, value: V) -> None:
        """Insert or update item in cache."""
        # Update existing T1 item
        if key in self.t1:
            self.t1.pop(key)
            self.t2[key] = value
            self.t2.move_to_end(key)
            return
        
        # Update existing T2 item
        if key in self.t2:
            self.t2[key] = value
            self.t2.move_to_end(key)
            return
        
        # Ghost hit in B1: increase target for T1
        if key in self.b1:
            self.p = min(self.capacity, self.p + max(1, len(self.b2) // len(self.b1)))
            self._replace(key)
            self.b1.pop(key)
            self.t2[key] = value
            self.t2.move_to_end(key)
            return
        
        # Ghost hit in B2: decrease target for T1
        if key in self.b2:
            self.p = max(0, self.p - max(1, len(self.b1) // len(self.b2)))
            self._replace(key)
            self.b2.pop(key)
            self.t2[key] = value
            self.t2.move_to_end(key)
            return
        
        # Complete miss: new item
        total_size = len(self.t1) + len(self.t2)
        
        # If at capacity, make room
        if total_size >= self.capacity:
            self._replace(key)
        
        # Add to T1 (recent)
        self.t1[key] = value
        self.t1.move_to_end(key)
    
    def _replace(self, key: K) -> None:
        """
        Replace algorithm: evict from T1 or T2 based on target p.
        Core of ARC's adaptive behavior.
        """
        t1_size = len(self.t1)
        
        if t1_size > 0 and (t1_size > self.p or (t1_size == self.p and key in self.b2)):
            # Evict from T1
            evict_key, _ = self.t1.popitem(last=False)
            self.b1[evict_key] = None
            self.b1.move_to_end(evict_key)
            
            # Trim B1 if too large
            if len(self.b1) > self.capacity:
                self.b1.popitem(last=False)
        else:
            # Evict from T2
            if self.t2:
                evict_key, _ = self.t2.popitem(last=False)
                self.b2[evict_key] = None
                self.b2.move_to_end(evict_key)
                
                # Trim B2 if too large
                if len(self.b2) > self.capacity:
                    self.b2.popitem(last=False)
    
    def __len__(self) -> int:
        return len(self.t1) + len(self.t2)
    
    def __contains__(self, key: K) -> bool:
        return key in self.t1 or key in self.t2
    
    def stats(self) -> dict:
        """Get cache statistics."""
        return {
            'capacity': self.capacity,
            't1_size': len(self.t1),
            't2_size': len(self.t2),
            'b1_size': len(self.b1),
            'b2_size': len(self.b2),
            'target_t1': self.p,
            'total_cached': len(self)
        }


# Demo
if __name__ == "__main__":
    cache = ARCCache[str, int](capacity=4)
    
    # Access pattern: A, B, C, A, D, E, A, B, C, D
    pattern = ['A', 'B', 'C', 'A', 'D', 'E', 'A', 'B', 'C', 'D']
    
    for i, key in enumerate(pattern):
        value = cache.get(key)
        if value is None:
            cache.put(key, i)
            print(f"Step {i+1}: Insert {key} -> {cache.stats()}")
        else:
            print(f"Step {i+1}: Hit {key}")
    
    print("\nFinal stats:", cache.stats())
    print("Cached items:", list(cache.t1.keys()), list(cache.t2.keys()))
```
**Why**: ARC outperforms LRU and LFU by adapting to workload patterns. Used in ZFS, IBM storage, and high-performance databases. No manual tuning required.
