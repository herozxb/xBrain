## 1. Hamiltonian Path

```python
# PROBLEM: Find a path that visits every vertex exactly once in a graph
# TIME: O(N!) SPACE: O(N)
def hamiltonian_path(graph):
    """
    Find Hamiltonian path using backtracking.
    A Hamiltonian path visits each vertex exactly once.
    
    Args:
        graph: adjacency list representation {vertex: [neighbors]}
    
    Returns:
        list of vertices forming Hamiltonian path, or None if not exists
    """
    n = len(graph)
    path = []
    visited = set()
    
    def backtrack(current):
        # Base case: all vertices visited
        if len(path) == n:
            return True
        
        # Try all neighbors
        for neighbor in graph.get(current, []):
            if neighbor not in visited:
                visited.add(neighbor)
                path.append(neighbor)
                
                # Recurse
                if backtrack(neighbor):
                    return True
                
                # Backtrack
                visited.remove(neighbor)
                path.pop()
        
        return False
    
    # Try starting from each vertex
    for start in graph:
        visited = {start}
        path = [start]
        if backtrack(start):
            return path
    
    return None
```
**Why**: Demonstrates backtracking on graphs; useful for route planning and circuit design.

## 2. Sweep Line Algorithm

```python
# PROBLEM: Count maximum overlapping intervals efficiently
# TIME: O(N log N) SPACE: O(N)
def max_overlapping_intervals(intervals):
    """
    Find maximum number of overlapping intervals using sweep line.
    
    Args:
        intervals: list of (start, end) tuples
    
    Returns:
        maximum number of overlapping intervals at any point
    """
    events = []
    
    # Create events: +1 for start, -1 for end
    for start, end in intervals:
        events.append((start, 1))   # interval starts
        events.append((end, -1))    # interval ends
    
    # Sort by position, end events before start events at same position
    events.sort(key=lambda x: (x[0], x[1]))
    
    max_overlap = 0
    current_overlap = 0
    
    for position, delta in events:
        current_overlap += delta
        max_overlap = max(max_overlap, current_overlap)
    
    return max_overlap
```
**Why**: Efficiently solves interval problems; essential for resource allocation and scheduling systems.

## 3. Splay Tree

```python
# PROBLEM: Self-adjusting BST with amortized O(log N) operations
# TIME: O(log N) amortized SPACE: O(N)
class SplayNode:
    def __init__(self, key, value=None):
        self.key = key
        self.value = value
        self.left = None
        self.right = None
        self.parent = None

class SplayTree:
    def __init__(self):
        self.root = None
    
    def _rotate_left(self, x):
        """Left rotation around node x."""
        y = x.right
        x.right = y.left
        if y.left:
            y.left.parent = x
        y.parent = x.parent
        if not x.parent:
            self.root = y
        elif x == x.parent.left:
            x.parent.left = y
        else:
            x.parent.right = y
        y.left = x
        x.parent = y
    
    def _rotate_right(self, x):
        """Right rotation around node x."""
        y = x.left
        x.left = y.right
        if y.right:
            y.right.parent = x
        y.parent = x.parent
        if not x.parent:
            self.root = y
        elif x == x.parent.right:
            x.parent.right = y
        else:
            x.parent.left = y
        y.right = x
        x.parent = y
    
    def _splay(self, x):
        """Move node x to root through rotations."""
        while x.parent:
            if not x.parent.parent:
                # Zig step
                if x == x.parent.left:
                    self._rotate_right(x.parent)
                else:
                    self._rotate_left(x.parent)
            elif x == x.parent.left and x.parent == x.parent.parent.left:
                # Zig-zig step
                self._rotate_right(x.parent.parent)
                self._rotate_right(x.parent)
            elif x == x.parent.right and x.parent == x.parent.parent.right:
                # Zig-zig step
                self._rotate_left(x.parent.parent)
                self._rotate_left(x.parent)
            elif x == x.parent.right and x.parent == x.parent.parent.left:
                # Zig-zag step
                self._rotate_left(x.parent)
                self._rotate_right(x.parent)
            else:
                # Zig-zag step
                self._rotate_right(x.parent)
                self._rotate_left(x.parent)
    
    def insert(self, key, value=None):
        """Insert key-value pair and splay the new node."""
        if not self.root:
            self.root = SplayNode(key, value)
            return
        
        # Standard BST insert
        current = self.root
        while current:
            if key < current.key:
                if not current.left:
                    current.left = SplayNode(key, value)
                    current.left.parent = current
                    self._splay(current.left)
                    return
                current = current.left
            elif key > current.key:
                if not current.right:
                    current.right = SplayNode(key, value)
                    current.right.parent = current
                    self._splay(current.right)
                    return
                current = current.right
            else:
                current.value = value
                self._splay(current)
                return
    
    def search(self, key):
        """Search for key and splay the found/accessed node."""
        if not self.root:
            return None
        
        current = self.root
        while current:
            if key < current.key:
                if not current.left:
                    self._splay(current)
                    return None
                current = current.left
            elif key > current.key:
                if not current.right:
                    self._splay(current)
                    return None
                current = current.right
            else:
                self._splay(current)
                return current.value
        
        return None
```
**Why**: Self-optimizing tree structure; recently accessed elements are faster to access again. Great for caches.

## 4. LRU Cache Variants - LFU Cache

```python
# PROBLEM: Evict least frequently used items when cache is full
# TIME: O(1) for get/put SPACE: O(capacity)
from collections import defaultdict, OrderedDict

class LFUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.min_freq = 0
        self.key_to_val = {}           # key -> value
        self.key_to_freq = {}          # key -> frequency
        self.freq_to_keys = defaultdict(OrderedDict)  # freq -> {key: None}
    
    def get(self, key):
        """Get value and update frequency."""
        if key not in self.key_to_val:
            return -1
        
        # Update frequency
        freq = self.key_to_freq[key]
        del self.freq_to_keys[freq][key]
        
        # Update min_freq if needed
        if not self.freq_to_keys[freq] and freq == self.min_freq:
            self.min_freq += 1
        
        # Add to new frequency bucket
        self.key_to_freq[key] = freq + 1
        self.freq_to_keys[freq + 1][key] = None
        
        return self.key_to_val[key]
    
    def put(self, key, value):
        """Put value and update frequency."""
        if self.capacity <= 0:
            return
        
        # If key exists, update and increase frequency
        if key in self.key_to_val:
            self.key_to_val[key] = value
            self.get(key)  # This updates frequency
            return
        
        # Evict if at capacity
        if len(self.key_to_val) >= self.capacity:
            # Remove least frequently used (and least recently used among those)
            oldest_key, _ = self.freq_to_keys[self.min_freq].popitem(last=False)
            del self.key_to_val[oldest_key]
            del self.key_to_freq[oldest_key]
        
        # Add new key
        self.key_to_val[key] = value
        self.key_to_freq[key] = 1
        self.freq_to_keys[1][key] = None
        self.min_freq = 1
```
**Why**: Alternative to LRU; useful when frequently accessed items should stay cached longer.

## 5. Saga Pattern

```python
# PROBLEM: Coordinate distributed transactions with compensating actions
# TIME: O(N) SPACE: O(N)
class SagaStep:
    def __init__(self, action, compensate):
        """
        A saga step with action and compensation.
        
        Args:
            action: callable that performs the action
            compensate: callable that undoes the action
        """
        self.action = action
        self.compensate = compensate

class SagaOrchestrator:
    def __init__(self):
        self.steps = []
        self.completed_steps = []
    
    def add_step(self, action, compensate):
        """Add a step to the saga."""
        self.steps.append(SagaStep(action, compensate))
        return self
    
    def execute(self):
        """
        Execute saga with automatic compensation on failure.
        
        Returns:
            (success: bool, result: any, error: Exception or None)
        """
        # Execute steps forward
        for i, step in enumerate(self.steps):
            try:
                result = step.action()
                self.completed_steps.append((i, step, result))
            except Exception as e:
                # Compensate in reverse order
                self._compensate()
                return False, None, e
        
        return True, self.completed_steps, None
    
    def _compensate(self):
        """Execute compensating transactions in reverse order."""
        for i, step, result in reversed(self.completed_steps):
            try:
                step.compensate(result)
            except Exception as e:
                # Log compensation failure but continue
                print(f"Compensation failed for step {i}: {e}")

# Example usage
if __name__ == "__main__":
    def book_hotel():
        print("Hotel booked")
        return {"hotel_id": 123}
    
    def cancel_hotel(result):
        print(f"Hotel {result['hotel_id']} cancelled")
    
    def book_flight():
        print("Flight booked")
        return {"flight_id": 456}
    
    def cancel_flight(result):
        print(f"Flight {result['flight_id']} cancelled")
    
    def charge_credit_card():
        # Simulate failure
        raise Exception("Credit card declined")
    
    def refund_credit_card(result):
        print("Credit card refunded")
    
    saga = SagaOrchestrator()
    saga.add_step(book_hotel, cancel_hotel)
    saga.add_step(book_flight, cancel_flight)
    saga.add_step(charge_credit_card, refund_credit_card)
    
    success, result, error = saga.execute()
    # Output: Hotel booked -> Flight booked -> Flight cancelled -> Hotel cancelled
```
**Why**: Maintains consistency in distributed systems without 2PC; each step can be undone if later steps fail.

## 6. Dependency Injection

```python
# PROBLEM: Decouple component dependencies for testable, modular code
# TIME: O(1) lookup SPACE: O(N)
from typing import Dict, Type, Callable, Any
from functools import wraps

class DIContainer:
    """
    Simple dependency injection container.
    Supports singleton and transient lifecycles.
    """
    
    def __init__(self):
        self._registrations: Dict[Type, Dict] = {}
        self._singletons: Dict[Type, Any] = {}
    
    def register_singleton(self, interface: Type, implementation: Type):
        """Register a singleton (single instance for lifetime)."""
        self._registrations[interface] = {
            'implementation': implementation,
            'lifecycle': 'singleton'
        }
    
    def register_transient(self, interface: Type, implementation: Type):
        """Register transient (new instance each time)."""
        self._registrations[interface] = {
            'implementation': implementation,
            'lifecycle': 'transient'
        }
    
    def register_instance(self, interface: Type, instance: Any):
        """Register an existing instance."""
        self._singletons[interface] = instance
        self._registrations[interface] = {
            'implementation': type(instance),
            'lifecycle': 'instance'
        }
    
    def resolve(self, interface: Type) -> Any:
        """
        Resolve dependency recursively.
        Automatically injects constructor dependencies.
        """
        if interface not in self._registrations:
            raise ValueError(f"Type {interface} not registered")
        
        registration = self._registrations[interface]
        
        # Return existing singleton or instance
        if registration['lifecycle'] in ('singleton', 'instance'):
            if interface in self._singletons:
                return self._singletons[interface]
        
        implementation = registration['implementation']
        
        # Auto-wire constructor dependencies
        init_signature = implementation.__init__.__annotations__
        dependencies = {}
        
        for param_name, param_type in init_signature.items():
            if param_name != 'return' and param_type in self._registrations:
                dependencies[param_name] = self.resolve(param_type)
        
        instance = implementation(**dependencies)
        
        # Cache singleton
        if registration['lifecycle'] == 'singleton':
            self._singletons[interface] = instance
        
        return instance

# Example usage
class ILogger:
    def log(self, message: str) -> None:
        pass

class ConsoleLogger(ILogger):
    def log(self, message: str) -> None:
        print(f"[LOG] {message}")

class Database:
    def __init__(self, logger: ILogger):
        self.logger = logger
    
    def query(self, sql: str):
        self.logger.log(f"Executing: {sql}")
        return "result"

# Configure container
container = DIContainer()
container.register_singleton(ILogger, ConsoleLogger)
container.register_transient(Database, Database)

# Resolve dependencies
db = container.resolve(Database)  # Logger is auto-injected
db.query("SELECT * FROM users")
```
**Why**: Enables loose coupling and easy testing; swap implementations without changing consumers.

## 7. Leaky Bucket Rate Limiter

```python
# PROBLEM: Rate limit requests with smooth traffic output
# TIME: O(1) SPACE: O(1)
import time
from threading import Lock

class LeakyBucket:
    """
    Leaky bucket rate limiter.
    Requests fill the bucket; bucket drains at constant rate.
    """
    
    def __init__(self, capacity: int, leak_rate: float):
        """
        Args:
            capacity: maximum bucket size (burst capacity)
            leak_rate: requests per second that drain from bucket
        """
        self.capacity = capacity
        self.leak_rate = leak_rate  # requests per second
        self.water = 0  # current water level
        self.last_leak = time.time()
        self.lock = Lock()
    
    def _leak(self):
        """Remove water that has leaked since last check."""
        now = time.time()
        elapsed = now - self.last_leak
        leaked = elapsed * self.leak_rate
        self.water = max(0, self.water - leaked)
        self.last_leak = now
    
    def allow_request(self) -> bool:
        """
        Check if request is allowed.
        
        Returns:
            True if request allowed, False if rate limited
        """
        with self.lock:
            self._leak()
            
            if self.water < self.capacity:
                self.water += 1
                return True
            
            return False
    
    def time_until_available(self) -> float:
        """Seconds until next request is allowed."""
        with self.lock:
            self._leak()
            if self.water < self.capacity:
                return 0.0
            return (self.water - self.capacity + 1) / self.leak_rate

# Example usage
if __name__ == "__main__":
    limiter = LeakyBucket(capacity=10, leak_rate=2)  # 10 burst, 2 req/s
    
    for i in range(15):
        if limiter.allow_request():
            print(f"Request {i}: Allowed")
        else:
            print(f"Request {i}: Rate limited (wait {limiter.time_until_available():.2f}s)")
        time.sleep(0.1)
```
**Why**: Smooths bursty traffic into steady stream; prevents API abuse while allowing reasonable bursts.

## 8. Consistent Hashing Load Balancer

```python
# PROBLEM: Distribute requests with minimal remapping when servers change
# TIME: O(log N) lookup SPACE: O(N)
import hashlib
from bisect import bisect_right
from typing import List, Optional

class ConsistentHashLoadBalancer:
    """
    Consistent hashing with virtual nodes.
    Distributes keys uniformly and minimizes remapping on changes.
    """
    
    def __init__(self, virtual_nodes: int = 150):
        """
        Args:
            virtual_nodes: number of virtual nodes per server for uniformity
        """
        self.virtual_nodes = virtual_nodes
        self.ring = {}          # hash -> server
        self.sorted_keys = []   # sorted hash values
        self.servers = set()
    
    def _hash(self, key: str) -> int:
        """Generate consistent hash for a key."""
        return int(hashlib.md5(key.encode()).hexdigest(), 16)
    
    def add_server(self, server: str) -> None:
        """Add server with virtual nodes to the ring."""
        self.servers.add(server)
        
        for i in range(self.virtual_nodes):
            virtual_key = f"{server}#{i}"
            hash_val = self._hash(virtual_key)
            self.ring[hash_val] = server
            self.sorted_keys.append(hash_val)
        
        self.sorted_keys.sort()
    
    def remove_server(self, server: str) -> None:
        """Remove server and all its virtual nodes."""
        self.servers.discard(server)
        
        for i in range(self.virtual_nodes):
            virtual_key = f"{server}#{i}"
            hash_val = self._hash(virtual_key)
            if hash_val in self.ring:
                del self.ring[hash_val]
                self.sorted_keys.remove(hash_val)
    
    def get_server(self, key: str) -> Optional[str]:
        """
        Find server responsible for a key.
        
        Args:
            key: the key to look up
        
        Returns:
            server name or None if ring is empty
        """
        if not self.ring:
            return None
        
        hash_val = self._hash(key)
        
        # Find first node with hash >= key's hash
        idx = bisect_right(self.sorted_keys, hash_val)
        
        # Wrap around to first node if past the end
        if idx >= len(self.sorted_keys):
            idx = 0
        
        return self.ring[self.sorted_keys[idx]]
    
    def get_distribution(self, keys: List[str]) -> dict:
        """Analyze distribution of keys across servers."""
        distribution = {server: 0 for server in self.servers}
        
        for key in keys:
            server = self.get_server(key)
            if server:
                distribution[server] += 1
        
        return distribution

# Example usage
if __name__ == "__main__":
    lb = ConsistentHashLoadBalancer(virtual_nodes=150)
    
    # Add servers
    lb.add_server("server1")
    lb.add_server("server2")
    lb.add_server("server3")
    
    # Route requests
    for user_id in ["user_1", "user_2", "user_3", "user_4", "user_5"]:
        server = lb.get_server(user_id)
        print(f"{user_id} -> {server}")
    
    # Check distribution
    test_keys = [f"key_{i}" for i in range(1000)]
    dist = lb.get_distribution(test_keys)
    print(f"Distribution: {dist}")
    
    # Add new server - only ~1/N keys remap
    lb.add_server("server4")
    new_dist = lb.get_distribution(test_keys)
    print(f"After adding server4: {new_dist}")
```
**Why**: Minimizes cache invalidation when servers change; only K/N keys remap when adding/removing server (K=keys, N=servers).

## 9. Two-Phase Commit

```python
# PROBLEM: Ensure atomic transaction across distributed databases
# TIME: O(N) with network latency SPACE: O(N)
from enum import Enum
from typing import List, Dict, Optional
import time

class TransactionState(Enum):
    INIT = "INIT"
    PREPARED = "PREPARED"
    COMMITTED = "COMMITTED"
    ABORTED = "ABORTED"

class Participant:
    """Database participant in 2PC transaction."""
    
    def __init__(self, name: str):
        self.name = name
        self.state = TransactionState.INIT
        self.transaction_data = None
    
    def prepare(self, data) -> bool:
        """
        Phase 1: Prepare to commit.
        Reserve resources and ensure commit is possible.
        """
        try:
            # Simulate resource reservation
            self.transaction_data = data
            self.state = TransactionState.PREPARED
            print(f"[{self.name}] Prepared for transaction")
            return True
        except Exception as e:
            print(f"[{self.name}] Prepare failed: {e}")
            self.state = TransactionState.ABORTED
            return False
    
    def commit(self) -> bool:
        """Phase 2: Commit the transaction."""
        if self.state != TransactionState.PREPARED:
            return False
        
        try:
            # Simulate committing data
            self.state = TransactionState.COMMITTED
            print(f"[{self.name}] Committed transaction")
            return True
        except Exception as e:
            print(f"[{self.name}] Commit failed: {e}")
            return False
    
    def rollback(self) -> bool:
        """Abort the transaction."""
        try:
            self.transaction_data = None
            self.state = TransactionState.ABORTED
            print(f"[{self.name}] Rolled back transaction")
            return True
        except Exception as e:
            print(f"[{self.name}] Rollback failed: {e}")
            return False

class TwoPhaseCoordinator:
    """Coordinator for two-phase commit protocol."""
    
    def __init__(self, participants: List[Participant]):
        self.participants = participants
        self.state = TransactionState.INIT
        self.timeout = 5.0  # seconds
    
    def execute_transaction(self, data) -> bool:
        """
        Execute distributed transaction with 2PC.
        
        Returns:
            True if committed successfully, False if aborted
        """
        # Phase 1: Prepare
        print("=== PHASE 1: PREPARE ===")
        all_prepared = True
        
        for participant in self.participants:
            if not participant.prepare(data):
                all_prepared = False
                break
        
        # Phase 2: Commit or Abort
        if all_prepared:
            print("\n=== PHASE 2: COMMIT ===")
            self.state = TransactionState.COMMITTED
            for participant in self.participants:
                participant.commit()
            return True
        else:
            print("\n=== PHASE 2: ABORT ===")
            self.state = TransactionState.ABORTED
            for participant in self.participants:
                if participant.state == TransactionState.PREPARED:
                    participant.rollback()
            return False

# Example usage
if __name__ == "__main__":
    # Simulate distributed transaction across 3 databases
    db1 = Participant("OrderDB")
    db2 = Participant("InventoryDB")
    db3 = Participant("PaymentDB")
    
    coordinator = TwoPhaseCoordinator([db1, db2, db3])
    
    # Execute distributed transaction
    order_data = {"order_id": 123, "amount": 99.99}
    success = coordinator.execute_transaction(order_data)
    
    print(f"\nTransaction result: {'SUCCESS' if success else 'FAILED'}")
```
**Why**: Guarantees atomicity across distributed databases; all commit or all rollback together.

# PROBLEM: Stream large datasets asynchronously without loading all into memory
# TIME: O(N) SPACE: O(1) - constant memory regardless of dataset size
import asyncio
from typing import AsyncGenerator

async def fetch_page(session, url: str, page: int) -> list:
    """Simulate fetching a page of data from API."""
    await asyncio.sleep(0.1)  # Simulate network latency
    # Return mock data
    return [{"id": page * 10 + i, "data": f"item_{page}_{i}"} for i in range(10)]

async def paginated_data_stream(
    session,
    base_url: str,
    total_pages: int
) -> AsyncGenerator[dict, None]:
    """
    Async generator that yields items from paginated API.
    Processes one page at a time, constant memory usage.
    
    Yields:
        Individual items from all pages
    """
    for page in range(total_pages):
        items = await fetch_page(session, base_url, page)
        
        for item in items:
            yield item  # Yield one item at a time

async def process_large_file(filepath: str) -> AsyncGenerator[str, None]:
    """
    Async generator for reading large file line by line.
    """
    with open(filepath, 'r') as f:
        for line in f:
            # Simulate async processing
            await asyncio.sleep(0.01)
            yield line.strip()

async def batch_processor(
    source: AsyncGenerator,
    batch_size: int = 100
) -> AsyncGenerator[list, None]:
    """
    Batch items from async generator into chunks.
    
    Args:
        source: async generator of items
        batch_size: items per batch
    
    Yields:
        List of items of batch_size
    """
    batch = []
    
    async for item in source:
        batch.append(item)
        
        if len(batch) >= batch_size:
            yield batch
            batch = []
    
    # Yield remaining items
    if batch:
        yield batch

async def transform_stream(
    source: AsyncGenerator,
    transformer: callable
) -> AsyncGenerator:
    """
    Apply transformation to each item in stream.
    """
    async for item in source:
        yield transformer(item)

async def filter_stream(
    source: AsyncGenerator,
    predicate: callable
) -> AsyncGenerator:
    """
    Filter items from async stream.
    """
    async for item in source:
        if predicate(item):
            yield item

# Example: Compose async generators into pipeline
async def data_pipeline():
    """Demonstrate async generator pipeline."""
    
    # Source: paginated data

        for i in range(50):

    # Consume the pipeline
