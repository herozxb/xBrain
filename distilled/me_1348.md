## 1. Tim Sort

```python
# PROBLEM: Efficient hybrid sorting algorithm combining merge sort and insertion sort,
# designed to perform well on real-world data with existing order patterns.
# TIME: O(n log n) worst/average, O(n) best (already sorted) SPACE: O(n)

def tim_sort(arr, min_run=32):
    """
    Tim Sort is Python's built-in sort algorithm. It divides the array into
    small runs, sorts them with insertion sort, then merges using merge sort.
    """
    n = len(arr)
    
    # Helper: insertion sort for small subarrays
    def insertion_sort(left, right):
        for i in range(left + 1, right + 1):
            key = arr[i]
            j = i - 1
            while j >= left and arr[j] > key:
                arr[j + 1] = arr[j]
                j -= 1
            arr[j + 1] = key
    
    # Helper: merge two sorted runs
    def merge(left, mid, right):
        left_copy = arr[left:mid + 1]
        right_copy = arr[mid + 1:right + 1]
        
        i = j = 0
        k = left
        
        while i < len(left_copy) and j < len(right_copy):
            if left_copy[i] <= right_copy[j]:
                arr[k] = left_copy[i]
                i += 1
            else:
                arr[k] = right_copy[j]
                j += 1
            k += 1
        
        # Copy remaining elements
        while i < len(left_copy):
            arr[k] = left_copy[i]
            i += 1
            k += 1
        while j < len(right_copy):
            arr[k] = right_copy[j]
            j += 1
            k += 1
    
    # Calculate minimum run size (power of 2 between 32 and 64)
    def calc_min_run(size):
        r = 0
        while size >= min_run:
            r |= size & 1
            size >>= 1
        return size + r
    
    if n < 2:
        return arr
    
    run_size = calc_min_run(n)
    
    # Sort individual runs with insertion sort
    for start in range(0, n, run_size):
        end = min(start + run_size - 1, n - 1)
        insertion_sort(start, end)
    
    # Merge runs of increasing size
    size = run_size
    while size < n:
        for left in range(0, n, 2 * size):
            mid = min(n - 1, left + size - 1)
            right = min(n - 1, left + 2 * size - 1)
            if mid < right:
                merge(left, mid, right)
        size *= 2
    
    return arr
```

**Why**: Tim Sort is the default sorting algorithm in Python, Java, and Android. It excels on partially ordered real-world data, making it more practical than textbook algorithms.

## 2. LFU Cache

```python
# PROBLEM: Design a cache that evicts the least frequently used items when full.
# When multiple items have same frequency, evict the least recently used among them.
# TIME: O(1) get/put SPACE: O(capacity)

from collections import defaultdict, OrderedDict

class LFUCache:
    """
    LFU Cache with O(1) operations using nested dictionaries.
    Maintains frequency buckets where each bucket is an OrderedDict for LRU.
    """
    
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.size = 0
        self.min_freq = 0
        
        # key -> (value, frequency)
        self.key_to_val_freq = {}
        
        # frequency -> OrderedDict of keys (maintains LRU order)
        self.freq_to_keys = defaultdict(OrderedDict)
    
    def get(self, key: int) -> int:
        """Retrieve value and update frequency."""
        if key not in self.key_to_val_freq:
            return -1
        
        val, freq = self.key_to_val_freq[key]
        
        # Remove from current frequency bucket
        del self.freq_to_keys[freq][key]
        
        # If this was the last key at min_freq, increment min_freq
        if not self.freq_to_keys[freq] and freq == self.min_freq:
            self.min_freq += 1
        
        # Add to next frequency bucket
        self.freq_to_keys[freq + 1][key] = None
        self.key_to_val_freq[key] = (val, freq + 1)
        
        return val
    
    def put(self, key: int, value: int) -> None:
        """Insert or update a key-value pair."""
        if self.capacity <= 0:
            return
        
        # If key exists, update value and frequency
        if key in self.key_to_val_freq:
            _, freq = self.key_to_val_freq[key]
            self.key_to_val_freq[key] = (value, freq)
            self.get(key)  # Update frequency
            return
        
        # Evict if at capacity
        if self.size >= self.capacity:
            # Remove LRU from minimum frequency bucket
            lru_key, _ = self.freq_to_keys[self.min_freq].popitem(last=False)
            del self.key_to_val_freq[lru_key]
            self.size -= 1
        
        # Insert new key
        self.key_to_val_freq[key] = (value, 1)
        self.freq_to_keys[1][key] = None
        self.min_freq = 1
        self.size += 1
```

**Why**: LFU cache is crucial in CDN caching, database buffer pools, and web browsers where access frequency indicates importance better than recency alone.

## 3. Leaky Bucket Rate Limiter

```python
# PROBLEM: Rate limiting algorithm that smooths burst traffic into steady flow.
# Requests queue up and "leak" at a constant rate, rejecting when bucket overflows.
# TIME: O(1) per request SPACE: O(bucket_size)

import time
from threading import Lock
from collections import deque

class LeakyBucket:
    """
    Leaky Bucket rate limiter: requests enter the bucket and leak out
    at a constant rate. If bucket is full, requests are dropped.
    """
    
    def __init__(self, capacity: int, leak_rate: float):
        """
        Args:
            capacity: Maximum number of requests in bucket
            leak_rate: Requests processed per second
        """
        self.capacity = capacity
        self.leak_rate = leak_rate  # requests per second
        self.bucket = deque()
        self.lock = Lock()
    
    def _leak(self):
        """Remove requests that have "leaked out" since last check."""
        now = time.time()
        while self.bucket:
            # Calculate time since oldest request
            elapsed = now - self.bucket[0]
            # Number of requests that should have leaked
            leaked = elapsed * self.leak_rate
            
            if leaked >= 1:
                self.bucket.popleft()
            else:
                break
    
    def allow_request(self) -> bool:
        """
        Try to add a request to the bucket.
        Returns True if accepted, False if bucket is full.
        """
        with self.lock:
            self._leak()
            
            if len(self.bucket) < self.capacity:
                self.bucket.append(time.time())
                return True
            return False
    
    def get_queue_size(self) -> int:
        """Get current number of requests waiting in bucket."""
        with self.lock:
            self._leak()
            return len(self.bucket)
```

**Why**: Leaky bucket enforces a smooth output rate, ideal for API rate limiting, network traffic shaping, and preventing sudden load spikes on backend services.

## 4. Dinic's Algorithm (Max Flow)

```python
# PROBLEM: Find maximum flow in a flow network from source to sink.
# TIME: O(V²E) general, O(E√V) for bipartite graphs SPACE: O(V + E)

from collections import deque

class DinicMaxFlow:
    """
    Dinic's algorithm for maximum flow using BFS level graph
    and DFS blocking flow with current edge optimization.
    """
    
    class Edge:
        def __init__(self, to: int, capacity: int, reverse_edge: int):
            self.to = to
            self.capacity = capacity
            self.reverse_edge = reverse_edge
    
    def __init__(self, num_nodes: int):
        self.n = num_nodes
        self.graph = [[] for _ in range(num_nodes)]
    
    def add_edge(self, from_node: int, to_node: int, capacity: int):
        """Add directed edge with given capacity."""
        forward = self.Edge(to_node, capacity, len(self.graph[to_node]))
        backward = self.Edge(from_node, 0, len(self.graph[from_node]))
        
        self.graph[from_node].append(forward)
        self.graph[to_node].append(backward)
    
    def _bfs_level(self, source: int, sink: int, level: list) -> bool:
        """Build level graph using BFS. Returns True if sink is reachable."""
        for i in range(self.n):
            level[i] = -1
        
        queue = deque([source])
        level[source] = 0
        
        while queue:
            node = queue.popleft()
            
            for edge in self.graph[node]:
                if level[edge.to] < 0 and edge.capacity > 0:
                    level[edge.to] = level[node] + 1
                    queue.append(edge.to)
        
        return level[sink] >= 0
    
    def _dfs_flow(self, node: int, sink: int, flow: int, 
                  level: list, start: list) -> int:
        """Find blocking flow using DFS with current edge optimization."""
        if node == sink:
            return flow
        
        # Process edges starting from 'start[node]'
        for i in range(start[node], len(self.graph[node])):
            edge = self.graph[node][i]
            
            if level[edge.to] == level[node] + 1 and edge.capacity > 0:
                # Recursively find flow
                pushed = self._dfs_flow(
                    edge.to, sink, 
                    min(flow, edge.capacity), 
                    level, start
                )
                
                if pushed > 0:
                    # Update residual capacities
                    edge.capacity -= pushed
                    self.graph[edge.to][edge.reverse_edge].capacity += pushed
                    return pushed
            
            start[node] += 1
        
        return 0
    
    def max_flow(self, source: int, sink: int) -> int:
        """Calculate maximum flow from source to sink."""
        flow = 0
        level = [-1] * self.n
        
        while self._bfs_level(source, sink, level):
            # Reset start indices for current edge optimization
            start = [0] * self.n
            
            while True:
                pushed = self._dfs_flow(source, sink, float('inf'), level, start)
                if pushed == 0:
                    break
                flow += pushed
            
            level = [-1] * self.n
        
        return flow
```

**Why**: Max flow algorithms are fundamental in network routing, bipartite matching, airline scheduling, image segmentation, and resource allocation problems.

## 5. Descriptors (Python Advanced)

```python
# PROBLEM: Create reusable attribute validation/access control without decorators.
# Descriptors allow custom behavior when getting/setting/deleting attributes.
# TIME: O(1) per access SPACE: O(1) overhead per descriptor instance

class ValidatedAttribute:
    """
    A descriptor that validates values on assignment.
    Demonstrates __get__, __set__, and __delete__ protocol.
    """
    
    def __init__(self, validator=None, default=None):
        self.validator = validator
        self.default = default
        self.attr_name = None
    
    def __set_name__(self, owner, name):
        """Called when descriptor is assigned to a class attribute."""
        self.attr_name = f"_descriptor_{name}"
    
    def __get__(self, obj, objtype=None):
        """Called when attribute is accessed: obj.attr"""
        if obj is None:
            return self  # Access through class, not instance
        return getattr(obj, self.attr_name, self.default)
    
    def __set__(self, obj, value):
        """Called when attribute is assigned: obj.attr = value"""
        if self.validator and not self.validator(value):
            raise ValueError(f"Invalid value: {value}")
        setattr(obj, self.attr_name, value)
    
    def __delete__(self, obj):
        """Called when attribute is deleted: del obj.attr"""
        if hasattr(obj, self.attr_name):
            delattr(obj, self.attr_name)


class RangeDescriptor:
    """
    Descriptor enforcing numeric range constraints.
    Uses weakref to avoid memory leaks with instance storage.
    """
    
    def __init__(self, min_val=None, max_val=None):
        self.min_val = min_val
        self.max_val = max_val
    
    def __set_name__(self, owner, name):
        self.name = name
    
    def __get__(self, obj, objtype=None):
        if obj is None:
            return self
        return obj.__dict__.get(self.name)
    
    def __set__(self, obj, value):
        if not isinstance(value, (int, float)):
            raise TypeError(f"{self.name} must be numeric")
        
        if self.min_val is not None and value < self.min_val:
            raise ValueError(f"{self.name} must be >= {self.min_val}")
        if self.max_val is not None and value > self.max_val:
            raise ValueError(f"{self.name} must be <= {self.max_val}")
        
        obj.__dict__[self.name] = value


# Example usage
class Product:
    price = RangeDescriptor(min_val=0)
    quantity = RangeDescriptor(min_val=0, max_val=1000)
    name = ValidatedAttribute(
        validator=lambda x: isinstance(x, str) and len(x) > 0,
        default=""
    )
    
    def __init__(self, name, price, quantity):
        self.name = name
        self.price = price
        self.quantity = quantity
```

**Why**: Descriptors power Python's `@property`, `@classmethod`, `@staticmethod`, and SQLAlchemy ORM fields. Essential for creating clean, reusable attribute behavior in libraries and frameworks.

## 6. Dependency Injection Container

```python
# PROBLEM: Decouple component creation from usage, enabling testable, modular code.
# DI containers manage object lifecycles and resolve dependencies automatically.
# TIME: O(1) resolution (amortized) SPACE: O(n) for registered services

from typing import Any, Callable, Dict, Type, TypeVar, Optional
from enum import Enum
from dataclasses import dataclass
from functools import wraps

T = TypeVar('T')

class Lifecycle(Enum):
    TRANSIENT = "transient"  # New instance each time
    SINGLETON = "singleton"  # Single shared instance
    SCOPED = "scoped"        # One instance per scope


@dataclass
class ServiceDescriptor:
    service_type: Type
    implementation: Any
    lifecycle: Lifecycle
    instance: Optional[Any] = None
    factory: Optional[Callable] = None


class DIContainer:
    """
    Lightweight dependency injection container supporting
    singleton, transient, and scoped lifecycles.
    """
    
    def __init__(self):
        self._services: Dict[Type, ServiceDescriptor] = {}
        self._singletons: Dict[Type, Any] = {}
    
    def register_singleton(self, service_type: Type[T], 
                           implementation: Optional[Type[T]] = None) -> None:
        """Register a singleton service (single shared instance)."""
        impl = implementation or service_type
        self._services[service_type] = ServiceDescriptor(
            service_type=service_type,
            implementation=impl,
            lifecycle=Lifecycle.SINGLETON
        )
    
    def register_transient(self, service_type: Type[T],
                           implementation: Optional[Type[T]] = None) -> None:
        """Register a transient service (new instance each request)."""
        impl = implementation or service_type
        self._services[service_type] = ServiceDescriptor(
            service_type=service_type,
            implementation=impl,
            lifecycle=Lifecycle.TRANSIENT
        )
    
    def register_factory(self, service_type: Type[T], 
                         factory: Callable[['DIContainer'], T],
                         lifecycle: Lifecycle = Lifecycle.TRANSIENT) -> None:
        """Register using a factory function."""
        self._services[service_type] = ServiceDescriptor(
            service_type=service_type,
            implementation=None,
            lifecycle=lifecycle,
            factory=factory
        )
    
    def resolve(self, service_type: Type[T]) -> T:
        """Resolve a service and its dependencies."""
        if service_type not in self._services:
            raise ValueError(f"Service {service_type.__name__} not registered")
        
        descriptor = self._services[service_type]
        
        # Return cached singleton if exists
        if descriptor.lifecycle == Lifecycle.SINGLETON:
            if service_type in self._singletons:
                return self._singletons[service_type]
        
        # Create new instance
        if descriptor.factory:
            instance = descriptor.factory(self)
        else:
            instance = self._create_instance(descriptor.implementation)
        
        # Cache singleton
        if descriptor.lifecycle == Lifecycle.SINGLETON:
            self._singletons[service_type] = instance
        
        return instance
    
    def _create_instance(self, implementation: Type[T]) -> T:
        """Create instance with resolved constructor dependencies."""
        import inspect
        
        sig = inspect.signature(implementation.__init__)
        kwargs = {}
        
        for param_name, param in sig.parameters.items():
            if param_name == 'self':
                continue
            
            param_type = param.annotation
            if param_type != inspect.Parameter.empty:
                if param_type in self._services:
                    kwargs[param_name] = self.resolve(param_type)
        
        return implementation(**kwargs)


def injectable(func: Callable) -> Callable:
    """Decorator to inject container dependencies into function."""
    @wraps(func)
    def wrapper(container: DIContainer, *args, **kwargs):
        import inspect
        sig = inspect.signature(func)
        
        bound_args = {}
        for param_name, param in sig.parameters.items():
            if param_name == 'container':
                continue
            if param_name in kwargs:
                bound_args[param_name] = kwargs[param_name]
            elif param.annotation in container._services:
                bound_args[param_name] = container.resolve(param.annotation)
        
        return func(container=container, **bound_args)
    
    return wrapper
```

**Why**: Dependency injection enables clean architecture, simplifies testing with mock injection, and is fundamental to frameworks like FastAPI, Django, and enterprise Java applications.

## 7. Boyer-Moore String Matching

```python
# PROBLEM: Efficient string search using bad character and good suffix heuristics.
# TIME: O(n/m) best (skipping ahead), O(nm) worst SPACE: O(alphabet_size)
# Average case is often sublinear due to skipping.

class BoyerMoore:
    """
    Boyer-Moore string matching with bad character and good suffix rules.
    Scans pattern right-to-left, skipping ahead when mismatch found.
    """
    
    def __init__(self, pattern: str):
        self.pattern = pattern
        self.m = len(pattern)
        
        # Precompute bad character table
        self.bad_char = self._build_bad_char_table()
        
        # Precompute good suffix table
        self.good_suffix = self._build_good_suffix_table()
    
    def _build_bad_char_table(self) -> dict:
        """
        For each character, record its rightmost position in pattern.
        If mismatch at position i with character c, shift to align
        rightmost c in pattern with the mismatched position.
        """
        table = {}
        for i, char in enumerate(self.pattern):
            table[char] = i
        return table
    
    def _build_good_suffix_table(self) -> list:
        """
        When a suffix of pattern matches but then mismatches,
        shift to next occurrence of that suffix in pattern,
        or to a prefix that matches suffix of the matched portion.
        """
        m = self.m
        table = [0] * (m + 1)
        
        # Case 1: matching suffix appears elsewhere in pattern
        suffix = [0] * (m + 1)
        
        # Find borders (prefix = suffix)
        i = m
        j = m + 1
        suffix[i] = j
        
        while i > 0:
            while j <= m and self.pattern[i - 1] != self.pattern[j - 1]:
                if table[j] == 0:
                    table[j] = j - i
                j = suffix[j]
            
            i -= 1
            j -= 1
            suffix[i] = j
        
        # Case 2: suffix of matched part matches prefix of pattern
        j = suffix[0]
        for i in range(m + 1):
            if table[i] == 0:
                table[i] = j
            if i == j:
                j = suffix[j]
        
        return table
    
    def search(self, text: str) -> list:
        """Find all occurrences of pattern in text."""
        n = len(text)
        if self.m == 0 or n < self.m:
            return []
        
        matches = []
        i = 0  # position in text
        
        while i <= n - self.m:
            j = self.m - 1  # position in pattern (scan right-to-left)
            
            # Match from right to left
            while j >= 0 and self.pattern[j] == text[i + j]:
                j -= 1
            
            if j < 0:
                # Full match found
                matches.append(i)
                # Shift by good suffix value for full match
                i += self.good_suffix[0]
            else:
                # Mismatch at position j
                # Calculate shift from bad character rule
                bad_char_shift = j - self.bad_char.get(text[i + j], -1)
                
                # Use maximum of bad character and good suffix shifts
                i += max(bad_char_shift, self.good_suffix[j + 1])
        
        return matches
```

**Why**: Boyer-Moore is the fastest string search algorithm for typical English text, used in grep, text editors, and DNA sequence analysis. Often runs in sublinear time by skipping characters.

## 8. Composite Pattern

```python
# PROBLEM: Treat individual objects and compositions uniformly. Build tree structures
# where clients ignore differences between leaf and composite nodes.
# TIME: O(n) traversal SPACE: O(n) for tree structure

from abc import ABC, abstractmethod
from typing import List, Optional, Iterator
from dataclasses import dataclass

# Component interface
class Graphic(ABC):
    """Base class for all graphic elements."""
    
    @abstractmethod
    def draw(self) -> str:
        """Render this graphic."""
        pass
    
    @abstractmethod
    def get_bounds(self) -> tuple:
        """Return (x, y, width, height) bounding box."""
        pass
    
    def is_composite(self) -> bool:
        """Check if this is a composite (can have children)."""
        return False


# Leaf
@dataclass
class Dot(Graphic):
    """A simple point graphic."""
    x: int
    y: int
    
    def draw(self) -> str:
        return f"● at ({self.x}, {self.y})"
    
    def get_bounds(self) -> tuple:
        return (self.x, self.y, 0, 0)


@dataclass
class Circle(Graphic):
    """A circle graphic."""
    x: int
    y: int
    radius: int
    
    def draw(self) -> str:
        return f"◯ center=({self.x}, {self.y}) radius={self.radius}"
    
    def get_bounds(self) -> tuple:
        return (
            self.x - self.radius,
            self.y - self.radius,
            self.radius * 2,
            self.radius * 2
        )


# Composite
class GraphicGroup(Graphic):
    """
    A composite that can contain other graphics.
    Supports recursive tree structures.
    """
    
    def __init__(self, name: str = "Group"):
        self._name = name
        self._children: List[Graphic] = []
    
    def add(self, graphic: Graphic) -> 'GraphicGroup':
        """Add a child graphic."""
        self._children.append(graphic)
        return self
    
    def remove(self, graphic: Graphic) -> None:
        """Remove a child graphic."""
        self._children.remove(graphic)
    
    def get_children(self) -> List[Graphic]:
        """Return all children."""
        return self._children.copy()
    
    def is_composite(self) -> bool:
        return True
    
    def draw(self) -> str:
        lines = [f"[{self._name}]"]
        for child in self._children:
            # Indent child drawings
            child_draw = child.draw()
            lines.append("  " + child_draw.replace("\n", "\n  "))
        return "\n".join(lines)
    
    def get_bounds(self) -> tuple:
        """Calculate bounding box encompassing all children."""
        if not self._children:
            return (0, 0, 0, 0)
        
        min_x = min_y = float('inf')
        max_x = max_y = float('-inf')
        
        for child in self._children:
            x, y, w, h = child.get_bounds()
            min_x = min(min_x, x)
            min_y = min(min_y, y)
            max_x = max(max_x, x + w)
            max_y = max(max_y, y + h)
        
        return (min_x, min_y, max_x - min_x, max_y - min_y)
    
    def __iter__(self) -> Iterator[Graphic]:
        """Iterate over children."""
        return iter(self._children)
    
    def find_at(self, x: int, y: int) -> Optional[Graphic]:
        """Find the topmost graphic at given coordinates."""
        # Search in reverse order (topmost first)
        for child in reversed(self._children):
            if child.is_composite():
                result = child.find_at(x, y)
                if result:
                    return result
            else:
                bx, by, bw, bh = child.get_bounds()
                if bx <= x <= bx + bw and by <= y <= by + bh:
                    return child
        return None
```

**Why**: Composite pattern is essential in GUI frameworks (widget trees), file systems (files/folders), organizational hierarchies, and any recursive tree structure where uniform treatment simplifies client code.

## 9. Async Generators & Context Managers

```python
# PROBLEM: Handle streaming data asynchronously with proper resource cleanup.
# Async generators yield values over time; async context managers ensure cleanup.
# TIME: O(1) per iteration SPACE: O(1) (streaming, not buffering)

import asyncio
from typing import AsyncIterator, AsyncContextManager
from contextlib import asynccontextmanager


class AsyncLineReader:
    """
    Async context manager for reading files line by line.
    Combines async context manager and async iterator protocols.
    """
    
    def __init__(self, filepath: str, chunk_size: int = 8192):
        self.filepath = filepath
        self.chunk_size = chunk_size
        self._file = None
        self._buffer = ""
    
    async def __aenter__(self):
        self._file = open(self.filepath, 'r')
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        if self._file:
            self._file.close()
        return False  # Don't suppress exceptions
    
    def __aiter__(self):
        return self
    
    async def __anext__(self) -> str:
        """Return next complete line from file."""
        while True:
            # Check buffer for complete line
            line_end = self._buffer.find('\n')
            if line_end != -1:
                line = self._buffer[:line_end]
                self._buffer = self._buffer[line_end + 1:]
                return line
            
            # Read more data (simulated async read)
            await asyncio.sleep(0)  # Yield control
            chunk = self._file.read(self.chunk_size)
            
            if not chunk:
                if self._buffer:
                    result = self._buffer
                    self._buffer = ""
                    return result
                raise StopAsyncIteration
            
            self._buffer += chunk


async def stream_range(start: int, end: int, 
                       delay: float = 0.1) -> AsyncIterator[int]:
    """
    Async generator that yields values over time.
    Simulates streaming data with delays.
    """
    for i in range(start, end + 1):
        await asyncio.sleep(delay)
        yield i


@asynccontextmanager
async def async_timer(name: str):
    """
    Async context manager for timing async code blocks.
    """
    start = asyncio.get_event_loop().time()
    print(f"[{name}] Starting...")
    
    try:
        yield
    finally:
        elapsed = asyncio.get_event_loop().time() - start
        print(f"[{name}] Completed in {elapsed:.3f}s")


async def process_stream():
    """
    Example combining async generators, context managers,
    and async iteration.
    """
    # Process streaming data
    async with async_timer("stream processing"):
        total = 0
        async for value in stream_range(1, 10, delay=0.05):
            total += value
            print(f"  Received: {value}, running total: {total}")
        
        print(f"Final total: {total}")
    
    # Demonstrate async comprehension
    squared = [x * x async for x in stream_range(1, 5, delay=0.02)]
    print(f"Squared values: {squared}")


class RateLimitedStream:
    """
    Async generator with rate limiting using token bucket.
    """
    
    def __init__(self, source: AsyncIterator, 
                 rate: float = 10.0,  # items per second
                 burst: int = 5):
        self.source = source
        self.rate = rate
        self.burst = burst
        self.tokens = burst
        self.last_time = None
    
    def __aiter__(self):
        return self
    
    async def __anext__(self):
        # Replenish tokens based on elapsed time
        now = asyncio.get_event_loop().time()
        
        if self.last_time is not None:
            elapsed = now - self.last_time
            self.tokens = min(self.burst, 
                            self.tokens + elapsed * self.rate)
        
        self.last_time = now
        
        # Wait if no tokens available
        if self.tokens < 1:
            wait_time = (1 - self.tokens) / self.rate
            await asyncio.sleep(wait_time)
            self.tokens = 0
        else:
            self.tokens -= 1
        
        return await self.source.__anext__()
```

**Why**: Async generators power real-time data pipelines, WebSocket streams, database cursors, and any I/O-bound streaming scenario. Essential for modern async Python applications.

## 10. Distributed Lock (Redlock-inspired)

```python
# PROBLEM: Acquire exclusive locks across distributed nodes.
# Prevents race conditions when multiple processes access shared resources.
# TIME: O(n) where n = number of Redis instances SPACE: O(1)

import time
import uuid
import hashlib
from typing import List, Optional
from dataclasses import dataclass
from threading import Lock


@dataclass
class LockResult:
    """Result of a distributed lock acquisition attempt."""
    acquired: bool
    lock_id: Optional[str] = None
    validity_time: float = 0.0
    error: Optional[str] = None


class MockRedisNode:
    """Simulated Redis node for demonstration."""
    
    def __init__(self, name: str):
        self.name = name
        self._locks = {}
        self._lock = Lock()
    
    def set_if_not_exists(self, key: str, value: str, 
                          ttl_ms: int) -> bool:
        """SET key value NX PX ttl - atomic set if not exists."""
        with self._lock:
            if key in self._locks:
                return False
            self._locks[key] = {
                'value': value,
                'expires': time.time() + ttl_ms / 1000
            }
            return True
    
    def get(self, key: str) -> Optional[str]:
        """Get value if exists and not expired."""
        with self._lock:
            if key not in self._locks:
                return None
            
            lock_info = self._locks[key]
            if time.time() > lock_info['expires']:
                del self._locks[key]
                return None
            
            return lock_info['value']
    
    def delete_if_match(self, key: str, value: str) -> bool:
        """DEL key if value matches (atomic unlock)."""
        with self._lock:
            if key not in self._locks:
                return False
            
            if self._locks[key]['value'] == value:
                del self._locks[key]
                return True
            
            return False


class DistributedLock:
    """
    Redlock-inspired distributed lock using multiple Redis nodes.
    Requires majority quorum to acquire lock.
    """
    
    def __init__(self, nodes: List[MockRedisNode], 
                 lock_name: str,
                 ttl_ms: int = 10000,
                 retry_delay_ms: int = 200,
                 retry_count: int = 3):
        self.nodes = nodes
        self.lock_name = lock_name
        self.ttl_ms = ttl_ms
        self.retry_delay_ms = retry_delay_ms
        self.retry_count = retry_count
        self.quorum = len(nodes) // 2 + 1
        self._lock_id = None
    
    def _generate_lock_id(self) -> str:
        """Generate unique lock identifier."""
        return str(uuid.uuid4())
    
    def _acquire_on_node(self, node: MockRedisNode, 
                         lock_id: str) -> bool:
        """Try to acquire lock on a single node."""
        return node.set_if_not_exists(
            self.lock_name, 
            lock_id, 
            self.ttl_ms
        )
    
    def _release_on_node(self, node: MockRedisNode, 
                         lock_id: str) -> bool:
        """Release lock on a single node if we own it."""
        return node.delete_if_match(self.lock_name, lock_id)
    
    def acquire(self) -> LockResult:
        """
        Attempt to acquire distributed lock.
        Must succeed on majority of nodes within validity period.
        """
        lock_id = self._generate_lock_id()
        
        for attempt in range(self.retry_count):
            start_time = time.time()
            acquired_count = 0
            
            # Try to acquire on all nodes in parallel
            for node in self.nodes:
                if self._acquire_on_node(node, lock_id):
                    acquired_count += 1
            
            # Calculate time elapsed and remaining validity
            elapsed_ms = (time.time() - start_time) * 1000
            validity_time = self.ttl_ms - elapsed_ms
            
            # Check if we have quorum and valid time remaining
            if acquired_count >= self.quorum and validity_time > 0:
                self._lock_id = lock_id
                return LockResult(
                    acquired=True,
                    lock_id=lock_id,
                    validity_time=validity_time / 1000
                )
            
            # Failed - release any locks we acquired
            for node in self.nodes:
                self._release_on_node(node, lock_id)
            
            # Wait before retry
            if attempt < self.retry_count - 1:
                time.sleep(self.retry_delay_ms / 1000)
        
        return LockResult(
            acquired=False,
            error="Failed to acquire quorum"
        )
    
    def release(self) -> bool:
        """Release the lock if we hold it."""
        if not self._lock_id:
            return False
        
        released_count = 0
        for node in self.nodes:
            if self._release_on_node(node, self._lock_id):
                released_count += 1
        
        self._lock_id = None
        return released_count >= self.quorum
    
    def extend(self, extend_ms: int = None) -> bool:
        """Extend the lock TTL (requires re-acquisition)."""
        if not self._lock_id:
            return False
        
        # In production, use Lua script for atomic extension
        # Here we just re-acquire with same lock_id
        return False  # Simplified for demo


# Context manager usage
class distributed_lock_context:
    """Context manager for distributed lock."""
    
    def __init__(self, lock: DistributedLock):
        self.lock = lock
        self.result = None
    
    def __enter__(self):
        self.result = self.lock.acquire()
        if not self.result.acquired:
            raise RuntimeError("Failed to acquire distributed lock")
        return self.result
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        self.lock.release()
        return False
```

**Why**: Distributed locks are critical in microservices for preventing concurrent access to shared resources like databases, file storage, or external APIs. Redlock provides safety even when individual nodes fail.
