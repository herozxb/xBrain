    -->
    
    <form id="dataForm" action="/submit" method="POST">
        <label for="username">Username:</label>
        <input type="text" id="username" name="username" required>
        
        <!-- Hidden inputs store data without UI visibility -->
        <input type="hidden" name="userId" value="12345">
        <input type="hidden" name="sessionId" value="abc-xyz-789">
        <input type="hidden" name="timestamp" id="formTimestamp">
        <input type="hidden" name="csrf_token" value="secure_token_here">
        
        <button type="submit">Submit</button>
    </form>
    
    <script>
        /**
         * Sets hidden field values dynamically before form submission
         * Useful for tracking, security tokens, and state management
         */
        function setHiddenFieldValue(inputId, value) {
            const hiddenInput = document.getElementById(inputId);
            if (hiddenInput && hiddenInput.type === 'hidden') {
                hiddenInput.value = value;
                return true;
            }
            return false;
        }
        
        /**
         * Gets hidden field value by input ID
         */
        function getHiddenFieldValue(inputId) {
            const hiddenInput = document.getElementById(inputId);
            return hiddenInput ? hiddenInput.value : null;
        }
        
        // Set timestamp when form is about to be submitted
        document.getElementById('dataForm').addEventListener('submit', function(e) {
            setHiddenFieldValue('formTimestamp', Date.now());
            console.log('Form submitted with hidden data:', {
                userId: getHiddenFieldValue('userId') || this.querySelector('[name="userId"]').value,
                sessionId: this.querySelector('[name="sessionId"]').value
            });
        });
        
        // Initialize timestamp on page load
        document.addEventListener('DOMContentLoaded', function() {
            setHiddenFieldValue('formTimestamp', Date.now());
        });
    </script>
</body>
</html>
```

**Why**: Hidden inputs securely transmit tracking IDs, CSRF tokens, session data, and timestamps without cluttering the UI or allowing user modification.

    <!-- 
        PROBLEM: Apply styles to entire table columns efficiently without repeating classes
        TIME: O(n) where n = number of columns SPACE: O(1)
    -->
    
    <table>
        <!-- colgroup defines column groups for styling -->
        <colgroup>
            <!-- First column: narrow ID column -->
            <col style="background-color: #f0f0f0; width: 50px;">
            <!-- Second column: medium name column -->
            <col style="background-color: #e8f4e8; width: 150px;">
            <!-- Third column spans 2 columns: wide description -->
            <col span="2" style="background-color: #fffde8; width: 200px;">
        </colgroup>
        
        <thead>
            <tr>
                <th>ID</th>
                <th>Name</th>
                <th>Email</th>
                <th>Status</th>
            </tr>
        </thead>
        <tbody>
            <tr>
                <td>1</td>
                <td>John Doe</td>
                <td>john@example.com</td>
                <td>Active</td>
            </tr>
            <tr>
                <td>2</td>
                <td>Jane Smith</td>
                <td>jane@example.com</td>
                <td>Pending</td>
            </tr>
        </tbody>
    </table>
    
    <script>
        /**
         * Dynamically creates colgroup for a table based on column config
         * @param {HTMLTableElement} table - Target table element
         * @param {Array} columns - Array of column config objects
         */
        function createColgroup(table, columns) {
            // Remove existing colgroup if present
            const existingColgroup = table.querySelector('colgroup');
            if (existingColgroup) {
                existingColgroup.remove();
            }
            
            // Create new colgroup
            const colgroup = document.createElement('colgroup');
            
            columns.forEach(config => {
                const col = document.createElement('col');
                
                // Apply span if specified
                if (config.span) {
                    col.setAttribute('span', config.span);
                }
                
                // Apply styles
                if (config.style) {
                    Object.assign(col.style, config.style);
                }
                
                // Apply CSS class
                if (config.className) {
                    col.className = config.className;
                }
                
                colgroup.appendChild(col);
            });
            
            // Insert at the beginning of table
            table.insertBefore(colgroup, table.firstChild);
            
            return colgroup;
        }
        
        // Example: Update column styles dynamically
        const myTable = document.querySelector('table');
        createColgroup(myTable, [
            { style: { backgroundColor: '#ffebee', width: '60px' } },
            { style: { backgroundColor: '#e3f2fd', width: '120px' } },
            { span: 2, style: { backgroundColor: '#f3e5f5', width: '180px' } }
        ]);
    </script>
</body>
</html>
```

**Why**: Colgroup efficiently styles entire columns at once, reducing CSS repetition and improving table rendering performance by defining column structure upfront.

    <script>
        /**
         * Creates a figure element with image and caption
         * @param {string} src - Image source URL
         * @param {string} alt - Alt text for accessibility
         * @param {string} caption - Figcaption text
         * @param {Object} options - Optional settings (width, height, id)
         * @returns {HTMLFigureElement}
         */
        function createFigure(src, alt, caption, options = {}) {
            // Create figure container
            const figure = document.createElement('figure');
            
            // Create image element
            const img = document.createElement('img');
            img.src = src;
            img.alt = alt;
            img.loading = 'lazy'; // Native lazy loading
            
            if (options.width) img.width = options.width;
            if (options.height) img.height = options.height;
            
            // Create caption element
            const figcaption = document.createElement('figcaption');
            figcaption.textContent = caption;
            
            // Assemble figure
            figure.appendChild(img);
            figure.appendChild(figcaption);
            
            if (options.id) figure.id = options.id;
            if (options.className) figure.className = options.className;
            
            return figure;
        }
        
        /**
         * Dynamically adds a figure to the page
         */
        function addFigureToPage(container, src, alt, caption, options = {}) {
            const figure = createFigure(src, alt, caption, options);
            container.appendChild(figure);
            return figure;
        }
        
        // Example usage: Add a figure dynamically
        document.addEventListener('DOMContentLoaded', function() {
            const container = document.body;
            addFigureToPage(
                container,
                'https://via.placeholder.com/300x200',

    <script>
        /**
         * Applies border styles to an element dynamically
         * @param {HTMLElement} element - Target element
         * @param {Object} borderConfig - Border configuration object
         */
        function applyBorder(element, borderConfig) {
            const {
                width = '1px',
                style = 'solid',
                color = '#000',
                radius = '0',
                top, right, bottom, left
            } = borderConfig;
            
            // Apply shorthand border if no sides specified
            if (!top && !right && !bottom && !left) {
                element.style.border = `${width} ${style} ${color}`;
            } else {
                // Apply individual sides
                if (top) element.style.borderTop = `${top.width || width} ${top.style || style} ${top.color || color}`;
                if (right) element.style.borderRight = `${right.width || width} ${right.style || style} ${right.color || color}`;
                if (bottom) element.style.borderBottom = `${bottom.width || width} ${bottom.style || style} ${bottom.color || color}`;
                if (left) element.style.borderLeft = `${left.width || width} ${left.style || style} ${left.color || color}`;
            }
            
            // Apply border radius
            if (radius) {
                if (typeof radius === 'object') {
                    element.style.borderRadius = `${radius.topLeft || 0} ${radius.topRight || 0} ${radius.bottomRight || 0} ${radius.bottomLeft || 0}`;
                } else {
                    element.style.borderRadius = radius;
                }
            }
            
            return element;
        }
        
        /**
         * Creates an animated border effect
         */
        function createAnimatedBorder(element, colors, duration = 2000) {
            let colorIndex = 0;
            
            setInterval(() => {
                colorIndex = (colorIndex + 1) % colors.length;
                element.style.borderColor = colors[colorIndex];
            }, duration / colors.length);
        }
        
        // Example: Apply custom border dynamically
        const demoElement = document.createElement('div');
        demoElement.className = 'demo-box';
        demoElement.textContent = 'Dynamic Border';
        document.body.appendChild(demoElement);
        
        applyBorder(demoElement, {
            width: '3px',
            style: 'solid',
            color: '#e83e8c',
            radius: { topLeft: '20px', bottomRight: '20px', topRight: '0', bottomLeft: '0' }
        });
        
        // Add animation
        createAnimatedBorder(demoElement, ['#e83e8c', '#6f42c1', '#007bff', '#28a745'], 2000);
    </script>
</body>
</html>
```

**Why**: CSS borders define element boundaries visually, enhance UI hierarchy, create visual separation, and provide interactive feedback through various styles and animations.

<body>
    <!-- 
        PROBLEM: Control spacing outside elements using CSS margins
        TIME: O(1) SPACE: O(1)
    -->
    
    <h2>Margin Examples</h2>
    
    <div class="container">
        <!-- Margin on all sides -->
        <div class="box" style="margin: 20px;">Margin: 20px all sides</div>
        
        <!-- Vertical | Horizontal -->
        <div class="box" style="margin: 10px 30px;">Margin: 10px vertical, 30px horizontal</div>
        
        <!-- Top | Horizontal | Bottom -->
        <div class="box" style="margin: 5px 20px 40px;">Margin: 5px top, 20px sides, 40px bottom</div>
        
        <!-- Auto horizontal centering -->
        <div class="spacer">
            <div class="box" style="margin: 20px auto; display: block; width: 200px;">
                Centered with auto margin
            </div>
        </div>
    </div>
    
    <script>
        /**
         * Applies margin styles to an element
         * @param {HTMLElement} element - Target element
         * @param {Object|string} margin - Margin value(s)
         */
        function setMargin(element, margin) {
            if (typeof margin === 'string') {
                element.style.margin = margin;
            } else {
                const { top, right, bottom, left, horizontal, vertical } = margin;
                
                if (top !== undefined) element.style.marginTop = top;
                if (right !== undefined) element.style.marginRight = right;
                if (bottom !== undefined) element.style.marginBottom = bottom;
                if (left !== undefined) element.style.marginLeft = left;
                if (horizontal !== undefined) {
                    element.style.marginLeft = horizontal;
                    element.style.marginRight = horizontal;
                }
                if (vertical !== undefined) {
                    element.style.marginTop = vertical;
                    element.style.marginBottom = vertical;
                }
            }
            return element;
        }
        
        /**
         * Centers an element horizontally using auto margins
         * @param {HTMLElement} element - Element to center
         * @param {string} width - Width of the element
         */
        function centerHorizontally(element, width) {
            element.style.display = 'block';
            element.style.width = width;
            element.style.marginLeft = 'auto';
            element.style.marginRight = 'auto';
            return element;
        }
        
        /**
         * Calculates total margin space of an element
         * @param {HTMLElement} element - Target element
         * @returns {Object} Margin values in pixels
         */
        function getComputedMargins(element) {
            const style = window.getComputedStyle(element);
            return {
                top: parseFloat(style.marginTop),
                right: parseFloat(style.marginRight),
                bottom: parseFloat(style.marginBottom),
                left: parseFloat(style.marginLeft),
                total: parseFloat(style.marginTop) + parseFloat(style.marginBottom) +
                       parseFloat(style.marginLeft) + parseFloat(style.marginRight)
            };
        }
        
        // Example: Dynamic margin adjustment
        const newBox = document.createElement('div');
        newBox.className = 'box';
        newBox.textContent = 'Dynamic Margins';
        document.querySelector('.container').appendChild(newBox);
        
        setMargin(newBox, {
            top: '30px',
            horizontal: '50px',
            bottom: '10px'
        });
        
        console.log('Computed margins:', getComputedMargins(newBox));
    </script>
</body>
</html>
```

**Why**: Margins create external spacing between elements, enable horizontal centering with auto values, and manage layout rhythm without affecting element dimensions.

## js_functions

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>JavaScript Functions</title>
</head>
<body>
    <!-- 
        PROBLEM: Define reusable blocks of code with various function syntaxes
        TIME: O(1) for definition, varies for execution SPACE: O(n) for closures
    -->
    
    <h2>Function Types in JavaScript</h2>
    <div id="output"></div>
    
    <script>
        const output = document.getElementById('output');
        
        /**
         * 1. Function Declaration - Hoisted, can be called before definition
         */
        function greet(name) {
            return `Hello, ${name}!`;
        }
        
        /**
         * 2. Function Expression - Not hoisted
         */
        const add = function(a, b) {
            return a + b;
        };
        
        /**
         * 3. Arrow Function - Concise syntax, no own 'this'
         */
        const multiply = (a, b) => a * b;
        
        /**
         * 4. Arrow Function with body
         */
        const divide = (a, b) => {
            if (b === 0) throw new Error('Division by zero');
            return a / b;
        };
        
        /**
         * 5. Immediately Invoked Function Expression (IIFE)
         */
        const counter = (function() {
            let count = 0;
            return {
                increment: () => ++count,
                decrement: () => --count,
                getCount: () => count
            };
        })();
        
        /**
         * 6. Generator Function - Yields multiple values
         */
        function* numberGenerator(max) {
            for (let i = 1; i <= max; i++) {
                yield i;
            }
        }
        
        /**
         * 7. Async Function - Returns a Promise
         */
        async function fetchData(url) {
            const response = await fetch(url);
            return response.json();
        }
        
        /**
         * 8. Higher-Order Function - Takes/returns functions
         */
        function createMultiplier(factor) {
            return (number) => number * factor;
        }
        
        /**
         * 9. Recursive Function - Calls itself
         */
        function factorial(n) {
            if (n <= 1) return 1;
            return n * factorial(n - 1);
        }
        
        /**
         * 10. Function with Default Parameters & Rest
         */
        function sum(message = 'Sum:', ...numbers) {
            const total = numbers.reduce((acc, n) => acc + n, 0);
            return `${message} ${total}`;
        }
        
        // Demonstration
        const results = [
            `Declaration: ${greet('World')}`,
            `Expression: 5 + 3 = ${add(5, 3)}`,
            `Arrow: 4 * 7 = ${multiply(4, 7)}`,
            `Arrow body: 20 / 4 = ${divide(20, 4)}`,
            `IIFE counter: ${counter.increment()}, ${counter.increment()}, ${counter.getCount()}`,
            `Generator: ${[...numberGenerator(5)].join(', ')}`,
            `Higher-Order: ${createMultiplier(5)(10)}`,
            `Recursive: 5! = ${factorial(5)}`,
            `Default & Rest: ${sum('Total:', 1, 2, 3, 4, 5)}`
        ];
        
        output.innerHTML = results.map(r => `<p>${r}</p>`).join('');
        
        // Export for testing
        if (typeof module !== 'undefined') {
            module.exports = { greet, add, multiply, divide, factorial, createMultiplier };
        }
    </script>
</body>
</html>
```

**Why**: Functions encapsulate reusable logic, enable abstraction, support functional programming patterns, and provide the foundation for modular, maintainable JavaScript code.

    <!--

    -->
    
    <h2>JavaScript Scope Examples</h2>
    <div id="output"></div>
    
    <script>
        const output = document.getElementById('output');
        const logs = [];
        
        // ==========================================
        // 1. GLOBAL SCOPE
        // ==========================================
        var globalVar = 'I am global (var)';
        let globalLet = 'I am global (let)';
        
        /**
         * Demonstrates different scope types in JavaScript
         */
        function demonstrateScope() {
            
            // ==========================================
            // 2. FUNCTION SCOPE
            // ==========================================
            var functionScoped = 'Only accessible in this function';
            
            // ==========================================
            // 3. BLOCK SCOPE (let, const)
            // ==========================================
            if (true) {
                let blockLet = 'Block-scoped let';
                const blockConst = 'Block-scoped const';
                var notBlockScoped = 'var ignores blocks';
                
                logs.push(`Inside block: ${blockLet}, ${blockConst}`);
            }
            
            // var is function-scoped, so this works:
            logs.push(`var outside block: ${notBlockScoped}`);
            
            // This would throw ReferenceError:
            // console.log(blockLet);
            
            // ==========================================
            // 4. LEXICAL SCOPE (Closures)
            // ==========================================
            function outerFunction(outerParam) {
                const outerVar = 'Outer variable';
                
                function innerFunction(innerParam) {
                    // Inner has access to outer's scope
                    logs.push(`Inner can access: ${outerParam}, ${outerVar}`);
                    return `${outerParam} - ${innerParam}`;
                }
                
                return innerFunction;
            }
            
            const closure = outerFunction('Outer Value');
            closure('Inner Value');
            
            // ==========================================
            // 5. SCOPE CHAIN
            // ==========================================
            const level1 = 'Level 1';
            
            function scopeChainDemo() {
                const level2 = 'Level 2';
                
                function level3Function() {
                    const level3 = 'Level 3';
                    // Can access all levels up the chain
                    logs.push(`Scope chain: ${level1}, ${level2}, ${level3}`);
                }
                
                level3Function();
            }
            
            scopeChainDemo();
            
            // ==========================================
            // 6. MODULE PATTERN (Private Scope)
            // ==========================================
            const counterModule = (function() {
                let count = 0; // Private variable
                
                return {
                    increment() { return ++count; },
                    decrement() { return --count; },
                    getCount() { return count; }
                };
            })();
            
            logs.push(`Module count: ${counterModule.increment()}, ${counterModule.increment()}`);
            // Cannot access: counterModule.count (private)
            
            // ==========================================
            // 7. TEMPORAL DEAD ZONE (TDZ)
            // ==========================================
            function tdzDemo() {
                // This would throw ReferenceError (TDZ):
                // console.log(tdzVar);
                let tdzVar = 'Now accessible';
                logs.push(`After declaration: ${tdzVar}`);
            }
            
            tdzDemo();
        }
        
        // Run demonstrations
        demonstrateScope();
        
        // Display results
        output.innerHTML = logs.map(log => `<p>${log}</p>`).join('');
        
        // Export module for testing
        if (typeof module !== 'undefined') {
            module.exports = { counterModule };
        }
    </script>

**Why**: Understanding scope prevents variable collisions, enables data privacy through closures, supports modular code organization, and helps avoid common bugs with hoisting and TDZ.

## dom_append_child

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>DOM appendChild</title>
    <style>
        .list-item { 
            padding: 10px; 
            margin: 5px 0; 
            background: #f0f0f0; 
            border-left: 3px solid #007bff; 
        }
        #container { padding: 20px; background: #fafafa; }
    </style>
</head>
<body>
    <!-- 
        PROBLEM: Add DOM elements as the last child of a parent element
        TIME: O(1) for single append SPACE: O(n) for n elements
    -->
    
    <h2>appendChild Examples</h2>
    
    <div id="container">
        <p>Original content</p>
    </div>
    
    <ul id="list">
        <li>Item 1</li>
    </ul>
    
    <button onclick="addListItem()">Add Item</button>
    <button onclick="addMultipleItems()">Add 5 Items</button>
    
    <script>
        /**
         * Appends a single element to a parent
         * @param {HTMLElement} parent - Parent container
         * @param {HTMLElement} child - Element to append
         * @returns {HTMLElement} The appended child
         */
        function appendElement(parent, child) {
            return parent.appendChild(child);
        }
        
        /**
         * Creates and appends an element in one operation
         * @param {HTMLElement} parent - Parent container
         * @param {string} tag - HTML tag name
         * @param {Object} attributes - Element attributes
         * @param {string} textContent - Text content
         */
        function createAndAppend(parent, tag, attributes = {}, textContent = '') {
            const element = document.createElement(tag);
            
            // Set attributes
            Object.entries(attributes).forEach(([key, value]) => {
                if (key === 'className') {
                    element.className = value;
                } else if (key === 'style' && typeof value === 'object') {
                    Object.assign(element.style, value);
                } else {
                    element.setAttribute(key, value);
                }
            });
            
            // Set text content
            if (textContent) {
                element.textContent = textContent;
            }
            
            parent.appendChild(element);
            return element;
        }
        
        /**
         * Efficiently appends multiple elements using DocumentFragment
         * @param {HTMLElement} parent - Parent container
         * @param {Array<HTMLElement>} children - Elements to append
         */
        function appendMultiple(parent, children) {
            // DocumentFragment reduces reflows - single DOM update
            const fragment = document.createDocumentFragment();
            
            children.forEach(child => {
                fragment.appendChild(child);
            });
            
            parent.appendChild(fragment);
            return children;
        }
        
        /**
         * Moves an existing element to a new parent
         * @param {HTMLElement} element - Element to move
         * @param {HTMLElement} newParent - New parent container
         */
        function moveElement(element, newParent) {
            // appendChild moves the element if it already has a parent
            newParent.appendChild(element);
            return element;
        }
        
        // Interactive examples
        let itemCount = 1;
        
        function addListItem() {
            const list = document.getElementById('list');
            const li = document.createElement('li');
            li.className = 'list-item';
            li.textContent = `Item ${++itemCount}`;
            list.appendChild(li);
        }
        
        function addMultipleItems() {
            const list = document.getElementById('list');
            const newItems = [];
            
            for (let i = 0; i < 5; i++) {
                const li = document.createElement('li');
                li.className = 'list-item';
                li.textContent = `Item ${++itemCount}`;
                newItems.push(li);
            }
            
            appendMultiple(list, newItems);
        }
        
        // Initial demo
        const container = document.getElementById('container');
        createAndAppend(container, 'div', { className: 'list-item' }, 'Dynamically appended element');
        
        // Example of moving an element
        const originalP = container.querySelector('p');
        const newSection = createAndAppend(container, 'section', { style: { marginTop: '10px', padding: '10px', background: '#e8f4e8' } });
        moveElement(originalP, newSection);
    </script>
</body>
</html>
```

**Why**: appendChild is the fundamental DOM method for adding elements, supports element movement between parents, and with DocumentFragment enables efficient batch operations minimizing reflows.

## event_mouseover

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Mouseover Event</title>
    <style>
        .hover-box {
            width: 150px;
            height: 100px;
            margin: 10px;
            display: inline-block;
            padding: 20px;
            background: #f0f0f0;
            border: 2px solid #ccc;
            transition: all 0.3s ease;
            cursor: pointer;
        }
        .hover-box.active {
            background: #007bff;
            color: white;
            transform: scale(1.05);
        }
        .tooltip {
            position: absolute;
            background: #333;
            color: white;
            padding: 8px 12px;
            border-radius: 4px;
            font-size: 14px;
            pointer-events: none;
            opacity: 0;
            transition: opacity 0.2s;
        }
        .tooltip.visible {
            opacity: 1;
        }
    </style>
</head>
<body>
    <!-- 
        PROBLEM: Detect when mouse enters an element and respond appropriately
        TIME: O(1) SPACE: O(1)
    -->
    
    <h2>Mouseover Event Examples</h2>
    
    <div class="hover-box" data-info="First box">Hover Box 1</div>
    <div class="hover-box" data-info="Second box">Hover Box 2</div>
    <div class="hover-box" data-info="Third box">Hover Box 3</div>
    
    <div id="tooltip" class="tooltip"></div>
    
    <p id="status">Status: Waiting for hover...</p>
    
    <script>
        /**
         * Sets up mouseover/mouseout event handlers
         * @param {HTMLElement} element - Target element
         * @param {Function} onOver - Callback for mouseover
         * @param {Function} onOut - Callback for mouseout
         */
        function setupHover(element, onOver, onOut) {
            element.addEventListener('mouseover', function(e) {
                this.classList.add('active');
                if (onOver) onOver.call(this, e);
            });
            
            element.addEventListener('mouseout', function(e) {
                this.classList.remove('active');
                if (onOut) onOut.call(this, e);
            });
        }
        
        /**
         * Creates a tooltip that follows the mouse
         * @param {string} selector - Element selector
         * @param {Function} contentFn - Function to get tooltip content
         */
        function createTooltip(selector, contentFn) {
            const tooltip = document.getElementById('tooltip');
            const elements = document.querySelectorAll(selector);
            
            elements.forEach(el => {
                el.addEventListener('mouseover', function(e) {
                    tooltip.textContent = contentFn(this);
                    tooltip.classList.add('visible');
                });
                
                el.addEventListener('mousemove', function(e) {
                    tooltip.style.left = (e.pageX + 15) + 'px';
                    tooltip.style.top = (e.pageY + 15) + 'px';
                });
                
                el.addEventListener('mouseout', function() {
                    tooltip.classList.remove('visible');
                });
            });
        }
        
        /**
         * Debounced hover handler to prevent rapid firing
         * @param {HTMLElement} element - Target element
         * @param {number} delay - Debounce delay in ms
         */
        function debouncedHover(element, delay = 100) {
            let timeout;
            
            element.addEventListener('mouseover', function() {
                clearTimeout(timeout);
                timeout = setTimeout(() => {
                    console.log(`Debounced hover on: ${this.textContent}`);
                }, delay);
            });
            
            element.addEventListener('mouseout', function() {
                clearTimeout(timeout);
            });
        }
        
        /**
         * mouseenter vs mouseover comparison
         * mouseover: bubbles, triggers for child elements
         * mouseenter: does not bubble, only for the element itself
         */
        function compareMouseEvents(container) {
            const statusEl = document.getElementById('status');
            
            container.addEventListener('mouseover', function(e) {
                statusEl.textContent = `mouseover: ${e.target.textContent}`;
                console.log('mouseover (bubbles):', e.target.textContent);
            });
            
            container.addEventListener('mouseenter', function(e) {
                console.log('mouseenter (no bubble):', this.textContent);
            });
        }
        
        // Initialize hover effects
        document.querySelectorAll('.hover-box').forEach(box => {
            setupHover(
                box,
                function() { console.log(`Hovering: ${this.textContent}`); },
                function() { console.log(`Left: ${this.textContent}`); }
            );
            
            debouncedHover(box, 150);
        });
        
        // Initialize tooltips
        createTooltip('.hover-box', el => el.dataset.info);
        
        // Compare mouse events
        compareMouseEvents(document.querySelector('.hover-box'));
    </script>
</body>
</html>
```

**Why**: Mouseover events enable interactive UI feedback, tooltips, hover states, and dynamic content preview. Understanding bubbling vs non-bubbling (mouseenter) is crucial for nested element handling.
