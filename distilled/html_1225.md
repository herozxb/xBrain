## js_template_literals_tagged

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Tagged Template Literals</title>
</head>
<body>
    <h2>Tagged Template Literals Demo</h2>
    <div id="output"></div>
    
    <script>
        // PROBLEM: Process template literals with custom logic
        // TIME: O(n) where n = number of template parts SPACE: O(n)
        
        /* Why: Tagged templates enable syntax highlighting, 
           internationalization, and custom string processing */
        
        // Basic tag function
        function highlight(strings, ...values) {
            let result = strings[0];
            for (let i = 0; i < values.length; i++) {
                result += `<mark>${values[i]}</mark>` + strings[i + 1];
            }
            return result;
        }
        
        // Usage
        const name = "Alice";
        const age = 30;
        const highlighted = highlight`User ${name} is ${age} years old`;
        
        // Escape HTML for safety
        function safeHtml(strings, ...values) {
            let result = strings[0];
            for (let i = 0; i < values.length; i++) {
                const escaped = String(values[i])
                    .replace(/&/g, '&amp;')
                    .replace(/</g, '&lt;')
                    .replace(/>/g, '&gt;')
                    .replace(/"/g, '&quot;');
                result += escaped + strings[i + 1];
            }
            return result;
        }
        
        const userInput = '<script>alert("XSS")<\/script>';
        const safe = safeHtml`User input: ${userInput}`;
        
        // Currency formatting
        function currency(strings, ...values) {
            return strings.reduce((result, str, i) => {
                const value = values[i] !== undefined 
                    ? '$' + values[i].toFixed(2) 
                    : '';
                return result + str + value;
            }, '');
        }
        
        const price = 29.99;
        const tax = 2.50;
        const total = currency`Price: ${price}, Tax: ${tax}`;
        
        // Template for i18n
        function i18n(strings, ...values) {
            const translations = {
                'Hello': 'Hola',
                'world': 'mundo'
            };
            
            let result = strings[0];
            for (let i = 0; i < values.length; i++) {
                const translated = translations[values[i]] || values[i];
                result += translated + strings[i + 1];
            }
            return result;
        }
        
        const greeting = i18n`${'Hello'}, ${'world'}!`;
        
        // Display results
        document.getElementById('output').innerHTML = `
            <h3>Examples:</h3>
            <p><strong>Highlight:</strong> ${highlighted}</p>
            <p><strong>Safe HTML:</strong> ${safe}</p>
            <p><strong>Currency:</strong> ${total}</p>
            <p><strong>i18n:</strong> ${greeting}</p>
            
            <h3>Code Examples:</h3>
            <pre>
// Highlight tag
highlight\`User \${name} is \${age} years old\`

// Safe HTML tag (prevents XSS)
safeHtml\`Input: \${userInput}\`

// Currency formatter
currency\`Price: \${price}, Tax: \${tax}\`

// Internationalization
i18n\`\${'Hello'}, \${'world'}!\`
            </pre>
        `;
    </script>
</body>
</html>
```

**Why**: Tagged templates provide powerful metaprogramming capabilities for DSLs, sanitization, and domain-specific formatting.

        // PROBLEM: Pass functions as arguments for async operations and customization

## js_array_unshift

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>JavaScript Array unshift()</title>
</head>
<body>
    <h2>Array unshift() Method Demo</h2>
    <div id="output"></div>
    
    <script>
        // PROBLEM: Add one or more elements to the beginning of an array
        // TIME: O(n) where n = array length SPACE: O(m) where m = elements added
        
        /* Why: Efficient prepending for queues, logs, and maintaining reverse chronological order */
        
        // Basic usage
        let fruits = ['banana', 'orange'];
        const newLength = fruits.unshift('apple');
        
        console.log(fruits);        // ['apple', 'banana', 'orange']
        console.log(newLength);      // 3
        
        // Add multiple elements
        let numbers = [4, 5];
        numbers.unshift(1, 2, 3);
        
        console.log(numbers);        // [1, 2, 3, 4, 5]
        
        // Use case: Task queue (FIFO)
        class TaskQueue {
            constructor() {
                this.tasks = [];
            }
            
            addUrgent(task) {
                this.tasks.unshift(task);
            }
            
            addNormal(task) {
                this.tasks.push(task);
            }
            
            processNext() {
                return this.tasks.shift();
            }
            
            get length() {
                return this.tasks.length;
            }
        }
        
        const queue = new TaskQueue();
        queue.addNormal('Task 1');
        queue.addNormal('Task 2');
        queue.addUrgent('URGENT Task');
        
        console.log('Queue:', queue.tasks);
        // ['URGENT Task', 'Task 1', 'Task 2']
        
        // Use case: Message log (newest first)
        class MessageLog {
            constructor() {
                this.messages = [];
            }
            
            addMessage(msg) {
                const timestamp = new Date().toLocaleTimeString();
                this.messages.unshift({ text: msg, time: timestamp });
            }
            
            getLastMessages(count = 5) {
                return this.messages.slice(0, count);
            }
        }
        
        const chat = new MessageLog();
        chat.addMessage('Hello!');
        setTimeout(() => chat.addMessage('How are you?'), 100);
        setTimeout(() => chat.addMessage('Good!'), 200);
        
        // Performance comparison
        const largeArray = new Array(100000);
        
        console.time('unshift');
        largeArray.unshift(1);
        console.timeEnd('unshift');
        
        console.time('push (for reference)');
        largeArray.push(1);
        console.timeEnd('push (for reference)');
        
        // Display results
        setTimeout(() => {
            document.getElementById('output').innerHTML = `
                <h3>Examples:</h3>
                <p><strong>Basic:</strong> ${JSON.stringify(fruits)}</p>
                <p><strong>Multiple:</strong> ${JSON.stringify(numbers)}</p>
                <p><strong>Queue:</strong> ${JSON.stringify(queue.tasks)}</p>
                
                <h3>Message Log:</h3>
                <ul>
                    ${chat.getLastMessages().map(m => 
                        `<li><strong>${m.time}</strong>: ${m.text}</li>`
                    ).join('')}
                </ul>
                
                <h3>Important Notes:</h3>
                <ul>
                    <li>unshift() modifies the original array</li>
                    <li>Returns the new length of the array</li>
                    <li>Time complexity: O(n) - shifts all elements</li>
                    <li>For better performance with large arrays, consider using push() + reverse()</li>
                </ul>
                
                <h3>Alternative Pattern (Better Performance):</h3>
                <pre>
// Instead of many unshifts
arr.unshift(item1);
arr.unshift(item2);

// Consider:
arr.push(item1, item2);
arr.reverse();
                </pre>
            `;
        }, 300);
    </script>
</body>
</html>
```

**Why**: `unshift()` provides intuitive array prepending for priority queues, logs, and reverse-chronological displays, though with O(n) cost.
