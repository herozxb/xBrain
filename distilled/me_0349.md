```python
# TIME: O(log n) for insert, delete, search SPACE: O(n)
class RedBlackTree:
    """Red-Black Tree implementation with automatic rebalancing"""
    
    class Node:
        def __init__(self, data, color='RED'):
            self.data = data        # Node value
            self.color = color      # 'RED' or 'BLACK'
            self.left = None        # Left child
            self.right = None       # Right child
            self.parent = None      # Parent node
    
    def __init__(self):
        self.NIL = self.Node(None, 'BLACK')  # Sentinel node
        self.root = self.NIL
    
    def insert(self, data):
        """Insert new node and maintain red-black properties"""
        node = self.Node(data)
        node.left = self.NIL
        node.right = self.NIL
        
        # Standard BST insert
        parent = None
        current = self.root
        
        while current != self.NIL:
            parent = current
            if node.data < current.data:
                current = current.left
            else:
                current = current.right
        
        node.parent = parent
        
        if parent is None:
            self.root = node
        elif node.data < parent.data:
            parent.left = node
        else:
            parent.right = node
        
        # Fix red-black violations
        self._fix_insert(node)
    
    def _fix_insert(self, node):
        """Rebalance tree after insertion"""
        while node.parent and node.parent.color == 'RED':
            if node.parent == node.parent.parent.left:
                uncle = node.parent.parent.right
                if uncle.color == 'RED':
                    # Case 1: Recolor
                    node.parent.color = 'BLACK'
                    uncle.color = 'BLACK'
                    node.parent.parent.color = 'RED'
                    node = node.parent.parent
                else:
                    if node == node.parent.right:
                        # Case 2: Left rotation
                        node = node.parent
                        self._left_rotate(node)
                    # Case 3: Right rotation
                    node.parent.color = 'BLACK'
                    node.parent.parent.color = 'RED'
                    self._right_rotate(node.parent.parent)
            else:
                # Mirror cases for right side
                uncle = node.parent.parent.left
                if uncle.color == 'RED':
                    node.parent.color = 'BLACK'
                    uncle.color = 'BLACK'
                    node.parent.parent.color = 'RED'
                    node = node.parent.parent
                else:
                    if node == node.parent.left:
                        node = node.parent
                        self._right_rotate(node)
                    node.parent.color = 'BLACK'
                    node.parent.parent.color = 'RED'
                    self._left_rotate(node.parent.parent)
        
        self.root.color = 'BLACK'
    
    def _left_rotate(self, x):
        """Left rotation around node x"""
        y = x.right
        x.right = y.left
        if y.left != self.NIL:
            y.left.parent = x
        y.parent = x.parent
        if x.parent is None:
            self.root = y
        elif x == x.parent.left:
            x.parent.left = y
        else:
            x.parent.right = y
        y.left = x
        x.parent = y
    
    def _right_rotate(self, x):
        """Right rotation around node x"""
        y = x.left
        x.left = y.right
        if y.right != self.NIL:
            y.right.parent = x
        y.parent = x.parent
        if x.parent is None:
            self.root = y
        elif x == x.parent.right:
            x.parent.right = y
        else:
            x.parent.left = y
        y.right = x
        x.parent = y
```

```python
# PROBLEM: Implement a disk-optimized self-balancing tree for databases
# TIME: O(log n) SPACE: O(n)
class BTreeNode:
    """B-Tree node with configurable minimum degree"""
    def __init__(self, t, leaf=False):
        self.t = t              # Minimum degree
        self.keys = []          # Array of keys
        self.children = []      # Array of child pointers
        self.leaf = leaf        # True if leaf node
        self.n = 0              # Current number of keys

class BTree:
    """B-Tree optimized for disk storage with O(log n) operations"""
    
    def __init__(self, t):
        self.t = t              # Minimum degree
        self.root = BTreeNode(t, leaf=True)
    
    def search(self, node, key):
        """Search for key in subtree rooted at node"""
        i = 0
        # Find first key >= search key
        while i < node.n and key > node.keys[i]:
            i += 1
        
        # Found key
        if i < node.n and key == node.keys[i]:
            return (node, i)
        
        # Key not found and this is leaf
        if node.leaf:
            return None
        
        # Recurse to appropriate child
        return self.search(node.children[i], key)
    
    def insert(self, key):
        """Insert key into B-Tree"""
        root = self.root
        
        # If root is full, split it
        if root.n == 2 * self.t - 1:
            new_root = BTreeNode(self.t)
            new_root.children.append(self.root)
            self.root = new_root
            self._split_child(new_root, 0)
            self._insert_non_full(new_root, key)
        else:
            self._insert_non_full(root, key)
    
    def _split_child(self, parent, i):
        """Split child at index i"""
        t = self.t
        full_child = parent.children[i]
        new_node = BTreeNode(t, leaf=full_child.leaf)
        
        # Move median key to parent
        parent.keys.insert(i, full_child.keys[t - 1])
        
        # Split keys
        new_node.keys = full_child.keys[t:(2*t - 1)]
        full_child.keys = full_child.keys[0:(t - 1)]
        
        # Split children if not leaf
        if not full_child.leaf:
            new_node.children = full_child.children[t:(2*t)]
            full_child.children = full_child.children[0:t]
        
        # Update counts
        new_node.n = t - 1
        full_child.n = t - 1
        parent.children.insert(i + 1, new_node)
        parent.n += 1
    
    def _insert_non_full(self, node, key):
        """Insert key into non-full node"""
        i = node.n - 1
        
        if node.leaf:
            # Insert key in correct position
            node.keys.append(0)
            while i >= 0 and key < node.keys[i]:
                node.keys[i + 1] = node.keys[i]
                i -= 1
            node.keys[i + 1] = key
            node.n += 1
        else:
            # Find child to recurse into
            while i >= 0 and key < node.keys[i]:
                i -= 1
            i += 1
            
            # Split child if full
            if node.children[i].n == 2 * self.t - 1:
                self._split_child(node, i)
                if key > node.keys[i]:
                    i += 1
            
            self._insert_non_full(node.children[i], key)
```

```python
from abc import ABC, abstractmethod

class Prototype(ABC):
    """Abstract prototype with clone method"""
    
    @abstractmethod
    def clone(self):
        """Create a copy of this object"""
        pass
    
    @abstractmethod
    def __str__(self):
        pass

class Shape(Prototype):
    """Concrete prototype - geometric shape"""
    
    def __init__(self, shape_type, color, x=0, y=0):
        self.shape_type = shape_type
        self.color = color
        self.x = x              # Position
        self.y = y
        self.dimensions = {}    # Shape-specific dimensions
    
    def clone(self):
        """Deep clone of shape object"""
        # Use deepcopy to clone nested objects
        return copy.deepcopy(self)
    
    def __str__(self):
        return f"{self.shape_type}(color={self.color}, pos=({self.x},{self.y}), dims={self.dimensions})"

class PrototypeRegistry:
    """Registry of pre-configured prototypes"""
    
    def __init__(self):
        self._prototypes = {}
    
    def register(self, name, prototype):
        """Register a prototype with given name"""
        self._prototypes[name] = prototype
    
    def clone(self, name):
        """Clone a registered prototype"""
        prototype = self._prototypes.get(name)
        if not prototype:
            raise ValueError(f"Prototype '{name}' not found")
        return prototype.clone()
```

```python
# TIME: O(n) for operations SPACE: O(n) for tree structure
from abc import ABC, abstractmethod

class FileSystemComponent(ABC):
    """Abstract base for files and directories"""
    
    def __init__(self, name):
        self.name = name
    
    @abstractmethod
    def display(self, indent=0):
        """Display component with indentation"""
        pass
    
    @abstractmethod
    def get_size(self):
        """Get size of component"""
        pass

class File(FileSystemComponent):
    """Leaf component - individual file"""
    
    def __init__(self, name, size):
        super().__init__(name)
        self.size = size
    
    def display(self, indent=0):
        print("  " * indent + f"📄 {self.name} ({self.size} KB)")
    
    def get_size(self):
        return self.size

class Directory(FileSystemComponent):
    """Composite component - contains other components"""
    
    def __init__(self, name):
        super().__init__(name)
        self.children = []
    
    def add(self, component):
        """Add child component"""
        self.children.append(component)
        return self  # Enable chaining
    
    def remove(self, component):
        """Remove child component"""
        self.children.remove(component)
    
    def display(self, indent=0):
        print("  " * indent + f"📁 {self.name}/")
        for child in self.children:
            child.display(indent + 1)
    
    def get_size(self):
        """Recursively calculate total size"""
        return sum(child.get_size() for child in self.children)
```

```python
        pass

    def play_mp4(self, filename):
        return f"Playing MP4 file: {filename}"

class OldAudioPlayer:
    """Another legacy player"""
    
    def play_wav(self, filename):
        return f"Playing WAV audio: {filename}"

# Adapter classes
class MediaAdapter(MediaPlayer):
    """Adapter to make AdvancedMediaPlayer compatible"""
    
    def __init__(self):
        self.advanced_player = AdvancedMediaPlayer()
    
    def play(self, filename):
        """Adapt to expected interface"""
        if filename.endswith('.vlc'):
            return self.advanced_player.play_vlc(filename)
        elif filename.endswith('.mp4'):
            return self.advanced_player.play_mp4(filename)
        else:
            return f"Unsupported format: {filename}"

class AudioAdapter(MediaPlayer):
    """Adapter for old audio player"""
    
    def __init__(self):
        self.old_player = OldAudioPlayer()
    
    def play(self, filename):
        if filename.endswith('.wav'):
            return self.old_player.play_wav(filename)
        return f"Unsupported audio format: {filename}"

# Client code
class AudioPlayer(MediaPlayer):
    """Unified player using adapters"""
    
    def __init__(self):
        self.media_adapter = MediaAdapter()
        self.audio_adapter = AudioAdapter()
    
    def play(self, filename):
        """Play any supported format"""
        if filename.endswith('.mp3'):
            return f"Playing MP3: {filename}"
        elif filename.endswith(('.vlc', '.mp4')):
            return self.media_adapter.play(filename)
        elif filename.endswith('.wav'):
            return self.audio_adapter.play(filename)
        else:
            return f"Invalid media: {filename}"

```

```python
# PROBLEM: Allow services to dynamically find each other in distributed systems
# TIME: O(1) lookup with hash map SPACE: O(n) for service registry
import time
import threading
from collections import defaultdict
from typing import Dict, List, Optional
import random

@dataclass
class ServiceInstance:
    """Service instance with metadata"""
    service_id: str
    host: str
    port: int
    health_url: str
    last_heartbeat: float
    metadata: dict

class ServiceDiscovery:
    """Distributed service discovery with health checking"""
    
    def __init__(self, heartbeat_timeout=30):
        self.services: Dict[str, List[ServiceInstance]] = defaultdict(list)
        self.heartbeat_timeout = heartbeat_timeout
        self.lock = threading.Lock()
    
    def register(self, service_name: str, instance: ServiceInstance):
        """Register a service instance"""
        with self.lock:
            self.services[service_name].append(instance)
            print(f"✅ Registered {service_name} at {instance.host}:{instance.port}")
    
    def deregister(self, service_name: str, service_id: str):
        """Remove a service instance"""
        with self.lock:
            instances = self.services[service_name]
            self.services[service_name] = [
                i for i in instances if i.service_id != service_id
            ]
            print(f"❌ Deregistered {service_name}/{service_id}")
    
    def discover(self, service_name: str) -> Optional[ServiceInstance]:
        """Discover a healthy service instance (round-robin)"""
        with self.lock:
            instances = self.services.get(service_name, [])
            healthy = [i for i in instances if self._is_healthy(i)]
            
            if not healthy:
                return None
            
            # Simple round-robin selection
            return random.choice(healthy)
    
    def discover_all(self, service_name: str) -> List[ServiceInstance]:
        """Discover all healthy instances of a service"""
        with self.lock:
            instances = self.services.get(service_name, [])
            return [i for i in instances if self._is_healthy(i)]
    
    def send_heartbeat(self, service_name: str, service_id: str):
        """Update heartbeat for service instance"""
        with self.lock:
            instances = self.services.get(service_name, [])
            for instance in instances:
                if instance.service_id == service_id:
                    instance.last_heartbeat = time.time()
                    break
    
    def _is_healthy(self, instance: ServiceInstance) -> bool:
        """Check if instance is healthy based on heartbeat"""
        return (time.time() - instance.last_heartbeat) < self.heartbeat_timeout
    
    def health_check_loop(self):
        """Background thread to remove unhealthy instances"""
        while True:
            time.sleep(10)
            with self.lock:
                for service_name in list(self.services.keys()):
                    before = len(self.services[service_name])
                    self.services[service_name] = [
                        i for i in self.services[service_name] 
                        if self._is_healthy(i)
                    ]
                    after = len(self.services[service_name])
                    if before != after:
                        print(f"🏥 {service_name}: removed {before - after} unhealthy instances")

class LoadBalancer:
    """Client-side load balancer using service discovery"""

        self.discovery = discovery
    
    def get_instance(self, service_name: str) -> Optional[ServiceInstance]:
```

```python
# PROBLEM: Replicate data across multiple database nodes for availability
# TIME: O(n) for replication SPACE: O(n) for replica storage
import threading
import queue
from typing import List, Dict
from enum import Enum

class OperationType(Enum):
    INSERT = "INSERT"
    UPDATE = "UPDATE"
    DELETE = "DELETE"

@dataclass
class WriteOperation:
    """Database write operation"""
    op_type: OperationType
    table: str
    key: str
    value: dict
    timestamp: float
    replica_id: str

class DatabaseNode:
    """Single database node (master or replica)"""
    
    def __init__(self, node_id: str, is_master=False):
        self.node_id = node_id
        self.is_master = is_master
        self.data: Dict[str, dict] = {}
        self.write_log: List[WriteOperation] = []
        self.replication_queue = queue.Queue()
    
    def write(self, op_type: OperationType, table: str, key: str, value: dict):
        """Execute write operation"""
        op = WriteOperation(op_type, table, key, value, time.time(), self.node_id)
        
        if op_type == OperationType.INSERT or op_type == OperationType.UPDATE:
            self.data[key] = value
        elif op_type == OperationType.DELETE:
            self.data.pop(key, None)
        
        self.write_log.append(op)
        return op
    
    def apply_replication(self, op: WriteOperation):
        """Apply replicated operation from master"""
        if op.op_type == OperationType.INSERT or op.op_type == OperationType.UPDATE:
            self.data[op.key] = op.value
        elif op.op_type == OperationType.DELETE:
            self.data.pop(op.key, None)
        
        self.write_log.append(op)
        print(f"📥 Replica {self.node_id} applied {op.op_type.value} on {op.key}")

class MasterSlaveReplication:
    """Master-slave replication with async propagation"""
    
    def __init__(self, master: DatabaseNode):
        self.master = master
        self.replicas: List[DatabaseNode] = []
        self.replication_thread = None
    
    def add_replica(self, replica: DatabaseNode):
        """Add replica node"""
        self.replicas.append(replica)
        print(f"✅ Added replica: {replica.node_id}")
    
    def write(self, table: str, key: str, value: dict, op_type=OperationType.INSERT):
        """Write to master and replicate to slaves"""
        # Write to master
        op = self.master.write(op_type, table, key, value)
        print(f"✍️ Master wrote {op_type.value} on {key}")
        
        # Async replication to slaves
        for replica in self.replicas:
            replica.replication_queue.put(op)
    
    def read_from_master(self, key: str) -> dict:
        """Strong consistency read from master"""
        return self.master.data.get(key)
    
    def read_from_replica(self, replica_id: str, key: str) -> dict:
        """Eventually consistent read from replica"""
        replica = next((r for r in self.replicas if r.node_id == replica_id), None)
        if replica:
            return replica.data.get(key)
        return None
    
    def start_replication(self):
        """Start background replication threads"""
        def replicate_to_slave(replica: DatabaseNode):
            while True:
                try:
                    op = replica.replication_queue.get(timeout=1)
                    replica.apply_replication(op)
                except queue.Empty:
                    continue
        
        for replica in self.replicas:
            thread = threading.Thread(
                target=replicate_to_slave, 
                args=(replica,), 
                daemon=True
            )
            thread.start()

# Example usage
master = DatabaseNode("master-1", is_master=True)
replica1 = DatabaseNode("replica-1")
replica2 = DatabaseNode("replica-2")

replication = MasterSlaveReplication(master)
replication.add_replica(replica1)
replication.add_replica(replica2)
replication.start_replication()

# Writes go to master, replicated to slaves
replication.write("users", "user:1", {"name": "Alice", "email": "alice@example.com"})
replication.write("users", "user:2", {"name": "Bob", "email": "bob@example.com"})

time.sleep(0.5)  # Wait for replication

# Read from master (strong consistency)
print(replication.read_from_master("user:1"))

# Read from replica (eventual consistency)
print(replication.read_from_replica("replica-1", "user:2"))
```

**Why**: Database replication provides high availability, fault tolerance, and read scalability. Master-slave pattern is used in MySQL, PostgreSQL streaming replication, and MongoDB replica sets.

## 9. Query Optimization

```python
# PROBLEM: Optimize database query execution with indexing and caching
# TIME: O(1) with index vs O(n) without SPACE: O(n) for indexes
from typing import Dict, List, Set, Tuple
from collections import defaultdict
import time

class QueryPlan:
    """Execution plan for a query"""
    
    def __init__(self, query_type: str, table: str, conditions: dict):
        self.query_type = query_type
        self.table = table
        self.conditions = conditions
        self.index_used = None
        self.estimated_cost = 0
        self.execution_steps = []

class QueryOptimizer:
    """Query optimizer with cost-based optimization"""
    
    def __init__(self):
        self.statistics: Dict[str, dict] = {}
        self.indexes: Dict[str, Dict[str, Set]] = defaultdict(lambda: defaultdict(set))
    
    def create_index(self, table: str, column: str, data: List[Tuple]):
        """Create index on table column"""
        for row_id, value in data:
            self.indexes[table][column].add((value, row_id))
        print(f"📊 Created index on {table}.{column}")
    
    def analyze_query(self, query: str, table_stats: dict) -> QueryPlan:
        """Analyze query and generate execution plan"""
        # Parse query (simplified)
        plan = QueryPlan("SELECT", "users", {"age": 25})
        
        # Check if index can be used
        if "age" in self.indexes.get("users", {}):
            plan.index_used = "users.age"
            plan.estimated_cost = 1  # O(1) with index
        else:
            plan.estimated_cost = table_stats.get("row_count", 1000)  # O(n) scan
        
        # Generate execution steps
        if plan.index_used:
            plan.execution_steps = [
                "INDEX_SCAN users.age",
                "FETCH rows",
                "RETURN results"
            ]
        else:
            plan.execution_steps = [
                "SEQUENTIAL_SCAN users",
                "FILTER age = 25",
                "RETURN results"
            ]
        
        return plan
    
    def execute_with_index(self, table: str, column: str, value) -> List[int]:
        """Execute query using index"""
        index = self.indexes[table][column]
        return [row_id for (val, row_id) in index if val == value]

class QueryCache:
    """LRU cache for query results"""
    
    def __init__(self, max_size=100):
        self.cache: Dict[str, dict] = {}
        self.max_size = max_size
        self.lru_list = []
    
    def get(self, query_hash: str) -> dict:
        """Get cached result"""
        if query_hash in self.cache:
            # Move to front (most recently used)
            self.lru_list.remove(query_hash)
            self.lru_list.insert(0, query_hash)
            print(f"🎯 Cache HIT: {query_hash}")
            return self.cache[query_hash]
        print(f"❌ Cache MISS: {query_hash}")
        return None
    
    def put(self, query_hash: str, result: dict):
        """Cache query result"""
        if len(self.cache) >= self.max_size:
            # Evict least recently used
            lru_key = self.lru_list.pop()
            del self.cache[lru_key]
        
        self.cache[query_hash] = result
        self.lru_list.insert(0, query_hash)
        print(f"💾 Cached result: {query_hash}")

class QueryExecutor:
    """Execute queries with optimization and caching"""
    
    def __init__(self):
        self.optimizer = QueryOptimizer()
        self.cache = QueryCache()
    
    def execute(self, query: str, data: List[dict]) -> List[dict]:
        """Execute query with optimization"""
        # Check cache first
        query_hash = hash(query)
        cached = self.cache.get(query_hash)
        if cached:
            return cached["results"]
        
        # Analyze and optimize
        plan = self.optimizer.analyze_query(query, {"row_count": len(data)})
        print(f"📋 Execution plan: {' -> '.join(plan.execution_steps)}")
        print(f"💰 Estimated cost: {plan.estimated_cost}")
        
        # Execute based on plan
        start_time = time.time()
        
        if plan.index_used:
            # Use index (simulated)
            results = [row for row in data if row.get("age") == 25]
        else:
            # Full scan
            results = [row for row in data if row.get("age") == 25]
        
        execution_time = time.time() - start_time
        print(f"⏱️ Execution time: {execution_time:.4f}s")
        
        # Cache results
        self.cache.put(query_hash, {"results": results, "time": execution_time})
        
        return results

# Example usage
executor = QueryExecutor()

# Sample data
users = [
    {"id": i, "name": f"User{i}", "age": 20 + (i % 30)} 
    for i in range(1000)
]

# Create index
executor.optimizer.create_index("users", "age", [(u["id"], u["age"]) for u in users])

# Execute query (first time - cache miss)
query = "SELECT * FROM users WHERE age = 25"
results1 = executor.execute(query, users)

# Execute same query (cache hit)
results2 = executor.execute(query, users)
```

**Why**: Query optimization reduces execution time through indexing, caching, and cost-based planning. Used in PostgreSQL, MySQL, and Oracle for query performance tuning.

## 10. NoSQL Patterns

```python
# PROBLEM: Design data patterns for NoSQL databases (document, key-value, column)
# TIME: O(1) for key-based access SPACE: O(n)
from typing import Dict, List, Any
from collections import defaultdict
import json

# Pattern 1: Document Store
class DocumentStore:
    """MongoDB-style document store"""
    
    def __init__(self):
        self.collections: Dict[str, Dict[str, dict]] = defaultdict(dict)
    
    def insert(self, collection: str, doc_id: str, document: dict):
        """Insert document into collection"""
        document["_id"] = doc_id
        self.collections[collection][doc_id] = document
        print(f"📄 Inserted {doc_id} into {collection}")
    
    def find(self, collection: str, query: dict) -> List[dict]:
        """Find documents matching query"""
        results = []
        for doc in self.collections[collection].values():
            if self._matches(doc, query):
                results.append(doc)
        return results
    
    def update(self, collection: str, doc_id: str, updates: dict):
        """Update document"""
        if doc_id in self.collections[collection]:
            self.collections[collection][doc_id].update(updates)
            print(f"✏️ Updated {doc_id}")
    
    def _matches(self, doc: dict, query: dict) -> bool:
        """Check if document matches query"""
        for key, value in query.items():
            if key not in doc or doc[key] != value:
                return False
        return True

# Pattern 2: Key-Value Store with TTL
class KeyValueStore:
    """Redis-style key-value store with expiration"""
    
    def __init__(self):
        self.store: Dict[str, Any] = {}
        self.ttl: Dict[str, float] = {}
    
    def set(self, key: str, value: Any, ttl_seconds: int = None):
        """Set key-value pair with optional TTL"""
        self.store[key] = value
        if ttl_seconds:
            self.ttl[key] = time.time() + ttl_seconds
        print(f"🔑 SET {key}")
    
    def get(self, key: str) -> Any:
        """Get value by key (returns None if expired)"""
        if key in self.ttl and time.time() > self.ttl[key]:
            del self.store[key]
            del self.ttl[key]
            print(f"⏰ Key expired: {key}")
            return None
        return self.store.get(key)
    
    def delete(self, key: str):
        """Delete key"""
        self.store.pop(key, None)
        self.ttl.pop(key, None)
        print(f"🗑️ Deleted {key}")

# Pattern 3: Column-Family Store
class ColumnFamilyStore:
    """Cassandra-style column family store"""
    
    def __init__(self):
        self.column_families: Dict[str, Dict[str, Dict[str, Any]]] = defaultdict(
            lambda: defaultdict(dict)
        )
    
    def insert(self, cf: str, row_key: str, columns: Dict[str, Any]):
        """Insert columns into row"""
        self.column_families[cf][row_key].update(columns)
        print(f"📊 Inserted into {cf}[{row_key}]: {list(columns.keys())}")
    
    def get_row(self, cf: str, row_key: str) -> Dict[str, Any]:
        """Get entire row"""
        return self.column_families[cf].get(row_key, {})
    
    def get_column(self, cf: str, row_key: str, column: str) -> Any:
        """Get specific column from row"""
        return self.column_families[cf].get(row_key, {}).get(column)
    
    def scan(self, cf: str) -> List[Dict[str, Any]]:
        """Scan all rows in column family"""
        return [
            {"row_key": k, "columns": v} 
            for k, v in self.column_families[cf].items()
        ]

# Pattern 4: Time-Series Data
class TimeSeriesStore:
    """Optimized for time-series data"""
    
    def __init__(self):
        self.series: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
    
    def write_point(self, metric: str, timestamp: float, value: Any, tags: dict = None):
        """Write data point"""
        self.series[metric].append({
            "timestamp": timestamp,
            "value": value,
            "tags": tags or {}
        })
        # Keep sorted by timestamp
        self.series[metric].sort(key=lambda x: x["timestamp"])
    
    def query_range(self, metric: str, start: float, end: float) -> List[dict]:
        """Query points in time range"""
        return [
            p for p in self.series[metric]
            if start <= p["timestamp"] <= end
        ]

# Example usage
print("=== Document Store ===")
doc_store = DocumentStore()
doc_store.insert("users", "user:1", {"name": "Alice", "age": 30, "city": "NYC"})
doc_store.insert("users", "user:2", {"name": "Bob", "age": 25, "city": "LA"})
print(doc_store.find("users", {"city": "NYC"}))

print("\n=== Key-Value Store ===")
kv_store = KeyValueStore()
kv_store.set("session:abc123", {"user_id": 1}, ttl_seconds=3600)
kv_store.set("config", {"theme": "dark"})
print(kv_store.get("session:abc123"))

print("\n=== Column Family Store ===")
cf_store = ColumnFamilyStore()
cf_store.insert("posts", "post:1", {"title": "Hello", "author": "Alice", "date": "2026-02-19"})
cf_store.insert("posts", "post:2", {"title": "World", "author": "Bob", "date": "2026-02-19"})
print(cf_store.get_row("posts", "post:1"))

print("\n=== Time-Series Store ===")
ts_store = TimeSeriesStore()
ts_store.write_point("cpu_usage", time.time() - 100, 75.5, {"host": "server1"})
ts_store.write_point("cpu_usage", time.time() - 50, 82.3, {"host": "server1"})
ts_store.write_point("cpu_usage", time.time(), 78.1, {"host": "server1"})
print(f"Recent points: {len(ts_store.query_range('cpu_usage', time.time() - 200, time.time()))}")
```

**Why**: NoSQL patterns optimize for specific use cases: documents for semi-structured data, key-value for caching, column families for wide-row analytics, and time-series for metrics. Used in MongoDB, Redis, Cassandra, and InfluxDB.
