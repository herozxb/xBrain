<!DOCTYPE html>

    // Basic destructuring
    const user = {

      email: 'alice@example.com',

    };
    
    // Extract specific properties
    const { name, age } = user;

    console.log(userName, userEmail); // 'Alice' 'alice@example.com'

    // Nested destructuring
    const person = {
      id: 1,
      info: {
        firstName: 'Bob',
        lastName: 'Smith',
        address: {
          street: '123 Main St',
          city: 'Boston'
        }
      }
    };
    
    const { info: { firstName, lastName, address: { city: personCity } } } = person;
    console.log(firstName, personCity); // 'Bob' 'Boston'
    
    // Destructuring in function parameters
    function displayUser({ name, age, email = 'N/A' }) {
      return `${name}, ${age} years old, Email: ${email}`;
    }
    
    console.log(displayUser(user)); // 'Alice, 30 years old, Email: alice@example.com'
    
    // Rest properties
    const { name: restName, ...rest } = user;
    console.log(rest); // { age: 30, email: '...', city: 'New York' }
    
    // Destructuring with computed property names
    const key = 'email';
    const { [key]: dynamicValue } = user;
    console.log(dynamicValue); // 'alice@example.com'
    
    // Display results
    output.innerHTML = `
      <h3>Examples:</h3>
      <ul>
        <li><strong>Basic:</strong> const { name, age } = user; → ${name}, ${age}</li>
        <li><strong>Renaming:</strong> const { name: userName } = user; → ${userName}</li>
        <li><strong>Defaults:</strong> const { country = 'USA' } = user; → ${country}</li>
        <li><strong>Nested:</strong> const { info: { firstName } } = person; → ${firstName}</li>
        <li><strong>Function params:</strong> displayUser({ name, age }) → ${displayUser(user)}</li>
        <li><strong>Rest:</strong> const { name, ...rest } = user; → ${JSON.stringify(rest)}</li>
      </ul>
    `;
  </script>

```

**Why**: Object destructuring provides a concise syntax to extract multiple properties into variables. It supports renaming, default values, nested structures, and rest properties, making code cleaner and more readable.

    // TIME: O(n) SPACE: O(1)
    
    const output = document.getElementById('output');
    let results = [];
    
    // Basic usage
    const fruits = ['apple', 'banana', 'orange', 'mango'];
    
    const hasApple = fruits.includes('apple');    // true
    const hasGrape = fruits.includes('grape');    // false

    // With fromIndex parameter
    const numbers = [1, 2, 3, 4, 5, 3];
    
    const hasThreeFrom0 = numbers.includes(3, 0);     // true (index 2)
    const hasThreeFrom3 = numbers.includes(3, 3);     // true (index 5)
    const hasThreeFrom6 = numbers.includes(3, 6);     // false (past end)
    
    results.push(`numbers.includes(3, 0): ${hasThreeFrom0}`);
    results.push(`numbers.includes(3, 3): ${hasThreeFrom3}`);
    results.push(`numbers.includes(3, 6): ${hasThreeFrom6}`);
    
    // Type sensitivity
    const mixed = [1, '1', true, 'true'];
    
    results.push(`mixed.includes(1): ${mixed.includes(1)}`);        // true
    results.push(`mixed.includes('1'): ${mixed.includes('1')}`);    // true
    results.push(`mixed.includes(true): ${mixed.includes(true)}`);  // true
    
    // NaN handling (unlike indexOf)
    const withNaN = [1, 2, NaN, 4];
    
    results.push(`withNaN.includes(NaN): ${withNaN.includes(NaN)}`);  // true
    results.push(`withNaN.indexOf(NaN): ${withNaN.indexOf(NaN)}`);    // -1
    
    // Sparse arrays
    const sparse = [1, , , 4];  // holes at index 1 and 2
    
    results.push(`sparse.includes(undefined): ${sparse.includes(undefined)}`); // false
    
    // Practical use: validation
    const allowedRoles = ['admin', 'editor', 'moderator'];
    
    function checkPermission(role) {
      return allowedRoles.includes(role);
    }
    
    results.push(`checkPermission('admin'): ${checkPermission('admin')}`);   // true
    results.push(`checkPermission('guest'): ${checkPermission('guest')}`);   // false
    
    // Display results
    output.innerHTML = `
      <div class="demo-box">
        <h3>Results:</h3>
        <ul>
          ${results.map(r => `<li><code>${r}</code></li>`).join('')}

      <div class="demo-box">
        <h3>Key Points:</h3>
        <ul>
          <li>Returns <span class="highlight">true</span> or <span class="highlight">false</span></li>
          <li>Uses <strong>strict equality</strong> (===)</li>
          <li>Second parameter: <code>fromIndex</code> (where to start searching)</li>
          <li><strong>Can find NaN</strong> (unlike indexOf)</li>
          <li>Time complexity: O(n)</li>
        </ul>
        
        <h3>vs indexOf:</h3>
        <ul>
          <li><code>includes()</code>: Returns boolean</li>
          <li><code>indexOf()</code>: Returns index or -1</li>
          <li><code>includes(NaN)</code> works; <code>indexOf(NaN)</code> returns -1</li>
        </ul>
      </div>
    `;

</html>
```

**Why**: `includes()` is the cleanest way to check array membership, returning a boolean. Unlike `indexOf()`, it properly handles `NaN` and communicates intent clearly: you want to know "if" something exists, not "where" it is.

  <script>
    // PROBLEM: Make an object immutable (cannot be changed)
    // TIME: O(1) SPACE: O(1)
    
    const output = document.getElementById('output');
    let logs = [];
    
    function log(msg, isError = false) {
      logs.push({ msg, isError });
    }
    
    // Create and freeze an object
    const config = {
      apiUrl: 'https://api.example.com',
      timeout: 5000,
      retries: 3
    };
    
    Object.freeze(config);
    
    log('Object frozen: config');
    
    // Attempt to modify (fails silently in non-strict mode)
    try {
      config.timeout = 10000;  // Won't work
      log(`After trying to change timeout: ${config.timeout}`, true);
    } catch (e) {
      log(`Error: ${e.message}`, true);
    }
    
    // Attempt to add property
    config.newProp = 'test';  // Won't work
    log(`After trying to add property: ${JSON.stringify(config)}`);
    
    // Attempt to delete property
    delete config.retries;    // Won't work
    log(`After trying to delete retries: ${config.retries}`);
    
    // Check if frozen
    log(`Object.isFrozen(config): ${Object.isFrozen(config)}`);
    
    // Shallow freeze limitation
    const user = {
      name: 'Alice',
      preferences: {
        theme: 'dark',
        language: 'en'
      }
    };
    
    Object.freeze(user);
    
    // Can still modify nested objects!
    user.preferences.theme = 'light';  // This works!
    log(`Nested object modified: ${user.preferences.theme}`, true);
    
    // Deep freeze function
    function deepFreeze(obj) {
      const propNames = Object.getOwnPropertyNames(obj);
      
      for (const name of propNames) {
        const value = obj[name];
        
        if (value && typeof value === 'object') {
          deepFreeze(value);
        }
      }
      
      return Object.freeze(obj);
    }
    
    // Apply deep freeze
    const deepUser = {
      name: 'Bob',
      settings: {
        theme: 'dark',
        notifications: true
      }
    };
    
    deepFreeze(deepUser);
    
    // Now nested object is also frozen
    try {
      deepUser.settings.theme = 'light';
      log(`Deep frozen nested modification: ${deepUser.settings.theme}`);
    } catch (e) {
      log(`Deep frozen error (strict mode): ${e.message}`, true);
    }
    
    log(`Object.isFrozen(deepUser.settings): ${Object.isFrozen(deepUser.settings)}`);
    
    // Display results
    output.innerHTML = `
      <div class="demo-box">
        <h3>Operations Log:</h3>
        <ul>
          ${logs.map(l => `
            <li class="${l.isError ? 'error-text' : 'success-text'}">
              ${l.msg}
            </li>
          `).join('')}
        </ul>
      </div>
      
      <div class="demo-box">
        <h3>What freeze() prevents:</h3>
        <ul>
          <li>❌ Adding new properties</li>
          <li>❌ Removing existing properties</li>
          <li>❌ Changing property values</li>
          <li>❌ Changing enumerability, configurability, or writability</li>
        </ul>
        
        <h3>Important Notes:</h3>
        <ul>
          <li><strong>Shallow freeze:</strong> Nested objects can still be modified</li>
          <li>Use <code>deepFreeze()</code> for complete immutability</li>
          <li>In strict mode, modifications throw TypeError</li>
          <li>Frozen objects can't be unfrozen</li>
        </ul>
        
        <h3>vs seal() and preventExtensions():</h3>
        <pre>
freeze():           No add, no delete, no modify
seal():             No add, no delete, can modify
preventExtensions(): No add, can delete, can modify
        </pre>
      </div>
    `;
  </script>
</body>
</html>

**Why**: `Object.freeze()` creates truly immutable objects, preventing accidental modifications to configuration, constants, or state. It's shallow by default, so use deep freeze for nested objects. Perfect for constants, configuration objects, and Redux state.

## 10. JavaScript String includes()

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>JavaScript String includes()</title>
  <style>
    .demo-box {
      padding: 15px;
      margin: 10px 0;
      background-color: #f5f5f5;
      border-radius: 4px;
    }
    
    .highlight {
      background-color: #fff3cd;
      padding: 2px 6px;
      border-radius: 3px;
    }
    
    .search-match {
      background-color: #c8e6c9;
      padding: 2px 4px;
    }
  </style>
</head>
<body>
  <h2>JavaScript String includes()</h2>
  
  <div>
    <input type="text" id="searchInput" placeholder="Search in text...">
    <button id="searchBtn">Search</button>
  </div>
  
  <div id="output"></div>
  
  <script>
    // PROBLEM: Check if a string contains another string
    // TIME: O(n*m) SPACE: O(1)
    // Where n = string length, m = search string length
    
    const output = document.getElementById('output');
    const searchInput = document.getElementById('searchInput');
    const searchBtn = document.getElementById('searchBtn');
    
    let results = [];
    
    // Basic usage
    const sentence = 'The quick brown fox jumps over the lazy dog';
    
    const hasFox = sentence.includes('fox');      // true
    const hasCat = sentence.includes('cat');      // false
    
    results.push(`sentence.includes('fox'): ${hasFox}`);
    results.push(`sentence.includes('cat'): ${hasCat}`);
    
    // Case sensitivity
    const hasFoxLower = sentence.includes('fox');      // true
    const hasFoxUpper = sentence.includes('FOX');      // false
    const hasFoxCaseInsensitive = sentence.toLowerCase().includes('FOX'.toLowerCase()); // true
    
    results.push(`Case sensitive - 'FOX': ${hasFoxUpper}`);
    results.push(`Case insensitive: ${hasFoxCaseInsensitive}`);
    
    // With position parameter
    const text = 'Hello world, hello universe';
    
    const hasHelloFrom0 = text.includes('Hello', 0);   // true
    const hasHelloFrom10 = text.includes('Hello', 10); // false (past first occurrence)
    const hasHelloFrom15 = text.includes('hello', 15); // true (finds second 'hello')
    
    results.push(`text.includes('Hello', 0): ${hasHelloFrom0}`);
    results.push(`text.includes('Hello', 10): ${hasHelloFrom10}`);
    results.push(`text.includes('hello', 15): ${hasHelloFrom15}`);
    
    // Empty string always returns true
    results.push(`'hello'.includes(''): ${'hello'.includes('')}`);  // true
    results.push(`''.includes(''): ${''.includes('')}`);            // true
    
    // Practical use: search/filter
    const products = [
      { name: 'iPhone 15', category: 'phone' },
      { name: 'MacBook Pro', category: 'laptop' },
      { name: 'iPad Air', category: 'tablet' },
      { name: 'Samsung Galaxy', category: 'phone' }
    ];
    
    function searchProducts(query) {
      return products.filter(p => 
        p.name.toLowerCase().includes(query.toLowerCase())
      );
    }
    
    results.push(`searchProducts('phone'): ${searchProducts('phone').length} results`);
    results.push(`searchProducts('a'): ${searchProducts('a').length} results`);
    
    // Interactive search
    function performSearch() {
      const query = searchInput.value;
      const sampleText = 'JavaScript is a versatile programming language used for web development';
      
      const found = sampleText.toLowerCase().includes(query.toLowerCase());
      
      output.innerHTML = `
        <div class="demo-box">
          <p><strong>Text:</strong> "${sampleText}"</p>
          <p><strong>Searching for:</strong> "${query}"</p>
          <p><strong>Found:</strong> 
            <span class="${found ? 'search-match' : 'highlight'}">${found}</span>
          </p>
        </div>
        
        <div class="demo-box">
          <h3>Results Log:</h3>
          <ul>
            ${results.map(r => `<li><code>${r}</code></li>`).join('')}
          </ul>
        </div>
        
        <div class="demo-box">
          <h3>Key Points:</h3>
          <ul>
            <li>Returns <span class="highlight">true</span> or <span class="highlight">false</span></li>
            <li><strong>Case-sensitive</strong> by default</li>
            <li>Second parameter: <code>position</code> (where to start)</li>
            <li>Empty string always returns <code>true</code></li>
            <li>Time complexity: O(n*m)</li>
          </ul>
          
          <h3>vs indexOf() and search():</h3>
          <ul>
            <li><code>includes()</code>: Returns boolean</li>
            <li><code>indexOf()</code>: Returns position or -1</li>
            <li><code>search()</code>: Works with regex</li>
          </ul>
        </div>
      `;
    }
    
    searchBtn.addEventListener('click', performSearch);
    searchInput.addEventListener('keyup', (e) => {
      if (e.key === 'Enter') performSearch();
    });
    
    // Initial display
    performSearch();
  </script>
</body>
</html>
```

**Why**: `String.includes()` is the simplest way to check if a string contains a substring. It returns a boolean, making it perfect for conditional checks and search functionality. It's case-sensitive and more readable than `indexOf() !== -1`.
