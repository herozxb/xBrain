## 1. Suffix Automaton

```python
# PROBLEM: Build a suffix automaton to support substring queries and counting distinct substrings
# TIME: O(n) for construction, O(m) for substring query SPACE: O(n * alphabet)
class SuffixAutomaton:
    """Suffix Automaton for efficient string processing."""
    
    class State:
        __slots__ = ['next', 'link', 'len']
        def __init__(self):
            self.next = {}  # transitions
            self.link = -1  # suffix link
            self.len = 0    # length of longest string in this state
    
    def __init__(self, s: str = ""):
        self.st = [self.State()]  # initial state (root)
        self.last = 0  # state corresponding to whole string
        for ch in s:
            self.extend(ch)
    
    def extend(self, c: str) -> None:
        """Add character c to the automaton."""
        p = self.last
        curr = len(self.st)
        self.st.append(self.State())
        self.st[curr].len = self.st[p].len + 1
        
        # Follow suffix links and add transitions
        while p != -1 and c not in self.st[p].next:
            self.st[p].next[c] = curr
            p = self.st[p].link
        
        if p == -1:
            # Reached root, set suffix link to root
            self.st[curr].link = 0
        else:
            q = self.st[p].next[c]
            if self.st[p].len + 1 == self.st[q].len:
                # q is a valid extension
                self.st[curr].link = q
            else:
                # Need to clone state q
                clone = len(self.st)
                self.st.append(self.State())
                self.st[clone].len = self.st[p].len + 1
                self.st[clone].next = self.st[q].next.copy()
                self.st[clone].link = self.st[q].link
                
                while p != -1 and self.st[p].next.get(c) == q:
                    self.st[p].next[c] = clone
                    p = self.st[p].link
                
                self.st[q].link = clone
                self.st[curr].link = clone
        
        self.last = curr
    
    def contains(self, pattern: str) -> bool:
        """Check if pattern is a substring of the original string."""
        state = 0
        for c in pattern:
            if c not in self.st[state].next:
                return False
            state = self.st[state].next[c]
        return True
    
    def count_distinct_substrings(self) -> int:
        """Count number of distinct substrings."""
        result = 0
        for i in range(1, len(self.st)):
            result += self.st[i].len - self.st[self.st[i].link].len
        return result
```
**Why**: Suffix automaton enables O(n) construction for all substring-related queries, supporting substring check, counting distinct substrings, and finding longest common substring efficiently.

## 2. Palindromic Tree (Eertree)

```python
# PROBLEM: Find all distinct palindromic substrings and their counts
# TIME: O(n) for construction SPACE: O(n * alphabet)
class PalindromicTree:
    """Palindromic Tree (Eertree) for palindrome processing."""
    
    class Node:
        __slots__ = ['next', 'link', 'len', 'count', 'num']
        def __init__(self):
            self.next = {}   # transitions
            self.link = 0    # suffix link
            self.len = 0     # palindrome length
            self.count = 0   # occurrence count
            self.num = 0     # number of palindromes to root
    
    def __init__(self, s: str):
        self.s = s
        self.nodes = [self.Node(), self.Node()]  # root(-1), root(0)
        self.nodes[0].len = -1  # imaginary root for odd-length
        self.nodes[1].len = 0   # empty string root for even-length
        self.nodes[0].link = 0
        self.nodes[1].link = 0
        
        self.last = 1  # node for longest suffix palindrome
        self.s_ptr = 0  # current position
        
        for i, ch in enumerate(s):
            self.add_char(i, ch)
    
    def get_link(self, node: int, pos: int) -> int:
        """Find the longest palindromic suffix."""
        while True:
            cur_len = self.nodes[node].len
            if pos - 1 - cur_len >= 0 and self.s[pos - 1 - cur_len] == self.s[pos]:
                return node
            node = self.nodes[node].link
        return node
    
    def add_char(self, pos: int, ch: str) -> bool:
        """Add character at position pos."""
        cur = self.get_link(self.last, pos)
        
        if ch in self.nodes[cur].next:
            # Palindrome already exists
            self.last = self.nodes[cur].next[ch]
            self.nodes[self.last].count += 1
            return False
        
        # Create new node
        new_node = len(self.nodes)
        self.nodes.append(self.Node())
        self.nodes[new_node].len = self.nodes[cur].len + 2
        self.nodes[cur].next[ch] = new_node
        
        if self.nodes[new_node].len == 1:
            # Single character palindrome
            self.nodes[new_node].link = 1
        else:
            # Find suffix link
            link_node = self.get_link(self.nodes[cur].link, pos)
            self.nodes[new_node].link = self.nodes[link_node].next[ch]
        
        self.nodes[new_node].num = self.nodes[self.nodes[new_node].link].num + 1
        self.nodes[new_node].count = 1
        self.last = new_node
        return True
    
    def count_distinct_palindromes(self) -> int:
        """Return number of distinct palindromic substrings."""
        return len(self.nodes) - 2
    
    def count_all_palindromes(self) -> int:
        """Return total count of all palindromic substrings."""
        result = 0
        for i in range(len(self.nodes) - 1, 1, -1):
            self.nodes[self.nodes[i].link].count += self.nodes[i].count
            result += self.nodes[i].count
        return result
```
**Why**: Eertree stores all distinct palindromic substrings in O(n) space, supporting queries like counting palindromes and finding palindrome statistics in linear time.

## 3. Ternary Search Trie

```python
# PROBLEM: Implement a space-efficient trie for string storage and prefix queries
# TIME: O(L) for search/insert where L is key length SPACE: O(N * alphabet)
class TernarySearchTrie:
    """Ternary Search Trie with space-efficient node representation."""
    
    class Node:
        __slots__ = ['char', 'left', 'mid', 'right', 'val', 'is_end']
        def __init__(self, ch: str):
            self.char = ch
            self.left = None
            self.mid = None
            self.right = None
            self.val = None      # associated value
            self.is_end = False  # marks end of key
    
    def __init__(self):
        self.root = None
        self._size = 0
    
    def put(self, key: str, val) -> None:
        """Insert key with associated value."""
        if not key:
            return
        self.root = self._put(self.root, key, val, 0)
    
    def _put(self, node: 'TernarySearchTrie.Node', key: str, val, d: int):
        ch = key[d]
        if node is None:
            node = self.Node(ch)
        
        if ch < node.char:
            node.left = self._put(node.left, key, val, d)
        elif ch > node.char:
            node.right = self._put(node.right, key, val, d)
        elif d < len(key) - 1:
            node.mid = self._put(node.mid, key, val, d + 1)
        else:
            if not node.is_end:
                self._size += 1
            node.val = val
            node.is_end = True
        return node
    
    def get(self, key: str):
        """Get value associated with key, or None if not found."""
        node = self._get(self.root, key, 0)
        return node.val if node and node.is_end else None
    
    def _get(self, node: 'TernarySearchTrie.Node', key: str, d: int):
        if node is None:
            return None
        ch = key[d]
        
        if ch < node.char:
            return self._get(node.left, key, d)
        elif ch > node.char:
            return self._get(node.right, key, d)
        elif d < len(key) - 1:
            return self._get(node.mid, key, d + 1)
        else:
            return node
    
    def contains(self, key: str) -> bool:
        """Check if key exists in trie."""
        return self.get(key) is not None or (self._get(self.root, key, 0) is not None 
                                             and self._get(self.root, key, 0).is_end)
    
    def starts_with(self, prefix: str) -> list:
        """Get all keys with given prefix."""
        node = self._get(self.root, prefix, 0)
        results = []
        if node is None:
            return results
        if node.is_end:
            results.append((prefix, node.val))
        self._collect(node.mid, prefix, results)
        return results
    
    def _collect(self, node: 'TernarySearchTrie.Node', prefix: str, results: list):
        if node is None:
            return
        self._collect(node.left, prefix, results)
        new_prefix = prefix + node.char
        if node.is_end:
            results.append((new_prefix, node.val))
        self._collect(node.mid, new_prefix, results)
        self._collect(node.right, prefix, results)
    
    def __len__(self) -> int:
        return self._size
```
**Why**: Ternary search tries balance space efficiency with fast lookups, using only 3 pointers per node instead of full alphabet arrays, ideal for large character sets.

        new_id = self._get_id(new_node)

            self.head = self.tail = new_node
            new_node.both = 0
        else:
            head_id = self._get_id(self.head)
            new_node.both = head_id
            self.head.both = self.head.both ^ new_id
            self.head = new_node
    
    def traverse_forward(self) -> list:
        """Return all elements in forward order."""
        result = []
        prev_id = 0
        curr = self.head
        
        while curr is not None:
            result.append(curr.data)
            next_id = curr.both ^ prev_id
            prev_id = self._get_id(curr)
            curr = self._from_id(next_id)

            result.append(curr.data)
            prev_id = curr.both ^ next_id
            next_id = self._get_id(curr)
            curr = self._from_id(prev_id)
        
        return result

    """Simplified XOR list using prev/next XOR with direct node storage."""
    
    class Node:
        __slots__ = ['data', 'ptr_xor']
        def __init__(self, data):
            self.data = data
            self.ptr_xor = None  # XOR of prev and next references
    
    def __init__(self):
        self.head = None
        self.tail = None
        self._nodes = []  # Keep references alive
    
    def _xor(self, a, b):
        return a if b is None else b if a is None else (a, b)
    
    def append(self, data) -> None:
        new_node = self.Node(data)
        self._nodes.append(new_node)
        
        if self.head is None:
            self.head = self.tail = new_node
            new_node.ptr_xor = None
        else:
            new_node.ptr_xor = self.tail
            self.tail.ptr_xor = (self.tail.ptr_xor, new_node) if self.tail.ptr_xor else new_node
            self.tail = new_node
    
    def to_list_forward(self) -> list:
        result = []
        prev = None
        curr = self.head
        
        while curr:
            result.append(curr.data)
            ptr = curr.ptr_xor
            if ptr is None:
                break
            if isinstance(ptr, tuple):
                next_node = ptr[1] if ptr[0] is prev else ptr[0]
            else:
                next_node = ptr if prev is None else None
            prev, curr = curr, next_node
        
        return result

## 5. Hash Table Open Addressing

```python
# PROBLEM: Implement a hash table using open addressing with linear probing
# TIME: O(1) average for insert/delete/search SPACE: O(n)
class HashTableOpenAddressing:
    """Hash table with open addressing and linear probing."""
    
    DELETED = object()  # Marker for deleted slots
    
    def __init__(self, capacity: int = 16, load_factor: float = 0.75):
        self.capacity = capacity
        self.load_factor = load_factor
        self.size = 0
        self.keys = [None] * capacity
        self.values = [None] * capacity
    
    def _hash(self, key) -> int:
        """Compute hash value for key."""
        return hash(key) % self.capacity
    
    def _probe(self, key, for_insert: bool = False) -> int:
        """Linear probing to find slot for key."""
        idx = self._hash(key)
        first_deleted = -1
        
        for _ in range(self.capacity):
            if self.keys[idx] is None:
                return first_deleted if first_deleted != -1 and for_insert else idx
            if self.keys[idx] is self.DELETED:
                if first_deleted == -1:
                    first_deleted = idx
            elif self.keys[idx] == key:
                return idx
            idx = (idx + 1) % self.capacity
        
        return first_deleted if first_deleted != -1 else -1
    
    def _resize(self) -> None:
        """Double the capacity and rehash all entries."""
        old_keys = self.keys
        old_values = self.values
        self.capacity *= 2
        self.keys = [None] * self.capacity
        self.values = [None] * self.capacity
        self.size = 0
        
        for k, v in zip(old_keys, old_values):
            if k is not None and k is not self.DELETED:
                self.put(k, v)
    
    def put(self, key, value) -> None:
        """Insert or update key-value pair."""
        if self.size >= self.capacity * self.load_factor:
            self._resize()
        
        idx = self._probe(key, for_insert=True)
        if idx == -1:
            self._resize()
            idx = self._probe(key, for_insert=True)
        
        if self.keys[idx] is None or self.keys[idx] is self.DELETED:
            self.size += 1
        self.keys[idx] = key
        self.values[idx] = value
    
    def get(self, key, default=None):
        """Get value for key, or default if not found."""
        idx = self._probe(key)
        if idx == -1 or self.keys[idx] is None or self.keys[idx] is self.DELETED:
            return default
        return self.values[idx]
    
    def remove(self, key) -> bool:
        """Remove key from table. Return True if removed."""
        idx = self._probe(key)
        if idx == -1 or self.keys[idx] is None or self.keys[idx] is self.DELETED:
            return False
        
        self.keys[idx] = self.DELETED
        self.values[idx] = None
        self.size -= 1
        return True
    
    def contains(self, key) -> bool:
        """Check if key exists in table."""
        idx = self._probe(key)
        return idx != -1 and self.keys[idx] is not None and self.keys[idx] is not self.DELETED
    
    def __len__(self) -> int:
        return self.size
    
    def __setitem__(self, key, value):
        self.put(key, value)
    
    def __getitem__(self, key):
        result = self.get(key, self)
        if result is self:
            raise KeyError(key)
        return result
    
    def __contains__(self, key) -> bool:
        return self.contains(key)
```
**Why**: Open addressing with linear probing provides cache-friendly hash table operations without pointer overhead, ideal for small to medium datasets with good hash distribution.

## 6. Hash Table Chaining

```python
# PROBLEM: Implement a hash table using separate chaining for collision resolution
# TIME: O(1) average for insert/delete/search SPACE: O(n)
class HashTableChaining:
    """Hash table with separate chaining for collision handling."""
    
    class Node:
        __slots__ = ['key', 'value', 'next']
        def __init__(self, key, value):
            self.key = key
            self.value = value
            self.next = None
    
    def __init__(self, capacity: int = 16, load_factor: float = 0.75):
        self.capacity = capacity
        self.load_factor = load_factor
        self.size = 0
        self.buckets = [None] * capacity
    
    def _hash(self, key) -> int:
        """Compute bucket index for key."""
        return hash(key) % self.capacity
    
    def _resize(self) -> None:
        """Double capacity and rehash all entries."""
        old_buckets = self.buckets
        self.capacity *= 2
        self.buckets = [None] * self.capacity
        self.size = 0
        
        for head in old_buckets:
            curr = head
            while curr:
                self.put(curr.key, curr.value)
                curr = curr.next
    
    def put(self, key, value) -> None:
        """Insert or update key-value pair."""
        if self.size >= self.capacity * self.load_factor:
            self._resize()
        
        idx = self._hash(key)
        curr = self.buckets[idx]
        
        # Check if key already exists
        while curr:
            if curr.key == key:
                curr.value = value
                return
            curr = curr.next
        
        # Insert new node at head
        new_node = self.Node(key, value)
        new_node.next = self.buckets[idx]
        self.buckets[idx] = new_node
        self.size += 1
    
    def get(self, key, default=None):
        """Get value for key, or default if not found."""
        idx = self._hash(key)
        curr = self.buckets[idx]
        
        while curr:
            if curr.key == key:
                return curr.value
            curr = curr.next
        return default
    
    def remove(self, key) -> bool:
        """Remove key from table. Return True if removed."""
        idx = self._hash(key)
        curr = self.buckets[idx]
        prev = None
        
        while curr:
            if curr.key == key:
                if prev:
                    prev.next = curr.next
                else:
                    self.buckets[idx] = curr.next
                self.size -= 1
                return True
            prev = curr
            curr = curr.next
        return False
    
    def contains(self, key) -> bool:
        """Check if key exists in table."""
        return self.get(key, self) is not self
    
    def keys(self) -> list:
        """Return all keys in table."""
        result = []
        for head in self.buckets:
            curr = head
            while curr:
                result.append(curr.key)
                curr = curr.next
        return result
    
    def values(self) -> list:
        """Return all values in table."""
        result = []
        for head in self.buckets:
            curr = head
            while curr:
                result.append(curr.value)
                curr = curr.next
        return result
    
    def items(self) -> list:
        """Return all key-value pairs."""
        result = []
        for head in self.buckets:
            curr = head
            while curr:
                result.append((curr.key, curr.value))
                curr = curr.next
        return result
    
    def __len__(self) -> int:
        return self.size
    
    def __setitem__(self, key, value):
        self.put(key, value)
    
    def __getitem__(self, key):
        result = self.get(key, self)
        if result is self:
            raise KeyError(key)
        return result
    
    def __contains__(self, key) -> bool:
        return self.contains(key)
```
**Why**: Separate chaining handles collisions gracefully and works well with poor hash functions, maintaining O(1) operations with low collision probability.

## 7. Y-Fast Trie

```python
# PROBLEM: Implement a space-efficient integer set with predecessor/successor queries
# TIME: O(log log U) for operations, U = universe size SPACE: O(n)
import bisect
from collections import defaultdict

class XFastTrieSimple:
    """Simplified X-Fast Trie for demonstration."""
    
    def __init__(self, max_bits: int = 32):
        self.max_bits = max_bits
        self.hash_tables = [defaultdict(set) for _ in range(max_bits + 1)]
        self.min_val = None
        self.max_val = None
    
    def insert(self, x: int) -> None:
        """Insert integer x."""
        for level in range(self.max_bits + 1):
            prefix = x >> (self.max_bits - level)
            self.hash_tables[level][prefix].add(x)
        
        if self.min_val is None or x < self.min_val:
            self.min_val = x
        if self.max_val is None or x > self.max_val:
            self.max_val = x
    
    def contains(self, x: int) -> bool:
        """Check if x is in the set."""
        return x in self.hash_tables[self.max_bits][x]


class YFastTrie:
    """Y-Fast Trie using balanced BSTs (simulated with sorted lists)."""
    
    class Bucket:
        """A bucket storing a subset of elements."""
        def __init__(self):
            self.elements = []  # Sorted list
            self.rep = None      # Representative (max element)
        
        def insert(self, x: int) -> None:
            """Insert x maintaining sorted order."""
            idx = bisect.bisect_left(self.elements, x)
            if idx == len(self.elements) or self.elements[idx] != x:
                self.elements.insert(idx, x)
            self.rep = self.elements[-1]
        
        def remove(self, x: int) -> bool:
            """Remove x from bucket."""
            idx = bisect.bisect_left(self.elements, x)
            if idx < len(self.elements) and self.elements[idx] == x:
                self.elements.pop(idx)
                self.rep = self.elements[-1] if self.elements else None
                return True
            return False
        
        def successor(self, x: int):
            """Find smallest element >= x."""
            idx = bisect.bisect_left(self.elements, x)
            return self.elements[idx] if idx < len(self.elements) else None
        
        def predecessor(self, x: int):
            """Find largest element < x."""
            idx = bisect.bisect_left(self.elements, x)
            return self.elements[idx - 1] if idx > 0 else None
    
    def __init__(self, max_bits: int = 32, bucket_size: int = 16):
        self.max_bits = max_bits
        self.bucket_size = bucket_size
        self.xfast = XFastTrieSimple(max_bits)
        self.buckets = {}  # rep -> Bucket
        self.size = 0
    
    def _find_bucket(self, x: int) -> 'YFastTrie.Bucket':
        """Find bucket that might contain x."""
        # Find successor representative
        succ_rep = self._find_successor_rep(x)
        if succ_rep is not None and succ_rep in self.buckets:
            return self.buckets[succ_rep]
        # Find predecessor bucket
        pred_rep = self._find_predecessor_rep(x)
        if pred_rep is not None and pred_rep in self.buckets:
            return self.buckets[pred_rep]
        return None
    
    def _find_successor_rep(self, x: int):
        """Find smallest representative >= x."""
        for level in range(self.xfast.max_bits + 1):
            prefix = x >> (self.xfast.max_bits - level)
            for r in sorted(self.buckets.keys()):
                if r >= x:
                    return r
        return None
    
    def _find_predecessor_rep(self, x: int):
        """Find largest representative < x."""
        preds = [r for r in self.buckets.keys() if r < x]
        return max(preds) if preds else None
    
    def insert(self, x: int) -> None:
        """Insert integer x."""
        bucket = self._find_bucket(x)
        
        if bucket is None:
            # Create new bucket
            bucket = self.Bucket()
            bucket.insert(x)
            self.buckets[bucket.rep] = bucket
            self.xfast.insert(bucket.rep)
        else:
            bucket.insert(x)
            # Update representative if needed
            old_rep = None
            for r in list(self.buckets.keys()):
                if self.buckets[r] is bucket:
                    old_rep = r
                    break
            
            if old_rep != bucket.rep:
                if old_rep is not None:
                    del self.buckets[old_rep]
                self.buckets[bucket.rep] = bucket
                self.xfast.insert(bucket.rep)
            
            # Split if too large
            if len(bucket.elements) > 2 * self.bucket_size:
                self._split_bucket(bucket)
        
        self.size += 1
    
    def _split_bucket(self, bucket: 'YFastTrie.Bucket') -> None:
        """Split bucket into two."""
        mid = len(bucket.elements) // 2
        new_bucket = self.Bucket()
        
        for x in bucket.elements[mid:]:
            new_bucket.insert(x)
        
        bucket.elements = bucket.elements[:mid]
        bucket.rep = bucket.elements[-1]
        
        self.buckets[bucket.rep] = bucket
        self.buckets[new_bucket.rep] = new_bucket
        self.xfast.insert(new_bucket.rep)
    
    def contains(self, x: int) -> bool:
        """Check if x is in the set."""
        bucket = self._find_bucket(x)
        if bucket is None:
            return False
        idx = bisect.bisect_left(bucket.elements, x)
        return idx < len(bucket.elements) and bucket.elements[idx] == x
    
    def successor(self, x: int):
        """Find smallest element >= x."""
        bucket = self._find_bucket(x)
        if bucket:
            succ = bucket.successor(x)
            if succ is not None:
                return succ
        # Check next bucket
        next_rep = self._find_successor_rep(x + 1)
        if next_rep and next_rep in self.buckets:
            return self.buckets[next_rep].elements[0]
        return None
    
    def predecessor(self, x: int):
        """Find largest element < x."""
        bucket = self._find_bucket(x)
        if bucket:
            pred = bucket.predecessor(x)
            if pred is not None:
                return pred
        # Check previous bucket
        prev_rep = self._find_predecessor_rep(x)
        if prev_rep and prev_rep in self.buckets:
            return self.buckets[prev_rep].elements[-1]
        return None

from collections import defaultdict

class XFastTrie:
    """X-Fast Trie for integer predecessor/successor queries."""
    
    class Node:
        __slots__ = ['left', 'right', 'is_leaf', 'value']
        def __init__(self):
            self.left = None   # descendant for bit 0
            self.right = None  # descendant for bit 1
            self.is_leaf = False
            self.value = None
    
    def __init__(self, max_bits: int = 32):
        self.max_bits = max_bits
        self.root = self.Node()
        self.hash_tables = [defaultdict(list) for _ in range(max_bits + 1)]
        self.min_val = None
        self.max_val = None
        self.size = 0
    
    def _get_bit(self, x: int, level: int) -> int:
        """Get bit at given level (0-indexed from MSB)."""
        return (x >> (self.max_bits - 1 - level)) & 1
    
    def insert(self, x: int) -> None:
        """Insert integer x."""
        if self.contains(x):
            return
        
        node = self.root
        path = []
        
        for level in range(self.max_bits):
            bit = self._get_bit(x, level)
            path.append((node, bit))
            
            if bit == 0:
                if node.left is None:
                    node.left = self.Node()
                node = node.left
            else:
                if node.right is None:
                    node.right = self.Node()
                node = node.right
            
            # Update hash table
            prefix = x >> (self.max_bits - level - 1)
            self.hash_tables[level + 1][prefix].append(node)
        
        node.is_leaf = True
        node.value = x
        self.hash_tables[self.max_bits][x].append(node)
        self.size += 1
        
        if self.min_val is None or x < self.min_val:
            self.min_val = x
        if self.max_val is None or x > self.max_val:
            self.max_val = x
    
    def contains(self, x: int) -> bool:
        """Check if x is in the set."""
        if x not in self.hash_tables[self.max_bits]:
            return False
        nodes = self.hash_tables[self.max_bits][x]
        return any(n.is_leaf and n.value == x for n in nodes)
    
    def _find_longest_prefix(self, x: int) -> tuple:
        """Binary search for longest matching prefix."""
        low, high = 0, self.max_bits
        best_level = 0
        best_node = self.root
        
        while low <= high:
            mid = (low + high) // 2
            prefix = x >> (self.max_bits - mid) if mid > 0 else 0
            
            if mid in self.hash_tables and prefix in self.hash_tables[mid]:
                best_level = mid
                best_node = self.hash_tables[mid][prefix][0]
                low = mid + 1
            else:
                high = mid - 1
        
        return best_level, best_node
    
    def successor(self, x: int) -> int:
        """Find smallest element > x."""
        if self.size == 0:
            return None
        if self.max_val is not None and x >= self.max_val:
            return None
        
        level, node = self._find_longest_prefix(x)
        
        # Traverse to find successor
        for l in range(level, self.max_bits):
            bit = self._get_bit(x, l)
            
            if bit == 0:
                if node.left:
                    # There's a smaller value, go right for successor
                    if node.right:
                        return self._find_min(node.right)
                    node = node.left
                else:
                    if node.right:
                        return self._find_min(node.right)
                    return None
            else:
                if node.right:
                    node = node.right
                else:
                    return None
        
        if node.is_leaf and node.value > x:
            return node.value
        
        return None
    
    def predecessor(self, x: int) -> int:
        """Find largest element < x."""
        if self.size == 0:
            return None
        if self.min_val is not None and x <= self.min_val:
            return None
        
        level, node = self._find_longest_prefix(x)
        
        for l in range(level, self.max_bits):
            bit = self._get_bit(x, l)
            
            if bit == 1:
                if node.right:
                    if node.left:
                        return self._find_max(node.left)
                    node = node.right
                else:
                    if node.left:
                        return self._find_max(node.left)
                    return None
            else:
                if node.left:
                    node = node.left
                else:
                    return None
        
        if node.is_leaf and node.value < x:
            return node.value
        
        return None
    
    def _find_min(self, node: 'XFastTrie.Node') -> int:
        """Find minimum value in subtree."""
        while not node.is_leaf:
            node = node.left if node.left else node.right
        return node.value

        """Find maximum value in subtree."""
        while not node.is_leaf:
            node = node.right if node.right else node.left
        return node.value

        return self.size

# PROBLEM: Implement a self-adjusting heap with O(log n) amortized operations
# TIME: O(log n) amortized for all operations SPACE: O(n)
class SkewHeap:
    """Skew Heap - self-adjusting leftist heap variant."""
    
    class Node:
        __slots__ = ['value', 'left', 'right']
        def __init__(self, value):
            self.value = value
            self.left = None
            self.right = None
    
    def __init__(self):
        self.root = None
        self._size = 0
    
    def _merge(self, h1: 'SkewHeap.Node', h2: 'SkewHeap.Node') -> 'SkewHeap.Node':
        """Merge two skew heaps."""
        if h1 is None:
            return h2
        if h2 is None:
            return h1
        
        # Ensure h1 has smaller root
        if h1.value > h2.value:
            h1, h2 = h2, h1
        
        # Swap children and recursively merge
        h1.left, h1.right = h1.right, h1.left
        h1.right = self._merge(h1.right, h2)
        
        return h1
    
    def push(self, value) -> None:
        """Insert value into heap."""
        new_node = self.Node(value)
        self.root = self._merge(self.root, new_node)
        self._size += 1
    
    def pop(self):
        """Remove and return minimum value."""
        if self.root is None:
            raise IndexError("pop from empty heap")
        
        result = self.root.value
        self.root = self._merge(self.root.left, self.root.right)
        self._size -= 1
        return result
    
    def peek(self):
        """Return minimum value without removing."""
        if self.root is None:
            raise IndexError("peek at empty heap")
        return self.root.value
    
    def merge(self, other: 'SkewHeap') -> None:
        """Merge another heap into this one."""
        self.root = self._merge(self.root, other.root)
        self._size += other._size
        other.root = None
        other._size = 0
    
    def is_empty(self) -> bool:
        return self.root is None
    
    def __len__(self) -> int:
        return self._size
    
    def __bool__(self) -> bool:
        return self.root is not None


class SkewHeapMax(SkewHeap):
    """Max-heap variant of Skew Heap."""
    
    class Node:
        __slots__ = ['value', 'left', 'right']
        def __init__(self, value):
            self.value = value
            self.left = None
            self.right = None
    
    def _merge(self, h1, h2):
        if h1 is None:
            return h2
        if h2 is None:
            return h1
        
        # Ensure h1 has larger root (for max-heap)
        if h1.value < h2.value:
            h1, h2 = h2, h1
        
        # Swap and merge
        h1.left, h1.right = h1.right, h1.left
        h1.right = self._merge(h1.right, h2)
        
        return h1


# Example usage for sorting
def skew_heapsort(arr: list) -> list:
    """Sort array using skew heap."""
    heap = SkewHeap()
    for x in arr:
        heap.push(x)
    
    result = []
    while heap:
        result.append(heap.pop())
    return result

# PROBLEM: Implement a cache-friendly linked list with block-based storage
# TIME: O(sqrt(n)) for random access, O(1) for sequential SPACE: O(n)
import math

class UnrolledLinkedList:
    """Unrolled linked list storing multiple elements per node."""
    
    class Block:
        __slots__ = ['elements', 'next', 'prev']
        def __init__(self, capacity: int):
            self.elements = []
            self.next = None
            self.prev = None
            self._capacity = capacity
        
        def is_full(self) -> bool:
            return len(self.elements) >= self._capacity
        
        def is_underfull(self, min_size: int) -> bool:
            return len(self.elements) < min_size
    
    def __init__(self, block_size: int = 32):
        self.block_size = block_size
        self.head = None
        self.tail = None
        self._size = 0
        self._num_blocks = 0
    
    def _create_block(self) -> 'Block':
        """Create a new block."""
        block = self.Block(self.block_size)
        self._num_blocks += 1
        return block
    
    def append(self, value) -> None:
        """Add element to end."""
        if self.tail is None:
            self.head = self.tail = self._create_block()
        
        if self.tail.is_full():
            new_block = self._create_block()
            self.tail.next = new_block
            new_block.prev = self.tail
            self.tail = new_block
        
        self.tail.elements.append(value)
        self._size += 1
    
    def prepend(self, value) -> None:
        """Add element to beginning."""
        if self.head is None:
            self.head = self.tail = self._create_block()
            self.head.elements.append(value)
        else:
            # Insert at head block
            if self.head.is_full():
                new_block = self._create_block()
                new_block.next = self.head
                self.head.prev = new_block
                self.head = new_block
            self.head.elements.insert(0, value)
        
        self._size += 1
    
    def _find_block(self, index: int) -> tuple:
        """Find block and local index for global index."""
        if index < 0:
            index += self._size
        if index < 0 or index >= self._size:
            raise IndexError("list index out of range")
        
        block = self.head
        while block:
            if index < len(block.elements):
                return block, index
            index -= len(block.elements)
            block = block.next
        
        raise IndexError("list index out of range")
    
    def __getitem__(self, index: int):
        """Get element at index."""
        block, local_idx = self._find_block(index)
        return block.elements[local_idx]
    
    def __setitem__(self, index: int, value) -> None:
        """Set element at index."""
        block, local_idx = self._find_block(index)
        block.elements[local_idx] = value
    
    def insert(self, index: int, value) -> None:
        """Insert element at index."""
        if index == 0:
            self.prepend(value)
            return
        if index >= self._size:
            self.append(value)
            return
        
        block, local_idx = self._find_block(index)
        
        if block.is_full():
            # Split block
            new_block = self._create_block()
            mid = len(block.elements) // 2
            
            new_block.elements = block.elements[mid:]
            block.elements = block.elements[:mid]
            
            new_block.next = block.next
            new_block.prev = block
            if block.next:
                block.next.prev = new_block
            block.next = new_block
            
            if block is self.tail:
                self.tail = new_block
            
            # Re-find correct block
            if local_idx >= len(block.elements):
                block = new_block
                local_idx -= len(block.elements)
        
        block.elements.insert(local_idx, value)
        self._size += 1
    
    def remove(self, index: int):
        """Remove and return element at index."""
        block, local_idx = self._find_block(index)
        value = block.elements.pop(local_idx)
        self._size -= 1
        
        # Merge with neighbor if underfull
        min_size = self.block_size // 2
        if block.is_underfull(min_size) and block.next:
            # Merge with next block
            block.elements.extend(block.next.elements)
            removed = block.next
            block.next = removed.next
            if removed.next:
                removed.next.prev = block
            if removed is self.tail:
                self.tail = block
            self._num_blocks -= 1
        
        return value
    
    def __delitem__(self, index: int) -> None:
        self.remove(index)
    
    def to_list(self) -> list:
        """Convert to Python list."""
        result = []
        block = self.head
        while block:
            result.extend(block.elements)
            block = block.next
        return result
    
    def __iter__(self):
        """Iterate over all elements."""
        block = self.head
        while block:
            yield from block.elements
            block = block.next
    
    def __len__(self) -> int:
        return self._size
    
    def __repr__(self) -> str:
        return f"UnrolledLinkedList({self.to_list()})"
```
**Why**: Unrolled linked lists optimize cache performance by storing multiple elements per node, reducing pointer overhead while maintaining efficient insertion/deletion.
