## WeakMap

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>WeakMap for Object Metadata</title>
</head>
<body>
    <h2>WeakMap - Private Data Storage</h2>
    <div id="output"></div>

    <script>
        // PROBLEM: Store metadata on objects without preventing garbage collection
        // TIME: O(1) SPACE: O(n)

        // WeakMap only accepts objects as keys
        const privateData = new WeakMap();

        // Create objects that will be used as keys
        const user1 = { name: 'Alice' };
        const user2 = { name: 'Bob' };

        // Store private metadata (not accessible from the object itself)
        privateData.set(user1, { loginCount: 5, lastLogin: '2026-02-27' });
        privateData.set(user2, { loginCount: 12, lastLogin: '2026-02-28' });

        // Retrieve private data
        const output = document.getElementById('output');
        
        [user1, user2].forEach(user => {
            const data = privateData.get(user);
            const item = document.createElement('p');
            item.textContent = `${user.name}: ${data.loginCount} logins, last: ${data.lastLogin}`;
            output.appendChild(item);
        });

        // Check existence and size
        console.log('Has user1:', privateData.has(user1)); // true
        
        // When user objects are garbage collected, WeakMap entries are automatically removed
        // This prevents memory leaks - ideal for caching object-related data

        // **Why**: WeakMap allows garbage collection of keys, preventing memory leaks
        // Perfect for caching, private data, and metadata on DOM elements
    </script>
</body>
</html>
```

## WeakSet

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>WeakSet for Object Tracking</title>
</head>
<body>
    <h2>WeakSet - Track Objects Without Memory Leaks</h2>
    <div id="output"></div>

    <script>
        // PROBLEM: Track processed objects without preventing garbage collection
        // TIME: O(1) SPACE: O(n)

        // WeakSet only holds object references weakly
        const processedItems = new WeakSet();

        // Simulate processing function
        function processItem(item) {
            if (processedItems.has(item)) {
                return 'Already processed';
            }
            processedItems.add(item);
            return 'Processing complete';
        }

        // Create objects to track
        const item1 = { id: 1, data: 'First' };
        const item2 = { id: 2, data: 'Second' };

        const output = document.getElementById('output');

        // First processing
        const result1 = document.createElement('p');
        result1.textContent = `Item 1 (first): ${processItem(item1)}`;
        output.appendChild(result1);

        // Duplicate check
        const result2 = document.createElement('p');
        result2.textContent = `Item 1 (second): ${processItem(item1)}`;
        output.appendChild(result2);

        // New item
        const result3 = document.createElement('p');
        result3.textContent = `Item 2: ${processItem(item2)}`;
        output.appendChild(result3);

        // Note: WeakSet has no size property, no iteration, no clear()
        // When item objects are no longer referenced elsewhere, they're GC'd
        // and automatically removed from the WeakSet

        // **Why**: WeakSet tracks objects without memory leaks
        // Ideal for marking objects as processed/visited/active
    </script>
</body>
</html>
```

## Proxy ownKeys

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Proxy ownKeys Trap</title>
</head>
<body>
    <h2>Proxy ownKeys - Control Property Enumeration</h2>
    <div id="output"></div>

    <script>
        // PROBLEM: Filter which properties appear in Object.keys() and for...in
        // TIME: O(n) SPACE: O(n)

        const output = document.getElementById('output');

        // Target object with mixed property types
        const user = {
            name: 'Alice',
            email: 'alice@example.com',
            _internalId: 12345,  // Private by convention
            password: 'secret',   // Sensitive data
            age: 30
        };

        // Proxy that hides properties starting with '_' or named 'password'
        const protectedUser = new Proxy(user, {
            ownKeys(target) {
                // Get all own keys from target
                const allKeys = Reflect.ownKeys(target);
                
                // Filter out private and sensitive properties
                return allKeys.filter(key => {
                    const keyStr = String(key);
                    return !keyStr.startsWith('_') && keyStr !== 'password';
                });
            },
            
            // Also need to handle getOwnPropertyDescriptor for Object.keys() to work
            getOwnPropertyDescriptor(target, prop) {
                if (prop.startsWith('_') || prop === 'password') {
                    return undefined; // Property doesn't exist for enumeration
                }
                return Reflect.getOwnPropertyDescriptor(target, prop);
            }
        });

        // Demonstrate the filtering
        const visibleKeys = document.createElement('p');
        visibleKeys.textContent = `Visible keys: ${Object.keys(protectedUser).join(', ')}`;
        output.appendChild(visibleKeys);

        const allValues = document.createElement('p');
        allValues.textContent = `Values: ${JSON.stringify(protectedUser, null, 2)}`;
        output.appendChild(allValues);

        // Direct access still works (ownKeys only affects enumeration)
        const directAccess = document.createElement('p');
        directAccess.textContent = `Direct password access: ${protectedUser.password}`;
        output.appendChild(directAccess);

        // **Why**: ownKeys trap controls what Object.keys(), for...in, and spread see
        // Perfect for hiding internal properties, implementing access control
    </script>
</body>
</html>
```

## Generator value and done

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Generator Value and Done Properties</title>
</head>
<body>
    <h2>Generator value & done - Understanding Iterator Results</h2>
    <div id="output"></div>

    <script>
        // PROBLEM: Understand the structure of generator iterator results
        // TIME: O(1) per next() call SPACE: O(1)

        const output = document.getElementById('output');

        // Generator function produces an iterator
        function* numberGenerator() {
            yield 1;
            yield 2;
            yield 3;
            // Generator completes after this
        }

        // Create iterator from generator
        const iterator = numberGenerator();

        // Each next() returns { value, done }
        let result = iterator.next();
        
        const step1 = document.createElement('p');
        step1.textContent = `First call: value=${result.value}, done=${result.done}`;
        output.appendChild(step1);
        // { value: 1, done: false }

        result = iterator.next();
        const step2 = document.createElement('p');
        step2.textContent = `Second call: value=${result.value}, done=${result.done}`;
        output.appendChild(step2);
        // { value: 2, done: false }

        result = iterator.next();
        const step3 = document.createElement('p');
        step3.textContent = `Third call: value=${result.value}, done=${result.done}`;
        output.appendChild(step3);
        // { value: 3, done: false }

        result = iterator.next();
        const step4 = document.createElement('p');
        step4.textContent = `Fourth call: value=${result.value}, done=${result.done}`;
        output.appendChild(step4);
        // { value: undefined, done: true }

        // Manual iteration pattern
        const manualIterator = numberGenerator();
        let manualResult;
        const manualOutput = document.createElement('div');
        manualOutput.innerHTML = '<strong>Manual iteration:</strong><br>';
        
        while (!(manualResult = manualIterator.next()).done) {
            const p = document.createElement('p');
            p.textContent = `  Got value: ${manualResult.value}`;
            manualOutput.appendChild(p);
        }
        
        const doneMsg = document.createElement('p');
        doneMsg.textContent = '  Iterator exhausted (done=true)';
        manualOutput.appendChild(doneMsg);
        output.appendChild(manualOutput);

        // **Why**: Understanding {value, done} is essential for manual iteration
        // Enables custom iteration logic beyond for...of loops
    </script>
</body>
</html>
```

## Closure Factory Pattern

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Closure Factory Pattern</title>
</head>
<body>
    <h2>Closure Factory - Create Configured Functions</h2>
    <div id="output"></div>

    <script>
        // PROBLEM: Create specialized functions from a general template
        // TIME: O(1) for creation, O(n) for execution SPACE: O(1)

        const output = document.getElementById('output');

        // Factory function that creates multiplier functions
        function createMultiplier(factor) {
            // 'factor' is captured in the closure
            return function(number) {
                return number * factor;
            };
        }

        // Create specialized functions using the factory
        const double = createMultiplier(2);
        const triple = createMultiplier(3);
        const quadruple = createMultiplier(4);

        // Each function remembers its 'factor' through closure
        const results = [
            `Double 5: ${double(5)}`,      // 10
            `Triple 5: ${triple(5)}`,      // 15
            `Quadruple 5: ${quadruple(5)}` // 20
        ];

        results.forEach(text => {
            const p = document.createElement('p');
            p.textContent = text;
            output.appendChild(p);
        });

        // More practical example: API URL factory
        function createApiFetcher(baseUrl) {
            return function(endpoint) {
                return `${baseUrl}${endpoint}`;
            };
        }

        const githubApi = createApiFetcher('https://api.github.com');
        const localApi = createApiFetcher('http://localhost:3000/api/');

        const apiExamples = document.createElement('div');
        apiExamples.innerHTML = `
            <h3>API Factory Example:</h3>
            <p>GitHub user: ${githubApi('/users/octocat')}</p>
            <p>Local users: ${localApi('users')}</p>
        `;
        output.appendChild(apiExamples);

        // Validator factory
        function createValidator(minLength, maxLength) {
            return function(text) {
                return text.length >= minLength && text.length <= maxLength;
            };
        }

        const usernameValidator = createValidator(3, 20);
        
        const validationExample = document.createElement('div');
        validationExample.innerHTML = `
            <h3>Validator Factory:</h3>
            <p>'ab' valid: ${usernameValidator('ab')}</p>
            <p>'alice' valid: ${usernameValidator('alice')}</p>
            <p>'this is way too long for username' valid: ${usernameValidator('this is way too long for username')}</p>
        `;
        output.appendChild(validationExample);

        // **Why**: Factory pattern with closures creates configurable, reusable functions
        // Eliminates code duplication while maintaining flexibility
    </script>
</body>
</html>
```
