```html
    </div>

  <script>
    class TabInterface {
      constructor(container) {
        this.container = container;
        this.tabs = container.querySelectorAll('[role="tab"]');
        this.panels = container.querySelectorAll('[role="tabpanel"]');
        
        this.init();
      }
      
      init() {
        // Add click handlers
        this.tabs.forEach(tab => {
          tab.addEventListener('click', () => this.selectTab(tab));
          
          // Keyboard navigation
          tab.addEventListener('keydown', (e) => this.handleKeydown(e));
        });
      }
      
      selectTab(selectedTab) {
        // Deselect all tabs
        this.tabs.forEach(tab => {
          tab.setAttribute('aria-selected', 'false');
          tab.setAttribute('tabindex', '-1');
        });
        
        // Hide all panels
        this.panels.forEach(panel => {
          panel.classList.add('hidden');
        });
        
        // Select clicked tab
        selectedTab.setAttribute('aria-selected', 'true');
        selectedTab.setAttribute('tabindex', '0');
        selectedTab.focus();
        
        // Show corresponding panel
        const panelId = selectedTab.getAttribute('aria-controls');
        document.getElementById(panelId).classList.remove('hidden');
      }
      
      handleKeydown(e) {
        const tabsArray = Array.from(this.tabs);
        const currentIndex = tabsArray.indexOf(e.target);
        let newIndex;
        
        switch(e.key) {
          case 'ArrowRight':
          case 'ArrowDown':
            newIndex = (currentIndex + 1) % tabsArray.length;
            this.selectTab(tabsArray[newIndex]);
            e.preventDefault();
            break;
            
          case 'ArrowLeft':
          case 'ArrowUp':
            newIndex = (currentIndex - 1 + tabsArray.length) % tabsArray.length;
            this.selectTab(tabsArray[newIndex]);
            e.preventDefault();
            break;
            
          case 'Home':
            this.selectTab(tabsArray[0]);
            e.preventDefault();
            break;
            
          case 'End':
            this.selectTab(tabsArray[tabsArray.length - 1]);
            e.preventDefault();
            break;
        }
      }
    }
    
    // Initialize tab interface
    new TabInterface(document.querySelector('.tab-container'));
  </script>
</body>
</html>
```

**Why**: Provides full keyboard navigation and screen reader support for tab interfaces, essential for accessibility compliance (WCAG).

```html
        'og:title': data.title,
        'og:description': data.description,
        'og:image': data.image,
        'og:url': data.url
```

```html
  <script>
    // Flex-grow calculation formula:
    // item size = base size + (available space × item's grow factor / total grow factors)
    
    function changeGrow(itemIndex, growValue) {
      const items = document.querySelectorAll('#interactiveContainer .item');
      items[itemIndex - 1].style.flexGrow = growValue;
      items[itemIndex - 1].textContent = `Item ${itemIndex} (${growValue}x)`;
    }
    
    function resetGrow() {
      const items = document.querySelectorAll('#interactiveContainer .item');
      items.forEach((item, index) => {
        item.style.flexGrow = 1;
        item.textContent = `Item ${index + 1}`;
      });
    }
    
    // Calculate actual width distribution
    function calculateDistribution(container) {
      const items = container.querySelectorAll('.item');
      const containerWidth = container.offsetWidth;
      const gapWidth = 10 * (items.length - 1);
      const availableWidth = containerWidth - gapWidth;
      
      // Get total flex-grow values
      let totalGrow = 0;
      const growValues = [];
      
      items.forEach(item => {
        const grow = parseFloat(getComputedStyle(item).flexGrow) || 0;
        growValues.push(grow);
        totalGrow += grow;
      });
      
      // Calculate each item's width
      const widths = growValues.map((grow, index) => {
        if (totalGrow === 0) return 0;
        return (availableWidth * grow / totalGrow);
      });
      
      return {
        containerWidth,
        availableWidth,
        totalGrow,
        distribution: widths
      };
    }
    
    // Log distribution for demo
    console.log('Equal grow distribution:', 
      calculateDistribution(document.querySelector('.equal-grow')));
  </script>
</body>
</html>
```

**Why**: Fundamental for responsive layouts - allows flexible distribution of space without fixed widths, adapts to different screen sizes automatically.

## 8. Promise Resolve

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Promise.resolve() Demo</title>
</head>
<body>
  <!-- PROBLEM: Create immediately resolved promises and convert values to promises
       TIME: O(1) SPACE: O(1) -->
  
  <div id="output" style="font-family: monospace; padding: 20px;"></div>

  <script>
    const output = document.getElementById('output');
    
    function log(message) {
      const div = document.createElement('div');
      div.textContent = message;
      output.appendChild(div);
    }
    
    // ============================================
    // Promise.resolve() creates an already-resolved Promise
    // Useful for starting promise chains or converting values
    // ============================================
    
    // 1. Create resolved promise with value
    const resolvedPromise = Promise.resolve(42);
    
    resolvedPromise.then(value => {
      log(`Resolved value: ${value}`); // 42
    });
    
    // 2. Convert non-promise value to promise
    function asyncOperation(maybePromise) {
      // Ensure we always work with a promise
      return Promise.resolve(maybePromise);
    }
    
    asyncOperation('immediate value')
      .then(val => log(`Converted: ${val}`));
    
    // 3. Flatten nested promises
    const nestedPromise = Promise.resolve(Promise.resolve(Promise.resolve('deep')));
    // Automatically unwraps to single promise
    nestedPromise.then(val => log(`Flattened: ${val}`));
    
    // 4. Starting promise chains with initial value
    Promise.resolve({ id: 1, name: 'Initial' })
      .then(data => {
        data.processed = true;
        return data;
      })
      .then(data => {
        data.timestamp = Date.now();
        return data;
      })
      .then(data => log(`Chain result: ${JSON.stringify(data)}`));
    
    // 5. Mock async function for testing
    function mockFetch(id) {
      // Simulates successful API response
      return Promise.resolve({
        id,
        data: 'Mock data',
        status: 200
      });
    }
    
    mockFetch(123).then(response => {
      log(`Mock response: ${JSON.stringify(response)}`);
    });
    
    // 6. Convert callback-style cache to promise
    const cache = {
      'user:1': { name: 'John' },
      'user:2': { name: 'Jane' }
    };
    
    function getFromCache(key) {
      const value = cache[key];
      if (value) {
        // Cache hit - return resolved promise
        return Promise.resolve(value);
      }
      // Cache miss - would normally fetch from server
      return Promise.reject(new Error('Not found'));
    }
    
    getFromCache('user:1')
      .then(user => log(`Cache hit: ${user.name}`))
      .catch(err => log(`Cache miss: ${err.message}`));
    
    // 7. Promise.resolve vs new Promise
    // These are equivalent:
    Promise.resolve('value');
    new Promise(resolve => resolve('value'));
    
    // But Promise.resolve is more concise and performs better
    
    // 8. Handling thenables (promise-like objects)
    const thenable = {
      then(resolve) {
        resolve('Thenable resolved!');
      }
    };
    
    Promise.resolve(thenable)
      .then(val => log(`Thenable: ${val}`));
    
    // 9. Error handling - Promise.resolve doesn't reject
    // To create rejected promise, use Promise.reject()
    Promise.resolve(new Error('This is still resolved'))
      .then(value => log(`Error as value: ${value.message}`));
    
    // 10. Real-world example: Conditional async flow
    function getData(useCache) {
      if (useCache && cache['user:1']) {
        // Return cached data as promise
        return Promise.resolve(cache['user:1']);
      }
      // Fetch from server
      return fetch('/api/user/1').then(r => r.json());
    }
    
    getData(true) // Use cache
      .then(data => log(`Conditional: ${JSON.stringify(data)}`));
  </script>
</body>
</html>
```

**Why**: Essential for promise chaining, converting values to promises, and creating mock/test data. Simplifies async code by ensuring consistent promise interfaces.

```html
<body>
  <!-- PROBLEM: Persist data in browser storage with key-value pairs
       TIME: O(1) SPACE: O(n) where n is storage size -->
  
  <div class="storage-demo">
    <h2>LocalStorage Demo</h2>
    
    <label>Key:</label>
    <input type="text" id="storageKey" placeholder="Enter key">
    
    <label>Value:</label>
    <textarea id="storageValue" placeholder="Enter value" rows="3"></textarea>
    
    <div>

    </div>
    
    <div id="notification" class="notification" style="display:none;"></div>
    
    <h3>Current Storage:</h3>
    <div class="storage-display" id="storageDisplay"></div>
  </div>

  <script>
    // ============================================
    // localStorage.setItem(key, value)
    // - Stores data with no expiration date
    // - Data persists even after browser close
    // - Limited to ~5MB per origin
    // - Only stores strings (must serialize objects)
    // ============================================
    
    const keyInput = document.getElementById('storageKey');
    const valueInput = document.getElementById('storageValue');
    const display = document.getElementById('storageDisplay');
    const notification = document.getElementById('notification');
    
    function showNotification(message, isError = false) {
      notification.textContent = message;
      notification.style.background = isError ? '#f8d7da' : '#d4edda';
      notification.style.borderColor = isError ? '#dc3545' : '#28a745';
      notification.style.display = 'block';
      setTimeout(() => notification.style.display = 'none', 3000);
    }
    
    // 1. Basic setItem
    function saveToStorage() {
      const key = keyInput.value.trim();
      const value = valueInput.value;
      
      if (!key) {
        showNotification('Please enter a key', true);
        return;
      }
      
      try {
        localStorage.setItem(key, value);
        showNotification(`Saved "${key}" successfully`);
        updateDisplay();
      } catch (error) {
        // Handle QuotaExceededError
        showNotification(`Error: ${error.message}`, true);
      }
    }
    
    // 2. Basic getItem
    function loadFromStorage() {
      const key = keyInput.value.trim();
      
      if (!key) {
        showNotification('Please enter a key', true);
        return;
      }
      
      const value = localStorage.getItem(key);
      
      if (value !== null) {
        valueInput.value = value;
        showNotification(`Loaded "${key}"`);
      } else {
        showNotification(`Key "${key}" not found`, true);
      }
    }
    
    // 3. Remove item
    function removeFromStorage() {
      const key = keyInput.value.trim();
      localStorage.removeItem(key);
      showNotification(`Removed "${key}"`);
      updateDisplay();
    }
    
    // 4. Clear all storage
    function clearStorage() {
      if (confirm('Clear all localStorage data?')) {
        localStorage.clear();
        showNotification('Storage cleared');
        updateDisplay();
      }
    }
    
    // Display current storage contents
    function updateDisplay() {
      const items = [];
      for (let i = 0; i < localStorage.length; i++) {
        const key = localStorage.key(i);
        const value = localStorage.getItem(key);
        items.push(`${key}: ${value}`);
      }
      display.textContent = items.join('\n') || '(empty)';
    }
    
    // ============================================
    // Advanced localStorage patterns
    // ============================================
    
    // 5. Store objects (JSON serialization required)
    function saveObject(key, obj) {
      try {
        const serialized = JSON.stringify(obj);
        localStorage.setItem(key, serialized);
        return true;
      } catch (error) {
        console.error('Failed to save object:', error);
        return false;
      }
    }
    
    function loadObject(key, defaultValue = null) {
      try {
        const serialized = localStorage.getItem(key);
        return serialized ? JSON.parse(serialized) : defaultValue;
      } catch (error) {
        console.error('Failed to load object:', error);
        return defaultValue;
      }
    }
    
    // 6. Storage with expiration
    function setWithExpiry(key, value, ttlMs) {
      const item = {
        value,
        expiry: Date.now() + ttlMs
      };
      localStorage.setItem(key, JSON.stringify(item));
    }
    
    function getWithExpiry(key) {
      const serialized = localStorage.getItem(key);
      if (!serialized) return null;
      
      const item = JSON.parse(serialized);
      
      // Check if expired
      if (Date.now() > item.expiry) {
        localStorage.removeItem(key);
        return null;
      }
      
      return item.value;
    }
    
    // 7. Listen for storage changes (across tabs)
    window.addEventListener('storage', function(e) {
      console.log('Storage changed:', {
        key: e.key,
        oldValue: e.oldValue,
        newValue: e.newValue,
        url: e.url
      });
      updateDisplay();
    });
    
    // 8. Check available space
    function getStorageSize() {
      let total = 0;
      for (let key in localStorage) {
        if (localStorage.hasOwnProperty(key)) {
          total += localStorage.getItem(key).length * 2; // UTF-16
        }
      }
      return (total / 1024).toFixed(2) + ' KB';
    }
```
