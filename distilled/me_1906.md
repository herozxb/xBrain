## VP-Tree (Vantage Point Tree)
```python
# PROBLEM: Build a metric space indexing structure for nearest neighbor search
# TIME: O(log n) average search, O(n log n) build SPACE: O(n)
import heapq
from typing import List, Tuple, Callable, Optional

class VPTreeNode:
    def __init__(self, point, threshold=0, left=None, right=None):
        self.point = point
        self.threshold = threshold
        self.left = left
        self.right = None

class VPTree:
    """Vantage Point Tree for efficient nearest neighbor search in metric spaces."""
    
    def __init__(self, points: List, distance_fn: Callable):
        self.distance = distance_fn
        self.root = self._build(points) if points else None
    
    def _build(self, points: List) -> Optional[VPTreeNode]:
        if not points:
            return None
        
        # Choose vantage point (first point)
        vp = points[0]
        if len(points) == 1:
            return VPTreeNode(vp)
        
        # Calculate distances to vantage point
        distances = [(p, self.distance(vp, p)) for p in points[1:]]
        distances.sort(key=lambda x: x[1])
        
        # Median as threshold
        mid = len(distances) // 2
        threshold = distances[mid][1]
        
        # Partition into inside and outside
        inside = [p for p, d in distances[:mid]]
        outside = [p for p, d in distances[mid:]]
        
        return VPTreeNode(
            vp, threshold,
            self._build(inside),
            self._build(outside)
        )
    
    def nearest_neighbor(self, query, k=1) -> List[Tuple]:
        """Find k nearest neighbors to query point."""
        if not self.root:
            return []
        
        heap = []  # Max-heap (negated distances)
        self._search(self.root, query, k, heap)
        return sorted([(-d, p) for p, d in heap])
    
    def _search(self, node: VPTreeNode, query, k: int, heap: List):
        if not node:
            return
        
        dist = self.distance(node.point, query)
        
        # Add to heap if not full or closer than farthest
        if len(heap) < k:
            heapq.heappush(heap, (node.point, -dist))
        elif dist < -heap[0][1]:
            heapq.heappushpop(heap, (node.point, -dist))
        
        # Decide which subtree(s) to search
        if dist < node.threshold:
            # Query inside: search inside first
            self._search(node.left, query, k, heap)
            if len(heap) < k or node.threshold - dist < -heap[0][1]:
                self._search(node.right, query, k, heap)
        else:
            # Query outside: search outside first
            self._search(node.right, query, k, heap)
            if len(heap) < k or dist - node.threshold < -heap[0][1]:
                self._search(node.left, query, k, heap)

# Usage example
def euclidean(a, b):
    return sum((x - y) ** 2 for x, y in zip(a, b)) ** 0.5
```
**Why**: VP-Trees provide efficient nearest neighbor search in O(log n) average time for any metric space, useful for similarity search, clustering, and recommendation systems.

## Cover Tree
```python
# PROBLEM: Index metric spaces with guaranteed logarithmic search time
# TIME: O(log n) search, O(n log n) build SPACE: O(n)
from typing import List, Tuple, Callable, Optional, Set

class CoverTreeNode:
    def __init__(self, point, level: int):
        self.point = point
        self.level = level
        self.children: List['CoverTreeNode'] = []

class CoverTree:
    """Cover tree with O(log n) nearest neighbor guarantee."""
    
    def __init__(self, points: List, distance_fn: Callable, base: float = 2):
        self.distance = distance_fn
        self.base = base
        self.root = None
        for p in points:
            self.insert(p)
    
    def insert(self, point) -> bool:
        if not self.root:
            self.root = CoverTreeNode(point, 0)
            return True
        
        # Find nearest neighbor to determine insertion level
        dist = self.distance(self.root.point, point)
        level = int(-math.log(dist) / math.log(self.base)) if dist > 0 else 0
        
        return self._insert(self.root, point, level, float('inf'))
    
    def _insert(self, node: CoverTreeNode, point, level: int, max_dist: float) -> bool:
        dist = self.distance(node.point, point)
        
        # Check if within cover distance
        if dist > max_dist:
            return False
        
        # Try to insert into existing children
        for child in node.children:
            if self._insert(child, point, level - 1, max_dist / self.base):
                return True
        
        # Create new child node
        node.children.append(CoverTreeNode(point, level))
        return True
    
    def nearest_neighbor(self, query) -> Tuple:
        """Find single nearest neighbor."""
        if not self.root:
            return None
        
        best = (self.root.point, self.distance(self.root.point, query))
        self._search(self.root, query, best)
        return best
    
    def _search(self, node: CoverTreeNode, query, best: Tuple):
        dist = self.distance(node.point, query)
        
        if dist < best[1]:
            best = (node.point, dist)
        
        # Search children in order of increasing distance
        for child in sorted(node.children, 
                          key=lambda c: self.distance(c.point, query)):
            child_dist = self.distance(child.point, query)
            if child_dist < best[1] + self.base ** child.level:
                self._search(child, query, best)
```

## Ball Tree
```python
# PROBLEM: Efficient nearest neighbor search for high-dimensional data
# TIME: O(log n) average search, O(n log n) build SPACE: O(n)
from typing import List, Tuple, Callable, Optional
import numpy as np

class BallTreeNode:
    def __init__(self, centroid, radius, points=None, left=None, right=None):
        self.centroid = centroid
        self.radius = radius
        self.points = points  # Leaf node points
        self.left = left
        self.right = right

class BallTree:
    """Ball tree for spatial indexing and nearest neighbor search."""
    
    def __init__(self, points: List, distance_fn: Callable, leaf_size: int = 20):
        self.distance = distance_fn
        self.leaf_size = leaf_size
        self.root = self._build(np.array(points)) if points else None
    
    def _build(self, points: np.ndarray) -> Optional[BallTreeNode]:
        if len(points) == 0:
            return None
        
        if len(points) <= self.leaf_size:
            # Leaf node
            centroid = points.mean(axis=0)
            radius = max(self.distance(centroid, p) for p in points)
            return BallTreeNode(centroid, radius, points=points)
        
        # Choose split dimension (max spread)
        spreads = points.max(axis=0) - points.min(axis=0)
        split_dim = np.argmax(spreads)
        
        # Split at median
        median = np.median(points[:, split_dim])
        left_pts = points[points[:, split_dim] < median]
        right_pts = points[points[:, split_dim] >= median]
        
        # Handle edge case
        if len(left_pts) == 0 or len(right_pts) == 0:
            centroid = points.mean(axis=0)
            radius = max(self.distance(centroid, p) for p in points)
            return BallTreeNode(centroid, radius, points=points)
        
        # Build children
        left = self._build(left_pts)
        right = self._build(right_pts)
        
        # Calculate centroid and radius
        centroid = points.mean(axis=0)
        radius = max(self.distance(centroid, p) for p in points)
        
        return BallTreeNode(centroid, radius, left=left, right=right)
    
    def query(self, point, k=1) -> List[Tuple[float, any]]:
        """Find k nearest neighbors."""
        if not self.root:
            return []
        
        heap = []
        self._search(self.root, np.array(point), k, heap)
        return sorted([(-d, p) for p, d in heap])
    
    def _search(self, node: BallTreeNode, point: np.ndarray, k: int, heap: List):
        dist = self.distance(node.centroid, point)
        
        # Pruning: skip if ball too far
        if heap and dist - node.radius > -heap[0][1]:
            return
        
        if node.points is not None:
            # Leaf: check all points
            for p in node.points:
                d = self.distance(p, point)
                if len(heap) < k:
                    heapq.heappush(heap, (p, -d))
                elif d < -heap[0][1]:
                    heapq.heappushpop(heap, (p, -d))
        else:
            # Internal: search closer child first
            left_dist = self.distance(node.left.centroid, point) if node.left else float('inf')
            right_dist = self.distance(node.right.centroid, point) if node.right else float('inf')
            
            if left_dist < right_dist:
                self._search(node.left, point, k, heap)
                self._search(node.right, point, k, heap)
            else:
                self._search(node.right, point, k, heap)
                self._search(node.left, point, k, heap)

import heapq
```
**Why**: Ball trees are efficient for high-dimensional data where KD-trees degrade, using hyperspherical bounding volumes that better match data distributions.

## Event Dispatcher Pattern
```python
# PROBLEM: Decouple event sources from handlers for flexible pub-sub system
# TIME: O(1) subscribe, O(n) dispatch where n = number of listeners
from typing import Callable, Dict, List, Any
from collections import defaultdict

class EventDispatcher:
    """Centralized event dispatcher with type-based routing."""
    
    def __init__(self):
        self._listeners: Dict[str, List[Callable]] = defaultdict(list)
        self._once_listeners: Dict[str, List[Callable]] = defaultdict(list)
    
    def on(self, event_type: str, listener: Callable) -> 'EventDispatcher':
        """Subscribe to event type permanently."""
        self._listeners[event_type].append(listener)
        return self
    
    def once(self, event_type: str, listener: Callable) -> 'EventDispatcher':
        """Subscribe to event type once (auto-removed after firing)."""
        self._once_listeners[event_type].append(listener)
        return self
    
    def off(self, event_type: str, listener: Callable) -> bool:
        """Unsubscribe listener from event type."""
        if listener in self._listeners[event_type]:
            self._listeners[event_type].remove(listener)
            return True
        return False
    
    def emit(self, event_type: str, *args, **kwargs) -> int:
        """Emit event to all listeners, returns count of listeners called."""
        count = 0
        
        # Call permanent listeners
        for listener in self._listeners[event_type]:
            listener(*args, **kwargs)
            count += 1
        
        # Call and remove one-time listeners
        for listener in self._once_listeners[event_type]:
            listener(*args, **kwargs)
            count += 1
        self._once_listeners[event_type].clear()
        
        return count
    
    def clear(self, event_type: str = None):
        """Clear all listeners for event type (or all if None)."""
        if event_type:
            self._listeners[event_type].clear()
            self._once_listeners[event_type].clear()
        else:
            self._listeners.clear()
            self._once_listeners.clear()

# Usage example
dispatcher = EventDispatcher()
dispatcher.on('user_login', lambda user: print(f"User logged in: {user}"))
dispatcher.emit('user_login', 'Alice')
```
**Why**: Event dispatchers enable loose coupling between components, allowing multiple independent handlers to respond to events without the emitter knowing about them.

## Listener Pattern
```python
# PROBLEM: Implement observer pattern with lifecycle management
# TIME: O(1) register/remove, O(n) notify
from typing import Protocol, List, Any
from abc import ABC, abstractmethod

class EventListener(Protocol):
    """Protocol for event listeners."""
    def on_event(self, event_type: str, data: Any) -> None:
        ...

class EventSource:
    """Subject that maintains list of listeners and notifies them."""
    
    def __init__(self):
        self._listeners: List[EventListener] = []
    
    def add_listener(self, listener: EventListener) -> None:
        """Register listener."""
        if listener not in self._listeners:
            self._listeners.append(listener)
    
    def remove_listener(self, listener: EventListener) -> bool:
        """Unregister listener."""
        try:
            self._listeners.remove(listener)
            return True
        except ValueError:
            return False
    
    def notify(self, event_type: str, data: Any = None) -> int:
        """Notify all listeners, returns count of listeners notified."""
        count = 0
        for listener in self._listeners[:]:  # Copy to allow removal during iteration
            listener.on_event(event_type, data)
            count += 1
        return count

class LoggingListener:
    """Concrete listener that logs events."""
    
    def __init__(self, name: str):
        self.name = name
    
    def on_event(self, event_type: str, data: Any) -> None:
        print(f"[{self.name}] Event: {event_type}, Data: {data}")

# Usage
source = EventSource()

```
**Why**: The listener pattern provides a clean interface for implementing the observer pattern with proper lifecycle management and type safety.

## Subscriber Pattern
```python
# PROBLEM: Type-safe pub-sub with channel-based message routing
# TIME: O(1) subscribe, O(m) publish where m = subscriber count
from typing import Callable, Dict, Set, Any, TypeVar
from dataclasses import dataclass
from datetime import datetime

T = TypeVar('T')

@dataclass
class Message:
    """Envelope for published messages."""
    channel: str
    data: Any
    timestamp: datetime
    metadata: dict

class Subscriber:
    """Subscriber with callback and optional filtering."""
    
    def __init__(self, callback: Callable[[Message], None], 
                 channels: Set[str] = None,
                 filter_fn: Callable[[Message], bool] = None):
        self.callback = callback
        self.channels = channels  # None = all channels
        self.filter_fn = filter_fn
        self.active = True
    
    def receives(self, message: Message) -> bool:
        """Check if subscriber accepts message."""
        if not self.active:
            return False
        if self.channels and message.channel not in self.channels:
            return False
        if self.filter_fn and not self.filter_fn(message):
            return False
        return True

class PubSub:
    """Publisher-Subscriber system with channels."""
    
    def __init__(self):
        self.subscribers: Set[Subscriber] = set()
        self._history: List[Message] = []
    
    def subscribe(self, callback: Callable, channels: Set[str] = None,
                  filter_fn: Callable = None) -> Subscriber:
        """Create and register subscriber."""
        sub = Subscriber(callback, channels, filter_fn)
        self.subscribers.add(sub)
        return sub
    
    def unsubscribe(self, subscriber: Subscriber) -> bool:
        """Remove subscriber."""
        return subscriber in self.subscribers and self.subscribers.remove(subscriber) is None
    
    def publish(self, channel: str, data: Any, metadata: dict = None) -> int:
        """Publish message to channel, returns delivery count."""
        message = Message(channel, data, datetime.now(), metadata or {})
        self._history.append(message)
        
        delivered = 0
        for sub in list(self.subscribers):
            if sub.receives(message):
                sub.callback(message)
                delivered += 1
        
        return delivered
    
    def get_history(self, channel: str = None) -> List[Message]:
        """Get message history for channel."""
        if channel:
            return [m for m in self._history if m.channel == channel]
        return self._history.copy()

# Usage
pubsub = PubSub()
pubsub.subscribe(print, channels={'alerts'})
pubsub.publish('alerts', 'System overload!')
```
**Why**: Subscriber pattern extends observer with channel-based routing, message filtering, and history tracking for more sophisticated event handling.

## Backend for Frontend (BFF)
```python
# PROBLEM: Create dedicated backend APIs tailored for specific frontend clients
# TIME: O(1) request routing SPACE: O(n) for n routes
from typing import Dict, Callable, Any, Optional
from dataclasses import dataclass
from abc import ABC, abstractmethod

@dataclass
class Request:
    path: str
    method: str
    headers: Dict[str, str]
    body: Any
    params: Dict[str, str]

@dataclass
class Response:
    status: int
    body: Any
    headers: Dict[str, str] = None

class BackendForFrontend:
    """BFF pattern implementation for client-specific APIs."""
    
    def __init__(self, client_type: str):
        self.client_type = client_type
        self.routes: Dict[str, Callable] = {}
        self.middleware: List[Callable] = []
        self._services: Dict[str, Any] = {}
    
    def route(self, path: str, method: str = 'GET'):
        """Decorator for route registration."""
        def decorator(fn: Callable) -> Callable:
            key = f"{method}:{path}"
            self.routes[key] = fn
            return fn
        return decorator
    
    def use(self, middleware: Callable):
        """Add middleware."""
        self.middleware.append(middleware)
        return self
    
    def register_service(self, name: str, service: Any):
        """Register downstream service."""
        self._services[name] = service
    
    def handle(self, request: Request) -> Response:
        """Route request through middleware to handler."""
        # Apply middleware
        for mw in self.middleware:
            request = mw(request)
            if isinstance(request, Response):
                return request
        
        # Find handler
        key = f"{request.method}:{request.path}"
        handler = self.routes.get(key)
        
        if not handler:
            return Response(404, {"error": "Not found"})
        
        try:
            return handler(request, self._services)
        except Exception as e:
            return Response(500, {"error": str(e)})

# Usage: Mobile BFF
mobile_bff = BackendForFrontend("mobile")

@mobile_bff.route("/user/profile", "GET")
def get_mobile_profile(req: Request, services: Dict) -> Response:
    # Mobile-optimized: fewer fields
    user = services['user_service'].get(req.params['id'])
    return Response(200, {
        "name": user.name,
        "avatar": user.avatar_mobile
    })
```
**Why**: BFF pattern creates client-specific backends that aggregate and transform data appropriately, reducing frontend complexity and network overhead.

## Sidecar Pattern
```python
# PROBLEM: Augment services with cross-cutting concerns without code changes
# TIME: O(1) delegation, O(n) for n sidecars
from typing import Callable, Any, List, Optional
from dataclasses import dataclass
import time

@dataclass
class ServiceRequest:
    service: str
    method: str
    params: dict
    metadata: dict = None

@dataclass  
class ServiceResponse:
    success: bool
    data: Any
    error: str = None
    latency_ms: float = 0

class Sidecar:
    """Sidecar that intercepts and augments service calls."""
    
    def __init__(self, name: str):
        self.name = name
        self.pre_hooks: List[Callable] = []
        self.post_hooks: List[Callable] = []
    
    def before(self, hook: Callable) -> 'Sidecar':
        """Add pre-processing hook."""
        self.pre_hooks.append(hook)
        return self
    
    def after(self, hook: Callable) -> 'Sidecar':
        """Add post-processing hook."""
        self.post_hooks.append(hook)
        return self
    
    def intercept(self, request: ServiceRequest, 
                  next_handler: Callable) -> ServiceResponse:
        """Apply hooks before/after service call."""
        # Pre-processing
        for hook in self.pre_hooks:
            request = hook(request) or request
        
        # Call service
        response = next_handler(request)
        
        # Post-processing
        for hook in self.post_hooks:
            response = hook(response) or response
        
        return response

class ServiceWithSidecars:
    """Main service wrapped with sidecars."""
    
    def __init__(self, service_fn: Callable):
        self.service_fn = service_fn
        self.sidecars: List[Sidecar] = []
    
    def add_sidecar(self, sidecar: Sidecar) -> 'ServiceWithSidecars':
        self.sidecars.append(sidecar)
        return self
    
    def call(self, request: ServiceRequest) -> ServiceResponse:
        """Execute request through sidecar chain."""
        def handler(req: ServiceRequest) -> ServiceResponse:
            start = time.time()
            try:
                data = self.service_fn(**req.params)
                latency = (time.time() - start) * 1000
                return ServiceResponse(True, data, latency_ms=latency)
            except Exception as e:
                return ServiceResponse(False, None, str(e))
        
        # Chain sidecars
        result = handler
        for sidecar in reversed(self.sidecars):
            prev_handler = result
            result = lambda req, s=sidecar, h=prev_handler: s.intercept(req, h)
        
        return result(request)

# Usage: Logging sidecar
logging_sidecar = Sidecar("logger")
logging_sidecar.before(lambda req: print(f"Calling {req.service}.{req.method}"))
logging_sidecar.after(lambda res: print(f"Result: {res.success}"))

service = ServiceWithSidecars(lambda x: x * 2)
service.add_sidecar(logging_sidecar)
```
**Why**: Sidecars add cross-cutting concerns (logging, monitoring, retry) without modifying core service logic, enabling reusable infrastructure components.

## PACELC Theorem
```python
# PROBLEM: Trade-off between consistency, availability, and latency in distributed systems
# TIME: O(1) decision SPACE: O(1)
from enum import Enum
from typing import Callable, Any, Tuple
from dataclasses import dataclass

class ConsistencyLevel(Enum):
    STRONG = "strong"
    EVENTUAL = "eventual"
    CAUSAL = "causal"

class AvailabilityMode(Enum):
    AVAILABLE = "available"
    CONSISTENT = "consistent"

@dataclass
class SystemState:
    partition: bool
    latency_sensitive: bool
    consistency: ConsistencyLevel
    availability: AvailabilityMode

class PACELCCoordinator:
    """
    PACELC: If Partition (P), trade Availability (A) for Consistency (C);
    Else (E), trade Latency (L) for Consistency (C).
    """
    
    def __init__(self, prefer_consistency: bool = True):
        self.prefer_consistency = prefer_consistency
        self.state = SystemState(
            partition=False,
            latency_sensitive=False,
            consistency=ConsistencyLevel.STRONG,
            availability=AvailabilityMode.AVAILABLE
        )
    
    def handle_partition(self, is_partitioned: bool) -> SystemState:
        """
        P: During partition, choose between A and C.
        """
        self.state.partition = is_partitioned
        
        if is_partitioned:
            if self.prefer_consistency:
                # Sacrifice availability for consistency
                self.state.availability = AvailabilityMode.CONSISTENT
                self.state.consistency = ConsistencyLevel.STRONG
            else:
                # Sacrifice consistency for availability
                self.state.availability = AvailabilityMode.AVAILABLE
                self.state.consistency = ConsistencyLevel.EVENTUAL
        else:
            self.state.availability = AvailabilityMode.AVAILABLE
        
        return self.state
    
    def handle_latency_tradeoff(self, latency_critical: bool) -> SystemState:
        """
        E: When no partition, trade Latency for Consistency.
        """
        self.state.latency_sensitive = latency_critical
        
        if not self.state.partition:
            if latency_critical:
                # Favor lower latency over strong consistency
                self.state.consistency = ConsistencyLevel.EVENTUAL
            else:
                # Favor strong consistency, accept higher latency
                self.state.consistency = ConsistencyLevel.STRONG
        
        return self.state
    
    def get_read_strategy(self) -> Tuple[Callable, Any]:
        """Get appropriate read strategy based on current state."""
        if self.state.consistency == ConsistencyLevel.STRONG:
            return self._read_strong, {"quorum": True}
        elif self.state.consistency == ConsistencyLevel.CAUSAL:
            return self._read_causal, {"vector_clock": True}
        else:
            return self._read_eventual, {"any_replica": True}
    
    def _read_strong(self, key: str, **kwargs) -> Any:
        """Read with quorum - strong consistency."""
        # Would contact majority of replicas
        return f"strong_read({key})"
    
    def _read_causal(self, key: str, **kwargs) -> Any:
        """Read with causal consistency using vector clocks."""
        return f"causal_read({key})"
    
    def _read_eventual(self, key: str, **kwargs) -> Any:
        """Read from any replica - eventual consistency."""
        return f"eventual_read({key})"

# Usage
coordinator = PACELCCoordinator(prefer_consistency=True)
state = coordinator.handle_partition(True)  # During partition
print(f"Consistency: {state.consistency}, Availability: {state.availability}")
```
**Why**: PACELC extends CAP by addressing normal operation trade-offs, recognizing that systems must balance latency vs. consistency even without partitions.

## BASE Properties
```python
# PROBLEM: Implement relaxed consistency model for high availability systems
# TIME: O(1) basic operations SPACE: O(n) for n entities
from typing import Dict, Any, Callable, List, Optional
from dataclasses import dataclass, field
from datetime import datetime, timedelta
import uuid

@dataclass
class Entity:
    """Entity with versioning for eventual consistency."""
    id: str
    data: Dict[str, Any]
    version: int = 1
    timestamp: datetime = field(default_factory=datetime.now)
    tombstone: bool = False

class BASEStore:
    """
    BASE: Basically Available, Soft state, Eventually consistent.
    """
    
    def __init__(self, sync_delay_seconds: int = 5):
        self.store: Dict[str, Entity] = {}
        self.pending_updates: List[Entity] = []
        self.sync_delay = timedelta(seconds=sync_delay_seconds)
        self.sync_handlers: List[Callable] = []
    
    def write(self, key: str, value: Dict[str, Any]) -> Entity:
        """Basically Available: Write succeeds immediately (to local)."""
        if key in self.store:
            entity = self.store[key]
            entity.data = value
            entity.version += 1
            entity.timestamp = datetime.now()
        else:
            entity = Entity(str(uuid.uuid4()), value)
            self.store[key] = entity
        
        self.pending_updates.append(entity)
        return entity
    
    def read(self, key: str) -> Optional[Entity]:
        """Read current (possibly stale) value."""
        return self.store.get(key)
    
    def delete(self, key: str) -> bool:
        """Soft delete with tombstone."""
        if key in self.store:
            entity = self.store[key]
            entity.tombstone = True
            entity.version += 1
            self.pending_updates.append(entity)
            return True
        return False
    
    def register_sync(self, handler: Callable):
        """Register handler for propagating updates."""
        self.sync_handlers.append(handler)
    
    def sync(self) -> int:
        """
        Eventually consistent: Propagate pending updates.
        Returns count of synced entities.
        """
        synced = 0
        cutoff = datetime.now() - self.sync_delay
        
        to_sync = [e for e in self.pending_updates if e.timestamp < cutoff]
        
        for entity in to_sync:
            for handler in self.sync_handlers:
                handler(entity)
            self.pending_updates.remove(entity)
            synced += 1
        
        return synced
    
    def merge(self, entity: Entity) -> bool:
        """
        Soft state: Merge incoming entity with local version.
        Last-write-wins with version check.
        """
        local = self.store.get(entity.id)
        
        if not local or entity.version > local.version:
            self.store[entity.id] = entity
            return True
        elif entity.version == local.version:
            # Conflict: timestamp wins
            if entity.timestamp > local.timestamp:
                self.store[entity.id] = entity
                return True
        
        return False
    
    def get_state(self) -> Dict[str, Any]:
        """Get current (soft) state for debugging."""
        return {
            "entities": len(self.store),
            "pending": len(self.pending_updates),
            "tombstones": sum(1 for e in self.store.values() if e.tombstone)
        }

# Usage
```
