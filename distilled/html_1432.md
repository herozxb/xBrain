```html
    /**
     * Get all bounding rectangles for an inline element
     * # PROBLEM: Get dimensions of each line in a multi-line inline element
     * # TIME: O(n) SPACE: O(n) where n = number of lines
     * 
     * @param {HTMLElement} element - The inline element to measure
     * @returns {DOMRectList} Collection of rects, one per line
     */
    function getInlineRects(element) {
      // For inline elements that wrap, returns one rect per line
      // Block elements return single rect (same as getBoundingClientRect)
      return element.getClientRects();
    }

    // Demo: Highlight each rect on hover
    const inlineEl = document.getElementById('inline-element');
    const infoEl = document.getElementById('info');
    
    inlineEl.addEventListener('mouseenter', function() {
      const rects = getInlineRects(this);
      let info = `<strong>Found ${rects.length} client rects:</strong><br>`;
      
      for (let i = 0; i < rects.length; i++) {
        const r = rects[i];
        info += `Rect ${i + 1}: (${r.left.toFixed(0)}, ${r.top.toFixed(0)}) to (${r.right.toFixed(0)}, ${r.bottom.toFixed(0)})<br>`;
      }
      
      infoEl.innerHTML = info;
    });
  </script>
</body>
</html>
```

**Why:** Critical for inline elements that span multiple lines. Unlike getBoundingClientRect (which returns one combined rect), this gives each line's dimensions - essential for text selection, highlighting, and precise inline element manipulation.

```html
    function getViewportSize() {

        </span>
      `;
    }

    updateViewportInfo();
    
    // Update on resize
    window.addEventListener('resize', updateViewportInfo);
```

**Why:** Fundamental for responsive design, modal positioning, and viewport-relative calculations. Different from screen dimensions - this reflects actual visible area after browser chrome, perfect for UI responsiveness.

```html
    /**
     * Get mouse position relative to entire document (including scrolled parts)
     * # PROBLEM: Track mouse position in document coordinates
     * # TIME: O(1) SPACE: O(1)
     * 
     * @param {MouseEvent} event - The mouse event object
     * @returns {Object} Page-relative x and y coordinates
     */
    function getMousePagePosition(event) {
      // pageX/Y = clientX/Y + scroll offsets
      // Works across all scrolled content
      return {
        x: event.pageX,
        y: event.pageY
      };
    }

    /**
     * Alternative: Calculate from clientX/Y if pageX/Y unavailable
     * @param {MouseEvent} event - Mouse event
     * @returns {Object} Calculated page coordinates
     */
    function calculatePagePosition(event) {
      return {
        x: event.clientX + window.scrollX,
        y: event.clientY + window.scrollY
      };
    }

    // Demo: Track mouse and update cursor position
    const tracker = document.getElementById('tracker');
    const cursor = document.getElementById('cursor');

    document.addEventListener('mousemove', function(e) {
      const pos = getMousePagePosition(e);
      
      tracker.innerHTML = `
        <strong>Mouse Position</strong><br>
        Page X: ${pos.x}<br>
        Page Y: ${pos.y}<br>
        Scroll Y: ${window.scrollY}<br>
        Client Y: ${e.clientY}
      `;
      
      // Move cursor to mouse position
      cursor.style.left = pos.x + 'px';
      cursor.style.top = pos.y + 'px';
    });
  </script>
</body>
</html>
```

**Why:** pageX/pageY include scroll offset, unlike clientX/clientY. Essential for drag-and-drop, drawing applications, and any interaction that needs absolute document position regardless of scrolling.

## Event Delegation

```html

    #todo-list {

    }

    }
  </style>
</head>
<body>
  <h2>Click items to complete, × to delete</h2>
  <ul id="todo-list">
    <li>Learn HTML <span class="delete-btn">×</span></li>
    <li>Learn CSS <span class="delete-btn">×</span></li>
    <li>Learn JavaScript <span class="delete-btn">×</span></li>
  </ul>
  <button id="add-btn">Add Item</button>

  <script>
    /**
     * Event delegation pattern - handle events on parent instead of children
     * # PROBLEM: Efficiently handle events on dynamic or many child elements
     * # TIME: O(1) per event SPACE: O(1)
     * 
     * Benefits:
     * 1. Single event listener instead of many
     * 2. Works for dynamically added elements
     * 3. Lower memory usage
     */
    
    const todoList = document.getElementById('todo-list');
    const addBtn = document.getElementById('add-btn');

    /**
     * Setup event delegation on parent element
     * @param {HTMLElement} parent - Parent container
     * @param {string} childSelector - CSS selector for target children
     * @param {string} eventType - Event type to listen for
     * @param {Function} handler - Event handler function
     */
    function delegate(parent, childSelector, eventType, handler) {
      parent.addEventListener(eventType, function(event) {
        // Find closest ancestor matching selector (includes event.target)
        const targetElement = event.target.closest(childSelector);
        
        // Check if found and it's within our parent
        if (targetElement && parent.contains(targetElement)) {
          handler.call(targetElement, event);
        }
      });
    }

    // Demo: Single listener handles all list items (including future ones)
    
    // Toggle completion on li click
    delegate(todoList, 'li', 'click', function(event) {
      // Ignore if clicking delete button
      if (event.target.classList.contains('delete-btn')) return;
      this.classList.toggle('completed');
    });

    // Delete on × click
    delegate(todoList, '.delete-btn', 'click', function(event) {
      event.stopPropagation();
      this.parentElement.remove();
    });

    // Add new items dynamically - delegation handles them automatically!
    let itemCount = 4;
    addBtn.addEventListener('click', function() {
      const li = document.createElement('li');
      li.innerHTML = `New item ${itemCount++} <span class="delete-btn">×</span>`;
      todoList.appendChild(li);
    });
  </script>
</body>
</html>
```

**Why:** Event delegation uses event bubbling to handle many elements with a single listener. Essential for dynamic content (added after page load) and performance when dealing with hundreds/thousands of elements.

```html
    /**
     * Create and dispatch custom events
     * # PROBLEM: Create application-specific events for decoupled communication
     * # TIME: O(1) SPACE: O(1)
     */

    /**
     * Create a custom event with data
     * @param {string} eventName - Name of the custom event
     * @param {Object} detail - Data to pass with the event
     * @param {boolean} bubbles - Whether event should bubble
     * @returns {CustomEvent} The custom event object

    function createCustomEvent(eventName, detail = {}, bubbles = true) {
      return new CustomEvent(eventName, {
        detail: detail,
        bubbles: bubbles,
        cancelable: true
      });
    }

    /**
     * Dispatch custom event on an element
     * @param {HTMLElement} element - Target element
     * @param {CustomEvent} event - Event to dispatch
     */
    function dispatchCustomEvent(element, event) {
      element.dispatchEvent(event);
    }

    const appRoot = document.body;
    const notification = document.getElementById('notification');
    const log = document.getElementById('log');

    // Trigger custom events

      const event = createCustomEvent('user:login', { username: 'John' });

    });

    document.getElementById('logout-btn').addEventListener('click', function() {
      const event = createCustomEvent('user:logout');
      dispatchCustomEvent(appRoot, event);
    });

    document.getElementById('purchase-btn').addEventListener('click', function() {
      const event = createCustomEvent('purchase:complete', {
        item: 'Premium Plan',
        amount: 99.99
      });
      dispatchCustomEvent(appRoot, event);
    });
  </script>

</html>
```

**Why:** Custom events enable loose coupling between components. Different parts of your app can communicate without direct references - essential for modular architecture, plugin systems, and state management.

```html
    /**
     * Storage event handling for cross-tab communication
     * # PROBLEM: Detect and react to localStorage changes from other tabs
     * # TIME: O(1) SPACE: O(1)
     * 
     * Note: storage event fires in ALL OTHER tabs/windows, not the current one
     */

    const log = document.getElementById('log');
    const input = document.getElementById('data-input');
    
    /**
     * Listen for storage changes from other tabs
     * @param {Function} callback - Handler receiving event details
     */
    function onStorageChange(callback) {
      window.addEventListener('storage', function(event) {
        // event.key - changed key name
        // event.oldValue - previous value
        // event.newValue - new value
        // event.url - page URL that made the change
        // event.storageArea - localStorage or sessionStorage
        
        callback({
          key: event.key,
          oldValue: event.oldValue,
          newValue: event.newValue,
          url: event.url
```

```html
    /**
     * Cache API for storing request/response pairs
     * # PROBLEM: Store and retrieve network resources for offline access
     * # TIME: O(1) for cache operations SPACE: O(n) where n = cached items
     */

    const CACHE_NAME = 'demo-cache-v1';
    const logEl = document.getElementById('log');
    const cacheListEl = document.getElementById('cache-list');

    function log(msg) {
      const time = new Date().toLocaleTimeString();
      logEl.innerHTML += `[${time}] ${msg}<br>`;
      logEl.scrollTop = logEl.scrollHeight;
    }

    /**
     * Open or create a cache
     * @param {string} cacheName - Name of the cache
     * @returns {Promise<Cache>} Cache object
     */
    async function openCache(cacheName) {
      return await caches.open(cacheName);
    }

    /**
     * Add a resource to cache
     * @param {string} cacheName - Cache name
     * @param {string} url - URL to cache
     */
    async function cacheResource(cacheName, url) {
      try {
        const cache = await openCache(cacheName);
        // add() fetches and caches in one operation
        await cache.add(url);
        log(`✓ Cached: ${url}`);
      } catch (error) {
        log(`✗ Error caching: ${error.message}`);
      }
    }

    /**
     * List all cached URLs in a cache
     * @param {string} cacheName - Cache name
     * @returns {Promise<string[]>} Array of cached URLs
     */
    async function listCachedItems(cacheName) {
      const cache = await openCache(cacheName);
      const requests = await cache.keys();
      return requests.map(req => req.url);
    }

    /**
     * Retrieve a cached response
     * @param {string} cacheName - Cache name
     * @param {string} url - URL to retrieve
     * @returns {Promise<Response|null>} Cached response or null
     */
    async function getCachedResponse(cacheName, url) {
      const cache = await openCache(cacheName);
      return await cache.match(url);
    }

    /**
     * Delete entire cache
     * @param {string} cacheName - Cache name to delete
     */
    async function deleteCache(cacheName) {
      await caches.delete(cacheName);
      log(`✓ Cache "${cacheName}" deleted`);
    }

      } else {
        log('Not found in cache');
      }
    });

      await deleteCache(CACHE_NAME);
      cacheListEl.innerHTML = '<em>No cached items</em>';
    });
```
