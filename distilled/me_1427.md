## [Longest Substring Without Repeating Characters]
```python
# PROBLEM: Find the length of the longest substring without repeating characters
# TIME: O(n) SPACE: O(min(m, n)) where m is charset size
def length_of_longest_substring(s: str) -> int:
    # Use sliding window with hash map to track character indices
    char_index = {}  # maps character to its most recent index
    left = 0
    max_len = 0
    
    for right, char in enumerate(s):
        # If character seen and its index is within current window
        if char in char_index and char_index[char] >= left:
            # Shrink window from left to exclude the duplicate
            left = char_index[char] + 1
        
        # Update character's latest index
        char_index[char] = right
        
        # Update maximum length found
        max_len = max(max_len, right - left + 1)
    
    return max_len
```
**Why**: Classic sliding window demonstrating dynamic window adjustment with hash map for O(1) lookups.

## [Container With Most Water]
```python
# PROBLEM: Find two lines that form container holding the most water
# TIME: O(n) SPACE: O(1)
def max_area(height: list[int]) -> int:
    # Two pointers approach: start from widest container
    left, right = 0, len(height) - 1
    max_water = 0
    
    while left < right:
        # Calculate current container area
        width = right - left
        container_height = min(height[left], height[right])
        max_water = max(max_water, width * container_height)
        
        # Move pointer with smaller height (bottleneck)
        # Moving the larger one can only decrease area
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1
    
    return max_water
```
**Why**: Two pointers technique showing greedy optimization - always eliminate the weaker wall.

## [Subsets with Backtracking]
```python
# PROBLEM: Generate all possible subsets (power set) of a given set
# TIME: O(n * 2^n) SPACE: O(n) for recursion stack
def subsets(nums: list[int]) -> list[list[int]]:
    result = []
    
    def backtrack(start: int, current: list[int]):
        # Add current subset to result (make a copy)
        result.append(current[:])
        
        # Try adding each remaining element
        for i in range(start, len(nums)):
            # Include nums[i] in current subset
            current.append(nums[i])
            
            # Recurse with next starting index
            backtrack(i + 1, current)
            
            # Backtrack: remove nums[i] to try other combinations
            current.pop()
    
    backtrack(0, [])
    return result
```
**Why**: Demonstrates backtracking pattern - choose, explore, unchoose for exhaustive search.

## [Number of Connected Components]
```python
# PROBLEM: Count connected components in an undirected graph
# TIME: O(n * α(n)) ≈ O(n) SPACE: O(n)
class UnionFind:
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.components = n
    
    def find(self, x: int) -> int:
        # Path compression: point to root directly
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x: int, y: int) -> bool:
        # Union by rank for balanced trees
        px, py = self.find(x), self.find(y)
        if px == py:
            return False  # Already connected
        
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        self.components -= 1
        return True

def count_components(n: int, edges: list[list[int]]) -> int:
    uf = UnionFind(n)
    for u, v in edges:
        uf.union(u, v)
    return uf.components
```
**Why**: Union-Find with path compression and union by rank for near-constant time operations.

## [Coin Change]
```python
# PROBLEM: Find minimum coins needed to make target amount
# TIME: O(amount * n_coins) SPACE: O(amount)
def coin_change(coins: list[int], amount: int) -> int:
    # dp[i] = minimum coins to make amount i
    # Initialize with impossible value (amount + 1)
    dp = [amount + 1] * (amount + 1)
    dp[0] = 0  # Base case: 0 coins for amount 0
    
    # Build up solutions for each amount
    for i in range(1, amount + 1):
        for coin in coins:
            if coin <= i:
                # Either don't use this coin, or use it
                dp[i] = min(dp[i], dp[i - coin] + 1)
    
    return dp[amount] if dp[amount] <= amount else -1
```
**Why**: Classic DP showing optimal substructure - solution builds from smaller subproblems.

## [Course Schedule - Cycle Detection]
```python
# PROBLEM: Determine if all courses can be finished given prerequisites
# TIME: O(V + E) SPACE: O(V + E)
def can_finish(num_courses: int, prerequisites: list[list[int]]) -> bool:
    # Build adjacency list for graph
    graph = [[] for _ in range(num_courses)]
    for course, prereq in prerequisites:
        graph[course].append(prereq)
    
    # States: 0=unvisited, 1=visiting, 2=visited
    state = [0] * num_courses
    
    def has_cycle(course: int) -> bool:
        if state[course] == 1:  # Back edge found
            return True
        if state[course] == 2:  # Already processed
            return False
        
        state[course] = 1  # Mark as visiting
        
        for neighbor in graph[course]:
            if has_cycle(neighbor):
                return True
        
        state[course] = 2  # Mark as visited
        return False
    
    return not any(has_cycle(c) for c in range(num_courses))
```
**Why**: DFS-based cycle detection in directed graph using three-color marking.

## [Binary Tree Level Order Traversal]
```python
# PROBLEM: Traverse binary tree level by level (BFS)
# TIME: O(n) SPACE: O(w) where w is max width
from collections import deque

def level_order(root) -> list[list[int]]:
    if not root:
        return []
    
    result = []
    queue = deque([root])
    
    while queue:
        level_size = len(queue)
        current_level = []
        
        # Process all nodes at current level
        for _ in range(level_size):
            node = queue.popleft()
            current_level.append(node.val)
            
            # Add children for next level
            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)
        
        result.append(current_level)
    
    return result
```
**Why**: BFS pattern using queue - level-by-level processing essential for shortest path problems.

## [Kth Largest Element]
```python
# PROBLEM: Find kth largest element in unsorted array
# TIME: O(n log k) SPACE: O(k)
import heapq

def find_kth_largest(nums: list[int], k: int) -> int:
    # Maintain min-heap of k largest elements
    # Smallest of those k is the kth largest overall
    min_heap = []
    
    for num in nums:
        heapq.heappush(min_heap, num)
        
        # Keep only k elements in heap
        if len(min_heap) > k:
            heapq.heappop(min_heap)
    
    return min_heap[0]  # Root is kth largest
```
**Why**: Heap usage for efficient k-element tracking without sorting entire array.

## [Implement Trie]
```python
# PROBLEM: Implement a trie (prefix tree) with insert, search, startsWith
# TIME: O(m) per operation where m is word length SPACE: O(alphabet_size * total_chars)
class TrieNode:
    def __init__(self):
        self.children = {}  # char -> TrieNode
        self.is_end = False

class Trie:
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word: str) -> None:
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
    
    def search(self, word: str) -> bool:
        node = self._traverse(word)
        return node is not None and node.is_end
    
    def starts_with(self, prefix: str) -> bool:
        return self._traverse(prefix) is not None
    
    def _traverse(self, prefix: str) -> TrieNode:
        node = self.root
        for char in prefix:
            if char not in node.children:
                return None
            node = node.children[char]
        return node
```
**Why**: Trie for efficient prefix-based queries - essential for autocomplete, dictionary lookup.

## [Climbing Stairs with Memoization]
```python
# PROBLEM: Count ways to climb n stairs taking 1 or 2 steps
# TIME: O(n) SPACE: O(n) with cache, O(1) optimized
from functools import lru_cache

@lru_cache(maxsize=None)
def climb_stairs_memo(n: int) -> int:
    # Base cases
    if n <= 2:
        return n
    
    # Recurrence: ways(n) = ways(n-1) + ways(n-2)
    return climb_stairs_memo(n - 1) + climb_stairs_memo(n - 2)

def climb_stairs_dp(n: int) -> int:
    # Space-optimized DP (same as Fibonacci)
    if n <= 2:
        return n
    
    prev, curr = 1, 2
    for _ in range(3, n + 1):
        prev, curr = curr, prev + curr
    
    return curr
```
**Why**: Shows both @lru_cache decorator usage and space-optimized iterative DP.
