## 1. Tarjan's Strongly Connected Components

```python
# PROBLEM: Find all strongly connected components in a directed graph
# TIME: O(V + E) SPACE: O(V)
def tarjan_scc(graph):
    """
    Tarjan's algorithm for finding strongly connected components.
    Uses DFS with discovery times and low links.
    """
    index_counter = [0]
    stack = []
    lowlinks = {}
    index = {}
    on_stack = {}
    sccs = []
    
    def strongconnect(node):
        # Set discovery time and low link value
        index[node] = index_counter[0]
        lowlinks[node] = index_counter[0]
        index_counter[0] += 1
        on_stack[node] = True
        stack.append(node)
        
        # Explore neighbors
        for successor in graph.get(node, []):
            if successor not in index:
                # Successor not visited; recurse
                strongconnect(successor)
                lowlinks[node] = min(lowlinks[node], lowlinks[successor])
            elif on_stack.get(successor, False):
                # Successor is on stack, update low link
                lowlinks[node] = min(lowlinks[node], index[successor])
        
        # If node is root of SCC, pop the SCC
        if lowlinks[node] == index[node]:
            scc = []
            while True:
                successor = stack.pop()
                on_stack[successor] = False
                scc.append(successor)
                if successor == node:
                    break
            sccs.append(scc)
    
    # Run on all nodes
    for node in graph:
        if node not in index:
            strongconnect(node)
    
    return sccs
```

**Why**: Efficient single-pass algorithm for finding SCCs, crucial for analyzing dependency graphs and detecting cycles in directed graphs.

## 2. Lowest Common Ancestor with Binary Lifting

```python
# PROBLEM: Find the lowest common ancestor of two nodes in a tree efficiently
# TIME: O(N log N) preprocessing, O(log N) per query SPACE: O(N log N)
class LCA:
    def __init__(self, tree, root):
        """
        Initialize LCA structure using binary lifting technique.
        tree: adjacency list representation
        root: root node of the tree
        """
        self.tree = tree
        self.n = len(tree)
        self.log = (self.n).bit_length()
        
        # parent[i][j] = 2^j-th ancestor of node i
        self.parent = [[-1] * self.log for _ in range(self.n)]
        self.depth = [0] * self.n
        
        # BFS to fill parent table
        self._bfs(root)
        self._build_sparse_table()
    
    def _bfs(self, root):
        """Fill immediate parents and depths using BFS."""
        queue = [root]
        visited = set([root])
        
        while queue:
            node = queue.pop(0)
            for neighbor in self.tree[node]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    self.parent[neighbor][0] = node
                    self.depth[neighbor] = self.depth[node] + 1
                    queue.append(neighbor)
    
    def _build_sparse_table(self):
        """Build sparse table for binary lifting."""
        for j in range(1, self.log):
            for i in range(self.n):
                if self.parent[i][j-1] != -1:
                    self.parent[i][j] = self.parent[self.parent[i][j-1]][j-1]
    
    def lca(self, u, v):
        """Find lowest common ancestor of nodes u and v."""
        # Ensure u is deeper
        if self.depth[u] < self.depth[v]:
            u, v = v, u
        
        # Lift u to same depth as v
        diff = self.depth[u] - self.depth[v]
        for i in range(self.log):
            if diff & (1 << i):
                u = self.parent[u][i]
        
        if u == v:
            return u
        
        # Lift both nodes together
        for i in range(self.log - 1, -1, -1):
            if self.parent[u][i] != self.parent[v][i]:
                u = self.parent[u][i]
                v = self.parent[v][i]
        
        return self.parent[u][0]
```

**Why**: Enables fast LCA queries after linear preprocessing, essential for tree path queries and distance calculations.

## 3. Heavy-Light Decomposition

```python
# PROBLEM: Decompose tree into paths for efficient path queries
# TIME: O(N) decomposition, O(log^2 N) per path query SPACE: O(N)
class HeavyLightDecomposition:
    def __init__(self, tree, values, root=0):
        """
        Initialize HLD for path queries on trees.
        tree: adjacency list
        values: node values
        root: root node
        """
        self.tree = tree
        self.values = values
        self.n = len(tree)
        self.root = root
        
        # Decomposition arrays
        self.parent = [-1] * self.n
        self.depth = [0] * self.n
        self.size = [0] * self.n
        self.heavy = [-1] * self.n
        self.head = [0] * self.n
        self.pos = [0] * self.n
        self.chain = []
        
        # First DFS: compute sizes and heavy edges
        self._dfs_size(root)
        
        # Second DFS: decompose into chains
        self._dfs_decompose(root, root)
    
    def _dfs_size(self, node):
        """Compute subtree sizes and identify heavy edges."""
        self.size[node] = 1
        max_size = 0
        
        for child in self.tree[node]:
            if child != self.parent[node]:
                self.parent[child] = node
                self.depth[child] = self.depth[node] + 1
                self._dfs_size(child)
                
                self.size[node] += self.size[child]
                
                # Update heavy edge
                if self.size[child] > max_size:
                    max_size = self.size[child]
                    self.heavy[node] = child
    
    def _dfs_decompose(self, node, head):
        """Decompose tree into heavy-light chains."""
        self.head[node] = self.pos[node] = len(self.chain)
        self.chain.append(self.values[node])
        
        # Recurse on heavy edge first
        if self.heavy[node] != -1:
            self._dfs_decompose(self.heavy[node], head)
        
        # Recurse on light edges
        for child in self.tree[node]:
            if child != self.parent[node] and child != self.heavy[node]:
                self._dfs_decompose(child, child)
    
    def query_path(self, u, v):
        """Query sum on path from u to v (example operation)."""
        result = 0
        
        while self.head[u] != self.head[v]:
            # Ensure v is deeper
            if self.depth[self.head[u]] > self.depth[self.head[v]]:
                u, v = v, u
            
            # Query chain from head of v to v
            result += self._query_chain(self.head[v], self.pos[v])
            v = self.parent[self.head[v]]
        
        # Query remaining path on same chain
        if self.depth[u] > self.depth[v]:
            u, v = v, u
        result += self._query_chain(self.pos[u], self.pos[v])
        
        return result
    
    def _query_chain(self, l, r):
        """Query a range in the chain (can be optimized with segment tree)."""
        return sum(self.chain[l:r+1])
```

**Why**: Enables O(log^2 N) path queries on trees, crucial for problems requiring frequent updates and queries on tree paths.

## 4. Kosaraju's Algorithm

```python
# PROBLEM: Find all strongly connected components using two-pass DFS
# TIME: O(V + E) SPACE: O(V)
def kosaraju_scc(graph, vertices):
    """
    Kosaraju's algorithm for finding SCCs.
    Uses two DFS passes: one on original graph, one on reversed.
    """
    visited = set()
    finish_stack = []
    
    # First pass: record finish times
    def dfs1(node):
        visited.add(node)
        for neighbor in graph.get(node, []):
            if neighbor not in visited:
                dfs1(neighbor)
        finish_stack.append(node)
    
    # Run DFS on all vertices
    for v in vertices:
        if v not in visited:
            dfs1(v)
    
    # Reverse the graph
    reversed_graph = {}
    for node in graph:
        for neighbor in graph[node]:
            if neighbor not in reversed_graph:
                reversed_graph[neighbor] = []
            reversed_graph[neighbor].append(node)
    
    # Second pass: process in reverse finish order
    visited.clear()
    sccs = []
    
    def dfs2(node, component):
        visited.add(node)
        component.append(node)
        for neighbor in reversed_graph.get(node, []):
            if neighbor not in visited:
                dfs2(neighbor, component)
    
    while finish_stack:
        node = finish_stack.pop()
        if node not in visited:
            component = []
            dfs2(node, component)
            sccs.append(component)
    
    return sccs
```

**Why**: Intuitive two-pass approach that's easier to understand than Tarjan's, useful for educational purposes and when reversed graph is needed.

## 5. Splay Tree

```python
# PROBLEM: Self-balancing BST with amortized O(log N) operations
# TIME: O(log N) amortized SPACE: O(N)
class SplayNode:
    def __init__(self, key, value=None):
        self.key = key
        self.value = value
        self.left = None
        self.right = None
        self.parent = None

class SplayTree:
    def __init__(self):
        """Initialize empty splay tree."""
        self.root = None
    
    def _rotate_left(self, x):
        """Left rotation."""
        y = x.right
        x.right = y.left
        if y.left:
            y.left.parent = x
        y.parent = x.parent
        if not x.parent:
            self.root = y
        elif x == x.parent.left:
            x.parent.left = y
        else:
            x.parent.right = y
        y.left = x
        x.parent = y
    
    def _rotate_right(self, x):
        """Right rotation."""
        y = x.left
        x.left = y.right
        if y.right:
            y.right.parent = x
        y.parent = x.parent
        if not x.parent:
            self.root = y
        elif x == x.parent.right:
            x.parent.right = y
        else:
            x.parent.left = y
        y.right = x
        x.parent = y
    
    def _splay(self, x):
        """Splay node x to root using zig-zig, zig-zag, or zig operations."""
        while x.parent:
            if not x.parent.parent:
                # Zig operation
                if x == x.parent.left:
                    self._rotate_right(x.parent)
                else:
                    self._rotate_left(x.parent)
            elif x == x.parent.left and x.parent == x.parent.parent.left:
                # Zig-zig
                self._rotate_right(x.parent.parent)
                self._rotate_right(x.parent)
            elif x == x.parent.right and x.parent == x.parent.parent.right:
                # Zig-zig
                self._rotate_left(x.parent.parent)
                self._rotate_left(x.parent)
            elif x == x.parent.right and x.parent == x.parent.parent.left:
                # Zig-zag
                self._rotate_left(x.parent)
                self._rotate_right(x.parent)
            else:
                # Zig-zag
                self._rotate_right(x.parent)
                self._rotate_left(x.parent)
    
    def insert(self, key, value=None):
        """Insert key-value pair and splay to root."""
        node = SplayNode(key, value)
        if not self.root:
            self.root = node
            return
        
        current = self.root
        while current:
            if key < current.key:
                if not current.left:
                    current.left = node
                    node.parent = current
                    break
                current = current.left
            elif key > current.key:
                if not current.right:
                    current.right = node
                    node.parent = current
                    break
                current = current.right
            else:
                current.value = value
                self._splay(current)
                return
        
        self._splay(node)
    
    def search(self, key):
        """Search for key and splay it to root if found."""
        current = self.root
        while current:
            if key < current.key:
                current = current.left
            elif key > current.key:
                current = current.right
            else:
                self._splay(current)
                return current.value
        return None
```

**Why**: Self-adjusting tree that brings recently accessed elements to root, excellent for caches and scenarios with locality of reference.

## 6. Red-Black Tree

```python
# PROBLEM: Balanced BST guaranteeing O(log N) operations
# TIME: O(log N) SPACE: O(N)
class RBNode:
    def __init__(self, key, value=None, color='RED'):
        self.key = key
        self.value = value
        self.color = color  # RED or BLACK
        self.left = None
        self.right = None
        self.parent = None

class RedBlackTree:
    def __init__(self):
        """Initialize empty red-black tree with NIL sentinel."""
        self.NIL = RBNode(None, color='BLACK')
        self.root = self.NIL
    
    def _rotate_left(self, x):
        """Left rotation around node x."""
        y = x.right
        x.right = y.left
        if y.left != self.NIL:
            y.left.parent = x
        y.parent = x.parent
        if x.parent == self.NIL:
            self.root = y
        elif x == x.parent.left:
            x.parent.left = y
        else:
            x.parent.right = y
        y.left = x
        x.parent = y
    
    def _rotate_right(self, x):
        """Right rotation around node x."""
        y = x.left
        x.left = y.right
        if y.right != self.NIL:
            y.right.parent = x
        y.parent = x.parent
        if x.parent == self.NIL:
            self.root = y
        elif x == x.parent.right:
            x.parent.right = y
        else:
            x.parent.left = y
        y.right = x
        x.parent = y
    
    def _fix_insert(self, k):
        """Fix red-black properties after insertion."""
        while k.parent.color == 'RED':
            if k.parent == k.parent.parent.right:
                u = k.parent.parent.left
                if u.color == 'RED':
                    u.color = 'BLACK'
                    k.parent.color = 'BLACK'
                    k.parent.parent.color = 'RED'
                    k = k.parent.parent
                else:
                    if k == k.parent.left:
                        k = k.parent
                        self._rotate_right(k)
                    k.parent.color = 'BLACK'
                    k.parent.parent.color = 'RED'
                    self._rotate_left(k.parent.parent)
            else:
                u = k.parent.parent.right
                if u.color == 'RED':
                    u.color = 'BLACK'
                    k.parent.color = 'BLACK'
                    k.parent.parent.color = 'RED'
                    k = k.parent.parent
                else:
                    if k == k.parent.right:
                        k = k.parent
                        self._rotate_left(k)
                    k.parent.color = 'BLACK'
                    k.parent.parent.color = 'RED'
                    self._rotate_right(k.parent.parent)
            if k == self.root:
                break
        self.root.color = 'BLACK'
    
    def insert(self, key, value=None):
        """Insert key-value pair maintaining red-black properties."""
        node = RBNode(key, value)
        node.left = self.NIL
        node.right = self.NIL
        
        y = self.NIL
        x = self.root
        
        while x != self.NIL:
            y = x
            if node.key < x.key:
                x = x.left
            else:
                x = x.right
        
        node.parent = y
        if y == self.NIL:
            self.root = node
        elif node.key < y.key:
            y.left = node
        else:
            y.right = node
        
        self._fix_insert(node)
    
    def search(self, key):
        """Search for key in tree."""
        node = self.root
        while node != self.NIL:
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                return node.value
        return None
```

**Why**: Guaranteed balanced tree with worst-case O(log N) operations, widely used in standard libraries (e.g., C++ std::map).

## 7. Skip List

```python
# PROBLEM: Probabilistic data structure for O(log N) average search/insert
# TIME: O(log N) average, O(N) worst SPACE: O(N)
import random

class SkipListNode:
    def __init__(self, key, value=None, level=0):
        self.key = key
        self.value = value
        self.forward = [None] * (level + 1)

class SkipList:
    def __init__(self, max_level=16, p=0.5):
        """
        Initialize skip list with given max level and probability.
        max_level: maximum height of the skip list
        p: probability of promoting to next level
        """
        self.max_level = max_level
        self.p = p
        self.level = 0
        self.head = SkipListNode(None, None, max_level)
    
    def _random_level(self):
        """Generate random level for new node."""
        level = 0
        while random.random() < self.p and level < self.max_level - 1:
            level += 1
        return level
    
    def insert(self, key, value=None):
        """Insert key-value pair into skip list."""
        update = [None] * (self.max_level)
        current = self.head
        
        # Find position for each level
        for i in range(self.level, -1, -1):
            while current.forward[i] and current.forward[i].key < key:
                current = current.forward[i]
            update[i] = current
        
        current = current.forward[0]
        
        # Update if key exists
        if current and current.key == key:
            current.value = value
            return
        
        # Insert new node
        new_level = self._random_level()
        if new_level > self.level:
            for i in range(self.level + 1, new_level + 1):
                update[i] = self.head
            self.level = new_level
        
        new_node = SkipListNode(key, value, new_level)
        
        for i in range(new_level + 1):
            new_node.forward[i] = update[i].forward[i]
            update[i].forward[i] = new_node
    
    def search(self, key):
        """Search for key in skip list."""
        current = self.head
        
        for i in range(self.level, -1, -1):
            while current.forward[i] and current.forward[i].key < key:
                current = current.forward[i]
        
        current = current.forward[0]
        
        if current and current.key == key:
            return current.value
        return None
    
    def delete(self, key):
        """Delete key from skip list."""
        update = [None] * (self.max_level)
        current = self.head
        
        for i in range(self.level, -1, -1):
            while current.forward[i] and current.forward[i].key < key:
                current = current.forward[i]
            update[i] = current
        
        current = current.forward[0]
        
        if not current or current.key != key:
            return False
        
        # Remove node from all levels
        for i in range(self.level + 1):
            if update[i].forward[i] != current:
                break
            update[i].forward[i] = current.forward[i]
        
        # Adjust level
        while self.level > 0 and not self.head.forward[self.level]:
            self.level -= 1
        
        return True
```

**Why**: Simple to implement alternative to balanced trees, used in Redis and LevelDB for ordered data structures.

## 8. Paxos Algorithm

```python
# PROBLEM: Distributed consensus algorithm for fault-tolerant systems
# TIME: O(N) messages per consensus SPACE: O(N) for state
import threading
import time
from collections import defaultdict

class PaxosProposer:
    def __init__(self, node_id, acceptors):
        """
        Initialize Paxos proposer.
        node_id: unique identifier for this proposer
        acceptors: list of acceptor IDs
        """
        self.node_id = node_id
        self.acceptors = acceptors
        self.proposal_number = 0
        self.highest_proposal = 0
        self.highest_value = None
    
    def prepare(self, value):
        """Phase 1a: Send prepare request to acceptors."""
        self.proposal_number += 1
        proposal_id = (self.proposal_number, self.node_id)
        
        promises = []
        for acceptor in self.acceptors:
            # Simulate sending prepare request
            promise = self._send_prepare(acceptor, proposal_id)
            if promise:
                promises.append(promise)
        
        # Check if we have majority
        if len(promises) > len(self.acceptors) // 2:
            return self._accept_phase(proposal_id, value, promises)
        return None
    
    def _send_prepare(self, acceptor, proposal_id):
        """Send prepare request to single acceptor."""
        # In real implementation, this would be a network call
        # Simulated response
        return {'acceptor': acceptor, 'promise': True}
    
    def _accept_phase(self, proposal_id, value, promises):
        """Phase 2: Send accept request."""
        # Use highest proposal value from promises, or our value
        for promise in promises:
            if 'accepted_value' in promise:
                if promise['accepted_proposal'] > self.highest_proposal:
                    self.highest_proposal = promise['accepted_proposal']
                    self.highest_value = promise['accepted_value']
        
        chosen_value = self.highest_value if self.highest_value else value
        
        accepts = 0
        for acceptor in self.acceptors:
            if self._send_accept(acceptor, proposal_id, chosen_value):
                accepts += 1
        
        if accepts > len(self.acceptors) // 2:
            return chosen_value
        return None
    
    def _send_accept(self, acceptor, proposal_id, value):
        """Send accept request to single acceptor."""
        return True

class PaxosAcceptor:
    def __init__(self, node_id):
        """
        Initialize Paxos acceptor.
        node_id: unique identifier for this acceptor
        """
        self.node_id = node_id
        self.promised_proposal = 0
        self.accepted_proposal = 0
        self.accepted_value = None
        self.lock = threading.Lock()
    
    def handle_prepare(self, proposal_id):
        """Handle prepare request (Phase 1b)."""
        with self.lock:
            if proposal_id > self.promised_proposal:
                self.promised_proposal = proposal_id
                return {
                    'promise': True,
                    'accepted_proposal': self.accepted_proposal,
                    'accepted_value': self.accepted_value
                }
            return {'promise': False}
    
    def handle_accept(self, proposal_id, value):
        """Handle accept request (Phase 2b)."""
        with self.lock:
            if proposal_id >= self.promised_proposal:
                self.promised_proposal = proposal_id
                self.accepted_proposal = proposal_id
                self.accepted_value = value
                return True
            return False
```

**Why**: Foundation for distributed consensus, used in distributed databases and coordination services like Chubby and etcd.

## 9. Write-Ahead Logging (WAL)

```python
# PROBLEM: Ensure durability and atomicity in database systems
# TIME: O(1) for append, O(N) for recovery SPACE: O(N) for log
import json
import os
from datetime import datetime
import threading

class WriteAheadLog:
    def __init__(self, log_file='wal.log'):
        """
        Initialize write-ahead log.
        log_file: path to WAL file
        """
        self.log_file = log_file
        self.lock = threading.Lock()
        self.current_lsn = 0
        
        # Create log file if not exists
        if not os.path.exists(log_file):
            open(log_file, 'w').close()
    
    def _get_next_lsn(self):
        """Get next log sequence number."""
        self.current_lsn += 1
        return self.current_lsn
    
    def log_insert(self, table, key, value):
        """Log an insert operation."""
        log_entry = {
            'lsn': self._get_next_lsn(),
            'timestamp': datetime.now().isoformat(),
            'operation': 'INSERT',
            'table': table,
            'key': key,
            'value': value
        }
        self._write_log(log_entry)
        return log_entry['lsn']
    
    def log_update(self, table, key, old_value, new_value):
        """Log an update operation."""
        log_entry = {
            'lsn': self._get_next_lsn(),
            'timestamp': datetime.now().isoformat(),
            'operation': 'UPDATE',
            'table': table,
            'key': key,
            'old_value': old_value,
            'new_value': new_value
        }
        self._write_log(log_entry)
        return log_entry['lsn']
    
    def log_delete(self, table, key, old_value):
        """Log a delete operation."""
        log_entry = {
            'lsn': self._get_next_lsn(),
            'timestamp': datetime.now().isoformat(),
            'operation': 'DELETE',
            'table': table,
            'key': key,
            'old_value': old_value
        }
        self._write_log(log_entry)
        return log_entry['lsn']
    
    def log_commit(self, transaction_id):
        """Log transaction commit."""
        log_entry = {
            'lsn': self._get_next_lsn(),
            'timestamp': datetime.now().isoformat(),
            'operation': 'COMMIT',
            'transaction_id': transaction_id
        }
        self._write_log(log_entry)
        return log_entry['lsn']
    
    def log_checkpoint(self, active_transactions):
        """Log checkpoint for recovery."""
        log_entry = {
            'lsn': self._get_next_lsn(),
            'timestamp': datetime.now().isoformat(),
            'operation': 'CHECKPOINT',
            'active_transactions': active_transactions
        }
        self._write_log(log_entry)
        return log_entry['lsn']
    
    def _write_log(self, log_entry):
        """Write log entry to disk."""
        with self.lock:
            with open(self.log_file, 'a') as f:
```
