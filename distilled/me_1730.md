## Intro Sort

```python
# PROBLEM: Hybrid sorting algorithm combining quicksort, heapsort, and insertion sort
# TIME: O(n log n) average, O(n log n) worst case SPACE: O(log n) for recursion

import math

def intro_sort(arr):
    """
    Introspective sort: starts with quicksort, switches to heapsort
    when recursion depth exceeds 2*log(n), uses insertion sort for small arrays.
    """
    if not arr or len(arr) <= 1:
        return arr
    
    max_depth = 2 * int(math.log2(len(arr))) if len(arr) > 1 else 0
    _intro_sort_helper(arr, 0, len(arr) - 1, max_depth)
    return arr

def _intro_sort_helper(arr, start, end, max_depth):
    """Recursive helper that chooses the appropriate sorting strategy."""
    size = end - start + 1
    
    # Use insertion sort for small arrays (more efficient)
    if size < 16:
        _insertion_sort(arr, start, end)
        return
    
    # Switch to heapsort if recursion depth is too deep
    if max_depth == 0:
        _heap_sort_partial(arr, start, end)
        return
    
    # Standard quicksort partition
    pivot = _partition(arr, start, end)
    _intro_sort_helper(arr, start, pivot - 1, max_depth - 1)
    _intro_sort_helper(arr, pivot + 1, end, max_depth - 1)

def _partition(arr, start, end):
    """Lomuto partition scheme with median-of-three pivot selection."""
    mid = (start + end) // 2
    # Median of three
    if arr[start] > arr[mid]:
        arr[start], arr[mid] = arr[mid], arr[start]
    if arr[mid] > arr[end]:
        arr[mid], arr[end] = arr[end], arr[mid]
    if arr[start] > arr[mid]:
        arr[start], arr[mid] = arr[mid], arr[start]
    
    # Place pivot at end-1
    arr[mid], arr[end - 1] = arr[end - 1], arr[mid]
    pivot = arr[end - 1]
    
    i = start
    for j in range(start, end - 1):
        if arr[j] <= pivot:
            arr[i], arr[j] = arr[j], arr[i]
            i += 1
    
    arr[i], arr[end - 1] = arr[end - 1], arr[i]
    return i

def _insertion_sort(arr, start, end):
    """Insertion sort for small subarrays."""
    for i in range(start + 1, end + 1):
        key = arr[i]
        j = i - 1
        while j >= start and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key

def _heap_sort_partial(arr, start, end):
    """Heapsort for the given range."""
    def heapify(root, heap_end):
        """Maintain heap property from root down."""
        while True:
            left = 2 * (root - start) + 1 + start
            right = left + 1
            largest = root
            
            if left <= heap_end and arr[left] > arr[largest]:
                largest = left
            if right <= heap_end and arr[right] > arr[largest]:
                largest = right
            
            if largest != root:
                arr[root], arr[largest] = arr[largest], arr[root]
                root = largest
            else:
                break
    
    # Build max heap
    size = end - start + 1
    for i in range(start + size // 2 - 1, start - 1, -1):
        heapify(i, end)
    
    # Extract elements from heap
    for i in range(end, start, -1):
        arr[start], arr[i] = arr[i], arr[start]
        heapify(start, i - 1)
```

**Why**: Intro sort is Python's built-in sorting algorithm. It combines the best of three sorting algorithms - quicksort's speed, heapsort's worst-case guarantee, and insertion sort's efficiency on small arrays. This hybrid approach ensures O(n log n) worst-case while maintaining practical performance.

## Huffman Coding

```python
# PROBLEM: Lossless data compression using variable-length prefix codes
# TIME: O(n log n) for encoding, O(n) for decoding SPACE: O(n) for tree

import heapq
from collections import Counter
from dataclasses import dataclass, field
from typing import Optional, Dict

@dataclass(order=True)
class HuffmanNode:
    """Node in the Huffman tree."""
    freq: int
    char: Optional[str] = field(default=None, compare=False)
    left: Optional['HuffmanNode'] = field(default=None, compare=False)
    right: Optional['HuffmanNode'] = field(default=None, compare=False)

class HuffmanCoding:
    """Huffman coding implementation for text compression."""
    
    def __init__(self):
        self.root: Optional[HuffmanNode] = None
        self.codes: Dict[str, str] = {}
    
    def build_tree(self, text: str) -> None:
        """Build Huffman tree from input text."""
        if not text:
            return
        
        # Count character frequencies
        freq_map = Counter(text)
        
        # Create leaf nodes and push to min-heap
        heap = [HuffmanNode(freq=count, char=char) 
                for char, count in freq_map.items()]
        heapq.heapify(heap)
        
        # Build tree by combining nodes
        while len(heap) > 1:
            left = heapq.heappop(heap)
            right = heapq.heappop(heap)
            merged = HuffmanNode(freq=left.freq + right.freq, 
                                left=left, right=right)
            heapq.heappush(heap, merged)
        
        self.root = heap[0] if heap else None
        self._generate_codes(self.root, "")
    
    def _generate_codes(self, node: Optional[HuffmanNode], code: str) -> None:
        """Recursively generate codes for each character."""
        if node is None:
            return
        
        # Leaf node: store the code
        if node.char is not None:
            self.codes[node.char] = code if code else "0"
            return
        
        # Internal node: traverse left (0) and right (1)
        self._generate_codes(node.left, code + "0")
        self._generate_codes(node.right, code + "1")
    
    def encode(self, text: str) -> str:
        """Encode text using Huffman codes."""
        if not self.codes:
            self.build_tree(text)
        
        return "".join(self.codes[char] for char in text)
    
    def decode(self, encoded: str) -> str:
        """Decode binary string back to original text."""
        if not self.root:
            return ""
        
        result = []
        current = self.root
        
        for bit in encoded:
            # Traverse left for 0, right for 1
            current = current.left if bit == "0" else current.right
            
            # Reached leaf node
            if current.char is not None:
                result.append(current.char)
                current = self.root
        
        return "".join(result)
    
    def get_compression_ratio(self, text: str) -> float:
        """Calculate compression ratio (original bits / encoded bits)."""
        if not text:
            return 0.0
        
        encoded = self.encode(text)
        original_bits = len(text) * 8  # Assuming 8-bit characters
        compressed_bits = len(encoded)
        
        return original_bits / compressed_bits if compressed_bits > 0 else 0.0
```

**Why**: Huffman coding is foundational to compression algorithms used in ZIP, JPEG, and MP3. It demonstrates greedy algorithm design and optimal prefix code generation. Understanding it helps grasp entropy-based compression and variable-length encoding.

## Treap

```python
# PROBLEM: Binary search tree with randomized priorities for balanced operations
# TIME: O(log n) expected for all operations SPACE: O(n)

import random
from typing import Optional, Tuple

class TreapNode:
    """Node in a Treap (Tree + Heap)."""
    def __init__(self, key, priority=None):
        self.key = key
        self.priority = priority if priority is not None else random.random()
        self.left: Optional['TreapNode'] = None
        self.right: Optional['TreapNode'] = None

class Treap:
    """
    Treap: Combines BST property (by key) with heap property (by priority).
    Random priorities ensure expected O(log n) operations.
    """
    
    def __init__(self):
        self.root: Optional[TreapNode] = None
    
    def rotate_right(self, node: TreapNode) -> TreapNode:
        """Right rotation to maintain heap property."""
        new_root = node.left
        node.left = new_root.right
        new_root.right = node
        return new_root
    
    def rotate_left(self, node: TreapNode) -> TreapNode:
        """Left rotation to maintain heap property."""
        new_root = node.right
        node.right = new_root.left
        new_root.left = node
        return new_root
    
    def insert(self, key) -> None:
        """Insert key into treap."""
        self.root = self._insert(self.root, key)
    
    def _insert(self, node: Optional[TreapNode], key) -> TreapNode:
        """Recursive insert with rotation to maintain heap property."""
        if node is None:
            return TreapNode(key)
        
        if key < node.key:
            node.left = self._insert(node.left, key)
            # Rotate if child has higher priority (heap property)
            if node.left.priority > node.priority:
                node = self.rotate_right(node)
        elif key > node.key:
            node.right = self._insert(node.right, key)
            if node.right.priority > node.priority:
                node = self.rotate_left(node)
        # If key == node.key, do nothing (no duplicates)
        
        return node
    
    def delete(self, key) -> None:
        """Delete key from treap."""
        self.root = self._delete(self.root, key)
    
    def _delete(self, node: Optional[TreapNode], key) -> Optional[TreapNode]:
        """Recursive delete using rotation to push node to leaf."""
        if node is None:
            return None
        
        if key < node.key:
            node.left = self._delete(node.left, key)
        elif key > node.key:
            node.right = self._delete(node.right, key)
        else:
            # Found the node to delete
            if node.left is None and node.right is None:
                return None  # Leaf: simply remove
            
            # Rotate with the child that has higher priority
            if node.left is None:
                node = self.rotate_left(node)
                node.left = self._delete(node.left, key)
            elif node.right is None:
                node = self.rotate_right(node)
                node.right = self._delete(node.right, key)
            elif node.left.priority > node.right.priority:
                node = self.rotate_right(node)
                node.right = self._delete(node.right, key)
            else:
                node = self.rotate_left(node)
                node.left = self._delete(node.left, key)
        
        return node
    
    def search(self, key) -> bool:
        """Check if key exists in treap."""
        return self._search(self.root, key)
    
    def _search(self, node: Optional[TreapNode], key) -> bool:
        """Standard BST search."""
        if node is None:
            return False
        if key == node.key:
            return True
        if key < node.key:
            return self._search(node.left, key)
        return self._search(node.right, key)
    
    def split(self, key) -> Tuple['Treap', 'Treap']:
        """Split treap into two: keys < key and keys >= key."""
        left_treap = Treap()
        right_treap = Treap()
        left_treap.root, right_treap.root = self._split(self.root, key)
        return left_treap, right_treap
    
    def _split(self, node: Optional[TreapNode], key) -> Tuple[Optional[TreapNode], Optional[TreapNode]]:
        """Split operation: returns (left_tree, right_tree)."""
        if node is None:
            return None, None
        
        if key <= node.key:
            left, right = self._split(node.left, key)
            node.left = right
            return left, node
        else:
            left, right = self._split(node.right, key)
            node.right = left
            return node, right
```

**Why**: Treaps provide expected O(log n) operations without complex rebalancing logic. Randomization guarantees balance probabilistically. Used in competitive programming and scenarios requiring split/merge operations that AVL or Red-Black trees don't support efficiently.

## Cuckoo Hashing

```python
# PROBLEM: Hash table with O(1) worst-case lookup using multiple hash functions
# TIME: O(1) worst-case lookup, O(1) amortized insert SPACE: O(n)

from typing import Optional, List, Callable
import hashlib

class CuckooHashTable:
    """
    Cuckoo Hashing: Uses two hash tables and can kick out existing entries
    to make room for new ones, like a cuckoo bird pushing eggs from a nest.
    """
    
    def __init__(self, size: int = 100, max_kicks: int = 500):
        self.size = size
        self.max_kicks = max_kicks  # Max displacement attempts before resize
        self.table1: List[Optional[tuple]] = [None] * size
        self.table2: List[Optional[tuple]] = [None] * size
        self.count = 0
        self.load_factor_threshold = 0.5
    
    def _hash1(self, key) -> int:
        """First hash function."""
        key_str = str(key).encode()
        return int(hashlib.md5(key_str).hexdigest(), 16) % self.size
    
    def _hash2(self, key) -> int:
        """Second hash function - must differ from first."""
        key_str = str(key).encode()
        return int(hashlib.sha1(key_str).hexdigest(), 16) % self.size
    
    def lookup(self, key) -> bool:
        """O(1) worst-case lookup - check both possible positions."""
        h1, h2 = self._hash1(key), self._hash2(key)
        
        if self.table1[h1] is not None and self.table1[h1][0] == key:
            return True
        if self.table2[h2] is not None and self.table2[h2][0] == key:
            return True
        return False
    
    def get(self, key):
        """Get value associated with key."""
        h1, h2 = self._hash1(key), self._hash2(key)
        
        if self.table1[h1] is not None and self.table1[h1][0] == key:
            return self.table1[h1][1]
        if self.table2[h2] is not None and self.table2[h2][0] == key:
            return self.table2[h2][1]
        return None
    
    def insert(self, key, value) -> bool:
        """Insert with cuckoo displacement if needed."""
        # Check if key already exists
        h1, h2 = self._hash1(key), self._hash2(key)
        
        if self.table1[h1] is not None and self.table1[h1][0] == key:
            self.table1[h1] = (key, value)
            return True
        if self.table2[h2] is not None and self.table2[h2][0] == key:
            self.table2[h2] = (key, value)
            return True
        
        # Try to insert into first table
        current_key, current_value = key, value
        
        for _ in range(self.max_kicks):
            h1 = self._hash1(current_key)
            
            if self.table1[h1] is None:
                self.table1[h1] = (current_key, current_value)
                self.count += 1
                return True
            
            # Kick out existing entry
            self.table1[h1], (current_key, current_value) = \
                (current_key, current_value), self.table1[h1]
            
            # Try second table
            h2 = self._hash2(current_key)
            if self.table2[h2] is None:
                self.table2[h2] = (current_key, current_value)
                self.count += 1
                return True
            
            # Kick out from second table
            self.table2[h2], (current_key, current_value) = \
                (current_key, current_value), self.table2[h2]
        
        # Failed after max kicks - need to resize
        self._resize()
        return self.insert(current_key, current_value)
    
    def delete(self, key) -> bool:
        """Delete key from either table."""
        h1, h2 = self._hash1(key), self._hash2(key)
        
        if self.table1[h1] is not None and self.table1[h1][0] == key:
            self.table1[h1] = None
            self.count -= 1
            return True
        if self.table2[h2] is not None and self.table2[h2][0] == key:
            self.table2[h2] = None
            self.count -= 1
            return True
        return False
    
    def _resize(self) -> None:
        """Double table size and rehash all entries."""
        old_table1 = self.table1
        old_table2 = self.table2
        
        self.size *= 2
        self.table1 = [None] * self.size
        self.table2 = [None] * self.size
        self.count = 0
        
        for entry in old_table1 + old_table2:
            if entry is not None:
                self.insert(entry[0], entry[1])
```

**Why**: Cuckoo hashing provides O(1) worst-case lookup, unlike chaining or linear probing. Important for real-time systems and databases where predictable lookup time matters. The concept of multiple hash functions and displacement appears in bloom filters and other probabilistic data structures.

## Composite Pattern

```python
# PROBLEM: Compose objects into tree structures to represent part-whole hierarchies
# TIME: O(n) for traversal operations SPACE: O(n)

from abc import ABC, abstractmethod
from typing import List, Optional

class FileSystemComponent(ABC):
    """Abstract base class for files and directories."""
    
    def __init__(self, name: str):
        self.name = name
    
    @abstractmethod
    def get_size(self) -> int:
        """Return size in bytes."""
        pass
    
    @abstractmethod
    def display(self, indent: str = "") -> None:
        """Display the component."""
        pass

class File(FileSystemComponent):
    """Leaf node: represents a single file."""
    
    def __init__(self, name: str, size: int):
        super().__init__(name)
        self._size = size
    
    def get_size(self) -> int:
        return self._size
    
    def display(self, indent: str = "") -> None:
        print(f"{indent}📄 {self.name} ({self._size} bytes)")

class Directory(FileSystemComponent):
    """Composite node: can contain files and other directories."""
    
    def __init__(self, name: str):
        super().__init__(name)
        self._children: List[FileSystemComponent] = []
    
    def add(self, component: FileSystemComponent) -> None:
        """Add a child component."""
        self._children.append(component)
    
    def remove(self, component: FileSystemComponent) -> None:
        """Remove a child component."""
        self._children.remove(component)
    
    def get_child(self, index: int) -> Optional[FileSystemComponent]:
        """Get child by index."""
        if 0 <= index < len(self._children):
            return self._children[index]
        return None
    
    def get_size(self) -> int:
        """Recursively calculate total size."""
        return sum(child.get_size() for child in self._children)
    
    def display(self, indent: str = "") -> None:
        """Display directory and all its contents."""
        print(f"{indent}📁 {self.name}/ ({self.get_size()} bytes total)")
        for child in self._children:
            child.display(indent + "  ")
    
    def find_by_name(self, name: str) -> Optional[FileSystemComponent]:
        """Search for component by name recursively."""
        if self.name == name:
            return self
        
        for child in self._children:
            if child.name == name:
                return child
            if isinstance(child, Directory):
                result = child.find_by_name(name)
                if result:
                    return result
        return None

# Usage example
if __name__ == "__main__":
    # Build file system structure
    root = Directory("root")
    
    docs = Directory("documents")
    docs.add(File("report.pdf", 2_500_000))
    docs.add(File("notes.txt", 15_000))
    
    photos = Directory("photos")
    photos.add(File("vacation.jpg", 3_200_000))
    photos.add(File("family.png", 1_800_000))
    
    root.add(docs)
    root.add(photos)
    root.add(File("readme.md", 5_000))
    
    # Display tree structure
    root.display()
    # 📁 root/ (7_520_000 bytes total)
    #   📁 documents/ (2_515_000 bytes total)
    #     📄 report.pdf (2_500_000 bytes)
    #     📄 notes.txt (15_000 bytes)
    #   📁 photos/ (5_000_000 bytes total)
    #     📄 vacation.jpg (3_200_000 bytes)
    #     📄 family.png (1_800_000 bytes)
    #   📄 readme.md (5_000 bytes)
```

**Why**: Composite pattern lets clients treat individual objects and compositions uniformly. Essential for UI frameworks (DOM trees), file systems, organizational hierarchies, and any tree-structured data. Simplifies client code by eliminating type checks.

## Bridge Pattern

```python
# PROBLEM: Decouple abstraction from implementation so both can vary independently
# TIME: O(1) for operations SPACE: O(n)

from abc import ABC, abstractmethod

# Implementation interface
class Renderer(ABC):
    """Abstraction for rendering shapes - the implementation side."""
    
    @abstractmethod
    def render_circle(self, radius: float) -> str:
        pass
    
    @abstractmethod
    def render_rectangle(self, width: float, height: float) -> str:
        pass

# Concrete implementations
class VectorRenderer(Renderer):
    """Renders shapes as vector graphics."""
    
    def render_circle(self, radius: float) -> str:
        return f"Drawing a circle of radius {radius} (vector)"
    
    def render_rectangle(self, width: float, height: float) -> str:
        return f"Drawing a {width}x{height} rectangle (vector)"

class RasterRenderer(Renderer):
    """Renders shapes as raster/pixel graphics."""
    
    def render_circle(self, radius: float) -> str:
        return f"Drawing a circle of radius {radius} (raster pixels)"
    
    def render_rectangle(self, width: float, height: float) -> str:
        return f"Drawing a {width}x{height} rectangle (raster pixels)"

# Abstraction
class Shape:
    """Abstract shape that uses a renderer - the abstraction side."""
    
    def __init__(self, renderer: Renderer):
        self.renderer = renderer
    
    @abstractmethod
    def draw(self) -> str:
        pass

# Refined abstractions
class Circle(Shape):
    """Circle shape that delegates rendering to the renderer."""
    
    def __init__(self, renderer: Renderer, radius: float):
        super().__init__(renderer)
        self.radius = radius
    
    def draw(self) -> str:
        return self.renderer.render_circle(self.radius)

class Rectangle(Shape):
    """Rectangle shape that delegates rendering to the renderer."""
    
    def __init__(self, renderer: Renderer, width: float, height: float):
        super().__init__(renderer)
        self.width = width
        self.height = height
    
    def draw(self) -> str:
        return self.renderer.render_rectangle(self.width, self.height)

# Usage
if __name__ == "__main__":
    # Same shapes can be rendered differently
    vector_renderer = VectorRenderer()
    raster_renderer = RasterRenderer()
    
    # Vector circle
    vector_circle = Circle(vector_renderer, 5.0)
    print(vector_circle.draw())
    # "Drawing a circle of radius 5.0 (vector)"
    
    # Raster circle (same shape, different rendering)
    raster_circle = Circle(raster_renderer, 5.0)
    print(raster_circle.draw())
    # "Drawing a circle of radius 5.0 (raster pixels)"
    
    # Can add new shapes without changing renderers
    # Can add new renderers without changing shapes
```

**Why**: Bridge pattern prevents a permanent binding between abstraction and implementation. Critical when both hierarchies evolve independently - adding new shapes doesn't require modifying renderers and vice versa. Used in GUI frameworks, database drivers, and device drivers.

## Bulkhead Isolation

```python
# PROBLEM: Isolate components to prevent cascading failures in distributed systems
# TIME: O(1) for task submission SPACE: O(pool_size)

import threading
import queue
from typing import Callable, Any, Optional
from dataclasses import dataclass
from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"      # Normal operation
    OPEN = "open"          # Failing, reject requests
    HALF_OPEN = "half_open"  # Testing if recovered

@dataclass
class TaskResult:
    """Result of a submitted task."""
    success: bool
    value: Any = None
    error: Optional[Exception] = None

class Bulkhead:
    """
    Bulkhead pattern: limits concurrent executions to prevent resource exhaustion.
    Like ship compartments - if one floods, others remain functional.
    """
    
    def __init__(self, name: str, max_concurrent: int = 10, max_queue: int = 100):
        self.name = name
        self.max_concurrent = max_concurrent
        self.semaphore = threading.Semaphore(max_concurrent)
        self.task_queue = queue.Queue(maxsize=max_queue)
        self.active_count = 0
        self.rejected_count = 0
        self._lock = threading.Lock()
    
    def submit(self, task: Callable, *args, **kwargs) -> TaskResult:
        """
        Submit a task to the bulkhead.
        Returns immediately if capacity available, rejects if full.
        """
        # Try to acquire semaphore (non-blocking)
        if not self.semaphore.acquire(blocking=False):
            with self._lock:
                self.rejected_count += 1
            return TaskResult(
                success=False, 
                error=Exception(f"Bulkhead '{self.name}' at capacity")
            )
        
        try:
            with self._lock:
                self.active_count += 1
            
            result = task(*args, **kwargs)
            return TaskResult(success=True, value=result)
        
        except Exception as e:
            return TaskResult(success=False, error=e)
        
        finally:
            with self._lock:
                self.active_count -= 1
            self.semaphore.release()
    
    def submit_async(self, task: Callable, *args, **kwargs) -> TaskResult:
        """Submit task asynchronously via queue."""
        try:
            self.task_queue.put_nowait((task, args, kwargs))
            return TaskResult(success=True, value="Queued")
        except queue.Full:
            with self._lock:
                self.rejected_count += 1
            return TaskResult(
                success=False, 
                error=Exception(f"Bulkhead '{self.name}' queue full")
            )
    
    def get_stats(self) -> dict:
        """Get bulkhead statistics."""
        with self._lock:
            return {
                "name": self.name,
                "active": self.active_count,
                "max_concurrent": self.max_concurrent,
                "queue_size": self.task_queue.qsize(),
                "rejected": self.rejected_count
            }

class BulkheadManager:
    """Manages multiple bulkheads for different service pools."""
    
    def __init__(self):
        self.bulkheads: dict[str, Bulkhead] = {}
        self._lock = threading.Lock()
    
    def get_or_create(self, name: str, max_concurrent: int = 10) -> Bulkhead:
        """Get existing bulkhead or create new one."""
        with self._lock:
            if name not in self.bulkheads:
                self.bulkheads[name] = Bulkhead(name, max_concurrent)
            return self.bulkheads[name]
    
    def get_all_stats(self) -> list[dict]:
        """Get statistics for all bulkheads."""
        return [bh.get_stats() for bh in self.bulkheads.values()]

# Usage
if __name__ == "__main__":
    manager = BulkheadManager()
    
    # Separate bulkheads for different services
    db_bulkhead = manager.get_or_create("database", max_concurrent=5)
    api_bulkhead = manager.get_or_create("external_api", max_concurrent=10)
    
    def risky_database_query(query: str):
        # Simulate database operation
        return f"Result for: {query}"
    
    # Submit task to database bulkhead
    result = db_bulkhead.submit(risky_database_query, "SELECT * FROM users")
    print(result)
    # TaskResult(success=True, value='Result for: SELECT * FROM users')
```

**Why**: Bulkhead isolation prevents a single failing component from taking down the entire system. Essential in microservices where you want to limit the blast radius of failures. Combined with circuit breakers and rate limiters for resilience.

## Descriptors

```python
# PROBLEM: Create reusable attribute access logic without inheritance
# TIME: O(1) for get/set SPACE: O(1) per instance

from typing import Any, Optional, Callable

class ValidatedAttribute:
    """
    Descriptor that validates values on assignment.
    Demonstrates __get__, __set__, __delete__ protocol.
    """
    
    def __init__(self, name: str, validator: Callable[[Any], bool], 
                 default: Any = None, error_msg: str = "Invalid value"):
        self.name = name
        self.validator = validator
        self.default = default
        self.error_msg = error_msg
        # Storage key for instance dictionary
        self.private_name = f"_validated_{name}"
    
    def __get__(self, obj, objtype=None) -> Any:
        """Called when attribute is accessed."""
        if obj is None:
            # Accessed on class, return descriptor itself
            return self
        
        return getattr(obj, self.private_name, self.default)
    
    def __set__(self, obj, value) -> None:
        """Called when attribute is assigned."""
        if not self.validator(value):
            raise ValueError(f"{self.name}: {self.error_msg}, got {value}")
        setattr(obj, self.private_name, value)
    
    def __delete__(self, obj) -> None:
        """Called when attribute is deleted."""
        if hasattr(obj, self.private_name):
            delattr(obj, self.private_name)

class NonNegativeInt(ValidatedAttribute):
    """Descriptor for non-negative integers."""
    
    def __init__(self, name: str, default: int = 0):
        super().__init__(
            name=name,
            validator=lambda x: isinstance(x, int) and x >= 0,
            default=default,
            error_msg="Must be a non-negative integer"
        )

class StringMaxLength(ValidatedAttribute):
    """Descriptor for strings with max length."""
    
    def __init__(self, name: str, max_length: int, default: str = ""):
        super().__init__(
            name=name,
            validator=lambda x: isinstance(x, str) and len(x) <= max_length,
            default=default,
            error_msg=f"Must be a string with max length {max_length}"
        )

class LazyAttribute:
    """Descriptor that computes value on first access and caches it."""
    
    def __init__(self, func: Callable):
        self.func = func
        self.private_name = f"_lazy_{func.__name__}"
    
    def __get__(self, obj, objtype=None):
        if obj is None:
            return self
        
        if not hasattr(obj, self.private_name):
            # Compute and cache
            value = self.func(obj)
            setattr(obj, self.private_name, value)
        
        return getattr(obj, self.private_name)

# Usage
class Product:
    """Example class using descriptors."""
    
    price = NonNegativeInt("price", default=0)
    name = StringMaxLength("name", max_length=50, default="")
    
    def __init__(self, name: str, price: int):
        self.name = name
        self.price = price
    
    @LazyAttribute
    def expensive_label(self) -> str:
        """Computed lazily on first access."""
        print("Computing expensive_label...")
        return "Premium" if self.price > 100 else "Standard"

# Demonstration
if __name__ == "__main__":
    product = Product("Widget", 50)
    
    print(product.price)  # 50
    product.price = 100   # Valid
    # product.price = -5  # Raises ValueError
    
    print(product.expensive_label)  # Prints "Computing..." then "Standard"
    print(product.expensive_label)  # Cached, no print, just "Standard"
```

**Why**: Descriptors power Python's property, classmethod, staticmethod, and slots. They provide a clean way to add validation, lazy computation, or custom access logic without boilerplate. Understanding descriptors is key to Python metaprogramming.

## Slots

```python
# PROBLEM: Reduce memory usage and speed up attribute access
# TIME: O(1) for attribute access (faster) SPACE: O(1) per instance (less memory)

from dataclasses import dataclass
from typing import ClassVar
import sys

# Standard class without __slots__
class PointWithoutSlots:
    """Regular class - uses __dict__ for attributes."""
    
    def __init__(self, x: float, y: float, z: float):
        self.x = x
        self.y = y
        self.z = z

# Class with __slots__
class PointWithSlots:
    """
    Slotted class - fixed attribute layout, no __dict__.
    - Faster attribute access (no dict lookup)
    - Less memory (no per-instance dict)
    - Prevents adding arbitrary attributes
    """
    __slots__ = ('x', 'y', 'z')
    
    def __init__(self, x: float, y: float, z: float):
        self.x = x
        self.y = y
        self.z = z
    
    def __repr__(self):
        return f"Point({self.x}, {self.y}, {self.z})"
    
    def magnitude(self) -> float:
        return (self.x**2 + self.y**2 + self.z**2) ** 0.5

# Slots with inheritance
class Point3DWithColor(PointWithSlots):
    """Inheriting from slotted class - must add own slots."""
    __slots__ = ('color',)
    
    def __init__(self, x: float, y: float, z: float, color: str):
        super().__init__(x, y, z)
        self.color = color

# Dataclass with slots (Python 3.10+)
@dataclass(slots=True, frozen=True)
class ImmutablePoint:
    """Frozen dataclass with slots - immutable and memory efficient."""
    x: float
    y: float
    z: float
    
    def magnitude(self) -> float:
        return (self.x**2 + self.y**2 + self.z**2) ** 0.5

# Comparison and demonstration
def demonstrate_slots():
    """Show memory and performance differences."""
    
    # Create instances
    p1 = PointWithoutSlots(1.0, 2.0, 3.0)
    p2 = PointWithSlots(1.0, 2.0, 3.0)
    
    # Regular class allows arbitrary attributes
    p1.extra = "can add anything"
    
    # Slotted class prevents this
    try:
        p2.extra = "this fails"
    except AttributeError as e:
        print(f"Slots prevent arbitrary attributes: {e}")
    
    # Memory comparison
    print(f"Without slots size: {sys.getsizeof(p1.__dict__)} bytes (dict only)")
    print(f"With slots size: {sys.getsizeof(p2)} bytes (no dict)")
    
    # Create many instances to see memory difference
    points_no_slots = [PointWithoutSlots(i, i, i) for i in range(10000)]
    points_with_slots = [PointWithSlots(i, i, i) for i in range(10000)]
    
    total_no_slots = sum(sys.getsizeof(p.__dict__) for p in points_no_slots)
    total_with_slots = sum(sys.getsizeof(p) for p in points_with_slots)
    
    print(f"\n10000 instances:")
    print(f"Without slots: ~{total_no_slots / 1024:.1f} KB")
    print(f"With slots: ~{total_with_slots / 1024:.1f} KB")
    
    # Performance: attribute access
    import timeit
    
    no_slots_time = timeit.timeit(
        "p.x + p.y + p.z", 
        globals={"p": points_no_slots[0]},
        number=10_000_000
    )
    
    slots_time = timeit.timeit(
        "p.x + p.y + p.z", 
        globals={"p": points_with_slots[0]},
        number=10_000_000
    )
    
    print(f"\nAttribute access (10M iterations):")
    print(f"Without slots: {no_slots_time:.3f}s")
    print(f"With slots: {slots_time:.3f}s")

if __name__ == "__main__":
    demonstrate_slots()
```

**Why**: `__slots__` reduces memory by 40-60% for classes with many instances. The speedup comes from bypassing dict lookups. Essential for data-heavy applications, game development, or when creating millions of objects. Dataclasses with `slots=True` (Python 3.10+) give the best of both worlds.
