## 4. html_audio_autoplay

```html
<!-- PROBLEM: Automatically play audio when page loads (with caveats)
     TIME: O(1) SPACE: O(1) -->
<audio id="backgroundAudio" autoplay loop>
  <source src="background-music.mp3" type="audio/mpeg">
  <source src="background-music.ogg" type="audio/ogg">
  Your browser does not support the audio element.
</audio>

<button id="toggleAudio">Mute</button>

<script>
  const audio = document.getElementById('backgroundAudio');
  const toggleBtn = document.getElementById('toggleAudio');
  
  // Note: Most browsers require user interaction before autoplay
  // So we handle autoplay blocking gracefully
  audio.play().catch(() => {
    toggleBtn.textContent = 'Play Music';
  });
  
  toggleBtn.addEventListener('click', () => {
    if (audio.paused) {
      audio.play();
      toggleBtn.textContent = 'Mute';
    } else {
      audio.pause();
      toggleBtn.textContent = 'Play Music';
    }
  });
</script>
```

**Why**: While autoplay is possible, modern browsers restrict it for user experience. This pattern provides graceful fallback with manual controls when autoplay is blocked.

## 7. js_error_handling

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Error Handling Patterns</title>
</head>
<body>
  <script>
    // PROBLEM: Gracefully handle runtime errors in JavaScript
    // TIME: O(1) SPACE: O(1)
    
    // 1. Try-Catch for synchronous code
    function divideNumbers(a, b) {
      try {
        if (b === 0) {
          throw new Error('Division by zero is not allowed');
        }
        return a / b;
      } catch (error) {
        console.error('Error:', error.message);
        return null;
      } finally {
        console.log('Division attempt completed');
      }
    }
    
    // 2. Async error handling with try-catch
    async function fetchUserData(userId) {
      try {
        const response = await fetch(`/api/users/${userId}`);
        
        if (!response.ok) {
          throw new Error(`HTTP error! status: ${response.status}`);
        }
        
        const data = await response.json();
        return data;
      } catch (error) {
        console.error('Failed to fetch user:', error.message);
        throw error; // Re-throw for caller to handle
      }
    }
    
    // 3. Promise error handling
    function fetchWithPromise() {
      fetch('/api/data')
        .then(response => response.json())
        .then(data => console.log(data))
        .catch(error => console.error('Fetch error:', error))
        .finally(() => console.log('Fetch completed'));
    }
    
    // 4. Custom error types
    class ValidationError extends Error {
      constructor(message, field) {
        super(message);
        this.name = 'ValidationError';
        this.field = field;
      }
    }
    
    function validateEmail(email) {
      if (!email.includes('@')) {
        throw new ValidationError('Invalid email format', 'email');
      }
      return true;
    }
    
    // 5. Global error handlers
    window.addEventListener('error', (event) => {
      console.error('Global error:', event.error);
      // Send to error tracking service
    });
    
    window.addEventListener('unhandledrejection', (event) => {
      console.error('Unhandled promise rejection:', event.reason);
      event.preventDefault(); // Prevent default browser handling
    });
    
    // 6. Error handling with error object
    function processArray(arr) {
      try {
        if (!Array.isArray(arr)) {
          throw new TypeError('Input must be an array');
        }
        
        return arr.map(item => {
          if (typeof item !== 'number') {
            throw new TypeError(`Item ${item} is not a number`);
          }
          return item * 2;
        });
      } catch (error) {
        console.error(`${error.name}: ${error.message}`);
        return [];
      }
    }
    
    // Usage examples
    console.log(divideNumbers(10, 2));  // 5
    console.log(divideNumbers(10, 0));  // null with error message
    
    try {
      validateEmail('invalid-email');
    } catch (error) {
      console.error(`${error.field}: ${error.message}`);
    }
  </script>
</body>
</html>
```

**Why**: Comprehensive error handling prevents application crashes, provides meaningful feedback to users and developers, and enables proper error logging and monitoring in production.

## 8. js_template_literals

```html
<!DOCTYPE html>
<html lang="en">

<body>
  <div id="output"></div>
  
  <script>
    // PROBLEM: Create dynamic strings with embedded expressions
    // TIME: O(1) SPACE: O(1)
    
    // 1. Basic template literal
    const name = 'Alice';
    const age = 30;
    const greeting = `Hello, ${name}! You are ${age} years old.`;
    console.log(greeting);
    
    // 2. Multi-line strings
    const htmlTemplate = `
      <div class="card">
        <h2>${name}</h2>
        <p>Age: ${age}</p>
      </div>
    `;
    document.getElementById('output').innerHTML = htmlTemplate;
    
    // 3. Expression interpolation
    const a = 5;
    const b = 10;
    console.log(`Sum: ${a + b}, Product: ${a * b}`);
    
    // 4. Function calls in template literals
    function toUpperCase(str) {
      return str.toUpperCase();
    }
    console.log(`Name in uppercase: ${toUpperCase(name)}`);
    
    // 5. Tagged template literals
    function highlight(strings, ...values) {
      return strings.reduce((result, string, i) => {
        const value = values[i] ? 
          `<span class="highlight">${values[i]}</span>` : '';
        return result + string + value;
      }, '');
    }
    
    const highlighted = highlight`Hello ${name}, you have ${5} new messages`;
    console.log(highlighted);
    
    // 6. Nested template literals
    const items = ['Apple', 'Banana', 'Orange'];
    const list = `
      <ul>
        ${items.map(item => `<li>${item}</li>`).join('\n        ')}
      </ul>
    `;
    console.log(list);
    
    // 7. Conditional expressions
    const isLoggedIn = true;
    const userMessage = `Welcome ${isLoggedIn ? name : 'Guest'}!`;
    console.log(userMessage);
    
    // 8. Template literal with object properties
    const user = {
      name: 'Bob',
      email: 'bob@example.com',
      role: 'admin'
    };
    
    const userInfo = `
      User Details:
      Name: ${user.name}
      Email: ${user.email}
      Role: ${user.role}
    `;
    console.log(userInfo);
    
    // 9. Raw strings with String.raw
    const path = String.raw`C:\Users\Documents\file.txt`;
    console.log(path); // Backslashes are preserved
    
    // 10. Template literal in function return
    function createEmailTemplate(recipient, subject, body) {
      return `
To: ${recipient}
Subject: ${subject}

${body}

Best regards,
Your Team
      `.trim();
    }
    
    const email = createEmailTemplate(
      'john@example.com',
      'Meeting Reminder',
      'Don\'t forget our meeting tomorrow at 10 AM.'
    );
    console.log(email);
  </script>
</body>
</html>
```

**Why**: Template literals provide cleaner syntax for string interpolation, multi-line strings, and embedded expressions compared to string concatenation, making code more readable and maintainable.

## 10. event_focus

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Focus Event Examples</title>
  <style>
    input, textarea {
      width: 300px;
      padding: 10px;
      margin: 10px 0;
      border: 2px solid #ddd;
      border-radius: 4px;
      transition: all 0.3s;
    }
    
    input:focus, textarea:focus {
      outline: none;
      border-color: #4CAF50;
      box-shadow: 0 0 8px rgba(76, 175, 80, 0.3);
      background-color: #f9f9f9;
    }
    
    .input-wrapper {
      position: relative;
      margin: 20px 0;
    }
    
    .input-label {
      position: absolute;
      left: 10px;
      top: 50%;
      transform: translateY(-50%);
      color: #999;
      transition: all 0.3s;
      pointer-events: none;
    }
    
    .input-wrapper input:focus + .input-label,
    .input-wrapper input:not(:placeholder-shown) + .input-label {
      top: -10px;
      font-size: 12px;
      color: #4CAF50;
    }
    
    .focus-indicator {
      opacity: 0;
      transition: opacity 0.3s;
    }
    
    .focused .focus-indicator {
      opacity: 1;
    }
  </style>
</head>
<body>
  <h2>Focus Event Handling</h2>
  
  <div class="input-wrapper">
    <input type="text" id="nameInput" placeholder=" ">
    <label class="input-label" for="nameInput">Name</label>
    <span class="focus-indicator">✓ Field focused</span>
  </div>
  
  <div class="input-wrapper">
    <input type="email" id="emailInput" placeholder=" ">
    <label class="input-label" for="emailInput">Email</label>
    <span class="focus-indicator">✓ Field focused</span>
  </div>
  
  <textarea id="messageInput" placeholder="Enter your message"></textarea>
  
  <div id="focusLog"></div>
  
  <script>
    // PROBLEM: Handle element focus and blur events for better UX
    // TIME: O(1) SPACE: O(1)
    
    const nameInput = document.getElementById('nameInput');
    const emailInput = document.getElementById('emailInput');
    const messageInput = document.getElementById('messageInput');
    const focusLog = document.getElementById('focusLog');
    
    // 1. Basic focus and blur event handlers
    nameInput.addEventListener('focus', (e) => {
      console.log('Name input focused');
      e.target.parentElement.classList.add('focused');
      logEvent('Name field focused');
    });
    
    nameInput.addEventListener('blur', (e) => {
      console.log('Name input lost focus');
      e.target.parentElement.classList.remove('focused');
      logEvent('Name field lost focus');
    });
    
    // 2. Focus with validation
    emailInput.addEventListener('blur', (e) => {
      const email = e.target.value;
      if (email && !email.includes('@')) {
        e.target.style.borderColor = '#f44336';
        logEvent('Invalid email format');
      } else {
        e.target.style.borderColor = '#4CAF50';
      }
    });
    
    // 3. Focusin and focusout (bubble vs capture)
    document.body.addEventListener('focusin', (e) => {
      console.log('Focusin event (bubbles):', e.target.id);
    });
    
    document.body.addEventListener('focusout', (e) => {
      console.log('Focusout event (bubbles):', e.target.id);
    });
    
    // 4. Programmatic focus management
    function focusFirstInput() {
      nameInput.focus();
    }
    
    function focusNextInput(current) {
      const inputs = [nameInput, emailInput, messageInput];
      const currentIndex = inputs.indexOf(current);
      const nextIndex = (currentIndex + 1) % inputs.length;
      inputs[nextIndex].focus();
    }
    
    // 5. Tab key navigation enhancement
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') {
        e.preventDefault();
        focusNextInput(document.activeElement);
      }
    });
    
    // 6. Auto-focus on page load
    window.addEventListener('load', () => {
      nameInput.focus();
    });
    
    // 7. Track focus history
    const focusHistory = [];
    
    document.addEventListener('focusin', (e) => {
      if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA') {
        focusHistory.push({
          element: e.target.id,
          timestamp: new Date().toISOString()
        });
      }
    });
    
    // 8. Focus trap for modals (accessibility)
    function trapFocus(modal) {
      const focusableElements = modal.querySelectorAll(
        'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
      );
      const firstElement = focusableElements[0];
      const lastElement = focusableElements[focusableElements.length - 1];
      
      modal.addEventListener('keydown', (e) => {
        if (e.key === 'Tab') {
          if (e.shiftKey && document.activeElement === firstElement) {
            e.preventDefault();
            lastElement.focus();
          } else if (!e.shiftKey && document.activeElement === lastElement) {
            e.preventDefault();
            firstElement.focus();
          }
        }
      });
    }
    
    // 9. Visual focus indicator for accessibility
    document.addEventListener('focusin', (e) => {
      if (e.target.matches('input, textarea, button, a')) {
        e.target.setAttribute('data-focused', 'true');
      }
    });
    
    document.addEventListener('focusout', (e) => {
      e.target.removeAttribute('data-focused');
    });
    
    // Helper function to log events
    function logEvent(message) {
      const logEntry = document.createElement('div');
      logEntry.textContent = `${new Date().toLocaleTimeString()}: ${message}`;
      focusLog.appendChild(logEntry);
      focusLog.scrollTop = focusLog.scrollHeight;
    }
    
    // 10. Debounced focus handler (for search inputs)
    let searchTimeout;
    const searchInput = document.createElement('input');
    searchInput.placeholder = 'Search...';
    document.body.appendChild(searchInput);
    
    searchInput.addEventListener('input', (e) => {
      clearTimeout(searchTimeout);
      searchTimeout = setTimeout(() => {
        console.log('Searching for:', e.target.value);
      }, 300);
    });
  </script>
</body>
</html>
```

**Why**: Focus events are crucial for form validation, accessibility (keyboard navigation), and enhancing user experience by providing visual feedback when users interact with input fields.
