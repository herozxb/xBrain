## 1. Quickhull Algorithm

```python
# PROBLEM: Find convex hull of a set of 2D points using divide-and-conquer
# TIME: O(n log n) average, O(n²) worst case SPACE: O(n)
def quickhull(points):
    """
    Compute the convex hull of a set of 2D points.
    Returns the vertices of the convex hull in counter-clockwise order.
    """
    if len(points) <= 2:
        return points
    
    # Find leftmost and rightmost points
    min_x = min(points, key=lambda p: p[0])
    max_x = max(points, key=lambda p: p[0])
    
    # Divide points into two sets based on line from min_x to max_x
    left_set = []
    right_set = []
    
    for point in points:
        if point == min_x or point == max_x:
            continue
        # Calculate cross product to determine side
        cross = cross_product(min_x, max_x, point)
        if cross > 0:
            left_set.append(point)
        elif cross < 0:
            right_set.append(point)
    
    # Recursively find hull points on each side
    hull = [min_x]
    find_hull(left_set, min_x, max_x, hull)
    hull.append(max_x)
    find_hull(right_set, max_x, min_x, hull)
    
    return hull

def cross_product(a, b, c):
    """Calculate cross product of vectors AB and AC."""
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])

def find_hull(points, p1, p2, hull):
    """Recursively find hull points in the set."""
    if not points:
        return
    
    # Find point with maximum distance from line p1-p2
    max_dist = -1
    max_point = None
    
    for point in points:
        dist = point_to_line_distance(p1, p2, point)
        if dist > max_dist:
            max_dist = dist
            max_point = point
    
    if max_point is None:
        return
    
    # Insert max_point into hull between p1 and p2
    hull_index = hull.index(p2)
    hull.insert(hull_index, max_point)
    
    # Divide points into two subsets
    left_set1 = []
    left_set2 = []
    
    for point in points:
        if point == max_point:
            continue
        # Check which side of the new lines the point is on
        cross1 = cross_product(p1, max_point, point)
        cross2 = cross_product(max_point, p2, point)
        
        if cross1 > 0:
            left_set1.append(point)
        elif cross2 > 0:
            left_set2.append(point)
    
    # Recursively process both subsets
    find_hull(left_set1, p1, max_point, hull)
    find_hull(left_set2, max_point, p2, hull)

def point_to_line_distance(p1, p2, p):
    """Calculate perpendicular distance from point p to line p1-p2."""
    return abs(cross_product(p1, p2, p)) / distance(p1, p2)

def distance(p1, p2):
    """Calculate Euclidean distance between two points."""
    return ((p2[0] - p1[0])**2 + (p2[1] - p1[1])**2) ** 0.5
```

**Why**: Quickhull is an efficient divide-and-conquer algorithm for convex hull computation, similar to quicksort in its approach. It's practical for real-world applications in computational geometry.

## 2. Interleaving String

```python
# PROBLEM: Check if string s3 is formed by interleaving s1 and s2
# TIME: O(m*n) SPACE: O(m*n) or O(min(m,n)) with optimization
def is_interleave(s1, s2, s3):
    """
    Determine if s3 can be formed by interleaving the characters of s1 and s2
    while maintaining the relative order of characters from each string.
    """
    m, n = len(s1), len(s2)
    
    # Length check
    if m + n != len(s3):
        return False
    
    # DP table: dp[i][j] = True if s3[0:i+j] can be formed by s1[0:i] and s2[0:j]
    dp = [[False] * (n + 1) for _ in range(m + 1)]
    
    # Base case: empty strings
    dp[0][0] = True
    
    # Fill first row: using only s2
    for j in range(1, n + 1):
        dp[0][j] = dp[0][j-1] and s2[j-1] == s3[j-1]
    
    # Fill first column: using only s1
    for i in range(1, m + 1):
        dp[i][0] = dp[i-1][0] and s1[i-1] == s3[i-1]
    
    # Fill rest of the table
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            # s3[i+j-1] can come from either s1[i-1] or s2[j-1]
            from_s1 = dp[i-1][j] and s1[i-1] == s3[i+j-1]
            from_s2 = dp[i][j-1] and s2[j-1] == s3[i+j-1]
            dp[i][j] = from_s1 or from_s2
    
    return dp[m][n]

# Space-optimized version using 1D DP
def is_interleave_optimized(s1, s2, s3):
    """Optimized version using O(n) space where n = len(s2)."""
    m, n = len(s1), len(s2)
    
    if m + n != len(s3):
        return False
    
    # Use shorter string for column dimension
    if m < n:
        s1, s2 = s2, s1
        m, n = n, m
    
    dp = [False] * (n + 1)
    
    for i in range(m + 1):
        for j in range(n + 1):
            if i == 0 and j == 0:
                dp[j] = True
            elif i == 0:
                dp[j] = dp[j-1] and s2[j-1] == s3[j-1]
            elif j == 0:
                dp[j] = dp[j] and s1[i-1] == s3[i-1]
            else:
                dp[j] = (dp[j] and s1[i-1] == s3[i+j-1]) or \
                        (dp[j-1] and s2[j-1] == s3[i+j-1])
    
    return dp[n]
```

**Why**: This is a classic DP problem that tests understanding of state transitions and space optimization. The interleaving constraint makes it more complex than typical string matching problems.

## 3. Regular Expression Matching

```python
# PROBLEM: Implement regex matching with '.' and '*' support
# TIME: O(m*n) SPACE: O(m*n)
def is_match(s, p):
    """
    Check if string s matches pattern p where:
    '.' matches any single character
    '*' matches zero or more of the preceding element
    """
    m, n = len(s), len(p)
    
    # dp[i][j] = True if s[0:i] matches p[0:j]
    dp = [[False] * (n + 1) for _ in range(m + 1)]
    
    # Empty pattern matches empty string
    dp[0][0] = True
    
    # Handle patterns like a*, a*b*, a*b*c* which can match empty string
    for j in range(2, n + 1):
        if p[j-1] == '*':
            dp[0][j] = dp[0][j-2]
    
    # Fill the DP table
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            # If current pattern char is '*', we have two choices:
            # 1. Match zero occurrences of preceding element
            # 2. Match one or more occurrences if preceding matches current s char
            if p[j-1] == '*':
                # Zero occurrences: ignore the char-* pair
                dp[i][j] = dp[i][j-2]
                # One or more: check if preceding char matches s[i-1]
                if p[j-2] == s[i-1] or p[j-2] == '.':
                    dp[i][j] = dp[i][j] or dp[i-1][j]
            # Direct match or '.' wildcard
            elif p[j-1] == s[i-1] or p[j-1] == '.':
                dp[i][j] = dp[i-1][j-1]
    
    return dp[m][n]

# Recursive solution with memoization (often more intuitive)
def is_match_recursive(s, p):
    """Recursive approach with memoization for better readability."""
    memo = {}
    
    def dp(i, j):
        # Check if pattern from j onwards matches string from i onwards
        if (i, j) in memo:
            return memo[(i, j)]
        
        # If pattern is exhausted, string must also be exhausted
        if j == len(p):
            return i == len(s)
        
        # Check if current characters match
        first_match = i < len(s) and (p[j] == s[i] or p[j] == '.')
        
        # Handle '*' wildcard
        if j + 1 < len(p) and p[j+1] == '*':
            # Either match zero times, or match once and continue
            result = dp(i, j+2) or (first_match and dp(i+1, j))
        else:
            # Simple match: advance both pointers
            result = first_match and dp(i+1, j+1)
        
        memo[(i, j)] = result
        return result
    
    return dp(0, 0)
```

**Why**: This problem elegantly combines DP with parsing logic. The '*' operator requires careful state management, making it an excellent test of algorithmic thinking and DP skills.

## 4. Sudoku Solver

```python
# PROBLEM: Solve a 9x9 Sudoku board using backtracking
# TIME: O(9^(n*n)) worst case SPACE: O(n*n) for recursion stack
def solve_sudoku(board):
    """
    Solve the Sudoku puzzle in-place using backtracking.
    board: 9x9 2D list where '.' represents empty cells
    Returns True if solvable, False otherwise.
    """
    def is_valid(row, col, num):
        """Check if placing num at board[row][col] is valid."""
        # Check row
        if num in board[row]:
            return False
        
        # Check column
        for i in range(9):
            if board[i][col] == num:
                return False
        
        # Check 3x3 box
        box_row, box_col = 3 * (row // 3), 3 * (col // 3)
        for i in range(box_row, box_row + 3):
            for j in range(box_col, box_col + 3):
                if board[i][j] == num:
                    return False
        
        return True
    
    def solve():
        """Backtracking solver."""
        # Find next empty cell
        for i in range(9):
            for j in range(9):
                if board[i][j] == '.':
                    # Try digits 1-9
                    for num in map(str, range(1, 10)):
                        if is_valid(i, j, num):
                            board[i][j] = num
                            
                            # Recursively solve rest of board
                            if solve():
                                return True
                            
                            # Backtrack if this path fails
                            board[i][j] = '.'
                    
                    # No valid number found, trigger backtracking
                    return False
        
        # All cells filled - solution found
        return True
    
    return solve()

# Optimized version using constraint propagation and bit manipulation
def solve_sudoku_optimized(board):
    """
    Optimized Sudoku solver using bit manipulation for constraint tracking.
    Each row, column, and box uses an integer as a bitmask for used digits.
    """
    # Bitmasks for tracking used digits in rows, columns, and boxes
    rows = [0] * 9
    cols = [0] * 9
    boxes = [0] * 9
    
    # Initialize constraints from existing numbers
    for i in range(9):
        for j in range(9):
            if board[i][j] != '.':
                num = int(board[i][j])
                bit = 1 << num
                rows[i] |= bit
                cols[j] |= bit
                boxes[(i//3)*3 + j//3] |= bit
    
    def get_candidates(row, col):
        """Get available digits for a cell using bitmask operations."""
        used = rows[row] | cols[col] | boxes[(row//3)*3 + col//3]
        candidates = []
        for num in range(1, 10):
            if not (used & (1 << num)):
                candidates.append(str(num))
        return candidates
    
    def solve():
        """Backtracking with constraint propagation."""
        min_candidates = None
        min_cell = None
        
        # Find cell with minimum remaining candidates (MRV heuristic)
        for i in range(9):
            for j in range(9):
                if board[i][j] == '.':
                    candidates = get_candidates(i, j)
                    if not candidates:
                        return False
                    if min_candidates is None or len(candidates) < len(min_candidates):
                        min_candidates = candidates
                        min_cell = (i, j)
        
        # All cells filled
        if min_cell is None:
            return True
        
        row, col = min_cell
        box_idx = (row//3)*3 + col//3
        
        # Try each candidate
        for num in min_candidates:
            bit = 1 << int(num)
            
            # Place number
            board[row][col] = num
            rows[row] |= bit
            cols[col] |= bit
            boxes[box_idx] |= bit
            
            if solve():
                return True
            
            # Remove number (backtrack)
            board[row][col] = '.'
            rows[row] ^= bit
            cols[col] ^= bit
            boxes[box_idx] ^= bit
        
        return False
    
    return solve()
```

**Why**: Sudoku is the quintessential backtracking problem. The optimized version demonstrates constraint propagation and the Minimum Remaining Values (MRV) heuristic, crucial for solving constraint satisfaction problems.

## 5. Knight's Tour

```python
# PROBLEM: Find a sequence of moves for a knight to visit every square on a chessboard
# TIME: O(8^(n*n)) SPACE: O(n*n)
def knight_tour(n):
    """
    Find a knight's tour on an n×n chessboard.
    Returns a board with move numbers, or None if no tour exists.
    """
    # Initialize board with -1 (unvisited)
    board = [[-1] * n for _ in range(n)]
    
    # All 8 possible knight moves
    moves = [
        (2, 1), (1, 2), (-1, 2), (-2, 1),
        (-2, -1), (-1, -2), (1, -2), (2, -1)
    ]
    
    def is_valid(x, y):
        """Check if position (x, y) is on the board and unvisited."""
        return 0 <= x < n and 0 <= y < n and board[x][y] == -1
    
    def count_onward_moves(x, y):
        """Count how many moves are possible from (x, y)."""
        count = 0
        for dx, dy in moves:
            nx, ny = x + dx, y + dy
            if is_valid(nx, ny):
                count += 1
        return count
    
    def solve(x, y, move_count):
        """
        Recursively find knight's tour using Warnsdorff's heuristic.
        Always move to the square with fewest onward moves.
        """
        if move_count == n * n:
            return True
        
        # Get all valid moves from current position
        next_moves = []
        for dx, dy in moves:
            nx, ny = x + dx, y + dy
            if is_valid(nx, ny):
                # Use Warnsdorff's heuristic: prefer squares with fewer onward moves
                onward = count_onward_moves(nx, ny)
                next_moves.append((onward, nx, ny))
        
        # Sort by number of onward moves (ascending)
        next_moves.sort()
        
        # Try each move
        for _, nx, ny in next_moves:
            board[nx][ny] = move_count
            
            if solve(nx, ny, move_count + 1):
                return True
            
            # Backtrack
            board[nx][ny] = -1
        
        return False
    
    # Start from corner (0, 0)
    board[0][0] = 0
    
    if solve(0, 0, 1):
        return board
    else:
        return None

def print_board(board):
    """Print the knight's tour board."""
    if board is None:
        print("No solution exists")
        return
    
    n = len(board)
    for row in board:
        print(' '.join(f'{cell:3d}' for cell in row))
```

**Why**: Knight's Tour showcases the power of heuristics in backtracking. Warnsdorff's heuristic dramatically reduces the search space, making an otherwise intractable problem solvable efficiently.

## 6. Unbounded Knapsack

```python
# PROBLEM: Maximize value in knapsack with unlimited items of each type
# TIME: O(n*W) SPACE: O(W)
def unbounded_knapsack(weights, values, capacity):
    """
    Solve unbounded knapsack problem where each item can be used unlimited times.
    Returns maximum value achievable.
    """
    n = len(weights)
    
    # dp[w] = maximum value achievable with capacity w
    dp = [0] * (capacity + 1)
    
    # For each capacity from 1 to capacity
    for w in range(1, capacity + 1):
        # Try each item
        for i in range(n):
            if weights[i] <= w:
                # Take item i and add to optimal solution for remaining capacity
                dp[w] = max(dp[w], values[i] + dp[w - weights[i]])
    
    return dp[capacity]

# Variation: Coin Change - Minimum number of coins
def coin_change_min_coins(coins, amount):
    """
    Find minimum number of coins needed to make up the amount.
    Returns -1 if not possible.
    """
    # dp[i] = min coins needed for amount i
    dp = [float('inf')] * (amount + 1)
    dp[0] = 0
    
    for i in range(1, amount + 1):
        for coin in coins:
            if coin <= i:
                dp[i] = min(dp[i], dp[i - coin] + 1)
    
    return dp[amount] if dp[amount] != float('inf') else -1

# Variation: Coin Change - Number of ways
def coin_change_ways(coins, amount):
    """
    Find number of ways to make up the amount.
    """
    # dp[i] = number of ways to make amount i
    dp = [0] * (amount + 1)
    dp[0] = 1  # One way to make amount 0: use no coins
    
    # Process each coin
    for coin in coins:
        # For each amount from coin to target
        for i in range(coin, amount + 1):
            dp[i] += dp[i - coin]
    
    return dp[amount]

# Variation: Rod Cutting - Maximize profit
def rod_cutting(prices, length):
    """
    Given rod prices for different lengths, find maximum profit from cutting rod.
    prices[i] = price for rod of length i+1
    """
    # dp[i] = max profit for rod of length i
    dp = [0] * (length + 1)
    
    for i in range(1, length + 1):
        max_profit = 0
        # Try cutting at each possible length
        for j in range(1, min(i, len(prices)) + 1):
            max_profit = max(max_profit, prices[j-1] + dp[i - j])
        dp[i] = max_profit
    
    return dp[length]
```

**Why**: Unbounded knapsack is a fundamental DP pattern with many real-world applications: coin change, rod cutting, resource allocation. Understanding this pattern enables solving a wide class of optimization problems.

## 7. Maximal Rectangle

```python
# PROBLEM: Find largest rectangle containing only 1s in a binary matrix
# TIME: O(rows * cols) SPACE: O(cols)
def maximal_rectangle(matrix):
    """
    Find the area of the largest rectangle containing only 1's.
    Uses the largest rectangle in histogram approach for each row.
    """
    if not matrix or not matrix[0]:
        return 0
    
    rows, cols = len(matrix), len(matrix[0])
    heights = [0] * cols
    max_area = 0
    
    for row in matrix:
        # Update heights for current row
        for j in range(cols):
            if row[j] == '1':
                heights[j] += 1
            else:
                heights[j] = 0
        
        # Find largest rectangle in current histogram
        max_area = max(max_area, largest_rectangle_area(heights))
    
    return max_area

def largest_rectangle_area(heights):
    """
    Find largest rectangle area in histogram using monotonic stack.
    """
    stack = []  # Stack stores indices
    max_area = 0
    heights = heights + [0]  # Add sentinel
    
    for i, h in enumerate(heights):
        # Maintain increasing stack
        while stack and heights[stack[-1]] > h:
            height = heights[stack.pop()]
            
            # Calculate width: from previous smaller to current
            if stack:
                width = i - stack[-1] - 1
            else:
                width = i
            
            max_area = max(max_area, height * width)
        
        stack.append(i)
    
    return max_area

# Alternative: DP approach
def maximal_rectangle_dp(matrix):
    """
    DP approach tracking left, right, and height boundaries.
    """
    if not matrix or not matrix[0]:
        return 0
    
    rows, cols = len(matrix), len(matrix[0])
    left = [0] * cols
    right = [cols] * cols
    height = [0] * cols
    max_area = 0
    
    for i in range(rows):
        cur_left, cur_right = 0, cols
        
        # Update height
        for j in range(cols):
            if matrix[i][j] == '1':
                height[j] += 1
            else:
                height[j] = 0
        
        # Update left boundary
        for j in range(cols):
            if matrix[i][j] == '1':
                left[j] = max(left[j], cur_left)
            else:
                left[j] = 0
                cur_left = j + 1
        
        # Update right boundary
        for j in range(cols - 1, -1, -1):
            if matrix[i][j] == '1':
                right[j] = min(right[j], cur_right)
            else:
                right[j] = cols
                cur_right = j
        
        # Calculate max area for this row
        for j in range(cols):
            max_area = max(max_area, height[j] * (right[j] - left[j]))
    
    return max_area
```

**Why**: This problem combines two important techniques: monotonic stacks and 2D DP. It demonstrates how to decompose a complex problem into simpler subproblems (largest rectangle in histogram).

## 8. Largest Rectangle in Histogram

```python
# PROBLEM: Find largest rectangular area in a histogram
# TIME: O(n) SPACE: O(n)
def largest_rectangle_histogram(heights):
    """
    Find the area of the largest rectangle in a histogram.
    Uses monotonic stack to find next smaller element on both sides.
    """
    n = len(heights)
    stack = []
    max_area = 0
    
    for i in range(n):
        # While current height is less than stack top, pop and calculate area
        while stack and heights[stack[-1]] > heights[i]:
            height = heights[stack.pop()]
            
            # Width extends from previous smaller (stack top) to current
            if stack:
                width = i - stack[-1] - 1
            else:
                width = i  # No previous smaller, extends to start
            
            max_area = max(max_area, height * width)
        
        stack.append(i)
    
    # Process remaining bars in stack
    while stack:
        height = heights[stack.pop()]
        if stack:
            width = n - stack[-1] - 1
        else:
            width = n
        max_area = max(max_area, height * width)
    
    return max_area

# Alternative: Precompute next smaller elements
def largest_rectangle_histogram_v2(heights):
    """
    Alternative approach using arrays for next smaller elements.
    """
    n = len(heights)
    
    # Find next smaller on left
    left_smaller = [-1] * n
    stack = []
    for i in range(n):
        while stack and heights[stack[-1]] >= heights[i]:
            stack.pop()
        left_smaller[i] = stack[-1] if stack else -1
        stack.append(i)
    
    # Find next smaller on right
    right_smaller = [n] * n
    stack = []
    for i in range(n - 1, -1, -1):
        while stack and heights[stack[-1]] >= heights[i]:
            stack.pop()
        right_smaller[i] = stack[-1] if stack else n
        stack.append(i)
    
    # Calculate max area
    max_area = 0
    for i in range(n):
        width = right_smaller[i] - left_smaller[i] - 1
        max_area = max(max_area, heights[i] * width)
    
    return max_area
```

**Why**: The monotonic stack technique is essential for solving problems involving "next greater/smaller element" patterns. This classic problem appears in many interviews and demonstrates efficient stack usage.

## 9. Container With Most Water (Optimized)

```python
# PROBLEM: Find two lines that form container holding the most water
# TIME: O(n) SPACE: O(1)
def max_area(height):
    """
    Find maximum water container area using two-pointer technique.
    Start from ends and move the shorter line inward.
    """
    left, right = 0, len(height) - 1
    max_water = 0
    
    while left < right:
        # Calculate current area
        width = right - left
        h = min(height[left], height[right])
        max_water = max(max_water, width * h)
        
        # Move the shorter line (optimization: can't do better keeping it)
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1
    
    return max_water

# Extended: Return indices and heights of the two lines
def max_area_with_details(height):
    """
    Return the area and details of the optimal container.
    """
    left, right = 0, len(height) - 1
    max_water = 0
    best_left, best_right = left, right
    
    while left < right:
        width = right - left
        h = min(height[left], height[right])
        current_area = width * h
        
        if current_area > max_water:
            max_water = current_area
            best_left, best_right = left, right
        
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1
    
    return {
        'max_area': max_water,
        'left_index': best_left,
        'right_index': best_right,
        'left_height': height[best_left],
        'right_height': height[best_right]
    }

# Proof of correctness: Why greedy works
"""
The two-pointer approach works because:
1. Starting from max width, we can only decrease width
2. To potentially get more area, we need greater height
3. The current max height is min(h[left], h[right])
4. Moving the taller line can't improve min height (it stays same or decreases)
5. Moving the shorter line might find a taller one
6. Therefore, always move the shorter line
"""
```

## 10. Online Stock Span

```python
# PROBLEM: Design a data structure for calculating stock span efficiently
# TIME: O(1) amortized per operation SPACE: O(n)
class StockSpanner:
    """
    Calculate span of stock prices: consecutive days with price <= today's price.
    Uses monotonic stack to track prices and their spans.
    """
    
    def __init__(self):
        # Stack of (price, span) pairs
        self.stack = []
    
    def next(self, price):
        """
        Process next price and return its span.
        Span = number of consecutive days with price <= current price.
        """
        span = 1
        
        # Pop all prices less than or equal to current price
        # Accumulate their spans (they are now part of current span)
        while self.stack and self.stack[-1][0] <= price:
            _, prev_span = self.stack.pop()
            span += prev_span
        
        self.stack.append((price, span))
        return span

# Alternative implementation with explicit tracking
class StockSpannerDetailed:
    """
    Alternative implementation tracking indices and using monotonic stack.
    """
    
    def __init__(self):
        self.prices = []  # All prices seen so far
        self.stack = []   # Stack of indices with decreasing prices
        self.day = 0
    
    def next(self, price):
        """Process next price and return its span."""
        self.prices.append(price)
        
        # Find previous greater element (strictly greater)
        while self.stack and self.prices[self.stack[-1]] <= price:
            self.stack.pop()
        
        # Calculate span
        if not self.stack:
            span = self.day + 1  # All previous days
        else:
            span = self.day - self.stack[-1]
        
        self.stack.append(self.day)
        self.day += 1
        
        return span

# Example usage and testing
def test_stock_spanner():
    """Test the stock spanner with example data."""
    spanner = StockSpanner()
    prices = [100, 80, 60, 70, 60, 75, 85]
    expected = [1, 1, 1, 2, 1, 4, 6]
    
    results = []
    for price in prices:
        results.append(spanner.next(price))
    
    assert results == expected, f"Expected {expected}, got {results}"
    print("All tests passed!")
    return results
```
