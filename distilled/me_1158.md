## Heap Sort

```python
# PROBLEM: Sort an array using heap-based comparison sort
# TIME: O(n log n) SPACE: O(1) in-place
def heap_sort(arr):
    """
    Heap sort using max-heap approach.
    1. Build max heap from array
    2. Extract max element repeatedly
    3. Heapify remaining elements
    """
    n = len(arr)
    
    # Build max heap (start from last non-leaf node)
    for i in range(n // 2 - 1, -1, -1):
        _heapify(arr, n, i)
    
    # Extract elements one by one
    for i in range(n - 1, 0, -1):
        # Move current root (max) to end
        arr[0], arr[i] = arr[i], arr[0]
        # Heapify reduced heap
        _heapify(arr, i, 0)
    
    return arr

def _heapify(arr, n, i):
    """Maintain max-heap property at index i."""
    largest = i
    left = 2 * i + 1
    right = 2 * i + 2
    
    # Find largest among root, left child, and right child
    if left < n and arr[left] > arr[largest]:
        largest = left
    if right < n and arr[right] > arr[largest]:
        largest = right
    
    # Swap and continue heapifying if root is not largest
    if largest != i:
        arr[i], arr[largest] = arr[largest], arr[i]
        _heapify(arr, n, largest)
```

**Why**: Heap sort guarantees O(n log n) worst-case performance and sorts in-place. Unlike quicksort, it has no worst-case O(n²) scenario. Ideal when worst-case performance matters more than average-case speed.

## A* Search Algorithm

```python
# PROBLEM: Find shortest path using heuristic-guided search
# TIME: O(E log V) typical, O(b^d) worst case SPACE: O(V)
import heapq
from typing import Dict, List, Tuple, Optional

def a_star_search(
    graph: Dict[str, List[Tuple[str, int]]],
    start: str,
    goal: str,
    heuristic: Dict[str, int]
) -> Optional[Tuple[List[str], int]]:
    """
    A* search algorithm for optimal pathfinding.
    
    Args:
        graph: Adjacency list {node: [(neighbor, cost), ...]}
        start: Starting node
        goal: Target node
        heuristic: Estimated distance from each node to goal
    
    Returns:
        (path, total_cost) or None if no path exists
    """
    # Priority queue: (f_score, g_score, node, path)
    # f_score = g_score + heuristic
    open_set = [(heuristic[start], 0, start, [start])]
    closed_set = set()
    
    while open_set:
        f_score, g_score, current, path = heapq.heappop(open_set)
        
        # Check if we reached the goal
        if current == goal:
            return path, g_score
        
        # Skip if already visited
        if current in closed_set:
            continue
        
        closed_set.add(current)
        
        # Explore neighbors
        for neighbor, cost in graph.get(current, []):
            if neighbor not in closed_set:
                new_g = g_score + cost
                new_f = new_g + heuristic.get(neighbor, 0)
                heapq.heappush(
                    open_set,
                    (new_f, new_g, neighbor, path + [neighbor])
                )
    
    return None  # No path found

# Example usage
if __name__ == "__main__":
    # Graph: node -> [(neighbor, edge_cost), ...]
    graph = {
        'A': [('B', 1), ('C', 4)],
        'B': [('A', 1), ('C', 2), ('D', 5)],
        'C': [('A', 4), ('B', 2), ('D', 1)],
        'D': [('B', 5), ('C', 1)]
    }
    
    # Heuristic: estimated distance to goal 'D'
    heuristic = {'A': 3, 'B': 2, 'C': 1, 'D': 0}
    
    result = a_star_search(graph, 'A', 'D', heuristic)
    print(f"Path: {result[0]}, Cost: {result[1]}")
```

**Why**: A* combines Dijkstra's optimality with heuristic guidance, making it much faster than uninformed search when good heuristics exist. Widely used in pathfinding (games, maps, robotics) because it finds optimal paths while exploring fewer nodes than BFS/Dijkstra.

## KMP Pattern Matching

```python
# PROBLEM: Find all occurrences of pattern in text efficiently
# TIME: O(n + m) SPACE: O(m) for failure function
from typing import List

def kmp_search(text: str, pattern: str) -> List[int]:
    """
    Knuth-Morris-Pratt string matching algorithm.
    
    Args:
        text: The text to search in
        pattern: The pattern to search for
    
    Returns:
        List of starting indices where pattern occurs
    """
    if not pattern:
        return list(range(len(text) + 1))
    
    # Build failure function (partial match table)
    failure = _build_failure_function(pattern)
    
    matches = []
    j = 0  # Index for pattern
    
    for i in range(len(text)):
        # Mismatch: use failure function to backtrack
        while j > 0 and text[i] != pattern[j]:
            j = failure[j - 1]
        
        # Match: advance pattern index
        if text[i] == pattern[j]:
            j += 1
        
        # Full pattern match found
        if j == len(pattern):
            matches.append(i - len(pattern) + 1)
            j = failure[j - 1]  # Continue searching
    
    return matches

def _build_failure_function(pattern: str) -> List[int]:
    """
    Build the failure function (LPS array).
    failure[i] = length of longest proper prefix that is also suffix
    for pattern[0..i]
    """
    m = len(pattern)
    failure = [0] * m
    j = 0  # Length of previous longest prefix-suffix
    
    for i in range(1, m):
        # Mismatch: backtrack to previous valid prefix
        while j > 0 and pattern[i] != pattern[j]:
            j = failure[j - 1]
        
        # Match: extend prefix-suffix
        if pattern[i] == pattern[j]:
            j += 1
            failure[i] = j
    
    return failure

# Example usage
if __name__ == "__main__":
    text = "ABABDABACDABABCABAB"
    pattern = "ABABCABAB"
    
    matches = kmp_search(text, pattern)
    print(f"Pattern found at indices: {matches}")
    # Output: [10]
```

**Why**: KMP achieves linear O(n + m) time by never backtracking in the text. The failure function allows intelligent skipping based on already-matched characters. Essential for pattern matching in large texts, DNA sequences, or any scenario where naive O(nm) is too slow.

## Fenwick Tree (Binary Indexed Tree)

```python
# PROBLEM: Efficiently compute prefix sums with point updates
# TIME: O(log n) for both query and update SPACE: O(n)
from typing import List

class FenwickTree:
    """
    Fenwick Tree (Binary Indexed Tree) for prefix sum queries.
    
    Supports:
    - Point update: add value to single element
    - Prefix sum query: sum of first k elements
    - Range sum query: sum of elements in range [i, j]
    
    Uses 1-based indexing internally for bitwise operations.
    """
    
    def __init__(self, size: int):
        """Initialize tree of given size."""
        self.n = size
        self.tree = [0] * (size + 1)  # 1-based indexing
    
    def update(self, index: int, delta: int) -> None:
        """
        Add delta to element at index (0-based).
        Propagates change to all relevant tree nodes.
        """
        i = index + 1  # Convert to 1-based
        while i <= self.n:
            self.tree[i] += delta
            i += i & (-i)  # Move to next responsible node
    
    def prefix_sum(self, index: int) -> int:
        """
        Get sum of elements [0, index] (0-based, inclusive).
        Uses binary representation to accumulate partial sums.
        """
        i = index + 1  # Convert to 1-based
        result = 0
        while i > 0:
            result += self.tree[i]
            i -= i & (-i)  # Move to parent
        return result
    
    def range_sum(self, left: int, right: int) -> int:
        """Get sum of elements in [left, right] (0-based, inclusive)."""
        if left == 0:
            return self.prefix_sum(right)
        return self.prefix_sum(right) - self.prefix_sum(left - 1)
    
    @classmethod
    def from_array(cls, arr: List[int]) -> 'FenwickTree':
        """Build Fenwick tree from array in O(n)."""
        ft = cls(len(arr))
        for i, val in enumerate(arr):
            ft.tree[i + 1] = val
        
        # Build tree structure
        for i in range(1, ft.n + 1):
            parent = i + (i & (-i))
            if parent <= ft.n:
                ft.tree[parent] += ft.tree[i]
        
        return ft

# Example usage
if __name__ == "__main__":
    arr = [1, 3, 5, 7, 9, 11]
    ft = FenwickTree.from_array(arr)
    
    print(f"Prefix sum [0..3]: {ft.prefix_sum(3)}")  # 1+3+5+7 = 16
    print(f"Range sum [2..4]: {ft.range_sum(2, 4)}")  # 5+7+9 = 21
    
    ft.update(2, 10)  # Add 10 to index 2
    print(f"After update, range sum [2..4]: {ft.range_sum(2, 4)}")  # 15+7+9 = 31
```

**Why**: Fenwick trees achieve the same O(log n) operations as segment trees but with simpler implementation and lower constant factors. Ideal for competitive programming and applications requiring frequent range sum queries with point updates (e.g., frequency counting, dynamic prefix sums).

## Prototype Pattern

```python
# PROBLEM: Clone objects without coupling to their concrete classes
# TIME: O(n) for deep copy SPACE: O(n) for cloned object
import copy
from abc import ABC, abstractmethod
from typing import Any, Dict

class Prototype(ABC):
    """Abstract prototype interface."""
    
    @abstractmethod
    def clone(self) -> 'Prototype':
        """Create a copy of this object."""
        pass
    
    @abstractmethod
    def shallow_clone(self) -> 'Prototype':
        """Create a shallow copy of this object."""
        pass

class Document(Prototype):
    """Concrete prototype representing a document."""
    
    def __init__(
        self,
        title: str,
        content: str,
        metadata: Dict[str, Any],
        attachments: list
    ):
        self.title = title
        self.content = content
        self.metadata = metadata  # Mutable - needs deep copy
        self.attachments = attachments  # Mutable list - needs deep copy
        self.created_at = "2026-02-19"
    
    def clone(self) -> 'Document':
        """
        Deep clone: creates independent copy including nested objects.
        Changes to clone won't affect original.
        """
        return copy.deepcopy(self)
    
    def shallow_clone(self) -> 'Document':
        """
        Shallow clone: copies top-level attributes only.
        Nested mutable objects are shared between original and clone.
        """
        return copy.copy(self)
    
    def __str__(self) -> str:
        return f"Document(title='{self.title}', metadata={self.metadata})"

class PrototypeRegistry:
    """Registry for managing prototype instances."""
    
    def __init__(self):
        self._prototypes: Dict[str, Prototype] = {}
    
    def register(self, name: str, prototype: Prototype) -> None:
        """Register a prototype with a name."""
        self._prototypes[name] = prototype
    
    def unregister(self, name: str) -> None:
        """Remove a prototype from registry."""
        del self._prototypes[name]
    
    def clone(self, name: str) -> Prototype:
        """Clone a registered prototype by name."""
        return self._prototypes[name].clone()

# Example usage
if __name__ == "__main__":
    # Create prototype document
    template = Document(
        title="Contract Template",
        content="This agreement is made on...",
        metadata={"version": "1.0", "author": "Legal Team"},
        attachments=["terms.pdf", "conditions.pdf"]
    )
    
    # Create registry and register prototype
    registry = PrototypeRegistry()
    registry.register("contract_template", template)
    
    # Clone for specific use
    contract1 = registry.clone("contract_template")
    contract1.title = "Contract - Client A"
    contract1.metadata["client"] = "Client A"
    
    contract2 = registry.clone("contract_template")
    contract2.title = "Contract - Client B"
    contract2.metadata["client"] = "Client B"
    
    # Original template unchanged
    print(f"Template: {template}")
    print(f"Contract 1: {contract1}")
    print(f"Contract 2: {contract2}")
    
    # Demonstrate deep vs shallow copy
    shallow = template.shallow_clone()
    shallow.metadata["version"] = "2.0"
    print(f"Original metadata after shallow clone modification: {template.metadata}")
    # Original is affected! (shared reference)
```

**Why**: Prototype pattern enables object cloning without knowing their concrete classes. Essential when object creation is expensive (database queries, network calls) or when you need to preserve object state. Python's `copy` module provides built-in support, making implementation straightforward.

## Builder Pattern

```python
# PROBLEM: Construct complex objects step-by-step with clear API
# TIME: O(1) per step, O(n) total SPACE: O(n) for built object
from typing import Optional, List

class Computer:
    """Complex product with many optional attributes."""
    
    def __init__(self):
        self.cpu: Optional[str] = None
        self.ram: Optional[int] = None  # GB
        self.storage: Optional[int] = None  # GB
        self.gpu: Optional[str] = None
        self.monitor: Optional[str] = None
        self.peripherals: List[str] = []
    
    def __str__(self) -> str:
        specs = [
            f"CPU: {self.cpu}",
            f"RAM: {self.ram}GB",
            f"Storage: {self.storage}GB",
        ]
        if self.gpu:
            specs.append(f"GPU: {self.gpu}")
        if self.monitor:
            specs.append(f"Monitor: {self.monitor}")
        if self.peripherals:
            specs.append(f"Peripherals: {', '.join(self.peripherals)}")
        return "\n".join(specs)

class ComputerBuilder:
    """Builder for constructing Computer objects step by step."""
    
    def __init__(self):
        self._computer = Computer()
    
    def set_cpu(self, cpu: str) -> 'ComputerBuilder':
        """Set the CPU model."""
        self._computer.cpu = cpu
        return self  # Enable method chaining
    
    def set_ram(self, gb: int) -> 'ComputerBuilder':
        """Set RAM capacity in GB."""
        if gb < 1:
            raise ValueError("RAM must be at least 1GB")
        self._computer.ram = gb
        return self
    
    def set_storage(self, gb: int) -> 'ComputerBuilder':
        """Set storage capacity in GB."""
        if gb < 1:
            raise ValueError("Storage must be at least 1GB")
        self._computer.storage = gb
        return self
    
    def set_gpu(self, gpu: str) -> 'ComputerBuilder':
        """Set the GPU model."""
        self._computer.gpu = gpu
        return self
    
    def set_monitor(self, monitor: str) -> 'ComputerBuilder':
        """Set the monitor type."""
        self._computer.monitor = monitor
        return self
    
    def add_peripheral(self, peripheral: str) -> 'ComputerBuilder':
        """Add a peripheral device."""
        self._computer.peripherals.append(peripheral)
        return self
    
    def build(self) -> Computer:
        """Build and validate the final Computer object."""
        # Validation
        if not self._computer.cpu:
            raise ValueError("CPU is required")
        if not self._computer.ram:
            raise ValueError("RAM is required")
        if not self._computer.storage:
            raise ValueError("Storage is required")
        
        # Return built computer and reset builder
        result = self._computer
        self._computer = Computer()
        return result

# Director class (optional) - encapsulates common build sequences
class ComputerDirector:
    """Director for creating predefined computer configurations."""
    
    @staticmethod
    def build_gaming_pc() -> Computer:
        """Create a gaming PC configuration."""
        return (ComputerBuilder()
            .set_cpu("Intel i9-14900K")
            .set_ram(32)
            .set_storage(2000)
            .set_gpu("NVIDIA RTX 4090")
            .set_monitor("27\" 4K 144Hz")
            .add_peripheral("Gaming Keyboard")
            .add_peripheral("Gaming Mouse")
            .add_peripheral("RGB Headset")
            .build())
    
    @staticmethod
    def build_office_pc() -> Computer:
        """Create an office PC configuration."""
        return (ComputerBuilder()
            .set_cpu("Intel i5-13400")
            .set_ram(16)
            .set_storage(512)
            .add_peripheral("Standard Keyboard")
            .add_peripheral("Standard Mouse")
            .build())

# Example usage
if __name__ == "__main__":
    # Custom build with method chaining
    custom_pc = (ComputerBuilder()
        .set_cpu("AMD Ryzen 9 7950X")
        .set_ram(64)
        .set_storage(4000)
        .set_gpu("AMD Radeon RX 7900 XTX")
        .build())
    
    print("Custom PC:")
    print(custom_pc)
    print()
    
    # Predefined configuration using director
    gaming_pc = ComputerDirector.build_gaming_pc()
    print("Gaming PC:")
    print(gaming_pc)
```

**Why**: Builder pattern handles complex object construction with many optional parameters. Unlike telescoping constructors or massive `__init__` methods, it provides readable, chainable API. Validation can be deferred to `build()`, ensuring objects are always in valid state. Ideal for configuration objects, SQL query builders, or any complex construction logic.

## Token Bucket Rate Limiter

```python
# PROBLEM: Limit API requests using token bucket algorithm
# TIME: O(1) for both allow and replenish SPACE: O(1)
import time
import threading
from typing import Optional

class TokenBucket:
    """
    Token Bucket Rate Limiter.
    
    - Bucket holds tokens up to capacity
    - Tokens replenish at fixed rate (refill_rate per second)
    - Each request consumes one token
    - Request rejected if no token available
    """
    
    def __init__(self, capacity: int, refill_rate: float):
        """
        Initialize token bucket.
        
        Args:
            capacity: Maximum number of tokens in bucket
            refill_rate: Tokens added per second
        """
        self.capacity = capacity
        self.refill_rate = refill_rate  # tokens/second
        self.tokens = capacity  # Start full
        self.last_refill = time.time()
        self.lock = threading.Lock()
    
    def _refill(self) -> None:
        """Replenish tokens based on elapsed time."""
        now = time.time()
        elapsed = now - self.last_refill
        
        # Calculate tokens to add
        tokens_to_add = elapsed * self.refill_rate
        self.tokens = min(self.capacity, self.tokens + tokens_to_add)
        self.last_refill = now
    
    def allow_request(self, tokens: int = 1) -> bool:
        """
        Check if request is allowed.
        
        Args:
            tokens: Number of tokens to consume (default: 1)
        
        Returns:
            True if request allowed, False if rate limited
        """
        with self.lock:
            self._refill()
            
            if self.tokens >= tokens:
                self.tokens -= tokens
                return True
            return False
    
    def wait_for_token(self, tokens: int = 1, timeout: Optional[float] = None) -> bool:
        """
        Block until token available or timeout.
        
        Args:
            tokens: Number of tokens needed
            timeout: Maximum time to wait (None = wait forever)
        
        Returns:
            True if token acquired, False if timeout
        """
        start_time = time.time()
        
        while True:
            if self.allow_request(tokens):
                return True
            
            # Check timeout
            if timeout is not None:
                elapsed = time.time() - start_time
                if elapsed >= timeout:
                    return False
            
            # Calculate sleep time until next token available
            with self.lock:
                tokens_needed = tokens - self.tokens
                if tokens_needed > 0:
                    sleep_time = tokens_needed / self.refill_rate
                    time.sleep(min(sleep_time, 0.1))  # Check frequently

class RateLimiterRegistry:
    """Manage multiple rate limiters for different resources."""
    
    def __init__(self):
        self.limiters = {}
        self.lock = threading.Lock()
    
    def get_limiter(
        self,
        name: str,
        capacity: int,
        refill_rate: float
    ) -> TokenBucket:
        """Get or create a rate limiter by name."""
        with self.lock:
            if name not in self.limiters:
                self.limiters[name] = TokenBucket(capacity, refill_rate)
            return self.limiters[name]

# Example usage
if __name__ == "__main__":
    # API rate limiter: 10 requests per second, burst up to 20
    api_limiter = TokenBucket(capacity=20, refill_rate=10)
    
    # Simulate requests
    for i in range(25):
        if api_limiter.allow_request():
            print(f"Request {i+1}: Allowed")
        else:
            print(f"Request {i+1}: Rate limited!")
            break
    
    print("\nWaiting for token...")
    if api_limiter.wait_for_token(timeout=2.0):
        print("Token acquired after waiting!")
    
    # Registry example
    registry = RateLimiterRegistry()
    user_limiter = registry.get_limiter("user_api", capacity=100, refill_rate=5)
    search_limiter = registry.get_limiter("search_api", capacity=50, refill_rate=2)
```

**Why**: Token bucket allows controlled bursts while maintaining average rate limit. Unlike fixed window algorithms, it handles burst traffic naturally and provides smooth rate limiting. Widely used for API rate limiting, network traffic shaping, and preventing resource exhaustion. Thread-safe implementation ensures correctness in concurrent environments.

## Optimistic Locking

```python
# PROBLEM: Handle concurrent updates without database locks
# TIME: O(1) for validation SPACE: O(1)
import time
import threading
from dataclasses import dataclass
from typing import Optional, Dict
from datetime import datetime

@dataclass
class VersionedEntity:
    """Entity with version number for optimistic locking."""
    id: str
    data: Dict
    version: int = 1
    updated_at: Optional[datetime] = None

class OptimisticLockException(Exception):
    """Raised when optimistic lock validation fails."""
    pass

class DataStore:
    """
    Thread-safe data store with optimistic locking.
    
    Workflow:
    1. Read entity (get current version)
    2. Modify data locally
    3. Save with version check
    4. If version mismatch, retry or fail
    """
    
    def __init__(self):
        self.entities: Dict[str, VersionedEntity] = {}
        self.lock = threading.Lock()
    
    def read(self, entity_id: str) -> Optional[VersionedEntity]:
        """
        Read entity with current version.
        Returns copy to prevent external modification.
        """
        with self.lock:
            entity = self.entities.get(entity_id)
            if entity:
                # Return copy with same version
                return VersionedEntity(
                    id=entity.id,
                    data=entity.data.copy(),
                    version=entity.version,
                    updated_at=entity.updated_at
                )
            return None
    
    def update(self, entity: VersionedEntity) -> VersionedEntity:
        """
        Update entity with optimistic lock validation.
        
        Raises:
            OptimisticLockException: if version mismatch (concurrent update)
        """
        with self.lock:
            current = self.entities.get(entity.id)
            
            # New entity
            if current is None:
                entity.version = 1
                entity.updated_at = datetime.now()
                self.entities[entity.id] = entity
                return entity
            
            # Optimistic lock check
            if current.version != entity.version:
                raise OptimisticLockException(
                    f"Version mismatch: expected {current.version}, "
                    f"got {entity.version}. Entity was modified by another transaction."
                )
            
            # Update with incremented version
            entity.version = current.version + 1
            entity.updated_at = datetime.now()
            self.entities[entity.id] = entity
            return entity
    
    def delete(self, entity_id: str, expected_version: int) -> bool:
        """Delete with version check."""
        with self.lock:
            current = self.entities.get(entity_id)
            if current and current.version == expected_version:
                del self.entities[entity_id]
                return True
            return False

def update_with_retry(
    store: DataStore,
    entity_id: str,
    update_func,
    max_retries: int = 3
) -> VersionedEntity:
    """
    Update entity with automatic retry on conflict.
    
    Args:
        store: Data store instance
        entity_id: Entity to update
        update_func: Function that modifies entity data
        max_retries: Maximum retry attempts
    
    Returns:
        Updated entity
    """
    for attempt in range(max_retries):
        # Read current state
        entity = store.read(entity_id)
        if entity is None:
            raise ValueError(f"Entity {entity_id} not found")
        
        # Apply update
        update_func(entity.data)
        
        # Try to save
        try:
            return store.update(entity)
        except OptimisticLockException:
            if attempt == max_retries - 1:
                raise
            time.sleep(0.1 * (attempt + 1))  # Exponential backoff
    
    raise RuntimeError("Max retries exceeded")

# Example usage
if __name__ == "__main__":
    store = DataStore()
    
    # Create initial entity
    entity = VersionedEntity(id="user:123", data={"name": "Alice", "email": "alice@example.com"})
    store.update(entity)
    
    # Successful update
    entity = store.read("user:123")
    entity.data["email"] = "alice.new@example.com"
    updated = store.update(entity)
    print(f"Updated: version {updated.version}, data: {updated.data}")
    
    # Simulate concurrent update (version mismatch)
    stale_entity = VersionedEntity(
        id="user:123",
        data={"name": "Alice", "email": "old@example.com"},
        version=1  # Stale version!
    )
    
    try:
        store.update(stale_entity)
    except OptimisticLockException as e:
        print(f"Lock exception: {e}")
    
    # Update with automatic retry
    def add_login_count(data):
        data["login_count"] = data.get("login_count", 0) + 1
    
    result = update_with_retry(store, "user:123", add_login_count)
    print(f"After retry update: {result.data}")
```

**Why**: Optimistic locking avoids database locks by validating versions at commit time. Ideal for low-contention scenarios where conflicts are rare. Unlike pessimistic locking, it doesn't block readers and scales better. Common in distributed systems, NoSQL databases, and web applications. Automatic retry with exponential backoff handles transient conflicts gracefully.

## Python Dataclasses

```python
# PROBLEM: Reduce boilerplate for data-holding classes
# TIME: O(1) for creation SPACE: O(n) for instance attributes
from dataclasses import dataclass, field, asdict, astuple
from typing import List, Optional, ClassVar
from datetime import datetime

@dataclass
class Product:
    """Basic dataclass with automatic __init__, __repr__, __eq__."""
    name: str
    price: float
    quantity: int = 1  # Default value
    
    def total_cost(self) -> float:
        """Instance method works normally."""
        return self.price * self.quantity

@dataclass(order=True)  # Enable comparison operators
class Priority:
    """Dataclass with ordering support (<, >, <=, >=)."""
    level: int
    name: str = field(compare=False)  # Exclude from comparison

@dataclass
class Order:
    """
    Advanced dataclass with various features.
    Demonstrates field customization and class variables.
    """
    # Class variable (not instance field)
    tax_rate: ClassVar[float] = 0.08
    
    # Fields with custom defaults
    order_id: int
    customer: str
    items: List[Product] = field(default_factory=list)
    created_at: datetime = field(default_factory=datetime.now)
    discount: float = 0.0
    notes: Optional[str] = None
    
    # Exclude from repr
    internal_id: str = field(repr=False, default="")
    
    # Custom property
    @property
    def subtotal(self) -> float:
        """Calculate subtotal before tax and discount."""
        return sum(item.total_cost() for item in self.items)
    
    @property
    def total(self) -> float:
        """Calculate final total with tax and discount."""
        return self.subtotal * (1 + self.tax_rate) * (1 - self.discount)
    
    def add_item(self, product: Product) -> None:
        """Add item to order."""
        self.items.append(product)
    
    # Post-init processing
    def __post_init__(self):
        """Validate and process after __init__."""
        if self.discount < 0 or self.discount > 1:
            raise ValueError("Discount must be between 0 and 1")
        if not self.internal_id:
            self.internal_id = f"ORD-{self.order_id}"

@dataclass(frozen=True)  # Immutable dataclass
class Address:
    """Immutable address - hashable, can be dict key."""
    street: str
    city: str
    country: str
    postal_code: str
    
    def __str__(self) -> str:
        return f"{self.street}, {self.city} {self.postal_code}, {self.country}"

# Inheritance with dataclasses
@dataclass
class DigitalProduct(Product):
    """Dataclass inheriting from another dataclass."""
    download_url: str
    file_size_mb: float
    license_key: Optional[str] = None

# Example usage
if __name__ == "__main__":
    # Basic dataclass
    laptop = Product(name="Laptop", price=999.99, quantity=2)
    print(f"{laptop.name}: ${laptop.total_cost():.2f}")
    
    # Ordered dataclass
    high = Priority(3, "High")
    low = Priority(1, "Low")
    print(f"high > low: {high > low}")
    
    # Complex dataclass with nested dataclasses
    order = Order(
        order_id=1001,
        customer="Alice",
        discount=0.1
    )
    order.add_item(Product("Mouse", 29.99, 2))
    order.add_item(Product("Keyboard", 79.99, 1))
    
    print(f"\nOrder {order.order_id}:")
    print(f"Subtotal: ${order.subtotal:.2f}")
    print(f"Total (with tax & discount): ${order.total:.2f}")
    
    # Immutable dataclass
    addr = Address("123 Main St", "San Francisco", "USA", "94105")
    print(f"\nAddress: {addr}")
    # addr.city = "LA"  # FrozenInstanceError!
    
    # Conversion utilities
    order_dict = asdict(order)
    order_tuple = astuple(order)
    
    # Inheritance
    ebook = DigitalProduct(
        name="Python Guide",
        price=29.99,
        download_url="https://example.com/book.pdf",
        file_size_mb=15.5
    )
    print(f"\nDigital product: {ebook.name}")
```

**Why**: Dataclasses eliminate boilerplate code for data-holding classes. They auto-generate `__init__`, `__repr__`, `__eq__`, and optionally comparison methods. Features like default factories, field customization, and frozen instances handle most use cases. More readable than named tuples, more features than regular classes with less code. Ideal for DTOs, configuration objects, API responses, and any data structure.
