## 1. js_proxy_getOwnPropertyDescriptor

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Proxy getOwnPropertyDescriptor Demo</title>
</head>
<body>
  <h2>Proxy getOwnPropertyDescriptor Trap</h2>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Intercept property descriptor requests on a proxy
    // TIME: O(1) SPACE: O(1)
    
    /**
     * Why: The getOwnPropertyDescriptor trap intercepts Object.getOwnPropertyDescriptor()
     * calls, allowing custom behavior when property descriptors are queried.
     * Useful for hiding private properties or providing virtual descriptors.
     */
    
    const target = {
      _secret: 'hidden value',
      publicData: 'visible data',
      count: 42
    };

    const handler = {
      // Trap: Intercept property descriptor requests
      getOwnPropertyDescriptor(targetObj, prop) {
        // Hide properties starting with underscore (private convention)
        if (prop.startsWith('_')) {
          console.log(`[Trap] Hiding descriptor for: ${prop}`);
          return undefined; // Property doesn't exist as far as caller knows
        }

        // Get the actual descriptor
        const descriptor = Object.getOwnPropertyDescriptor(targetObj, prop);
        
        if (descriptor) {
          console.log(`[Trap] Returning descriptor for: ${prop}`);
          // Optionally modify the descriptor
          // Make all visible properties non-enumerable
          return {
            ...descriptor,
            enumerable: true,
            configurable: true
          };
        }

        return undefined;
      }
    };

    const proxy = new Proxy(target, handler);

    // Demo: Query property descriptors
    const output = document.getElementById('output');
    const results = [];

    results.push('=== getOwnPropertyDescriptor Demo ===\n');

    // Try to get descriptor for public property
    const publicDesc = Object.getOwnPropertyDescriptor(proxy, 'publicData');
    results.push(`publicData descriptor: ${JSON.stringify(publicDesc, null, 2)}`);

    // Try to get descriptor for private property (will return undefined)
    const privateDesc = Object.getOwnPropertyDescriptor(proxy, '_secret');
    results.push(`_secret descriptor: ${privateDesc}`);

    // Check if property "exists" using descriptor
    results.push(`\n'secret' in proxy: ${Reflect.has(proxy, '_secret')}`);

    // List all "own" property keys (uses ownKeys trap if defined)
    results.push(`\nVisible keys: ${Object.keys(proxy)}`);

    output.textContent = results.join('\n');
  </script>
</body>
</html>
```

## 2. js_proxy_defineProperty

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Proxy defineProperty Demo</title>
</head>
<body>
  <h2>Proxy defineProperty Trap</h2>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Intercept and control property definitions on a proxy
    // TIME: O(1) SPACE: O(1)
    
    /**
     * Why: The defineProperty trap intercepts Object.defineProperty() calls,
     * enabling validation, logging, or transformation of property definitions.
     * Essential for creating validated objects or implementing reactive systems.
     */
    
    const target = {};

    const handler = {
      // Trap: Intercept property definitions
      defineProperty(targetObj, prop, descriptor) {
        console.log(`[Trap] Attempting to define: ${prop}`);
        
        // Validation: Prevent defining properties starting with underscore
        if (prop.startsWith('_')) {
          console.log(`[Trap] Rejected: Private properties not allowed via proxy`);
          return false; // Throws TypeError in strict mode
        }

        // Validation: Ensure value types for specific properties
        if (prop === 'age' && descriptor.value !== undefined) {
          if (typeof descriptor.value !== 'number' || descriptor.value < 0) {
            console.log(`[Trap] Rejected: Age must be a positive number`);
            return false;
          }
        }

        // Validation: Ensure email has valid format
        if (prop === 'email' && descriptor.value !== undefined) {
          const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
          if (!emailRegex.test(descriptor.value)) {
            console.log(`[Trap] Rejected: Invalid email format`);
            return false;
          }
        }

        // Log the definition
        console.log(`[Trap] Allowed: Defining ${prop}`);
        
        // Actually define the property on target
        return Object.defineProperty(targetObj, prop, descriptor);
      }
    };

    const proxy = new Proxy(target, handler);

    // Demo: Define properties with validation
    const output = document.getElementById('output');
    const results = [];

    results.push('=== defineProperty Demo ===\n');

    // Valid: Define a string property
    Object.defineProperty(proxy, 'name', {
      value: 'Alice',
      writable: true,
      enumerable: true,
      configurable: true
    });
    results.push(`name defined: ${proxy.name}`);

    // Valid: Define age with number
    Object.defineProperty(proxy, 'age', {
      value: 25,
      writable: true
    });
    results.push(`age defined: ${proxy.age}`);

    // Invalid: Try to define negative age
    try {
      Object.defineProperty(proxy, 'age', { value: -5 });
    } catch (e) {
      results.push(`\nRejected negative age: ${e.message}`);
    }

    // Valid: Define email with correct format
    Object.defineProperty(proxy, 'email', {
      value: 'alice@example.com'
    });
    results.push(`email defined: ${proxy.email}`);

    // Invalid: Try to define invalid email
    try {
      Object.defineProperty(proxy, 'email', { value: 'not-an-email' });
    } catch (e) {
      results.push(`\nRejected invalid email: ${e.message}`);
    }

    // Invalid: Try to define private property
    try {
      Object.defineProperty(proxy, '_private', { value: 'secret' });
    } catch (e) {
      results.push(`\nRejected private property: ${e.message}`);
    }

    results.push(`\nFinal object: ${JSON.stringify(proxy, null, 2)}`);

    output.textContent = results.join('\n');
  </script>
</body>
</html>
```

## 3. js_generator_done

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Generator done Property Demo</title>
</head>
<body>
  <h2>Generator done Property</h2>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Track generator completion status using the done property
    // TIME: O(1) per iteration SPACE: O(1)
    
    /**
     * Why: The done property in iterator results indicates whether iteration
     * is complete. Essential for manually iterating generators or implementing
     * custom iteration logic with proper termination handling.
     */
    
    // Generator function that yields multiple values
    function* numberGenerator(max) {
      for (let i = 1; i <= max; i++) {
        yield i;
      }
      // Generator automatically returns { value: undefined, done: true }
    }

    // Generator with explicit return value
    function* generatorWithReturn() {
      yield 'first';
      yield 'second';
      return 'final'; // This becomes the last value when done: true
    }

    // Generator that yields infinite sequence
    function* infiniteSequence() {
      let i = 0;
      while (true) {
        yield i++;
      }
    }

    // Demo
    const output = document.getElementById('output');
    const results = [];

    results.push('=== Generator done Property Demo ===\n');

    // Example 1: Basic iteration using done
    results.push('--- Basic Generator ---');
    const gen1 = numberGenerator(3);
    let iteration1 = gen1.next();
    
    while (!iteration1.done) {
      results.push(`value: ${iteration1.value}, done: ${iteration1.done}`);
      iteration1 = gen1.next();
    }
    results.push(`value: ${iteration1.value}, done: ${iteration1.done} (exhausted)`);

    // Example 2: Generator with return value
    results.push('\n--- Generator with Return ---');
    const gen2 = generatorWithReturn();
    let iteration2;
    
    do {
      iteration2 = gen2.next();
      results.push(`value: ${JSON.stringify(iteration2.value)}, done: ${iteration2.done}`);
    } while (!iteration2.done);

    // Example 3: Manual iteration with limit
    results.push('\n--- Infinite Sequence (limited) ---');
    const gen3 = infiniteSequence();
    
    for (let i = 0; i < 5; i++) {
      const result = gen3.next();
      results.push(`value: ${result.value}, done: ${result.done}`);
    }

    // Example 4: Check done without consuming
    results.push('\n--- Peeking at done status ---');
    const gen4 = numberGenerator(2);
    let peek = gen4.next();
    
    results.push(`First: value=${peek.value}, done=${peek.done}`);
    results.push(`Can continue: ${!peek.done}`);
    
    peek = gen4.next();
    results.push(`Second: value=${peek.value}, done=${peek.done}`);
    
    peek = gen4.next();
    results.push(`Third: value=${peek.value}, done=${peek.done}`);
    results.push(`Generator exhausted: ${peek.done}`);

    output.textContent = results.join('\n');
  </script>
</body>
```

## 4. js_generator_return

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Generator return Method Demo</title>
</head>
<body>
  <h2>Generator return() Method</h2>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Prematurely terminate a generator with a return value
    // TIME: O(1) SPACE: O(1)
    
    /**
     * Why: The return() method forcefully completes a generator, executing
     * any finally blocks. Useful for cleanup, cancellation, or early
     * termination of generator-based operations.
     */
    
    // Generator with try-finally for cleanup
    function* resourceGenerator() {
      try {
        yield 'Resource acquired';
        yield 'Processing step 1';
        yield 'Processing step 2';
        yield 'Processing step 3';
        return 'Normal completion';
      } finally {
        // This ALWAYS runs, even with early return()
        console.log('[Cleanup] Releasing resources...');
      }
    }

    // Generator simulating async operations
    function* taskRunner() {
      const tasks = ['Task A', 'Task B', 'Task C', 'Task D'];
      
      for (const task of tasks) {
        const shouldContinue = yield task;
        if (!shouldContinue) {
          return 'Cancelled by user';
        }
      }
      
      return 'All tasks completed';
    }

    // Demo
    const output = document.getElementById('output');
    const results = [];

    results.push('=== Generator return() Method Demo ===\n');

    // Example 1: Early termination triggers finally
    results.push('--- Early Termination with Cleanup ---');
    const gen1 = resourceGenerator();
    
    results.push(`Step 1: ${gen1.next().value}`);
    results.push(`Step 2: ${gen1.next().value}`);
    
    // Force early return
    const returnResult = gen1.return('Emergency stop!');
    results.push(`\nreturn() called:`);
    results.push(`  value: ${returnResult.value}`);
    results.push(`  done: ${returnResult.done}`);

    // Subsequent calls return same state
    results.push(`\nAfter return():`);
    results.push(`  next(): ${JSON.stringify(gen1.next())}`);

    // Example 2: Cancellation pattern
    results.push('\n--- Task Cancellation Pattern ---');
    const gen2 = taskRunner();
    
    let status;
    status = gen2.next(); // Start
    results.push(`Task: ${status.value}, done: ${status.done}`);
    
    status = gen2.next(true); // Continue
    results.push(`Task: ${status.value}, done: ${status.done}`);
    
    // Cancel
    status = gen2.return('User cancelled');
    results.push(`\nCancelled: ${status.value}, done: ${status.done}`);

    // Example 3: Using return for timeout simulation
    results.push('\n--- Timeout Pattern ---');
    function* dataFetch() {
      yield 'Connecting...';
      yield 'Authenticating...';
      yield 'Fetching data...';
      return 'Data received';
    }

    const gen3 = dataFetch();
    results.push(gen3.next().value);
    
    // Simulate timeout after 2 steps
    results.push(gen3.next().value);
    const timeout = gen3.return('Connection timed out');
    results.push(`\nTimeout: ${timeout.value}`);
    results.push(`Generator state - done: ${timeout.done}`);

    output.textContent = results.join('\n');
  </script>
```

## 5. js_function_prototype

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Function Prototype Demo</title>
</head>
<body>
  <h2>Function.prototype in JavaScript</h2>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Understand and utilize Function.prototype for method sharing
    // TIME: O(1) for property access SPACE: O(1)
    
    /**
     * Why: Every function in JavaScript has a prototype property used for
     * inheritance when the function is used as a constructor. Understanding
     * Function.prototype is essential for OOP and method sharing patterns.
     */
    
    // 1. Basic constructor with prototype methods
    function Animal(name, species) {
      this.name = name;
      this.species = species;
    }

    // Add methods to prototype (shared across all instances)
    Animal.prototype.speak = function() {
      return `${this.name} makes a sound`;
    };

    Animal.prototype.describe = function() {
      return `${this.name} is a ${this.species}`;
    };

    // 2. Extending prototype (inheritance)
    function Dog(name, breed) {
      Animal.call(this, name, 'Dog');
      this.breed = breed;
    }

    // Set up inheritance chain
    Dog.prototype = Object.create(Animal.prototype);
    Dog.prototype.constructor = Dog;

    // Override parent method
    Dog.prototype.speak = function() {
      return `${this.name} barks: Woof!`;
    };

    // Add new method
    Dog.prototype.fetch = function() {
      return `${this.name} fetches the ball`;
    };

    // 3. Adding methods to built-in prototypes (use with caution!)
    // Only for demonstration - extending native prototypes is controversial
    Function.prototype.describe = function() {
      return `Function: ${this.name || 'anonymous'}`;
    };

    // Demo
    const output = document.getElementById('output');
    const results = [];

    results.push('=== Function.prototype Demo ===\n');

    // Example 1: Basic prototype usage
    results.push('--- Basic Constructor ---');
    const genericAnimal = new Animal('Generic', 'Unknown');
    results.push(genericAnimal.describe());
    results.push(genericAnimal.speak());

    // Example 2: Inheritance
    results.push('\n--- Inheritance ---');
    const buddy = new Dog('Buddy', 'Golden Retriever');
    results.push(buddy.describe());
    results.push(buddy.speak());
    results.push(buddy.fetch());

    // Example 3: Prototype chain inspection
    results.push('\n--- Prototype Chain ---');
    results.push(`buddy instanceof Dog: ${buddy instanceof Dog}`);
    results.push(`buddy instanceof Animal: ${buddy instanceof Animal}`);
    results.push(`buddy.hasOwnProperty('name'): ${buddy.hasOwnProperty('name')}`);

    results.push('\n--- Shared Methods ---');
    const dog1 = new Dog('Max', 'Beagle');

    results.push(`dog1.speak === dog2.speak: ${dog1.speak === dog2.speak}`);

    // Example 5: Function.prototype methods
```

## 6. js_constructor_pattern

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Constructor Pattern Demo</title>
</head>
<body>
  <h2>Constructor Pattern in JavaScript</h2>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Create multiple objects with the same structure using constructors
    // TIME: O(1) for creation SPACE: O(n) for n instances
    
    /**
     * Why: The constructor pattern is fundamental for creating multiple objects
     * with shared behavior. It combines constructor functions with prototypes
     * for memory-efficient method sharing.
     */
    
    // Pattern 1: Traditional Constructor Function
    function User(name, email, age) {
      // Instance properties (unique to each instance)
      this.name = name;
      this.email = email;
      this.age = age;
      this.createdAt = new Date();
      
      // Private-like variables (not on this)
      let _password = '';
      this.setPassword = function(pwd) {
        _password = pwd;
      };
      this.checkPassword = function(pwd) {
        return _password === pwd;
      };
    }

    // Prototype methods (shared across instances)
    User.prototype.getDisplayName = function() {
      return `${this.name} <${this.email}>`;
    };

    User.prototype.isAdult = function() {
      return this.age >= 18;
    };

    User.prototype.getAccountAge = function() {
      const now = new Date();
      const diff = now - this.createdAt;
      return Math.floor(diff / (1000 * 60 * 60 * 24)); // Days
    };

    // Pattern 2: Constructor with Validation
    function BankAccount(accountNumber, initialBalance, owner) {
      // Validation in constructor
      if (!accountNumber || accountNumber.length < 8) {
        throw new Error('Invalid account number');
      }
      if (initialBalance < 0) {
        throw new Error('Initial balance cannot be negative');
      }
      
      this.accountNumber = accountNumber;
      this.balance = initialBalance;
      this.owner = owner;
      this.transactions = [];
      
      // Record initial deposit
      if (initialBalance > 0) {
        this.transactions.push({
          type: 'DEPOSIT',
          amount: initialBalance,
          date: new Date(),
          description: 'Initial deposit'
        });
      }
    }

    BankAccount.prototype.deposit = function(amount) {
      if (amount <= 0) throw new Error('Amount must be positive');
      this.balance += amount;
      this.transactions.push({
        type: 'DEPOSIT',
        amount,
        date: new Date()
      });
      return this.balance;
    };

    BankAccount.prototype.withdraw = function(amount) {
      if (amount <= 0) throw new Error('Amount must be positive');
      if (amount > this.balance) throw new Error('Insufficient funds');
      this.balance -= amount;
      this.transactions.push({
        type: 'WITHDRAWAL',
        amount,
        date: new Date()
      });
      return this.balance;
    };

    BankAccount.prototype.getStatement = function() {
      return this.transactions.map(t => 
        `${t.date.toISOString().split('T')[0]} | ${t.type}: $${t.amount}`
      ).join('\n');
    };

    // Pattern 3: ES6 Class (syntactic sugar for constructor pattern)
    class Product {
      constructor(name, price, category) {
        this.name = name;
        this.price = price;
        this.category = category;
        this.inStock = true;
      }

      applyDiscount(percent) {
        this.price *= (1 - percent / 100);
        return this.price;
      }

      toString() {
        return `${this.name} ($${this.price.toFixed(2)})`;
      }
    }

    // Demo
    const output = document.getElementById('output');
    const results = [];

    results.push('=== Constructor Pattern Demo ===\n');

    // Example 1: User constructor
    results.push('--- User Constructor ---');
    const user1 = new User('Alice', 'alice@example.com', 25);
    const user2 = new User('Bob', 'bob@example.com', 17);
    
    user1.setPassword('secret123');
    
    results.push(user1.getDisplayName());
    results.push(`Is adult: ${user1.isAdult()}`);
    results.push(`Password check: ${user1.checkPassword('secret123')}`);
    results.push(`\n${user2.getDisplayName()}`);
    results.push(`Is adult: ${user2.isAdult()}`);

    // Example 2: BankAccount with validation
    results.push('\n--- BankAccount Constructor ---');
    try {
      const account = new BankAccount('12345678', 1000, 'Alice');
      results.push(`Account: ${account.accountNumber}`);
      results.push(`Owner: ${account.owner}`);
      results.push(`Balance: $${account.balance}`);
      
      account.deposit(500);
      account.withdraw(200);
      
      results.push(`\nNew balance: $${account.balance}`);
      results.push('\nTransaction History:');
      results.push(account.getStatement());
    } catch (e) {
      results.push(`Error: ${e.message}`);
    }

    // Example 3: ES6 Class syntax
    results.push('\n--- ES6 Class (Constructor Pattern) ---');
    const laptop = new Product('Laptop', 999.99, 'Electronics');
    results.push(laptop.toString());
    laptop.applyDiscount(10);
    results.push(`After 10% discount: $${laptop.price.toFixed(2)}`);

    // Example 4: instanceof check
    results.push('\n--- Type Checking ---');
    results.push(`user1 instanceof User: ${user1 instanceof User}`);
    results.push(`laptop instanceof Product: ${laptop instanceof Product}`);
    results.push(`User.prototype.isPrototypeOf(user1): ${User.prototype.isPrototypeOf(user1)}`);

    output.textContent = results.join('\n');
  </script>
</body>
</html>
```

## 7. js_closure_module_pattern

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Closure Module Pattern Demo</title>
</head>
<body>
  <h2>Closure Module Pattern</h2>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Create private state and expose controlled public API
    // TIME: O(1) for method calls SPACE: O(n) for encapsulated data
    
    /**
     * Why: The module pattern uses closures to create private state that
     * cannot be accessed directly from outside. Only the exposed methods
     * can interact with the private data, ensuring encapsulation.
     */
    
    // Pattern 1: Basic Module Pattern (IIFE)
    const Counter = (function() {
      // Private state
      let count = 0;
      const history = [];
      
      // Private helper
      function log(action) {
        history.push({
          action,
          timestamp: new Date(),
          value: count
        });
      }
      
      // Public API
      return {
        increment() {
          count++;
          log('increment');
          return count;
        },
        decrement() {
          count--;
          log('decrement');
          return count;
        },
        getValue() {
          return count;
        },
        reset() {
          count = 0;
          log('reset');
        },
        getHistory() {
          return [...history]; // Return copy, not reference
        }
      };
    })();

    // Pattern 2: Module Factory (creates new module instances)
    function createShoppingCart() {
      // Private state
      const items = [];
      
      // Private methods
      function findItemIndex(id) {
        return items.findIndex(item => item.id === id);
      }
      
      function calculateTotal() {
        return items.reduce((sum, item) => sum + (item.price * item.quantity), 0);
      }
      
      // Public API
      return {
        addItem(id, name, price, quantity = 1) {
          const existingIndex = findItemIndex(id);
          if (existingIndex >= 0) {
            items[existingIndex].quantity += quantity;
          } else {
            items.push({ id, name, price, quantity });
          }
          return this.getItemCount();
        },
        
        removeItem(id) {
          const index = findItemIndex(id);
          if (index >= 0) {
            items.splice(index, 1);
            return true;
          }
          return false;
        },
        
        updateQuantity(id, quantity) {
          const index = findItemIndex(id);
          if (index >= 0) {
            if (quantity <= 0) {
              return this.removeItem(id);
            }
            items[index].quantity = quantity;
            return true;
          }
          return false;
        },
        
        getItemCount() {
          return items.reduce((sum, item) => sum + item.quantity, 0);
        },
        
        getTotal() {
          return calculateTotal();
        },
        
        getItems() {
          return items.map(item => ({ ...item })); // Return copies
        },
        
        clear() {
          items.length = 0;
        }
      };
    }

    // Pattern 3: Revealing Module Pattern
    const ConfigManager = (function() {
      // Private
      const config = {};
      const watchers = {};
      
      function set(key, value) {
        const oldValue = config[key];
        config[key] = value;
        
        // Notify watchers
        if (watchers[key]) {
          watchers[key].forEach(callback => callback(value, oldValue));
        }
      }
      
      function get(key, defaultValue = undefined) {
        return key in config ? config[key] : defaultValue;
      }
      
      function remove(key) {
        delete config[key];
        delete watchers[key];
      }
      
      function watch(key, callback) {
        if (!watchers[key]) {
          watchers[key] = [];
        }
        watchers[key].push(callback);
        
        // Return unsubscribe function
        return function unwatch() {
          const index = watchers[key].indexOf(callback);
          if (index >= 0) watchers[key].splice(index, 1);
        };
      }
```

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>IIFE Closure Demo</title>
</head>
<body>
  <h2>Immediately Invoked Function Expressions (IIFE)</h2>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Create isolated scope and execute code immediately
    // TIME: O(n) where n is IIFE body SPACE: O(1) for scope creation
    
    /**
     * Why: IIFEs create a new scope, preventing variable leakage to the
     * global namespace. They're useful for initialization, creating private
     * variables, and avoiding global pollution in modular code.
     */
    
    // Basic IIFE syntax variations
    // (function() { })();  // Classic
    // (() => { })();       // Arrow function
    // (function() { }());  // Douglas Crockford style
    // !function() { }();   // Unary operator
    
    const output = document.getElementById('output');
    const results = [];

    results.push('=== IIFE Closure Demo ===\n');

    // Example 1: Basic IIFE for scope isolation
    results.push('--- Scope Isolation ---');
    
    (function() {
      const privateVar = 'I am private';
      results.push(`Inside IIFE: ${privateVar}`);
    })();
    
    results.push(`Outside IIFE: privateVar is ${typeof privateVar}`);

    // Example 2: IIFE with parameters
    results.push('\n--- IIFE with Parameters ---');
    
    const result = (function(a, b) {
      return a + b;
    })(5, 3);
    
    results.push(`IIFE returned: ${result}`);

    // Example 3: IIFE for initialization
    results.push('\n--- Initialization Pattern ---');
    
    const AppConfig = (function() {
      // Private configuration
      let config = {
        apiUrl: 'https://api.example.com',
        timeout: 5000,
        retries: 3
      };
      
      // Initialize (runs immediately)
      console.log('[IIFE] Initializing app config...');
      
      // Return public interface
      return {
        get: (key) => config[key],
        set: (key, value) => { config[key] = value; },
        getAll: () => ({ ...config })
      };
    })();
    
    results.push(`API URL: ${AppConfig.get('apiUrl')}`);
    AppConfig.set('debug', true);
    results.push(`Debug mode: ${AppConfig.get('debug')}`);

    // Example 4: IIFE for avoiding closure pitfalls in loops
    results.push('\n--- Loop Closure Fix ---');
    
    // Wrong way (closure captures loop variable)
    results.push('Without IIFE (prints 5,5,5,5,5):');
    const wrong = [];
    for (var i = 0; i < 5; i++) {
      wrong.push(() => i);
    }
    results.push(`  [${wrong.map(f => f()).join(', ')}]`);
    
    // Correct way with IIFE
    results.push('With IIFE (prints 0,1,2,3,4):');
    const correct = [];
    for (var j = 0; j < 5; j++) {
      correct.push((function(n) {
        return () => n;
      })(j));
    }
    results.push(`  [${correct.map(f => f()).join(', ')}]`);
    
    // Modern alternative: let instead of var
    results.push('With let (modern approach):');
    const modern = [];
    for (let k = 0; k < 5; k++) {
      modern.push(() => k);
    }
    results.push(`  [${modern.map(f => f()).join(', ')}]`);

    // Example 5: IIFE for library wrapping
    results.push('\n--- Library Wrapper Pattern ---');
    
    const MyLibrary = (function(window, document) {
      // Private
      const version = '1.0.0';
      
      function privateHelper(msg) {
        return `[${version}] ${msg}`;
      }
      
      // Public API
      return {
        log(message) {
          console.log(privateHelper(message));
          return privateHelper(message);
        },
        getVersion() {
          return version;
        }
      };
    })(window, document); // Pass globals as parameters
    
    results.push(MyLibrary.log('Hello from library'));
    results.push(`Version: ${MyLibrary.getVersion()}`);

    // Example 6: IIFE with async
    results.push('\n--- Async IIFE ---');
    
    (async function() {
      const data = await Promise.resolve('Async data loaded');
      results.push(`Async IIFE result: ${data}`);
      output.textContent = results.join('\n');
    })();

    // Note: Output is updated after async completes
    setTimeout(() => {
      output.textContent = results.join('\n');
    }, 100);
  </script>
</body>
</html>
```

```html
    /**
     * Why: Math.sqrt() returns the square root of a number. Essential for

     */
    
    const output = document.getElementById('output');
    const results = [];

    results.push('=== Math.sqrt() Demo ===\n');

    // Example 1: Basic usage

    // Example 3: Euclidean distance
    results.push('\n--- Euclidean Distance ---');
    function distance(x1, y1, x2, y2) {
      const dx = x2 - x1;
      const dy = y2 - y1;
      return Math.sqrt(dx * dx + dy * dy);
    }
    
    results.push(`Distance (0,0) to (3,4): ${distance(0, 0, 3, 4)}`);
    results.push(`Distance (0,0) to (1,1): ${distance(0, 0, 1, 1).toFixed(4)}`);
    results.push(`Distance (1,2) to (4,6): ${distance(1, 2, 4, 6)}`);

    // Example 4: 3D distance
    results.push('\n--- 3D Distance ---');
    function distance3D(p1, p2) {
      const dx = p2.x - p1.x;
      const dy = p2.y - p1.y;
      const dz = p2.z - p1.z;
      return Math.sqrt(dx * dx + dy * dy + dz * dz);
    }
    
    const point1 = { x: 0, y: 0, z: 0 };
    const point2 = { x: 1, y: 2, z: 2 };
    results.push(`3D distance: ${distance3D(point1, point2).toFixed(4)}`);

    // Example 5: Vector magnitude (length)
    results.push('\n--- Vector Magnitude ---');
    function magnitude(vector) {
      return Math.sqrt(
        vector.reduce((sum, component) => sum + component * component, 0)
      );
    }
    
    results.push(`| [3, 4] | = ${magnitude([3, 4])}`);
    results.push(`| [1, 1, 1] | = ${magnitude([1, 1, 1]).toFixed(4)}`);
    results.push(`| [1, 2, 3, 4] | = ${magnitude([1, 2, 3, 4]).toFixed(4)}`);

    function standardDeviation(arr) {
      const n = arr.length;
      const mean = arr.reduce((a, b) => a + b, 0) / n;
      const squaredDiffs = arr.map(x => Math.pow(x - mean, 2));
      const variance = squaredDiffs.reduce((a, b) => a + b, 0) / n;
      return Math.sqrt(variance);
    }
    
    const data = [2, 4, 4, 4, 5, 5, 7, 9];

    function solveQuadratic(a, b, c) {
      const discriminant = b * b - 4 * a * c;
      
      if (discriminant < 0) {
        return { real: false, solutions: [] };
      }
      
      const sqrtDisc = Math.sqrt(discriminant);
      const x1 = (-b + sqrtDisc) / (2 * a);
      const x2 = (-b - sqrtDisc) / (2 * a);
      
      return { 
        real: true, 
        solutions: discriminant === 0 ? [x1] : [x1, x2] 
      };
    }
    
    // x² - 5x + 6 = 0 → x = 2 or x = 3
    const q1 = solveQuadratic(1, -5, 6);
    results.push(`x² - 5x + 6 = 0: x = ${q1.solutions.join(', ')}`);
    
    // x² + 1 = 0 (no real solutions)
    const q2 = solveQuadratic(1, 0, 1);
    results.push(`x² + 1 = 0: ${q2.real ? q2.solutions : 'No real solutions'}`);

    output.textContent = results.join('\n');
  </script>
</body>
</html>
```
