## Greedy - Jump Game
```python
# PROBLEM: Determine if you can reach the last index
# TIME: O(n) SPACE: O(1)

def can_jump(nums: list) -> bool:
    """
    Check if you can reach the last index by jumping.
    
    Args:
        nums: List where nums[i] is max jump length
    
    Returns:
        True if last index is reachable
    """
    max_reach = 0
    
    for i, jump in enumerate(nums):
        # Cannot reach current position
        if i > max_reach:
            return False
        
        # Update furthest reachable
        max_reach = max(max_reach, i + jump)
        
        # Already can reach end
        if max_reach >= len(nums) - 1:
            return True
    
    return True
```

**Why**: Greedy tracks maximum reachable index at each step, O(n) single pass.

## Topological Sort - Course Schedule
```python
# PROBLEM: Determine if all courses can be finished
# TIME: O(V + E) SPACE: O(V + E)

from collections import defaultdict, deque

def can_finish(num_courses: int, prerequisites: list) -> bool:
    """
    Check if all courses can be completed given prerequisites.
    
    Args:
        num_courses: Total number of courses
        prerequisites: List of [course, prerequisite] pairs
    
    Returns:
        True if all courses can be finished
    """
    # Build graph and indegree count
    graph = defaultdict(list)
    indegree = [0] * num_courses
    
    for course, prereq in prerequisites:
        graph[prereq].append(course)
        indegree[course] += 1
    
    # Start with courses having no prerequisites
    queue = deque([i for i in range(num_courses) if indegree[i] == 0])
    completed = 0
    
    while queue:
        course = queue.popleft()
        completed += 1
        
        # Reduce indegree for dependent courses
        for next_course in graph[course]:
            indegree[next_course] -= 1
            if indegree[next_course] == 0:
                queue.append(next_course)
    
    return completed == num_courses
```

**Why**: Kahn's algorithm uses BFS to process nodes with zero indegree, detecting cycles.

## Segment Tree - Range Sum Query
```python
# PROBLEM: Query sum of range with updates
# TIME: O(log n) query/update SPACE: O(n)

class SegmentTree:
    """
    Segment tree for range sum queries with updates.
    """
    
    def __init__(self, nums: list):
        self.n = len(nums)
        self.tree = [0] * (4 * self.n)
        self._build(nums, 0, 0, self.n - 1)
    
    def _build(self, nums: list, node: int, start: int, end: int) -> None:
        """Build segment tree recursively."""
        if start == end:
            self.tree[node] = nums[start]
            return
        
        mid = (start + end) // 2
        left_child = 2 * node + 1
        right_child = 2 * node + 2
        
        self._build(nums, left_child, start, mid)
        self._build(nums, right_child, mid + 1, end)
        
        self.tree[node] = self.tree[left_child] + self.tree[right_child]
    
    def update(self, index: int, value: int) -> None:
        """Update value at index."""
        self._update(0, 0, self.n - 1, index, value)
    
    def _update(self, node: int, start: int, end: int, 
                index: int, value: int) -> None:
        if start == end:
            self.tree[node] = value
            return
        
        mid = (start + end) // 2
        
        if index <= mid:
            self._update(2 * node + 1, start, mid, index, value)
        else:
            self._update(2 * node + 2, mid + 1, end, index, value)
        
        self.tree[node] = self.tree[2 * node + 1] + self.tree[2 * node + 2]
    
    def query(self, left: int, right: int) -> int:
        """Query sum in range [left, right]."""
        return self._query(0, 0, self.n - 1, left, right)
    
    def _query(self, node: int, start: int, end: int,
               left: int, right: int) -> int:
        # No overlap
        if right < start or left > end:
            return 0
        
        # Complete overlap
        if left <= start and end <= right:
            return self.tree[node]
        
        # Partial overlap
        mid = (start + end) // 2
        
        left_sum = self._query(2 * node + 1, start, mid, left, right)
        right_sum = self._query(2 * node + 2, mid + 1, end, left, right)
        
        return left_sum + right_sum
```

**Why**: Segment tree enables O(log n) range queries and updates for associative operations.

## Disjoint Set (Union-Find)
```python
# PROBLEM: Efficient union and find operations
# TIME: O(α(n)) amortized SPACE: O(n)

class DisjointSet:
    """
    Union-Find with path compression and union by rank.
    """
    
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank = [0] * n
    
    def find(self, x: int) -> int:
        """Find root with path compression."""
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x: int, y: int) -> bool:
        """
        Union two sets.
        
        Returns:
            True if merged, False if already in same set
        """
        px, py = self.find(x), self.find(y)
        
        if px == py:
            return False
        
        # Union by rank
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        
        self.parent[py] = px
        
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        return True
    
    def connected(self, x: int, y: int) -> bool:
        """Check if x and y are in same set."""
        return self.find(x) == self.find(y)
```

**Why**: Path compression and union by rank achieve near O(1) amortized time.

## Command Pattern
```python
# PROBLEM: Encapsulate requests as objects
# TIME: O(1) SPACE: O(n) for command history

from abc import ABC, abstractmethod

class Command(ABC):
    """Abstract command interface."""
    
    @abstractmethod
    def execute(self) -> None:
        pass
    
    @abstractmethod
    def undo(self) -> None:
        pass


class Light:
    """Receiver - the object being controlled."""
    
    def turn_on(self) -> None:
        print("Light is ON")
    
    def turn_off(self) -> None:
        print("Light is OFF")


class LightOnCommand(Command):
    """Concrete command to turn light on."""
    
    def __init__(self, light: Light):
        self.light = light
    
    def execute(self) -> None:
        self.light.turn_on()
    
    def undo(self) -> None:
        self.light.turn_off()


class RemoteControl:
    """Invoker - triggers commands."""
    
    def __init__(self):
        self.history = []
    
    def execute_command(self, command: Command) -> None:
        command.execute()
        self.history.append(command)
    
    def undo_last(self) -> None:
        if self.history:
            self.history.pop().undo()
```

**Why**: Command pattern decouples sender from receiver, enables undo/redo.

## State Pattern
```python
# PROBLEM: Change behavior based on internal state
# TIME: O(1) SPACE: O(1)

from abc import ABC, abstractmethod

class State(ABC):
    """Abstract state interface."""
    
    @abstractmethod
    def handle(self) -> str:
        pass


class RedState(State):
    def handle(self) -> str:
        return "Stop! Next: Green"


class YellowState(State):
    def handle(self) -> str:
        return "Caution! Next: Red"


class GreenState(State):
    def handle(self) -> str:
        return "Go! Next: Yellow"


class TrafficLight:
    """Context - maintains current state."""
    
    def __init__(self):
        self.states = {
            'red': RedState(),
            'yellow': YellowState(),
            'green': GreenState()
        }
        self.current = 'red'
    
    def change(self) -> str:
        result = self.states[self.current].handle()
        
        # Transition to next state
        transitions = {'red': 'green', 'green': 'yellow', 'yellow': 'red'}
        self.current = transitions[self.current]
        
        return result
```

**Why**: State pattern encapsulates state-specific behavior, making transitions explicit.

## Mediator Pattern
```python
# PROBLEM: Centralize communication between objects
# TIME: O(1) SPACE: O(n) for colleagues

from abc import ABC, abstractmethod

class Mediator(ABC):
    """Abstract mediator interface."""
    
    @abstractmethod
    def send(self, message: str, colleague: 'Colleague') -> None:
        pass


class Colleague(ABC):
    """Abstract colleague interface."""
    
    def __init__(self, mediator: Mediator):
        self.mediator = mediator
    
    @abstractmethod
    def receive(self, message: str) -> None:
        pass


class ChatRoom(Mediator):
    """Concrete mediator - chat room."""
    
    def __init__(self):
        self.users = []
    
    def add_user(self, user: Colleague) -> None:
        self.users.append(user)
    
    def send(self, message: str, sender: Colleague) -> None:
        for user in self.users:
            if user != sender:
                user.receive(message)


class User(Colleague):
    """Concrete colleague - chat user."""
    
    def __init__(self, mediator: Mediator, name: str):
        super().__init__(mediator)
        self.name = name
    
    def send(self, message: str) -> None:
        self.mediator.send(message, self)
    
    def receive(self, message: str) -> None:
        print(f"{self.name} received: {message}")
```

**Why**: Mediator reduces direct connections between objects, centralizing communication.

## Message Queue
```python
# PROBLEM: Asynchronous message processing
# TIME: O(1) for enqueue/dequeue SPACE: O(n)

from collections import deque
from threading import Lock
import time

class MessageQueue:
    """
    Thread-safe message queue implementation.
    """
    
    def __init__(self):
        self.queue = deque()
        self.lock = Lock()
    
    def enqueue(self, message: any) -> None:
        """Add message to queue."""
        with self.lock:
            self.queue.append(message)
    
    def dequeue(self) -> any:
        """Remove and return first message."""
        with self.lock:
            if self.queue:
                return self.queue.popleft()
            return None
    
    def peek(self) -> any:
        """View first message without removing."""
        with self.lock:
            if self.queue:
                return self.queue[0]
            return None
    
    def size(self) -> int:
        """Get queue size."""
        with self.lock:
            return len(self.queue)


class PubSub:
    """Publisher-Subscriber implementation."""
    
    def __init__(self):
        self.subscribers = {}
    
    def subscribe(self, topic: str, callback: callable) -> None:
        """Subscribe to a topic."""
        if topic not in self.subscribers:
            self.subscribers[topic] = []
        self.subscribers[topic].append(callback)
    
    def publish(self, topic: str, message: any) -> None:
        """Publish message to topic."""
        if topic in self.subscribers:
            for callback in self.subscribers[topic]:
                callback(message)
```

**Why**: Message queue decouples producers from consumers, enabling async processing.

## Connection Pool
```python
# PROBLEM: Reuse database connections efficiently
# TIME: O(1) acquire/release SPACE: O(pool_size)

from threading import Lock
from queue import Queue
import time

class Connection:
    """Simulated database connection."""
    
    def __init__(self, id: int):
        self.id = id
        self.created_at = time.time()
    
    def execute(self, query: str) -> str:
        return f"Connection {self.id}: Executed '{query}'"
    
    def close(self) -> None:
        print(f"Connection {self.id} closed")


class ConnectionPool:
    """
    Thread-safe connection pool.
    """
    
    def __init__(self, max_size: int = 10):
        self.max_size = max_size
        self.pool = Queue(maxsize=max_size)
        self.lock = Lock()
        self.created = 0
        
        # Pre-create connections
        for _ in range(max_size // 2):
            self._create_connection()
    
    def _create_connection(self) -> Connection:
        """Create new connection."""
        with self.lock:
            if self.created < self.max_size:
                conn = Connection(self.created)
                self.created += 1
                self.pool.put(conn)
                return conn
        return None
    
    def acquire(self, timeout: float = None) -> Connection:
        """Get connection from pool."""
        try:
            return self.pool.get(timeout=timeout)
        except:
            # Try to create new if under limit
            if self.created < self.max_size:
                return self._create_connection()
            raise Exception("Connection pool exhausted")
    
    def release(self, conn: Connection) -> None:
        """Return connection to pool."""
        if conn:
            self.pool.put(conn)
```

**Why**: Connection pooling avoids overhead of creating new connections for each request.
