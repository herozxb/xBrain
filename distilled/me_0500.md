## 1. Trie Variants (Compressed Trie / Radix Tree)

```python
# PROBLEM: Implement a compressed trie (radix tree) that merges nodes with single children to save space
# TIME: O(L) for insert/search where L is key length SPACE: O(N * L) where N is number of keys
class CompressedTrieNode:
    def __init__(self):
        self.children = {}  # Maps character to (substring, node) tuple
        self.is_end = False
    
    def insert(self, word):
        """Insert word into compressed trie, merging nodes when possible."""
        if not word:
            self.is_end = True
            return
        
        # Find matching prefix among children
        for char, (sub, child) in list(self.children.items()):
            # Find common prefix between word and stored substring
            common_len = 0
            min_len = min(len(word), len(sub))
            while common_len < min_len and word[common_len] == sub[common_len]:
                common_len += 1
            
            if common_len > 0:
                # Split the node
                if common_len == len(sub):
                    # Entire substring matches, continue in child
                    child.insert(word[common_len:])
                    return
                else:
                    # Partial match, need to split
                    new_node = CompressedTrieNode()
                    # Old child becomes child of new node with remaining suffix
                    new_node.children[sub[common_len]] = (sub[common_len:], child)
                    # New node replaces old entry
                    self.children[char] = (sub[:common_len], new_node)
                    # Insert remaining word into new node
                    new_node.insert(word[common_len:])
                    return
        
        # No matching prefix found, create new leaf
        self.children[word[0]] = (word, CompressedTrieNode())
        self.children[word[0]][1].is_end = True
    
    def search(self, word):
        """Search for word in compressed trie."""
        if not word:
            return self.is_end
        
        first_char = word[0] if word else None
        if first_char not in self.children:
            return False
        
        sub, child = self.children[first_char]
        if word.startswith(sub):
            return child.search(word[len(sub):])
        return False
```
**Why**: Compressed tries reduce memory by merging single-child chains, improving cache efficiency and space usage for large dictionaries.

## 2. Two-Queue Cache (2Q Cache)

```python
# PROBLEM: Implement 2Q cache that separates hot and cold data to prevent cache pollution from sequential scans
# TIME: O(1) for get/put SPACE: O(K) where K is capacity
from collections import deque, defaultdict

class TwoQCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.kin_ratio = 0.25  # Ratio for "in" queue
        
        # Three queues
        self.queue_in = deque()  # For first-time access (FIFO)
        self.queue_out = deque()  # Ghost queue for recent evictions
        self.queue_hot = deque()  # For frequently accessed (LRU)
        
        # Track positions
        self.cache = {}  # key -> value
        self.key_location = {}  # key -> 'in', 'out', or 'hot'
    
    def get(self, key):
        """Retrieve item, promoting to hot queue if found."""
        if key not in self.cache:
            return -1
        
        location = self.key_location.get(key)
        
        if location == 'hot':
            # Move to front of hot queue (LRU behavior)
            self.queue_hot.remove(key)
            self.queue_hot.appendleft(key)
        elif location == 'in':
            # Promote to hot queue
            self.queue_in.remove(key)
            self._move_to_hot(key)
        # 'out' means it's evicted, treat as miss
        
        return self.cache[key]
    
    def put(self, key, value):
        """Insert item, using 2Q algorithm to manage queues."""
        if key in self.cache:
            self.cache[key] = value
            # Update position if exists
            self.get(key)  # Trigger promotion logic
            return
        
        self.cache[key] = value
        
        # Check if recently evicted (in ghost queue)
        if key in self.key_location and self.key_location[key] == 'out':
            # Recently accessed again, promote to hot
            self.queue_out.remove(key)
            self._move_to_hot(key)
        else:
            # New key, add to "in" queue
            self._add_to_in(key)
    
    def _add_to_in(self, key):
        """Add key to the 'in' queue."""
        self.queue_in.appendleft(key)
        self.key_location[key] = 'in'
        
        # Evict from 'in' if over capacity
        kin_capacity = int(self.capacity * self.kin_ratio)
        while len(self.queue_in) > kin_capacity:
            evicted = self.queue_in.pop()
            del self.key_location[evicted]
            # Move to ghost queue
            self.queue_out.appendleft(evicted)
            self.key_location[evicted] = 'out'
    
    def _move_to_hot(self, key):
        """Move key to hot queue."""
        self.queue_hot.appendleft(key)
        self.key_location[key] = 'hot'
        
        # Evict from hot if over capacity
        hot_capacity = self.capacity - int(self.capacity * self.kin_ratio)
        while len(self.queue_hot) > hot_capacity:
            evicted = self.queue_hot.pop()
            del self.cache[evicted]
            del self.key_location[evicted]
```
**Why**: 2Q cache prevents sequential scans from polluting the cache by keeping recently accessed data separate from frequently accessed data, improving hit rates for workloads with mixed access patterns.

## 3. IoC Container (Inversion of Control)

```python
# PROBLEM: Implement a lightweight IoC container for dependency injection with singleton and transient lifecycles
# TIME: O(1) for resolve after registration SPACE: O(N) for registered services
class IoCContainer:
    def __init__(self):
        self._registrations = {}  # service_name -> (factory, lifecycle)
        self._singletons = {}  # service_name -> instance
        self._resolving = set()  # Detect circular dependencies
    
    def register_singleton(self, name, factory):
        """Register a service with singleton lifecycle."""
        self._registrations[name] = (factory, 'singleton')
        return self
    
    def register_transient(self, name, factory):
        """Register a service with transient lifecycle (new instance each time)."""
        self._registrations[name] = (factory, 'transient')
        return self
    
    def register_instance(self, name, instance):
        """Register an existing instance directly."""
        self._singletons[name] = instance
        self._registrations[name] = (None, 'instance')
        return self
    
    def resolve(self, name):
        """Resolve a service by name, handling dependencies recursively."""
        if name not in self._registrations:
            raise ValueError(f"Service '{name}' not registered")
        
        factory, lifecycle = self._registrations[name]
        
        # Return existing instance for singletons
        if lifecycle == 'singleton' and name in self._singletons:
            return self._singletons[name]
        
        if lifecycle == 'instance':
            return self._singletons[name]
        
        # Detect circular dependencies
        if name in self._resolving:
            raise ValueError(f"Circular dependency detected for '{name}'")
        
        self._resolving.add(name)
        try:
            # Create new instance using factory
            instance = factory(self)
            
            # Store singleton
            if lifecycle == 'singleton':
                self._singletons[name] = instance
            
            return instance
        finally:
            self._resolving.discard(name)
    
    def try_resolve(self, name):
        """Try to resolve, returning None if not found."""
        try:
            return self.resolve(name)
        except ValueError:
            return None

# Usage example
class Database:
    def __init__(self, connection_string):
        self.connection_string = connection_string
    
    def query(self, sql):
        return f"Result of: {sql}"

class UserRepository:
    def __init__(self, db):
        self.db = db
    
    def find_by_id(self, user_id):
        return self.db.query(f"SELECT * FROM users WHERE id = {user_id}")

# Setup container
container = IoCContainer()
container.register_singleton('db', lambda c: Database("postgresql://localhost/app"))
container.register_singleton('user_repo', lambda c: UserRepository(c.resolve('db')))
```
**Why**: IoC containers decouple component creation from usage, making code more testable, maintainable, and flexible by centralizing dependency management.

## 4. Timeout Pattern (Circuit Breaker with Timeout)

```python
# PROBLEM: Implement timeout wrapper with fallback for resilient service calls
# TIME: O(1) per call SPACE: O(1)
import time
import threading
from functools import wraps

class TimeoutException(Exception):
    pass

class TimeoutExecutor:
    def __init__(self, timeout_seconds, fallback=None):
        self.timeout = timeout_seconds
        self.fallback = fallback
    
    def execute(self, func, *args, **kwargs):
        """Execute function with timeout and optional fallback."""
        result = [None]
        exception = [None]
        
        def worker():
            try:
                result[0] = func(*args, **kwargs)
            except Exception as e:
                exception[0] = e
        
        thread = threading.Thread(target=worker, daemon=True)
        thread.start()
        thread.join(timeout=self.timeout)
        
        if thread.is_alive():
            # Timeout occurred
            if self.fallback:
                return self.fallback(*args, **kwargs)
            raise TimeoutException(f"Operation timed out after {self.timeout}s")
        
        if exception[0]:
            raise exception[0]
        
        return result[0]
    
    def __call__(self, func):
        """Use as decorator."""
        @wraps(func)
        def wrapper(*args, **kwargs):
            return self.execute(func, *args, **kwargs)
        return wrapper

# Alternative using signal (Unix only)
class TimeoutSignal:
    def __init__(self, timeout_seconds, fallback=None):
        self.timeout = timeout_seconds
        self.fallback = fallback
    
    def execute(self, func, *args, **kwargs):
        """Execute with timeout using signals (main thread only)."""
        import signal
        
        def timeout_handler(signum, frame):
            raise TimeoutException("Operation timed out")
        
        # Set signal handler
        old_handler = signal.signal(signal.SIGALRM, timeout_handler)
        signal.setitimer(signal.ITIMER_REAL, self.timeout)
        
        try:
            result = func(*args, **kwargs)
            return result
        except TimeoutException:
            if self.fallback:
                return self.fallback(*args, **kwargs)
            raise
        finally:
            signal.setitimer(signal.ITIMER_REAL, 0)
            signal.signal(signal.SIGALRM, old_handler)

# Usage
def slow_operation():
    time.sleep(10)
    return "Done"

def fallback_operation():
    return "Fallback result"

executor = TimeoutExecutor(timeout_seconds=2, fallback=fallback_operation)
result = executor.execute(slow_operation)  # Returns "Fallback result"
```
**Why**: Timeouts prevent cascading failures in distributed systems by limiting how long a system waits for unresponsive services, with fallbacks providing graceful degradation.

## 6. Redlock Algorithm (Distributed Lock)

```python
# PROBLEM: Implement Redlock algorithm for distributed locking across multiple Redis instances
# TIME: O(N) for lock/unlock where N is number of Redis nodes SPACE: O(1)
import time
import uuid
import hashlib

class RedisNode:
    """Simulated Redis node for demonstration."""
    def __init__(self, name):
        self.name = name
        self.locks = {}
    
    def set_lock(self, key, value, ttl_ms):
        """Set lock if not exists (NX), with TTL (PX)."""
        if key not in self.locks or self.locks[key]['expires'] < time.time() * 1000:
            self.locks[key] = {
                'value': value,
                'expires': time.time() * 1000 + ttl_ms
            }
            return True
        return False
    
    def del_lock(self, key, value):
        """Delete lock only if value matches (Lua script simulation)."""
        if key in self.locks and self.locks[key]['value'] == value:
            del self.locks[key]
            return True
        return False

class Redlock:
    def __init__(self, redis_nodes, retry_count=3, retry_delay=200):
        """
        Initialize Redlock with multiple Redis nodes.
        
        Args:
            redis_nodes: List of Redis node instances
            retry_count: Number of retry attempts
            retry_delay: Delay between retries in milliseconds
        """
        self.nodes = redis_nodes
        self.quorum = len(redis_nodes) // 2 + 1
        self.retry_count = retry_count
        self.retry_delay = retry_delay / 1000.0
    
    def lock(self, resource, ttl_ms):
        """
        Acquire distributed lock using Redlock algorithm.
        
        Args:
            resource: Lock resource name
            ttl_ms: Lock TTL in milliseconds
        
        Returns:
            Lock value if acquired, None otherwise
        """
        lock_value = self._generate_lock_value()
        
        for attempt in range(self.retry_count):
            start_time = time.time()
            
            # Try to acquire lock on all nodes
            acquired = 0
            for node in self.nodes:
                if node.set_lock(resource, lock_value, ttl_ms):
                    acquired += 1
            
            # Check if quorum reached
            elapsed_ms = (time.time() - start_time) * 1000
            validity = ttl_ms - elapsed_ms
            
            if acquired >= self.quorum and validity > 0:
                return {
                    'resource': resource,
                    'value': lock_value,
                    'validity': validity
                }
            
            # Failed to acquire, release partial locks
            self._unlock_internal(resource, lock_value)
            
            # Wait before retry
            if attempt < self.retry_count - 1:
                time.sleep(self.retry_delay)
        
        return None
    
    def unlock(self, lock):
        """Release the distributed lock."""
        self._unlock_internal(lock['resource'], lock['value'])
    
    def _unlock_internal(self, resource, value):
        """Unlock on all nodes."""
        for node in self.nodes:
            node.del_lock(resource, value)
    
    def _generate_lock_value(self):
        """Generate unique lock value."""
        return str(uuid.uuid4())

# Usage
nodes = [RedisNode(f"redis-{i}") for i in range(5)]
redlock = Redlock(nodes)

# Acquire lock
lock = redlock.lock("my_resource", ttl_ms=10000)
if lock:
    try:
        # Critical section
        print(f"Lock acquired: {lock['value']}")
    finally:
        redlock.unlock(lock)
```
**Why**: Redlock provides distributed mutual exclusion across unreliable networks by requiring quorum agreement, making it resilient to individual node failures and network partitions.

```python
# PROBLEM: Implement transaction manager with ACID properties simulation
# TIME: O(1) for begin/commit, O(N) for rollback of N operations SPACE: O(N) for transaction log
from enum import Enum
from collections import defaultdict
import copy

class TransactionState(Enum):
    ACTIVE = 1
    COMMITTED = 2
    ROLLED_BACK = 3

class Transaction:
    def __init__(self, tx_id):
        self.tx_id = tx_id
        self.state = TransactionState.ACTIVE
        self.operations = []  # Log of operations
        self.snapshot = {}  # Data snapshot for isolation
    
    def log_operation(self, operation_type, key, old_value, new_value):
        """Log operation for potential rollback."""
        self.operations.append({
            'type': operation_type,
            'key': key,
            'old_value': old_value,
            'new_value': new_value
        })

class TransactionManager:
    def __init__(self):
        self.data = {}  # Actual data storage
        self.locks = defaultdict(list)  # Row-level locks
        self.tx_counter = 0
        self.active_transactions = {}
    
    def begin(self):
        """Start a new transaction."""
        self.tx_counter += 1
        tx = Transaction(self.tx_counter)
        tx.snapshot = copy.deepcopy(self.data)  # Snapshot isolation
        self.active_transactions[tx.tx_id] = tx
        return tx
    
    def read(self, tx, key):
        """Read data within transaction (snapshot isolation)."""
        if tx.state != TransactionState.ACTIVE:
            raise ValueError("Transaction not active")
        
        # Read from transaction's snapshot
        return tx.snapshot.get(key)
    
    def write(self, tx, key, value):
        """Write data within transaction."""
        if tx.state != TransactionState.ACTIVE:
            raise ValueError("Transaction not active")
        
        # Acquire lock
        if not self._acquire_lock(tx, key, 'write'):
            raise ValueError(f"Could not acquire lock for key: {key}")
        
        old_value = tx.snapshot.get(key)
        tx.log_operation('write', key, old_value, value)
        tx.snapshot[key] = value
    
    def commit(self, tx):
        """Commit transaction (apply changes to actual data)."""
        if tx.state != TransactionState.ACTIVE:
            return False
        
        # Durability: Write-ahead logging would go here
        
        # Apply changes to actual data
        for op in tx.operations:
            if op['type'] == 'write':
                self.data[op['key']] = op['new_value']
        
        # Release locks
        self._release_locks(tx)
        
        tx.state = TransactionState.COMMITTED
        del self.active_transactions[tx.tx_id]
        return True
    
    def rollback(self, tx):
        """Rollback transaction (undo all changes)."""
        if tx.state != TransactionState.ACTIVE:
            return False
        
        # Undo operations in reverse order
        for op in reversed(tx.operations):
            # Changes are only in snapshot, no actual data modification needed
            pass
        
        # Release locks
        self._release_locks(tx)
        
        tx.state = TransactionState.ROLLED_BACK
        del self.active_transactions[tx.tx_id]
        return True
    
    def _acquire_lock(self, tx, key, lock_type):
        """Acquire lock on key."""
        # Simplified: check if any conflicting lock exists
        for lock_tx_id, lock_t in self.locks[key]:
            if lock_tx_id != tx.tx_id:
                if lock_t == 'write' or lock_type == 'write':
                    return False
        
        self.locks[key].append((tx.tx_id, lock_type))
        return True
    
    def _release_locks(self, tx):
        """Release all locks held by transaction."""
        for key in list(self.locks.keys()):
            self.locks[key] = [
                (tid, lt) for tid, lt in self.locks[key]
                if tid != tx.tx_id
            ]
```
**Why**: Understanding ACID transactions is crucial for building reliable database systems and ensuring data integrity in concurrent environments.

## 8. Connection Pooling

```python
# PROBLEM: Implement database connection pool with health checking and connection validation
# TIME: O(1) for get/return connection (amortized) SPACE: O(P) where P is pool size
import threading
import time
from queue import Queue, Empty
from contextlib import contextmanager

class Connection:
    """Simulated database connection."""
    def __init__(self, conn_id, config):
        self.conn_id = conn_id
        self.config = config
        self.connected = False
        self.last_used = time.time()
        self.connect()
    
    def connect(self):
        """Establish connection."""
        # Simulated connection
        self.connected = True
        return True
    
    def close(self):
        """Close connection."""
        self.connected = False
    
    def is_valid(self):
        """Check if connection is still valid."""
        return self.connected
    
    def execute(self, query):
        """Execute query on connection."""
        if not self.connected:
            raise ConnectionError("Connection not established")
        self.last_used = time.time()
        return f"Result of: {query}"

class ConnectionPool:
    def __init__(self, config, min_size=5, max_size=20, 
                 idle_timeout=300, connection_timeout=30):
        self.config = config
        self.min_size = min_size
        self.max_size = max_size
        self.idle_timeout = idle_timeout
        self.connection_timeout = connection_timeout
        
        self._pool = Queue()
        self._active_connections = set()
        self._lock = threading.Lock()
        self._total_created = 0
        
        # Initialize minimum connections
        self._initialize_pool()
        
        # Start maintenance thread
        self._maintenance_thread = threading.Thread(
            target=self._maintenance, daemon=True
        )
        self._maintenance_thread.start()
    
    def _initialize_pool(self):
        """Create initial connections."""
        for _ in range(self.min_size):
            conn = self._create_connection()
            self._pool.put(conn)
    
    def _create_connection(self):
        """Create a new connection."""
        with self._lock:
            if self._total_created >= self.max_size:
                raise ValueError("Maximum pool size reached")
            self._total_created += 1
            conn_id = self._total_created
        
        return Connection(conn_id, self.config)
    
    def get_connection(self):
        """Get a connection from pool with timeout."""
        try:
            # Try to get existing connection
            conn = self._pool.get(block=True, timeout=self.connection_timeout)
            
            # Validate connection
            if not conn.is_valid():
                with self._lock:
                    self._total_created -= 1
                conn = self._create_connection()
            
            with self._lock:
                self._active_connections.add(conn.conn_id)
            
            return conn
        
        except Empty:
            # Try to create new connection if under max
            with self._lock:
                if self._total_created < self.max_size:
                    conn = self._create_connection()
                    self._active_connections.add(conn.conn_id)
                    return conn
            
            raise TimeoutError("Connection pool exhausted")
    
    def return_connection(self, conn):
        """Return connection to pool."""
        with self._lock:
            self._active_connections.discard(conn.conn_id)
        
        if conn.is_valid():
            self._pool.put(conn)
        else:
            with self._lock:
                self._total_created -= 1
    
    @contextmanager
    def connection(self):
        """Context manager for automatic connection handling."""
        conn = self.get_connection()
        try:
            yield conn
        finally:
            self.return_connection(conn)
    
    def _maintenance(self):
        """Periodic maintenance to clean up idle connections."""
        while True:
            time.sleep(60)  # Check every minute
            
            current_time = time.time()
            temp_list = []
            
            # Check all idle connections
            while not self._pool.empty():
                try:
                    conn = self._pool.get_nowait()
                    idle_time = current_time - conn.last_used
                    
                    # Close if idle too long and above minimum
                    if idle_time > self.idle_timeout:
                        with self._lock:
                            if self._total_created > self.min_size:
                                conn.close()
                                self._total_created -= 1
                                continue
                    
                    temp_list.append(conn)
                except Empty:
                    break
            
            # Return valid connections to pool
            for conn in temp_list:
                self._pool.put(conn)
    
    def close_all(self):
        """Close all connections in pool."""
        while not self._pool.empty():
            try:
                conn = self._pool.get_nowait()
                conn.close()
            except Empty:
                break

# Usage
pool = ConnectionPool(
    config={"host": "localhost", "port": 5432},
    min_size=5,
    max_size=20
)

# Using context manager
```

## 9. ThreadPoolExecutor Pattern (Advanced)

```python
# PROBLEM: Implement custom thread pool with work stealing, priority queue, and cancellation
# TIME: O(log N) for task submission to priority queue SPACE: O(N) for pending tasks
import threading
import queue
from concurrent.futures import Future
from dataclasses import dataclass, field
from typing import Callable, Any, Optional
import time

@dataclass(order=True)
class PrioritizedTask:
    priority: int
    task_id: int = field(compare=False)
    func: Callable = field(compare=False)
    args: tuple = field(compare=False)
    kwargs: dict = field(compare=False)
    future: Future = field(compare=False)

class ThreadPoolExecutorAdvanced:
    def __init__(self, max_workers=None, enable_work_stealing=True):
        self.max_workers = max_workers or 4
        self.enable_work_stealing = enable_work_stealing
        
        self._work_queue = queue.PriorityQueue()
        self._task_counter = 0
        self._task_counter_lock = threading.Lock()
        
        self.workers = []
        self.local_queues = {}  # Thread-local work queues
        self.shutdown_flag = threading.Event()
        
        self._create_workers()
    
    def _create_workers(self):
        """Create worker threads."""
        for i in range(self.max_workers):
            worker = threading.Thread(
                target=self._worker_loop,
                args=(i,),
                daemon=True
            )
            worker.start()
            self.workers.append(worker)
    
    def _worker_loop(self, worker_id):
        """Main worker loop with work stealing."""
        local_queue = queue.Queue()
        self.local_queues[worker_id] = local_queue
        
        while not self.shutdown_flag.is_set():
            task = None
            
            # Try local queue first
            try:
                task = local_queue.get_nowait()
            except queue.Empty:
                pass
            
            # Then try global queue
            if task is None:
                try:
                    task = self._work_queue.get(timeout=0.1)
                except queue.Empty:
                    # Work stealing: try to steal from other workers
                    if self.enable_work_stealing:
                        task = self._steal_work(worker_id)
                    continue
            
            if task:
                self._execute_task(task)
    
    def _steal_work(self, thief_id):
        """Attempt to steal work from other workers."""
        for worker_id, local_queue in self.local_queues.items():
            if worker_id != thief_id:
                try:
                    # Steal from the end (oldest tasks)
                    return local_queue.get_nowait()
                except queue.Empty:
                    continue
        return None
    
    def _execute_task(self, task):
        """Execute task and set result/exception in future."""
        if task.future.cancelled():
            return
        
        try:
            result = task.func(*task.args, **task.kwargs)
            task.future.set_result(result)
        except Exception as e:
            task.future.set_exception(e)
    
    def submit(self, func, *args, priority=0, **kwargs):
        """Submit task with optional priority (lower = higher priority)."""
        with self._task_counter_lock:
            self._task_counter += 1
            task_id = self._task_counter
        
        future = Future()
        task = PrioritizedTask(
            priority=priority,
            task_id=task_id,
            func=func,
            args=args,
            kwargs=kwargs,
            future=future
        )
        
        self._work_queue.put(task)
        return future
    
    def map(self, func, iterable, timeout=None):
        """Apply function to each item in iterable."""
        futures = [
            self.submit(func, item)
            for item in iterable
        ]
        
        results = []
        for future in futures:
            try:
                results.append(future.result(timeout=timeout))
            except Exception:
                # Cancel remaining futures
                for f in futures:
                    f.cancel()
                raise
        
        return results
    
    def shutdown(self, wait=True, cancel_futures=False):
        """Shutdown the executor."""
        self.shutdown_flag.set()
        
        if cancel_futures:
            # Drain remaining tasks
            while not self._work_queue.empty():
                try:
                    task = self._work_queue.get_nowait()
                    task.future.cancel()
                except queue.Empty:
                    break
        
        if wait:
            for worker in self.workers:
                worker.join(timeout=5)

# Usage
pool = ThreadPoolExecutorAdvanced(max_workers=4)

# Submit tasks with priorities
future1 = pool.submit(lambda x: x * 2, 10, priority=1)
future2 = pool.submit(lambda x: x * 3, 20, priority=0)  # Higher priority

# Map over iterable
results = pool.map(lambda x: x ** 2, [1, 2, 3, 4, 5])
print(results)  # [1, 4, 9, 16, 25]

pool.shutdown()
```
**Why**: Advanced thread pools with work stealing and priority queuing optimize CPU utilization, reduce latency for high-priority tasks, and handle uneven workloads efficiently.

## 10. JWT Patterns (JSON Web Token Implementation)

```python
# PROBLEM: Implement JWT generation, validation, and refresh token rotation
# TIME: O(1) for encode/decode SPACE: O(K) for key storage
import base64
import hashlib
import hmac
import json
import time
from typing import Optional, Dict, Any

class JWTHandler:
    def __init__(self, secret_key, algorithm='HS256', 
                 access_token_expire=3600, refresh_token_expire=604800):
        self.secret_key = secret_key
        self.algorithm = algorithm
        self.access_token_expire = access_token_expire
        self.refresh_token_expire = refresh_token_expire
        self.blacklist = set()  # In production, use Redis
        self.refresh_tokens = {}  # user_id -> refresh_token
    
    def _base64_encode(self, data):
        """URL-safe base64 encode."""
        json_str = json.dumps(data, separators=(',', ':'))
        return base64.urlsafe_b64encode(json_str.encode()).rstrip(b'=').decode()
    
    def _base64_decode(self, data):
        """URL-safe base64 decode."""
        # Add padding
        padding = 4 - len(data) % 4
        if padding != 4:
            data += '=' * padding
        
        json_str = base64.urlsafe_b64decode(data.encode())
        return json.loads(json_str)
    
    def _sign(self, data):
        """Sign data using HMAC."""
        if self.algorithm == 'HS256':
            digest = hashlib.sha256
        elif self.algorithm == 'HS384':
            digest = hashlib.sha384
        elif self.algorithm == 'HS512':
            digest = hashlib.sha512
        else:
            raise ValueError(f"Unsupported algorithm: {self.algorithm}")
        
        return hmac.new(
            self.secret_key.encode(),
            data.encode(),
            digest
        ).hexdigest()
    
    def encode(self, payload, expires_in=None):
        """Generate JWT token."""
        # Header
        header = {
            'alg': self.algorithm,
            'typ': 'JWT'
        }
        
        # Add expiration
        now = int(time.time())
        if expires_in:
            payload['exp'] = now + expires_in
        elif 'exp' not in payload:
            payload['exp'] = now + self.access_token_expire
        
        payload['iat'] = now  # Issued at
        
        # Encode header and payload
        header_encoded = self._base64_encode(header)
        payload_encoded = self._base64_encode(payload)
        
        # Create signature
        message = f"{header_encoded}.{payload_encoded}"
        signature = self._sign(message)
        
        return f"{message}.{signature}"
    
    def decode(self, token, verify_exp=True):
        """Decode and validate JWT token."""
        # Check blacklist
        if token in self.blacklist:
            raise ValueError("Token has been revoked")
        
        parts = token.split('.')
        if len(parts) != 3:
            raise ValueError("Invalid token format")
        
        header_encoded, payload_encoded, signature = parts
        
        # Verify signature
        message = f"{header_encoded}.{payload_encoded}"
        expected_signature = self._sign(message)
        
        if not hmac.compare_digest(signature, expected_signature):
            raise ValueError("Invalid signature")
        
        # Decode payload
        payload = self._base64_decode(payload_encoded)
        
        # Verify expiration
        if verify_exp and 'exp' in payload:
            if payload['exp'] < time.time():
                raise ValueError("Token has expired")
        
        return payload
    
    def generate_tokens(self, user_id, additional_claims=None):
        """Generate access and refresh token pair."""
        claims = {'sub': user_id}
        if additional_claims:
            claims.update(additional_claims)
        
        # Access token (short-lived)
        access_token = self.encode(claims.copy())
        
        # Refresh token (long-lived)
        refresh_claims = claims.copy()
        refresh_claims['type'] = 'refresh'
        refresh_token = self.encode(refresh_claims, self.refresh_token_expire)
        
        # Store refresh token
        self.refresh_tokens[user_id] = refresh_token
        
        return {
            'access_token': access_token,
            'refresh_token': refresh_token,
            'token_type': 'Bearer',
            'expires_in': self.access_token_expire
        }
    
    def refresh_access_token(self, refresh_token):
        """Generate new access token using refresh token."""
        payload = self.decode(refresh_token)
        
        if payload.get('type') != 'refresh':
            raise ValueError("Invalid refresh token")
        
        user_id = payload['sub']
        
        # Verify refresh token matches stored one
        if self.refresh_tokens.get(user_id) != refresh_token:
            raise ValueError("Refresh token has been rotated")
        
        # Generate new access token
        new_claims = {'sub': user_id}
        new_access_token = self.encode(new_claims)
        
        # Optional: Rotate refresh token for better security
        new_refresh_token = self.encode(
            {'sub': user_id, 'type': 'refresh'},
            self.refresh_token_expire
        )
        self.refresh_tokens[user_id] = new_refresh_token
        self.blacklist.add(refresh_token)  # Invalidate old refresh token
        
        return {
            'access_token': new_access_token,
            'refresh_token': new_refresh_token,
            'token_type': 'Bearer',
            'expires_in': self.access_token_expire
        }
    
    def revoke_token(self, token):
        """Revoke (blacklist) a token."""
        self.blacklist.add(token)
        
        # Also remove from refresh tokens
        try:
            payload = self.decode(token, verify_exp=False)
            if payload.get('sub') in self.refresh_tokens:
                del self.refresh_tokens[payload['sub']]
        except:
            pass

# Usage
jwt_handler = JWTHandler(
    secret_key="your-secret-key-here",
    algorithm='HS256',
    access_token_expire=3600,  # 1 hour
    refresh_token_expire=604800  # 7 days
)

# Generate tokens
tokens = jwt_handler.generate_tokens(user_id="user123")
print(f"Access Token: {tokens['access_token']}")

# Decode token
payload = jwt_handler.decode(tokens['access_token'])
print(f"User ID: {payload['sub']}")

# Refresh token
new_tokens = jwt_handler.refresh_access_token(tokens['refresh_token'])
```
**Why**: JWT patterns enable stateless authentication with secure token rotation, allowing scalable distributed systems without server-side session storage while maintaining security through refresh token mechanisms.
