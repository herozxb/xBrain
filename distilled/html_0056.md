## html5_fullscreen_exitFullscreen

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Exit Fullscreen Demo</title>
</head>
<body>
  <div id="fullscreen-container" style="width:100%; height:300px; background:#2d3748; display:flex; align-items:center; justify-content:center;">
    <button id="btn-exit">Exit Fullscreen</button>
  </div>

  <script>
    /**
     * PROBLEM: Exit fullscreen mode programmatically
     * TIME: O(1) SPACE: O(1)
     * 
     * WHY: Required when building custom video players, presentations,
     * or immersive UIs where user needs an explicit exit button.
     */
    function exitFullscreen() {
      // Cross-browser compatible fullscreen exit
      // document.exitFullscreen is the standard method
      if (document.exitFullscreen) {
        // Standard API for modern browsers
        document.exitFullscreen();
      } else if (document.webkitExitFullscreen) {
        // Safari/Chrome legacy prefix
        document.webkitExitFullscreen();
      } else if (document.msExitFullscreen) {
        // IE/Edge legacy prefix
        document.msExitFullscreen();
      } else if (document.mozCancelFullScreen) {
        // Firefox legacy prefix
        document.mozCancelFullScreen();
      }
    }

    // Attach click handler
    document.getElementById('btn-exit').addEventListener('click', exitFullscreen);

    // Listen for fullscreen changes
    document.addEventListener('fullscreenchange', () => {
      console.log('Fullscreen state changed');
    });
  </script>
</body>
</html>
```

**Why:** Use `exitFullscreen()` when building custom media controls or kiosk-mode applications where users need an obvious way to leave fullscreen without using keyboard shortcuts (F11/Esc).

## html5_visibilitychange_event

```html

     * reducing API polling, or saving resources when page is hidden.
     */
    function handleVisibilityChange() {
      const statusEl = document.getElementById('status');
      
      // document.hidden returns true when page is not visible
      // document.visibilityState returns 'visible' | 'hidden' | 'prerender'

        console.log('User switched away - pause animations/videos');
        
        // Example: Pause a video player

        // startPolling();

```

**Why:** The `visibilitychange` event helps optimize performance by pausing non-essential tasks when users aren't viewing the page, saving battery on mobile devices and reducing unnecessary network requests.

## js_error_message

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Error Message Demo</title>
</head>
<body>
  <button id="btn-test">Throw Test Error</button>
  <p id="result"></p>

  <script>
    /**
     * PROBLEM: Extract and display meaningful error messages from Error objects
     * TIME: O(1) SPACE: O(1)
     * 
     * WHY: Error.message provides human-readable error descriptions for
     * debugging, logging, and displaying user-friendly error notifications.
     */

    function demonstrateErrorMessage() {
      try {
        // Intentionally cause an error
        throw new Error('Something went wrong with the operation');
      } catch (error) {
        // error.message contains the error description string
        const errorMessage = error.message;
        
        // Display to user
        document.getElementById('result').textContent = 
          `Error caught: ${errorMessage}`;
        
        // Log to console
        console.error('Error message:', errorMessage);
        
        return errorMessage;
      }
    }

    // Different error types have different default messages
    function showErrorTypes() {
      const errors = [
        new Error('Generic error'),
        new TypeError('Type mismatch'),
        new RangeError('Value out of range'),
        new ReferenceError('Variable not defined')
      ];

      errors.forEach(err => {
        // Each error type has a meaningful .message property
        console.log(`${err.name}: ${err.message}`);
      });
    }

    document.getElementById('btn-test').addEventListener('click', () => {
      demonstrateErrorMessage();
      showErrorTypes();
    });
  </script>
</body>
</html>
```

**Why:** `error.message` is the primary way to access error descriptions in JavaScript. Use it for user notifications, server-side logging, and debugging to understand what went wrong in your application.

## js_regexp_flags_m

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>RegExp Multiline Flag Demo</title>
</head>
<body>
  <h2>RegExp 'm' (multiline) Flag</h2>
  <pre id="output"></pre>

  <script>
    /**
     * PROBLEM: Match start/end of each line in multi-line text
     * TIME: O(n) SPACE: O(1)
     * 
     * WHY: Without 'm' flag, ^ and $ only match start/end of entire string.
     * With 'm', they match at every line boundary, enabling line-by-line validation.
     */

    const multilineText = `First line
Second line
Third line
Fourth line`;

    // Without 'm' flag - ^ only matches start of entire string
    const singleLinePattern = /^Second/;
    console.log('Without m flag:', singleLinePattern.test(multilineText));
    // Output: false (no match at string start)

    // With 'm' flag - ^ matches start of each line
    const multilinePattern = /^Second/m;
    console.log('With m flag:', multilinePattern.test(multilineText));
    // Output: true (matches start of second line)

    // Find all lines starting with capital letters
    function findLinesStartingWithCapital(text) {
      const pattern = /^[A-Z]/gm; // 'g' for global, 'm' for multiline
      const matches = text.match(pattern);
      return matches;
    }

    // Count lines ending with specific text
    function countLinesEndingWith(text, suffix) {
      const pattern = new RegExp(`${suffix}$`, 'gm');
      const matches = text.match(pattern);
      return matches ? matches.length : 0;
    }

    // Demonstration
    const log = document.getElementById('output');
    log.textContent = `Text:
${multilineText}

Lines starting with 'Second': ${findLinesStartingWithCapital(multilineText)}
Pattern /^Second/m test: ${multilinePattern.test(multilineText)}`;
  </script>
</body>
</html>
```

**Why:** The `m` flag transforms `^` and `$` anchors to work per-line instead of per-string. Essential for processing logs, CSV data, form inputs, and any multi-line content where line boundaries matter.

## js_date_getMinutes

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Date getMinutes() Demo</title>
</head>
<body>
  <h2>Current Time: <span id="clock"></span></h2>
  <button id="btn-check">Check Minute Value</button>

  <script>
    /**
     * PROBLEM: Extract the minutes component (0-59) from a Date object
     * TIME: O(1) SPACE: O(1)
     * 
     * WHY: Essential for time display, scheduling, alarms, and any
     * time-sensitive logic like "run every 15 minutes past the hour".
     */

    // Get current minutes
    function getCurrentMinutes() {
      const now = new Date();
      // getMinutes() returns 0-59
      return now.getMinutes();
    }

    // Format time with leading zero
    function formatTime(date) {
      const hours = date.getHours();
      const minutes = date.getMinutes();
      const seconds = date.getSeconds();
      
      // Pad with leading zero if needed
      const pad = (n) => n.toString().padStart(2, '0');
      
      return `${pad(hours)}:${pad(minutes)}:${pad(seconds)}`;
    }

    // Check if we're at quarter-hour mark
    function isQuarterHour() {
      const minutes = new Date().getMinutes();
      return minutes % 15 === 0;
    }

    // Run task at specific minute past hour
    function shouldRunAtMinute(targetMinute) {
      const currentMinute = new Date().getMinutes();
      return currentMinute === targetMinute;
    }

    // Update clock display
    function updateClock() {
      document.getElementById('clock').textContent = formatTime(new Date());
    }

    // Initialize
    updateClock();
    setInterval(updateClock, 1000);

    document.getElementById('btn-check').addEventListener('click', () => {
      const mins = getCurrentMinutes();
      alert(`Current minute: ${mins}\nIs quarter hour: ${isQuarterHour()}`);
    });
  </script>
</body>
</html>
```

**Why:** `getMinutes()` is the standard way to extract minutes from JavaScript Date objects. Use it for clocks, timers, scheduled tasks, and time-based conditional logic in applications.

## js_math_pow

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Math.pow() Demo</title>
</head>
<body>
  <h2>Math.pow() - Exponentiation</h2>
  <input type="number" id="base" placeholder="Base" value="2">
  <input type="number" id="exp" placeholder="Exponent" value="10">
  <button id="btn-calc">Calculate</button>
  <p id="result"></p>

  <script>
    /**
     * PROBLEM: Calculate base raised to exponent power
     * TIME: O(log n) for implementation SPACE: O(1)
     * 
     * WHY: Essential for mathematical calculations, geometry,
     * compound interest, growth rates, and binary operations.
     */

    // Basic exponentiation
    function power(base, exponent) {
      // Math.pow(base, exponent) returns base^exponent
      return Math.pow(base, exponent);
    }

    // Calculate compound interest
    function compoundInterest(principal, rate, years) {
      // A = P(1 + r)^n
      return principal * Math.pow(1 + rate, years);
    }

    // Calculate circle area (r^2 * π)
    function circleArea(radius) {
      return Math.PI * Math.pow(radius, 2);
    }

    // Convert between number bases (using powers)
    function binaryToDecimal(binaryStr) {
      let result = 0;
      for (let i = 0; i < binaryStr.length; i++) {
        const bit = parseInt(binaryStr[i]);
        const power = binaryStr.length - 1 - i;
        result += bit * Math.pow(2, power);
      }
      return result;
    }

    // Alternative: ES6 exponentiation operator (**)
    // 2 ** 10 === Math.pow(2, 10)

    document.getElementById('btn-calc').addEventListener('click', () => {
      const base = parseFloat(document.getElementById('base').value);
      const exp = parseFloat(document.getElementById('exp').value);
      
      const result = Math.pow(base, exp);
      const es6Result = base ** exp; // Same result
      
      document.getElementById('result').textContent = 
        `${base}^${exp} = ${result.toLocaleString()}`;
      
      // Demo other calculations
      console.log('Compound interest ($1000 at 5% for 10 years):', 
        compoundInterest(1000, 0.05, 10).toFixed(2));
      console.log('Circle area (radius 5):', circleArea(5).toFixed(2));
      console.log('Binary 1010 to decimal:', binaryToDecimal('1010'));
    });
  </script>
</body>
</html>
```

**Why:** `Math.pow()` calculates exponents for any mathematical operation requiring powers. Modern JavaScript also offers the `**` operator, but `Math.pow()` remains useful for dynamic exponent values and clarity in mathematical expressions.

## js_map_delete

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Map.delete() Demo</title>
</head>
<body>
  <h2>Map.delete() - Remove Entries</h2>
  <button id="btn-demo">Run Demo</button>
  <pre id="output"></pre>

  <script>
    /**
     * PROBLEM: Remove a key-value pair from a Map by key
     * TIME: O(1) average SPACE: O(1)
     * 
     * WHY: Maps provide ordered, iterable key-value storage with
     * any value as key. delete() is the standard way to remove entries.
     */

    function demonstrateMapDelete() {
      // Create a Map with user data
      const userMap = new Map([
        ['user1', { name: 'Alice', age: 30 }],
        ['user2', { name: 'Bob', age: 25 }],
        ['user3', { name: 'Charlie', age: 35 }]
      ]);

      const log = [];
      
      log.push(`Initial size: ${userMap.size}`);
      log.push(`Has 'user2': ${userMap.has('user2')}`);
      
      // Delete an entry - returns true if key existed, false otherwise
      const deleted = userMap.delete('user2');
      log.push(`Deleted 'user2': ${deleted}`);
      
      log.push(`Size after delete: ${userMap.size}`);
      log.push(`Has 'user2' after delete: ${userMap.has('user2')}`);
      
      // Trying to delete non-existent key returns false
      const notDeleted = userMap.delete('nonexistent');
      log.push(`Delete nonexistent key: ${notDeleted}`);

      // Display remaining entries
      log.push('\nRemaining entries:');
      userMap.forEach((value, key) => {
        log.push(`  ${key}: ${JSON.stringify(value)}`);
      });

      document.getElementById('output').textContent = log.join('\n');
    }

    // Practical example: Cache with TTL
    class CacheWithExpiry {
      constructor() {
        this.cache = new Map();
      }

      set(key, value, ttlMs) {
        this.cache.set(key, {
          value,
          expiry: Date.now() + ttlMs
        });
      }

      get(key) {
        const item = this.cache.get(key);
        if (!item) return undefined;
        
        if (Date.now() > item.expiry) {
          this.cache.delete(key); // Remove expired entry
          return undefined;
        }
        
        return item.value;
      }
    }

    document.getElementById('btn-demo').addEventListener('click', demonstrateMapDelete);
  </script>
</body>
</html>
```

**Why:** `Map.delete()` efficiently removes entries while maintaining Map integrity. Unlike plain objects, Maps preserve insertion order and allow any value as a key, making delete() essential for caches, registries, and state management.

## js_proxy_constructor

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Proxy Constructor Demo</title>
</head>
<body>
  <h2>Proxy Constructor - Object Interception</h2>
  <button id="btn-demo">Run Demo</button>
  <pre id="output"></pre>

  <script>
    /**
     * PROBLEM: Intercept and customize fundamental object operations
     * TIME: O(1) for proxy creation SPACE: O(1) per trap
     * 
     * WHY: Proxies enable powerful patterns: validation, logging,
     * virtual properties, access control, and reactive data binding.
     */

    function demonstrateProxy() {
      const log = [];

      // Target object
      const target = {
        name: 'John',
        age: 25,
        _secret: 'hidden'
      };

      // Handler with traps
      const handler = {
        // Intercept property access
        get(obj, prop) {
          log.push(`Getting: ${prop}`);
          if (prop.startsWith('_')) {
            return undefined; // Hide private properties
          }
          return obj[prop];
        },

        // Intercept property assignment
        set(obj, prop, value) {
          log.push(`Setting: ${prop} = ${value}`);
          
          // Validation example
          if (prop === 'age' && (value < 0 || value > 150)) {
            throw new Error('Invalid age');
          }
          
          obj[prop] = value;
          return true; // Indicate success
        },

        // Intercept property check
        has(obj, prop) {
          log.push(`Checking has: ${prop}`);
          return prop in obj && !prop.startsWith('_');
        }
      };

      // Create proxy using new Proxy(target, handler)
      const proxy = new Proxy(target, handler);

      // Use the proxy
      log.push(`Name: ${proxy.name}`);
      log.push(`Age: ${proxy.age}`);
      
      proxy.age = 26;
      log.push(`New age: ${proxy.age}`);
      
      log.push(`Has 'name': ${'name' in proxy}`);
      log.push(`Has '_secret': ${'_secret' in proxy}`);
      log.push(`Access '_secret': ${proxy._secret}`);

      document.getElementById('output').textContent = log.join('\n');
    }

    // Practical example: Reactive object
    function createReactiveObject(obj, callback) {
      return new Proxy(obj, {
        set(target, prop, value) {
          const oldValue = target[prop];
          target[prop] = value;
          callback(prop, oldValue, value);
          return true;
        }
      });
    }

    document.getElementById('btn-demo').addEventListener('click', demonstrateProxy);
  </script>
</body>
</html>
```

**Why:** `new Proxy(target, handler)` creates a wrapper that intercepts operations on objects. Use it for validation, logging, access control, virtual properties, and building reactive frameworks like Vue's reactivity system.

## js_generator_function

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Generator Function Demo</title>
</head>
<body>
  <h2>Generator Functions - Lazy Iteration</h2>
  <button id="btn-demo">Run Demo</button>
  <pre id="output"></pre>

  <script>
    /**
     * PROBLEM: Create iterators that yield values on-demand
     * TIME: O(1) per yield SPACE: O(1) per iteration state
     * 
     * WHY: Generators pause execution between yields, enabling
     * infinite sequences, lazy evaluation, and memory-efficient iteration.
     */

    // Generator function syntax: function*
    function* numberGenerator(start, end) {
      for (let i = start; i <= end; i++) {
        yield i; // Pause and return value
      }
    }

    // Infinite sequence generator
    function* infiniteSequence() {
      let n = 0;
      while (true) {
        yield n++;
      }
    }

    // Fibonacci generator
    function* fibonacci() {
      let [a, b] = [0, 1];
      while (true) {
        yield a;
        [a, b] = [b, a + b];
      }
    }

    // Range generator with step
    function* range(start, end, step = 1) {
      for (let i = start; step > 0 ? i <= end : i >= end; i += step) {
        yield i;
      }
    }

    function demonstrateGenerator() {
      const log = [];

      // Create generator instance
      const gen = numberGenerator(1, 5);
      log.push('Generator created');
      
      // Use next() to get values one at a time
      log.push(`First: ${JSON.stringify(gen.next())}`); // {value: 1, done: false}
      log.push(`Second: ${JSON.stringify(gen.next())}`); // {value: 2, done: false}
      
      // Use for...of to iterate all remaining
      log.push('Remaining values:');
      for (const num of gen) {
        log.push(`  ${num}`);
      }

      // Fibonacci first 10 numbers
      log.push('\nFirst 10 Fibonacci:');
      const fib = fibonacci();
      for (let i = 0; i < 10; i++) {
        log.push(`  ${fib.next().value}`);
      }

      document.getElementById('output').textContent = log.join('\n');
    }

    // Practical: Process large datasets in chunks
    function* chunkProcessor(array, chunkSize) {
      for (let i = 0; i < array.length; i += chunkSize) {
        yield array.slice(i, i + chunkSize);
      }
    }

    document.getElementById('btn-demo').addEventListener('click', demonstrateGenerator);
  </script>
</body>
</html>
```

**Why:** Generator functions (`function*`) create iterables that pause at each `yield`, enabling lazy evaluation, infinite sequences, and memory-efficient processing of large datasets. Essential for streaming data, custom iterables, and async iteration patterns.
