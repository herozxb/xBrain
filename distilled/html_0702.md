    /**
     * Creates a downloadable link for dynamic content.
     * Generates a blob URL and triggers download with specified filename.
     * 
     * @param {string} content - The content to download
     * @param {string} filename - The download filename
     * @param {string} mimeType - MIME type (default: text/plain)
     * @returns {HTMLAnchorElement} The anchor element for download
     */
    function createDownloadLink(content, filename, mimeType = 'text/plain') {
      // Create blob from content
      const blob = new Blob([content], { type: mimeType });
      
      // Generate object URL for the blob
      const blobUrl = URL.createObjectURL(blob);
      
      // Create and configure anchor element
      const anchor = document.createElement('a');
      anchor.href = blobUrl;
      anchor.download = filename;
      anchor.textContent = `Download ${filename}`;
      
      // Clean up blob URL when anchor is removed (memory management)
      anchor.addEventListener('click', () => {
        // Revoke URL after a short delay to allow download to start
        setTimeout(() => URL.revokeObjectURL(blobUrl), 100);
      });
      
      return anchor;
    }

    /**
     * Programmatically triggers a file download.
     * Useful for download buttons that aren't anchor elements.
     * 
     * @param {Blob|string} data - Blob or string content to download
     * @param {string} filename - The download filename
     */
    function triggerDownload(data, filename) {
      const blob = data instanceof Blob ? data : new Blob([data]);
      const url = URL.createObjectURL(blob);
      
      // Create temporary anchor and trigger click
      const tempLink = document.createElement('a');
      tempLink.href = url;
      tempLink.download = filename;
      tempLink.style.display = 'none';
      
      document.body.appendChild(tempLink);
      tempLink.click();
      document.body.removeChild(tempLink);
      
      // Clean up
      URL.revokeObjectURL(url);
    }

    /**
     * Downloads a canvas as a PNG image.
     * 
     * @param {HTMLCanvasElement} canvas - The canvas to download
     * @param {string} filename - The download filename
     */
    function downloadCanvas(canvas, filename = 'canvas.png') {

  <script>
    /**
     * Finds all elements that are adjacent siblings of a specific type.
     * Mirrors the CSS + combinator behavior in JavaScript.
     * 
     * @param {HTMLElement} parent - Container to search within
     * @param {string} firstSelector - Selector for the first element
     * @param {string} secondSelector - Selector for the adjacent sibling
     * @returns {Element[]} Array of matching adjacent siblings
     */
    function findAdjacentSiblings(parent, firstSelector, secondSelector) {
      const firstElements = parent.querySelectorAll(firstSelector);
      const matches = [];
      
      firstElements.forEach((element) => {
        const nextSibling = element.nextElementSibling;
        if (nextSibling && nextSibling.matches(secondSelector)) {
          matches.push(nextSibling);
        }
      });
      
      return matches;
    }

    /**
     * Applies styles to adjacent siblings programmatically.
     * Useful when CSS can't be used directly.
     * 
     * @param {HTMLElement} parent - Container element
     * @param {string} firstSelector - First element selector
     * @param {string} secondSelector - Adjacent sibling selector
     * @param {Object} styles - CSS styles to apply
     */
    function styleAdjacentSiblings(parent, firstSelector, secondSelector, styles) {
      const siblings = findAdjacentSiblings(parent, firstSelector, secondSelector);
      
      siblings.forEach((sibling) => {
        Object.assign(sibling.style, styles);
      });
      
      console.log(`Styled ${siblings.length} adjacent siblings`);
    }

    /**
     * Creates a dynamic adjacent sibling relationship observer.
     * Reapplies styles when DOM changes.
     * 
     * @param {HTMLElement} parent - Container to observe
     * @param {string} firstSelector - First element selector
     * @param {string} secondSelector - Adjacent sibling selector
     * @param {Object} styles - CSS styles to apply
     * @returns {MutationObserver} The observer instance
     */
    function observeAdjacentSiblings(parent, firstSelector, secondSelector, styles) {
      const applyStyles = () => {
        // Reset previous styles (optional, based on use case)
        styleAdjacentSiblings(parent, firstSelector, secondSelector, styles);
      };
      
      const observer = new MutationObserver(applyStyles);
      observer.observe(parent, { childList: true, subtree: true });
      
      // Initial application
      applyStyles();
      
      return observer;
    }

      'label',
      { cursor: 'pointer', userSelect: 'none' }

  <script>
    /**
     * Finds all general siblings matching a selector after a reference element.
     * Mirrors the CSS ~ combinator behavior in JavaScript.
     * 
     * @param {Element} element - Reference element
     * @param {string} siblingSelector - Selector for matching siblings
     * @returns {Element[]} Array of all matching following siblings
     */
    function findGeneralSiblings(element, siblingSelector) {
      const siblings = [];
      let current = element.nextElementSibling;
      
      while (current) {
        if (current.matches(siblingSelector)) {
          siblings.push(current);
        }
        current = current.nextElementSibling;
      }
      
      return siblings;
    }

    /**
     * Applies a callback to all general siblings of an element.
     * Useful for interactive state management.
     * 
     * @param {Element} element - Reference element
     * @param {string} siblingSelector - Selector for siblings
     * @param {Function} callback - Function to apply to each sibling
     */
    function forEachGeneralSibling(element, siblingSelector, callback) {
      const siblings = findGeneralSiblings(element, siblingSelector);
      siblings.forEach((sibling, index) => {
        callback(sibling, index, siblings.length);
      });
    }

    /**
     * Creates a hover effect where hovered item stays prominent
     * and all following siblings are dimmed.
     * 
     * @param {HTMLElement} container - Container with sibling items
     * @param {string} itemSelector - Selector for items
     */
    function setupFollowingSiblingDim(container, itemSelector) {
      const items = container.querySelectorAll(itemSelector);
      
      items.forEach((item) => {
        item.addEventListener('mouseenter', () => {
          // Reset all items first
          items.forEach(i => {
            i.style.opacity = '1';
            i.style.transform = 'scale(1)';
          });
          
          // Dim all following siblings
          forEachGeneralSibling(item, itemSelector, (sibling) => {
            sibling.style.opacity = '0.5';
            sibling.style.transform = 'scale(0.98)';
          });
        });
      });
      
      container.addEventListener('mouseleave', () => {
        items.forEach(item => {
          item.style.opacity = '1';
          item.style.transform = 'scale(1)';
        });
      });
    }

    /**
     * Finds elements between two siblings (inclusive or exclusive).
     * Useful for range selection in lists.
     * 
     * @param {Element} start - Starting element
     * @param {Element} end - Ending element
     * @param {boolean} inclusive - Include start and end elements
     * @returns {Element[]} Elements between start and end
     */
    function findSiblingsBetween(start, end, inclusive = true) {
      const between = [];
      let current = inclusive ? start : start.nextElementSibling;
      
      while (current && current !== (inclusive ? end : end.nextElementSibling)) {
        between.push(current);
        current = current.nextElementSibling;
      }
      
      return between;
    }

    // Setup interactive demo

    setupFollowingSiblingDim(cardContainer, '.card');

    const followingSteps = findGeneralSiblings(activeStep, '.step');
    console.log(`Steps after active: ${followingSteps.length}`);
  </script>

    /**
     * Checks if an element or any of its descendants has focus.
     * JavaScript equivalent of :focus-within.
     * 
     * @param {HTMLElement} element - Element to check
     * @returns {boolean} True if element contains focus
     */
    function containsFocus(element) {
      return element.contains(document.activeElement);
    }

    /**
     * Adds a class to an element when it contains focus.
     * Useful for browsers without :focus-within support.
     * 
     * @param {HTMLElement} element - Element to observe
     * @param {string} className - Class to add (default: 'has-focus')
     * @returns {Function} Cleanup function
     */
    function polyfillFocusWithin(element, className = 'has-focus') {
      const updateClass = () => {
        if (containsFocus(element)) {
          element.classList.add(className);
        } else {
          element.classList.remove(className);
        }
      };

      // Add focus/blur listeners to element and all descendants
      const events = ['focus', 'blur'];
      const handler = () => setTimeout(updateClass, 0);

      element.addEventListener('focusin', handler);
      element.addEventListener('focusout', handler);

      // Initial check
      updateClass();

      // Return cleanup function
      return () => {
        element.removeEventListener('focusin', handler);
        element.removeEventListener('focusout', handler);
      };
    }

    /**
     * Creates a focus-within aware container with callbacks.
     * Useful for complex interactive components.
     * 
     * @param {HTMLElement} container - Container element
     * @param {Object} callbacks - onFocusIn and onFocusOut callbacks
     * @returns {Function} Cleanup function
     */
    function onFocusWithin(container, callbacks) {
      const { onFocusIn, onFocusOut } = callbacks;
      let hasFocus = false;

      const checkFocus = () => {
        const nowHasFocus = containsFocus(container);
        
        if (nowHasFocus && !hasFocus && onFocusIn) {
          onFocusIn(container);
        } else if (!nowHasFocus && hasFocus && onFocusOut) {
          onFocusOut(container);
        }
        
        hasFocus = nowHasFocus;
      };

      container.addEventListener('focusin', checkFocus);
      container.addEventListener('focusout', () => setTimeout(checkFocus, 0));

      return () => {
        container.removeEventListener('focusin', checkFocus);
        container.removeEventListener('focusout', checkFocus);
      };
    }

    // Example: Log when form group receives/loses focus
    const formGroup = document.querySelector('.form-group');
    onFocusWithin(formGroup, {
      onFocusIn: (el) => console.log('Form group received focus'),
      onFocusOut: (el) => console.log('Form group lost focus')
    });

  <script>
    /**
     * Creates a CSS linear-gradient string from configuration.
     * Handles direction, color stops, and transparency.
     * 
     * @param {Object} config - Gradient configuration
     * @returns {string} CSS gradient string
     */
    function createLinearGradient(config) {
      const { direction = 'to right', colors = [] } = config;
      
      // Format color stops
      const stops = colors.map((color, index) => {
        if (typeof color === 'string') {
          return color;
        }
        // Color object with position: { color: '#fff', position: 50 }
        return `${color.color} ${color.position}%`;
      });
      
      return `linear-gradient(${direction}, ${stops.join(', ')})`;
    }

    /**
     * Generates a random hex color.
     * 
     * @returns {string} Random hex color (e.g., '#a3f2c1')
     */
    function randomColor() {
      return '#' + Math.floor(Math.random() * 16777215)
        .toString(16)
        .padStart(6, '0');
    }

    /**
     * Generates a random linear gradient with specified color count.
     * 
     * @param {number} colorCount - Number of colors (default: 2)
     * @param {number} angle - Gradient angle in degrees (default: random)
     * @returns {string} CSS gradient string
     */
    function randomLinearGradient(colorCount = 2, angle = null) {
      const colors = Array.from({ length: colorCount }, () => randomColor());
      const gradientAngle = angle !== null ? angle : Math.floor(Math.random() * 360);
      
      return createLinearGradient({
        direction: `${gradientAngle}deg`,
        colors
      });
    }

    /**
     * Creates a gradient that transitions between an array of colors.
     * Automatically calculates even stop positions.
     * 
     * @param {string[]} colors - Array of color strings
     * @param {string} direction - CSS gradient direction
     * @returns {string} CSS gradient string
     */
    function evenGradient(colors, direction = 'to right') {
      const stops = colors.map((color, index) => {
        const position = Math.round((index / (colors.length - 1)) * 100);
        return { color, position };
      });
      
      return createLinearGradient({ direction, colors: stops });
    }

    /**
     * Creates a gradient with hard color stripes (no smooth transition).
     * 
     * @param {string[]} colors - Array of colors
     * @param {string} direction - CSS gradient direction
     * @returns {string} CSS gradient string
     */
    function stripedGradient(colors, direction = 'to right') {
      const stops = [];
      const step = 100 / colors.length;
      
      colors.forEach((color, index) => {
        const start = index * step;
        const end = (index + 1) * step;
        stops.push({ color, position: start });
        stops.push({ color, position: end });
      });
      
      return createLinearGradient({ direction, colors: stops });
    }

    /**
     * Animates a gradient by shifting colors.
     * Creates a "moving" gradient effect.
     * 
     * @param {HTMLElement} element - Element to animate
     * @param {string[]} colors - Colors to cycle through
     * @param {number} duration - Duration of one cycle (ms)
     * @returns {number} Animation frame ID
     */
    function animateGradient(element, colors, duration = 3000) {
      let offset = 0;
      const shift = () => {
        offset = (offset + 1) % 360;
        const gradient = createLinearGradient({
          direction: `${offset}deg`,
          colors
        });
        element.style.background = gradient;
        return requestAnimationFrame(shift);
      };
      
      return shift();
    }

    }));

    console.log('Striped:', stripedGradient(['#ff0000', '#00ff00', '#0000ff']));
