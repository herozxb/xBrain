<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>HTML Source Element</title>
</head>
<body>
    <!-- PROBLEM: Provide multiple video formats for browser compatibility
         TIME: O(1) SPACE: O(1) -->
    <video controls width="640" poster="thumbnail.jpg">
        <!-- Modern browsers: WebM (better compression) -->
        <source src="video.webm" type="video/webm">
        
        <!-- Safari/iOS: MP4 (H.264) -->
        <source src="video.mp4" type="video/mp4">
        
        <!-- Older browsers: OGG -->
        <source src="video.ogv" type="video/ogg">
        
        <!-- Ultimate fallback -->
        <p>Your browser doesn't support HTML5 video. 
           <a href="video.mp4">Download the video</a>.</p>
    </video>
    <script>
        // Why: Source element provides format fallbacks for media elements
        // Browser selects first supported format automatically
        const video = document.querySelector('video');
        const sources = video.querySelectorAll('source');
        
        sources.forEach((source, index) => {
            console.log(`Source ${index + 1}: ${source.type}`);
        });
        
        video.addEventListener('loadedmetadata', () => {
            const currentSource = video.querySelector(`source[src="${video.currentSrc.split('/').pop()}"]`);
            console.log(`Playing: ${currentSource ? currentSource.type : 'unknown format'}`);
        });
    </script>
</body>
</html>

## html_input_file
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>HTML Input File</title>
    <style>
        /* PROBLEM: File upload with preview and validation
           TIME: O(n) where n = file size SPACE: O(m) where m = preview size */
        .file-preview {
            max-width: 300px;
            max-height: 200px;
            margin-top: 1rem;
            display: none;
        }
        .file-info {
            margin-top: 0.5rem;
            font-family: monospace;
            font-size: 0.9rem;
        }
        .error {
            color: #e74c3c;
            margin-top: 0.5rem;
        }
    </style>
</head>
<body>
    <label for="fileInput">Upload an image (max 5MB):</label>
    <input type="file" id="fileInput" accept="image/*">
    
    <div id="error" class="error"></div>
    <img id="preview" class="file-preview" alt="Preview">
    <div id="fileInfo" class="file-info"></div>
    <script>
        // Why: File input enables client-side file handling
        // Accept attribute filters file types, improving UX
        const fileInput = document.getElementById('fileInput');
        const preview = document.getElementById('preview');
        const fileInfo = document.getElementById('fileInfo');
        const errorDiv = document.getElementById('error');
        
        const MAX_SIZE = 5 * 1024 * 1024; // 5MB
        
        fileInput.addEventListener('change', (e) => {
            const file = e.target.files[0];
            errorDiv.textContent = '';
            
            if (!file) return;
            
            // Validate file size
            if (file.size > MAX_SIZE) {
                errorDiv.textContent = `File too large: ${(file.size / 1024 / 1024).toFixed(2)}MB (max 5MB)`;
                return;
            }
            
            // Display file info
            fileInfo.textContent = `Name: ${file.name} | Size: ${(file.size / 1024).toFixed(2)}KB | Type: ${file.type}`;
            
            // Create preview for images
            if (file.type.startsWith('image/')) {
                const reader = new FileReader();
                reader.onload = (e) => {
                    preview.src = e.target.result;
                    preview.style.display = 'block';
                };
                reader.readAsDataURL(file);
            } else {
                preview.style.display = 'none';
            }
        });
    </script>
</body>
</html>
```

            cursor: pointer;
            border-radius: 4px;
        }
        .controls button:hover {
            background: #2980b9;
        }

            flex: 1;
            height: 8px;
            margin: auto 0;
        }
    </style>
</head>
<body>
    <div class="video-container">
        <video id="video" poster="poster.jpg">
            <source src="sample.mp4" type="video/mp4">
            <source src="sample.webm" type="video/webm">
        </video>
        
        <div class="controls">
            <button id="playPause">Play</button>
            <button id="mute">Mute</button>
            <input type="range" id="progress" class="progress" value="0" min="0" max="100">
            <button id="fullscreen">Fullscreen</button>
        </div>
    </div>
    <script>
        // Why: Video element enables native multimedia playback
        // JavaScript API provides full control over playback
        const video = document.getElementById('video');
        const playPauseBtn = document.getElementById('playPause');
        const muteBtn = document.getElementById('mute');
        const progress = document.getElementById('progress');
        const fullscreenBtn = document.getElementById('fullscreen');
        
        playPauseBtn.addEventListener('click', () => {
            if (video.paused) {
                video.play();
                playPauseBtn.textContent = 'Pause';
            } else {
                video.pause();
                playPauseBtn.textContent = 'Play';
            }
        });
        
        muteBtn.addEventListener('click', () => {
            video.muted = !video.muted;
            muteBtn.textContent = video.muted ? 'Unmute' : 'Mute';
        });
        
        video.addEventListener('timeupdate', () => {
            const percent = (video.currentTime / video.duration) * 100;
            progress.value = percent;
        });
        
        progress.addEventListener('input', () => {
            const time = (progress.value / 100) * video.duration;
            video.currentTime = time;
        });
        
        fullscreenBtn.addEventListener('click', () => {
            if (video.requestFullscreen) {
                video.requestFullscreen();
            }
        });
    </script>
</body>
</html>
```

            align-items: center;

            transition: transform 0.1s;
        }

            transform: scale(0.95);
        }

            background: #e74c3c;
        }
        .info {
            margin-top: 1rem;
            font-family: monospace;
        }
    </style>
</head>
<body>
    <div class="click-box" id="clickBox">Click Me!</div>
    <div class="info">
        <div>Clicks: <span id="clickCount">0</span></div>
        <div>Last click: <span id="lastClick">-</span></div>
        <div>Double clicks: <span id="doubleClickCount">0</span></div>
    </div>
    <script>
        // Why: Click event is fundamental for user interaction
        // Distinguish between single and double clicks
        const clickBox = document.getElementById('clickBox');
        const clickCountEl = document.getElementById('clickCount');
        const lastClickEl = document.getElementById('lastClick');
        const doubleClickCountEl = document.getElementById('doubleClickCount');
        
        let clickCount = 0;
        let doubleClickCount = 0;
        let clickTimer = null;
        
        // Single click handler
        clickBox.addEventListener('click', (e) => {
            clickCount++;
            clickCountEl.textContent = clickCount;
            
            // Get click position
            const rect = clickBox.getBoundingClientRect();
            const x = e.clientX - rect.left;
            const y = e.clientY - rect.top;
            lastClickEl.textContent = `(${Math.round(x)}, ${Math.round(y)})`;
            
            // Visual feedback
            clickBox.style.background = `hsl(${Math.random() * 360}, 70%, 50%)`;
        });
        
        // Double click handler
        clickBox.addEventListener('dblclick', (e) => {
            doubleClickCount++;
            doubleClickCountEl.textContent = doubleClickCount;
            
            // Visual feedback for double click
            clickBox.classList.add('double-click');
            setTimeout(() => clickBox.classList.remove('double-click'), 300);
        });
        
        // Right-click context menu
        clickBox.addEventListener('contextmenu', (e) => {
            e.preventDefault();
            console.log('Right-click detected at:', e.clientX, e.clientY);
        });
    </script>
</body>
</html>
```
