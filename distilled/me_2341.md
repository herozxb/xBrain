## 1. Trie String Matching
```python
# PROBLEM: Implement efficient string matching using a Trie data structure
# TIME: O(m) for search where m is pattern length, O(n*m) to build where n is number of words
# SPACE: O(ALPHABET_SIZE * m * n) for storing all characters
class TrieNode:
    """Node for Trie data structure."""
    def __init__(self):
        self.children = {}  # Map character to child node
        self.is_end = False  # Marks end of a word

class Trie:
    """Trie-based string matching for efficient prefix and exact searches."""
    
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word: str) -> None:
        """Insert a word into the trie."""
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
    
    def search(self, word: str) -> bool:
        """Check if exact word exists in trie."""
        node = self.root
        for char in word:
            if char not in node.children:
                return False
            node = node.children[char]
        return node.is_end
    
    def starts_with(self, prefix: str) -> bool:
        """Check if any word starts with given prefix."""
        node = self.root
        for char in prefix:
            if char not in node.children:
                return False
            node = node.children[char]
        return True

def trie_string_match(words: list[str], queries: list[str]) -> list[bool]:
    """
    Match multiple queries against a set of words using Trie.
    Returns list of booleans indicating if each query exists.
    """
    trie = Trie()
    for word in words:
        trie.insert(word)
    return [trie.search(q) for q in queries]
```
**Why**: Trie provides O(m) string matching regardless of total number of words, ideal for autocomplete, spell checkers, and prefix-based searches.

## 2. Fractional Knapsack
```python
# PROBLEM: Select items to maximize value with weight constraint, allowing fractions
# TIME: O(n log n) for sorting items by value/weight ratio
# SPACE: O(1) additional space (O(n) for sorted copy)
def fractional_knapsack(weights: list[float], values: list[float], capacity: float) -> float:
    """
    Solve fractional knapsack problem using greedy approach.
    Items can be taken in fractions to maximize total value.
    """
    # Create list of (value/weight ratio, weight, value)
    items = [(v / w, w, v) for w, v in zip(weights, values) if w > 0]
    
    # Sort by value/weight ratio in descending order
    items.sort(reverse=True, key=lambda x: x[0])
    
    total_value = 0.0
    remaining_capacity = capacity
    
    for ratio, weight, value in items:
        if remaining_capacity <= 0:
            break
        
        # Take as much as possible of this item
        take = min(weight, remaining_capacity)
        total_value += take * ratio
        remaining_capacity -= take
    
    return total_value
```
**Why**: Greedy algorithm works optimally for fractional knapsack (unlike 0/1 knapsack), taking highest value-density items first ensures maximum value.

## 3. Regular Expression DP
```python
# PROBLEM: Match string against pattern with '.' and '*' using dynamic programming
# TIME: O(m * n) where m = len(string), n = len(pattern)
# SPACE: O(m * n) for DP table
def is_match(s: str, p: str) -> bool:
    """
    Match string s against pattern p.
    '.' matches any single character
    '*' matches zero or more of the preceding element
    """
    m, n = len(s), len(p)
    # dp[i][j] = True if s[0:i] matches p[0:j]
    dp = [[False] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = True  # Empty string matches empty pattern
    
    # Handle patterns like a*, a*b*, a*b*c* that can match empty string
    for j in range(2, n + 1):
        if p[j - 1] == '*':
            dp[0][j] = dp[0][j - 2]
    
    # Fill DP table
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if p[j - 1] == '*':
                # * can match zero occurrences
                dp[i][j] = dp[i][j - 2]
                # Or * can match one more occurrence if preceding char matches
                if p[j - 2] == '.' or p[j - 2] == s[i - 1]:
                    dp[i][j] = dp[i][j] or dp[i - 1][j]
            elif p[j - 1] == '.' or p[j - 1] == s[i - 1]:
                # Direct match or wildcard
                dp[i][j] = dp[i - 1][j - 1]
    
    return dp[m][n]
```
**Why**: DP handles the complexity of * operator (zero or more matches) systematically, avoiding exponential backtracking of naive recursion.

## 4. Baby Step Giant Step
```python
# PROBLEM: Solve discrete logarithm: find x where g^x ≡ h (mod p)
# TIME: O(sqrt(p)) using meet-in-the-middle approach
# SPACE: O(sqrt(p)) for storing baby steps
def baby_step_giant_step(g: int, h: int, p: int) -> int:
    """
    Solve discrete logarithm problem: g^x ≡ h (mod p)
    Returns x if solution exists, -1 otherwise.
    Uses meet-in-the-middle optimization.
    """
    import math
    
    if h == 1:
        return 0
    
    # Calculate m = ceil(sqrt(p))
    m = int(math.ceil(math.sqrt(p)))
    
    # Baby steps: compute g^j mod p for j = 0, 1, ..., m-1
    baby_steps = {}
    power = 1
    for j in range(m):
        if power not in baby_steps:  # Keep smallest j
            baby_steps[power] = j
        power = (power * g) % p
    
    # Precompute g^(-m) mod p using Fermat's little theorem
    g_inv_m = pow(g, p - 1 - m, p)  # g^(p-1-m) ≡ g^(-m) mod p
    
    # Giant steps: compute h * (g^(-m))^i for i = 0, 1, ..., m-1
    giant_step = h
    for i in range(m):
        if giant_step in baby_steps:
            return i * m + baby_steps[giant_step]
        giant_step = (giant_step * g_inv_m) % p
    
    return -1  # No solution found
```
**Why**: Meet-in-the-middle reduces O(p) brute force to O(sqrt(p)), essential for cryptographic applications where p is large.

## 5. Pollard's Rho Algorithm
```python
# PROBLEM: Factor large composite numbers efficiently
# TIME: O(n^(1/4)) expected time for finding a factor
# SPACE: O(1) additional space
def pollard_rho(n: int) -> int:
    """
    Find a non-trivial factor of composite n using Pollard's Rho algorithm.
    Returns 1 if n is prime (or fails to find factor).
    """
    import random
    import math
    
    if n < 2:
        return 1
    if n % 2 == 0:
        return 2
    
    x = random.randint(2, n - 1)
    y = x
    c = random.randint(1, n - 1)
    d = 1
    
    # Floyd's cycle detection
    while d == 1:
        # Tortoise: x = f(x)
        x = (x * x + c) % n
        # Hare: y = f(f(y))
        y = (y * y + c) % n
        y = (y * y + c) % n
        # Check GCD
        d = math.gcd(abs(x - y), n)
    
    return d if d != n else 1

def factorize(n: int) -> list[int]:
    """Factor n into prime factors using Pollard's Rho."""
    import math
    
    if n <= 1:
        return []
    
    # Check small primes first
    for p in [2, 3, 5, 7, 11, 13, 17, 19]:
        while n % p == 0:
            n //= p
            yield p
    
    if n == 1:
        return
    
    # Use Pollard's Rho for larger factors
    stack = [n]
    while stack:
        num = stack.pop()
        if num == 1:
            continue
        # Check if prime using simple trial division
        if all(num % p != 0 for p in range(2, int(math.sqrt(num)) + 1)):
            yield num
        else:
            factor = pollard_rho(num)
            if factor > 1:
                stack.append(factor)
                stack.append(num // factor)
            else:
                # Fallback: try harder
                stack.append(num)
```
**Why**: Pollard's Rho is faster than trial division for large numbers, used in cryptographic attacks and integer factorization.

## 6. Arithmetic Coding
```python
# PROBLEM: Compress data using arithmetic coding for optimal entropy encoding
# TIME: O(n) for encoding/decoding where n is message length
# SPACE: O(k) for symbol probability table where k is alphabet size
from collections import Counter
from typing import Dict, Tuple

def arithmetic_encode(message: str, probs: Dict[str, float]) -> Tuple[float, float]:
    """
    Encode message using arithmetic coding.
    Returns (low, high) interval representing the encoded value.
    """
    # Build cumulative probability ranges
    symbols = sorted(probs.keys())
    cum_prob = {symbols[0]: (0.0, probs[symbols[0]])}
    for i in range(1, len(symbols)):
        prev_high = cum_prob[symbols[i-1]][1]
        cum_prob[symbols[i]] = (prev_high, prev_high + probs[symbols[i]])
    
    # Encode message
    low, high = 0.0, 1.0
    for char in message:
        range_size = high - low
        char_low, char_high = cum_prob[char]
        low = low + range_size * char_low
        high = low + range_size * (char_high - char_low)
    
    return low, high

def arithmetic_decode(value: float, length: int, probs: Dict[str, float]) -> str:
    """
    Decode arithmetic coded value back to original message.
    """
    # Build cumulative probability ranges
    symbols = sorted(probs.keys())
    cum_prob = {symbols[0]: (0.0, probs[symbols[0]])}
    for i in range(1, len(symbols)):
        prev_high = cum_prob[symbols[i-1]][1]
        cum_prob[symbols[i]] = (prev_high, prev_high + probs[symbols[i]])
    
    # Decode message
    result = []
    for _ in range(length):
        # Find symbol containing current value
        for symbol, (low, high) in cum_prob.items():
            if low <= value < high:
                result.append(symbol)
                # Rescale value to symbol's range
                value = (value - low) / (high - low)
                break
    
    return ''.join(result)

def build_probabilities(text: str) -> Dict[str, float]:
    """Build probability distribution from text."""
    counter = Counter(text)
    total = len(text)
    return {char: count / total for char, count in counter.items()}
```
**Why**: Arithmetic coding achieves near-optimal compression by encoding entire messages as single floating-point values, better than Huffman for skewed distributions.

## 7. Gaussian Elimination
```python
# PROBLEM: Solve system of linear equations Ax = b using row reduction
# TIME: O(n^3) for n equations with n unknowns
# SPACE: O(n^2) for augmented matrix
def gaussian_elimination(A: list[list[float]], b: list[float]) -> list[float]:
    """
    Solve system of linear equations using Gaussian elimination with partial pivoting.
    Returns solution vector x, or raises exception if no unique solution.
    """
    n = len(A)
    
    # Create augmented matrix [A|b]
    aug = [row[:] + [b[i]] for i, row in enumerate(A)]
    
    # Forward elimination with partial pivoting
    for col in range(n):
        # Find pivot (maximum absolute value in column)
        max_row = max(range(col, n), key=lambda r: abs(aug[r][col]))
        
        if abs(aug[max_row][col]) < 1e-10:
            raise ValueError("Matrix is singular or nearly singular")
        
        # Swap rows
        aug[col], aug[max_row] = aug[max_row], aug[col]
        
        # Eliminate column entries below pivot
        for row in range(col + 1, n):
            factor = aug[row][col] / aug[col][col]
            for j in range(col, n + 1):
                aug[row][j] -= factor * aug[col][j]
    
    # Back substitution
    x = [0.0] * n
    for i in range(n - 1, -1, -1):
        x[i] = aug[i][n]
        for j in range(i + 1, n):
            x[i] -= aug[i][j] * x[j]
        x[i] /= aug[i][i]
    
    return x
```
**Why**: Gaussian elimination is fundamental for solving linear systems, matrix inversion, and computing determinants with O(n³) complexity.

## 8. Discrete Logarithm (Shanks' Algorithm variant)
```python
# PROBLEM: Compute discrete logarithm in multiplicative group mod p
# TIME: O(sqrt(p) log p) with optimizations
# SPACE: O(sqrt(p))
def discrete_log_shanks(g: int, h: int, p: int) -> int:
    """
    Find x such that g^x ≡ h (mod p) using optimized Shanks' algorithm.
    Enhanced version with early termination and memory optimization.
    """
    import math
    
    # Handle special cases
    if h % p == 1:
        return 0
    
    n = int(math.ceil(math.sqrt(p - 1)))
    
    # Baby step: build hash table g^j → j
    table = {}
    gj = 1
    for j in range(n):
        if gj not in table:
            table[gj] = j
        gj = (gj * g) % p
    
    # Precompute g^(-n)
    gn_inv = pow(g, n * (p - 2), p)  # Using Fermat's theorem
    
    # Giant step: search for h * (g^(-n))^i in table
    gamma = h % p
    for i in range(n):
        if gamma in table:
            result = i * n + table[gamma]
            # Verify result
            if pow(g, result, p) == h % p:
                return result
        gamma = (gamma * gn_inv) % p
    
    raise ValueError("No discrete logarithm exists")
```
**Why**: Efficient discrete logarithm computation is critical in cryptography for attacking Diffie-Hellman and ElGamal systems.

## 9. Fermat's Little Theorem Applications
```python
# PROBLEM: Efficient modular arithmetic operations using Fermat's theorem
# TIME: O(log p) for modular inverse using fast exponentiation
# SPACE: O(1)
def mod_inverse_fermat(a: int, p: int) -> int:
    """
    Compute modular inverse using Fermat's Little Theorem.
    Works when p is prime: a^(-1) ≡ a^(p-2) (mod p)
    """
    return pow(a, p - 2, p)

def is_prime_fermat(n: int, k: int = 5) -> bool:
    """
    Probabilistic primality test using Fermat's theorem.
    Tests k random witnesses.
    Returns True if probably prime, False if definitely composite.
    """
    import random
    
    if n < 2:
        return False
    if n == 2 or n == 3:
        return True
    if n % 2 == 0:
        return False
    
    # Test k random witnesses
    for _ in range(k):
        a = random.randint(2, n - 2)
        if pow(a, n - 1, n) != 1:
            return False
    
    return True

def modular_division(numerator: int, denominator: int, p: int) -> int:
    """
    Compute (numerator / denominator) mod p efficiently.
    Uses Fermat's theorem for inverse computation.
    """
    return (numerator * mod_inverse_fermat(denominator, p)) % p
```
**Why**: Fermat's theorem enables O(log p) modular inverses critical for RSA, elliptic curve cryptography, and modular arithmetic operations.

## 10. Euler's Totient Function
```python
# PROBLEM: Count integers from 1 to n that are coprime to n
# TIME: O(sqrt(n)) for single computation
# SPACE: O(1)
def euler_totient(n: int) -> int:
    """
    Compute Euler's totient function φ(n).
    Counts integers k in [1, n] where gcd(n, k) = 1.
    Uses formula: φ(n) = n * ∏(1 - 1/p) for all prime factors p.
    """
    result = n
    p = 2
    
    while p * p <= n:
        if n % p == 0:
            # Remove all occurrences of p from n
            while n % p == 0:
                n //= p
            # Apply formula: multiply by (1 - 1/p)
            result -= result // p
        p += 1
    
    # If n is still > 1, then it's a prime factor
    if n > 1:
        result -= result // n
    
    return result

def euler_totient_range(max_n: int) -> list[int]:
    """
    Compute totient for all numbers 1 to max_n using sieve.
    TIME: O(n log log n), SPACE: O(n)
    """
    phi = list(range(max_n + 1))
    
    for i in range(2, max_n + 1):
        if phi[i] == i:  # i is prime
            for j in range(i, max_n + 1, i):
                phi[j] -= phi[j] // i
    
    return phi

def count_coprime_pairs(n: int) -> int:
    """
    Count pairs (a, b) where 1 ≤ a < b ≤ n and gcd(a, b) = 1.
    Uses cumulative totient sum.
    """
    phi = euler_totient_range(n)
    return sum(phi[2:n+1])
```
**Why**: Euler's totient is fundamental in number theory and cryptography, used in RSA key generation, counting primitive roots, and analyzing group structures.
