## [Child Selector]

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Child Selector</title>
  <style>
    /* PROBLEM: How to select only direct children of an element?
       TIME: O(n) SPACE: O(1) where n = number of elements */
    
    /* Child combinator (>) selects only direct children */
    .parent > li {
      background: #e74c3c;
      color: white;
      padding: 10px;
      margin: 5px;
    }
    
    /* This will NOT affect nested grandchildren */
    .parent > li > ul {
      background: #95a5a6;
      padding: 10px;
    }
    
    /* Grandchildren styling - separate from children */
    .parent > li > ul > li {
      background: #3498db;
      color: white;
      list-style: none;
      padding: 5px;
      margin: 3px;
    }
  </style>
</head>
<body>
  <h2>Child Selector (>) Demo</h2>
  <ul class="parent">
    <li>Direct Child 1</li>
    <li>Direct Child 2
      <ul>
        <li>Grandchild 2.1</li>
        <li>Grandchild 2.2</li>
      </ul>
    </li>
    <li>Direct Child 3</li>
  </ul>
</body>
</html>
```

**Why**: The child selector (`>`) is more specific than descendant selector (space). It only targets immediate children, preventing unwanted styling of deeply nested elements.

## [Arrow This Binding]

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Arrow Function This Binding</title>
</head>
<body>
  <h2>Arrow Function This Binding Demo</h2>
  <button id="btn1">Regular Function</button>
  <button id="btn2">Arrow Function</button>
  <div id="output"></div>

  <script>
    /* PROBLEM: How does arrow function 'this' differ from regular functions?
       TIME: O(1) SPACE: O(1) */
    
    const output = document.getElementById('output');
    
    // Object with regular method
    const handler1 = {
      name: 'Regular Function',
      greet: function() {
        // Regular function has its own 'this'
        output.innerHTML += `${this.name}: ${this}<br>`;
      }
    };
    
    // Object with arrow method
    const handler2 = {
      name: 'Arrow Function',
      greet: () => {
        // Arrow function inherits 'this' from enclosing scope (window)
        output.innerHTML += `${this?.name || 'Window'}: ${this}<br>`;
      }
    };
    
    // Event handler comparison
    const button1 = document.getElementById('btn1');
    const button2 = document.getElementById('btn2');
    
    // Regular function: 'this' = button element
    button1.addEventListener('click', function() {
      output.innerHTML += `Regular: this = ${this.tagName}<br>`;
    });
    
    // Arrow function: 'this' = enclosing scope (window/global)
    button2.addEventListener('click', () => {
      output.innerHTML += `Arrow: this = ${this === window ? 'Window' : 'Other'}<br>`;
    });
    
    // Practical use: preserving context in callbacks
    class Timer {
      constructor() {
        this.seconds = 0;
      }
      
      startRegular() {
        // Wrong: loses 'this' context
        setInterval(function() {
          // this.seconds is undefined
          output.innerHTML += `Regular: ${this?.seconds ?? 'undefined'}<br>`;
        }, 1000);
      }
      
      startArrow() {
        // Correct: arrow preserves 'this'
        setInterval(() => {
          this.seconds++;
          output.innerHTML += `Arrow: ${this.seconds} seconds<br>`;
        }, 1000);
      }
    }
    
    const timer = new Timer();
    // timer.startArrow(); // Uncomment to test
  </script>
</body>
</html>
```

**Why**: Arrow functions don't have their own `this` binding - they inherit from the enclosing scope. Essential for callbacks where you need to preserve the outer context.

## [Promise Then]

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Promise Then</title>
</head>
<body>
  <h2>Promise Then Demo</h2>
  <button id="fetchBtn">Fetch Data</button>
  <div id="output"></div>

  <script>
    /* PROBLEM: How to chain asynchronous operations with promises?
       TIME: O(1) for synchronous, O(n) for chained operations
       SPACE: O(1) per promise in chain */
    
    const output = document.getElementById('output');
    
    // Simulate async data fetch
    function fetchUser(id) {
      return new Promise((resolve, reject) => {
        setTimeout(() => {
          if (id > 0) {
            resolve({ id, name: `User ${id}` });
          } else {
            reject(new Error('Invalid user ID'));
          }
        }, 500);
      });
    }
    
    // Simulate fetching user posts
    function fetchPosts(userId) {
      return new Promise((resolve) => {
        setTimeout(() => {
          resolve([
            { id: 1, title: 'First Post', userId },
            { id: 2, title: 'Second Post', userId }
          ]);
        }, 300);
      });
    }
    
    document.getElementById('fetchBtn').addEventListener('click', () => {
      output.innerHTML = 'Fetching...<br>';
      
      // Promise chain with .then()
      fetchUser(1)
        .then((user) => {
          output.innerHTML += `User: ${user.name}<br>`;
          return fetchPosts(user.id); // Return another promise
        })
        .then((posts) => {
          output.innerHTML += `Posts: ${posts.length}<br>`;
          posts.forEach(post => {
            output.innerHTML += `  - ${post.title}<br>`;
          });
          return posts.length; // Can return non-promise values
        })
        .then((count) => {
          output.innerHTML += `Total: ${count} posts<br>`;
        })
        .catch((error) => {
          // Catches errors from any .then() in the chain
          output.innerHTML += `Error: ${error.message}<br>`;
        })
        .finally(() => {
          // Always executes
          output.innerHTML += 'Operation complete<br>';
        });
    });
  </script>
</body>
</html>
```

**Why**: `.then()` enables promise chaining, allowing sequential async operations. Each `.then()` returns a new promise, creating a readable async flow without callback hell.

## [Array Map]

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Array Map</title>
</head>
<body>
  <h2>Array Map Demo</h2>
  <div id="output"></div>

  <script>
    /* PROBLEM: How to transform all elements in an array?
       TIME: O(n) SPACE: O(n) where n = array length */
    
    const output = document.getElementById('output');
    
    // Basic transformation
    const numbers = [1, 2, 3, 4, 5];
    const doubled = numbers.map(num => num * 2);
    output.innerHTML += `Doubled: ${doubled}<br>`;
    
    // Object transformation
    const users = [
      { name: 'Alice', age: 25 },
      { name: 'Bob', age: 30 },
      { name: 'Charlie', age: 35 }
    ];
    
    const names = users.map(user => user.name);
    output.innerHTML += `Names: ${names}<br>`;
    
    // Adding index parameter
    const indexed = users.map((user, index) => ({
      ...user,
      id: index + 1
    }));
    output.innerHTML += `With IDs: ${JSON.stringify(indexed)}<br>`;
';
    
    // Extracting specific properties
    const products = [
      { name: 'Laptop', price: 999, inStock: true },
      { name: 'Phone', price: 599, inStock: false },
      { name: 'Tablet', price: 399, inStock: true }
    ];
    
    const prices = products.map(p => p.price);
    output.innerHTML += `Prices: $${prices.join(', $')}<br>`;
    
    // Creating HTML elements
    const listItems = products
      .map(p => `<li>${p.name}: $${p.price}</li>`)
      .join('');
    output.innerHTML += `<ul>${listItems}</ul>`;
    
    // Chaining with filter (map returns new array)
    const affordableProducts = products
      .filter(p => p.price < 700)
      .map(p => p.name);
    output.innerHTML += `Affordable: ${affordableProducts}<br>`;
  </script>
</body>
</html>
```

**Why**: `map()` creates a new array by transforming each element. It's immutable (doesn't modify original), chainable, and declarative - core to functional programming in JavaScript.

## [Object Keys]

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Object Keys</title>
</head>
<body>
  <h2>Object Keys Demo</h2>
  <div id="output"></div>

  <script>
    /* PROBLEM: How to iterate over object properties?
       TIME: O(n) SPACE: O(n) where n = number of properties */
    
    const output = document.getElementById('output');
    
    // Basic object
    const user = {
      name: 'Alice',
      age: 30,
      email: 'alice@example.com',
      active: true
    };
    
    // Get all keys as array
    const keys = Object.keys(user);
    output.innerHTML += `Keys: ${keys}<br>`;
    
    // Iterate over keys
    keys.forEach(key => {
      output.innerHTML += `  ${key}: ${user[key]}<br>`;
    });
    
    // Count properties
    output.innerHTML += `Property count: ${keys.length}<br>`;
    
    // Check if property exists
    output.innerHTML += `Has email: ${keys.includes('email')}<br>`;
    
    // Practical use: transform object to array
    const keyValuePairs = Object.keys(user).map(key => ({
      key,
      value: user[key]
    }));
    output.innerHTML += `Pairs: ${JSON.stringify(keyValuePairs)}<br>`;
    
    // Filter keys
    const stringKeys = Object.keys(user).filter(key => 
      typeof user[key] === 'string'
    );
    output.innerHTML += `String keys: ${stringKeys}<br>`;
    
    // Works with arrays (returns indices)
    const arr = ['a', 'b', 'c'];
    output.innerHTML += `Array keys: ${Object.keys(arr)}<br>`;
    
    // Nested object iteration
    const settings = {
      theme: 'dark',
      notifications: { email: true, sms: false },
      privacy: { public: false }
    };
    
    function getNestedKeys(obj, prefix = '') {
      return Object.keys(obj).flatMap(key => {
        const path = prefix ? `${prefix}.${key}` : key;
        return typeof obj[key] === 'object' && obj[key] !== null
          ? getNestedKeys(obj[key], path)
          : [path];
      });
    }
    
    output.innerHTML += `Nested keys: ${getNestedKeys(settings)}<br>`;
  </script>
</body>
</html>
```

**Why**: `Object.keys()` returns an array of enumerable property names. Essential for iteration, validation, and transforming objects into different data structures.

## [String Slice]

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>String Slice</title>
</head>
<body>
  <h2>String Slice Demo</h2>
  <div id="output"></div>

  <script>
    /* PROBLEM: How to extract a portion of a string?
       TIME: O(n) where n = length of extracted string
       SPACE: O(n) for new string */
    
    const output = document.getElementById('output');
    
    const text = 'Hello, World!';
    
    // Basic slicing
    output.innerHTML += `Original: "${text}"<br>`;
    output.innerHTML += `slice(0, 5): "${text.slice(0, 5)}"<br>`;
    output.innerHTML += `slice(7): "${text.slice(7)}"<br>`;
    
    // Negative indices (count from end)
    output.innerHTML += `slice(-6): "${text.slice(-6)}"<br>`;
    output.innerHTML += `slice(-6, -1): "${text.slice(-6, -1)}"<br>`;
    
    // Common use cases
    const filename = 'document.pdf';
    const extension = filename.slice(filename.lastIndexOf('.'));
    output.innerHTML += `Extension: "${extension}"<br>`;
    
    // Truncate with ellipsis
    function truncate(str, maxLength) {
      if (str.length <= maxLength) return str;
      return str.slice(0, maxLength - 3) + '...';
    }
    output.innerHTML += `Truncated: "${truncate('This is a very long text', 15)}"<br>`;
    
    // Extract domain from email
    const email = 'user@example.com';
    const domain = email.slice(email.indexOf('@') + 1);
    output.innerHTML += `Domain: "${domain}"<br>`;
    
    // Capitalize first letter
    function capitalize(str) {
      return str.slice(0, 1).toUpperCase() + str.slice(1);
    }
    output.innerHTML += `Capitalized: "${capitalize('hello')}"<br>`;
    
    // Get last N characters
    function getLastN(str, n) {
      return str.slice(-n);
    }
    output.innerHTML += `Last 3: "${getLastN('abcdef', 3)}"<br>`;
    
    // Remove first and last character
    const trimmed = text.slice(1, -1);
    output.innerHTML += `Trimmed: "${trimmed}"<br>`;
  </script>
</body>
</html>
```

**Why**: `slice()` extracts a section of a string without modifying the original. Negative indices make it easy to work from the end. Immutable and safer than `substr()` or `substring()`.

## [Geolocation Get Current Position]

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Geolocation getCurrentPosition</title>
  <style>
    #map {
      width: 100%;
      height: 300px;
      background: #ecf0f1;
      display: flex;
      align-items: center;
      justify-content: center;
      margin: 20px 0;
    }
  </style>
</head>
<body>
  <h2>Geolocation getCurrentPosition Demo</h2>
  <button id="locationBtn">Get My Location</button>
  <div id="output"></div>
  <div id="map">Map placeholder</div>

  <script>
    /* PROBLEM: How to get user's current geographical position?
       TIME: O(1) for API call, async response varies
       SPACE: O(1) */
    
    const output = document.getElementById('output');
    const map = document.getElementById('map');
    
    // Check if geolocation is supported
    if (!navigator.geolocation) {
      output.innerHTML = 'Geolocation not supported by your browser';
    }
    
    document.getElementById('locationBtn').addEventListener('click', () => {
      output.innerHTML = 'Getting location...';
      
      // Success callback
      function success(position) {
        const { latitude, longitude, accuracy, altitude, heading, speed } = position.coords;
        const timestamp = new Date(position.timestamp);
        
        output.innerHTML = `
          <strong>Location Found:</strong><br>
          Latitude: ${latitude.toFixed(6)}<br>
          Longitude: ${longitude.toFixed(6)}<br>
          Accuracy: ${accuracy.toFixed(2)} meters<br>
          ${altitude ? `Altitude: ${altitude.toFixed(2)} meters<br>` : ''}
          ${heading ? `Heading: ${heading.toFixed(2)}°<br>` : ''}
          ${speed ? `Speed: ${speed.toFixed(2)} m/s<br>` : ''}
          Timestamp: ${timestamp.toLocaleString()}
        `;
        
        // Display on map (placeholder)
        map.innerHTML = `Lat: ${latitude.toFixed(4)}, Lng: ${longitude.toFixed(4)}`;
      }
      
      // Error callback
      function error(error) {
        const errors = {
          1: 'PERMISSION_DENIED: User denied geolocation',
          2: 'POSITION_UNAVAILABLE: Position unavailable',
          3: 'TIMEOUT: Request timed out'
        };
        output.innerHTML = `Error: ${errors[error.code]}`;
      }
      
      // Options
      const options = {
        enableHighAccuracy: true,  // Use GPS if available
        timeout: 10000,            // Max wait time (ms)
        maximumAge: 0              // Don't use cached position
      };
      
      // Get current position
      navigator.geolocation.getCurrentPosition(success, error, options);
    });
    
    // Watch position for continuous updates
    let watchId;
    
    function startWatching() {
      watchId = navigator.geolocation.watchPosition(
        (position) => {
          console.log('Updated:', position.coords);
        },
        null,
        { enableHighAccuracy: true }
      );
    }
    
    function stopWatching() {
      if (watchId) {
        navigator.geolocation.clearWatch(watchId);
      }
    }
  </script>
</body>
</html>
```

**Why**: `getCurrentPosition()` retrieves the user's location once. Requires HTTPS and user permission. Essential for location-aware applications like maps, delivery, and local recommendations.
