## [1] :not() Pseudo-Class Filter

```html
<!-- PROBLEM: Exclude specific elements from styling -->
<!-- TIME: O(n) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>:not() Pseudo-Class</title>
  <style>
    /* Style all list items except those with .special class */
    ul li:not(.special) {
      color: #666;
      background: #f5f5f5;
      padding: 8px;
      margin: 4px 0;
    }
    
    /* Style all buttons except disabled ones */
    button:not([disabled]) {
      background: #007bff;
      color: white;
      padding: 10px 20px;
      border: none;
      cursor: pointer;
    }
    
    button:not([disabled]):hover {
      background: #0056b3;
    }
    
    /* Hide elements that are not active */
    .tab-panel:not(.active) {
      display: none;
    }
  </style>
</head>
```

**Why**: The `:not()` pseudo-class allows excluding elements from CSS selection without adding extra classes. It reduces DOM manipulation and keeps styles cleaner. Useful for form validation, state management, and conditional styling.

## [7] Object Destructuring

```html
<!-- PROBLEM: Extract values from objects concisely -->
<!-- TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Object Destructuring</title>
</head>
<body>
  <h2>Object Destructuring Examples</h2>
  <div id="output"></div>
  
  <script>
    const output = document.getElementById('output');
    
    // Basic destructuring
    const user = {
      name: 'Alice',
      age: 30,
      email: 'alice@example.com'
    };
    
    const { name, age, email } = user;
    output.innerHTML += `<p><strong>Basic:</strong> ${name}, ${age}, ${email}</p>`;
    
    // Renaming variables
    const { name: userName, age: userAge } = user;
    output.innerHTML += `<p><strong>Renamed:</strong> ${userName}, ${userAge}</p>`;
    
    // Default values
    const { name: n, country = 'USA' } = user;
    output.innerHTML += `<p><strong>Default:</strong> ${n}, ${country}</p>`;
    
    // Nested destructuring
    const company = {
      name: 'TechCorp',
      address: {
        city: 'San Francisco',
        state: 'CA'
      }
    };
    
    const { name: companyName, address: { city, state } } = company;
    output.innerHTML += `<p><strong>Nested:</strong> ${companyName}, ${city}, ${state}</p>`;
    
    // Destructuring in function parameters
    function greet({ name, age }) {
      return `Hello ${name}, you are ${age} years old`;
    }
    
    output.innerHTML += `<p><strong>Function param:</strong> ${greet(user)}</p>`;
    
    // Rest properties
    const { name: firstName, ...rest } = user;
    output.innerHTML += `<p><strong>Rest:</strong> ${firstName}, ${JSON.stringify(rest)}</p>`;
    
    // Destructuring with computed property names
    const key = 'email';
    const { [key]: userEmail } = user;
    output.innerHTML += `<p><strong>Computed:</strong> ${userEmail}</p>`;
    
    // Swapping variables (with array destructuring)
    let a = 1, b = 2;
    [a, b] = [b, a];
    output.innerHTML += `<p><strong>Swap:</strong> a=${a}, b=${b}</p>`;
    
    // Destructuring API response
    const apiResponse = {
      data: {
        users: ['Alice', 'Bob', 'Charlie'],
        total: 3
      },
      status: 200
    };
    
    const { data: { users, total }, status } = apiResponse;
    output.innerHTML += `<p><strong>API:</strong> Status ${status}, Users: ${users.join(', ')}</p>`;
  </script>
</body>
</html>
```

**Why**: Object destructuring extracts values concisely, reducing code verbosity. Essential for API responses, function parameters, and configuration objects. Supports renaming, defaults, nested extraction, and rest operators. Makes code more readable and maintainable.

```html
  <script>
    const output = document.getElementById('output');
    
    // Simulated API call
    function fetchUserAPI(shouldFail = false) {
      return new Promise((resolve, reject) => {
        setTimeout(() => {
          if (shouldFail) {
            reject(new Error('Network error: Failed to fetch user'));
          } else {
            resolve({ id: 1, name: 'Alice', email: 'alice@example.com' });
          }

    // Basic try/catch
    async function fetchUserSuccess() {
      output.innerHTML = '<div class="card warning">Loading...</div>';
      
      try {
        const user = await fetchUserAPI(false);
        output.innerHTML = `

            <h3>Error!</h3>
            <p>${error.message}</p>
          </div>
        `;
      }
    }
    
    // Handling errors

      try {
        const user = await fetchUserAPI(true);
        output.innerHTML = `<div class="card success">${JSON.stringify(user)}</div>`;
      } catch (error) {
        output.innerHTML = `
          <div class="card error">
            <h3>Caught Error!</h3>
            <p>${error.message}</p>
            <p>Stack: ${error.stack}</p>
          </div>
        `;
      }
    }
    
    // Multiple async operations
    async function fetchMultiple() {
      output.innerHTML = '<div class="card warning">Loading all...</div>';
      
      try {
        // Parallel execution
        const [user1, user2, user3] = await Promise.all([
          fetchUserAPI(false),
          fetchUserAPI(false),
          fetchUserAPI(false)
        ]);
        
        output.innerHTML = `
          <div class="card success">
            <h3>All Users Loaded</h3>
            <p>${user1.name}, ${user2.name}, ${user3.name}</p>
          </div>
        `;
      } catch (error) {
        output.innerHTML = `
          <div class="card error">
            <h3>One or more failed</h3>
            <p>${error.message}</p>
          </div>
        `;
      }
    }
    
    // Retry logic
    async function fetchWithRetry(maxRetries = 3) {
      let attempts = 0;
      
      while (attempts < maxRetries) {
        try {
          attempts++;
          output.innerHTML = `<div class="card warning">Attempt ${attempts}/${maxRetries}...</div>`;
          
          const user = await fetchUserAPI(attempts < 3); // Fail first 2 times
          
          output.innerHTML = `
            <div class="card success">
              <h3>Success after ${attempts} attempts!</h3>
              <p>${user.name}</p>
            </div>
          `;
          return user;
        } catch (error) {
          if (attempts >= maxRetries) {
            output.innerHTML = `
              <div class="card error">
                <h3>Failed after ${maxRetries} attempts</h3>
                <p>${error.message}</p>
              </div>
            `;
            throw error;
          }
          
          // Wait before retry
          await new Promise(resolve => setTimeout(resolve, 500));
        }
      }
    }
    
    // Global error handler
    window.addEventListener('unhandledrejection', event => {
      console.error('Unhandled promise rejection:', event.reason);
      event.preventDefault();
    });
  </script>
</body>
</html>
```

**Why**: Async/await with try/catch provides synchronous-looking error handling for async code. More readable than promise chains. Supports retry logic, parallel operations, and comprehensive error information. Essential for robust API interactions and user feedback.
