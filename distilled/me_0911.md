## [Divide and Conquer - Maximum Subarray]

```python
# PROBLEM: Find the contiguous subarray with maximum sum
# TIME: O(n log n) SPACE: O(log n) recursion depth
def max_subarray_dc(nums):
    """Find maximum subarray sum using divide and conquer"""
    
    def helper(left, right):
        # Base case: single element
        if left == right:
            return nums[left]
        
        mid = (left + right) // 2
        
        # Recursively find max in left and right halves
        left_max = helper(left, mid)
        right_max = helper(mid + 1, right)
        
        # Find max crossing middle: extend from mid in both directions
        # Left extension (mid -> left)
        left_sum = float('-inf')
        curr = 0
        for i in range(mid, left - 1, -1):
            curr += nums[i]
            left_sum = max(left_sum, curr)
        
        # Right extension (mid+1 -> right)
        right_sum = float('-inf')
        curr = 0
        for i in range(mid + 1, right + 1):
            curr += nums[i]
            right_sum = max(right_sum, curr)
        
        # Cross sum = max left extension + max right extension
        cross_max = left_sum + right_sum
        
        # Return max of three candidates
        return max(left_max, right_max, cross_max)
    
    return helper(0, len(nums) - 1)
```

**Why**: Divide and conquer breaks complex problems into smaller subproblems. Classic pattern for optimization problems. Though Kadane's algorithm is O(n), this demonstrates the paradigm used in merge sort, quicksort, and Strassen's matrix multiplication.

## [Kruskal's Algorithm - Minimum Spanning Tree]

```python
# PROBLEM: Find minimum spanning tree of a weighted undirected graph
# TIME: O(E log E) for sorting edges SPACE: O(V) for union-find
class UnionFind:
    """Union-Find (Disjoint Set Union) with path compression and union by rank"""
    
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
    
    def find(self, x):
        """Find root with path compression"""
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        """Union by rank: return True if merged, False if already connected"""
        px, py = self.find(x), self.find(y)
        if px == py:
            return False
        
        # Union by rank: attach shorter tree to taller
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        return True

def kruskal(n, edges):
    """
    Find MST using Kruskal's algorithm
    n: number of vertices (0 to n-1)
    edges: list of (weight, u, v) tuples
    Returns: MST weight and list of edges in MST
    """
    # Sort edges by weight (greedy: pick smallest first)
    edges.sort()
    
    uf = UnionFind(n)
    mst_edges = []
    mst_weight = 0
    
    for weight, u, v in edges:
        # Add edge if it doesn't create a cycle
        if uf.union(u, v):
            mst_edges.append((u, v, weight))
            mst_weight += weight
            
            # MST has exactly n-1 edges
            if len(mst_edges) == n - 1:
                break
    
    return mst_weight, mst_edges
```

**Why**: MST algorithms optimize network design (cables, roads, pipelines) by connecting all nodes with minimum total cost. Kruskal's greedy approach + union-find is elegant and efficient. Used in clustering algorithms and network routing.

## [Bellman-Ford Algorithm - Single Source Shortest Path]

```python
# PROBLEM: Find shortest paths from source to all vertices (handles negative weights)
# TIME: O(V*E) SPACE: O(V)
def bellman_ford(n, edges, source):
    """
    Bellman-Ford algorithm for shortest paths with negative weights
    n: number of vertices
    edges: list of (u, v, weight) directed edges
    source: starting vertex
    Returns: distances dict and boolean for negative cycle detection
    """
    # Initialize distances: source=0, others=infinity
    dist = {i: float('inf') for i in range(n)}
    dist[source] = 0
    
    # Relax all edges (n-1) times
    # After i iterations, we have shortest paths using at most i edges
    for _ in range(n - 1):
        updated = False
        for u, v, w in edges:
            if dist[u] != float('inf') and dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                updated = True
        
        # Early termination if no updates
        if not updated:
            break
    
    # Check for negative cycles: if we can still relax, cycle exists
    has_negative_cycle = False
    for u, v, w in edges:
        if dist[u] != float('inf') and dist[u] + w < dist[v]:
            has_negative_cycle = True
            break
    
    return dist, has_negative_cycle

# Example: Detect arbitrage in currency exchange
def find_arbitrage(currencies, rates):
    """
    rates[i][j] = exchange rate from currency i to j
    Returns True if arbitrage opportunity exists
    """
    n = len(currencies)
    edges = []
    
    # Convert to log space: arbitrage exists if negative cycle
    # product of rates > 1 iff sum of log(rates) > 0
    for i in range(n):
        for j in range(n):
            if i != j and rates[i][j] > 0:
                edges.append((i, j, -math.log(rates[i][j])))
    
    _, has_cycle = bellman_ford(n, edges, 0)
    return has_cycle
```

**Why**: Unlike Dijkstra, Bellman-Ford handles negative weights and detects negative cycles. Essential for currency arbitrage detection, network routing with costs, and scenarios with penalty weights. Slower than Dijkstra but more versatile.

## [AVL Tree - Self-Balancing BST]

```python
# PROBLEM: Maintain balanced BST for guaranteed O(log n) operations
# TIME: O(log n) insert/delete/search SPACE: O(n)
class AVLNode:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None
        self.height = 1  # Height of node (leaf has height 1)

class AVLTree:
    """Self-balancing AVL tree with rotation-based balancing"""
    
    def get_height(self, node):
        """Get height of node (None has height 0)"""
        return node.height if node else 0
    
    def get_balance(self, node):
        """Get balance factor: left height - right height"""
        return self.get_height(node.left) - self.get_height(node.right) if node else 0
    
    def update_height(self, node):
        """Update height based on children"""
        node.height = 1 + max(self.get_height(node.left), self.get_height(node.right))
    
    def rotate_right(self, y):
        """Right rotation: left child becomes root"""
        x = y.left
        T2 = x.right
        
        # Perform rotation
        x.right = y
        y.left = T2
        
        # Update heights (y first, then x)
        self.update_height(y)
        self.update_height(x)
        
        return x  # New root
    
    def rotate_left(self, x):
        """Left rotation: right child becomes root"""
        y = x.right
        T2 = y.left
        
        # Perform rotation
        y.left = x
        x.right = T2
        
        # Update heights
        self.update_height(x)
        self.update_height(y)
        
        return y  # New root
    
    def insert(self, node, key):
        """Insert key and rebalance tree"""
        # Standard BST insertion
        if not node:
            return AVLNode(key)
        
        if key < node.key:
            node.left = self.insert(node.left, key)
        elif key > node.key:
            node.right = self.insert(node.right, key)
        else:
            return node  # Duplicate keys not allowed
        
        # Update height
        self.update_height(node)
        
        # Check balance factor
        balance = self.get_balance(node)
        
        # Four cases of imbalance:
        
        # Left Left: right rotation
        if balance > 1 and key < node.left.key:
            return self.rotate_right(node)
        
        # Right Right: left rotation
        if balance < -1 and key > node.right.key:
            return self.rotate_left(node)
        
        # Left Right: left rotation on left child, then right rotation
        if balance > 1 and key > node.left.key:
            node.left = self.rotate_left(node.left)
            return self.rotate_right(node)
        
        # Right Left: right rotation on right child, then left rotation
        if balance < -1 and key < node.right.key:
            node.right = self.rotate_right(node.right)
            return self.rotate_left(node)
        
        return node  # No rotation needed
    
    def search(self, node, key):
        """Search for key in tree"""
        if not node or node.key == key:
            return node
        
        if key < node.key:
            return self.search(node.left, key)
        return self.search(node.right, key)
```

**Why**: AVL trees guarantee O(log n) operations by maintaining balance through rotations. Used in databases, file systems, and in-memory sorted structures. Red-Black trees are more common in practice (less rigid balancing), but AVL offers faster lookups.

## [Mediator Pattern - Decouple Component Communication]

```python
# PROBLEM: Reduce direct dependencies between components by centralizing communication
# TIME: O(1) for simple routing SPACE: O(n) for n components
class ChatMediator:
    """Mediator for chat room: users don't communicate directly"""
    
    def __init__(self):
        self.users = {}  # name -> User object
    
    def register(self, user):
        """Register user with mediator"""
        self.users[user.name] = user
    
    def send(self, message, sender, recipient=None):
        """
        Route message through mediator
        If recipient is None, broadcast to all
        """
        if recipient:
            # Direct message
            if recipient in self.users:
                self.users[recipient].receive(message, sender)
        else:
            # Broadcast to all except sender
            for name, user in self.users.items():
                if name != sender:
                    user.receive(message, sender)

class User:
    """User component that only talks to mediator"""
    
    def __init__(self, name, mediator):
        self.name = name
        self.mediator = mediator
        mediator.register(self)
    
    def send(self, message, recipient=None):
        """Send message through mediator"""
        self.mediator.send(message, self.name, recipient)
    
    def receive(self, message, sender):
        """Receive message from mediator"""
        print(f"[{self.name}] From {sender}: {message}")

# Usage
mediator = ChatMediator()
alice = User("Alice", mediator)
bob = User("Bob", mediator)
charlie = User("Charlie", mediator)

alice.send("Hello everyone!")  # Broadcast
bob.send("Hi Alice!", "Alice")  # Direct message
```

**Why**: Mediator pattern decouples components by centralizing communication logic. Prevents "spaghetti code" in systems with many interacting objects. Used in UI frameworks (dialog coordinators), air traffic control systems, and chat applications. Promotes single responsibility: components focus on their job, mediator handles interactions.

## [Chain of Responsibility - Request Processing Pipeline]

```python
# PROBLEM: Process requests through a chain of handlers dynamically
# TIME: O(n) for n handlers in chain SPACE: O(1) per handler
from abc import ABC, abstractmethod

class Handler(ABC):
    """Abstract handler in the chain"""
    
    def __init__(self):
        self.next_handler = None
    
    def set_next(self, handler):
        """Set next handler in chain"""
        self.next_handler = handler
        return handler  # Enable chaining: h1.set_next(h2).set_next(h3)
    
    def handle(self, request):
        """
        Process request or pass to next handler
        Returns True if handled, False otherwise
        """
        if self._process(request):
            return True
        
        if self.next_handler:
            return self.next_handler.handle(request)
        
        return False  # End of chain, unhandled
    
    @abstractmethod
    def _process(self, request):
        """Override to implement specific handling logic"""
        pass

class AuthHandler(Handler):
    """Check if user is authenticated"""
    
    def _process(self, request):
        if not request.get('authenticated'):
            print("❌ Authentication failed")
            return True  # Handled (blocked)
        print("✓ Authenticated")
        return False  # Pass to next

class PermissionHandler(Handler):
    """Check if user has required permissions"""
    
    def _process(self, request):
        required = request.get('required_permission')
        user_perms = request.get('permissions', [])
        
        if required and required not in user_perms:
            print(f"❌ Missing permission: {required}")
            return True
        print("✓ Permissions OK")
        return False

class RateLimitHandler(Handler):
    """Check rate limiting"""
    
    def __init__(self, max_requests=10):
        super().__init__()
        self.max_requests = max_requests
        self.request_counts = {}
    
    def _process(self, request):
        user = request.get('user')
        self.request_counts[user] = self.request_counts.get(user, 0) + 1
        
        if self.request_counts[user] > self.max_requests:
            print(f"❌ Rate limit exceeded for {user}")
            return True
        print(f"✓ Rate limit OK ({self.request_counts[user]}/{self.max_requests})")
        return False

class DataHandler(Handler):
    """Process the actual request data"""
    
    def _process(self, request):
        print(f"✓ Processing data: {request.get('data')}")
        return True

# Setup chain
auth = AuthHandler()
perm = PermissionHandler()
rate = RateLimitHandler(max_requests=5)
data = DataHandler()

auth.set_next(rate).set_next(perm).set_next(data)

# Test requests
request1 = {
    'user': 'alice',
    'authenticated': True,
    'permissions': ['read', 'write'],
    'required_permission': 'read',
    'data': 'Important document'
}

request2 = {
    'user': 'bob',
    'authenticated': False,
    'data': 'Should not reach here'
}

auth.handle(request1)  # Full chain
auth.handle(request2)  # Blocked at auth
```

**Why**: Chain of Responsibility decouples senders from receivers by allowing multiple objects to handle a request. Used in middleware pipelines (Express.js), event bubbling in GUIs, logging frameworks, and authentication/authorization flows. Flexible: chain can be modified at runtime without changing client code.

## [Publish-Subscribe Pattern - Decoupled Event Distribution]

```python
# PROBLEM: Broadcast events to multiple subscribers without coupling
# TIME: O(k) for k subscribers SPACE: O(n*k) for n topics with k subscribers each
from collections import defaultdict
from typing import Callable, Any

class PubSub:
    """Publish-Subscribe system for event-driven architecture"""
    
    def __init__(self):
        self.topics = defaultdict(list)  # topic -> [subscribers]
        self.subscriber_ids = {}  # id -> (topic, callback)
        self.next_id = 0
    
    def subscribe(self, topic: str, callback: Callable[[Any], None]) -> int:
        """
        Subscribe to topic with callback function
        Returns subscription ID for unsubscribing
        """
        sub_id = self.next_id
        self.next_id += 1
        
        self.topics[topic].append(callback)
        self.subscriber_ids[sub_id] = (topic, callback)
        
        return sub_id
    
    def unsubscribe(self, sub_id: int):
        """Remove subscriber by ID"""
        if sub_id not in self.subscriber_ids:
            return False
        
        topic, callback = self.subscriber_ids[sub_id]
        self.topics[topic].remove(callback)
        del self.subscriber_ids[sub_id]
        
        return True
    
    def publish(self, topic: str, message: Any):
        """Publish message to all subscribers of topic"""
        if topic not in self.topics:
            return 0
        
        # Notify all subscribers (async in real implementations)
        for callback in self.topics[topic]:
            try:
                callback(message)
            except Exception as e:
                print(f"Error in subscriber: {e}")
        
        return len(self.topics[topic])

# Example: Stock price notification system
class StockTrader:
    def __init__(self, name, pubsub):
        self.name = name
        self.pubsub = pubsub
    
    def watch_stock(self, symbol):
        """Subscribe to stock price updates"""
        def on_price_update(price):
            print(f"  [{self.name}] {symbol} is now ${price:.2f}")
        
        return self.pubsub.subscribe(f"stock.{symbol}", on_price_update)
    
    def on_market_event(self, event):
        """Handle market-wide events"""
        print(f"  [{self.name}] Market alert: {event}")

# Usage
pubsub = PubSub()

# Create traders
alice = StockTrader("Alice", pubsub)
bob = StockTrader("Bob", pubsub)

# Subscribe to stocks
alice.watch_stock("AAPL")
alice.watch_stock("GOOGL")
bob.watch_stock("AAPL")
bob.watch_stock("MSFT")

# Subscribe to market events
pubsub.subscribe("market.alert", alice.on_market_event)

# Publish updates
print("📈 AAPL price update:")
pubsub.publish("stock.AAPL", 150.25)

print("\n📈 GOOGL price update:")
pubsub.publish("stock.GOOGL", 2800.50)

print("\n📢 Market alert:")
pubsub.publish("market.alert", "High volatility detected")
```

**Why**: Pub-Sub enables loosely coupled, event-driven systems. Publishers don't know subscribers exist. Used in message queues (RabbitMQ, Kafka), UI event systems, microservices communication, and real-time applications (chat, live updates). Scales well: add/remove subscribers without touching publishers.

## [API Gateway Pattern - Unified Entry Point]

```python
# PROBLEM: Provide single entry point for multiple microservices
# TIME: O(1) routing + backend latency SPACE: O(n) for n routes
from functools import wraps
import time
from collections import defaultdict

class APIGateway:
    """API Gateway with routing, rate limiting, and authentication"""
    
    def __init__(self):
        self.routes = {}  # path -> (service_url, method)
        self.middleware = []  # Pre-processing functions
        self.rate_limiter = RateLimiter(requests_per_minute=60)
        self.auth_service = None
    
    def register_route(self, path, service_url, method='GET'):
        """Register microservice endpoint"""
        self.routes[path] = (service_url, method)
    
    def add_middleware(self, func):
        """Add middleware function (auth, logging, etc.)"""
        self.middleware.append(func)
    
    def request(self, path, method='GET', headers=None, body=None):
        """
        Process incoming request through gateway
        """
        # Check if route exists
        if path not in self.routes:
            return {'status': 404, 'error': 'Not Found'}
        
        service_url, expected_method = self.routes[path]
        
        # Validate method
        if method != expected_method:
            return {'status': 405, 'error': 'Method Not Allowed'}
        
        # Run middleware chain
        context = {
            'path': path,
            'method': method,
            'headers': headers or {},
            'body': body,
            'client_id': headers.get('X-Client-ID') if headers else None
        }
        
        for middleware in self.middleware:
            result = middleware(context)
            if result:  # Middleware can block request
                return result
        
        # Route to microservice (simulated)
        return self._forward_to_service(service_url, context)
    
    def _forward_to_service(self, service_url, context):
        """Simulate forwarding request to microservice"""
        return {
            'status': 200,
            'service': service_url,
            'data': f"Response from {service_url}"
        }

class RateLimiter:
    """Token bucket rate limiter"""
    
    def __init__(self, requests_per_minute):
        self.rpm = requests_per_minute
        self.requests = defaultdict(list)
    
    def check(self, client_id):
        """Check if client is within rate limit"""
        now = time.time()
        window_start = now - 60  # 1 minute window
        
        # Remove old requests
        self.requests[client_id] = [
            t for t in self.requests[client_id] if t > window_start
        ]
        
        # Check limit
        if len(self.requests[client_id]) >= self.rpm:
            return False
        
        self.requests[client_id].append(now)
        return True

# Middleware functions
def auth_middleware(context):
    """Check authentication token"""
    token = context['headers'].get('Authorization')
    if not token or not token.startswith('Bearer '):
        return {'status': 401, 'error': 'Unauthorized'}
    context['user'] = token.split()[1]  # Extract user from token
    return None

def rate_limit_middleware(rate_limiter):
    """Factory: create rate limiting middleware"""
    def middleware(context):
        client_id = context.get('client_id') or context.get('user', 'anonymous')
        if not rate_limiter.check(client_id):
            return {'status': 429, 'error': 'Rate Limit Exceeded'}
        return None
    return middleware

def logging_middleware(context):
    """Log all requests"""
    print(f"[{time.time():.2f}] {context['method']} {context['path']}")
    return None

# Setup gateway
gateway = APIGateway()
limiter = RateLimiter(requests_per_minute=100)

# Register microservices
gateway.register_route('/api/users', 'http://user-service:8001', 'GET')
gateway.register_route('/api/orders', 'http://order-service:8002', 'POST')
gateway.register_route('/api/products', 'http://product-service:8003', 'GET')

# Add middleware (order matters!)
gateway.add_middleware(logging_middleware)
gateway.add_middleware(auth_middleware)
gateway.add_middleware(rate_limit_middleware(limiter))

# Test requests
headers = {
    'Authorization': 'Bearer user123',
    'X-Client-ID': 'client_abc'
}

print("✓ Valid request:")
response = gateway.request('/api/users', 'GET', headers)
print(response)

print("\n✗ No auth:")
response = gateway.request('/api/orders', 'POST', {})
print(response)

print("\n✗ Invalid route:")
response = gateway.request('/api/unknown', 'GET', headers)
print(response)
```

**Why**: API Gateway centralizes cross-cutting concerns: authentication, rate limiting, logging, load balancing, and request routing. Essential for microservices architecture. Provides single entry point for clients, simplifying frontend integration. Used in AWS API Gateway, Kong, Nginx. Enables protocol translation (REST → gRPC), caching, and SSL termination.

# PROBLEM: Implement efficient database index for range queries
# TIME: O(log n) search/insert/delete SPACE: O(n)
class BTreeNode:
    """B-Tree node with keys and children"""
    
    def __init__(self, leaf=False):
        self.keys = []  # List of keys (row IDs or indexed values)
        self.children = []  # Child pointers
        self.leaf = leaf  # True if leaf node
        self.values = []  # For leaf nodes: store actual values/pointers

class BTree:
    """
    B-Tree for database indexing
    Order m: each node has at most m children and m-1 keys
    """
    
    def __init__(self, order=3):
        self.root = BTreeNode(leaf=True)
        self.order = order  # Minimum degree
    
    def search(self, key, node=None):
        """Search for key in B-Tree"""
        if node is None:
            node = self.root
        
        # Find first key >= search key
        i = 0
        while i < len(node.keys) and key > node.keys[i]:
            i += 1
        
        # Key found
        if i < len(node.keys) and key == node.keys[i]:
            return node.values[i] if node.leaf else node
        
        # Key not found and leaf
        if node.leaf:
            return None
        
        # Recurse to child
        return self.search(key, node.children[i])
    
    def insert(self, key, value):
        """Insert key-value pair"""
        root = self.root
        
        # If root is full, split
        if len(root.keys) == (2 * self.order - 1):
            new_root = BTreeNode()
            new_root.children.append(self.root)
            self._split_child(new_root, 0)
            self.root = new_root
        
        self._insert_non_full(self.root, key, value)
    
    def _split_child(self, parent, index):
        """Split full child node"""
        order = self.order
        full_child = parent.children[index]
        new_child = BTreeNode(leaf=full_child.leaf)
        
        # Move middle key up to parent
        mid = order - 1
        parent.keys.insert(index, full_child.keys[mid])
        parent.children.insert(index + 1, new_child)
        
        # Split keys and children
        new_child.keys = full_child.keys[mid + 1:]
        full_child.keys = full_child.keys[:mid]
        
        if full_child.leaf:
            new_child.values = full_child.values[mid + 1:]
            full_child.values = full_child.values[:mid]
        else:
            new_child.children = full_child.children[mid + 1:]
            full_child.children = full_child.children[:mid + 1]
    
    def _insert_non_full(self, node, key, value):
        """Insert into non-full node"""
        if node.leaf:
            # Find position and insert
            i = len(node.keys) - 1
            node.keys.append(None)
            node.values.append(None)
            
            while i >= 0 and key < node.keys[i]:
                node.keys[i + 1] = node.keys[i]
                node.values[i + 1] = node.values[i]
                i -= 1
            
            node.keys[i + 1] = key
            node.values[i + 1] = value
        else:
            # Find child to insert into
            i = len(node.keys) - 1
            while i >= 0 and key < node.keys[i]:
                i -= 1
            i += 1
            
            # Split child if full
            if len(node.children[i].keys) == (2 * self.order - 1):
                self._split_child(node, i)
                if key > node.keys[i]:
                    i += 1
            
            self._insert_non_full(node.children[i], key, value)
    
    def range_query(self, start, end):
        """Find all keys in range [start, end]"""
        results = []
        self._range_query_helper(self.root, start, end, results)
        return results
    
    def _range_query_helper(self, node, start, end, results):
        """Recursive range query"""
        i = 0
        while i < len(node.keys) and node.keys[i] < start:
            i += 1
        
        if node.leaf:
            # Collect all keys in range
            while i < len(node.keys) and node.keys[i] <= end:
                results.append((node.keys[i], node.values[i]))
                i += 1
        else:
            # Recurse to children
            while i < len(node.keys) and node.keys[i] <= end:
                self._range_query_helper(node.children[i], start, end, results)
                if node.keys[i] >= start:
                    results.append((node.keys[i], node.keys[i]))  # Or stored value
                i += 1
            if i < len(node.children):
                self._range_query_helper(node.children[i], start, end, results)
