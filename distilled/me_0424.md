## Manacher's Algorithm

```python
# PROBLEM: Find the longest palindromic substring in O(n) time
# TIME: O(n) SPACE: O(n)
def manacher_algorithm(s):
    """
    Find the longest palindromic substring using Manacher's algorithm.
    
    The algorithm transforms the string by inserting special characters
    to handle both odd and even length palindromes uniformly.
    """
    if not s:
        return ""
    
    # Transform string: "abc" -> "^#a#b#c#$"
    # ^ and $ are sentinels to avoid boundary checks
    transformed = '^#' + '#'.join(s) + '#$'
    n = len(transformed)
    
    # Array to store palindrome radii
    p = [0] * n
    center = right = 0
    
    for i in range(1, n - 1):
        # Mirror index on the other side of center
        mirror = 2 * center - i
        
        # If i is within right boundary, use mirror value
        if i < right:
            p[i] = min(right - i, p[mirror])
        
        # Attempt to expand palindrome centered at i
        while transformed[i + p[i] + 1] == transformed[i - p[i] - 1]:
            p[i] += 1
        
        # Update center and right boundary if palindrome expands past right
        if i + p[i] > right:
            center = i
            right = i + p[i]
    
    # Find the maximum length palindrome
    max_len = max(p)
    center_idx = p.index(max_len)
    
    # Extract the original palindrome
    start = (center_idx - max_len) // 2
    return s[start:start + max_len]
```

**Why**: Linear time palindrome detection using center expansion with smart boundary tracking and mirroring properties.

## Graham Scan

```python
# PROBLEM: Compute convex hull of a set of 2D points
# TIME: O(n log n) SPACE: O(n)
def graham_scan(points):
    """
    Compute the convex hull using Graham scan algorithm.
    
    Steps:
    1. Find the point with lowest y-coordinate (and leftmost if tie)
    2. Sort points by polar angle with respect to the start point
    3. Process points maintaining convexity using cross product
    """
    if len(points) < 3:
        return points
    
    # Find the point with lowest y (and leftmost if tie)
    start = min(points, key=lambda p: (p[1], p[0]))
    
    def polar_angle(p):
        """Calculate polar angle from start point."""
        import math
        return math.atan2(p[1] - start[1], p[0] - start[0])
    
    def distance(p):
        """Calculate squared distance from start point."""
        return (p[0] - start[0])**2 + (p[1] - start[1])**2
    
    # Sort by polar angle, then by distance for collinear points
    sorted_points = sorted(points, key=lambda p: (polar_angle(p), distance(p)))
    
    def cross_product(o, a, b):
        """
        Calculate cross product of OA and OB vectors.
        Returns positive for counter-clockwise, negative for clockwise.
        """
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
    
    # Build convex hull
    hull = []
    for point in sorted_points:
        # Remove last point while it makes a non-left turn
        while len(hull) > 1 and cross_product(hull[-2], hull[-1], point) <= 0:
            hull.pop()
        hull.append(point)
    
    return hull
```

**Why**: Efficient convex hull computation using angular sorting and stack-based incremental construction with orientation tests.

## Jarvis March (Gift Wrapping)

```python
# PROBLEM: Compute convex hull using gift wrapping algorithm
# TIME: O(nh) where h is hull size SPACE: O(h)
def jarvis_march(points):
    """
    Compute convex hull using Jarvis march (gift wrapping) algorithm.
    
    Works by starting from leftmost point and wrapping around
    by always selecting the most counter-clockwise point.
    """
    if len(points) < 3:
        return points
    
    def cross_product(o, a, b):
        """Cross product of OA and OB vectors."""
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
    
    # Find leftmost point to start
    start = min(points, key=lambda p: p[0])
    hull = [start]
    
    while True:
        # Select any point as candidate (not the last hull point)
        candidate = points[0] if points[0] != hull[-1] else points[1]
        
        # Find most counter-clockwise point
        for point in points:
            if point == hull[-1]:
                continue
            cross = cross_product(hull[-1], candidate, point)
            # Update candidate if point is more counter-clockwise
            if cross < 0 or (cross == 0 and 
                           abs(point[0] - hull[-1][0]) + abs(point[1] - hull[-1][1]) >
                           abs(candidate[0] - hull[-1][0]) + abs(candidate[1] - hull[-1][1])):
                candidate = point
        
        # If we've returned to start, we're done
        if candidate == start:
            break
        
        hull.append(candidate)
    
    return hull
```

**Why**: Output-sensitive algorithm optimal for few hull points; wraps around convex hull like gift wrapping paper.

## Quickhull Algorithm

```python
# PROBLEM: Compute convex hull using divide and conquer approach
# TIME: O(n log n) average, O(n²) worst SPACE: O(n)
def quickhull(points):
    """
    Compute convex hull using Quickhull algorithm.
    
    Recursively finds the farthest point from a line segment
    and partitions points on either side.
    """
    if len(points) < 3:
        return points
    
    def distance_from_line(p1, p2, p):
        """Calculate perpendicular distance of point p from line p1-p2."""
        return abs((p2[1] - p1[1]) * p[0] - (p2[0] - p1[0]) * p[1] + 
                   p2[0] * p1[1] - p2[1] * p1[0])
    
    def side_of_line(p1, p2, p):
        """Determine which side of line p1-p2 the point p is on."""
        return (p2[1] - p1[1]) * p[0] - (p2[0] - p1[0]) * p[1] + \
               p2[0] * p1[1] - p2[1] * p1[0]
    
    def quickhull_recursive(points, p1, p2, side):
        """Recursively find hull points on one side of line p1-p2."""
        max_dist = 0
        farthest = None
        
        # Find farthest point from line on given side
        for p in points:
            dist = distance_from_line(p1, p2, p)
            if side_of_line(p1, p2, p) == side and dist > max_dist:
                max_dist = dist
                farthest = p
        
        # Base case: no point found on this side
        if farthest is None:
            return []
        
        # Recursively find points in the two new regions
        result = [farthest]
        result.extend(quickhull_recursive(points, p1, farthest, 
                                         -side_of_line(p1, farthest, p2)))
        result.extend(quickhull_recursive(points, farthest, p2,
                                         -side_of_line(farthest, p2, p1)))
        
        return result
    
    # Find leftmost and rightmost points
    min_x = min(points, key=lambda p: p[0])
    max_x = max(points, key=lambda p: p[0])
    
    # Find hull points on both sides of the line
    hull = [min_x, max_x]
    hull.extend(quickhull_recursive(points, min_x, max_x, 1))
    hull.extend(quickhull_recursive(points, min_x, max_x, -1))
    
    # Remove duplicates and sort (for consistent output)
    hull = list(set(hull))
    
    return hull
```

**Why**: Divide-and-conquer approach similar to quicksort; efficient in practice with good average-case performance.

## Chan's Algorithm

```python
# PROBLEM: Compute convex hull in O(n log h) worst-case time
# TIME: O(n log h) where h is hull size SPACE: O(n)
def chans_algorithm(points):
    """
    Compute convex hull using Chan's algorithm.
    
    Combines Graham scan and Jarvis march to achieve
    O(n log h) worst-case time complexity.
    """
    import math
    
    def graham_scan_limited(pts, k):
        """Graham scan limited to k points on hull."""
        if len(pts) < 3:
            return pts
        
        start = min(pts, key=lambda p: (p[1], p[0]))
        
        def polar_angle(p):
            return math.atan2(p[1] - start[1], p[0] - start[0])
        
        sorted_pts = sorted(pts, key=polar_angle)
        
        def cross_product(o, a, b):
            return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
        
        hull = []
        for p in sorted_pts:
            while len(hull) > 1 and cross_product(hull[-2], hull[-1], p) <= 0:
                hull.pop()
            hull.append(p)
            if len(hull) > k:
                return None  # Exceeded limit
        
        return hull
    
    def jarvis_step(hulls, current):
        """Find next convex hull point using Jarvis-style wrapping."""
        def cross_product(o, a, b):
            return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
        
        candidate = None
        for hull in hulls:
            if not hull:
                continue
            # Find extreme point in this hull
            extreme = max(hull, key=lambda p: cross_product(current, 
                           (current[0] + 1, current[1]), p))
            if candidate is None or cross_product(current, candidate, extreme) < 0:
                candidate = extreme
        
        return candidate
    
    if len(points) < 3:
        return points
    
    n = len(points)
    
    # Try increasing values of h until we find the correct hull size
    for t in range(1, n + 1):
        h = 2 ** (2 ** t)  # Exponential search
        
        # Partition into groups and compute mini-hulls
        m = max(1, n // h)
        groups = [points[i:i + m] for i in range(0, n, m)]
        
        hulls = []
        for group in groups:
            hull = graham_scan_limited(group, h)
            if hull is None:
                break
            hulls.append(hull)
        else:
            # Jarvis march over the mini-hulls
            start = min(points, key=lambda p: (p[1], p[0]))
            result = [start]
            
            for _ in range(h):
                next_pt = jarvis_step(hulls, result[-1])
                if next_pt == start or next_pt is None:
                    return result
                result.append(next_pt)
            
            # If we didn't complete the hull, h was too small
            continue
        
        # If we get here, one of the mini-hulls was too big, try larger h
        # Actually, we should continue to next iteration
        pass
    
    return points  # Fallback
```

**Why**: Optimal O(n log h) worst-case algorithm combining Graham scan and Jarvis march with exponential search.

## AA Tree

```python
# PROBLEM: Implement self-balancing BST with simplified balancing rules
# TIME: O(log n) for all operations SPACE: O(n)
class AATreeNode:
    def __init__(self, key, value=None):
        self.key = key
        self.value = value
        self.left = None
        self.right = None
        self.level = 1  # Level instead of color (simpler than RB tree)

class AATree:
    """
    AA tree - a simplified form of red-black tree using levels.
    
    Invariants:
    - Level of left child is exactly one less than parent
    - Level of right child is equal or one less than parent
    - Level of right grandchild is less than grandparent
    - Level > 1 nodes must have two children
    """
    
    def __init__(self):
        self.root = None
    
    def skew(self, node):
        """
        Right rotation to fix left horizontal link.
        Level of node equals level of left child.
        """
        if node is None or node.left is None:
            return node
        
        if node.level == node.left.level:
            # Rotate right
            new_root = node.left
            node.left = new_root.right
            new_root.right = node
            return new_root
        
        return node
    
    def split(self, node):
        """
        Left rotation to fix consecutive right horizontal links.
        Level of node equals level of right child's right child.
        """
        if node is None or node.right is None or node.right.right is None:
            return node
        
        if node.level == node.right.right.level:
            # Rotate left and increase level
            new_root = node.right
            node.right = new_root.left
            new_root.left = node
            new_root.level += 1
            return new_root
        
        return node
    
    def insert(self, key, value=None):
        """Insert key-value pair into tree."""
        self.root = self._insert(self.root, key, value)
    
    def _insert(self, node, key, value):
        """Recursive insert with rebalancing."""
        if node is None:
            return AATreeNode(key, value)
        
        if key < node.key:
            node.left = self._insert(node.left, key, value)
        elif key > node.key:
            node.right = self._insert(node.right, key, value)
        else:
            node.value = value  # Update existing
            return node
        
        # Rebalance
        node = self.skew(node)
        node = self.split(node)
        
        return node
    
    def delete(self, key):
        """Delete key from tree."""
        self.root = self._delete(self.root, key)
    
    def _delete(self, node, key):
        """Recursive delete with rebalancing."""
        if node is None:
            return None
        
        if key < node.key:
            node.left = self._delete(node.left, key)
        elif key > node.key:
            node.right = self._delete(node.right, key)
        else:
            # Found node to delete
            if node.left is None:
                return node.right
            elif node.right is None:
                return node.left
            else:
                # Two children: find successor
                successor = self._find_min(node.right)
                node.key = successor.key
                node.value = successor.value
                node.right = self._delete(node.right, successor.key)
        
        # Rebalance after deletion
        if node is not None:
            # Decrease level if needed
            min_level = min(
                node.left.level if node.left else 0,
                node.right.level if node.right else 0
            ) + 1
            
            if node.level > min_level:
                node.level = min_level
                if node.right and node.right.level > node.level:
                    node.right.level = node.level
            
            node = self.skew(node)
            node = self.split(node)
        
        return node
    
    def _find_min(self, node):
        """Find minimum key in subtree."""
        while node.left:
            node = node.left
        return node
    
    def search(self, key):
        """Search for key and return associated value."""
        node = self.root
        while node:
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                return node.value
        return None
```

**Why**: Simplified self-balancing tree using levels instead of colors; easier to implement than red-black trees with same performance.

## AVL Tree

```python
# PROBLEM: Implement strictly balanced BST with height difference ≤ 1
# TIME: O(log n) for all operations SPACE: O(n)
class AVLNode:
    def __init__(self, key, value=None):
        self.key = key
        self.value = value
        self.left = None
        self.right = None
        self.height = 1

class AVLTree:
    """
    AVL tree - self-balancing BST with strict balance condition.
    
    Balance factor = height(left) - height(right)
    Must be -1, 0, or 1 for all nodes.
    """
    
    def __init__(self):
        self.root = None
    
    def get_height(self, node):
        """Get height of node (0 for None)."""
        return node.height if node else 0
    
    def get_balance(self, node):
        """Get balance factor of node."""
        if not node:
            return 0
        return self.get_height(node.left) - self.get_height(node.right)
    
    def update_height(self, node):
        """Update height based on children."""
        if node:
            node.height = 1 + max(self.get_height(node.left),
                                  self.get_height(node.right))
    
    def rotate_right(self, y):
        """Right rotation around node y."""
        x = y.left
        T2 = x.right
        
        # Perform rotation
        x.right = y
        y.left = T2
        
        # Update heights
        self.update_height(y)
        self.update_height(x)
        
        return x
    
    def rotate_left(self, x):
        """Left rotation around node x."""
        y = x.right
        T2 = y.left
        
        # Perform rotation
        y.left = x
        x.right = T2
        
        # Update heights
        self.update_height(x)
        self.update_height(y)
        
        return y
    
    def insert(self, key, value=None):
        """Insert key-value pair into tree."""
        self.root = self._insert(self.root, key, value)
    
    def _insert(self, node, key, value):
        """Recursive insert with rebalancing."""
        if not node:
            return AVLNode(key, value)
        
        if key < node.key:
            node.left = self._insert(node.left, key, value)
        elif key > node.key:
            node.right = self._insert(node.right, key, value)
        else:
            node.value = value  # Update existing
            return node
        
        # Update height
        self.update_height(node)
        
        # Get balance factor
        balance = self.get_balance(node)
        
        # Left Left Case
        if balance > 1 and key < node.left.key:
            return self.rotate_right(node)
        
        # Right Right Case
        if balance < -1 and key > node.right.key:
            return self.rotate_left(node)
        
        # Left Right Case
        if balance > 1 and key > node.left.key:
            node.left = self.rotate_left(node.left)
            return self.rotate_right(node)
        
        # Right Left Case
        if balance < -1 and key < node.right.key:
            node.right = self.rotate_right(node.right)
            return self.rotate_left(node)
        
        return node
    
    def delete(self, key):
        """Delete key from tree."""
        self.root = self._delete(self.root, key)
    
    def _delete(self, node, key):
        """Recursive delete with rebalancing."""
        if not node:
            return None
        
        if key < node.key:
            node.left = self._delete(node.left, key)
        elif key > node.key:
            node.right = self._delete(node.right, key)
        else:
            # Node to delete found
            if not node.left:
                return node.right
            elif not node.right:
                return node.left
            else:
                # Node with two children
                temp = self._find_min(node.right)
                node.key = temp.key
                node.value = temp.value
                node.right = self._delete(node.right, temp.key)
        
        # Update height
        self.update_height(node)
        
        # Get balance factor
        balance = self.get_balance(node)
        
        # Left Left Case
        if balance > 1 and self.get_balance(node.left) >= 0:
            return self.rotate_right(node)
        
        # Left Right Case
        if balance > 1 and self.get_balance(node.left) < 0:
            node.left = self.rotate_left(node.left)
            return self.rotate_right(node)
        
        # Right Right Case
        if balance < -1 and self.get_balance(node.right) <= 0:
            return self.rotate_left(node)
        
        # Right Left Case
        if balance < -1 and self.get_balance(node.right) > 0:
            node.right = self.rotate_right(node.right)
            return self.rotate_left(node)
        
        return node
    
    def _find_min(self, node):
        """Find minimum key node in subtree."""
        while node.left:
            node = node.left
        return node
    
    def search(self, key):
        """Search for key and return value."""
        node = self.root
        while node:
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                return node.value
        return None
```

**Why**: Strictly balanced BST guaranteeing O(log n) operations; useful when frequent lookups require guaranteed performance.

## Scapegoat Tree

```python
# PROBLEM: Implement self-balancing BST without extra space per node
# TIME: O(log n) amortized SPACE: O(n)
import math

class ScapegoadNode:
    def __init__(self, key, value=None):
        self.key = key
        self.value = value
        self.left = None
        self.right = None

class ScapegoatTree:
    """
    Scapegoat tree - self-balancing BST using occasional rebuilding.
    
    No balance information stored in nodes.
    Rebuilds subtree when it becomes too unbalanced.
    """
    
    def __init__(self, alpha=0.7):
        """
        Initialize tree with balance parameter alpha (0.5 < alpha < 1.0).
        Lower alpha = more balanced = more rebuilding.
        """
        self.root = None
        self.alpha = alpha
        self.size = 0
        self.max_size = 0  # Track max size since last rebuild
    
    def size_of(self, node):
        """Get size of subtree rooted at node."""
        if not node:
            return 0
        return 1 + self.size_of(node.left) + self.size_of(node.right)
    
    def flatten(self, node, arr):
        """Flatten subtree into sorted array."""
        if not node:
            return
        self.flatten(node.left, arr)
        arr.append(node)
        self.flatten(node.right, arr)
    
    def build_balanced(self, arr, start, end):
        """Build balanced tree from sorted array."""
        if start > end:
            return None
        
        mid = (start + end) // 2
        node = arr[mid]
        node.left = self.build_balanced(arr, start, mid - 1)
        node.right = self.build_balanced(arr, mid + 1, end)
        
        return node
    
    def rebuild(self, node):
        """Rebuild subtree to make it balanced."""
        arr = []
        self.flatten(node, arr)
        return self.build_balanced(arr, 0, len(arr) - 1)
    
    def insert(self, key, value=None):
        """Insert key-value pair."""
        # Track path to insertion point
        path = []
        node = self.root
        parent = None
        
        while node:
            path.append(node)
            parent = node
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                node.value = value  # Update existing
                return
        
        # Create new node
        new_node = ScapegoadNode(key, value)
        
        if not self.root:
            self.root = new_node
        elif key < parent.key:
            parent.left = new_node
        else:
            parent.right = new_node
        
        self.size += 1
        self.max_size = max(self.max_size, self.size)
        
        # Check if tree needs rebuilding
        if self.size > self.alpha * self.max_size:
            # Find scapegoat node
            for i, node in enumerate(path):
                size = self.size_of(node)
                left_size = self.size_of(node.left)
                right_size = self.size_of(node.right)
                
                # Check if this node is unbalanced
                if left_size > self.alpha * size or right_size > self.alpha * size:
                    # Rebuild this subtree
                    rebuilt = self.rebuild(node)
                    
                    if i == 0:
                        self.root = rebuilt
                    else:
                        parent = path[i - 1]
                        if node == parent.left:
                            parent.left = rebuilt
                        else:
                            parent.right = rebuilt
                    
                    self.max_size = self.size
                    break
    
    def delete(self, key):
        """Delete key from tree."""
        self.root = self._delete(self.root, key)
        
        # Rebuild if tree is too sparse
        if self.size < self.alpha * self.max_size:
            self.root = self.rebuild(self.root)
            self.max_size = self.size
    
    def _delete(self, node, key):
        """Recursive delete."""
        if not node:
            return None
        
        if key < node.key:
            node.left = self._delete(node.left, key)
        elif key > node.key:
            node.right = self._delete(node.right, key)
        else:
            # Found node to delete
            if not node.left:
                self.size -= 1
                return node.right
            elif not node.right:
                self.size -= 1
                return node.left
            else:
                # Two children
                successor = self._find_min(node.right)
                node.key = successor.key
                node.value = successor.value
                node.right = self._delete(node.right, successor.key)
        
        return node
    
    def _find_min(self, node):
        """Find minimum key node in subtree."""
        while node.left:
            node = node.left
        return node
    
    def search(self, key):
        """Search for key and return value."""
        node = self.root
        while node:
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                return node.value
        return None
```

**Why**: Memory-efficient self-balancing tree; no per-node overhead and amortized O(log n) operations with occasional rebuilding.

# TIME: O(log n) for all operations SPACE: O(n)
class Node234:
    def __init__(self):
        self.keys = []       # 1-3 keys per node

class Tree234:
    """
    2-3-4 tree - B-tree of order 4.
    
    Each node has:
    - 1-3 keys
    - 2-4 children (if not leaf)
    - All leaves at same depth
    """
    
    def __init__(self):
        self.root = Node234()
    
    def search(self, key):
        """Search for key in tree."""
        return self._search(self.root, key)
    
    def _search(self, node, key):
        """Recursive search."""
        # Find position in current node
        i = 0
        while i < len(node.keys) and key > node.keys[i]:
            i += 1
        
        # Check if found
        if i < len(node.keys) and key == node.keys[i]:
            return True
        
        # If leaf and not found, key doesn't exist
        if node.is_leaf:
            return False
        
        # Search in appropriate child
        return self._search(node.children[i], key)
    
    def insert(self, key):
        """Insert key into tree."""
        root = self.root
        
        # If root is full, split it first
        if len(root.keys) == 3:
            new_root = Node234()
            new_root.is_leaf = False
            new_root.children.append(self.root)
            self.root = new_root
            self._split_child(new_root, 0)
        
        self._insert_non_full(self.root, key)
    
    def _split_child(self, parent, index):
        """Split full child at index."""
        full_child = parent.children[index]
        new_child = Node234()
        
        # Move middle key up to parent
        parent.keys.insert(index, full_child.keys[1])
        
        # New child gets rightmost key
        new_child.keys = [full_child.keys[2]]
        full_child.keys = [full_child.keys[0]]
        
        # Split children if not leaf
        if not full_child.is_leaf:
            new_child.is_leaf = False
            new_child.children = full_child.children[2:]
            full_child.children = full_child.children[:2]
        
        # Insert new child
        parent.children.insert(index + 1, new_child)
    
    def _insert_non_full(self, node, key):
        """Insert into non-full node."""
        if node.is_leaf:
            # Insert key in sorted position
            i = len(node.keys) - 1
            node.keys.append(0)
            while i >= 0 and key < node.keys[i]:
                node.keys[i + 1] = node.keys[i]
                i -= 1
            node.keys[i + 1] = key
        else:
            # Find child to descend into
            i = len(node.keys) - 1
            while i >= 0 and key < node.keys[i]:
                i -= 1
            i += 1
            
            # Split child if full
            if len(node.children[i].keys) == 3:
                self._split_child(node, i)
                if key > node.keys[i]:
                    i += 1
            
            self._insert_non_full(node.children[i], key)
    
    def traverse(self):
        """In-order traversal of tree."""
        result = []
        self._traverse(self.root, result)
        return result
    
    def _traverse(self, node, result):
        """Recursive in-order traversal."""
        if node.is_leaf:
            result.extend(node.keys)
        else:
            for i in range(len(node.keys)):
                self._traverse(node.children[i], result)
                result.append(node.keys[i])
            self._traverse(node.children[len(node.keys)], result)
```

**Why**: Simple B-tree variant with guaranteed balance; used as basis for red-black trees in practice.

## B* Tree

# PROBLEM: Implement B-tree variant with better space utilization
# TIME: O(log n) for all operations SPACE: O(n)
class BStarNode:
    def __init__(self, order):
        self.order = order
        self.keys = []
        self.children = []
        self.is_leaf = True
    
    def is_full(self):
        return len(self.keys) >= 2 * self.order - 1
    
    def is_minimal(self):
        return len(self.keys) < self.order - 1

class BStarTree:
    """
    B* tree - B-tree variant requiring nodes to be at least 2/3 full.
    
    Better space utilization than B-tree but more complex operations.
    Split distributes keys to sibling when possible.
    """
    
    def __init__(self, order=3):
        """
        Initialize B* tree with minimum degree.
        order: minimum degree (min children = order)
        """
        self.order = order
        self.root = BStarNode(order)
    
    def search(self, key):
        """Search for key in tree."""
        return self._search(self.root, key)
    
    def _search(self, node, key):
        """Recursive search."""
        i = 0
        while i < len(node.keys) and key > node.keys[i]:
            i += 1
        
        if i < len(node.keys) and key == node.keys[i]:
            return True
        
        if node.is_leaf:
            return False
        
        return self._search(node.children[i], key)
    
    def insert(self, key):
        """Insert key into tree."""
        root = self.root
        
        if root.is_full():
            # Split root
            new_root = BStarNode(self.order)
            new_root.is_leaf = False
            new_root.children.append(self.root)
            self.root = new_root
            self._split_child(new_root, 0)
        
        self._insert_non_full(self.root, key)
    
    def _split_child(self, parent, index):
        """Split child, trying to redistribute with sibling first."""
        child = parent.children[index]
        
        # Try to redistribute with left sibling
        if index > 0:
            left_sibling = parent.children[index - 1]
            if not left_sibling.is_full():
                # Redistribute
                left_sibling.keys.append(parent.keys[index - 1])
                parent.keys[index - 1] = child.keys[0]
                child.keys.pop(0)
                
                if not child.is_leaf:
                    left_sibling.children.append(child.children[0])
                    child.children.pop(0)
                return
        
        # Try to redistribute with right sibling
        if index < len(parent.children) - 1:
            right_sibling = parent.children[index + 1]
            if not right_sibling.is_full():
                # Redistribute
                right_sibling.keys.insert(0, parent.keys[index])
                parent.keys[index] = child.keys[-1]
                child.keys.pop()
                
                if not child.is_leaf:
                    right_sibling.children.insert(0, child.children[-1])
                    child.children.pop()
                return
        
        # Must split: create new node and distribute keys
        new_node = BStarNode(self.order)
        
        # Split at 2/3 point for better balance
        split_point = len(child.keys) * 2 // 3
        
        # Move keys to new node
        new_node.keys = child.keys[split_point + 1:]
        middle_key = child.keys[split_point]
        child.keys = child.keys[:split_point]
        
        # Move children if not leaf
        if not child.is_leaf:
            new_node.is_leaf = False
            new_node.children = child.children[split_point + 1:]
            child.children = child.children[:split_point + 1]
        
        # Insert middle key into parent
        parent.keys.insert(index, middle_key)
        parent.children.insert(index + 1, new_node)
    
    def _insert_non_full(self, node, key):
        """Insert into non-full node."""
        if node.is_leaf:
            # Insert in sorted position
            i = len(node.keys) - 1
            node.keys.append(0)
            while i >= 0 and key < node.keys[i]:
                node.keys[i + 1] = node.keys[i]
                i -= 1
            node.keys[i + 1] = key
        else:
            # Find child
            i = len(node.keys) - 1
            while i >= 0 and key < node.keys[i]:
                i -= 1
            i += 1
            
            # Split child if full
            if node.children[i].is_full():
                self._split_child(node, i)
                if key > node.keys[i]:
                    i += 1
            
            self._insert_non_full(node.children[i], key)
    
    def traverse(self):
        """In-order traversal."""
        result = []
        self._traverse(self.root, result)
        return result
    
    def _traverse(self, node, result):
        """Recursive traversal."""
        if node.is_leaf:
            result.extend(node.keys)
        else:
            for i in range(len(node.keys)):
                self._traverse(node.children[i], result)
                result.append(node.keys[i])
            self._traverse(node.children[len(node.keys)], result)
```

**Why**: Higher space utilization (2/3 full minimum) than B-trees through redistribution before splitting.
