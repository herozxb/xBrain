## [Maximum Sum Subarray]
```python
# PROBLEM: Find the maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)
def max_sum_subarray(arr, k):
    """
    Sliding window technique - maintain a window of size k,
    slide it across the array while tracking the maximum sum.
    """
    if len(arr) < k:
        return None
    
    # Compute sum of first window
    window_sum = sum(arr[:k])
    max_sum = window_sum
    
    # Slide the window: subtract leftmost, add new rightmost
    for i in range(k, len(arr)):
        window_sum = window_sum - arr[i - k] + arr[i]
        max_sum = max(max_sum, window_sum)
    
    return max_sum
```
**Why**: Classic sliding window pattern - avoids recomputing entire window sum each time, reducing O(n*k) to O(n).

## [Container With Most Water]
```python
# PROBLEM: Find two lines that form container holding the most water
# TIME: O(n) SPACE: O(1)
def max_area(height):
    """
    Two pointers from both ends. Move the shorter line inward
    because that's the only way to potentially increase area.
    """
    left, right = 0, len(height) - 1
    max_water = 0
    
    while left < right:
        # Area = min(height) * width
        width = right - left
        current_water = min(height[left], height[right]) * width
        max_water = max(max_water, current_water)
        
        # Move the shorter line - only way to possibly increase area
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1
    
    return max_water
```
**Why**: Two pointers greedy approach - moving the taller line can never increase area since it's bounded by the shorter one.

## [Permutations]
```python
# PROBLEM: Generate all permutations of a list of distinct integers
# TIME: O(n! * n) SPACE: O(n) for recursion stack
def permute(nums):
    """
    Backtracking: try each number at each position,
    backtrack by unmarking and trying the next option.
    """
    result = []
    
    def backtrack(path, used):
        # Base case: complete permutation
        if len(path) == len(nums):
            result.append(path[:])  # Copy the path
            return
        
        # Try each unused number
        for i in range(len(nums)):
            if used[i]:
                continue
            
            # Choose
            used[i] = True
            path.append(nums[i])
            
            # Explore
            backtrack(path, used)
            
            # Unchoose (backtrack)
            path.pop()
            used[i] = False
    
    backtrack([], [False] * len(nums))
    return result
```
**Why**: Backtracking explores all possibilities systematically. The "choose-explore-unchoose" pattern is fundamental.

## [Number of Connected Components]
```python
# PROBLEM: Count connected components in an undirected graph
# TIME: O(n * α(n)) ≈ O(n) SPACE: O(n)
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.components = n
    
    def find(self, x):
        # Path compression: point directly to root
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        # Union by rank: attach smaller tree to larger
        px, py = self.find(x), self.find(y)
        if px == py:
            return False  # Already connected
        
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        self.components -= 1
        return True

def count_components(n, edges):
    uf = UnionFind(n)
    for u, v in edges:
        uf.union(u, v)
    return uf.components
```
**Why**: Union-Find is optimal for dynamic connectivity problems. Path compression + union by rank gives near-constant time operations.

## [Longest Common Subsequence]
```python
# PROBLEM: Find the length of the longest common subsequence between two strings
# TIME: O(m * n) SPACE: O(m * n), can be optimized to O(min(m, n))
def lcs(text1, text2):
    """
    DP: If characters match, extend previous LCS.
    If not, take max of skipping one char from either string.
    """
    m, n = len(text1), len(text2)
    
    # dp[i][j] = LCS length of text1[:i] and text2[:j]
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if text1[i - 1] == text2[j - 1]:
                # Characters match: extend the LCS
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                # Take max of skipping char from either string
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    
    return dp[m][n]
```
**Why**: Classic 2D DP - each cell depends on left, top, and diagonal. The recurrence captures "match or skip" decision at each position.

## [Dijkstra's Shortest Path]
```python
# PROBLEM: Find shortest path from source to all nodes in weighted graph
# TIME: O((V + E) log V) SPACE: O(V)
import heapq

def dijkstra(n, edges, source):
    """
    Greedy BFS with min-heap. Always process the node
    with smallest known distance first.
    """
    # Build adjacency list
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u].append((v, w))
    
    # Distance array, initialized to infinity
    dist = [float('inf')] * n
    dist[source] = 0
    
    # Min-heap: (distance, node)
    heap = [(0, source)]
    
    while heap:
        d, u = heapq.heappop(heap)
        
        # Skip if we've found a better path already
        if d > dist[u]:
            continue
        
        # Relax all neighbors
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(heap, (dist[v], v))
    
    return dist
```
**Why**: Greedy + heap = optimal for non-negative weights. Processing nodes in order of distance guarantees correctness.

## [Binary Tree Level Order Traversal]
```python
# PROBLEM: Traverse binary tree level by level, left to right
# TIME: O(n) SPACE: O(w) where w is max width
from collections import deque

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def level_order(root):
    """
    BFS with queue. Process all nodes at current level
    before moving to the next level.
    """
    if not root:
        return []
    
    result = []
    queue = deque([root])
    
    while queue:
        level_size = len(queue)
        current_level = []
        
        # Process all nodes at current level
        for _ in range(level_size):
            node = queue.popleft()
            current_level.append(node.val)
            
            # Add children for next level
            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)
        
        result.append(current_level)
    
    return result
```
**Why**: BFS naturally processes level by level. Tracking queue size before processing lets us separate levels cleanly.

## [Merge K Sorted Lists]
```python
# PROBLEM: Merge k sorted linked lists into one sorted list
# TIME: O(n log k) where n = total nodes, k = number of lists
# SPACE: O(k) for the heap
import heapq

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def merge_k_lists(lists):
    """
    Min-heap of size k. Always extract the smallest element
    from the heap and add the next element from that list.
    """
    # Min-heap stores (value, list_index, node)
    heap = []
    for i, node in enumerate(lists):
        if node:
            heapq.heappush(heap, (node.val, i, node))
    
    dummy = ListNode()
    current = dummy
    
    while heap:
        val, i, node = heapq.heappop(heap)
        current.next = node
        current = current.next
        
        # Push next node from the same list
        if node.next:
            heapq.heappush(heap, (node.next.val, i, node.next))
    
    return dummy.next
```
**Why**: Heap efficiently maintains which list has the smallest next element. O(log k) per extraction gives O(n log k) total.

## [Trie - Prefix Search]
```python
# PROBLEM: Implement a trie with insert, search, and startsWith
# TIME: O(m) per operation where m = word length
# SPACE: O(alphabet_size * total_characters)
class TrieNode:
    def __init__(self):
        self.children = {}  # char -> TrieNode
        self.is_end = False

class Trie:
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word):
        """Insert word into trie, creating nodes as needed."""
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
    
    def search(self, word):
        """Return True if word exists in trie."""
        node = self.root
        for char in word:
            if char not in node.children:
                return False
            node = node.children[char]
        return node.is_end
    
    def starts_with(self, prefix):
        """Return True if any word starts with prefix."""
        node = self.root
        for char in prefix:
            if char not in node.children:
                return False
            node = node.children[char]
        return True
```
**Why**: Trie provides O(m) prefix matching regardless of how many words are stored. Essential for autocomplete, spell checkers.

## [LRU Cache]
```python
# PROBLEM: Implement LRU cache with O(1) get and put operations
# TIME: O(1) for both get and put
# SPACE: O(capacity)
class Node:
    """Doubly linked list node."""
    def __init__(self, key=0, val=0):
        self.key = key
        self.val = val
        self.prev = None
        self.next = None

class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}  # key -> Node
        
        # Dummy head and tail for easier manipulation
        self.head = Node()
        self.tail = Node()
        self.head.next = self.tail
        self.tail.prev = self.head
    
    def _remove(self, node):
        """Remove node from its position in the list."""
        node.prev.next = node.next
        node.next.prev = node.prev
    
    def _add_to_front(self, node):
        """Add node right after head (most recently used)."""
        node.prev = self.head
        node.next = self.head.next
        self.head.next.prev = node
        self.head.next = node
    
    def get(self, key):
        """Get value and mark as recently used."""
        if key not in self.cache:
            return -1
        
        node = self.cache[key]
        # Move to front (most recently used)
        self._remove(node)
        self._add_to_front(node)
        return node.val
    
    def put(self, key, value):
        """Add/update key-value pair, evict LRU if over capacity."""
        if key in self.cache:
            # Update existing node
            node = self.cache[key]
            node.val = value
            self._remove(node)
            self._add_to_front(node)
        else:
            # Add new node
            if len(self.cache) >= self.capacity:
                # Evict LRU (node before tail)
                lru = self.tail.prev
                self._remove(lru)
                del self.cache[lru.key]
            
            node = Node(key, value)
            self.cache[key] = node
            self._add_to_front(node)
```
**Why**: Hash map + doubly linked list gives O(1) access and O(1) eviction. The list maintains usage order; the map enables fast lookup.
