                const td = document.createElement('td');

                // Toggle selection
                this.classList.toggle('selected');

        function renderTable(dataArray) {
            const tbody = document.getElementById('tableBody');
            tbody.innerHTML = ''; // Clear existing rows

            dataArray.forEach(item => {
                fragment.appendChild(createTableRow(item));
            });
            
            tbody.appendChild(fragment);
        }
        
        // Initial render
        renderTable(data);

        function addRow(newItem) {
            const tbody = document.getElementById('tableBody');

                        height: this.naturalHeight,

           TIME: O(1) for type checking SPACE: O(1) */
        
        const output = document.getElementById('output');
        const results = [];
        
        // Primitive types (immutable)
        
        // 1. String
        const str = "Hello, World!";
        results.push(`String: ${str} (${typeof str})`);
        
        // 2. Number (integer and float)
        const int = 42;
        const float = 3.14159;
        const infinity = Infinity;
        const notANumber = NaN;
        results.push(`Number: ${int}, ${float}, ${infinity}, ${notANumber} (${typeof int})`);
        
        // 3. Boolean
        const bool = true;
        results.push(`Boolean: ${bool} (${typeof bool})`);
        
        // 4. Undefined (declared but not assigned)
        let undef;
        results.push(`Undefined: ${undef} (${typeof undef})`);
        
        // 5. Null (intentional absence)
        const nullVal = null;
        results.push(`Null: ${nullVal} (${typeof nullVal})`); // typeof null is "object" (bug)
        
        // 6. Symbol (unique identifier)
        const sym = Symbol('description');
        results.push(`Symbol: ${sym.toString()} (${typeof sym})`);
        
        // 7. BigInt (large integers)
        const bigNum = 9007199254740991n;
        results.push(`BigInt: ${bigNum} (${typeof bigNum})`);
        
        // Reference types (mutable)
        
        // 1. Object
        const obj = { name: 'John', age: 30 };
        results.push(`Object: ${JSON.stringify(obj)} (${typeof obj})`);
        
        // 2. Array (type of object)
        const arr = [1, 2, 3, 'four'];
        results.push(`Array: ${JSON.stringify(arr)} (${typeof arr})`);
        
        // 3. Function (type of function)
        const func = function(x) { return x * 2; };
        results.push(`Function: ${func.name} (${typeof func})`);
        
        // 4. Date
        const date = new Date();
        results.push(`Date: ${date.toISOString()} (${typeof date})`);
        
        // 5. RegExp
        const regex = /pattern/g;
        results.push(`RegExp: ${regex} (${typeof regex})`);
        
        // Type checking utilities
        
        // Better type checking
        function getType(value) {
            return Object.prototype.toString.call(value).slice(8, -1);
        }
        
        results.push(`\n--- Better Type Checking ---`);
        results.push(`null: ${getType(null)}`);
        results.push(`array: ${getType([1, 2, 3])}`);
        results.push(`date: ${getType(new Date())}`);
        results.push(`regex: ${getType(/test/)}`);
        
        // Type conversion
        results.push(`\n--- Type Conversion ---`);
        
        // String to Number
        results.push(`String "42" to Number: ${Number("42")} (${typeof Number("42")})`);
        results.push(`ParseInt("42px"): ${parseInt("42px")}`);
        
        // Number to String
        results.push(`Number 42 to String: ${String(42)} (${typeof String(42)})`);
        
        // To Boolean
        results.push(`Truthy "hello": ${Boolean("hello")}`);
        results.push(`Falsy 0: ${Boolean(0)}`);
        
        // Display results
        output.textContent = results.join('\n');
        
        // Deep type check function
        function isPrimitive(value) {

        const logs = [];
        
        function log(message) {
            const time = new Date().toLocaleTimeString();
            logs.unshift(`[${time}] ${message}`);
            if (logs.length > 10) logs.pop();
            output.innerHTML = logs.join('<br>');

            this.style.background = '#2ecc71';
        });
        
        box2.addEventListener('mouseleave', function() {
            log('Mouse left Box 2');
            this.style.background = '#3498db';
        });

        const box3 = document.getElementById('box3');
        
        box3.addEventListener('dblclick', function() {
            log('Box 3 double-clicked!');
            alert('Double click detected!');

            once: true,      // Only fire once
            capture: false,  // Use bubbling (default)
            passive: true    // Won't call preventDefault

            log('Page scrolled');
        }, { passive: true });

        function handleClick(e) {
