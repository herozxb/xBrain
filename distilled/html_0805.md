## html_form_enctype

```html
<!--
  PROBLEM: How to specify how form data should be encoded before sending to server
  TIME: O(1) SPACE: O(1)
-->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Form Enctype Demo</title>
</head>
<body>
  <!--
    enctype attribute specifies how form data is encoded:
    - application/x-www-form-urlencoded (default): URL encoded
    - multipart/form-data: Required for file uploads
    - text/plain: Spaces become +, no special character encoding
  -->
  
  <!-- Default: URL encoded - good for text data -->
  <form action="/submit" method="POST" enctype="application/x-www-form-urlencoded">
    <label>Username: <input type="text" name="username"></label>
    <button type="submit">Submit (URL Encoded)</button>
  </form>
  
  <!-- Multipart: Required when uploading files -->
  <form action="/upload" method="POST" enctype="multipart/form-data">
    <label>File: <input type="file" name="document"></label>
    <button type="submit">Upload File</button>
  </form>
  
  <!-- Text/Plain: Minimal encoding, rarely used -->
  <form action="/contact" method="POST" enctype="text/plain">
    <label>Message: <textarea name="message"></textarea></label>
    <button type="submit">Send Plain Text</button>
  </form>
</body>
</html>
```

**Why**: Understanding enctype is critical for file uploads - using the wrong encoding will corrupt binary data or fail to send files properly.

## html_form_method

```html
<!--
  PROBLEM: How to choose between GET and POST HTTP methods for form submission
  TIME: O(1) SPACE: O(1)
-->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Form Method Demo</title>
</head>
<body>
  <!--
    method attribute defines HTTP method:
    - GET: Data in URL query string (default)
    - POST: Data in request body
    
    GET: Use for safe, idempotent operations (search, filters)
    POST: Use for state-changing operations (create, update, delete)
  -->
  
  <!-- GET: Parameters visible in URL, bookmarkable, cacheable -->
  <form action="/search" method="GET">
    <label>Search: <input type="search" name="q" placeholder="Enter query"></label>
    <button type="submit">Search</button>
    <!-- URL becomes: /search?q=your+query -->
  </form>
  
  <!-- POST: Data in body, not visible in URL, more secure for sensitive data -->
  <form action="/login" method="POST">
    <label>Email: <input type="email" name="email" required></label>
    <label>Password: <input type="password" name="password" required></label>
    <button type="submit">Login</button>
    <!-- Credentials sent in request body, not URL -->
  </form>
  
  <!--
    Key differences:
    - GET: Limited URL length (~2000 chars), cacheable, bookmarkable
    - POST: No size limit, not cached, not bookmarkable
    - GET: Should NOT modify server state
    - POST: CAN modify server state
  -->
</body>
</html>
```

**Why**: Choosing the correct method is fundamental to RESTful design, security (credentials in URL), and proper HTTP semantics.

## html_audio_playlist

```html
<!--
  PROBLEM: How to create an audio player with playlist functionality
  TIME: O(n) for n tracks SPACE: O(1)
-->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Audio Playlist Demo</title>
  <style>
    .playlist { list-style: none; padding: 0; }
    .playlist li { padding: 10px; cursor: pointer; border-bottom: 1px solid #ddd; }
    .playlist li.active { background: #007bff; color: white; }
    .playlist li:hover { background: #f0f0f0; }
    .playlist li.active:hover { background: #0056b3; }
  </style>
</head>
<body>
  <audio id="audioPlayer" controls></audio>
  
  <ul class="playlist" id="playlist">
    <li data-src="track1.mp3">Track 1 - Introduction</li>
    <li data-src="track2.mp3">Track 2 - Main Theme</li>
    <li data-src="track3.mp3">Track 3 - Conclusion</li>
  </ul>
  
  <script>
    // Get DOM elements
    const audio = document.getElementById('audioPlayer');
    const playlist = document.getElementById('playlist');
    const tracks = playlist.querySelectorAll('li');
    
    let currentTrack = 0;
    
    // Load and play a track by index
    function loadTrack(index) {
      // Clamp index to valid range
      if (index < 0) index = tracks.length - 1;
      if (index >= tracks.length) index = 0;
      
      currentTrack = index;
      
      // Update active state in playlist
      tracks.forEach((track, i) => {
        track.classList.toggle('active', i === index);
      });
      
      // Load the audio source and play
      audio.src = tracks[index].dataset.src;
      audio.play();
    }
    
    // Click handler for playlist items
    tracks.forEach((track, index) => {
      track.addEventListener('click', () => loadTrack(index));
    });
    
    // Auto-advance to next track when current ends
    audio.addEventListener('ended', () => {
      loadTrack(currentTrack + 1);
    });
    
    // Initialize first track (don't autoplay)
    tracks[0].classList.add('active');
    audio.src = tracks[0].dataset.src;
  </script>
</body>
</html>
```

**Why**: Playlists require coordinating audio events with UI state - understanding ended events and track management is essential for media apps.

## css_line_height

```html
<!--
  PROBLEM: How to control vertical spacing between lines of text
  TIME: O(1) SPACE: O(1)
-->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Line Height Demo</title>
  <style>
    /*
      line-height controls the vertical space between lines.
      Values can be:
      - Normal (browser default, usually ~1.2)
      - Unitless number (multiplier of font-size, RECOMMENDED)
      - Length (px, em, etc.)
      - Percentage (%)
      
      BEST PRACTICE: Use unitless numbers for inheritance
    */
    
    .tight { line-height: 1; }       /* Compact, lines touch */
    .normal { line-height: 1.5; }    /* Comfortable reading */
    .relaxed { line-height: 2; }     /* Spacious */
    
    .demo {
      font-size: 18px;
      border: 1px solid #ccc;
      padding: 10px;
      margin: 10px 0;
    }
    
    /* Unitless is best: inherits as multiplier */
    .parent {
      line-height: 1.5;
      font-size: 20px;
    }
    .child {
      font-size: 14px;
      /* line-height: 1.5 * 14px = 21px (inherited as ratio) */
    }
    
    /* Fixed units can cause issues with inheritance */
    .parent-fixed {
      line-height: 30px;  /* Fixed value */
      font-size: 20px;
    }
    .child-fixed {
      font-size: 40px;  /* Line height still 30px - text overlaps! */
    }
  </style>
</head>
<body>
  <div class="demo tight">
    <strong>line-height: 1 (Tight)</strong><br>
    This text has tight line spacing. Lines are close together.
    Good for headings but hard to read for long paragraphs.
  </div>
  
  <div class="demo normal">
    <strong>line-height: 1.5 (Normal)</strong><br>
    This text has comfortable line spacing. Good for body text.
    The recommended value for readable paragraphs.
  </div>
  
  <div class="demo relaxed">
    <strong>line-height: 2 (Relaxed)</strong><br>
    This text has spacious line spacing. Good for emphasis
    or decorative purposes.
  </div>
</body>
</html>
```

**Why**: Proper line-height is crucial for readability - too tight strains eyes, too loose breaks reading flow. Unitless values prevent inheritance bugs.

## js_type_conversion

```html
<!--
  PROBLEM: How JavaScript converts between different data types explicitly and implicitly
  TIME: O(1) SPACE: O(1)
-->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Type Conversion Demo</title>
</head>
<body>
  <script>
    // ============================================
    // EXPLICIT CONVERSION (Type Casting)
    // ============================================
    
    // To String
    String(123);          // "123"
    String(true);         // "true"
    String(null);         // "null"
    String(undefined);    // "undefined"
    (123).toString();     // "123"
    
    // To Number
    Number("123");        // 123
    Number("12.5");       // 12.5
    Number("12px");       // NaN (invalid)
    Number(true);         // 1
    Number(false);        // 0
    Number(null);         // 0
    Number(undefined);    // NaN
    parseInt("123");      // 123
    parseInt("12.9");     // 12 (truncates)
    parseInt("12px");     // 12 (parses until non-digit)
    parseFloat("12.5");   // 12.5
    parseFloat("12.5px"); // 12.5
    
    // To Boolean
    Boolean(1);           // true
    Boolean(0);           // false
    Boolean("");          // false (empty string)
    Boolean("hello");     // true (non-empty string)
    Boolean(null);        // false
    Boolean(undefined);   // false
    Boolean({});          // true (objects are truthy)
    Boolean([]);          // true (arrays are truthy)
    
    // ============================================
    // IMPLICIT CONVERSION (Coercion)
    // ============================================
    
    // String concatenation with +
    "5" + 3;              // "53" (number coerced to string)
    "5" + true;           // "5true"
    
    // Numeric conversion with other operators
    "5" - 3;              // 2 (string coerced to number)
    "5" * "2";            // 10
    "5" / "2";            // 2.5
    
    // Loose equality (==) triggers coercion
    5 == "5";             // true (string coerced to number)
    null == undefined;    // true (special case)
    0 == false;           // true
    0 == "";              // true
    0 == null;            // false!
    
    // Strict equality (===) - NO coercion
    5 === "5";            // false (different types)
    null === undefined;   // false
    
    // Logical operators
    "hello" && 0;         // 0 (returns last evaluated value)
    "" || "default";      // "default" (returns first truthy)
    !!"hello";            // true (double negation to boolean)
    
    // ============================================
    // COMMON PITFALLS
    // ============================================
    
    // Array to string
    String([1, 2, 3]);    // "1,2,3"
    [1] + [2];            // "12" (arrays to strings, then concatenate)
    
    // Object to primitive
    String({});           // "[object Object]"
    Number({});           // NaN
    
    // Weird coercions
    [] == false;          // true ([] becomes "", then 0)
    ![] == false;         // true (![] is false)
    [] == ![];            // true (both become false/0)
  </script>
</body>
</html>
```

**Why**: Type coercion is a major source of bugs in JavaScript. Understanding explicit vs implicit conversion prevents unexpected behavior and helps write predictable code.

## dom_parent_element

```html
<!--
  PROBLEM: How to access and traverse parent elements in the DOM tree
  TIME: O(1) SPACE: O(1)
-->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Parent Element Demo</title>
</head>
<body>
  <div id="grandparent" class="level-0">
    <div id="parent" class="level-1">
      <div id="child" class="level-2">
        <span id="target">Click me</span>
      </div>
    </div>
  </div>
  
  <ul id="list">
    <li class="item">Item 1</li>
    <li class="item">Item 2</li>
    <li class="item">Item 3</li>
  </ul>
  
  <script>
    // ============================================
    // PARENT ELEMENT VS PARENT NODE
    // ============================================
    
    const target = document.getElementById('target');
    
    // parentElement: Returns Element parent or null
    // More commonly used for DOM manipulation
    target.parentElement;          // <div id="child">
    target.parentElement.parentElement;  // <div id="parent">
    target.parentElement.parentElement.parentElement;  // <div id="grandparent">
    
    // parentNode: Returns any Node parent (Element, Document, etc.)
    // Returns Document or DocumentFragment at root
    target.parentNode;             // Same as parentElement for elements
    document.documentElement.parentNode;  // Document (not an Element!)
    
    // Key difference:
    document.documentElement.parentElement;  // null
    document.documentElement.parentNode;     // #document
    
    // ============================================
    // PRACTICAL USE CASES
    // ============================================
    
    // 1. Find closest ancestor matching a selector
    function findAncestor(element, selector) {
      let parent = element.parentElement;
      
      while (parent) {
        if (parent.matches(selector)) {
          return parent;
        }
        parent = parent.parentElement;
      }
      
      return null;  // No matching ancestor found
    }
    
    // Usage
    findAncestor(target, '.level-1');  // <div id="parent">
    
    // 2. closest() - Modern built-in method (recommended)
    target.closest('.level-1');  // <div id="parent">
    target.closest('div');       // <div id="child"> (nearest div ancestor)
    target.closest('body');      // <body>
    target.closest('.nonexistent');  // null
    
    // Note: closest() includes the element itself
    target.closest('#target');   // Returns target itself!
    
    // 3. Remove element and its parent
    function removeElementAndParent(element) {
      const parent = element.parentElement;
      if (parent) {
        parent.remove();
      }
    }
    
    // 4. Get element's depth in DOM tree
    function getDOMDepth(element) {
      let depth = 0;
      let current = element;
      
      while (current.parentElement) {
        depth++;
        current = current.parentElement;
      }
      
      return depth;
    }
    
    // 5. Check if element is descendant of another
    function isDescendantOf(element, potentialAncestor) {
      let parent = element.parentElement;
      
      while (parent) {
        if (parent === potentialAncestor) {
          return true;
        }
        parent = parent.parentElement;
      }
      
      return false;
    }
    
    // Modern approach using contains()
    document.body.contains(target);  // true
    
    // 6. Event delegation - check event origin
    document.getElementById('list').addEventListener('click', (e) => {
      // Find the clicked list item
      const item = e.target.closest('.item');
      
      if (item) {
        console.log('Clicked:', item.textContent);
      }
    });
    
    // 7. Traverse up to find specific parent type
    function findParentForm(element) {
      return element.closest('form');
    }
    
    // 8. Get all ancestors as array
    function getAncestors(element) {
      const ancestors = [];
      let parent = element.parentElement;
      
      while (parent) {
        ancestors.push(parent);
        parent = parent.parentElement;
      }
      
      return ancestors;
    }
    
    // Usage
    getAncestors(target);
    // [div#child, div#parent, div#grandparent, body, html]
  </script>
</body>
</html>
```

**Why**: Parent traversal is essential for event delegation, DOM manipulation (removing/inserting elements), and implementing features like tree navigation or hierarchical UIs.

## dom_children_elements

```html
<!--
  PROBLEM: How to access and iterate over child elements in the DOM
  TIME: O(n) for n children SPACE: O(1)
-->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Children Elements Demo</title>
</head>
<body>
  <div id="container">
    <p class="item">First paragraph</p>
    Text node (not an element)
    <p class="item">Second paragraph</p>
    <span class="item">A span</span>
    <!-- Comment node -->
    <p class="item">Third paragraph</p>
  </div>
  
  <ul id="list">
    <li>Item 1</li>
    <li>Item 2</li>
    <li>Item 3</li>
  </ul>
  
  <script>
    const container = document.getElementById('container');
    
    // ============================================
    // CHILDREN VS CHILDNODES
    // ============================================
    
    // children: HTMLCollection of ELEMENT children only
    container.children;
    // HTMLCollection(4) [p.item, p.item, span.item, p.item]
    // Length: 4 (only elements, ignores text/comments)
    
    // childNodes: NodeList of ALL child nodes
    container.childNodes;
    // NodeList(9) [text, p.item, text, p.item, text, span.item, comment, text, p.item]
    // Includes text nodes, comments, etc.
    
    // ============================================
    // ACCESSING SPECIFIC CHILDREN
    // ============================================
    
    // First/last element children
    container.firstElementChild;  // <p class="item">First paragraph</p>
    container.lastElementChild;   // <p class="item">Third paragraph</p>
    
    // First/last child nodes (may be text/comment)
    container.firstChild;   // Text node "Text node (not an element)"
    container.lastChild;    // May be text node with whitespace
    
    // Child count
    container.childElementCount;  // 4 (elements only)
    container.children.length;    // 4 (same as above)
    
    // ============================================
    // ITERATING OVER CHILDREN
    // ============================================
    
    // Method 1: for...of (recommended)
    for (const child of container.children) {
      console.log(child.tagName, child.className);
    }
    
    // Method 2: Convert to array for array methods
    const childrenArray = Array.from(container.children);
    childrenArray
      .filter(child => child.tagName === 'P')
      .forEach(child => child.style.color = 'blue');
    
    // Method 3: Traditional for loop
    for (let i = 0; i < container.children.length; i++) {
      container.children[i].dataset.index = i;
    }
    
    // ============================================
    // PRACTICAL USE CASES
    // ============================================
    
    // 1. Add class to all children
    function addClassToAllChildren(parent, className) {
      for (const child of parent.children) {
        child.classList.add(className);
      }
    }
    
    // 2. Remove all children
    function removeAllChildren(parent) {
      while (parent.firstChild) {
```
