```python
    pivot = arr[high]
    # Pointer for smaller element position
    i = low - 1
    # Iterate through partition
    for j in range(low, high):
        # If current element <= pivot
        if arr[j] <= pivot:
            # Increment smaller element index
            i += 1
            # Swap elements
            arr[i], arr[j] = arr[j], arr[i]
```

```python
# TIME: O(V + E) SPACE: O(V)
from collections import deque

def bfs(graph, start):

    visited = set()
    # Queue for BFS traversal
    queue = deque([start])
    # Result list for traversal order
    result = []
    # Continue until queue is empty
    while queue:
        # Dequeue front node
        node = queue.popleft()
        # Skip if already visited
        if node in visited:
            continue
        # Mark as visited
        visited.add(node)
        # Add to result
        result.append(node)
        # Add neighbors to queue
        for neighbor in graph[node]:
            if neighbor not in visited:
                queue.append(neighbor)
    return result
```
**Why**: Explores nearest nodes first, finds shortest path in unweighted graphs

```python
# TIME: O(N!) SPACE: O(N²)
def solve_n_queens(n):
    def is_safe(board, row, col):
        # Check column
        for i in range(row):
            if board[i][col]:
                return False
        # Check upper left diagonal
        i, j = row - 1, col - 1
        while i >= 0 and j >= 0:
            if board[i][j]:
                return False
            i -= 1
            j -= 1
        # Check upper right diagonal
        i, j = row - 1, col + 1
        while i >= 0 and j < n:
            if board[i][j]:
                return False
            i -= 1
            j += 1
        return True
    
    def solve(board, row):
        # Base case: all queens placed
        if row == n:
            return True
        # Try each column
        for col in range(n):
            if is_safe(board, row, col):
                board[row][col] = 1
                if solve(board, row + 1):
                    return True
                board[row][col] = 0  # Backtrack
        return False
    
    board = [[0] * n for _ in range(n)]
    solve(board, 0)
    return board
```
**Why**: Systematic exploration with pruning, finds all valid configurations

## [Topological Sort - Kahn's Algorithm]
```python
# PROBLEM: Order tasks with dependencies
# TIME: O(V + E) SPACE: O(V)
from collections import deque, defaultdict

def topological_sort(n, edges):
    # Build adjacency list and in-degree count
    graph = defaultdict(list)
    in_degree = [0] * n
    for u, v in edges:
        graph[u].append(v)
        in_degree[v] += 1
    # Queue of nodes with no dependencies
    queue = deque([i for i in range(n) if in_degree[i] == 0])
    result = []
    # Process nodes
    while queue:
        node = queue.popleft()
        result.append(node)
        # Reduce in-degree of neighbors
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    # Check for cycle
    return result if len(result) == n else []
```
**Why**: Handles dependencies efficiently, detects cycles

## [Dijkstra's Algorithm - Shortest Path]
```python
# PROBLEM: Find shortest path from source
# TIME: O((V + E) log V) SPACE: O(V)
import heapq

def dijkstra(n, edges, source):
    # Build adjacency list
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
    # Distance array initialized to infinity
    dist = [float('inf')] * n
    dist[source] = 0
    # Priority queue: (distance, node)
    pq = [(0, source)]
    while pq:
        d, u = heapq.heappop(pq)
        # Skip if already found shorter path
        if d > dist[u]:
            continue
        # Relax neighbors
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    return dist
```
**Why**: Optimal for non-negative weights, priority queue ensures efficiency

```python
# PROBLEM: Group words that are anagrams
# TIME: O(n * k log k) where k is max word length SPACE: O(n*k)
from collections import defaultdict

def group_anagrams(strs):
    # Map sorted word to list of anagrams
    groups = defaultdict(list)
    for word in strs:
        # Sort characters to create key
        key = ''.join(sorted(word))
        groups[key].append(word)
    return list(groups.values())
```
**Why**: Sorted string as key groups all anagrams together

## [Recursion - Tower of Hanoi]
```python
# PROBLEM: Move disks between pegs
# TIME: O(2^n) SPACE: O(n) recursion depth
def tower_of_hanoi(n, source, auxiliary, target):
    if n == 1:
        # Base case: move single disk
        print(f"Move disk 1 from {source} to {target}")
        return
    # Move n-1 disks from source to auxiliary
    tower_of_hanoi(n - 1, source, target, auxiliary)
    # Move largest disk to target
    print(f"Move disk {n} from {source} to {target}")
    # Move n-1 disks from auxiliary to target
    tower_of_hanoi(n - 1, auxiliary, source, target)
```
**Why**: Classic recursion example, demonstrates divide and conquer

```python
class Timer:
```

## [Property Decorator - Encapsulation]
```python
# PROBLEM: Control attribute access
# TIME: O(1) SPACE: O(1)
class Circle:
    def __init__(self, radius):
        self._radius = radius
    
    @property
    def radius(self):
        # Getter
        return self._radius
    
    @radius.setter
    def radius(self, value):
        # Setter with validation
        if value < 0:
            raise ValueError("Radius cannot be negative")
        self._radius = value
    
    @property
    def area(self):
        # Computed property (read-only)
        return 3.14159 * self._radius ** 2
```

## [Abstract Base Class]

```python
# PROBLEM: Define interface for subclasses
# TIME: O(1) SPACE: O(1)
from abc import ABC, abstractmethod

class Shape(ABC):
    @abstractmethod
    def area(self):
        pass
    
    @abstractmethod
    def perimeter(self):
        pass

class Rectangle(Shape):
    def __init__(self, width, height):
        self.width = width
        self.height = height
    
    def area(self):
        return self.width * self.height
    
    def perimeter(self):
        return 2 * (self.width + self.height)
```
**Why**: Enforces interface contract, prevents instantiation of incomplete classes

## [Magic Methods - Operator Overloading]

```python
# PROBLEM: Define custom behavior for operators
# TIME: O(1) SPACE: O(1)
class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __add__(self, other):
        # Define + operator
        return Vector(self.x + other.x, self.y + other.y)
    
    def __repr__(self):
        # Define string representation
        return f"Vector({self.x}, {self.y})"
    
    def __eq__(self, other):
        # Define == operator
        return self.x == other.x and self.y == other.y
    
    def __len__(self):
        # Define len() behavior
        return int((self.x**2 + self.y**2)**0.5)
```
**Why**: Makes classes behave like built-in types, intuitive API

## [Error Handling - Custom Exceptions]

```python
# PROBLEM: Create domain-specific exceptions
# TIME: O(1) SPACE: O(1)
class InsufficientFunds(Exception):
    def __init__(self, balance, amount):
        self.balance = balance
        self.amount = amount
        super().__init__(
            f"Insufficient funds: ${balance} < ${amount}"
        )

class BankAccount:
    def __init__(self, balance=0):
        self.balance = balance
    
    def withdraw(self, amount):
        if amount > self.balance:
            raise InsufficientFunds(self.balance, amount)
        self.balance -= amount
        return self.balance

# Usage with try-except
try:
    account = BankAccount(100)
    account.withdraw(150)
except InsufficientFunds as e:
    print(f"Error: {e}")
```
**Why**: Meaningful error messages, proper exception hierarchy

```python
import threading
import time

def worker(name, delay):
```

## [File Handling - CSV and JSON]
```python
# PROBLEM: Read and write structured data
# TIME: O(n) where n is data size SPACE: O(n)
import csv
import json

# CSV operations
def write_csv(filename, data):
    with open(filename, 'w', newline='') as f:
        writer = csv.DictWriter(f, fieldnames=data[0].keys())
        writer.writeheader()
        writer.writerows(data)

def read_csv(filename):
    with open(filename, 'r') as f:
        return list(csv.DictReader(f))

# JSON operations
def write_json(filename, data):
    with open(filename, 'w') as f:
        json.dump(data, f, indent=2)

def read_json(filename):
    with open(filename, 'r') as f:
        return json.load(f)
```
**Why**: Standard data interchange formats, simple APIs
