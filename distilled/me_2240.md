## Chain of Responsibility Variants

```python
# PROBLEM: Pass request through chain of handlers until one handles it
# TIME: O(n) SPACE: O(1)
class Handler:
    """Abstract base handler for chain of responsibility pattern."""
    
    def __init__(self):
        self._next_handler = None
    
    def set_next(self, handler):
        """Set the next handler in the chain."""
        self._next_handler = handler
        return handler  # Enable chaining
    
    def handle(self, request):
        """Handle request or pass to next handler."""
        if self._next_handler:
            return self._next_handler.handle(request)
        return None


class AuthenticationHandler(Handler):
    """Handles authentication validation."""
    
    def handle(self, request):
        if not request.get('authenticated'):
            return "Authentication failed"
        return super().handle(request)


class AuthorizationHandler(Handler):
    """Handles authorization checks."""
    
    def handle(self, request):
        if request.get('role') != 'admin':
            return "Authorization failed"
        return super().handle(request)


class RateLimitHandler(Handler):
    """Handles rate limiting checks."""
    
    def __init__(self, max_requests=100):
        super().__init__()
        self.max_requests = max_requests
        self.request_count = {}
    
    def handle(self, request):
        user = request.get('user', 'anonymous')
        count = self.request_count.get(user, 0)
        
        if count >= self.max_requests:
            return f"Rate limit exceeded for {user}"
        
        self.request_count[user] = count + 1
        return super().handle(request)


def create_handler_chain():
    """Create and return configured handler chain."""
    auth = AuthenticationHandler()
    authz = AuthorizationHandler()
    rate = RateLimitHandler(max_requests=10)
    
    # Build chain: rate -> auth -> authz
    rate.set_next(auth).set_next(authz)
    return rate
```

**Why**: Demonstrates flexible request processing where each handler can either process the request or pass it along the chain. Useful for middleware pipelines, validation chains, and event handling systems.

## Interpreter Pattern

```python
# PROBLEM: Evaluate expressions in a custom language or notation
# TIME: O(n) where n is expression length SPACE: O(d) where d is depth
from abc import ABC, abstractmethod


class Expression(ABC):
    """Abstract expression interface."""
    
    @abstractmethod
    def interpret(self, context):
        """Interpret the expression in given context."""
        pass


class Number(Expression):
    """Terminal expression for numeric values."""
    
    def __init__(self, value):
        self.value = value
    
    def interpret(self, context):
        return self.value


class Variable(Expression):
    """Terminal expression for variable references."""
    
    def __init__(self, name):
        self.name = name
    
    def interpret(self, context):
        return context.get(self.name, 0)


class Add(Expression):
    """Non-terminal expression for addition."""
    
    def __init__(self, left, right):
        self.left = left
        self.right = right
    
    def interpret(self, context):
        return self.left.interpret(context) + self.right.interpret(context)


class Subtract(Expression):
    """Non-terminal expression for subtraction."""
    
    def __init__(self, left, right):
        self.left = left
        self.right = right
    
    def interpret(self, context):
        return self.left.interpret(context) - self.right.interpret(context)


class Multiply(Expression):
    """Non-terminal expression for multiplication."""
    
    def __init__(self, left, right):
        self.left = left
        self.right = right
    
    def interpret(self, context):
        return self.left.interpret(context) * self.right.interpret(context)


class Parser:
    """Simple expression parser for arithmetic operations."""
    
    def __init__(self, expression):
        self.tokens = expression.split()
        self.pos = 0
    
    def parse(self):
        """Parse the expression and return AST."""
        return self._parse_expression()
    
    def _parse_expression(self):
        """Parse addition and subtraction."""
        left = self._parse_term()
        
        while self.pos < len(self.tokens) and self.tokens[self.pos] in ['+', '-']:
            op = self.tokens[self.pos]
            self.pos += 1
            right = self._parse_term()
            
            if op == '+':
                left = Add(left, right)
            else:
                left = Subtract(left, right)
        
        return left
    
    def _parse_term(self):
        """Parse multiplication."""
        left = self._parse_factor()
        
        while self.pos < len(self.tokens) and self.tokens[self.pos] == '*':
            self.pos += 1
            right = self._parse_factor()
            left = Multiply(left, right)
        
        return left
    
    def _parse_factor(self):
        """Parse numbers and variables."""
        token = self.tokens[self.pos]
        self.pos += 1
        
        if token.isdigit():
            return Number(int(token))
        else:
            return Variable(token)


def evaluate_expression(expr_string, context=None):
    """Evaluate arithmetic expression with variables."""
    if context is None:
        context = {}
    
    parser = Parser(expr_string)
    ast = parser.parse()
    return ast.interpret(context)
```

**Why**: Implements a simple interpreter for arithmetic expressions with variables. Useful for building domain-specific languages, query parsers, and rule engines.

## Iterator Pattern Variants

```python
# PROBLEM: Traverse collections without exposing internal structure
# TIME: O(1) per iteration SPACE: O(1)
from abc import ABC, abstractmethod


class Iterator(ABC):
    """Abstract iterator interface."""
    
    @abstractmethod
    def has_next(self):
        """Check if more elements exist."""
        pass
    
    @abstractmethod
    def next(self):
        """Return next element."""
        pass


class Iterable(ABC):
    """Abstract iterable interface."""
    
    @abstractmethod
    def create_iterator(self):
        """Create iterator for this collection."""
        pass


class ListIterator(Iterator):
    """Forward iterator for list."""
    
    def __init__(self, collection):
        self.collection = collection
        self.index = 0
    
    def has_next(self):
        return self.index < len(self.collection)
    
    def next(self):
        if not self.has_next():
            raise StopIteration
        item = self.collection[self.index]
        self.index += 1
        return item


class ReverseIterator(Iterator):
    """Reverse iterator for list."""
    
    def __init__(self, collection):
        self.collection = collection
        self.index = len(collection) - 1
    
    def has_next(self):
        return self.index >= 0
    
    def next(self):
        if not self.has_next():
            raise StopIteration
        item = self.collection[self.index]
        self.index -= 1
        return item


class SkipIterator(Iterator):
    """Iterator that skips every n elements."""
    
    def __init__(self, collection, skip=1):
        self.collection = collection
        self.index = 0
        self.skip = skip
    
    def has_next(self):
        return self.index < len(self.collection)
    
    def next(self):
        if not self.has_next():
            raise StopIteration
        item = self.collection[self.index]
        self.index += self.skip + 1
        return item


class FilterIterator(Iterator):
    """Iterator that filters elements based on predicate."""
    
    def __init__(self, collection, predicate):
        self.collection = collection
        self.predicate = predicate
        self.index = 0
        self._find_next()
    
    def _find_next(self):
        """Find next matching element."""
        while self.index < len(self.collection):
            if self.predicate(self.collection[self.index]):
                return
            self.index += 1
    
    def has_next(self):
        return self.index < len(self.collection)
    
    def next(self):
        if not self.has_next():
            raise StopIteration
        item = self.collection[self.index]
        self.index += 1
        self._find_next()
        return item


class NumberCollection(Iterable):
    """Collection with multiple iterator types."""
    
    def __init__(self, items):
        self.items = items
    
    def create_iterator(self, iterator_type='forward'):
        """Create iterator of specified type."""
        iterators = {
            'forward': ListIterator,
            'reverse': ReverseIterator,
            'skip': SkipIterator,
            'filter': FilterIterator
        }
        
        iterator_class = iterators.get(iterator_type, ListIterator)
        
        if iterator_type == 'skip':
            return iterator_class(self.items, skip=1)
        elif iterator_type == 'filter':
            return iterator_class(self.items, lambda x: x % 2 == 0)
        
        return iterator_class(self.items)
```

**Why**: Provides multiple iteration strategies for collections without exposing internal representation. Supports forward, reverse, skip, and filtered iteration patterns.

## Memento Pattern

```python
# PROBLEM: Capture and restore object state without violating encapsulation
# TIME: O(n) for state snapshot SPACE: O(n) for state storage
class Memento:
    """Stores internal state of Originator object."""
    
    def __init__(self, state):
        self._state = state
        self._timestamp = self._get_timestamp()
    
    def _get_timestamp(self):
        """Get current timestamp."""
        from datetime import datetime
        return datetime.now()
    
    def get_state(self):
        """Return saved state."""
        return self._state.copy()
    
    def get_timestamp(self):
        """Return when memento was created."""
        return self._timestamp


class Originator:
    """Object whose state needs to be saved."""
    
    def __init__(self):
        self._state = {}
    
    def set_state(self, key, value):
        """Set state value."""
        self._state[key] = value
    
    def get_state(self, key):
        """Get state value."""
        return self._state.get(key)
    
    def create_memento(self):
        """Create memento with current state."""
        return Memento(self._state.copy())
    
    def restore_from_memento(self, memento):
        """Restore state from memento."""
        self._state = memento.get_state()
    
    def __str__(self):
        return f"Originator(state={self._state})"


class Caretaker:
    """Manages mementos and provides undo/redo functionality."""
    
    def __init__(self, originator):
        self.originator = originator
        self._history = []
        self._current = -1
    
    def save(self):
        """Save current state."""
        # Remove any redo states
        self._history = self._history[:self._current + 1]
        
        # Add new state
        memento = self.originator.create_memento()
        self._history.append(memento)
        self._current += 1
    
    def undo(self):
        """Restore previous state."""
        if self._current > 0:
            self._current -= 1
            self.originator.restore_from_memento(self._history[self._current])
            return True
        return False
    
    def redo(self):
        """Restore next state."""
        if self._current < len(self._history) - 1:
            self._current += 1
            self.originator.restore_from_memento(self._history[self._current])
            return True
        return False
    
    def get_history(self):
        """Get list of all saved states with timestamps."""
        return [(i, m.get_timestamp()) for i, m in enumerate(self._history)]


class TransactionManager:
    """Example: Database transaction with rollback capability."""
    
    def __init__(self):
        self.originator = Originator()
        self.caretaker = Caretaker(self.originator)
    
    def begin_transaction(self):
        """Start new transaction."""
        self.caretaker.save()
    
    def commit(self):
        """Commit current state."""
        self.caretaker.save()
    
    def rollback(self):
        """Rollback to previous state."""
        return self.caretaker.undo()
    
    def update(self, key, value):
        """Update state."""
        self.originator.set_state(key, value)
```

**Why**: Enables undo/redo functionality and transaction rollback without exposing internal state. Essential for text editors, database transactions, and game state management.

## Prototype Pattern

```python
# PROBLEM: Create objects by cloning existing instances
# TIME: O(n) for deep copy SPACE: O(n) for cloned object
import copy
from abc import ABC, abstractmethod


class Prototype(ABC):
    """Abstract prototype interface."""
    
    @abstractmethod
    def clone(self):
        """Create copy of this object."""
        pass


class Document(Prototype):
    """Concrete prototype for document objects."""
    
    def __init__(self, title, content, metadata=None):
        self.title = title
        self.content = content
        self.metadata = metadata or {}
        self.created_at = self._get_timestamp()
    
    def _get_timestamp(self):
        """Get current timestamp."""
        from datetime import datetime
        return datetime.now()
    
    def clone(self, **kwargs):
        """Create shallow or deep copy with optional overrides."""
        # Deep copy to avoid shared references
        cloned = copy.deepcopy(self)
        
        # Apply any overrides
        for key, value in kwargs.items():
            if hasattr(cloned, key):
                setattr(cloned, key, value)
        
        return cloned
    
    def __str__(self):
        return f"Document(title='{self.title}', content='{self.content[:30]}...')"


class Shape(Prototype):
    """Prototype for geometric shapes."""
    
    def __init__(self, x, y, color='black'):
        self.x = x
        self.y = y
        self.color = color
    
    def clone(self, **kwargs):
        """Clone shape with optional parameter changes."""
        cloned = copy.deepcopy(self)
        for key, value in kwargs.items():
            if hasattr(cloned, key):
                setattr(cloned, key, value)
        return cloned
    
    def draw(self):
        """Draw the shape."""
        return f"Drawing {self.color} shape at ({self.x}, {self.y})"


class Circle(Shape):
    """Circle prototype."""
    
    def __init__(self, x, y, radius, color='black'):
        super().__init__(x, y, color)
        self.radius = radius
    
    def draw(self):
        return f"Drawing {self.color} circle at ({self.x}, {self.y}) with radius {self.radius}"


class PrototypeRegistry:
    """Registry for managing prototype objects."""
    
    def __init__(self):
        self._prototypes = {}
    
    def register(self, name, prototype):
        """Register a prototype with given name."""
        self._prototypes[name] = prototype
    
    def unregister(self, name):
        """Remove prototype from registry."""
        self._prototypes.pop(name, None)
    
    def clone(self, name, **kwargs):
        """Clone registered prototype."""
        prototype = self._prototypes.get(name)
        if not prototype:
            raise ValueError(f"Prototype '{name}' not found")
        return prototype.clone(**kwargs)
    
    def list_prototypes(self):
        """List all registered prototypes."""
        return list(self._prototypes.keys())


def create_prototype_factory():
    """Create and populate prototype registry."""
    registry = PrototypeRegistry()
    
    # Register document templates
    registry.register('invoice', Document(
        title='Invoice Template',
        content='Invoice #{{number}}\nDate: {{date}}\nAmount: ${{amount}}',
        metadata={'type': 'financial', 'department': 'accounting'}
    ))
    
    registry.register('report', Document(
        title='Monthly Report',
        content='Report for {{month}} {{year}}\n\n{{content}}',
        metadata={'type': 'report', 'confidential': False}
    ))
    
    # Register shape prototypes
    registry.register('small_circle', Circle(0, 0, 5, 'blue'))
    registry.register('large_circle', Circle(0, 0, 50, 'red'))
    
    return registry
```

**Why**: Avoids expensive object creation by cloning existing instances. Useful for creating objects with complex initialization, templates, and game object spawning.

```python
# PROBLEM: Implement custom iteration protocol for custom containers
# TIME: O(1) per iteration SPACE: O(1)
class FibonacciIterator:
    """Iterator that generates Fibonacci numbers on demand."""
    
    def __init__(self, max_count=None):
        self.max_count = max_count
        self.count = 0
        self.prev = 0
        self.curr = 1
    
    def __iter__(self):
        """Return iterator object."""
        return self
    
    def __next__(self):
        """Return next Fibonacci number."""
        if self.max_count is not None and self.count >= self.max_count:
            raise StopIteration
        
        if self.count == 0:
            self.count += 1
            return 0
        elif self.count == 1:
            self.count += 1
            return 1
        
        # Generate next Fibonacci number
        result = self.prev + self.curr
        self.prev, self.curr = self.curr, result
        self.count += 1
        return result


class RangeIterator:
    """Custom range iterator with step support."""
    
    def __init__(self, start, stop=None, step=1):
        if stop is None:
            self.current = 0
            self.stop = start
        else:
            self.current = start
            self.stop = stop
        self.step = step
    
    def __iter__(self):
        return self
    
    def __next__(self):
        if (self.step > 0 and self.current >= self.stop) or \
           (self.step < 0 and self.current <= self.stop):
            raise StopIteration
        
        result = self.current
        self.current += self.step
        return result


class TreeIterator:
    """In-order iterator for binary tree."""
    
    def __init__(self, root):
        self.root = root
        self.stack = []
        self.current = root
    
    def __iter__(self):
        return self
    
    def __next__(self):
        # Go to leftmost node
        while self.current is not None:
            self.stack.append(self.current)
            self.current = self.current.left
        
        # No more nodes
        if not self.stack:
            raise StopIteration
        
        # Visit node
        node = self.stack.pop()
        self.current = node.right
        return node.value


class TreeNode:
    """Binary tree node for TreeIterator."""
    
    def __init__(self, value, left=None, right=None):
        self.value = value
        self.left = left
        self.right = right


class ChunkedIterator:
    """Iterator that yields items in chunks."""
    
    def __init__(self, iterable, chunk_size):
        self.iterable = iter(iterable)
        self.chunk_size = chunk_size
    
    def __iter__(self):
        return self
    
    def __next__(self):
        chunk = []
        try:
            for _ in range(self.chunk_size):
                chunk.append(next(self.iterable))
            return chunk
        except StopIteration:
            if chunk:
                return chunk
            raise


def fibonacci_sequence(max_count=None):
    """Generator function for Fibonacci sequence."""
    prev, curr = 0, 1
    count = 0
    
    while max_count is None or count < max_count:
        yield prev
        prev, curr = curr, prev + curr
        count += 1
```

**Why**: Demonstrates Python's iterator protocol implementation for various use cases: infinite sequences, tree traversal, and chunked processing. More memory-efficient than loading entire collections.

## Generator Implementation

```python
# PROBLEM: Generate values lazily for memory efficiency
# TIME: O(1) per yield SPACE: O(1) (constant memory)
def read_lines_lazy(filepath):
    """Generator that yields file lines one at a time."""
    with open(filepath, 'r') as file:
        for line in file:
            yield line.strip()


def batch_generator(items, batch_size):
    """Generator that yields items in batches."""
    batch = []
    for item in items:
        batch.append(item)
        if len(batch) == batch_size:
            yield batch
            batch = []
    
    if batch:  # Yield remaining items
        yield batch


def infinite_counter(start=0, step=1):
    """Generator for infinite counter."""
    current = start
    while True:
        yield current
        current += step


def fibonacci_generator():
    """Generator for infinite Fibonacci sequence."""
    prev, curr = 0, 1
    while True:
        yield prev
        prev, curr = curr, prev + curr


def flatten_generator(nested_list):
    """Recursively flatten nested lists using generator."""
    for item in nested_list:
        if isinstance(item, list):
            yield from flatten_generator(item)
        else:
            yield item


def prime_generator():
    """Generator for infinite prime numbers."""
    def is_prime(n):
        if n < 2:
            return False
        if n == 2:
            return True
        if n % 2 == 0:
            return False
        for i in range(3, int(n**0.5) + 1, 2):
            if n % i == 0:
                return False
        return True
    
    num = 2
    while True:
        if is_prime(num):
            yield num
        num += 1


def sliding_window_generator(iterable, window_size):
    """Generator yielding sliding windows over iterable."""
    window = []
    for item in iterable:
        window.append(item)
        if len(window) == window_size:
            yield tuple(window)
            window.pop(0)


def permutation_generator(items):
    """Generator yielding all permutations of items."""
    if len(items) <= 1:
        yield items
    else:
        for i, item in enumerate(items):
            for perm in permutation_generator(items[:i] + items[i+1:]):
                yield [item] + perm


def cycle_generator(iterable):
    """Generator that cycles through iterable infinitely."""
    saved = list(iterable)
    while True:
        for item in saved:
            yield item


class GeneratorPipeline:
    """Chain multiple generators together."""
    
    def __init__(self):
        self.generators = []
    
    def add(self, gen_func):
        """Add generator function to pipeline."""
        self.generators.append(gen_func)
        return self
    
    def execute(self, initial_data):
        """Execute pipeline on initial data."""
        data = initial_data
        for gen_func in self.generators:
            data = gen_func(data)
        return data


# Example usage: Pipeline for data processing
def create_data_pipeline():
    """Create data processing pipeline."""
    pipeline = GeneratorPipeline()
    
    # Add filter step
    pipeline.add(lambda data: (x for x in data if x > 0))
    
    # Add transformation step
    pipeline.add(lambda data: (x * 2 for x in data))
    
    # Add batch step
    pipeline.add(lambda data: batch_generator(data, 5))
    
    return pipeline
```

**Why**: Implements lazy evaluation for memory-efficient processing of large datasets and infinite sequences. Essential for streaming data processing, file handling, and data pipelines.

## Descriptor Patterns

```python
# PROBLEM: Implement attribute access control and validation
# TIME: O(1) for get/set SPACE: O(1) per instance
class ValidatedAttribute:
    """Descriptor that validates assigned values."""
    
    def __init__(self, validator=None, default=None):
        self.validator = validator
        self.default = default
        self.name = None
    
    def __set_name__(self, owner, name):
        """Store attribute name."""
        self.name = f'_{name}'
    
    def __get__(self, obj, objtype=None):
        """Get attribute value."""
        if obj is None:
            return self
        return getattr(obj, self.name, self.default)
    
    def __set__(self, obj, value):
        """Set attribute with validation."""
        if self.validator and not self.validator(value):
            raise ValueError(f"Invalid value: {value}")
        setattr(obj, self.name, value)


class TypedAttribute:
    """Descriptor that enforces type checking."""
    
    def __init__(self, expected_type, default=None):
        self.expected_type = expected_type
        self.default = default
        self.name = None
    
    def __set_name__(self, owner, name):
        self.name = f'_{name}'
    
    def __get__(self, obj, objtype=None):
        if obj is None:
            return self
        return getattr(obj, self.name, self.default)
    
    def __set__(self, obj, value):
        if not isinstance(value, self.expected_type):
            raise TypeError(
                f"Expected {self.expected_type}, got {type(value)}"
            )
        setattr(obj, self.name, value)


class ReadOnlyAttribute:
    """Descriptor for read-only attributes."""
    
    def __init__(self, default=None):
        self.default = default
        self.name = None
    
    def __set_name__(self, owner, name):
        self.name = f'_{name}'
    
    def __get__(self, obj, objtype=None):
        if obj is None:
            return self
        return getattr(obj, self.name, self.default)
    
    def __set__(self, obj, value):
        if hasattr(obj, self.name):
            raise AttributeError(f"{self.name} is read-only")
        setattr(obj, self.name, value)


class CachedAttribute:
    """Descriptor that caches computed values."""
    
    def __init__(self, func):
        self.func = func
        self.name = None
    
    def __set_name__(self, owner, name):
        self.name = f'_{name}'
    
    def __get__(self, obj, objtype=None):
        if obj is None:
            return self
        
        # Check if value is cached
        if hasattr(obj, self.name):
            return getattr(obj, self.name)
        
        # Compute and cache value
        value = self.func(obj)
        setattr(obj, self.name, value)
        return value


class Person:
    """Example class using descriptors."""
    
    # Validated attribute: age must be positive
    age = ValidatedAttribute(
        validator=lambda x: isinstance(x, int) and 0 <= x <= 150,
        default=0
    )
    
    # Typed attribute: name must be string
    name = TypedAttribute(str, default="")
    
    # Read-only attribute: ID cannot be changed
    id = ReadOnlyAttribute()
    
    def __init__(self, name, age, person_id):
        self.name = name
        self.age = age
        self.id = person_id
    
    @CachedAttribute
    def full_info(self):
        """Expensive computation cached by descriptor."""
        # Simulate expensive operation
        import time
        time.sleep(0.1)
        return f"{self.name} (age: {self.age}, id: {self.id})"
```

**Why**: Implements reusable attribute access control, type checking, validation, and caching without subclassing. Core mechanism behind Python's property decorator and many frameworks.

## Property Patterns

```python
# PROBLEM: Implement computed and controlled attributes
# TIME: O(1) to O(n) depending on computation SPACE: O(1)
class Temperature:
    """Class demonstrating property patterns for unit conversion."""
    
    def __init__(self, celsius=0):
        self._celsius = celsius
    
    @property
    def celsius(self):
        """Get temperature in Celsius."""
        return self._celsius
    
    @celsius.setter
    def celsius(self, value):

class LazyProperty:
    """Decorator for lazy-loaded properties."""
    
    def __init__(self, func):
        self.func = func
        self.name = func.__name__

        value = self.func(obj)

class CachedProperty:
    """Property that caches result and allows reset."""
    
    def __init__(self, func):
        self.func = func
        self.name = f'_cached_{func.__name__}'
    
    def __get__(self, obj, objtype=None):
        if obj is None:
            return self
        
        # Return cached value if exists
        if hasattr(obj, self.name):
            return getattr(obj, self.name)
        
        # Compute and cache
        value = self.func(obj)
        setattr(obj, self.name, value)
        return value
    
    def __delete__(self, obj):
        """Clear cached value."""
        if hasattr(obj, self.name):
            delattr(obj, self.name)

class ObservableProperty:
    """Property that notifies observers on change."""
    
    def __init__(self, default=None):
        self.default = default
        self.name = None
        self.callbacks = []
    
    def __set_name__(self, owner, name):
        self.name = f'_{name}'
    
    def __get__(self, obj, objtype=None):
        if obj is None:
            return self
        return getattr(obj, self.name, self.default)
    
    def __set__(self, obj, value):
        old_value = getattr(obj, self.name, self.default)
        setattr(obj, self.name, value)
        
        # Notify observers
        for callback in self.callbacks:
            callback(obj, old_value, value)
    
    def observe(self, callback):
        """Register observer callback."""
        self.callbacks.append(callback)


class Counter:

    def increment(self):
```

```python
# PROBLEM: Implement hash-based index for fast key lookups
# TIME: O(1) average for get/put/delete SPACE: O(n)
class HashIndex:
    """Hash index with collision handling using chaining."""
    
    def __init__(self, capacity=1000, load_factor=0.75):
        self.capacity = capacity
        self.load_factor = load_factor
        self.size = 0
        self.buckets = [[] for _ in range(capacity)]
    
    def _hash(self, key):
        """Compute hash value for key."""
        return hash(key) % self.capacity
    
    def _resize(self):
        """Resize hash table when load factor exceeded."""
        old_buckets = self.buckets
        self.capacity *= 2
        self.buckets = [[] for _ in range(self.capacity)]
        self.size = 0
        
        # Rehash all entries
        for bucket in old_buckets:
            for key, value in bucket:
                self.put(key, value)
    
    def put(self, key, value):
        """Insert or update key-value pair."""
        # Check if resize needed
        if self.size / self.capacity > self.load_factor:
            self._resize()
        
        index = self._hash(key)
        bucket = self.buckets[index]
        
        # Update if key exists
        for i, (k, v) in enumerate(bucket):
            if k == key:
                bucket[i] = (key, value)
                return
        
        # Insert new key
        bucket.append((key, value))
        self.size += 1
    
    def get(self, key, default=None):
        """Retrieve value by key."""
        index = self._hash(key)
        bucket = self.buckets[index]
        
        for k, v in bucket:
            if k == key:
                return v
        
        return default
    
    def delete(self, key):
        """Remove key-value pair."""
        index = self._hash(key)
        bucket = self.buckets[index]
        
        for i, (k, v) in enumerate(bucket):
            if k == key:
                del bucket[i]
                self.size -= 1
                return True
        
        return False
    
    def contains(self, key):
        """Check if key exists."""
        return self.get(key) is not None
    
    def keys(self):
        """Return all keys."""
        for bucket in self.buckets:
            for key, _ in bucket:
                yield key
    
    def values(self):
        """Return all values."""
        for bucket in self.buckets:
            for _, value in bucket:
                yield value
    
    def items(self):
        """Return all key-value pairs."""
        for bucket in self.buckets:
            for key, value in bucket:
                yield key, value


class LinearProbingHashIndex:
    """Hash index using linear probing for collision resolution."""
    
    def __init__(self, capacity=1000):
        self.capacity = capacity
        self.size = 0
        self.keys = [None] * capacity
        self.values = [None] * capacity
        self.deleted = [False] * capacity  # Tombstone markers
    
    def _hash(self, key):
        return hash(key) % self.capacity
    
    def _probe(self, key):
        """Find slot for key using linear probing."""
        index = self._hash(key)
        
        for i in range(self.capacity):
            probe_index = (index + i) % self.capacity
            
            # Empty slot or same key
            if self.keys[probe_index] is None or self.keys[probe_index] == key:
                return probe_index
        
        raise Exception("Hash table full")
    
    def put(self, key, value):
        """Insert or update key-value pair."""
        if self.size >= self.capacity * 0.7:
            self._resize()
        
        index = self._hash(key)
        
        for i in range(self.capacity):
            probe_index = (index + i) % self.capacity
            
            if self.keys[probe_index] is None:
                self.keys[probe_index] = key
                self.values[probe_index] = value
                self.deleted[probe_index] = False
                self.size += 1
                return
            
            elif self.keys[probe_index] == key:
                self.values[probe_index] = value
                return
        
        raise Exception("Hash table full")
    
    def get(self, key, default=None):
        """Retrieve value by key."""
        index = self._hash(key)
        
        for i in range(self.capacity):
            probe_index = (index + i) % self.capacity
            
            if self.keys[probe_index] is None and not self.deleted[probe_index]:
                return default
            
            if self.keys[probe_index] == key:
                return self.values[probe_index]
        
        return default
    
    def delete(self, key):
        """Remove key-value pair (lazy deletion)."""
        index = self._hash(key)
        
        for i in range(self.capacity):
            probe_index = (index + i) % self.capacity
            
            if self.keys[probe_index] is None and not self.deleted[probe_index]:
                return False
            
            if self.keys[probe_index] == key:
                self.keys[probe_index] = None
                self.values[probe_index] = None
                self.deleted[probe_index] = True
                self.size -= 1
                return True
        
        return False
    
    def _resize(self):
        """Resize hash table."""
        old_keys = self.keys
        old_values = self.values
        
        self.capacity *= 2
        self.keys = [None] * self.capacity
        self.values = [None] * self.capacity
        self.deleted = [False] * self.capacity
        self.size = 0
        
        for k, v in zip(old_keys, old_values):
            if k is not None:
                self.put(k, v)


class DatabaseIndex:
    """Hash index for database tables with secondary indexes."""
    
    def __init__(self):
        self.primary_index = HashIndex()
        self.secondary_indexes = {}
    
    def create_secondary_index(self, column):
        """Create secondary index on column."""
        self.secondary_indexes[column] = HashIndex()
    
    def insert(self, primary_key, record):
        """Insert record with primary and secondary indexes."""
        self.primary_index.put(primary_key, record)
        
        # Update secondary indexes
        for column, index in self.secondary_indexes.items():
            if column in record:
                secondary_key = record[column]
                # Store list of primary keys for this secondary value
                existing = index.get(secondary_key, [])
                existing.append(primary_key)
                index.put(secondary_key, existing)
    
    def get_by_primary(self, primary_key):
        """Retrieve record by primary key."""
        return self.primary_index.get(primary_key)
    
    def get_by_secondary(self, column, value):
        """Retrieve records by secondary index."""
        if column not in self.secondary_indexes:
            raise ValueError(f"No secondary index on {column}")
        
        primary_keys = self.secondary_indexes[column].get(value, [])
        return [self.primary_index.get(pk) for pk in primary_keys]
```
