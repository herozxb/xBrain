    </form>

    <script>
        // Manual validation approach when using novalidate
        document.getElementById('myForm').addEventListener('submit', function(e) {
            e.preventDefault();
            
            const email = document.getElementById('email');
            const age = document.getElementById('age');
            
            // Custom validation logic
            if (!email.value) {
                alert('Email is required');
                return;
            }
            
            if (age.value && (age.value < 18 || age.value > 100)) {
                alert('Age must be between 18 and 100');
                return;

                age: age.value
            });
        });
    </script>

</html>
```
**Why**: The `novalidate` attribute allows custom JavaScript validation while bypassing browser's built-in HTML5 validation. Useful when you need complete control over validation logic, error messages, or want to submit data for server-side validation only.

    </div>
    
    <ul id="dropdown-menu" role="menu" hidden>
        <li role="menuitem">Option 1</li>
        <li role="menuitem">Option 2</li>
        <li role="menuitem">Option 3</li>
    </ul>

    <script>
        function setupARIAButton(elementId, onClick) {
            const btn = document.getElementById(elementId);
            if (!btn) return;
            
            // Click handler
            btn.addEventListener('click', (e) => {
                if (btn.getAttribute('aria-disabled') === 'true') return;
                onClick?.(e);
            });
            
            // Keyboard support (Enter and Space)
            btn.addEventListener('keydown', (e) => {
                if (btn.getAttribute('aria-disabled') === 'true') return;
                
                if (e.key === 'Enter' || e.key === ' ') {
                    e.preventDefault();
                    onClick?.(e);
                }
            });
        }
        
        // Toggle button example
        setupARIAButton('customBtn', function() {
            const isPressed = this.getAttribute('aria-pressed') === 'true';
            this.setAttribute('aria-pressed', !isPressed);
            console.log('Toggled:', !isPressed);
        });
        
        // Menu button example
        const menuBtn = document.getElementById('menuBtn');
        const menu = document.getElementById('dropdown-menu');
        
        menuBtn.addEventListener('click', () => {
            const expanded = menuBtn.getAttribute('aria-expanded') === 'true';
            menuBtn.setAttribute('aria-expanded', !expanded);
            menu.hidden = expanded;
        });
    </script>
</body>
</html>
```
**Why**: `role="button"` makes non-button elements (like divs) accessible to assistive technologies. Requires `tabindex="0"` and keyboard handlers (Enter/Space) since native button behavior isn't automatic. Use native `<button>` when possible.

## [HTML Link Rel Preload]
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">

    <!-- Preconnect to external domains -->

    <link rel="preconnect" href="https://cdn.example.com" crossorigin>
    
    <!-- DNS prefetch for non-critical external resources -->
    <link rel="dns-prefetch" href="https://analytics.google.com">
    
    <!-- Prefetch for next page navigation -->
    <link rel="prefetch" href="next-page.html" as="document">
</head>
<body>
    <h1>Optimized Resource Loading</h1>
    
    <!-- Font usage (already preloaded above) -->
    <style>
        @font-face {
            font-family: 'Roboto';
            src: url('fonts/roboto.woff2') format('woff2');
            font-display: swap;
        }
        body { font-family: 'Roboto', sans-serif; }
    </style>
    
    <script>
        // Check preload support
        const supportsPreload = (function() {
            const link = document.createElement('link');
            return link.relList && link.relList.supports && link.relList.supports('preload');
        })();
        
        // Dynamic preload function
        function preloadResource(url, as, crossorigin = false) {
            const link = document.createElement('link');
            link.rel = 'preload';
            link.href = url;
            link.as = as;
            if (crossorigin) link.crossOrigin = 'anonymous';
            document.head.appendChild(link);
            
            console.log(`Preloading: ${url} as ${as}`);
        }
        
        // Example: Preload image on hover
        document.querySelector('a[href="gallery.html"]')?.addEventListener('mouseenter', () => {
            preloadResource('gallery-thumb.jpg', 'image');
        });
        
        // Monitor resource timing
        window.addEventListener('load', () => {
            const entries = performance.getEntriesByType('resource');
            const preloaded = entries.filter(e => e.initiatorType === 'link');
            console.log(`Preloaded resources: ${preloaded.length}`);
        });
    </script>
    
    <!-- Deferred script (will use preloaded file) -->
    <script src="app.js" defer></script>
</body>
</html>
```
**Why**: `rel="preload"` tells browsers to download critical resources early, before they're needed. Improves Core Web Vitals by reducing render-blocking and loading latency. Use `as` attribute to set correct priority and prevent double-fetching.

## [JS Let vs Var]
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Let vs Var Differences</title>
</head>
<body>
    <h2>JavaScript: let vs var</h2>
    <pre id="output"></pre>
    
    <script>
        /* PROBLEM: Understand scoping differences between let and var
           TIME: O(1) SPACE: O(1) */
        
        const output = document.getElementById('output');
        const logs = [];
        function log(msg) { logs.push(msg); console.log(msg); }
        
        // ===== 1. BLOCK SCOPE =====
        log('--- Block Scope ---');
        
        function blockScopeExample() {
            if (true) {
                var varVariable = 'var is function-scoped';
                let letVariable = 'let is block-scoped';
            }
            
            log('var outside block: ' + varVariable); // ✓ Works
            // log('let outside block: ' + letVariable); // ✗ ReferenceError
        }
        blockScopeExample();
        
        // ===== 2. HOISTING =====
        log('\n--- Hoisting ---');
        
        function hoistingExample() {
            // var is hoisted with undefined
            log('var before declaration: ' + varHoisted); // undefined
            var varHoisted = 'now defined';
            
            // let is hoisted but in Temporal Dead Zone
            // log('let before declaration: ' + letHoisted); // ReferenceError
            let letHoisted = 'now defined';
        }
        hoistingExample();
        
        // ===== 3. GLOBAL OBJECT =====
        log('\n--- Global Object ---');
        
        var globalVar = 'attached to window';
        let globalLet = 'NOT attached to window';
        
        log('window.globalVar: ' + (typeof window.globalVar !== 'undefined'));
        log('window.globalLet: ' + (typeof window.globalLet !== 'undefined'));
        
        // ===== 4. REDECLARATION =====
        log('\n--- Redeclaration ---');
        
        var x = 1;
        var x = 2; // ✓ Allowed (silently overwrites)
        log('var redeclared: ' + x);
        
        let y = 1;
        // let y = 2; // ✗ SyntaxError: Identifier 'y' already declared
        
        // ===== 5. LOOP SCOPE =====
        log('\n--- Loop Scope (Classic Interview Question) ---');
        
        // With var - all callbacks reference same variable
        log('var in loop:');
        for (var i = 0; i < 3; i++) {
            setTimeout(() => log('var i = ' + i), 10);
        }
        
        // With let - each iteration has its own binding
        setTimeout(() => {
            log('\nlet in loop:');
            for (let j = 0; j < 3; j++) {
                setTimeout(() => log('let j = ' + j), 10);
            }
        }, 100);
        
        // ===== 6. BEST PRACTICES =====
        log('\n--- Best Practices ---');
        log('✓ Use const by default');
        log('✓ Use let when reassignment needed');
        log('✗ Avoid var in modern code');
        log('✓ Declare at top of scope');
        log('✓ Initialize before use');
        
        // Display all logs
        setTimeout(() => {
            output.textContent = logs.join('\n');
        }, 500);
    </script>
</body>
</html>
```
**Why**: `let` provides block scoping, preventing common bugs in loops and conditionals. `var` is function-scoped and hoisted, leading to confusing behavior. Use `const` by default, `let` when mutation needed, avoid `var` in ES6+ code.
