```python
# PROBLEM: Find the length of the longest substring without repeating characters
# TIME: O(n) SPACE: O(min(m, n)) where m is charset size

def length_of_longest_substring(s: str) -> int:
    """
    Sliding window technique with hash set.
    Expand right pointer, shrink left when duplicate found.
    """
    char_set = set()
    left = 0
    max_len = 0
    
    for right in range(len(s)):
        # Shrink window from left while duplicate exists
        while s[right] in char_set:
            char_set.remove(s[left])
            left += 1
        
        # Add current character and update max
        char_set.add(s[right])
        max_len = max(max_len, right - left + 1)
    
    return max_len
```
**Why:** Classic sliding window pattern. Demonstrates dynamic window adjustment and set-based duplicate detection.

## [Two Sum II - Sorted Array]
```python
# PROBLEM: Find two numbers in sorted array that sum to target
# TIME: O(n) SPACE: O(1)

def two_sum_sorted(numbers: list[int], target: int) -> tuple[int, int]:
    """
    Two pointers approach on sorted array.
    Move left pointer up if sum too small, right down if too large.
    """
    left, right = 0, len(numbers) - 1
    
    while left < right:
        current_sum = numbers[left] + numbers[right]
        
        if current_sum == target:
            return (left + 1, right + 1)  # 1-indexed
        elif current_sum < target:
            left += 1  # Need larger sum
        else:
            right -= 1  # Need smaller sum
    
    return (-1, -1)  # No solution
```
**Why:** Fundamental two-pointer technique. Exploits sorted property to eliminate quadratic search.

## [Subsets with Backtracking]
```python
# PROBLEM: Generate all possible subsets (power set)
# TIME: O(n * 2^n) SPACE: O(n) recursion depth

def subsets(nums: list[int]) -> list[list[int]]:
    """
    Backtracking to explore all combinations.
    At each step: include element or skip it.
    """
    result = []
    
    def backtrack(start: int, path: list[int]):
        # Add current subset to result (copy it!)
        result.append(path[:])
        
        # Try adding each remaining element
        for i in range(start, len(nums)):
            path.append(nums[i])      # Choose
            backtrack(i + 1, path)    # Explore
            path.pop()                # Unchoose (backtrack)
    
    backtrack(0, [])
    return result
```
**Why:** Pure backtracking template. Shows choose-explore-unchoose pattern essential for combinatorial problems.

## [Number of Connected Components]
```python
# PROBLEM: Count connected components in undirected graph
# TIME: O(n * α(n)) ≈ O(n) SPACE: O(n)

class UnionFind:
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.components = n
    
    def find(self, x: int) -> int:
        """Path compression: point nodes directly to root."""
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x: int, y: int) -> bool:
        """Union by rank. Returns True if merged."""
        px, py = self.find(x), self.find(y)
        if px == py:
            return False  # Already connected
        
        # Attach smaller tree under larger
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        self.components -= 1
        return True

def count_components(n: int, edges: list[list[int]]) -> int:
    uf = UnionFind(n)
    for u, v in edges:
        uf.union(u, v)
    return uf.components
```
**Why:** Union-Find with path compression and union by rank. Near-constant time operations for connectivity queries.

## [Longest Increasing Subsequence]
```python
# PROBLEM: Find length of longest strictly increasing subsequence
# TIME: O(n log n) SPACE: O(n)

def length_of_lis(nums: list[int]) -> int:
    """
    Patience sorting approach with binary search.
    tails[i] = smallest tail of LIS of length i+1
    """
    if not nums:
        return 0
    
    tails = []
    
    for num in nums:
        # Binary search for insertion point
        left, right = 0, len(tails)
        while left < right:
            mid = (left + right) // 2
            if tails[mid] < num:
                left = mid + 1
            else:
                right = mid
        
        # Extend or replace
        if left == len(tails):
            tails.append(num)
        else:
            tails[left] = num
    
    return len(tails)
```
**Why:** DP optimization using binary search. Shows how to reduce O(n²) DP to O(n log n) with clever data structure.

## [Course Schedule - Cycle Detection]
```python
# PROBLEM: Determine if all courses can be finished (detect cycle in DAG)
# TIME: O(V + E) SPACE: O(V + E)

from collections import defaultdict, deque

def can_finish(num_courses: int, prerequisites: list[list[int]]) -> bool:
    """
    Kahn's algorithm (topological sort via BFS).
    If we can order all nodes, no cycle exists.
    """
    # Build graph and in-degree count
    graph = defaultdict(list)
    in_degree = [0] * num_courses
    
    for course, prereq in prerequisites:
        graph[prereq].append(course)
        in_degree[course] += 1
    
    # Start with nodes having no prerequisites
    queue = deque([i for i in range(num_courses) if in_degree[i] == 0])
    completed = 0
    
    while queue:
        node = queue.popleft()
        completed += 1
        
        # Reduce in-degree of neighbors
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    
    return completed == num_courses
```
**Why:** Topological sort via BFS. Essential for dependency resolution and detecting cycles in directed graphs.

## [Maximum Path Sum in Binary Tree]
```python
# PROBLEM: Find max path sum where path can go through root
# TIME: O(n) SPACE: O(h) recursion depth

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def max_path_sum(root: TreeNode) -> int:
    """
    Post-order traversal. At each node, compute:
    1. Max path through this node (could be answer)
    2. Max path extending upward to parent
    """
    max_sum = float('-inf')
    
    def dfs(node: TreeNode) -> int:
        nonlocal max_sum
        if not node:
            return 0
        
        # Get max gains from children (ignore negative paths)
        left_gain = max(0, dfs(node.left))
        right_gain = max(0, dfs(node.right))
        
        # Path through this node as "root" of path
        current_path = node.val + left_gain + right_gain
        max_sum = max(max_sum, current_path)
        
        # Return max path extending upward
        return node.val + max(left_gain, right_gain)
    
    dfs(root)
    return max_sum
```
**Why:** Tree traversal with global state. Demonstrates handling paths that don't necessarily go through root.

```python
# PROBLEM: Merge k sorted linked lists into one sorted list
# TIME: O(n log k) SPACE: O(k) where n = total nodes

import heapq
from typing import Optional

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next
    
    # For heap comparison
    def __lt__(self, other):
        return self.val < other.val

def merge_k_lists(lists: list[Optional[ListNode]]) -> Optional[ListNode]:
    """
    Min-heap to always get smallest among k fronts.
    Push next node when current is popped.
    """
    heap = []
    for head in lists:
        if head:
            heapq.heappush(heap, head)
    
    dummy = ListNode()
    current = dummy
    
    while heap:
        smallest = heapq.heappop(heap)
        current.next = smallest
        current = current.next
        
        if smallest.next:
            heapq.heappush(heap, smallest.next)
    
    return dummy.next
```
**Why:** Heap-based k-way merge. Shows how min-heap efficiently maintains order across multiple sorted sequences.

```python
# PROBLEM: Implement prefix tree with insert, search, and startsWith
# TIME: O(m) per operation where m = word length SPACE: O(alphabet_size * total_chars)

class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_word = False

class Trie:
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word: str) -> None:
        """Add word to trie, creating nodes as needed."""
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_word = True
    
    def search(self, word: str) -> bool:
        """Check if complete word exists."""
        node = self._traverse(word)
        return node.is_word if node else False
    
    def starts_with(self, prefix: str) -> bool:
        """Check if any word starts with prefix."""
        return self._traverse(prefix) is not None
    
    def _traverse(self, prefix: str) -> Optional[TrieNode]:
        """Helper: get node at end of prefix, or None."""
        node = self.root
        for char in prefix:
            if char not in node.children:
                return None
            node = node.children[char]
        return node
```
**Why:** Trie fundamentals. Optimal for prefix-based operations, autocomplete, and dictionary implementations.

## [LRU Cache]
```python
# PROBLEM: Design cache with O(1) get and put, evicting least recently used
# TIME: O(1) for get/put SPACE: O(capacity)

from collections import OrderedDict

class LRUCache:
    """
    OrderedDict maintains insertion order.
    move_to_end() marks as recently used.
    """
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = OrderedDict()
    
    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1
        
        # Mark as recently used
        self.cache.move_to_end(key)
        return self.cache[key]
    
    def put(self, key: int, value: int) -> None:
        if key in self.cache:
            # Update and mark recent
            self.cache.move_to_end(key)
        else:
            # Evict LRU if at capacity
            if len(self.cache) >= self.capacity:
                self.cache.popitem(last=False)
        
        self.cache[key] = value
        self.cache.move_to_end(key)
```
**Why:** LRU cache pattern using OrderedDict. Shows hash map + doubly linked list concept in Pythonic way.
