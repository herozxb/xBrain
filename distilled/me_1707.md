## [Maximum Sum Subarray of Size K]

```python
# PROBLEM: Find maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)
def max_sum_subarray_k(nums, k):
    # Edge case: array too short
    if len(nums) < k:
        return 0
    
    # Calculate sum of first window
    window_sum = sum(nums[:k])
    max_sum = window_sum
    
    # Slide window through array, updating sum in O(1)
    for i in range(k, len(nums)):
        # Add new element, remove old element
        window_sum += nums[i] - nums[i - k]
        max_sum = max(max_sum, window_sum)
    
    return max_sum
```
**Why**: Classic sliding window pattern - maintain window sum efficiently instead of recalculating.

## [Container With Most Water]

```python
# PROBLEM: Find two lines that form container holding most water
# TIME: O(n) SPACE: O(1)
def max_area(height):
    # Two pointers from both ends
    left, right = 0, len(height) - 1
    max_water = 0
    
    while left < right:
        # Width times min height determines water capacity
        width = right - left
        container_height = min(height[left], height[right])
        max_water = max(max_water, width * container_height)
        
        # Move pointer with smaller height (bottleneck)
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1
    
    return max_water
```
**Why**: Two pointers greedily eliminate impossible pairs - moving smaller pointer is optimal.

## [Generate Parentheses]

```python
# PROBLEM: Generate all valid combinations of n pairs of parentheses
# TIME: O(4^n / sqrt(n)) SPACE: O(n) recursion depth
def generate_parentheses(n):
    result = []
    
    def backtrack(current, open_count, close_count):
        # Valid combination complete
        if len(current) == 2 * n:
            result.append(current)
            return
        
        # Can add opening parenthesis if under limit
        if open_count < n:
            backtrack(current + '(', open_count + 1, close_count)
        
        # Can add closing if it won't create invalid sequence
        if close_count < open_count:
            backtrack(current + ')', open_count, close_count + 1)
    
    backtrack('', 0, 0)
    return result
```
**Why**: Backtracking with constraints - only add ')' when it maintains validity.

## [Number of Islands]

```python
# PROBLEM: Count connected islands in 2D grid (1=land, 0=water)
# TIME: O(m * n) SPACE: O(m * n) worst case recursion
class UnionFind:
    def __init__(self, grid):
        self.parent = {}
        self.rank = {}
        self.count = 0
        
        # Initialize each land cell as separate component
        for i in range(len(grid)):
            for j in range(len(grid[0])):
                if grid[i][j] == '1':
                    self.parent[(i, j)] = (i, j)
                    self.rank[(i, j)] = 0
                    self.count += 1
    
    def find(self, x):
        # Path compression for efficiency
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        # Union by rank to keep trees balanced
        px, py = self.find(x), self.find(y)
        if px == py:
            return
        
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        self.count -= 1

def num_islands(grid):
    if not grid:
        return 0
    
    uf = UnionFind(grid)
    directions = [(0, 1), (1, 0)]  # Only check right and down
    
    for i in range(len(grid)):
        for j in range(len(grid[0])):
            if grid[i][j] == '1':
                for di, dj in directions:
                    ni, nj = i + di, j + dj
                    if ni < len(grid) and nj < len(grid[0]) and grid[ni][nj] == '1':
                        uf.union((i, j), (ni, nj))
    
    return uf.count
```
**Why**: Union-Find efficiently merges connected components with path compression.

## [Longest Palindromic Subsequence]

```python
# PROBLEM: Find length of longest subsequence that is a palindrome
# TIME: O(n^2) SPACE: O(n^2)
def longest_palindrome_subseq(s):
    n = len(s)
    # dp[i][j] = longest palindromic subseq in s[i:j+1]
    dp = [[0] * n for _ in range(n)]
    
    # Base case: single characters are palindromes of length 1
    for i in range(n):
        dp[i][i] = 1
    
    # Fill table for increasing substring lengths
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            
            if s[i] == s[j]:
                # Match: add 2 to inner substring result
                dp[i][j] = dp[i + 1][j - 1] + 2
            else:
                # No match: take best of excluding either end
                dp[i][j] = max(dp[i + 1][j], dp[i][j - 1])
    
    return dp[0][n - 1]
```
**Why**: DP on intervals - build solutions from smaller substrings to larger ones.

## [Dijkstra's Shortest Path]

```python
# PROBLEM: Find shortest path from source to all nodes in weighted graph
# TIME: O((V + E) log V) SPACE: O(V + E)
import heapq
from collections import defaultdict

def dijkstra(n, edges, source):
    # Build adjacency list
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))  # Undirected graph
    
    # Distance array initialized to infinity
    dist = [float('inf')] * n
    dist[source] = 0
    
    # Priority queue: (distance, node)
    pq = [(0, source)]
    
    while pq:
        d, node = heapq.heappop(pq)
        
        # Skip if we've found better path already
        if d > dist[node]:
            continue
        
        # Relax all neighbors
        for neighbor, weight in graph[node]:
            new_dist = dist[node] + weight
            if new_dist < dist[neighbor]:
                dist[neighbor] = new_dist
                heapq.heappush(pq, (new_dist, neighbor))
    
    return dist
```
**Why**: Greedy approach with min-heap always processes closest unvisited node first.

## [Lowest Common Ancestor]

```python
# PROBLEM: Find lowest common ancestor of two nodes in binary tree
# TIME: O(n) SPACE: O(h) recursion depth
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def lowest_common_ancestor(root, p, q):
    # Base case: reached end or found target
    if not root or root == p or root == q:
        return root
    
    # Search both subtrees
    left = lowest_common_ancestor(root.left, p, q)
    right = lowest_common_ancestor(root.right, p, q)
    
    # If both sides return non-null, current node is LCA
    if left and right:
        return root
    
    # Otherwise return non-null result from subtree
    return left if left else right
```
**Why**: Post-order traversal naturally finds LCA where paths to both nodes meet.

## [Top K Frequent Elements]

```python
# PROBLEM: Find k most frequent elements in array
# TIME: O(n log k) SPACE: O(n)
import heapq
from collections import Counter

def top_k_frequent(nums, k):
    # Count frequency of each element
    count = Counter(nums)
    
    # Min-heap of size k keeps only top k elements
    heap = []
    for num, freq in count.items():
        heapq.heappush(heap, (freq, num))
        if len(heap) > k:
            heapq.heappop(heap)  # Remove least frequent
    
    # Extract elements from heap
    return [num for freq, num in heap]
```
**Why**: Min-heap of size k efficiently maintains top k without sorting all elements.

## [Trie Word Search]

```python
# PROBLEM: Insert and search words with prefix matching
# TIME: O(m) per operation where m = word length SPACE: O(alphabet_size * total_chars)
class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_end = False

class Trie:
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word):
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
    
    def search(self, word):
        node = self.root
        for char in word:
            if char not in node.children:
                return False
            node = node.children[char]
        return node.is_end
    
    def starts_with(self, prefix):
        node = self.root
        for char in prefix:
            if char not in node.children:
                return False
            node = node.children[char]
        return True
```
**Why**: Trie enables O(m) prefix operations critical for autocomplete and dictionary apps.

## [LRU Cache]

```python
# PROBLEM: Implement cache with O(1) get/put and capacity eviction
# TIME: O(1) for get/put SPACE: O(capacity)
from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        # OrderedDict maintains insertion order
        self.cache = OrderedDict()
    
    def get(self, key):
        if key not in self.cache:
            return -1
        
        # Move to end (most recently used)
        self.cache.move_to_end(key)
        return self.cache[key]
    
    def put(self, key, value):
        if key in self.cache:
            # Update and move to end
            self.cache.move_to_end(key)
        else:
            # Evict least recently used if at capacity
            if len(self.cache) >= self.capacity:
                self.cache.popitem(last=False)
        
        self.cache[key] = value
```
**Why**: OrderedDict provides O(1) ordered operations perfect for LRU eviction policy.
