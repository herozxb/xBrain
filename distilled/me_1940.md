## 1. Minimum Spanning Tree Variants

```python
# PROBLEM: Find minimum spanning tree using Prim's algorithm with optimization
# TIME: O(E log V) SPACE: O(V)
import heapq
from typing import List, Tuple

def prim_mst_optimized(graph: List[List[Tuple[int, int]]]) -> Tuple[int, List[Tuple[int, int, int]]]:
    """
    Optimized Prim's algorithm for MST.
    Uses adjacency list with heap for efficient edge selection.
    
    Args:
        graph: Adjacency list where graph[u] = [(v, weight), ...]
    
    Returns:
        Tuple of (total_weight, list of edges (u, v, weight) in MST)
    """
    n = len(graph)
    if n == 0:
        return 0, []
    
    # Track visited vertices
    visited = [False] * n
    # Min-heap: (weight, u, v)
    min_heap = [(0, 0, 0)]  # Start from vertex 0
    mst_edges = []
    total_weight = 0
    
    while min_heap and len(mst_edges) < n:
        weight, u, v = heapq.heappop(min_heap)
        
        if visited[v]:
            continue
        
        # Add vertex to MST
        visited[v] = True
        if u != v:  # Skip the initial dummy edge
            mst_edges.append((u, v, weight))
            total_weight += weight
        
        # Add all edges from new vertex to heap
        for neighbor, edge_weight in graph[v]:
            if not visited[neighbor]:
                heapq.heappush(min_heap, (edge_weight, v, neighbor))
    
    return total_weight, mst_edges
```
**Why**: Optimized Prim's uses O(E log V) time with heap; better for dense graphs than Kruskal.

## 2. Strongly Connected Components

```python
# PROBLEM: Find all strongly connected components in a directed graph
# TIME: O(V + E) SPACE: O(V)
from typing import List, Dict

def kosaraju_scc(graph: Dict[int, List[int]]) -> List[List[int]]:
    """
    Kosaraju's algorithm to find SCCs in directed graph.
    Two-pass DFS: first pass gets finish order, second on transpose.
    
    Args:
        graph: Adjacency list representation
    
    Returns:
        List of SCCs, each SCC is a list of vertices
    """
    n = len(graph)
    visited = [False] * n
    finish_order = []
    
    # First DFS: record finish times
    def dfs1(v: int):
        visited[v] = True
        for u in graph[v]:
            if not visited[u]:
                dfs1(u)
        finish_order.append(v)
    
    for v in range(n):
        if not visited[v]:
            dfs1(v)
    
    # Build transpose graph
    transpose = {v: [] for v in range(n)}
    for v in range(n):
        for u in graph[v]:
            transpose[u].append(v)
    
    # Second DFS on transpose in reverse finish order
    visited = [False] * n
    sccs = []
    
    def dfs2(v: int, component: List[int]):
        visited[v] = True
        component.append(v)
        for u in transpose[v]:
            if not visited[u]:
                dfs2(u, component)
    
    for v in reversed(finish_order):
        if not visited[v]:
            component = []
            dfs2(v, component)
            sccs.append(component)
    
    return sccs
```
**Why**: Kosaraju's two-pass DFS is conceptually simple and optimal O(V + E) time.

## 3. Edmonds-Karp Algorithm

```python
# PROBLEM: Find maximum flow in a flow network
# TIME: O(VE²) SPACE: O(V²)
from collections import deque
from typing import List

def edmonds_karp(capacity: List[List[int]], source: int, sink: int) -> int:
    """
    Edmonds-Karp max flow using BFS to find augmenting paths.
    Guarantees O(VE²) time complexity.
    
    Args:
        capacity: Capacity matrix capacity[u][v]
        source: Source vertex
        sink: Sink vertex
    
    Returns:
        Maximum flow value
    """
    n = len(capacity)
    # Residual capacity matrix
    residual = [row[:] for row in capacity]
    max_flow = 0
    
    def bfs() -> List[int]:
        """Find augmenting path using BFS, return parent array."""
        parent = [-1] * n
        parent[source] = source
        queue = deque([source])
        
        while queue:
            u = queue.popleft()
            for v in range(n):
                if parent[v] == -1 and residual[u][v] > 0:
                    parent[v] = u
                    if v == sink:
                        return parent
                    queue.append(v)
        return parent
    
    # Keep augmenting flow while path exists
    while True:
        parent = bfs()
        
        # No augmenting path found
        if parent[sink] == -1:
            break
        
        # Find bottleneck capacity
        path_flow = float('inf')
        v = sink
        while v != source:
            u = parent[v]
            path_flow = min(path_flow, residual[u][v])
            v = u
        
        # Update residual capacities
        v = sink
        while v != source:
            u = parent[v]
            residual[u][v] -= path_flow
            residual[v][u] += path_flow
            v = u
        
        max_flow += path_flow
    
    return max_flow
```
**Why**: BFS-based augmenting paths ensure shortest path, guaranteeing polynomial time complexity.

## 4. Hopcroft-Karp Algorithm

```python
# PROBLEM: Find maximum bipartite matching
# TIME: O(E√V) SPACE: O(V + E)
from collections import deque
from typing import List, Tuple, Dict

def hopcroft_karp(graph: Dict[int, List[int]], n_left: int, n_right: int) -> List[Tuple[int, int]]:
    """
    Hopcroft-Karp algorithm for maximum bipartite matching.
    Uses BFS to build layers and DFS for augmenting paths.
    
    Args:
        graph: Adjacency list from left partition to right
        n_left: Number of vertices in left partition
        n_right: Number of vertices in right partition
    
    Returns:
        List of matched pairs (left_vertex, right_vertex)
    """
    INF = float('inf')
    # Pair arrays: -1 means unmatched
    pair_left = [-1] * n_left
    pair_right = [-1] * n_right
    dist = [0] * n_left
    
    def bfs() -> bool:
        """Build layered graph using BFS."""
        queue = deque()
        for u in range(n_left):
            if pair_left[u] == -1:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        
        dist_null = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist_null:
                for v in graph[u]:
                    if pair_right[v] == -1:
                        dist_null = dist[u] + 1
                    elif dist[pair_right[v]] == INF:
                        dist[pair_right[v]] = dist[u] + 1
                        queue.append(pair_right[v])
        
        return dist_null != INF
    
    def dfs(u: int) -> bool:
        """Find augmenting path using DFS."""
        for v in graph[u]:
            if pair_right[v] == -1 or (dist[pair_right[v]] == dist[u] + 1 and dfs(pair_right[v])):
                pair_left[u] = v
                pair_right[v] = u
                return True
        dist[u] = INF
        return False
    
    # Main algorithm
    matching = 0
    while bfs():
        for u in range(n_left):
            if pair_left[u] == -1 and dfs(u):
                matching += 1
    
    # Build result
    result = [(u, pair_left[u]) for u in range(n_left) if pair_left[u] != -1]
    return result
```
**Why**: BFS layering enables multiple augmenting paths per phase, achieving O(E√V) complexity.

## 5. Hungarian Algorithm

```python
# PROBLEM: Find optimal assignment in bipartite graph with costs
# TIME: O(n³) SPACE: O(n²)
from typing import List, Tuple

def hungarian_algorithm(cost_matrix: List[List[int]]) -> Tuple[int, List[Tuple[int, int]]]:
    """
    Hungarian algorithm for minimum cost assignment.
    Works on n×n cost matrix.
    
    Args:
        cost_matrix: n×n matrix of assignment costs
    
    Returns:
        Tuple of (total_cost, list of (row, col) assignments)
    """
    n = len(cost_matrix)
    # Potential values
    u = [0] * (n + 1)
    v = [0] * (n + 1)
    # Matching: p[j] = row assigned to column j
    p = [0] * (n + 1)
    # Path for augmenting
    way = [0] * (n + 1)
    
    for i in range(1, n + 1):
        p[0] = i
        j0 = 0  # Virtual column
        minv = [float('inf')] * (n + 1)
        used = [False] * (n + 1)
        
        while p[j0] != 0:
            used[j0] = True
            i0 = p[j0]
            delta = float('inf')
            j1 = 0
            
            for j in range(1, n + 1):
                if not used[j]:
                    cur = cost_matrix[i0 - 1][j - 1] - u[i0] - v[j]
                    if cur < minv[j]:
                        minv[j] = cur
                        way[j] = j0
                    if minv[j] < delta:
                        delta = minv[j]
                        j1 = j
            
            for j in range(n + 1):
                if used[j]:
                    u[p[j]] += delta
                    v[j] -= delta
                else:
                    minv[j] -= delta
            
            j0 = j1
        
        # Augmenting path
        while j0:
            j1 = way[j0]
            p[j0] = p[j1]
            j0 = j1
    
    # Build assignment result
    assignment = []
    total_cost = 0
    for j in range(1, n + 1):
        if p[j] != 0:
            assignment.append((p[j] - 1, j - 1))
            total_cost += cost_matrix[p[j] - 1][j - 1]
    
    return total_cost, assignment
```
**Why**: Optimal O(n³) algorithm for assignment problem; minimum cost bipartite matching.

## 6. Shortest Path in DAG

```python
# PROBLEM: Find shortest paths in directed acyclic graph
# TIME: O(V + E) SPACE: O(V)
from typing import List, Tuple, Dict
from collections import deque

def shortest_path_dag(graph: Dict[int, List[Tuple[int, int]]], source: int, n: int) -> List[int]:
    """
    Single source shortest path in DAG using topological sort.
    Works with negative edges (no negative cycles possible in DAG).
    
    Args:
        graph: Adjacency list with (vertex, weight) tuples
        source: Source vertex
        n: Number of vertices
    
    Returns:
        Distance array, dist[v] = shortest distance from source
    """
    # Step 1: Topological sort using Kahn's algorithm
    in_degree = [0] * n
    for u in range(n):
        for v, _ in graph.get(u, []):
            in_degree[v] += 1
    
    queue = deque([v for v in range(n) if in_degree[v] == 0])
    topo_order = []
    
    while queue:
        u = queue.popleft()
        topo_order.append(u)
        for v, _ in graph.get(u, []):
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)
    
    # Step 2: Relax edges in topological order
    INF = float('inf')
    dist = [INF] * n
    dist[source] = 0
    
    for u in topo_order:
        if dist[u] != INF:
            for v, weight in graph.get(u, []):
                if dist[v] > dist[u] + weight:
                    dist[v] = dist[u] + weight
    
    return dist
```
**Why**: Topological order ensures each edge relaxed once; handles negative weights in DAG.

## 7. All Pairs Shortest Path

```python
# PROBLEM: Find shortest paths between all pairs of vertices
# TIME: O(V³) SPACE: O(V²)
from typing import List

def floyd_warshall(matrix: List[List[int]]) -> List[List[int]]:
    """
    Floyd-Warshall algorithm for all-pairs shortest paths.
    
    Args:
        matrix: Adjacency matrix, matrix[i][j] = weight or INF if no edge
    
    Returns:
        Distance matrix where dist[i][j] = shortest distance from i to j
    """
    n = len(matrix)
    # Initialize distance matrix
    dist = [row[:] for row in matrix]
    
    # Set diagonal to 0
    for i in range(n):
        dist[i][i] = 0
    
    # Floyd-Warshall DP: dist[i][j] via intermediate k
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    return dist

def detect_negative_cycle(dist: List[List[int]]) -> bool:
    """Check if graph has negative cycle (any dist[i][i] < 0)."""
    n = len(dist)
    for i in range(n):
        if dist[i][i] < 0:
            return True
    return False
```
**Why**: Simple O(V³) DP; detects negative cycles; works for dense graphs.

## 8. Johnson's Algorithm

```python
# PROBLEM: Find all-pairs shortest paths with potential negative edges
# TIME: O(V² log V + VE) SPACE: O(V²)
from typing import List, Tuple, Dict
import heapq

def johnson_algorithm(graph: Dict[int, List[Tuple[int, int]]], n: int) -> List[List[int]]:
    """
    Johnson's algorithm for all-pairs shortest paths.
    Reweights edges to remove negatives, then runs Dijkstra from each vertex.
    
    Args:
        graph: Adjacency list with (vertex, weight) tuples
        n: Number of vertices
    
    Returns:
        Distance matrix for all pairs
    """
    # Step 1: Add virtual vertex connected to all others with weight 0
    augmented = {v: edges[:] for v, edges in graph.items()}
    augmented[n] = [(v, 0) for v in range(n)]  # Virtual vertex n
    
    # Step 2: Run Bellman-Ford from virtual vertex
    h = bellman_ford_simple(augmented, n, n + 1)
    if h is None:
        return None  # Negative cycle detected
    
    # Step 3: Reweight edges
    reweighted = {v: [] for v in range(n)}
    for u in range(n):
        for v, w in graph.get(u, []):
            reweighted[u].append((v, w + h[u] - h[v]))
    
    # Step 4: Run Dijkstra from each vertex
    dist_matrix = []
    for source in range(n):
        dist = dijkstra_simple(reweighted, source, n)
        # Re-adjust distances
        for v in range(n):
            if dist[v] != float('inf'):
                dist[v] = dist[v] - h[source] + h[v]
        dist_matrix.append(dist)
    
    return dist_matrix

def bellman_ford_simple(graph: Dict[int, List[Tuple[int, int]]], source: int, n: int) -> List[int]:
    """Bellman-Ford to find shortest distances from source."""
    INF = float('inf')
    dist = [INF] * n
    dist[source] = 0
    
    for _ in range(n - 1):
        for u in range(n):
            if dist[u] != INF:
                for v, w in graph.get(u, []):
                    if dist[v] > dist[u] + w:
                        dist[v] = dist[u] + w
    
    # Check for negative cycle
    for u in range(n):
        for v, w in graph.get(u, []):
            if dist[u] != INF and dist[v] > dist[u] + w:
                return None
    
    return dist

def dijkstra_simple(graph: Dict[int, List[Tuple[int, int]]], source: int, n: int) -> List[int]:
    """Simple Dijkstra for non-negative weights."""
    INF = float('inf')
    dist = [INF] * n
    dist[source] = 0
    heap = [(0, source)]
    visited = [False] * n
    
    while heap:
        d, u = heapq.heappop(heap)
        if visited[u]:
            continue
        visited[u] = True
        
        for v, w in graph.get(u, []):
            if not visited[v] and dist[v] > d + w:
                dist[v] = d + w
                heapq.heappush(heap, (dist[v], v))
    
    return dist
```
**Why**: Combines Bellman-Ford reweighting with Dijkstra for O(VE log V), better than Floyd-Warshall for sparse graphs.

## 9. K-Centers Problem

```python
# PROBLEM: Choose k centers to minimize maximum distance to nearest center
# TIME: O(n²k) SPACE: O(n)
from typing import List, Tuple

def k_centers_greedy(dist: List[List[int]], k: int) -> Tuple[List[int], int]:
    """
    Greedy 2-approximation for k-centers problem.
    Selects centers to minimize maximum distance to nearest center.
    
    Args:
        dist: Distance matrix where dist[i][j] = distance between i and j
        k: Number of centers to select
    
    Returns:
        Tuple of (list of center indices, maximum distance to nearest center)
    """
    n = len(dist)
    if k >= n:
        return list(range(n)), 0
    
    centers = [0]  # Start with first vertex
    # Distance from each vertex to nearest center
    min_dist = [float('inf')] * n
    
    for _ in range(1, k):
        # Update distances to nearest center
        for v in range(n):
            min_dist[v] = min(min_dist[v], dist[v][centers[-1]])
        
        # Find vertex farthest from all centers
        farthest = max(range(n), key=lambda v: min_dist[v])
        centers.append(farthest)
    
    # Calculate final maximum distance
    for v in range(n):
        min_dist[v] = min(min_dist[v], dist[v][centers[-1]])
    
    max_distance = max(min_dist)
    return centers, max_distance
```
**Why**: Greedy 2-approximation is best possible unless P=NP; optimal for k-centers metric.

## 10. Traveling Salesman Approximation

```python
# PROBLEM: Find approximate shortest tour visiting all vertices
# TIME: O(n² log n) SPACE: O(n²)
from typing import List, Tuple

def tsp_mst_approx(dist: List[List[int]]) -> Tuple[int, List[int]]:
    """
    2-approximation for TSP using MST-based approach.
    Works for metric TSP (triangle inequality holds).
    
    Args:
        dist: Distance matrix
    
    Returns:
        Tuple of (tour_length, tour as list of vertices)
    """
    n = len(dist)
    if n <= 1:
        return 0, list(range(n))
    
    # Step 1: Build MST using Prim's
    mst = build_mst(dist)
    
    # Step 2: DFS traversal of MST (preorder)
    tour = []
    visited = [False] * n
    
    def dfs(v: int):
        visited[v] = True
        tour.append(v)
        for u in mst[v]:
            if not visited[u]:
                dfs(u)
    
    dfs(0)
    
    # Step 3: Calculate tour length (return to start)
    tour_length = 0
    for i in range(len(tour)):
        tour_length += dist[tour[i]][tour[(i + 1) % n]]
    
    return tour_length, tour

def build_mst(dist: List[List[int]]) -> List[List[int]]:
    """Build MST adjacency list from distance matrix."""
    n = len(dist)
    import heapq
    
    visited = [False] * n
    mst = [[] for _ in range(n)]
    heap = [(0, 0, 0)]  # (weight, parent, vertex)
    
    while heap:
        weight, parent, v = heapq.heappop(heap)
        if visited[v]:
            continue
        visited[v] = True
        
        if parent != v:
            mst[parent].append(v)
            mst[v].append(parent)
        
        for u in range(n):
            if not visited[u]:
                heapq.heappush(heap, (dist[v][u], v, u))
    
    return mst
```
**Why**: MST-based 2-approximation guaranteed for metric TSP; Christofides improves to 1.5-approximation.
