## [readAsDataURL]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>FileReader readAsDataURL Example</title>
</head>
<body>
    <input type="file" id="fileInput" accept="image/*">
    <div id="preview"></div>

    <script>
        // PROBLEM: Read file as Base64-encoded Data URL
        // TIME: O(n) SPACE: O(n) where n = file size
        
        const fileInput = document.getElementById('fileInput');
        const preview = document.getElementById('preview');

        fileInput.addEventListener('change', (event) => {
            const file = event.target.files[0];
            
            if (!file) return;

            const reader = new FileReader();

            // readAsDataURL converts file to Base64 Data URL
            // Format: data:[mimeType];base64,[encodedData]
            reader.readAsDataURL(file);

            // Success: result contains the Data URL string
            reader.onload = (e) => {
                const img = document.createElement('img');
                img.src = e.target.result; // Base64 image
                img.style.maxWidth = '400px';
                preview.innerHTML = '';
                preview.appendChild(img);
            };

            // Error handling
            reader.onerror = (e) => {
                console.error('Error reading file:', e.target.error);
            };
        });
    </script>
</body>
</html>
```

**Why**: Enables previewing images client-side without server upload. Useful for profile pictures, document previews, and form validation.

## [readAsArrayBuffer]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>FileReader readAsArrayBuffer Example</title>
</head>
<body>
    <input type="file" id="fileInput">
    <div id="output"></div>

    <script>
        // PROBLEM: Read file as raw binary ArrayBuffer
        // TIME: O(n) SPACE: O(n) where n = file size

        const fileInput = document.getElementById('fileInput');
        const output = document.getElementById('output');

        fileInput.addEventListener('change', (event) => {
            const file = event.target.files[0];
            
            if (!file) return;

            const reader = new FileReader();

            // readAsArrayBuffer returns raw binary data
            // Useful for parsing custom file formats, WebAssembly, etc.
            reader.readAsArrayBuffer(file);

            reader.onload = (e) => {
                const arrayBuffer = e.target.result;
                
                // Convert to Uint8Array for byte-level access
                const bytes = new Uint8Array(arrayBuffer);
                
                // Display first 100 bytes in hex
                const hex = Array.from(bytes.slice(0, 100))
                    .map(b => b.toString(16).padStart(2, '0'))
                    .join(' ');
                
                output.innerHTML = `
                    <p><strong>File:</strong> ${file.name}</p>
                    <p><strong>Size:</strong> ${bytes.length} bytes</p>
                    <p><strong>First 100 bytes (hex):</strong></p>
                    <code>${hex}</code>
                `;
            };

            reader.onerror = (e) => {
                console.error('Error reading file:', e.target.error);
            };
        });
    </script>
</body>
</html>
```

**Why**: Essential for binary file processing, custom parsers, WebAssembly modules, and cryptographic operations.

## [FileList]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>FileList Example</title>
</head>
<body>
    <input type="file" id="fileInput" multiple>
    <ul id="fileList"></ul>

    <script>
        // PROBLEM: Iterate over selected files
        // TIME: O(n) SPACE: O(1) where n = number of files

        const fileInput = document.getElementById('fileInput');
        const fileList = document.getElementById('fileList');

        fileInput.addEventListener('change', (event) => {
            // FileList is array-like but NOT an Array
            const files = event.target.files;

            // Clear previous list
            fileList.innerHTML = '';

            // Iterate using for loop
            for (let i = 0; i < files.length; i++) {
                const file = files[i];
                
                const li = document.createElement('li');
                li.innerHTML = `
                    <strong>${file.name}</strong><br>
                    Type: ${file.type || 'unknown'}<br>
                    Size: ${formatBytes(file.size)}<br>
                    Last Modified: ${new Date(file.lastModified).toLocaleString()}
                `;
                fileList.appendChild(li);
            }

            // Convert to Array for array methods
            const fileArray = Array.from(files);
            console.log('Total files:', fileArray.length);
        });

        function formatBytes(bytes) {
            if (bytes === 0) return '0 Bytes';
            const k = 1024;
            const sizes = ['Bytes', 'KB', 'MB', 'GB'];
            const i = Math.floor(Math.log(bytes) / Math.log(k));
            return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
        }
    </script>
</body>
</html>
```

**Why**: Understanding FileList structure is crucial for handling multi-file uploads and validating selections.

## [onunhandledrejection]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>window onunhandledrejection Example</title>
</head>
<body>
    <button id="triggerBtn">Trigger Unhandled Rejection</button>
    <div id="log"></div>

    <script>
        // PROBLEM: Catch unhandled promise rejections globally
        // TIME: O(1) SPACE: O(1)

        // Global handler for unhandled promise rejections
        window.addEventListener('unhandledrejection', (event) => {
            console.error('Unhandled rejection:', event.reason);
            
            // Prevent default browser logging
            event.preventDefault();

            // Log to UI
            const log = document.getElementById('log');
            log.innerHTML += `<p style="color:red">
                <strong>Caught unhandled rejection:</strong> ${event.reason}
            </p>`;
        });

        document.getElementById('triggerBtn').addEventListener('click', () => {
            // This promise rejection has no .catch()
            // It will be caught by the global handler
            new Promise((resolve, reject) => {
                setTimeout(() => {
                    reject('Oops! Forgot to handle this rejection');
                }, 100);
            });
        });

        // Example: fetch without catch
        function fetchWithoutCatch() {
            // If this fails, unhandledrejection fires
            fetch('https://nonexistent-api.example.com/data');
        }
    </script>
</body>
</html>
```

**Why**: Essential for production error monitoring. Logs promise failures that developers forgot to handle with .catch().

## [Promise Unhandled Rejection]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Promise Unhandled Rejection Tracking</title>
</head>
<body>
    <button id="goodBtn">Handled Promise</button>
    <button id="badBtn">Unhandled Rejection</button>
    <div id="output"></div>

    <script>
        // PROBLEM: Track and report unhandled promise rejections
        // TIME: O(1) SPACE: O(1)

        // Track rejection reasons
        const rejectionLog = [];

        window.addEventListener('unhandledrejection', (event) => {
            const rejection = {
                reason: event.reason,
                timestamp: new Date().toISOString(),
                stack: event.reason?.stack || 'No stack trace'
            };

            rejectionLog.push(rejection);

            // Send to error tracking service
            sendToErrorService(rejection);

            // Log to console
            console.error('Unhandled Rejection Detected:', rejection);

            // Prevent default
            event.preventDefault();

            updateOutput();
        });

        function sendToErrorService(error) {
            // Simulated error tracking service
            console.log('Sending to error service:', error);
        }

        function updateOutput() {
            const output = document.getElementById('output');
            output.innerHTML = `
                <h3>Rejection Log (${rejectionLog.length}):</h3>
                <pre>${JSON.stringify(rejectionLog, null, 2)}</pre>
            `;
        }

        // Handled promise - won't trigger global handler
        document.getElementById('goodBtn').addEventListener('click', () => {
            fetch('https://invalid-url.example.com')
                .then(response => response.json())
                .catch(err => {
                    console.log('Handled error:', err.message);
                });
        });

        // Unhandled promise - will trigger global handler
        document.getElementById('badBtn').addEventListener('click', () => {
            async function badAsync() {
                const response = await fetch('https://invalid-url.example.com');
                return response.json();
            }
            // No .catch() - will bubble to global handler
            badAsync();
        });
    </script>
</body>
</html>
```

**Why**: Critical for monitoring app health in production. Helps identify promise chains missing error handlers.

## [RegExp Unicode Flag]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>RegExp Unicode Flag Example</title>
</head>
<body>
    <input type="text" id="input" placeholder="Type text with emojis or Chinese characters">
    <div id="output"></div>

    <script>
        // PROBLEM: Match Unicode characters properly
        // TIME: O(n) SPACE: O(1)

        const input = document.getElementById('input');
        const output = document.getElementById('output');

        input.addEventListener('input', (e) => {
            const text = e.target.value;

            // Without 'u' flag: matches only half of surrogate pairs
            const badRegex = /[😀-🙏]/;
            
            // With 'u' flag: properly handles Unicode code points
            const goodRegex = /[😀-🙏]/u;

            // Match any emoji using Unicode property escapes (requires 'u' flag)
            const emojiRegex = /\p{Emoji}/u;

            // Match Chinese characters (requires 'u' flag)
            const chineseRegex = /\p{Script=Han}/u;

            const results = {
                'Emoji matches': text.match(emojiRegex) || [],
                'Chinese characters': text.match(chineseRegex) || [],
                'Has emoji in range': goodRegex.test(text)
            };

            output.innerHTML = `<pre>${JSON.stringify(results, null, 2)}</pre>`;
        });

        // Example: length with and without Unicode awareness
        function showUnicodeDifferences() {
            const emoji = '😀';
            
            console.log('Without u flag:');
            console.log('  Length:', emoji.length); // 2 (surrogate pairs)
            
            console.log('With u flag and [...spread]:');
            console.log('  Length:', [...emoji].length); // 1 (actual character)
        }
    </script>
</body>
</html>
```

**Why**: Essential for international apps handling emojis, Chinese, Arabic, or any non-ASCII characters correctly.

## [RegExp Sticky Flag]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>RegExp Sticky Flag Example</title>
</head>
<body>
    <div id="output"></div>

    <script>
        // PROBLEM: Match only at current lastIndex position
        // TIME: O(n) SPACE: O(1)

        const text = '123abc456def789ghi';

        // Regular global regex: searches anywhere after lastIndex
        const globalRegex = /\d+/g;

        // Sticky regex: must match EXACTLY at lastIndex
        const stickyRegex = /\d+/y;

        // Demonstrate difference
        let output = '<h3>Global vs Sticky Flag</h3>';

        // Global regex
        globalRegex.lastIndex = 3; // Start at position 3
        const globalMatch = globalRegex.exec(text);
        output += `<p>Global (lastIndex=3): Matched "${globalMatch[0]}" at position ${globalMatch.index}</p>`;

        // Sticky regex - same position
        stickyRegex.lastIndex = 3; // Position 3 is 'a', not a digit
        const stickyMatch1 = stickyRegex.exec(text);
        output += `<p>Sticky (lastIndex=3): ${stickyMatch1 === null ? 'No match (expected)' : stickyMatch1[0]}</p>`;

        // Sticky regex - correct position
        stickyRegex.lastIndex = 0; // Position 0 is '1', a digit
        const stickyMatch2 = stickyRegex.exec(text);
        output += `<p>Sticky (lastIndex=0): Matched "${stickyMatch2[0]}" at position ${stickyMatch2.index}</p>`;

        // Parsing tokens sequentially
        output += '<h3>Sequential Token Parsing</h3>';
        const tokens = '123,456,789';
        const stickyTokenRegex = /\d+|,/y;
        const parsedTokens = [];

        while (true) {
            const match = stickyTokenRegex.exec(tokens);
            if (!match) break;
            parsedTokens.push(match[0]);
        }

        output += `<p>Tokens: ${JSON.stringify(parsedTokens)}</p>`;

        document.getElementById('output').innerHTML = output;
    </script>
</body>
</html>
```

**Why**: Perfect for writing parsers and tokenizers. Ensures matches are contiguous without gaps.

    </script>

## [Date setFullYear]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Date setFullYear Example</title>
</head>
<body>
    <div id="output"></div>

    <script>
        // PROBLEM: Set the year of a date object
        // TIME: O(1) SPACE: O(1)

        let output = '<h3>setFullYear() Method</h3>';

        const date = new Date();
        output += `<p>Original: ${date.toLocaleDateString()}</p>`;

        // setFullYear(year) - set just the year
        const result1 = date.setFullYear(2030);
        output += `<p>After setFullYear(2030): ${date.toLocaleDateString()}</p>`;
        output += `<p>Returns: ${result1} (timestamp)</p>`;

        // setFullYear(year, month) - year and month (0-indexed)
        date.setFullYear(2025, 11); // December 2025
        output += `<p>After setFullYear(2025, 11): ${date.toLocaleDateString()}</p>`;

        // setFullYear(year, month, day) - year, month, and day
        date.setFullYear(2025, 11, 31); // December 31, 2025
        output += `<p>After setFullYear(2025, 11, 31): ${date.toLocaleDateString()}</p>`;

        // Handle year overflow/underflow automatically
        const date2 = new Date(2025, 0, 31); // Jan 31, 2025
        output += `<p>Before: ${date2.toLocaleDateString()}</p>`;
        date2.setFullYear(2025, 1, 31); // Feb 31 doesn't exist
        output += `<p>After setFullYear(2025, 1, 31): ${date2.toLocaleDateString()} (auto-adjusted)</p>`;

        // Leap year handling
        const leapYear = new Date(2024, 1, 29); // Feb 29, 2024 (leap year)
        output += `<p>Leap year date: ${leapYear.toLocaleDateString()}</p>`;
        leapYear.setFullYear(2025); // 2025 is not a leap year
        output += `<p>After setFullYear(2025): ${leapYear.toLocaleDateString()} (adjusted to Mar 1)</p>`;

        // Birth year calculator
        const birthday = new Date();
        birthday.setFullYear(1990, 5, 15);
        const age = new Date().getFullYear() - birthday.getFullYear();
        output += `<p>Person born in 1990 is approximately ${age} years old</p>`;

        document.getElementById('output').innerHTML = output;
    </script>
</body>
</html>
```

**Why**: Commonly used for birthday inputs, expiration dates, and date-of-birth calculations in forms.

## [Date setMonth]

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Date setMonth Example</title>
</head>
<body>
    <div id="output"></div>

    <script>
        // PROBLEM: Set the month of a date (0-11)
        // TIME: O(1) SPACE: O(1)

        let output = '<h3>setMonth() Method</h3>';

        const date = new Date(2025, 0, 15); // Jan 15, 2025
        output += `<p>Original: ${date.toLocaleDateString()}</p>`;

        // setMonth(month) - month is 0-indexed (0=Jan, 11=Dec)
        const result1 = date.setMonth(6); // July
        output += `<p>After setMonth(6): ${date.toLocaleDateString()}</p>`;
        output += `<p>Returns: ${result1} (timestamp)</p>`;

        // setMonth(month, day) - set month and day
        date.setMonth(11, 25); // December 25
        output += `<p>After setMonth(11, 25): ${date.toLocaleDateString()}</p>`;

        // Month overflow wraps to next year
        const date2 = new Date(2025, 11, 1); // Dec 1, 2025
        output += `<p>Before: ${date2.toLocaleDateString()}</p>`;
        date2.setMonth(12); // Month 12 = January of next year
        output += `<p>After setMonth(12): ${date2.toLocaleDateString()} (wrapped to next year)</p>`;

        // Negative months wrap to previous year
        const date3 = new Date(2025, 0, 1); // Jan 1, 2025
        date3.setMonth(-1); // -1 = December of previous year
        output += `<p>setMonth(-1) on Jan: ${date3.toLocaleDateString()} (previous year)</p>`;

        // Practical example: subscription renewal
        const subscription = new Date();
        output += `<p>Today: ${subscription.toLocaleDateString()}</p>`;
        subscription.setMonth(subscription.getMonth() + 1); // Add 1 month
        output += `<p>Next billing: ${subscription.toLocaleDateString()}</p>`;

        // Month name helper
        const monthNames = ['January', 'February', 'March', 'April', 'May', 'June',
                           'July', 'August', 'September', 'October', 'November', 'December'];
        
        function getMonthName(date) {
            return monthNames[date.getMonth()];
        }

        date.setMonth(2); // March
        output += `<p>Month name: ${getMonthName(date)}</p>`;

        document.getElementById('output').innerHTML = output;
    </script>
</body>
</html>
```

**Why**: Common in calendar apps, subscription systems, and any feature needing month-based navigation.
