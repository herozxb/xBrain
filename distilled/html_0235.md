    <script>
        /**
         * Handle circular references when using JSON.stringify()
         * PROBLEM: JSON.stringify() throws error on circular object references
         * TIME: O(n) SPACE: O(n) where n is object depth
         * 
         * Why: Circular references occur when objects reference each other,
         * causing infinite loops during serialization. Common in DOM, graphs,
         * and linked data structures.
         */

        // Demonstrate the circular reference error
        function demonstrateCircular() {
            const obj = { name: 'Alice' };
            obj.self = obj; // Circular reference!
            
            try {
                JSON.stringify(obj);
            } catch (error) {
                document.getElementById('output').textContent = 
                    `Error: ${error.message}\n\n` +
                    `Type: TypeError\n` +
                    `Cause: Converting circular structure to JSON`;
            }
        }

        /**
         * Solution: Custom replacer function to handle circular references
         */
        function demonstrateSolution() {
            const obj = { name: 'Alice', age: 30 };
            obj.self = obj; // Circular reference
            obj.friend = { name: 'Bob' };
            obj.friend.backRef = obj; // Another circular reference

            // Method 1: Using WeakSet to track visited objects
            function safeStringify(obj) {
                const seen = new WeakSet();
                return JSON.stringify(obj, (key, value) => {
                    if (typeof value === 'object' && value !== null) {
                        if (seen.has(value)) {
                            return '[Circular]';
                        }
                        seen.add(value);
                    }
                    return value;
                }, 2);
            }

            const result = safeStringify(obj);
            document.getElementById('output').textContent = 
                `Safe JSON with circular handling:\n${result}`;
        }

        /**
         * Alternative: getCircularReplacer function (reusable)
         */
        const getCircularReplacer = () => {
            const seen = new WeakSet();
            return (key, value) => {
                if (typeof value === 'object' && value !== null) {
                    if (seen.has(value)) {
                        return '[Circular Reference]';
                    }
                    seen.add(value);
                }
                return value;
            };
        };

        // Usage example
        const circularObj = { a: 1 };
        circularObj.b = circularObj;
        console.log('Safe stringify:', 
            JSON.stringify(circularObj, getCircularReplacer()));
    </script>

         *

        // Create a generator function
        function* numberGenerator() {

            yield 2;
            yield 3;

            output += `  value: ${result.value}\n`;
            output += `  done: ${result.done}\n\n`;

                output += '=== Generator Complete ===\n';
                output += 'Click Reset to start over.\n';

        }

            gen = numberGenerator();
            stepCount = 0;
            document.getElementById('output').textContent =

        }

        resetGenerator();

        // Example: Generator with value passing

         * Implement private variables using closures
         * PROBLEM: Create truly private state in JavaScript
         * TIME: O(1) SPACE: O(1)
         * 
         * Why: Before ES2022 private fields (#), closures were the only way
         * to achieve true privacy. Still useful for encapsulation and
         * the module pattern.
         */

        // Counter with private state
        function createCounter() {
            let count = 0; // Private variable
            
            return {
                increment() { return ++count; },
                decrement() { return --count; },
                getCount() { return count; },
                reset() { count = 0; }
            };
        }

        // Bank account with private balance
        function createBankAccount(initialBalance) {
            let balance = initialBalance; // Private
            let transactions = []; // Private
            
            return {
                deposit(amount) {
                    if (amount > 0) {
                        balance += amount;
                        transactions.push({ type: 'deposit', amount, balance });
                        return `Deposited $${amount}. Balance: $${balance}`;
                    }
                    return 'Invalid amount';
                },
                withdraw(amount) {
                    if (amount > 0 && amount <= balance) {
                        balance -= amount;
                        transactions.push({ type: 'withdraw', amount, balance });
                        return `Withdrew $${amount}. Balance: $${balance}`;
                    }
                    return 'Insufficient funds or invalid amount';
                },
                getBalance() { return balance; },
                getHistory() { return [...transactions]; }
            };
        }

        function demonstratePrivateVariables() {
            let output = '=== Closure Private Variables Demo ===\n\n';

            // Counter demo
            output += '--- Counter Example ---\n';
            const counter1 = createCounter();
            const counter2 = createCounter();

            output += `counter1.increment(): ${counter1.increment()}\n`;
            output += `counter2.increment(): ${counter2.increment()}\n`;

            output += `${account.withdraw(30)}\n`;

            output += `\nTransaction History:\n`;

            });

        }
