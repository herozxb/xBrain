## FormData.entries()

```html
<!-- PROBLEM: Iterate through all key-value pairs in FormData -->
<!-- TIME: O(n) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>FormData.entries() Demo</title>
</head>
<body>
    <form id="myForm">
        <input type="text" name="username" value="john_doe">
        <input type="email" name="email" value="john@example.com">
        <input type="number" name="age" value="25">
        <button type="submit">Submit</button>
    </form>
    
    <div id="output"></div>

    <script>
        // Get form element
        const form = document.getElementById('myForm');
        const output = document.getElementById('output');
        
        // Listen for form submission
        form.addEventListener('submit', function(event) {
            event.preventDefault(); // Prevent default form submission
            
            // Create FormData object from form
            const formData = new FormData(form);
            
            // Use entries() to get iterator of all key-value pairs
            // entries() returns an iterator that yields [key, value] pairs
            let displayText = '<h3>Form Data Entries:</h3><ul>';
            
            // Iterate through all entries using for...of loop
            for (const [key, value] of formData.entries()) {
                // Each entry is an array: [propertyName, propertyValue]
                displayText += `<li><strong>${key}:</strong> ${value}</li>`;
            }
            
            displayText += '</ul>';
            output.innerHTML = displayText;
            
            // Alternative: Convert to array for array methods
            const entriesArray = Array.from(formData.entries());
            console.log('All entries as array:', entriesArray);
            // Output: [['username', 'john_doe'], ['email', 'john@example.com'], ['age', '25']]
        });
    </script>
</body>
</html>
```

**Why:** `FormData.entries()` provides an iterator to access all form data as key-value pairs, useful for debugging, logging, or processing form data before submission.

```html
            if (hasEmptyFields) {
                console.warn('Warning: Some fields are empty');
            }

            console.log('Total values:', allValues.length);
            console.log('Values:', allValues);
            // Output: ['Alice', '30', 'New York']
        }
    </script>

</html>
```

**Why:** `FormData.values()` extracts only the values without keys, useful for bulk value processing, validation, or data transformation operations.

## Service Worker Fetch Interception

```html
<!-- PROBLEM: Intercept and cache network requests with Service Worker -->
<!-- TIME: O(1) SPACE: O(cache size) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">

    <h1>Service Worker Fetch Interception</h1>
    <button id="fetchBtn">Fetch Data</button>
    <div id="content"></div>

    <script>
        // Register service worker
        if ('serviceWorker' in navigator) {
            navigator.serviceWorker.register('/sw.js')
                .then(registration => {
                    console.log('Service Worker registered:', registration.scope);
                })
                .catch(error => {
                    console.error('Service Worker registration failed:', error);
                });
        }

        // Fetch button handler
        document.getElementById('fetchBtn').addEventListener('click', async () => {
            try {
                // This request will be intercepted by service worker
                const response = await fetch('https://api.example.com/data');
                const data = await response.json();
                document.getElementById('content').innerHTML = 
                    `<pre>${JSON.stringify(data, null, 2)}</pre>`;
            } catch (error) {
                console.error('Fetch error:', error);
            }
        });
    </script>
</body>
</html>

<!-- sw.js (Service Worker file) -->
<script>
// Service Worker: sw.js
const CACHE_NAME = 'my-cache-v1';
const urlsToCache = [
    '/',
    '/styles.css',
    '/script.js'
];

// Install event - cache resources
self.addEventListener('install', event => {
    event.waitUntil(
        caches.open(CACHE_NAME)
            .then(cache => cache.addAll(urlsToCache))
    );
});

// Fetch event - intercept network requests
self.addEventListener('fetch', event => {
    event.respondWith(
        // Try cache first
        caches.match(event.request)
            .then(cachedResponse => {
                if (cachedResponse) {
                    console.log('Serving from cache:', event.request.url);
                    return cachedResponse;
                }
                
                // If not in cache, fetch from network
                return fetch(event.request)
                    .then(networkResponse => {
                        // Clone response for caching
                        const responseToCache = networkResponse.clone();
                        
                        // Cache the fetched resource
                        caches.open(CACHE_NAME)
                            .then(cache => {
                                cache.put(event.request, responseToCache);
                            });
                        
                        return networkResponse;
                    });
            })
            .catch(error => {
                console.error('Fetch failed:', error);
                // Return offline fallback if available
                return caches.match('/offline.html');
            })
    );
});
</script>
```

**Why:** Service Worker fetch interception enables offline functionality, caching strategies, and request/response modification for better performance and reliability.

## Push API

```html
<!-- PROBLEM: Send push notifications to users -->
<!-- TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Push API Demo</title>
</head>
<body>
    <h1>Push Notification Demo</h1>
    <button id="subscribeBtn">Subscribe to Push Notifications</button>
    <button id="unsubscribeBtn">Unsubscribe</button>
    <div id="status"></div>

    <script>
        // Check for Push API support
        if (!('PushManager' in window)) {
            document.getElementById('status').textContent = 
                'Push API not supported in this browser';
        }

        const subscribeBtn = document.getElementById('subscribeBtn');
        const unsubscribeBtn = document.getElementById('unsubscribeBtn');
        const statusDiv = document.getElementById('status');

        // Subscribe to push notifications
        subscribeBtn.addEventListener('click', async () => {
            try {
                // Request notification permission
                const permission = await Notification.requestPermission();
                
                if (permission !== 'granted') {
                    statusDiv.textContent = 'Notification permission denied';
                    return;
                }

                // Register service worker
                const registration = await navigator.serviceWorker.register('/sw.js');
                
                // Get push subscription
                const subscription = await registration.pushManager.subscribe({
                    userVisibleOnly: true,
                    applicationServerKey: urlBase64ToUint8Array(
                        'YOUR_PUBLIC_VAPID_KEY_HERE'
                    )
                });

                // Send subscription to server
                await fetch('/api/subscribe', {
                    method: 'POST',
                    body: JSON.stringify(subscription),
                    headers: { 'Content-Type': 'application/json' }
                });

                statusDiv.textContent = 'Successfully subscribed to push notifications!';
                console.log('Subscription:', subscription);
            } catch (error) {
                statusDiv.textContent = 'Subscription failed: ' + error.message;
            }
        });

        // Unsubscribe from push notifications
        unsubscribeBtn.addEventListener('click', async () => {
            try {
                const registration = await navigator.serviceWorker.ready;
                const subscription = await registration.pushManager.getSubscription();
                
                if (subscription) {
                    await subscription.unsubscribe();
                    statusDiv.textContent = 'Successfully unsubscribed';
                }
            } catch (error) {
                statusDiv.textContent = 'Unsubscribe failed: ' + error.message;
            }
        });

        // Helper function to convert VAPID key
        function urlBase64ToUint8Array(base64String) {
            const padding = '='.repeat((4 - base64String.length % 4) % 4);
            const base64 = (base64String + padding)
                .replace(/\-/g, '+')
                .replace(/_/g, '/');

            const rawData = window.atob(base64);
            const outputArray = new Uint8Array(rawData.length);

            for (let i = 0; i < rawData.length; ++i) {
                outputArray[i] = rawData.charCodeAt(i);
            }
            return outputArray;
        }
    </script>
</body>
</html>
```

## Notification.requestPermission()

```html
<!-- PROBLEM: Request user permission for notifications -->
<!-- TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Notification Permission Demo</title>
</head>
<body>
    <h1>Notification Permission Request</h1>
    <button id="requestBtn">Request Permission</button>
    <button id="checkBtn">Check Permission Status</button>
    <div id="result"></div>

    <script>
        const resultDiv = document.getElementById('result');

        // Request notification permission
        document.getElementById('requestBtn').addEventListener('click', async () => {
            try {
                // Check if notifications are supported
                if (!('Notification' in window)) {
                    resultDiv.innerHTML = '<p style="color: red;">Notifications not supported</p>';
                    return;
                }

                // Request permission using Promise-based API
                const permission = await Notification.requestPermission();
                
                // Handle the permission result
                handlePermissionResult(permission);
            } catch (error) {
                resultDiv.innerHTML = `<p style="color: red;">Error: ${error.message}</p>`;
            }
        });

        // Check current permission status
        document.getElementById('checkBtn').addEventListener('click', () => {
            if ('Notification' in window) {
                resultDiv.innerHTML = `
                    <p>Current Permission Status: <strong>${Notification.permission}</strong></p>
                    <ul>
                        <li>granted - User allowed notifications</li>
                        <li>denied - User blocked notifications</li>
                        <li>default - User hasn't decided yet</li>
                    </ul>
                `;
            } else {
                resultDiv.innerHTML = '<p>Notifications not supported</p>';
            }
        });

        // Handle permission result
        function handlePermissionResult(permission) {
            let message = '';
            
            switch (permission) {
                case 'granted':
                    message = '<p style="color: green;">✓ Permission granted! You can send notifications.</p>';
                    // Show test notification
                    new Notification('Permission Granted!', {
                        body: 'You will now receive notifications',
                        icon: '/icon.png'
                    });
                    break;
                    
                case 'denied':
                    message = '<p style="color: red;">✗ Permission denied. User blocked notifications.</p>';
                    break;
                    
                case 'default':
                    message = '<p style="color: orange;">? Permission dismissed. User closed the dialog.</p>';
                    break;
            }
            
            resultDiv.innerHTML = message;
        }

        // Check permission on page load
        window.addEventListener('load', () => {
            if ('Notification' in window) {
                console.log('Initial permission:', Notification.permission);
            }
        });
    </script>
</body>
</html>
```

**Why:** `Notification.requestPermission()` is required before sending notifications, respecting user privacy and giving them control over notification preferences.

## Notification.show()

```html
<!-- PROBLEM: Display desktop notifications to users -->
<!-- TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Show Notification Demo</title>
</head>
<body>
    <h1>Desktop Notifications</h1>
    <button id="notifyBtn">Show Notification</button>
    <button id="customBtn">Show Custom Notification</button>
    <div id="status"></div>

    <script>
        // Show basic notification
        document.getElementById('notifyBtn').addEventListener('click', () => {
            showNotification('Basic Notification', {
                body: 'This is a simple notification',
                icon: '/icon.png'
            });
        });

        // Show custom notification with all options
        document.getElementById('customBtn').addEventListener('click', () => {
            showNotification('Custom Notification', {
                body: 'This notification has all features',
                icon: '/icon.png',
                badge: '/badge.png',
                image: '/large-image.jpg',
                tag: 'unique-notification-id',
                requireInteraction: true,
                actions: [
                    { action: 'open', title: 'Open', icon: '/open.png' },
                    { action: 'close', title: 'Close', icon: '/close.png' }
                ],
                vibrate: [200, 100, 200],
                timestamp: Date.now()
            });
        });

        // Generic function to show notification
        function showNotification(title, options = {}) {
            // Check permission first
            if (Notification.permission !== 'granted') {
                document.getElementById('status').textContent = 
                    'Please grant notification permission first';
                return;
            }

            // Create and show notification
            const notification = new Notification(title, options);

            // Handle notification click
            notification.onclick = (event) => {
                event.preventDefault();
                console.log('Notification clicked');
                
                // Open or focus window
                window.open('https://example.com', '_blank');
                notification.close();
            };

            // Handle notification close
            notification.onclose = (event) => {
                console.log('Notification closed');
            };

            // Handle errors
            notification.onerror = (event) => {
                console.error('Notification error:', event);
            };

            // Handle action button clicks
            notification.onaction = (event) => {
                console.log('Action clicked:', event.action);
                if (event.action === 'open') {
                    window.open('https://example.com');
                }
            };

            document.getElementById('status').textContent = 
                'Notification displayed successfully';
        }

        // Request permission on page load
        if ('Notification' in window && Notification.permission === 'default') {
            Notification.requestPermission();
        }
    </script>
</body>
</html>
```

**Why:** `new Notification()` displays system-level notifications with rich content (images, actions, badges), enabling engaging user communication even when the browser is not focused.
