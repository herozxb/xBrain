```html
<script>
// Lazy load Vimeo videos for better performance
document.addEventListener('DOMContentLoaded', () => {
  const iframes = document.querySelectorAll('iframe[src*="vimeo.com"]');
  
  // Use Intersection Observer to load videos only when visible
  const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        const iframe = entry.target;
        if (iframe.dataset.src) {
          iframe.src = iframe.dataset.src;
          observer.unobserve(iframe);
        }
      }
    });
  }, { rootMargin: '100px' });
  
  iframes.forEach(iframe => {
    if (iframe.loading === 'lazy') {
      iframe.dataset.src = iframe.src;
      iframe.src = '';
      observer.observe(iframe);
    }
  });
});
</script>
```

**Why**: Responsive Vimeo embeds with lazy loading improve page load performance while maintaining proper aspect ratios across all devices.

```html
<!-- PROBLEM: Detect when mouse button is pressed for drag interactions -->
<!-- TIME: O(1) per event SPACE: O(1) -->

<script>
const box = document.getElementById('draggable-box');
const coords = document.getElementById('coordinates');
let isDragging = false;
let offsetX, offsetY;

// mousedown: initiate drag
box.addEventListener('mousedown', (e) => {
  isDragging = true;
  
  // Calculate offset from mouse to element corner
  const rect = box.getBoundingClientRect();
  offsetX = e.clientX - rect.left;
  offsetY = e.clientY - rect.top;
  
  // Add dragging class
  box.classList.add('dragging');
  
  // Position at current location
  box.style.left = `${e.clientX - offsetX}px`;
  box.style.top = `${e.clientY - offsetY}px`;
  
  coords.textContent = `Dragging started at (${e.clientX}, ${e.clientY})`;
  
  // Prevent text selection during drag
  e.preventDefault();
});

// mousemove: continue drag (on document for smooth tracking)
document.addEventListener('mousemove', (e) => {
  if (!isDragging) return;
  
  box.style.left = `${e.clientX - offsetX}px`;
  box.style.top = `${e.clientY - offsetY}px`;
  
  coords.textContent = `Position: (${e.clientX}, ${e.clientY})`;
});

// mouseup: end drag
document.addEventListener('mouseup', (e) => {
  if (isDragging) {
    isDragging = false;
    box.classList.remove('dragging');
    coords.textContent = `Dropped at (${e.clientX}, ${e.clientY})`;
  }
});

// Detect which mouse button was pressed
box.addEventListener('mousedown', (e) => {
  switch(e.button) {
    case 0: console.log('Left button'); break;
    case 1: console.log('Middle button'); break;
    case 2: console.log('Right button'); break;
  }
});
</script>
```

**Why**: The mousedown event enables interactive drag-and-drop interfaces, custom context menus, and gesture detection for enhanced user experiences.

```html
  color: #dc3545;
  font-size: 0.875rem;
}
</style>

<script>
const form = document.getElementById('mixedValidation');

// Custom validation logic
function validateEmail(email) {
  const re = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return re.test(email);
}

function validateAge(age) {
  const num = parseInt(age);
  return !isNaN(num) && num >= 18 && num <= 120;
}

// Real-time validation
document.getElementById('emailInput').addEventListener('blur', (e) => {
  const error = document.getElementById('emailError');
  if (e.target.value && !validateEmail(e.target.value)) {
    e.target.classList.add('invalid');
    error.textContent = 'Please enter a valid email';
  } else {
    e.target.classList.remove('invalid');
    error.textContent = '';
  }
});

// Submit handler with custom validation
form.addEventListener('submit', (e) => {
  e.preventDefault();
  
  const email = document.getElementById('emailInput');
  const age = document.getElementById('ageInput');
  let isValid = true;
  
  // Custom validation check
  if (!validateEmail(email.value)) {
    email.classList.add('invalid');
    document.getElementById('emailError').textContent = 'Invalid email format';
    isValid = false;
  }
  
  // Check validity using Constraint Validation API
  if (!age.checkValidity() && age.value) {
    age.classList.add('invalid');
    document.getElementById('ageError').textContent = 
      `Age must be between ${age.min} and ${age.max}`;
    isValid = false;
  }
  
  if (isValid) {
    console.log('Form submitted!', {
      email: email.value,
      age: age.value
    });
    alert('Form submitted successfully!');
  }
});

// Check if form would be valid without novalidate
document.getElementById('submitBtn').addEventListener('click', () => {
  console.log('Form checkValidity:', form.checkValidity());
  console.log('Form reportValidity:', form.reportValidity());
});
</script>
```

**Why**: The novalidate attribute enables custom validation logic while still leveraging HTML5 validation APIs programmatically for flexibility in form handling.

## JavaScript Generators

```html
<!-- PROBLEM: Create iterators with lazy evaluation and pausable execution -->
<!-- TIME: O(1) per yield SPACE: O(1) per generator instance -->
<div id="generator-demo">
  <h3>Generator Examples</h3>
  <button id="generateNumbers">Generate Numbers</button>
  <button id="generateFibonacci">Fibonacci Sequence</button>
  <button id="generatePaginated">Paginated Data</button>
  <div id="output"></div>
</div>

<script>
// Basic generator function (uses function*)
function* numberGenerator(max) {
  for (let i = 1; i <= max; i++) {
    // yield pauses execution and returns a value
    yield i;
  }
  // Generator automatically returns { done: true } after loop
}

// Infinite Fibonacci generator
function* fibonacciGenerator() {
  let prev = 0, curr = 1;
  while (true) {
    yield curr;
    [prev, curr] = [curr, prev + curr];
  }
}

// Generator for paginated API data
function* paginateData(data, pageSize) {
  for (let i = 0; i < data.length; i += pageSize) {
    // Simulate async delay (in real app, would be fetch)
    yield {
      page: Math.floor(i / pageSize) + 1,
      data: data.slice(i, i + pageSize),
      hasMore: i + pageSize < data.length
    };
  }
}

// Generator with return value
function* countdown(n) {
  while (n > 0) {
    yield n;
    n--;
  }
  return 'Liftoff!'; // Final return value
}

// Demo handlers
const output = document.getElementById('output');

document.getElementById('generateNumbers').addEventListener('click', () => {
  const gen = numberGenerator(5);
  const results = [];
  
  // Manual iteration
  let result = gen.next();
  while (!result.done) {
    results.push(result.value);
    result = gen.next();
  }
  
  output.innerHTML = `
    <p><strong>Numbers 1-5:</strong> ${results.join(', ')}</p>
    <p><strong>Using spread:</strong> ${[...numberGenerator(5)].join(', ')}</p>
    <p><strong>Using for...of:</strong></p>
    <pre>${(() => {
      let str = '';
      for (const num of numberGenerator(3)) str += num + ' ';
      return str;
    })()}</pre>
  `;
});

document.getElementById('generateFibonacci').addEventListener('click', () => {
  const fib = fibonacciGenerator();
  const firstTen = [];
  
  for (let i = 0; i < 10; i++) {
    firstTen.push(fib.next().value);
  }
  
  output.innerHTML = `
    <p><strong>First 10 Fibonacci:</strong> ${firstTen.join(', ')}</p>
    <p><em>Note: Generator can produce infinite values</em></p>
  `;
});

document.getElementById('generatePaginated').addEventListener('click', () => {
  // Sample data
  const allData = Array.from({ length: 25 }, (_, i) => `Item ${i + 1}`);
  const pager = paginateData(allData, 5);
  
  let html = '<p><strong>Paginated Results:</strong></p>';
  
  for (const page of pager) {
    html += `
      <div style="margin: 0.5rem 0; padding: 0.5rem; background: #f0f0f0; border-radius: 4px;">
        Page ${page.page}: ${page.data.join(', ')}
        ${page.hasMore ? '(more available)' : '(last page)'}
      </div>
    `;
  }
  
  output.innerHTML = html;
});

// Generator delegation with yield*
function* innerGen() {
  yield 'a';
  yield 'b';
}

function* outerGen() {
  yield 1;
  yield* innerGen(); // Delegates to inner generator
  yield 2;
}

console.log('Delegation:', [...outerGen()]); // [1, 'a', 'b', 2]

// Generator as async-like control flow
function* taskRunner() {
  const task1 = yield 'Starting task 1';
  console.log('Task 1 result:', task1);
  
  const task2 = yield 'Starting task 2';
  console.log('Task 2 result:', task2);
  
  return 'All tasks complete';
}
</script>
```

**Why**: Generators enable lazy evaluation of infinite sequences, memory-efficient iteration, and elegant control flow patterns without loading entire datasets into memory.

```html
<!-- Custom Element using Shadow DOM -->
<script>
class ShadowCard extends HTMLElement {
  constructor() {
    super();
    
    // Create shadow root (open mode allows JS access from outside)
    this.shadow = this.attachShadow({ mode: 'open' });
    
    // Clone template content
    const template = document.getElementById('card-template');
    this.shadow.appendChild(template.content.cloneNode(true));
    
    // Add event listener within shadow DOM
    this.shadow.getElementById('inner-btn').addEventListener('click', () => {
      this.dispatchEvent(new CustomEvent('card-click', {
        bubbles: true,
        composed: true, // Allows event to cross shadow boundary
        detail: { message: 'Button in shadow DOM clicked' }
      }));
    });
  }
  
  // Observed attributes
  static get observedAttributes() {
    return ['title'];
  }
  
  attributeChangedCallback(name, oldValue, newValue) {
    if (name === 'title' && this.shadow) {
      this.shadow.querySelector('h3').textContent = newValue;
    }
  }
}

// Register custom element
customElements.define('shadow-card', ShadowCard);

// Create and append instance
const host = document.getElementById('shadow-host');
const card = document.createElement('shadow-card');
card.setAttribute('title', 'My Custom Card');
card.innerHTML = '<span slot="content" class="highlight">Slotted content!</span>';
host.appendChild(card);

// Listen for events from shadow DOM
card.addEventListener('card-click', (e) => {
  console.log('Event from shadow:', e.detail);
  alert('Shadow DOM button clicked!');
});

// Demonstrate style isolation
const demo = document.createElement('div');
demo.innerHTML = `
  <h3>Demo: Style Isolation</h3>
  <p>This H3 has global styles (if any)</p>
  <shadow-card title="Isolated Component">
    <span slot="content">Styles don't leak in or out!</span>
  </shadow-card>
  <p>Global paragraph is unaffected by shadow styles</p>
`;
host.appendChild(demo);

// Different shadow modes
// open: shadowRoot accessible via element.shadowRoot
// closed: shadowRoot returns null (used internally)

console.log('Open shadow root:', card.shadowRoot); // Returns ShadowRoot
</script>
```
