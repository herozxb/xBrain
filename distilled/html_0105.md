## js_date_getSeconds

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Date getSeconds()</title>

<body>
  <h1>Get Current Seconds</h1>
  <p id="output"></p>

  <script>
    // PROBLEM: Extract seconds (0-59) from a Date object
    // TIME: O(1) SPACE: O(1)
    
    /**
     * Returns the seconds value from a given date.
     * @param {Date} date - The date object to extract seconds from
     * @returns {number} Seconds (0-59)
     */
    function getSeconds(date) {
      // getSeconds() returns an integer from 0 to 59
      // representing the seconds in the specified date
      return date.getSeconds();
    }

    // Example usage
    const now = new Date();
    const seconds = getSeconds(now);
    
    document.getElementById('output').textContent = 
      `Current seconds: ${seconds}`;
    
    console.log(`Seconds: ${seconds}`); // 0-59
  </script>
</body>
</html>
```

**Why**: `getSeconds()` is useful for time-sensitive operations, timers, and displaying precise time components in UI clocks.

```html
          'Cannot calculate square root of negative number';
      } else {
        document.getElementById('output').textContent = 
          `√${input} = ${result}`;
      }
    }

    // Initialize
    calculate();
  </script>

```

**Why**: `Math.sqrt()` is fundamental for geometry calculations, physics simulations, distance computations, and statistical analysis.

## js_math_cbrt

```html
  <input type="number" id="input" placeholder="Enter a number" value="27">
  <button onclick="calculate()">Calculate</button>
  <p id="output"></p>

  <script>
    // PROBLEM: Calculate the cube root of a number
    // TIME: O(1) SPACE: O(1)
    
    /**
     * Calculates the cube root of a number.
     * @param {number} num - The number to calculate cube root for
     * @returns {number} Cube root (works with negative numbers)
     */
    function calculateCbrt(num) {
      // Math.cbrt() returns the cube root of a number
      // Unlike sqrt, it handles negative numbers correctly
      // ∛(-8) = -2
      return Math.cbrt(num);
    }

    function calculate() {
      const input = parseFloat(document.getElementById('input').value);
      const result = calculateCbrt(input);
      
      document.getElementById('output').textContent = 
        `∛${input} = ${result.toFixed(4)}`;
    }

    // Initialize
    calculate();
    
    // Example outputs:
    console.log(Math.cbrt(27));   // 3
    console.log(Math.cbrt(-8));   // -2
    console.log(Math.cbrt(0));    // 0
  </script>
</body>
</html>
```

**Why**: `Math.cbrt()` is useful for volume calculations, 3D geometry, and solving cubic equations. Unlike `sqrt()`, it handles negative numbers correctly.

```html
      updateOutput();
    }

    // Initialize
    updateOutput();
    
    console.log('Before clear:', userMap.size); // 3
    userMap.clear();
    console.log('After clear:', userMap.size);  // 0
  </script>
```

**Why**: `Map.clear()` efficiently resets a Map to empty state, useful for caching systems, state management, and resetting data structures.

## js_map_size

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Map.size</title>
</head>
<body>
  <h1>Map Size Tracker</h1>
  <input type="text" id="key" placeholder="Key">
  <input type="text" id="value" placeholder="Value">
  <button onclick="addEntry()">Add</button>
  <p id="output"></p>

  <script>
    // PROBLEM: Get the number of entries in a Map
    // TIME: O(1) SPACE: O(1)
    
    /**
     * Returns the number of key-value pairs in a Map.
     * @param {Map} map - The Map to check
     * @returns {number} The count of entries
     */
    function getMapSize(map) {
      // size is a property (not a method) that returns
      // the number of elements in the Map
      return map.size;
    }

    const dataMap = new Map();

    function updateOutput() {
      const size = getMapSize(dataMap);
      document.getElementById('output').textContent = 
        `Map contains ${size} ${size === 1 ? 'entry' : 'entries'}`;
    }

    function addEntry() {
      const key = document.getElementById('key').value;
      const value = document.getElementById('value').value;
      
      if (key && value) {
        dataMap.set(key, value);
        updateOutput();
        // Clear inputs
        document.getElementById('key').value = '';
        document.getElementById('value').value = '';
      }
    }

    // Initialize
    updateOutput();
    
    // Example
    const myMap = new Map([['a', 1], ['b', 2], ['c', 3]]);
    console.log(myMap.size); // 3
  </script>
</body>
</html>
```

**Why**: `Map.size` provides O(1) access to entry count, essential for validation, iteration limits, and monitoring data structure capacity.

```html
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">

  <script>
    // PROBLEM: Identify the type of an error
    // TIME: O(1) SPACE: O(1)
    
    /**
     * Returns the name property of an error.
     * @param {Error} error - The error object
     * @returns {string} The error name/type
     */
    function getErrorName(error) {
      // The name property identifies the error type
      // Common values: Error, TypeError, ReferenceError, 
      // SyntaxError, RangeError, URIError
      return error.name;
    }

    function testErrors() {
      const errors = [
        new Error('Generic error'),
        new TypeError('Type mismatch'),
        new ReferenceError('Variable not found'),
        new SyntaxError('Invalid syntax'),
        new RangeError('Out of range')
      ];
      
      let output = '';
      errors.forEach(err => {
        output += `${getErrorName(err)}: ${err.message}\n`;
      });
      
      document.getElementById('output').textContent = output;
    }

    // Example: Handle different error types
    function safeOperation(value) {
      try {
        if (typeof value !== 'number') {
          throw new TypeError('Expected a number');
        }
        return value * 2;
      } catch (error) {
        console.log(`Caught ${error.name}: ${error.message}`);
        return null;
      }
    }

    safeOperation('string'); // Logs: Caught TypeError: Expected a number
  </script>
</body>
</html>
```

**Why**: `Error.name` enables type-based error handling, logging categorization, and building robust error recovery systems.

## js_error_toString

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Error.toString()</title>
</head>
<body>
  <h1>Error String Representation</h1>
  <button onclick="demonstrate()">Show Error Strings</button>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Get string representation of an error
    // TIME: O(1) SPACE: O(n) where n is message length
    
    /**
     * Returns the string representation of an error.
     * @param {Error} error - The error object
     * @returns {string} String in format "name: message"
     */
    function errorToString(error) {
      // toString() returns a string in the format:
      // "ErrorName: ErrorMessage"
      return error.toString();
    }

    function demonstrate() {
      const errors = [
        new Error('Something went wrong'),
        new TypeError('Cannot read property of undefined'),
        new RangeError('Array index out of bounds')
      ];
      
      let output = 'Error.toString() Demo:\n\n';
      
      errors.forEach(err => {
        output += `toString(): ${errorToString(err)}\n`;
        output += `name: ${err.name}\n`;
        output += `message: ${err.message}\n`;
        output += '---\n';
      });
      
      document.getElementById('output').textContent = output;
    }

    // Practical example: Logging errors
    function logError(error) {
      const timestamp = new Date().toISOString();
      console.error(`[${timestamp}] ${error.toString()}`);
    }

    try {
      JSON.parse('invalid json');
    } catch (e) {
      logError(e); // [2026-02-28T17:05:00.000Z] SyntaxError: Unexpected token...
    }
  </script>
</body>
</html>
```

**Why**: `Error.toString()` provides a standardized error format for logging, debugging, and displaying user-friendly error messages.

## css_env_function

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>CSS env() Function</title>

    /* PROBLEM: Access browser-defined environment variables
       TIME: O(1) SPACE: O(1) */
    
    body {
      margin: 0;
      padding: 0;
      font-family: system-ui, sans-serif;
    }
    
    /* 
     * env() allows access to user-agent defined environment variables
     * Common values: safe-area-inset-*, titlebar-area-*
     * Useful for notched devices and progressive web apps
     */
    
    .header {
      background: linear-gradient(135deg, #667eea 0%, #764ba2 100%);
      color: white;
      padding: 20px;
      
      /* Safe area insets for notched devices (iPhone X+) */
      padding-top: env(safe-area-inset-top, 20px);
      padding-left: env(safe-area-inset-left, 20px);
      padding-right: env(safe-area-inset-right, 20px);
    }
    
    .content {
      padding: 20px;
      
      /* Bottom safe area for devices with home indicator */
      padding-bottom: env(safe-area-inset-bottom, 20px);
    }
    
    .footer {
      position: fixed;
      bottom: 0;
      left: 0;
      right: 0;
      background: #333;
      color: white;
      padding: 15px;
      text-align: center;
      
      /* Ensure footer is above home indicator */
      padding-bottom: calc(15px + env(safe-area-inset-bottom, 0px));
    }
    
    /* fallback example */
    .box {
      /* If env var not defined, use 10px fallback */
      margin: env(custom-spacing, 10px);
    }
  </style>
</head>
<body>
  <header class="header">
    <h1>CSS env() Function</h1>
  </header>
  
  <main class="content">
    <h2>Environment Variables in CSS</h2>
    <p>The <code>env()</code> function accesses browser-defined environment variables.</p>
    <p>Primary use: Safe area insets for notched devices.</p>
  </main>
  
  <footer class="footer">
    Safe Area Aware Footer
  </footer>
</body>
</html>
```

**Why**: `env()` ensures content isn't obscured by device notches, home indicators, or rounded corners on modern devices, critical for PWA and mobile-first design.

## css_fit_content

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>CSS fit-content Function</title>
  <style>
    /* PROBLEM: Create responsive width that fits content
       TIME: O(1) SPACE: O(1) */
    
    body {
      font-family: system-ui, sans-serif;
      padding: 20px;
    }
    
    .container {
      border: 2px dashed #ccc;
      padding: 10px;
      margin-bottom: 20px;
    }
    
    /* 
     * fit-content() sets width to fit content
     * but not exceed max-width
     * Equivalent to: min(max-content, max(min-content, argument))
     */
    
    .fit-content-box {
      /* Width will be as wide as content, max 300px */
      width: fit-content(300px);
      
      background: linear-gradient(135deg, #f093fb 0%, #f5576c 100%);
      color: white;
      padding: 15px 25px;
      border-radius: 8px;
    }
    
    .fit-content-auto {
      /* Shorthand for fit-content(auto) */
      width: fit-content;
      
      background: #4ecdc4;
      color: white;
      padding: 15px 25px;
      border-radius: 8px;
      margin-top: 10px;
    }
    
    /* Practical example: Responsive buttons */
    .btn-group {
      display: flex;
      flex-direction: column;
      gap: 10px;
    }
    
    .btn {
      width: fit-content(200px);
      padding: 12px 24px;
      border: none;
      border-radius: 6px;
      background: #667eea;
      color: white;
```
