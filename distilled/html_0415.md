## js_spread_operator_objects
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>JS Spread Operator with Objects</title>
</head>
<body>
    <script>
        /* PROBLEM: Shallow copy and merge objects
           TIME: O(n) SPACE: O(n) where n = number of properties */

        // Original object
        const person = {
            name: 'Alice',
            age: 30,
            city: 'New York'
        };

        // Shallow copy using spread operator
        const personCopy = { ...person };
        console.log('Shallow copy:', personCopy);
        // Output: { name: 'Alice', age: 30, city: 'New York' }

        // Merge objects - later properties override earlier ones
        const additionalInfo = {
            age: 31,           // Will override original age
            occupation: 'Engineer'
        };

        const updatedPerson = { ...person, ...additionalInfo };
        console.log('Merged object:', updatedPerson);
        // Output: { name: 'Alice', age: 31, city: 'New York', occupation: 'Engineer' }

        // Add new properties while copying
        const personWithEmail = {
            ...person,
            email: 'alice@example.com'
        };
        console.log('With new property:', personWithEmail);

        // Conditional property spreading
        const includePhone = true;
        const personWithOptional = {
            ...person,
            ...(includePhone && { phone: '555-1234' })
        };
        console.log('Conditional spread:', personWithOptional);

        // Nested object merging (requires nested spread)
        const nestedObj = {
            user: { name: 'Bob', details: { age: 25 } }
        };

        const deepMerged = {
            ...nestedObj,
            user: {
                ...nestedObj.user,
                details: {
                    ...nestedObj.user.details,
                    city: 'Boston'
                }
            }
        };
        console.log('Deep merged:', deepMerged);

        // Rest properties in object destructuring
        const { name, ...rest } = person;
        console.log('Name:', name);  // 'Alice'
        console.log('Rest:', rest);  // { age: 30, city: 'New York' }

        // Practical use: default configuration with overrides
        function createUser(userData) {
            const defaults = {
                role: 'user',
                active: true,
                createdAt: new Date().toISOString()
            };
            return { ...defaults, ...userData };
        }

        const newUser = createUser({ name: 'Charlie', role: 'admin' });
        console.log('New user with defaults:', newUser);
    </script>
</body>
</html>
```
**Why:** The spread operator (`...`) for objects provides an elegant syntax for shallow copying, merging, and updating objects immutably, essential for React state management and functional programming patterns.

## js_requestAnimationFrame
```html
<!DOCTYPE html>

<head>
    <meta charset="UTF-8">

        }

            padding: 10px 20px;

            cursor: pointer;
        }
    </style>
</head>
<body>
    <h2>requestAnimationFrame Demo</h2>
    
    <div id="box"></div>
    <div id="progress">
        <div id="progressBar"></div>
    </div>
    
    <div class="controls">
        <button id="startBtn">Start Animation</button>
        <button id="stopBtn">Stop Animation</button>
    </div>
    
    <p>Frame count: <span id="frameCount">0</span></p>
    
    <script>
        const box = document.getElementById('box');
        const progressBar = document.getElementById('progressBar');
        const frameCountEl = document.getElementById('frameCount');
        
        let animationId = null;
        let frameCount = 0;
        let startTime = null;
        let position = 0;
        let direction = 1;
        
        function animate(timestamp) {
            // Initialize start time on first frame
            if (!startTime) startTime = timestamp;
            
            // Calculate elapsed time
            const elapsed = timestamp - startTime;
            
            // Update position (move back and forth)
            position += direction * 2;
            
            // Reverse direction at boundaries
            if (position >= 250 || position <= 0) {
                direction *= -1;
            }
            
            // Apply transform
            box.style.transform = `translateX(${position}px)`;
            
            // Update progress bar based on position
            const progress = (position / 250) * 100;
            progressBar.style.width = `${progress}%`;
            
            // Update frame counter
            frameCount++;
            frameCountEl.textContent = frameCount;
            
            // Continue animation loop
            animationId = requestAnimationFrame(animate);
        }
        
        document.getElementById('startBtn').addEventListener('click', () => {
            if (!animationId) {
                startTime = null;
                animationId = requestAnimationFrame(animate);
            }
        });
        
        document.getElementById('stopBtn').addEventListener('click', () => {
            if (animationId) {
                cancelAnimationFrame(animationId);
                animationId = null;
                console.log(`Animation stopped. Total frames: ${frameCount}`);
            }
        });
        
        // Cleanup on page unload
        window.addEventListener('beforeunload', () => {
            if (animationId) {
                cancelAnimationFrame(animationId);
            }
        });
    </script>
</body>
</html>
```
**Why:** `requestAnimationFrame` synchronizes animations with the browser's refresh rate (typically 60fps), providing smoother animations than `setInterval` while being CPU-efficient and pausing when the tab is inactive.

```html
    <script>
        /* PROBLEM: Find last element matching condition
           TIME: O(n) SPACE: O(1) */

        // Sample data: transactions ordered by time
        const transactions = [
            { id: 1, type: 'deposit', amount: 100 },
            { id: 2, type: 'withdrawal', amount: 50 },

        );

        // Output: 3

        // Practical use: Find most recent active user
        const users = [
            { name: 'Alice', status: 'inactive', lastLogin: '2024-01-01' },
            { name: 'Bob', status: 'active', lastLogin: '2024-01-15' },

            { name: 'Diana', status: 'active', lastLogin: '2024-01-20' }
        ];

        const lastActiveUser = users.findLast(user => user.status === 'active');
        console.log('Most recent active user:', lastActiveUser);
        // Output: { name: 'Diana', status: 'active', lastLogin: '2024-01-20' }

        // Find last number greater than threshold
        const numbers = [1, 5, 8, 3, 12, 7, 4];
        const lastGreaterThanFive = numbers.findLast(n => n > 5);
        console.log('Last number > 5:', lastGreaterThanFive);
        // Output: 7

        // Element not found case
        const notFound = transactions.findLast(tx => tx.amount > 1000);
        console.log('Not found result:', notFound);
        // Output: undefined

        const notFoundIndex = transactions.findLastIndex(tx => tx.amount > 1000);
        console.log('Not found index:', notFoundIndex);
        // Output: -1

        // Comparison with traditional approach
        // Before findLast, you had to reverse or iterate manually:
        const traditionalWay = [...transactions]
            .reverse()
            .find(tx => tx.type === 'withdrawal');
        console.log('Traditional approach result:', traditionalWay);
    </script>
</body>
</html>
```
**Why:** `findLast()` and `findLastIndex()` search arrays from the end, perfect for finding the most recent occurrence or last match without reversing arrays or writing manual loops, improving code readability.

## js_object_hasOwn
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>JS Object.hasOwn Method</title>
</head>
<body>
    <script>
        /* PROBLEM: Safely check if object has own property
           TIME: O(1) SPACE: O(1) */

        // Create object with prototype
        const person = Object.create({
            inheritedProp: 'I am from prototype'
        });
        
        person.name = 'Alice';
        person.age = 30;
        person.city = 'New York';

        // Object.hasOwn: Safe property check
        console.log('Has own name:', Object.hasOwn(person, 'name'));
        // Output: true

        console.log('Has own inherited:', Object.hasOwn(person, 'inheritedProp'));
        // Output: false (inherited, not own)

        // Comparison with hasOwnProperty
        console.log('hasOwnProperty name:', person.hasOwnProperty('name'));
        // Output: true

        // Problem with hasOwnProperty: can be overridden
        const dangerous = {
            hasOwnProperty: () => false,  // Malicious override
            secretKey: 'hidden'
        };

        // This returns false (incorrect!)
        console.log('Overridden hasOwnProperty:', 
            dangerous.hasOwnProperty('secretKey'));

        // Object.hasOwn still works correctly
        console.log('Safe check with hasOwn:', 
            Object.hasOwn(dangerous, 'secretKey'));
        // Output: true

        // Works with null-prototype objects
        const nullProto = Object.create(null);
        nullProto.value = 42;

        // hasOwnProperty would throw error here
        try {
            console.log('hasOwnProperty on null proto:', 
                nullProto.hasOwnProperty('value'));
        } catch (e) {
            console.log('Error with hasOwnProperty:', e.message);
        }

        // hasOwn works fine
        console.log('hasOwn on null proto:', Object.hasOwn(nullProto, 'value'));
        // Output: true

        // Practical use: Validate config objects
        function validateConfig(config) {
            const requiredProps = ['host', 'port', 'database'];
            
            for (const prop of requiredProps) {
                if (!Object.hasOwn(config, prop)) {
                    throw new Error(`Missing required property: ${prop}`);
                }
            }
            
            return true;
        }

        const config = { host: 'localhost', port: 5432, database: 'myapp' };
        console.log('Config valid:', validateConfig(config));

        // Use with objects that have null values
        const sparse = { a: 1, b: null, c: undefined };
        console.log('Has own b:', Object.hasOwn(sparse, 'b'));  // true
        console.log('Has own d:', Object.hasOwn(sparse, 'd'));  // false
    </script>
</body>
</html>
```
**Why:** `Object.hasOwn()` is the modern, safer replacement for `hasOwnProperty`, working correctly even when objects have overridden prototype methods or use `Object.create(null)`, providing reliable property existence checks.

```html
    <script>
        // Select all elements matching CSS selector
        // Returns a static NodeList (not live)
        
        function highlightAll() {
            // Select all .card elements
            const cards = document.querySelectorAll('.card');
            
            cards.forEach(card => {
                card.classList.add('highlighted');
            });
            
            updateStats(`Total cards: ${cards.length}`);
        }
        
        function highlightHighPriority() {
            // Reset first
            resetHighlight();
            
            // Complex selector: cards with high priority attribute
            const highPriorityCards = document.querySelectorAll(
                '.card[data-priority="high"]'
            );
            
            highPriorityCards.forEach(card => {
                card.classList.add('highlighted');
            });
            
            updateStats(`High priority cards: ${highPriorityCards.length}`);
        }
        
        function resetHighlight() {
            const cards = document.querySelectorAll('.card.highlighted');
            
            cards.forEach(card => {
                card.classList.remove('highlighted');
            });
            
            updateStats('Selection cleared');
        }
        
        function updateStats(message) {
            document.getElementById('stats').textContent = message;
        }
        
        // Multiple selectors (comma-separated)
        const headingsAndButtons = document.querySelectorAll('h2, button');
        console.log(`Found ${headingsAndButtons.length} headings and buttons`);
        
        // Descendant selectors
        const statsContent = document.querySelectorAll('.stats #stats');
        console.log(`Stats element:`, statsContent[0]);
        
        // Pseudo-classes
        const firstCard = document.querySelectorAll('.card:nth-child(odd)');
        console.log(`Odd-positioned cards: ${firstCard.length}`);
        
        // Convert NodeList to Array for more methods
        const cardsArray = Array.from(document.querySelectorAll('.card'));
        const cardTexts = cardsArray.map(card => card.textContent);
        console.log('Card texts:', cardTexts);
        
        // Note: NodeList is static, not live
        const allCards = document.querySelectorAll('.card');
        console.log(`Initial count: ${allCards.length}`);
        
        // Adding new card doesn't update the NodeList
        const newCard = document.createElement('div');
        newCard.className = 'card';
        newCard.textContent = 'New Card';
        document.body.appendChild(newCard);
        
        console.log(`After adding card: ${allCards.length} (still same)`);
        console.log(`Live collection: ${document.getElementsByClassName('card').length}`);
    </script>
</body>
</html>
```

**Why:** `querySelectorAll()` selects multiple elements using CSS selector syntax, returning a static NodeList that supports `forEach()`, enabling powerful batch operations on element collections with familiar selector patterns.
