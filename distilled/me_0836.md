## 1. Maximum Sum Subarray of Size K
```python
# PROBLEM: Find maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)
def max_sum_subarray_k(arr, k):
    """
    Sliding window technique for fixed-size window.
    Maintains sum of current window, slides by removing left and adding right.
    """
    if len(arr) < k:
        return None
    
    # Initialize first window
    window_sum = sum(arr[:k])
    max_sum = window_sum
    
    # Slide window from k to end
    for i in range(k, len(arr)):
        # Remove leftmost element, add new right element
        window_sum = window_sum - arr[i - k] + arr[i]
        max_sum = max(max_sum, window_sum)
    
    return max_sum
```
**Why**: Classic sliding window pattern. Foundation for many substring/subarray problems.

## 2. Two Sum (Sorted Array)
```python
# PROBLEM: Find two numbers that sum to target in sorted array
# TIME: O(n) SPACE: O(1)
def two_sum_sorted(arr, target):
    """
    Two pointers from both ends, move based on comparison with target.
    Only works on sorted arrays - enables O(n) instead of O(n²) brute force.
    """
    left, right = 0, len(arr) - 1
    
    while left < right:
        current_sum = arr[left] + arr[right]
        
        if current_sum == target:
            return (left, right)  # Found the pair
        elif current_sum < target:
            left += 1   # Need larger sum, move left pointer right
        else:
            right -= 1  # Need smaller sum, move right pointer left
    
    return None  # No valid pair found
```
**Why**: Two-pointer technique is essential for sorted array problems, container with most water, etc.

## 3. Permutations (Backtracking)
```python
# PROBLEM: Generate all permutations of a list
# TIME: O(n! × n) SPACE: O(n) for recursion depth
def permutations(nums):
    """
    Backtracking: try each choice, recurse, undo choice.
    Build permutation by swapping elements at each position.
    """
    result = []
    
    def backtrack(start):
        # Base case: all positions filled
        if start == len(nums):
            result.append(nums[:])  # Copy current state
            return
        
        # Try each element at current position
        for i in range(start, len(nums)):
            # Place element at position start
            nums[start], nums[i] = nums[i], nums[start]
            
            # Recurse to fill next position
            backtrack(start + 1)
            
            # Undo placement (backtrack)
            nums[start], nums[i] = nums[i], nums[start]
    
    backtrack(0)
    return result
```
**Why**: Backtracking is the core pattern for combinations, subsets, N-Queens, Sudoku solver.

## 4. Number of Islands (Union-Find)
```python
# PROBLEM: Count connected islands in 2D grid (1=land, 0=water)
# TIME: O(m×n×α(m×n)) SPACE: O(m×n)
class UnionFind:
    """Disjoint Set Union with path compression and union by rank."""
    
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.count = 0  # Number of disjoint sets
    
    def find(self, x):
        """Find root with path compression."""
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        """Union by rank, return True if merged."""
        px, py = self.find(x), self.find(y)
        if px == py:
            return False
        
        # Attach smaller tree under larger tree
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        self.count -= 1
        return True

def num_islands(grid):
    """
    Union-Find approach: each land cell is a node.
    Union adjacent lands, count remaining sets.
    """
    if not grid or not grid[0]:
        return 0
    
    rows, cols = len(grid), len(grid[0])
    uf = UnionFind(rows * cols)
    
    # Count initial lands
    land_count = 0
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == '1':
                land_count += 1
    
    uf.count = land_count
    
    # Union adjacent lands
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] != '1':
                continue
            
            # Check right and down neighbors
            if c + 1 < cols and grid[r][c + 1] == '1':
                uf.union(r * cols + c, r * cols + (c + 1))
            if r + 1 < rows and grid[r + 1][c] == '1':
                uf.union(r * cols + c, (r + 1) * cols + c)
    
    return uf.count
```
**Why**: Union-Find excels at dynamic connectivity problems, cycle detection, minimum spanning trees.

## 5. Longest Common Subsequence (Dynamic Programming)
```python
# PROBLEM: Find length of longest common subsequence between two strings
# TIME: O(m×n) SPACE: O(m×n) or O(min(m,n)) with optimization
def longest_common_subsequence(text1, text2):
    """
    DP: dp[i][j] = LCS length of text1[:i] and text2[:j]
    If chars match: dp[i][j] = dp[i-1][j-1] + 1
    Else: dp[i][j] = max(dp[i-1][j], dp[i][j-1])
    """
    m, n = len(text1), len(text2)
    
    # Space-optimized: only keep previous row
    prev = [0] * (n + 1)
    
    for i in range(1, m + 1):
        curr = [0] * (n + 1)
        for j in range(1, n + 1):
            if text1[i - 1] == text2[j - 1]:
                # Characters match: extend LCS
                curr[j] = prev[j - 1] + 1
            else:
                # Take max of excluding char from either string
                curr[j] = max(prev[j], curr[j - 1])
        prev = curr
    
    return prev[n]
```
**Why**: Classic DP pattern. Foundation for edit distance, shortest common supersequence.

## 6. Dijkstra's Shortest Path
```python
# PROBLEM: Find shortest path from source to all nodes in weighted graph
# TIME: O((V+E) log V) SPACE: O(V)
import heapq

def dijkstra(graph, source):
    """
    Greedy BFS with min-heap: always process closest unvisited node.
    Works only with non-negative edge weights.
    
    graph: {node: [(neighbor, weight), ...]}
    """
    # Distance from source to each node
    dist = {node: float('inf') for node in graph}
    dist[source] = 0
    
    # Min-heap: (distance, node)
    heap = [(0, source)]
    visited = set()
    
    while heap:
        d, node = heapq.heappop(heap)
        
        # Skip if already processed with shorter distance
        if node in visited:
            continue
        visited.add(node)
        
        # Relax all edges from current node
        for neighbor, weight in graph[node]:
            if neighbor in visited:
                continue
            
            new_dist = d + weight
            if new_dist < dist[neighbor]:
                dist[neighbor] = new_dist
                heapq.heappush(heap, (new_dist, neighbor))
    
    return dist
```
**Why**: Shortest path algorithm for weighted graphs. Used in navigation, network routing.

## 7. Lowest Common Ancestor (Binary Tree)
```python
# PROBLEM: Find lowest common ancestor of two nodes in binary tree
# TIME: O(n) SPACE: O(h) where h is tree height
def lowest_common_ancestor(root, p, q):
    """
    Post-order traversal: LCA is first node where p and q
    are in different subtrees (or one is the node itself).
    """
    if not root:
        return None
    
    # If current node is p or q, it could be LCA
    if root == p or root == q:
        return root
    
    # Search in both subtrees
    left = lowest_common_ancestor(root.left, p, q)
    right = lowest_common_ancestor(root.right, p, q)
    
    # If both subtrees found target, current node is LCA
    if left and right:
        return root
    
    # Otherwise return non-None result from subtrees
    return left if left else right
```
**Why**: Tree traversal pattern. Essential for BST operations, tree distance queries.

## 8. Kth Largest Element (Heap)
```python
# PROBLEM: Find kth largest element in unsorted array
# TIME: O(n log k) SPACE: O(k)
import heapq

def kth_largest(nums, k):
    """
    Min-heap of size k: smallest element in heap is kth largest overall.
    More efficient than sorting when k << n.
    """
    # Build min-heap with first k elements
    heap = nums[:k]
    heapq.heapify(heap)
    
    # Process remaining elements
    for num in nums[k:]:
        # If num is larger than smallest in heap, replace it
        if num > heap[0]:
            heapq.heapreplace(heap, num)
    
    # Root of heap is kth largest
    return heap[0]
```
**Why**: Heap pattern for top-k problems. Alternative: QuickSelect for O(n) average.

## 9. Word Search with Trie
```python
# PROBLEM: Find all words from dictionary in 2D board (Boggle solver)
# TIME: O(m×n×4^L) where L is max word length SPACE: O(total chars in dict)
class TrieNode:
    def __init__(self):
        self.children = {}
        self.word = None  # Stores complete word at end node

def build_trie(words):
    """Build trie from word list for efficient prefix matching."""
    root = TrieNode()
    for word in words:
        node = root
        for ch in word:
            if ch not in node.children:
                node.children[ch] = TrieNode()
            node = node.children[ch]
        node.word = word  # Mark end of word
    return root

def find_words(board, words):
    """
    DFS from each cell, following trie prefixes.
    Trie enables early termination when prefix invalid.
    """
    result = []
    trie = build_trie(words)
    rows, cols = len(board), len(board[0])
    
    def dfs(r, c, node):
        ch = board[r][c]
        
        # Dead end: char not in trie path
        if ch not in node.children:
            return
        
        next_node = node.children[ch]
        
        # Found a complete word
        if next_node.word:
            result.append(next_node.word)
            next_node.word = None  # Avoid duplicates
        
        # Mark visited
        board[r][c] = '#'
        
        # Explore 4 directions
        for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
            nr, nc = r + dr, c + dc
            if 0 <= nr < rows and 0 <= nc < cols:
                dfs(nr, nc, next_node)
        
        # Backtrack
        board[r][c] = ch
    
    for r in range(rows):
        for c in range(cols):
            dfs(r, c, trie)
    
    return result
```
**Why**: Trie enables efficient prefix matching. Used in autocomplete, spell checkers.

## 10. Memoized Fibonacci (LRU Cache)
```python
# PROBLEM: Compute nth Fibonacci number efficiently
# TIME: O(n) SPACE: O(n)
from functools import lru_cache

@lru_cache(maxsize=None)
def fibonacci(n):
    """
    LRU cache automatically memoizes results.
    Transforms exponential O(2^n) to linear O(n).
    Cache is shared across calls within same session.
    """
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

# Alternative: Manual memoization with dict
def fibonacci_manual(n, memo=None):
    """Explicit memoization for more control."""
    if memo is None:
        memo = {}
    
    if n in memo:
        return memo[n]
    
    if n < 2:
        return n
    
    result = fibonacci_manual(n - 1, memo) + fibonacci_manual(n - 2, memo)
    memo[n] = result
    return result
```
**Why**: Memoization pattern with Python's built-in decorator. Apply to any recursive problem with overlapping subproblems.
