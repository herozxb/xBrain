## 1. Maximum Sum Subarray of Size K
```python
# PROBLEM: Find the maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)
def max_sum_subarray_k(arr, k):
    """
    Use sliding window to maintain a window of size k.
    Slide the window by adding the next element and removing the first.
    Track the maximum sum seen so far.
    """
    if len(arr) < k:
        return -1
    
    # Compute sum of first window
    window_sum = sum(arr[:k])
    max_sum = window_sum
    
    # Slide window from k to end of array
    for i in range(k, len(arr)):
        # Add next element, remove first element of previous window
        window_sum += arr[i] - arr[i - k]
        max_sum = max(max_sum, window_sum)
    
    return max_sum
```
**Why**: Classic sliding window pattern for fixed-size subarray problems. O(n) beats brute force O(n*k).

## 2. Container With Most Water

```python
# PROBLEM: Find two lines that form container holding the most water
# TIME: O(n) SPACE: O(1)
def max_area(heights):
    """
    Use two pointers from both ends. Area = min(height) * width.
    Move the shorter line inward since it limits the area.
    The taller line might find a better pair.
    """
    left, right = 0, len(heights) - 1
    max_area = 0
    
    while left < right:
        # Calculate current area
        width = right - left
        h = min(heights[left], heights[right])
        max_area = max(max_area, width * h)
        
        # Move the shorter pointer inward
        if heights[left] < heights[right]:
            left += 1
        else:
            right -= 1
    
    return max_area
```
**Why**: Two pointers eliminate impossible pairs efficiently. Greedy choice is optimal here.

## 3. Generate All Permutations

```python
# PROBLEM: Generate all permutations of a list of numbers
# TIME: O(n! * n) SPACE: O(n) for recursion depth
def permute(nums):
    """
    Use backtracking: swap elements to generate permutations.
    Fix one position at a time, recurse on remaining positions.
    Backtrack by swapping back to restore state.
    """
    result = []
    
    def backtrack(start):
        # Base case: all positions fixed
        if start == len(nums):
            result.append(nums[:])  # Make a copy
            return
        
        # Try each element at position 'start'
        for i in range(start, len(nums)):
            # Swap current element to position 'start'
            nums[start], nums[i] = nums[i], nums[start]
            
            # Recurse on remaining positions
            backtrack(start + 1)
            
            # Backtrack: restore original order
            nums[start], nums[i] = nums[i], nums[start]
    
    backtrack(0)
    return result
```
**Why**: Backtracking explores all possibilities systematically. Swap-based approach is space-efficient.

## 4. Number of Connected Components

```python
# PROBLEM: Count connected components in an undirected graph
# TIME: O(n * α(n)) ≈ O(n) SPACE: O(n)
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.components = n
    
    def find(self, x):
        """Path compression: point nodes directly to root."""
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        """Union by rank: attach shorter tree under taller one."""
        px, py = self.find(x), self.find(y)
        if px == py:
            return  # Already connected
        
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        self.components -= 1

def count_components(n, edges):
    """
    Union Find: each edge merges two components.
    Final count is number of disjoint sets.
    """
    uf = UnionFind(n)
    for u, v in edges:
        uf.union(u, v)
    return uf.components
```
**Why**: Union Find is optimal for dynamic connectivity. Near-constant time per operation.

```python
# PROBLEM: Find the length of the longest common subsequence of two strings
# TIME: O(m * n) SPACE: O(min(m, n)) with space optimization
def lcs(text1, text2):
    """
    DP approach: if chars match, extend LCS by 1.
    Otherwise, take max of skipping one char from either string.
    Space optimization: only keep current and previous row.
    """
    # Ensure text1 is shorter for space optimization
    if len(text1) > len(text2):
        text1, text2 = text2, text1
    
    m, n = len(text1), len(text2)
    # Only two rows needed
    prev = [0] * (m + 1)
    curr = [0] * (m + 1)
    
    for j in range(1, n + 1):
        for i in range(1, m + 1):
            if text1[i - 1] == text2[j - 1]:
                # Characters match: extend LCS
                curr[i] = prev[i - 1] + 1
            else:
                # Take max of skipping char from either string
                curr[i] = max(prev[i], curr[i - 1])
        prev, curr = curr, [0] * (m + 1)
    
    return prev[m]
```
**Why**: Classic 2D DP with space optimization. Foundation for string comparison problems.

## 6. Dijkstra's Shortest Path
```python
# PROBLEM: Find shortest path from source to all nodes in weighted graph
# TIME: O((V + E) log V) SPACE: O(V)
import heapq

def dijkstra(n, edges, source):
    """
    Greedy approach: always process node with smallest known distance.
    Use min-heap for efficient minimum extraction.
    Works only with non-negative edge weights.
    """
    # Build adjacency list
    graph = {i: [] for i in range(n)}
    for u, v, w in edges:
        graph[u].append((v, w))
    
    # Initialize distances
    dist = [float('inf')] * n
    dist[source] = 0
    
    # Min-heap: (distance, node)
    heap = [(0, source)]
    
    while heap:
        d, u = heapq.heappop(heap)
        
        # Skip if we found a better path already
        if d > dist[u]:
            continue
        
        # Relax all neighbors
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(heap, (dist[v], v))
    
    return dist
```
**Why**: Optimal for single-source shortest path with non-negative weights. Heap enables efficient priority queue.

## 7. Lowest Common Ancestor in Binary Tree
```python
# PROBLEM: Find lowest common ancestor of two nodes in a binary tree
# TIME: O(n) SPACE: O(h) where h is tree height
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def lowest_common_ancestor(root, p, q):
    """
    Recursively search for p and q.
    If current node is p or q, return it.
    If both subtrees return non-null, current is LCA.
    If only one subtree returns non-null, propagate it up.
    """
    # Base case: reached end or found target
    if not root or root == p or root == q:
        return root
    
    # Search in left and right subtrees
    left = lowest_common_ancestor(root.left, p, q)
    right = lowest_common_ancestor(root.right, p, q)
    
    # If both found, current node is LCA
    if left and right:
        return root
    
    # Otherwise, return non-null result (or None if both null)
    return left if left else right
```
**Why**: Elegant recursive solution. Single traversal finds both nodes and their ancestor.

## 8. Merge K Sorted Linked Lists

```python
# PROBLEM: Merge k sorted linked lists into one sorted list
# TIME: O(n log k) where n = total nodes, k = number of lists
# SPACE: O(k) for the heap
import heapq

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def merge_k_lists(lists):
    """
    Use min-heap to always get the smallest element.
    Push first node of each list into heap.
    Pop minimum, add to result, push its next node.
    """
    # Min-heap: (value, list_index, node)
    # list_index breaks ties and prevents ListNode comparison
    heap = []
    for i, node in enumerate(lists):
        if node:
            heapq.heappush(heap, (node.val, i, node))
    
    # Dummy head simplifies list construction
    dummy = ListNode(0)
    curr = dummy
    
    while heap:
        val, i, node = heapq.heappop(heap)
        curr.next = ListNode(val)
        curr = curr.next
        
        # Push next node from same list
        if node.next:
            heapq.heappush(heap, (node.next.val, i, node.next))
    
    return dummy.next
```
**Why**: Heap efficiently merges multiple sorted sequences. Better than sequential merging O(n*k).

## 9. Trie for Prefix Search
```python
# PROBLEM: Implement a trie for efficient prefix-based word search
# TIME: O(m) for insert/search/startsWith where m = word length
# SPACE: O(alphabet_size * total_characters)
class TrieNode:
    def __init__(self):
        self.children = {}  # char -> TrieNode
        self.is_end = False

class Trie:
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word):
        """Insert word into trie, creating nodes as needed."""
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
    
    def search(self, word):
        """Check if exact word exists in trie."""
        node = self.root
        for char in word:
            if char not in node.children:
                return False
            node = node.children[char]
        return node.is_end
    
    def starts_with(self, prefix):
        """Check if any word starts with given prefix."""
        node = self.root
        for char in prefix:
            if char not in node.children:
                return False
            node = node.children[char]
        return True
```
**Why**: Trie provides O(m) prefix operations regardless of dictionary size. Essential for autocomplete, spell checkers.

## 10. LRU Cache Implementation
```python
# PROBLEM: Design a cache with O(1) get and put, evicting least recently used
# TIME: O(1) for both get and put
# SPACE: O(capacity)
class Node:
    """Doubly linked list node for O(1) removal."""
    def __init__(self, key=0, val=0):
        self.key = key
        self.val = val
        self.prev = None
        self.next = None

class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}  # key -> Node
        
        # Dummy head and tail simplify edge cases
        self.head = Node()
        self.tail = Node()
        self.head.next = self.tail
        self.tail.prev = self.head
    
    def _remove(self, node):
        """Remove node from linked list."""
        node.prev.next = node.next
        node.next.prev = node.prev
    
    def _add_to_front(self, node):
        """Add node right after head (most recent position)."""
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node
    
    def get(self, key):
        """Return value and move to front (mark as recently used)."""
        if key not in self.cache:
            return -1
        
        node = self.cache[key]
        # Move to front: remove then add
        self._remove(node)
        self._add_to_front(node)
        return node.val
    
    def put(self, key, value):
        """Add/update key-value pair, evict LRU if over capacity."""
        if key in self.cache:
            # Update existing: remove old node
            self._remove(self.cache[key])
        
        # Add new node
        node = Node(key, value)
        self.cache[key] = node
        self._add_to_front(node)
        
        # Evict LRU if over capacity
        if len(self.cache) > self.capacity:
            # LRU is at tail (before dummy tail)
            lru = self.tail.prev
            self._remove(lru)
            del self.cache[lru.key]
```
**Why**: Combines hash map (O(1) lookup) with doubly linked list (O(1) reordering). Standard interview question pattern.
