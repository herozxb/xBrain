## 1. Compressed Trie (Radix Trie)

```python
# PROBLEM: Implement a space-efficient trie that compresses chains of single-child nodes
# TIME: O(L) for insert/search where L is key length, O(1) for prefix check
# SPACE: O(N * L) where N is number of keys, but compressed saves 40-60% space
class CompressedTrieNode:
    """Node for compressed trie storing edge labels instead of single characters."""
    def __init__(self):
        self.children = {}  # char -> (label, node)
        self.is_end = False

class CompressedTrie:
    """Space-efficient trie compressing single-child chains."""
    
    def __init__(self):
        self.root = CompressedTrieNode()
    
    def insert(self, word: str) -> None:
        """Insert word, compressing paths where possible."""
        node = self.root
        i = 0
        
        while i < len(word):
            char = word[i]
            
            # No existing edge, create new compressed path
            if char not in node.children:
                node.children[char] = (word[i:], CompressedTrieNode())
                node.children[char][1].is_end = True
                return
            
            label, child = node.children[char]
            
            # Find common prefix between label and remaining word
            j = 0
            while j < len(label) and i + j < len(word) and label[j] == word[i + j]:
                j += 1
            
            if j == len(label):
                # Full label match, continue deeper
                node = child
                i += j
            else:
                # Partial match, split the node
                # Create new intermediate node
                mid = CompressedTrieNode()
                
                # Update parent to point to mid with shortened label
                node.children[char] = (label[:j], mid)
                
                # Move child under mid with remaining label
                mid.children[label[j]] = (label[j:], child)
                
                # Add new branch if there's remaining word
                if i + j < len(word):
                    new_child = CompressedTrieNode()
                    new_child.is_end = True
                    mid.children[word[i + j]] = (word[i + j:], new_child)
                else:
                    mid.is_end = True
                
                return
        
        node.is_end = True
    
    def search(self, word: str) -> bool:
        """Check if exact word exists in trie."""
        node = self.root
        i = 0
        
        while i < len(word):
            char = word[i]
            if char not in node.children:
                return False
            
            label, child = node.children[char]
            
            # Check if label matches word segment
            if not word[i:].startswith(label):
                return False
            
            i += len(label)
            node = child
        
        return node.is_end
    
    def starts_with(self, prefix: str) -> bool:
        """Check if any word starts with given prefix."""
        node = self.root
        i = 0
        
        while i < len(prefix):
            char = prefix[i]
            if char not in node.children:
                return False
            
            label, child = node.children[char]
            
            # Check if label is compatible with prefix
            remaining = prefix[i:]
            if not remaining.startswith(label[:len(remaining)] if len(remaining) < len(label) else label):
                return False
            
            i += len(label)
            node = child
        
        return True
```

**Why**: Compressed tries reduce memory by 40-60% compared to standard tries by merging single-child chains. Used in text editors for autocomplete and spell-checkers.

## 2. Scalable Bloom Filter

```python
# PROBLEM: Implement a bloom filter that grows when fill ratio exceeds threshold
# TIME: O(k) for add/contains where k is number of hash functions
# SPACE: O(m) bits initially, grows by factor when full
import hashlib
from bitarray import bitarray
import math

class ScalableBloomFilter:
    """Bloom filter that scales by adding new filters when capacity reached."""
    
    def __init__(self, initial_capacity: int = 1000, error_rate: float = 0.001, 
                 scale_factor: int = 2):
        """
        Initialize scalable bloom filter.
        
        Args:
            initial_capacity: Expected number of elements initially
            error_rate: Target false positive rate
            scale_factor: Growth multiplier when expanding
        """
        self.error_rate = error_rate
        self.scale_factor = scale_factor
        self.filters = []
        self._add_filter(initial_capacity, error_rate)
    
    def _add_filter(self, capacity: int, error_rate: float) -> None:
        """Create and add new bloom filter."""
        # Calculate optimal size and hash count
        # m = -n * ln(p) / (ln(2)^2)
        # k = (m/n) * ln(2)
        size = int(-capacity * math.log(error_rate) / (math.log(2) ** 2))
        k = int((size / capacity) * math.log(2))
        
        self.filters.append({
            'bits': bitarray(size),
            'k': k,
            'count': 0,
            'capacity': capacity
        })
        self.filters[-1]['bits'].setall(False)
    
    def _hashes(self, item: str, size: int, k: int) -> list:
        """Generate k hash values using double hashing."""
        # Use MD5 for first hash, SHA256 for second
        h1 = int(hashlib.md5(item.encode()).hexdigest(), 16)
        h2 = int(hashlib.sha256(item.encode()).hexdigest(), 16)
        
        # Double hashing: h(i) = h1 + i * h2
        return [(h1 + i * h2) % size for i in range(k)]
    
    def add(self, item: str) -> None:
        """Add item to filter, creating new filter if needed."""
        # Add to newest filter if it has space
        if self.filters[-1]['count'] >= self.filters[-1]['capacity']:
            # Create new, larger filter
            new_capacity = self.filters[-1]['capacity'] * self.scale_factor
            # Tighten error rate for new filters
            self._add_filter(new_capacity, self.error_rate * 0.9)
        
        current = self.filters[-1]
        for pos in self._hashes(item, len(current['bits']), current['k']):
            current['bits'][pos] = True
        current['count'] += 1
    
    def contains(self, item: str) -> bool:
        """Check if item might exist in any filter."""
        # Check all filters - false positives accumulate but stay bounded
        for f in self.filters:
            if all(f['bits'][pos] for pos in self._hashes(item, len(f['bits']), f['k'])):
                return True
        return False
    
    def size_info(self) -> dict:
        """Return information about current filter sizes."""
        return {
            'num_filters': len(self.filters),
            'total_bits': sum(len(f['bits']) for f in self.filters),
            'total_items': sum(f['count'] for f in self.filters)
        }
```

**Why**: Standard bloom filters have fixed capacity. Scalable variants auto-grow while maintaining bounded error rates. Essential for unknown or growing datasets.

## 3. Inversion of Control Container

```python
# PROBLEM: Implement IoC container for dependency injection and service management
# TIME: O(1) for registration, O(D) for resolution where D is dependency depth
# SPACE: O(S) where S is number of registered services
from typing import Any, Callable, Dict, Type, TypeVar, Optional
from dataclasses import dataclass
from enum import Enum

T = TypeVar('T')

class Lifestyle(Enum):
    """Service lifecycle management strategy."""
    TRANSIENT = "transient"  # New instance each request
    SINGLETON = "singleton"  # Single instance shared
    SCOPED = "scoped"        # One instance per scope

@dataclass
class ServiceDescriptor:
    """Describes how to create and manage a service."""
    interface: Type
    implementation: Type
    lifestyle: Lifestyle
    factory: Optional[Callable] = None
    instance: Optional[Any] = None

class IoCContainer:
    """Lightweight IoC container with automatic dependency resolution."""
    
    def __init__(self):
        self._services: Dict[Type, ServiceDescriptor] = {}
        self._singletons: Dict[Type, Any] = {}
        self._scopes: Dict[str, Dict[Type, Any]] = {}
    
    def register(self, interface: Type[T], implementation: Type[T], 
                 lifestyle: Lifestyle = Lifestyle.TRANSIENT) -> None:
        """Register service with lifestyle."""
        self._services[interface] = ServiceDescriptor(
            interface=interface,
            implementation=implementation,
            lifestyle=lifestyle
        )
    
    def register_factory(self, interface: Type[T], factory: Callable, 
                        lifestyle: Lifestyle = Lifestyle.TRANSIENT) -> None:
        """Register service with custom factory."""
        self._services[interface] = ServiceDescriptor(
            interface=interface,
            implementation=None,
            lifestyle=lifestyle,
            factory=factory
        )
    
    def register_instance(self, interface: Type[T], instance: T) -> None:
        """Register existing instance as singleton."""
        self._services[interface] = ServiceDescriptor(
            interface=interface,
            implementation=None,
            lifestyle=Lifestyle.SINGLETON,
            instance=instance
        )
        self._singletons[interface] = instance
    
    def resolve(self, interface: Type[T], scope_id: Optional[str] = None) -> T:
        """Resolve service with automatic dependency injection."""
        if interface not in self._services:
            raise ValueError(f"Service {interface.__name__} not registered")
        
        descriptor = self._services[interface]
        
        # Handle singleton
        if descriptor.lifestyle == Lifestyle.SINGLETON:
            if interface in self._singletons:
                return self._singletons[interface]
            
            instance = self._create_instance(descriptor, scope_id)
            self._singletons[interface] = instance
            return instance
        
        # Handle scoped
        if descriptor.lifestyle == Lifestyle.SCOPED:
            if not scope_id:
                raise ValueError("Scope ID required for scoped services")
            
            if scope_id not in self._scopes:
                self._scopes[scope_id] = {}
            
            if interface in self._scopes[scope_id]:
                return self._scopes[scope_id][interface]
            
            instance = self._create_instance(descriptor, scope_id)
            self._scopes[scope_id][interface] = instance
            return instance
        
        # Handle transient
        return self._create_instance(descriptor, scope_id)
    
    def _create_instance(self, descriptor: ServiceDescriptor, 
                        scope_id: Optional[str]) -> Any:
        """Create instance with dependency injection."""
        # Use factory if provided
        if descriptor.factory:
            return descriptor.factory(self)
        
        # Use pre-created instance
        if descriptor.instance:
            return descriptor.instance
        
        # Auto-resolve constructor dependencies
        impl = descriptor.implementation
        constructor = impl.__init__
        
        # Get type hints for constructor parameters
        hints = {}
        if hasattr(constructor, '__annotations__'):
            hints = {k: v for k, v in constructor.__annotations__.items() 
                    if k != 'return'}
        
        # Resolve dependencies
        deps = {}
        for param_name, param_type in hints.items():
            deps[param_name] = self.resolve(param_type, scope_id)
        
        return impl(**deps)
    
    def create_scope(self, scope_id: str) -> 'Scope':
        """Create a new scope for scoped services."""
        if scope_id not in self._scopes:
            self._scopes[scope_id] = {}
        return Scope(scope_id, self)
    
    def dispose_scope(self, scope_id: str) -> None:
        """Dispose all scoped services for given scope."""
        if scope_id in self._scopes:
            del self._scopes[scope_id]

class Scope:
    """Manages scoped service lifetime."""
    
    def __init__(self, scope_id: str, container: IoCContainer):
        self.scope_id = scope_id
        self.container = container
    
    def resolve(self, interface: Type[T]) -> T:
        """Resolve service within this scope."""
        return self.container.resolve(interface, self.scope_id)
    
    def dispose(self):
        """Dispose this scope."""
        self.container.dispose_scope(self.scope_id)
```

**Why**: IoC containers decouple component dependencies, making code testable and maintainable. Automatic resolution eliminates manual wiring. Essential for large applications.

## 4. Specification Pattern

```python
# PROBLEM: Implement composable business rules that can be combined
# TIME: O(1) for composition, O(C) for satisfaction where C is condition complexity
# SPACE: O(R) where R is number of composed rules
from abc import ABC, abstractmethod
from typing import Generic, TypeVar, List, Callable
from dataclasses import dataclass

T = TypeVar('T')

class Specification(ABC, Generic[T]):
    """Base specification for business rules."""
    
    @abstractmethod
    def is_satisfied_by(self, candidate: T) -> bool:
        """Check if candidate satisfies this specification."""
        pass
    
    def and_spec(self, other: 'Specification[T]') -> 'Specification[T]':
        """Combine with AND logic."""
        return AndSpecification(self, other)
    
    def or_spec(self, other: 'Specification[T]') -> 'Specification[T]':
        """Combine with OR logic."""
        return OrSpecification(self, other)
    
    def not_spec(self) -> 'Specification[T]':
        """Negate this specification."""
        return NotSpecification(self)
    
    def filter(self, candidates: List[T]) -> List[T]:
        """Filter candidates by this specification."""
        return [c for c in candidates if self.is_satisfied_by(c)]

class AndSpecification(Specification[T]):
    """Combines two specifications with AND logic."""
    
    def __init__(self, left: Specification[T], right: Specification[T]):
        self.left = left
        self.right = right
    
    def is_satisfiedby(self, candidate: T) -> bool:
        return self.left.is_satisfied_by(candidate) and \
               self.right.is_satisfied_by(candidate)

class OrSpecification(Specification[T]):
    """Combines two specifications with OR logic."""
    
    def __init__(self, left: Specification[T], right: Specification[T]):
        self.left = left
        self.right = right
    
    def is_satisfiedby(self, candidate: T) -> bool:
        return self.left.is_satisfied_by(candidate) or \
               self.right.is_satisfied_by(candidate)

class NotSpecification(Specification[T]):
    """Negates a specification."""
    
    def __init__(self, spec: Specification[T]):
        self.spec = spec
    
    def is_satisfied_by(self, candidate: T) -> bool:
        return not self.spec.is_satisfied_by(candidate)

class LambdaSpecification(Specification[T]):
    """Wraps a lambda as a specification."""
    
    def __init__(self, predicate: Callable[[T], bool]):
        self.predicate = predicate
    
    def is_satisfied_by(self, candidate: T) -> bool:
        return self.predicate(candidate)

# Example: E-commerce domain
@dataclass
class Product:
    """Product entity."""
    id: int
    name: str
    price: float
    category: str
    in_stock: bool
    rating: float

class PriceRangeSpecification(Specification[Product]):
    """Product price within range."""
    
    def __init__(self, min_price: float, max_price: float):
        self.min_price = min_price
        self.max_price = max_price
    
    def is_satisfied_by(self, candidate: Product) -> bool:
        return self.min_price <= candidate.price <= self.max_price

class InStockSpecification(Specification[Product]):
    """Product is in stock."""
    
    def is_satisfied_by(self, candidate: Product) -> bool:
        return candidate.in_stock

class CategorySpecification(Specification[Product]):
    """Product belongs to category."""
    
    def __init__(self, category: str):
        self.category = category
    
    def is_satisfied_by(self, candidate: Product) -> bool:
        return candidate.category == self.category

class HighRatedSpecification(Specification[Product]):
    """Product has high rating."""
    
    def __init__(self, min_rating: float = 4.0):
        self.min_rating = min_rating
    
    def is_satisfied_by(self, candidate: Product) -> bool:
        return candidate.rating >= self.min_rating

# Usage example
def find_featured_products(products: List[Product]) -> List[Product]:
    """Find products matching featured criteria."""
    # Build composite specification
    featured = (
        InStockSpecification()
        .and_spec(HighRatedSpecification(4.5))
        .and_spec(PriceRangeSpecification(10, 1000))
    )
    
    return featured.filter(products)
```

**Why**: Specification pattern encapsulates business rules into reusable, composable units. Enables complex queries without scattering logic. Makes validation and filtering declarative and testable.

## 5. Microservices Orchestration

```python
# PROBLEM: Coordinate multiple microservices with saga pattern
# TIME: O(S) for orchestration where S is number of steps
# SPACE: O(L) for saga state where L is log size
from typing import List, Callable, Any, Dict, Optional
from dataclasses import dataclass, field
from enum import Enum
import asyncio
from datetime import datetime

class StepStatus(Enum):
    """Status of saga step."""
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    COMPENSATED = "compensated"

@dataclass
class SagaStep:
    """Single step in saga with compensation."""
    name: str
    action: Callable
    compensate: Callable
    status: StepStatus = StepStatus.PENDING
    result: Any = None
    error: Optional[Exception] = None

@dataclass
class SagaContext:
    """Shared context for saga execution."""
    saga_id: str
    steps: List[SagaStep] = field(default_factory=list)
    current_step: int = 0
    started_at: Optional[datetime] = None
    completed_at: Optional[datetime] = None
    data: Dict[str, Any] = field(default_factory=dict)
    
    def get_result(self, step_name: str) -> Any:
        """Get result from completed step."""
        for step in self.steps:
            if step.name == step_name:
                return step.result
        return None

class SagaOrchestrator:
    """Orchestrates microservices saga with compensation."""
    
    def __init__(self):
        self.active_sagas: Dict[str, SagaContext] = {}
    
    async def execute(self, steps: List[tuple]) -> SagaContext:
        """
        Execute saga steps with automatic compensation on failure.
        
        Args:
            steps: List of (name, action, compensate) tuples
        
        Returns:
            SagaContext with execution results
        """
        saga_id = f"saga_{datetime.now().timestamp()}"
        context = SagaContext(
            saga_id=saga_id,
            steps=[SagaStep(name, action, compensate) 
                   for name, action, compensate in steps]
        )
        
        self.active_sagas[saga_id] = context
        context.started_at = datetime.now()
        
        try:
            # Execute steps forward
            for i, step in enumerate(context.steps):
                context.current_step = i
                step.status = StepStatus.RUNNING
                
                try:
                    # Pass context to action for data sharing
                    step.result = await step.action(context)
                    step.status = StepStatus.COMPLETED
                    context.data[f"{step.name}_result"] = step.result
                
                except Exception as e:
                    step.status = StepStatus.FAILED
                    step.error = e
                    # Trigger compensation
                    await self._compensate(context)
                    raise
            
            context.completed_at = datetime.now()
            return context
        
        finally:
            del self.active_sagas[saga_id]
    
    async def _compensate(self, context: SagaContext) -> None:
        """Execute compensations in reverse order."""
        # Compensate completed steps in reverse
        for step in reversed(context.steps[:context.current_step + 1]):
            if step.status == StepStatus.COMPLETED:
                try:
                    await step.compensate(context)
                    step.status = StepStatus.COMPENSATED
                except Exception as e:
                    # Log compensation failure but continue
                    print(f"Compensation failed for {step.name}: {e}")

# Example: Order processing microservices
class OrderService:

    async def create_order(self, order_data: dict) -> SagaContext:

        # Simulate microservice call

        return {"inventory_id": "inv_123", "available": True}
```

## 6. Service Mesh Sidecar

```python
# PROBLEM: Implement service mesh features: load balancing, circuit breaking, retries
# TIME: O(1) for request routing, O(H) for health checks
# SPACE: O(S) for service registry
from typing import Dict, List, Optional
from dataclasses import dataclass
from enum import Enum
import asyncio
import random
from datetime import datetime, timedelta
import aiohttp

class ServiceStatus(Enum):
    """Service instance health status."""
    HEALTHY = "healthy"
    UNHEALTHY = "unhealthy"
    DRAINING = "draining"

@dataclass
class ServiceInstance:
    """Single service instance."""
    id: str
    host: str
    port: int
    weight: int = 1
    status: ServiceStatus = ServiceStatus.HEALTHY
    last_check: Optional[datetime] = None
    request_count: int = 0
    error_count: int = 0

class ServiceRegistry:
    """Service discovery and health tracking."""
    
    def __init__(self):
        self.services: Dict[str, List[ServiceInstance]] = {}
    
    def register(self, service_name: str, instance: ServiceInstance) -> None:
        """Register service instance."""
        if service_name not in self.services:
            self.services[service_name] = []
        self.services[service_name].append(instance)
    
    def get_healthy(self, service_name: str) -> List[ServiceInstance]:
        """Get healthy instances for service."""
        if service_name not in self.services:
            return []
        return [i for i in self.services[service_name] 
                if i.status == ServiceStatus.HEALTHY]

class LoadBalancer:
    """Load balancing strategies for service selection."""
    
    @staticmethod
    def round_robin(instances: List[ServiceInstance], 
                    last_idx: int) -> tuple:
        """Round-robin selection."""
        if not instances:
            return None, last_idx
        idx = (last_idx + 1) % len(instances)
        return instances[idx], idx
    
    @staticmethod
    def weighted_random(instances: List[ServiceInstance]) -> Optional[ServiceInstance]:
        """Weighted random selection."""
        if not instances:
            return None
        
        total_weight = sum(i.weight for i in instances)
        r = random.uniform(0, total_weight)
        
        cumulative = 0
        for instance in instances:
            cumulative += instance.weight
            if r <= cumulative:
                return instance
        
        return instances[0]
    
    @staticmethod
    def least_connections(instances: List[ServiceInstance]) -> Optional[ServiceInstance]:
        """Select instance with fewest active requests."""
        if not instances:
            return None
        return min(instances, key=lambda i: i.request_count)

class CircuitBreaker:
    """Circuit breaker for fault tolerance."""
    
    def __init__(self, failure_threshold: int = 5, 
                 timeout_seconds: int = 60):
        self.failure_threshold = failure_threshold
        self.timeout_seconds = timeout_seconds
        self.failures = 0
        self.state = "closed"
        self.opened_at: Optional[datetime] = None
    
    async def call(self, func, *args, **kwargs):
        """Execute function with circuit breaker protection."""
        if self.state == "open":
            # Check if timeout elapsed
            if datetime.now() - self.opened_at > timedelta(seconds=self.timeout_seconds):
                self.state = "half-open"
            else:
                raise Exception("Circuit breaker is open")
        
        try:
            result = await func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise e
    
    def _on_success(self):
        """Handle successful call."""
        self.failures = 0
        self.state = "closed"
    
    def _on_failure(self):
        """Handle failed call."""
        self.failures += 1
        if self.failures >= self.failure_threshold:
            self.state = "open"
            self.opened_at = datetime.now()

class RetryPolicy:
    """Retry configuration for transient failures."""
    
    def __init__(self, max_retries: int = 3, 
                 backoff_base: float = 0.1,
                 retryable_errors: List[str] = None):
        self.max_retries = max_retries
        self.backoff_base = backoff_base
        self.retryable_errors = retryable_errors or ["timeout", "connection"]
    
    async def execute(self, func, *args, **kwargs):
        """Execute with retries and exponential backoff."""
        last_error = None
        
        for attempt in range(self.max_retries + 1):
            try:
                return await func(*args, **kwargs)
            except Exception as e:
                last_error = e
                
                # Check if error is retryable
                error_str = str(e).lower()
                if not any(err in error_str for err in self.retryable_errors):
                    raise e
                
                # Exponential backoff
                if attempt < self.max_retries:
                    delay = self.backoff_base * (2 ** attempt)
                    await asyncio.sleep(delay)
        
        raise last_error

class ServiceMeshProxy:
    """Sidecar proxy with mesh features."""
    
    def __init__(self, registry: ServiceRegistry):
        self.registry = registry
        self.circuit_breakers: Dict[str, CircuitBreaker] = {}
        self.retry_policy = RetryPolicy()
        self.lb_indices: Dict[str, int] = {}
    
    async def call(self, service_name: str, endpoint: str, 
                   method: str = "GET", data: dict = None) -> dict:
        """
        Make service call with mesh features:
        - Load balancing
        - Circuit breaking
        - Retries
        - Metrics
        """
        # Get healthy instances
        instances = self.registry.get_healthy(service_name)
        if not instances:
            raise Exception(f"No healthy instances for {service_name}")
        
        # Load balance
        if service_name not in self.circuit_breakers:
            self.circuit_breakers[service_name] = CircuitBreaker()
        
        instance, idx = LoadBalancer.round_robin(
            instances, 
            self.lb_indices.get(service_name, -1)
        )
        self.lb_indices[service_name] = idx
        
        if not instance:
            raise Exception("No instance selected")
        
        # Track request
        instance.request_count += 1
        
        # Build URL
        url = f"http://{instance.host}:{instance.port}{endpoint}"
        
        # Execute with circuit breaker and retries
        circuit = self.circuit_breakers[service_name]
        
        async def make_request():
            async with aiohttp.ClientSession() as session:
                async with session.request(method, url, json=data) as resp:
                    instance.request_count -= 1
                    if resp.status >= 500:
                        instance.error_count += 1
                        raise Exception(f"Server error: {resp.status}")
                    return await resp.json()
        
        try:
            return await circuit.call(
                self.retry_policy.execute,
                make_request
            )
        except Exception as e:
            instance.request_count -= 1
            instance.error_count += 1
            raise e
```

**Why**: Service mesh sidecars handle cross-cutting concerns: discovery, load balancing, circuit breaking, retries. Keeps application code focused on business logic. Essential for microservices reliability.

## 7. Database Transaction Manager

```python
# PROBLEM: Implement transaction isolation levels and concurrency control
# TIME: O(L) for lock acquisition where L is lock depth
# SPACE: O(T) for transaction state
from typing import Dict, Set, Optional, Any
from dataclasses import dataclass, field
from enum import Enum
from datetime import datetime
import threading

class IsolationLevel(Enum):
    """SQL transaction isolation levels."""
    READ_UNCOMMITTED = "read_uncommitted"
    READ_COMMITTED = "read_committed"
    REPEATABLE_READ = "repeatable_read"
    SERIALIZABLE = "serializable"

class TransactionState(Enum):
    """Transaction lifecycle state."""
    ACTIVE = "active"
    COMMITTED = "committed"
    ROLLED_BACK = "rolled_back"

@dataclass
class Lock:
    """Row-level lock."""
    resource_id: str
    transaction_id: str
    lock_type: str  # 'shared' or 'exclusive'
    acquired_at: datetime = field(default_factory=datetime.now)

@dataclass
class Transaction:
    """Database transaction context."""
    id: str
    isolation_level: IsolationLevel
    state: TransactionState = TransactionState.ACTIVE
    locks: Set[str] = field(default_factory=set)
    snapshots: Dict[str, Any] = field(default_factory=dict)
    started_at: datetime = field(default_factory=datetime.now)
    changes: Dict[str, Any] = field(default_factory=dict)

class TransactionManager:
    """Manages transactions with isolation levels and locking."""
    
    def __init__(self):
        self.transactions: Dict[str, Transaction] = {}
        self.locks: Dict[str, Lock] = {}
        self.data_store: Dict[str, Any] = {}
        self.lock_manager = threading.Lock()
        self.version_counter = 0
    
    def begin(self, isolation_level: IsolationLevel = 
              IsolationLevel.READ_COMMITTED) -> str:
        """Start new transaction."""
        tx_id = f"tx_{self.version_counter}"
        self.version_counter += 1
        
        self.transactions[tx_id] = Transaction(
            id=tx_id,
            isolation_level=isolation_level
        )
        
        return tx_id
    
    def read(self, tx_id: str, key: str) -> Optional[Any]:
        """Read data respecting isolation level."""
        tx = self.transactions[tx_id]
        
        if tx.state != TransactionState.ACTIVE:
            raise Exception("Transaction not active")
        
        # Check isolation level for visibility
        if tx.isolation_level == IsolationLevel.READ_UNCOMMITTED:
            # Can read uncommitted changes
            if key in self.data_store:
                return self.data_store[key]
        
        elif tx.isolation_level == IsolationLevel.READ_COMMITTED:
            # Only committed data
            if key in self.data_store:
                # In real DB, check commit status
                return self.data_store[key]
        
        elif tx.isolation_level == IsolationLevel.REPEATABLE_READ:
            # Use snapshot from first read
            if key in tx.snapshots:
                return tx.snapshots[key]
            
            if key in self.data_store:
                tx.snapshots[key] = self.data_store[key]
                return tx.snapshots[key]
        
        elif tx.isolation_level == IsolationLevel.SERIALIZABLE:
            # Acquire shared lock
            self._acquire_lock(tx_id, key, 'shared')
            
            if key in self.data_store:
                return self.data_store[key]
        
        return None
    
    def write(self, tx_id: str, key: str, value: Any) -> None:
        """Write data with exclusive lock."""
        tx = self.transactions[tx_id]
        
        if tx.state != TransactionState.ACTIVE:
            raise Exception("Transaction not active")
        
        # Acquire exclusive lock
        self._acquire_lock(tx_id, key, 'exclusive')
        
        # Store change in transaction
        tx.changes[key] = value
        
        # For READ_UNCOMMITTED, write immediately
        if tx.isolation_level == IsolationLevel.READ_UNCOMMITTED:
            self.data_store[key] = value
    
    def _acquire_lock(self, tx_id: str, resource_id: str, 
                      lock_type: str) -> None:
        """Acquire lock with deadlock prevention."""
        with self.lock_manager:
            # Check for conflicting locks
            if resource_id in self.locks:
                existing = self.locks[resource_id]
                
                if existing.transaction_id != tx_id:
                    # Conflict: exclusive blocks both, shared blocks exclusive
                    if (existing.lock_type == 'exclusive' or 
                        (existing.lock_type == 'shared' and lock_type == 'exclusive')):
                        raise Exception(f"Lock conflict on {resource_id}")
            
            # Acquire lock
            self.locks[resource_id] = Lock(
                resource_id=resource_id,
                transaction_id=tx_id,
                lock_type=lock_type
            )
            
            self.transactions[tx_id].locks.add(resource_id)
    
    def commit(self, tx_id: str) -> None:
        """Commit transaction changes."""
        tx = self.transactions[tx_id]
        
        if tx.state != TransactionState.ACTIVE:
            raise Exception("Transaction not active")
        
        # Apply all changes
        for key, value in tx.changes.items():
            self.data_store[key] = value
        
        # Release locks
        self._release_locks(tx_id)
        
        tx.state = TransactionState.COMMITTED
        tx.changes.clear()
    
    def rollback(self, tx_id: str) -> None:
        """Rollback transaction changes."""
        tx = self.transactions[tx_id]
        
        if tx.state != TransactionState.ACTIVE:
            raise Exception("Transaction not active")
        
        # Discard changes
        tx.changes.clear()
        tx.snapshots.clear()
        
        # Release locks
        self._release_locks(tx_id)
        
        tx.state = TransactionState.ROLLED_BACK
    
    def _release_locks(self, tx_id: str) -> None:
        """Release all locks held by transaction."""
        tx = self.transactions[tx_id]
        
        with self.lock_manager:
            for resource_id in tx.locks:
                if resource_id in self.locks:
                    del self.locks[resource_id]
            
            tx.locks.clear()
```

**Why**: Transaction isolation prevents concurrency anomalies: dirty reads, non-repeatable reads, phantom reads. Locking ensures ACID properties. Essential for financial and critical business systems.

```python
    operation: str  # INSERT, UPDATE, DELETE
    table: str
    data: Dict

class ReplicationManager:
    """Manages master-slave replication."""

        self.master = master
        self.master.is_master = True
        self.master.state = ReplicaState.ONLINE

        self.replication_log: List[WriteOperation] = []

        """Add slave replica."""
        slave.state = ReplicaState.SYNCING
        self.slaves.append(slave)
    
    async def write(self, operation: str, table: str, 
                    data: Dict) -> int:
        """

        """

        seq = self.sequence_counter
        self.sequence_counter += 1
        
        op = WriteOperation(
            sequence=seq,
            operation=operation,
            table=table,
            data=data
        )
        
        # Write to master
        self.replication_log.append(op)
        self.master.position = seq
        
        # Replicate to slaves asynchronously
        await self._replicate_to_slaves(op)
        
        return seq

        """Send operation to all slaves."""
        tasks = []
        
        for slave in self.slaves:
            if slave.state in [ReplicaState.ONLINE, ReplicaState.SYNCING]:
                tasks.append(self._send_to_slave(slave, op))
        
        # Wait for all replications
        await asyncio.gather(*tasks, return_exceptions=True)

                            op: WriteOperation) -> None:
        """Send single operation to slave."""
        try:
            # Simulate network call
            await asyncio.sleep(0.05)
            
            # Update slave position
            slave.position = op.sequence
            slave.last_sync = datetime.now()
            slave.state = ReplicaState.ONLINE
            
        except Exception as e:
            slave.state = ReplicaState.OFFLINE
            print(f"Replication failed to {slave.id}: {e}")

                   max_lag_seconds: float = 10.0) -> Dict:

        Read with routing preference.

            query: SQL query
            read_preference: 'master', 'slave', 'nearest'

            Query results

            return await self._read_from_master(query)
        
        elif read_preference == "slave":
            # Find suitable slave
            slave = self._select_slave(max_lag_seconds)
            if slave:
                return await self._read_from_slave(slave, query)
            else:
                # Fallback to master
                return await self._read_from_master(query)
        
        elif read_preference == "nearest":
            # Choose based on latency (simplified: use first healthy slave)
            slave = self._select_slave(max_lag_seconds)
            if slave:
                return await self._read_from_slave(slave, query)
            return await self._read_from_master(query)
        
        raise ValueError(f"Invalid read preference: {read_preference}")
    
    def _select_slave(self, max_lag_seconds: float) -> Optional[ReplicaNode]:
        """Select healthy slave within lag tolerance."""

            s for s in self.slaves
            if s.state == ReplicaState.ONLINE 
            and s.lag_seconds <= max_lag_seconds
        ]
        
        if not candidates:
            return None
        
        # Select slave with lowest lag
        return min(candidates, key=lambda s: s.lag_seconds)

        await asyncio.sleep(0.02)  # Simulate query

                               query: str) -> Dict:

        await asyncio.sleep(0.02)
        return {"source": f"slave:{slave.id}", "data": {}}
    
    def check_replication_health(self) -> Dict:
        """Check replication status."""
        return {
            "master": {
                "id": self.master.id,
                "state": self.master.state.value,
                "position": self.master.position
            },
            "slaves": [

                    "state": s.state.value,
                    "lag_seconds": s.lag_seconds,

                for s in self.slaves
            ],
            "log_size": len(self.replication_log)
        }
```

```python
from abc import ABC, abstractmethod
from typing import TypeVar, Generic, Type, Dict, Any, List, ClassVar
from dataclasses import dataclass
import inspect

class HookMixin:

    def post_create(self) -> None:

        self.left = left
        self.right = right

    def __init__(self, left: Specification[T], right: Specification[T]):

        self.right = right
    
    def is_satisfied_by(self, candidate: T) -> bool:
        return self.left.is_satisfied_by(candidate) or \
               self.right.is_satisfied_by(candidate)

class NotSpec(Specification[T]):
    def __init__(self, spec: Specification[T]):
        self.spec = spec
    
    def is_satisfied_by(self, candidate: T) -> bool:
        return not self.spec.is_satisfied_by(candidate)

# Concrete implementation
@dataclass
class User(Entity):
    """User entity."""
    
    id: str
    username: str
    email: str
    active: bool = True
    
    def get_id(self) -> str:
        return self.id
    
    def validate(self) -> List[str]:
        errors = super().validate()
        
        if '@' not in self.email:
            errors.append("Invalid email format")
        
        if len(self.username) < 3:
            errors.append("Username too short")
        
        return errors

class UserRepository(Repository[User]):
    """User repository implementation."""
    
    def __init__(self):
        self._storage: Dict[str, User] = {}
        self._entity_class = User
    
    async def find_by_id(self, id: str) -> Optional[User]:
        return self._storage.get(id)
    
    async def find_all(self) -> List[User]:
        return list(self._storage.values())
    
    async def save(self, user: User) -> User:
        user.pre_save()
        self._storage[user.id] = user
        user.post_save()
        return user
    
    async def delete(self, id: str) -> bool:
        if id in self._storage:
            del self._storage[id]
            return True
        return False
    
    async def _do_create(self, **kwargs) -> User:
        return User(**kwargs)
    
    async def find_by_spec(self, spec: Specification[User]) -> List[User]:
        """Find users matching specification."""
        return [u for u in self._storage.values() 
                if spec.is_satisfied_by(u)]

# Specifications
class ActiveUserSpec(Specification[User]):
    def is_satisfied_by(self, user: User) -> bool:
        return user.active

class EmailDomainSpec(Specification[User]):
    def __init__(self, domain: str):
        self.domain = domain
    
    def is_satisfied_by(self, user: User) -> bool:
        return user.email.endswith(f"@{self.domain}")
```

**Why**: ABC with mixins provides reusable functionality without multiple inheritance complexity. Specifications enable flexible queries. Repository pattern abstracts data access. Essential for DDD and clean architecture.
