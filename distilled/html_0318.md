```html
    <h1>Favicon Management Demo</h1>
    
    <button onclick="changeFavicon('/favicon-dark.svg')">Dark Mode Favicon</button>
    <button onclick="changeFavicon('/favicon.svg')">Light Mode Favicon</button>

    <script>
        // Dynamically change favicon
        function changeFavicon(href) {
            const link = document.querySelector("link[rel*='icon']") || 
                         document.createElement('link');
            link.type = 'image/svg+xml';
            link.rel = 'icon';
            link.href = href;
            document.head.appendChild(link);
        }

        // Auto-switch favicon based on color scheme
        function setupAutoFavicon() {
            const darkQuery = window.matchMedia('(prefers-color-scheme: dark)');
            
            const updateFavicon = (isDark) => {
                changeFavicon(isDark ? '/favicon-dark.svg' : '/favicon.svg');
            };
            
            updateFavicon(darkQuery.matches);
            darkQuery.addEventListener('change', (e) => updateFavicon(e.matches));
        }

        // Validate favicon exists
        async function validateFavicon(href) {
            try {
                const response = await fetch(href, { method: 'HEAD' });
                return response.ok;
            } catch {
                return false;
            }
        }

        setupAutoFavicon();
    </script>
</body>
</html>
```
**Why**: Favicons appear in browser tabs, bookmarks, and mobile home screens. Use multiple sizes/formats for cross-device compatibility. SVG favicons are resolution-independent.

## css_selector_attribute_contains
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Attribute Contains Selector Demo</title>
    <style>
        /* PROBLEM: Select elements with attribute containing specific text
           TIME: O(n) SPACE: O(1) */
        
        /* Contains substring anywhere */
        [class*="button"] {
            background: #3498db;
            color: white;
            padding: 10px 20px;
            border: none;
            border-radius: 4px;
            margin: 5px;
            cursor: pointer;
        }

        /* Contains word (space-separated) */
        [class~="primary"] {
            background: #2ecc71;
        }

        /* Starts with prefix */
        [class^="btn-"] {
            font-weight: bold;
        }

        /* Ends with suffix */
        [class$="-large"] {
            padding: 15px 30px;
            font-size: 1.2rem;
        }

        /* Starts with value or value- */
        [class|="icon"] {
            display: inline-flex;
            align-items: center;
        }

        /* Case-insensitive matching */
        [data-type*="card" i] {
            border: 2px solid #9b59b6;
            padding: 15px;
            margin: 10px 0;
        }
    </style>
</head>
<body>
    <h2>Attribute Contains Selectors</h2>
    
    <button class="my-button">Contains "button"</button>
    <button class="btn-primary">Starts with "btn-" and contains word "primary"</button>
    <button class="btn-large">Starts with "btn-", ends with "-large"</button>
    
    <div class="icon-home">Icon (|= match)</div>
    <div class="icon-user">Another icon</div>
    
    <div data-type="Card">Card (case-insensitive)</div>
    <div data-type="profile-card">Profile Card</div>

    <script>
        // Query elements with attribute containing text
        function getByAttributeContains(attr, value) {
            return document.querySelectorAll(`[${attr}*="${value}"]`);
        }

        // Query with case-insensitive flag
        function getByAttributeContainsCI(attr, value) {
            return document.querySelectorAll(`[${attr}*="${value}" i]`);
        }

        // Check if element matches attribute selector
        function hasAttributeContaining(element, attr, value) {
            const attrValue = element.getAttribute(attr) || '';
            return attrValue.includes(value);
        }

        // Usage examples
        console.log('Buttons:', getByAttributeContains('class', 'button').length);
        console.log('Cards:', getByAttributeContainsCI('data-type', 'card').length);
    </script>
</body>
</html>
```
**Why**: Attribute selectors with `*=` match any element whose attribute contains the substring. Combine with `~=` (word), `^=` (prefix), `$=` (suffix), `|=` (prefix or prefix-) for precise targeting.

```html
        // Convert HSLA to RGBA
        function hslaToRgba(h, s, l, a) {
            s /= 100;
            l /= 100;
            const k = n => (n + h / 30) % 12;
            const f = n => l - s * Math.min(l, 1 - l) * Math.max(-1, Math.min(k(n) - 3, Math.min(9 - k(n), 1)));
            return {
                r: Math.round(255 * f(0)),
                g: Math.round(255 * f(8)),
                b: Math.round(255 * f(4)),
                a: a
            };
        }

        // Generate HSLA color string
        function hsla(h, s, l, a = 1) {
            return `hsla(${h}, ${s}%, ${l}%, ${a})`;
        }

        // Create color palette variations
        function createPalette(hue, steps = 5) {
            const palette = [];
            for (let i = 0; i < steps; i++) {
                const lightness = 20 + (i * 60 / (steps - 1));
                palette.push(hsla(hue, 80, lightness));
            }
            return palette;
        }

        // Example: Generate complementary colors
        console.log('Complementary:', hsla(0, 100, 50), hsla(180, 100, 50));
        console.log('Palette:', createPalette(210));

```
**Why**: HSLA separates hue, saturation, and lightness making it easier to create color variations. Alpha adds transparency. More intuitive than RGB for color manipulation and theming.

## css_box_sizing_content_box
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Box Sizing Demo</title>
    <style>
        /* PROBLEM: Control how width/height includes padding and border
           TIME: O(1) SPACE: O(1) */
        
        .container {
            width: 300px;
            background: #f0f0f0;
            margin-bottom: 20px;
        }

        /* content-box: width = content only (default) */
        .content-box {
            box-sizing: content-box;
            width: 200px;
            padding: 20px;
            border: 5px solid #3498db;
            background: #e74c3c;
            color: white;
        }

        /* border-box: width = content + padding + border */
        .border-box {
            box-sizing: border-box;
            width: 200px;
            padding: 20px;
            border: 5px solid #2ecc71;
            background: #9b59b6;
            color: white;
        }

        /* Best practice: apply border-box to all elements */
        *, *::before, *::after {
            box-sizing: border-box;
        }

        .comparison {
            display: flex;
            gap: 20px;
            margin-top: 20px;
        }

        .comparison > div {
            width: 150px;
            padding: 20px;
            border: 10px solid #333;
            background: #3498db;
            color: white;
        }

        .comparison .content { box-sizing: content-box; }
        .comparison .border { box-sizing: border-box; }

        .info {
            font-size: 12px;
            color: #666;
            margin-top: 5px;
        }
    </style>
</head>
<body>
    <h2>Box Sizing Comparison</h2>
    
    <div class="container">
        <div class="content-box">
            Content Box
            <div class="info">Total: 250px (200 + 20*2 + 5*2)</div>
        </div>
    </div>

    <div class="container">
        <div class="border-box">
            Border Box
            <div class="info">Total: 200px (fixed width)</div>
        </div>
    </div>

    <h2>Side by Side</h2>
    <div class="comparison">
        <div class="content">
            Content Box<br>
            <small>Wider than 150px</small>
        </div>
        <div class="border">
            Border Box<br>
            <small>Exactly 150px</small>
        </div>
    </div>

    <script>
        // Calculate actual element size
        function getActualSize(element) {
            const styles = window.getComputedStyle(element);
            const boxSizing = styles.boxSizing;
            
            const width = parseFloat(styles.width);
            const padding = parseFloat(styles.paddingLeft) + parseFloat(styles.paddingRight);
            const border = parseFloat(styles.borderLeftWidth) + parseFloat(styles.borderRightWidth);
            
            return {
                boxSizing,
                contentWidth: width,
                paddingWidth: padding,
                borderWidth: border,
                totalWidth: boxSizing === 'border-box' 
                    ? width 
                    : width + padding + border
            };
        }

        // Toggle box-sizing
        function toggleBoxSizing(element) {
            const current = window.getComputedStyle(element).boxSizing;
            element.style.boxSizing = current === 'border-box' ? 'content-box' : 'border-box';
        }

        // Demo
        const contentBox = document.querySelector('.content-box');
        const borderBox = document.querySelector('.border-box');
        
        console.log('Content box:', getActualSize(contentBox));
        console.log('Border box:', getActualSize(borderBox));
    </script>
</body>
</html>
```
**Why**: `box-sizing: content-box` (default) adds padding/border to width, causing layout issues. `border-box` includes them in the declared width, making layouts predictable. Use `*, *::before, *::after { box-sizing: border-box; }` as a global reset.

```html
    <script>
        // Apply translateY transform
        function translateY(element, distance) {
            element.style.transform = `translateY(${distance}px)`;
        }

        // Get current translateY value
        function getTranslateY(element) {
            const style = window.getComputedStyle(element);
            const matrix = new DOMMatrix(style.transform);
            return matrix.m42; // Y translation
        }

        // Animate translateY
        function animateTranslateY(element, from, to, duration = 300) {
            element.style.transition = `transform ${duration}ms ease`;
            element.style.transform = `translateY(${from}px)`;
            
            requestAnimationFrame(() => {
                requestAnimationFrame(() => {
                    element.style.transform = `translateY(${to}px)`;
                });
            });
        }

        // Bounce effect
        function bounce(element, height = 20, times = 3) {
            let count = 0;
            const interval = setInterval(() => {
                const y = count % 2 === 0 ? -height : 0;
                element.style.transform = `translateY(${y}px)`;
                count++;
                if (count >= times * 2) {
                    clearInterval(interval);
                    element.style.transform = 'translateY(0)';
                }
            }, 150);
        }

        // Smooth scroll indicator
        const hoverBox = document.querySelector('.hover-lift');
        console.log('Current Y:', getTranslateY(hoverBox));
    </script>
```

**Why**: `translateY()` moves elements vertically without affecting document flow. GPU-accelerated for smooth animations. Common uses: hover effects, centering tricks, entrance animations. Combine with `translateX()` or use `translate()` for both axes.
