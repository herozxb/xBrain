## 1. Zone Map Implementation

```python
# PROBLEM: Efficiently skip irrelevant data blocks using min/max statistics
# TIME: O(n) for building, O(1) for query TIME: O(?) SPACE: O(?)
class ZoneMap:
    """
    Zone map indexes data blocks by maintaining min/max statistics.
    Enables predicate pushdown to skip entire blocks during queries.
    """
    
    def __init__(self, block_size: int = 1000):
        self.block_size = block_size
        self.zones = []  # List of (min_val, max_val, block_id)
    
    def build(self, data: list) -> None:
        """Build zone map from sorted or unsorted data."""
        for i in range(0, len(data), self.block_size):
            block = data[i:i + self.block_size]
            self.zones.append({
                'min_val': min(block),
                'max_val': max(block),
                'block_id': len(self.zones),
                'start_idx': i,
                'end_idx': min(i + self.block_size - 1, len(data) - 1)
            })
    
    def get_relevant_blocks(self, op: str, value: any) -> list:
        """
        Return block IDs that might contain rows matching the predicate.
        op: '<', '>', '=', '<=', '>=', 'between'
        """
        relevant = []
        
        for zone in self.zones:
            min_v, max_v = zone['min_val'], zone['max_val']
            
            # Check if predicate can possibly match this zone
            if op == '=':
                if min_v <= value <= max_v:
                    relevant.append(zone['block_id'])
            elif op == '<':
                if min_v < value:
                    relevant.append(zone['block_id'])
            elif op == '>':
                if max_v > value:
                    relevant.append(zone['block_id'])
            elif op == '<=':
                if min_v <= value:
                    relevant.append(zone['block_id'])
            elif op == '>=':
                if max_v >= value:
                    relevant.append(zone['block_id'])
        
        return relevant
    
    def get_stats(self) -> dict:
        """Return zone map statistics."""
        return {
            'total_zones': len(self.zones),
            'block_size': self.block_size
        }
```

**Why**: Zone maps enable databases to skip entire blocks of data during query execution, dramatically reducing I/O for analytical workloads. Used in columnar databases like Parquet.

## 2. Row Storage Implementation

```python
# PROBLEM: Store and retrieve data in row-oriented format for OLTP workloads
# TIME: O(1) for insert, O(n) for column scan TIME: O(?) SPACE: O(?)
class RowStorage:
    """
    Row-oriented storage engine optimized for transactional operations.
    Stores complete records together for efficient point queries.
    """
    
    def __init__(self, schema: dict):
        """
        schema: {'id': 'int', 'name': 'str', 'age': 'int'}
        """
        self.schema = schema
        self.rows = []
        self.index = {}  # Primary key index
        self.deleted = set()  # Track deleted rows
    
    def insert(self, row: dict) -> int:
        """Insert a row and return row_id."""
        # Validate schema
        for col, dtype in self.schema.items():
            if col not in row:
                raise ValueError(f"Missing column: {col}")
        
        row_id = len(self.rows)
        self.rows.append(row.copy())
        
        # Build index on first column (assuming it's PK)
        pk = row[list(self.schema.keys())[0]]
        self.index[pk] = row_id
        
        return row_id
    
    def get_by_pk(self, pk: any) -> dict:
        """Retrieve row by primary key - O(1)."""
        if pk not in self.index:
            return None
        
        row_id = self.index[pk]
        if row_id in self.deleted:
            return None
        
        return self.rows[row_id].copy()
    
    def update(self, pk: any, updates: dict) -> bool:
        """Update row by primary key."""
        if pk not in self.index:
            return False
        
        row_id = self.index[pk]
        if row_id in self.deleted:
            return False
        
        for col, val in updates.items():
            if col in self.schema:
                self.rows[row_id][col] = val
        
        return True
    
    def delete(self, pk: any) -> bool:
        """Soft delete row by primary key."""
        if pk not in self.index:
            return False
        
        row_id = self.index[pk]
        self.deleted.add(row_id)
        return True
    
    def scan_column(self, col_name: str) -> list:
        """Scan and return all values for a column - O(n)."""
        if col_name not in self.schema:
            return []
        
        return [
            row[col_name] 
            for i, row in enumerate(self.rows) 
            if i not in self.deleted
        ]
```

**Why**: Row storage is optimal for OLTP workloads with frequent point queries and updates. Entire records are stored together, minimizing disk seeks for transactional operations.

## 3. Delta Encoding

```python
# PROBLEM: Compress sorted data by storing differences between consecutive values
# TIME: O(n) encode/decode TIME: O(?) SPACE: O(?)
class DeltaEncoding:
    """
    Delta encoding compresses sequences by storing differences.
    Effective for timestamp series, sorted IDs, and incremental values.
    """
    
    @staticmethod
    def encode(data: list) -> dict:
        """
        Encode sorted data using delta compression.
        Returns base value and deltas.
        """
        if not data:
            return {'base': None, 'deltas': []}
        
        # Ensure data is sorted for optimal compression
        sorted_data = sorted(data)
        
        base = sorted_data[0]
        deltas = []
        
        for i in range(1, len(sorted_data)):
            delta = sorted_data[i] - sorted_data[i-1]
            deltas.append(delta)
        
        return {
            'base': base,
            'deltas': deltas,
            'original_length': len(data)
        }
    
    @staticmethod
    def decode(encoded: dict) -> list:
        """Decode delta-encoded data back to original values."""
        if encoded['base'] is None:
            return []
        
        result = [encoded['base']]
        current = encoded['base']
        
        for delta in encoded['deltas']:
            current += delta
            result.append(current)
        
        return result
    
    @staticmethod
    def encode_with_dictionary(data: list) -> dict:
        """
        Delta encoding with dictionary compression for repeated deltas.
        Best for data with repeating patterns.
        """
        if not data:
            return {'base': None, 'dict': [], 'indices': []}
        
        sorted_data = sorted(data)
        base = sorted_data[0]
        
        # Build delta dictionary
        delta_dict = {}
        dict_list = []
        indices = []
        
        for i in range(1, len(sorted_data)):
            delta = sorted_data[i] - sorted_data[i-1]
            
            if delta not in delta_dict:
                delta_dict[delta] = len(dict_list)
                dict_list.append(delta)
            
            indices.append(delta_dict[delta])
        
        return {
            'base': base,
            'dict': dict_list,
            'indices': indices
        }
```

**Why**: Delta encoding achieves high compression ratios for sorted or incremental data like timestamps, IDs, and sensor readings. Used in columnar databases and time-series storage.

## 4. Bit Packing

```python
# PROBLEM: Store integers using minimum bits required per value
# TIME: O(n) pack/unpack TIME: O(?) SPACE: O(?)
class BitPacking:
    """
    Bit packing compresses integers by using only required bits.
    Essential for columnar databases and memory-efficient storage.
    """
    
    @staticmethod
    def calculate_bits_needed(max_val: int) -> int:
        """Calculate bits needed to represent values up to max_val."""
        if max_val == 0:
            return 1
        return max_val.bit_length()
    
    @staticmethod
    def pack(values: list) -> dict:
        """
        Pack list of integers into compact bit array.
        Returns packed bytes and metadata.
        """
        if not values:
            return {'data': b'', 'bits_per_value': 0, 'count': 0}
        
        max_val = max(values)
        bits_per_value = BitPacking.calculate_bits_needed(max_val)
        
        # Calculate total bits and bytes needed
        total_bits = len(values) * bits_per_value
        total_bytes = (total_bits + 7) // 8
        
        # Pack values into bytes
        packed = bytearray(total_bytes)
        bit_offset = 0
        
        for val in values:
            # Write bits_per_value bits starting at bit_offset
            for i in range(bits_per_value):
                if val & (1 << i):
                    byte_idx = (bit_offset + i) // 8
                    bit_idx = 7 - ((bit_offset + i) % 8)
                    packed[byte_idx] |= (1 << bit_idx)
            
            bit_offset += bits_per_value
        
        return {
            'data': bytes(packed),
            'bits_per_value': bits_per_value,
            'count': len(values)
        }
    
    @staticmethod
    def unpack(packed: dict) -> list:
        """Unpack bit-packed data back to integer list."""
        if packed['count'] == 0:
            return []
        
        data = packed['data']
        bits_per_value = packed['bits_per_value']
        count = packed['count']
        
        values = []
        bit_offset = 0
        
        for _ in range(count):
            val = 0
            
            # Read bits_per_value bits
            for i in range(bits_per_value):
                byte_idx = (bit_offset + i) // 8
                bit_idx = 7 - ((bit_offset + i) % 8)
                
                if byte_idx < len(data) and (data[byte_idx] & (1 << bit_idx)):
                    val |= (1 << i)
            
            values.append(val)
            bit_offset += bits_per_value
        
        return values
    
    @staticmethod
    def compression_ratio(values: list) -> float:
        """Calculate compression ratio for given values."""
        if not values:
            return 0.0
        
        original_size = len(values) * 32  # Assume 32-bit integers
        packed = BitPacking.pack(values)
        compressed_size = len(packed['data']) * 8
        
        return original_size / compressed_size if compressed_size > 0 else 0.0
```

**Why**: Bit packing reduces memory footprint by 2-8x for integer columns with limited value ranges. Fundamental optimization in columnar databases and data warehouses.

## 5. NamedTuple Patterns

```python
# PROBLEM: Create memory-efficient immutable data structures with named fields
# TIME: O(1) access TIME: O(?) SPACE: O(?)
from collections import namedtuple
from typing import NamedTuple, Optional

class Point(NamedTuple):
    """Immutable 2D point with named coordinates."""
    x: float
    y: float
    
    def distance_to(self, other: 'Point') -> float:
        """Calculate Euclidean distance to another point."""
        return ((self.x - other.x) ** 2 + (self.y - other.y) ** 2) ** 0.5
    
    def translate(self, dx: float, dy: float) -> 'Point':
        """Return new translated point (immutable)."""
        return Point(self.x + dx, self.y + dy)

class Record(NamedTuple):
    """Database record with optional fields."""
    id: int
    name: str
    value: Optional[float] = None
    
    def with_value(self, new_value: float) -> 'Record':
        """Return new record with updated value."""
        return Record(self.id, self.name, new_value)

# Factory pattern with namedtuple
def create_point_tuple():
    """Factory creating different namedtuple types."""
    Point2D = namedtuple('Point2D', ['x', 'y'])
    Point3D = namedtuple('Point3D', ['x', 'y', 'z'])
    
    return Point2D, Point3D

# Inheritance pattern with namedtuple base
class ColorPoint(Point):
    """Point with color - extends base namedtuple."""
    
    def __new__(cls, x: float, y: float, color: str = 'black'):
        self = super().__new__(cls, x, y)
        self.color = color
        return self
    
    def __repr__(self):
        return f'ColorPoint(x={self.x}, y={self.y}, color={self.color})'

# Usage patterns
def namedtuple_patterns():
    """Demonstrate various namedtuple patterns."""
    # Creation
    p1 = Point(3.0, 4.0)
    p2 = Point(x=1.0, y=2.0)
    
    # Access patterns
    x, y = p1  # Unpacking
    coord = p1[0]  # Index access
    x_val = p1.x  # Named access
    
    # Methods
    dist = p1.distance_to(p2)
    p3 = p1.translate(1.0, 1.0)
    
    # Memory comparison
    import sys
    tuple_size = sys.getsizeof((3.0, 4.0))
    namedtuple_size = sys.getsizeof(p1)
    
    return {
        'point': p1,
        'distance': dist,
        'translated': p3,
        'memory_overhead': namedtuple_size - tuple_size
    }
```

**Why**: NamedTuples provide immutable, memory-efficient structures with named fields. Ideal for data records, coordinates, and return values where immutability and clarity matter.

## 6. Adapter Pattern

```python
# PROBLEM: Make incompatible interfaces work together without modifying existing code
# TIME: O(1) for adaptation TIME: O(?) SPACE: O(?)
from abc import ABC, abstractmethod

# Target interface expected by client
class MediaPlayer(ABC):
    @abstractmethod
    def play(self, filename: str) -> str:
        pass

# Adaptee - existing class with incompatible interface
class AdvancedMediaPlayer:
    """Legacy media player with different interface."""
    
    def play_vlc(self, filename: str) -> str:
        return f"Playing VLC file: {filename}"
    
    def play_mp4(self, filename: str) -> str:
        return f"Playing MP4 file: {filename}"

# Adapter - bridges the gap
class MediaAdapter(MediaPlayer):
    """Adapter makes AdvancedMediaPlayer compatible with MediaPlayer."""
    
    def __init__(self, audio_type: str):
        self.advanced_player = AdvancedMediaPlayer()
        self.audio_type = audio_type
    
    def play(self, filename: str) -> str:
        """Adapt play() to appropriate advanced method."""
        if self.audio_type.lower() == 'vlc':
            return self.advanced_player.play_vlc(filename)
        elif self.audio_type.lower() == 'mp4':
            return self.advanced_player.play_mp4(filename)
        else:
            return f"Unsupported format: {self.audio_type}"

# Object adapter pattern
class LegacyDatabase:
    """Legacy database with incompatible query interface."""
    
    def execute_query(self, sql: str) -> list:
        return [f"Result for: {sql}"]

class ModernDatabaseInterface(ABC):
    @abstractmethod
    def find(self, table: str, conditions: dict) -> list:
        pass

class DatabaseAdapter(ModernDatabaseInterface):
    """Adapter converting modern interface to legacy."""
    
    def __init__(self, legacy_db: LegacyDatabase):
        self.legacy_db = legacy_db
    
    def find(self, table: str, conditions: dict) -> list:
        """Convert modern query to legacy SQL."""
        where_clauses = [f"{k}='{v}'" for k, v in conditions.items()]
        where = " AND ".join(where_clauses)
        sql = f"SELECT * FROM {table} WHERE {where}"
        return self.legacy_db.execute_query(sql)

# Two-way adapter
class TwoWayAdapter(MediaPlayer, AdvancedMediaPlayer):
    """Adapter working in both directions."""
    
    def play(self, filename: str) -> str:
        if filename.endswith('.vlc'):
            return self.play_vlc(filename)
        elif filename.endswith('.mp4'):
            return self.play_mp4(filename)
        return f"Unknown format: {filename}"
```

**Why**: Adapter pattern enables integration of legacy systems, third-party libraries, and different APIs without modifying existing code. Promotes code reuse and backward compatibility.

## 7. Proxy Pattern

```python
# PROBLEM: Control access to an object with a surrogate or placeholder
# TIME: O(1) overhead TIME: O(?) SPACE: O(?)
import time
from abc import ABC, abstractmethod

# Subject interface
class Image(ABC):
    @abstractmethod
    def display(self) -> str:
        pass
    
    @abstractmethod
    def get_info(self) -> dict:
        pass

# Real subject
class RealImage(Image):
    """Actual image that's expensive to load."""
    
    def __init__(self, filename: str):
        self.filename = filename
        self._load_from_disk()
    
    def _load_from_disk(self) -> None:
        """Simulate expensive loading operation."""
        time.sleep(0.1)  # Simulate disk I/O
        self.data = f"Image data for {self.filename}"
        self.size = len(self.data)
    
    def display(self) -> str:
        return f"Displaying: {self.data}"
    
    def get_info(self) -> dict:
        return {
            'filename': self.filename,
            'size': self.size
        }

# Virtual Proxy - lazy loading
class ImageProxy(Image):
    """Proxy that loads real image on first use."""
    
    def __init__(self, filename: str):
        self.filename = filename
        self._real_image = None
    
    def _ensure_loaded(self) -> None:
        """Lazy initialization."""
        if self._real_image is None:
            self._real_image = RealImage(self.filename)
    
    def display(self) -> str:
        self._ensure_loaded()
        return self._real_image.display()
    
    def get_info(self) -> dict:
        # Return basic info without loading
        if self._real_image:
            return self._real_image.get_info()
        return {'filename': self.filename, 'size': 'not loaded'}

# Protection Proxy - access control
class ProtectedImageProxy(Image):
    """Proxy with access control."""
    
    def __init__(self, filename: str, allowed_users: list):
        self.real_image = RealImage(filename)
        self.allowed_users = allowed_users
        self.current_user = None
    
    def set_user(self, username: str) -> None:
        self.current_user = username
    
    def display(self) -> str:
        if self.current_user not in self.allowed_users:
            return "Access denied"
        return self.real_image.display()
    
    def get_info(self) -> dict:
        return self.real_image.get_info()

# Caching Proxy
class CachingProxy:
    """Proxy that caches results."""
    
    def __init__(self, service):
        self.service = service
        self.cache = {}
    
    def request(self, key: str) -> any:
        if key not in self.cache:
            self.cache[key] = self.service.request(key)
        return self.cache[key]
    
    def clear_cache(self) -> None:
        self.cache.clear()

# Smart Proxy - reference counting
class SmartProxy:
    """Proxy with reference counting and cleanup."""
    
    instances = {}
    ref_counts = {}
    
    def __new__(cls, key: str):
        if key not in cls.instances:
            cls.instances[key] = super().__new__(cls)
            cls.instances[key].key = key
            cls.ref_counts[key] = 0
        
        cls.ref_counts[key] += 1
        return cls.instances[key]
    
    def __del__(self):
        cls = self.__class__
        if self.key in cls.ref_counts:
            cls.ref_counts[self.key] -= 1
            if cls.ref_counts[self.key] == 0:
                del cls.instances[self.key]
                del cls.ref_counts[self.key]
```

**Why**: Proxy pattern provides controlled access to objects, enabling lazy loading, access control, caching, and logging without changing the real object's code.

## 8. Chain of Responsibility Pattern

```python
# PROBLEM: Pass requests through a chain of handlers until one handles it
# TIME: O(n) worst case for chain traversal TIME: O(?) SPACE: O(?)
from abc import ABC, abstractmethod

# Handler interface
class Handler(ABC):
    def __init__(self):
        self._next_handler = None
    
    def set_next(self, handler: 'Handler') -> 'Handler':
        """Set next handler in chain."""
        self._next_handler = handler
        return handler
    
    def handle(self, request: dict) -> str:
        """Handle request or pass to next handler."""
        result = self._handle_request(request)
        
        if result is None and self._next_handler:
            return self._next_handler.handle(request)
        
        return result or "Request not handled"
    
    @abstractmethod
    def _handle_request(self, request: dict) -> str:
        pass

# Concrete handlers
class AuthenticationHandler(Handler):
    def _handle_request(self, request: dict) -> str:
        if 'token' not in request:
            return "Authentication failed: No token"
        
        if not self._validate_token(request['token']):
            return "Authentication failed: Invalid token"
        
        return None  # Pass to next
    
    def _validate_token(self, token: str) -> bool:
        return token.startswith('valid_')

class AuthorizationHandler(Handler):
    def _handle_request(self, request: dict) -> str:
        if request.get('role') != 'admin':
            return "Authorization failed: Admin role required"
        
        return None  # Pass to next

class RateLimitHandler(Handler):
    def __init__(self, max_requests: int = 10):
        super().__init__()
        self.max_requests = max_requests
        self.request_counts = {}
    
    def _handle_request(self, request: dict) -> str:
        user_id = request.get('user_id', 'anonymous')
        
        if user_id not in self.request_counts:
            self.request_counts[user_id] = 0
        
        self.request_counts[user_id] += 1
        
        if self.request_counts[user_id] > self.max_requests:
            return f"Rate limit exceeded for user {user_id}"
        
        return None  # Pass to next

class LoggingHandler(Handler):
    def _handle_request(self, request: dict) -> str:
        print(f"[LOG] Processing request: {request.get('action', 'unknown')}")
        return None  # Always pass to next

class BusinessLogicHandler(Handler):
    def _handle_request(self, request: dict) -> str:
        action = request.get('action')
        
        if action == 'create':
            return f"Created resource: {request.get('data')}"
        elif action == 'delete':
            return f"Deleted resource: {request.get('resource_id')}"
        else:
            return f"Executed action: {action}"

# Chain builder
def build_handler_chain() -> Handler:
    """Build and return the handler chain."""
    auth = AuthenticationHandler()
    authz = AuthorizationHandler()
    rate_limit = RateLimitHandler()
    logger = LoggingHandler()
    business = BusinessLogicHandler()
    
    # Build chain: auth -> authz -> rate_limit -> logger -> business
    auth.set_next(authz).set_next(rate_limit).set_next(logger).set_next(business)
    
    return auth

# Usage
def process_request(request: dict) -> str:
    chain = build_handler_chain()
    return chain.handle(request)
```

**Why**: Chain of Responsibility decouples senders from receivers, allows dynamic reconfiguration of handling logic, and follows single responsibility principle for each handler.

## 9. Mediator Pattern

```python
# PROBLEM: Define centralized communication between objects to reduce coupling
# TIME: O(1) for simple routing TIME: O(?) SPACE: O(?)
from abc import ABC, abstractmethod
from typing import Dict, List, Callable

# Mediator interface
class Mediator(ABC):
    @abstractmethod
    def notify(self, sender: object, event: str, data: any) -> None:
        pass

# Concrete mediator
class ChatMediator(Mediator):
    """Mediator for chat room communication."""
    
    def __init__(self):
        self.users: Dict[str, 'User'] = {}
    
    def add_user(self, user: 'User') -> None:
        self.users[user.name] = user
    
    def notify(self, sender: object, event: str, data: any) -> None:
        if event == 'message':
            # Broadcast to all users except sender
            sender_name = sender.name if hasattr(sender, 'name') else 'Unknown'
            for name, user in self.users.items():
                if name != sender_name:
                    user.receive(data['message'], sender_name)
        
        elif event == 'private_message':
            # Send to specific user
            recipient = data.get('to')
            if recipient in self.users:
                sender_name = sender.name if hasattr(sender, 'name') else 'Unknown'
                self.users[recipient].receive(data['message'], sender_name, private=True)

# Colleague base class
class User:
    def __init__(self, mediator: Mediator, name: str):
        self.mediator = mediator
        self.name = name
        self.messages: List[dict] = []
        mediator.add_user(self) if hasattr(mediator, 'add_user') else None
    
    def send(self, message: str) -> None:
        """Send message to all users via mediator."""
        print(f"{self.name} sends: {message}")
        self.mediator.notify(self, 'message', {'message': message})
    
    def send_private(self, message: str, to: str) -> None:
        """Send private message via mediator."""
        self.mediator.notify(self, 'private_message', {
            'message': message,
            'to': to
        })
    
    def receive(self, message: str, from_user: str, private: bool = False) -> None:
        """Receive message."""
        msg_type = "Private" if private else "Public"
        print(f"{self.name} received {msg_type} from {from_user}: {message}")
        self.messages.append({
            'from': from_user,
            'message': message,
            'type': msg_type
        })

# Event Mediator for UI components
class UIMediator(Mediator):
    """Mediator for UI component interactions."""
    
    def __init__(self):
        self.components: Dict[str, object] = {}
        self.handlers: Dict[str, List[Callable]] = {}
    
    def register(self, component: object, name: str) -> None:
        self.components[name] = component
    
    def subscribe(self, event: str, handler: Callable) -> None:
        if event not in self.handlers:
            self.handlers[event] = []
        self.handlers[event].append(handler)
    
    def notify(self, sender: object, event: str, data: any) -> None:
        """Notify all subscribers of an event."""
        if event in self.handlers:
            for handler in self.handlers[event]:
                handler(data)
    
    def get_component(self, name: str) -> object:
        return self.components.get(name)

# UI Components
class Button:
    def __init__(self, mediator: UIMediator, name: str):
        self.mediator = mediator
        self.name = name
        self.enabled = True
        mediator.register(self, name)
    
    def click(self) -> None:
        if self.enabled:
```

```python
# PROBLEM: Define algorithm skeleton with overridable steps in subclasses
# TIME: O(n) where n is algorithm steps TIME: O(?) SPACE: O(?)
from abc import ABC, abstractmethod

# Abstract class with template method
class DataProcessor(ABC):
    """Template for data processing pipeline."""
    
    # Template method - defines algorithm skeleton
    def process(self, data: list) -> dict:
        """Main processing pipeline - not overridable."""
        result = {
            'input_size': len(data),
            'steps': []
        }
        
        # Step 1: Load
        loaded_data = self.load_data(data)
        result['steps'].append('load')
        
        # Step 2: Validate
        if not self.validate_data(loaded_data):
            result['error'] = 'Validation failed'
            return result
        result['steps'].append('validate')
        
        # Step 3: Transform (customizable)
        transformed = self.transform_data(loaded_data)
        result['steps'].append('transform')
        
        # Step 4: Process (customizable)
        processed = self.process_data(transformed)
        result['steps'].append('process')
        
        # Step 5: Save
        saved = self.save_data(processed)
        result['steps'].append('save')
        
        # Hook method (optional override)
        self.after_process(result)
        
        result['success'] = True
        result['output'] = saved
        return result
    
    # Default implementations (can be overridden)
    def load_data(self, data: list) -> list:
        """Default load - just return data."""
        return data
    
    def validate_data(self, data: list) -> bool:
        """Default validation - check non-empty."""
        return len(data) > 0
    
    # Abstract methods (must be implemented)
    @abstractmethod
    def transform_data(self, data: list) -> list:
        pass
    
    @abstractmethod
    def process_data(self, data: list) -> list:
        pass
    
    def save_data(self, data: list) -> list:
        """Default save - return data."""
        return data
    
    # Hook method (optional override)
    def after_process(self, result: dict) -> None:
        """Called after processing completes."""
        pass

# Concrete implementations
class NumberProcessor(DataProcessor):
    """Process numeric data."""
    
    def transform_data(self, data: list) -> list:
        """Filter and convert to numbers."""
        return [float(x) for x in data if str(x).replace('.', '').isdigit()]
    
    def process_data(self, data: list) -> list:
        """Calculate statistics."""
        if not data:
            return []
        
        return {
            'mean': sum(data) / len(data),
            'max': max(data),
            'min': min(data),
            'count': len(data)
        }

class TextProcessor(DataProcessor):
    """Process text data."""
    
    def validate_data(self, data: list) -> bool:
        """Validate that all items are strings."""
        return all(isinstance(x, str) for x in data)
    
    def transform_data(self, data: list) -> list:
        """Normalize text."""
        return [x.lower().strip() for x in data]
    
    def process_data(self, data: list) -> list:
        """Count word frequencies."""
        word_count = {}
        for text in data:
            for word in text.split():
                word_count[word] = word_count.get(word, 0) + 1
        return word_count
    
    def after_process(self, result: dict) -> None:
        """Log completion."""
        print(f"Text processing complete: {result['input_size']} items")

# File processor with hooks
class FileProcessor(DataProcessor):
    """Process file data with custom hooks."""
    
    def __init__(self):
        self.processed_files = []
    
    def load_data(self, data: list) -> list:
        """Load file contents."""
        return [{'path': f, 'content': f"Content of {f}"} for f in data]
    
    def transform_data(self, data: list) -> list:
        """Extract metadata."""
        for item in data:
            item['size'] = len(item['content'])
            item['lines'] = item['content'].count('\n') + 1
        return data
    
    def process_data(self, data: list) -> list:
        """Process files."""
        self.processed_files = [item['path'] for item in data]
        return data
    
    def after_process(self, result: dict) -> None:
        """Save processed file list."""
        result['processed_files'] = self.processed_files
```
