## [TimSort - Python's Adaptive Sorting Algorithm]

```python
# PROBLEM: Implement TimSort, a hybrid stable sorting algorithm derived from 
# merge sort and insertion sort, designed to perform well on many kinds of 
# real-world data. It's the default sorting algorithm in Python and Java.
# TIME: O(n log n) worst case, O(n) best case SPACE: O(n)

def timsort(arr):
    """
    TimSort: Adaptive, stable, hybrid sorting algorithm.
    
    Key concepts:
    1. Uses insertion sort for small chunks (runs)
    2. Identifies natural runs (already sorted sequences)
    3. Merges runs using modified merge sort
    4. Maintains stability by preserving order of equal elements
    """
    if not arr or len(arr) <= 1:
        return arr
    
    min_run = _calculate_min_run(len(arr))
    n = len(arr)
    
    # Step 1: Sort individual subarrays of size min_run using insertion sort
    for start in range(0, n, min_run):
        end = min(start + min_run - 1, n - 1)
        _insertion_sort(arr, start, end)
    
    # Step 2: Merge sorted runs starting from size min_run
    size = min_run
    while size < n:
        # Pick starting point of left sub array
        for left in range(0, n, 2 * size):
            # Find ending point of left sub array
            mid = min(n - 1, left + size - 1)
            # Find ending point of right sub array
            right = min(n - 1, left + 2 * size - 1)
            
            # Merge subarrays arr[left...mid] and arr[mid+1...right]
            if mid < right:
                _merge(arr, left, mid, right)
        
        size = 2 * size
    
    return arr

def _calculate_min_run(n):
    """
    Calculate minimum run length for TimSort.
    Returns a value in range 32-64 such that n/min_run is a power of 2.
    """
    r = 0
    while n >= 64:
        r |= n & 1  # Track remainder
        n >>= 1
    return n + r

def _insertion_sort(arr, left, right):
    """
    Insertion sort for small arrays.
    Efficient for nearly sorted data and small datasets.
    """
    for i in range(left + 1, right + 1):
        key = arr[i]
        j = i - 1
        
        # Move elements greater than key one position ahead
        while j >= left and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        
        arr[j + 1] = key

def _merge(arr, left, mid, right):
    """
    Merge two sorted subarrays arr[left...mid] and arr[mid+1...right].
    Uses Galloping mode optimization for better performance.
    """
    # Calculate lengths of two subarrays
    len1 = mid - left + 1
    len2 = right - mid
    
    # Create temporary arrays
    left_arr = arr[left:mid + 1]
    right_arr = arr[mid + 1:right + 1]
    
    i, j, k = 0, 0, left
    
    # Merge temp arrays back into arr[left...right]
    while i < len1 and j < len2:
        # Use <= to maintain stability
        if left_arr[i] <= right_arr[j]:
            arr[k] = left_arr[i]
            i += 1
        else:
            arr[k] = right_arr[j]
            j += 1
        k += 1
    
    # Copy remaining elements
    while i < len1:
        arr[k] = left_arr[i]
        i += 1
        k += 1
    
    while j < len2:
        arr[k] = right_arr[j]
        j += 1
        k += 1

# Example usage
if __name__ == "__main__":
    test_data = [64, 34, 25, 12, 22, 11, 90, 1, 5, 10]
    print(f"Original: {test_data}")
    timsort(test_data)
    print(f"Sorted:   {test_data}")
```

**Why**: TimSort is the default sorting algorithm in Python, Java, and Android. Understanding it reveals why Python's `sorted()` performs so well on real-world data - it exploits existing order in the input.

## [Boyer-Moore String Search Algorithm]

```python
# PROBLEM: Implement the Boyer-Moore string search algorithm, which uses
# bad character and good suffix heuristics for efficient pattern matching.
# TIME: O(n/m) best case, O(nm) worst case SPACE: O(k) where k = alphabet size

class BoyerMoore:
    """
    Boyer-Moore: Fast string search using two heuristics.
    
    Key concepts:
    1. Bad Character Rule: Skip alignments when mismatched character
       appears in pattern, align pattern to match
    2. Good Suffix Rule: Use already matched suffix to skip alignments
    3. Scans right-to-left within pattern
    4. Can skip sections of text, making sublinear O(n/m) possible
    """
    
    def __init__(self, pattern):
        self.pattern = pattern
        self.pattern_len = len(pattern)
        # Preprocess pattern for both heuristics
        self.bad_char = self._build_bad_char_table()
        self.good_suffix = self._build_good_suffix_table()
    
    def _build_bad_char_table(self):
        """
        Build bad character shift table.
        Maps each character to its rightmost position in pattern.
        """
        NO_OF_CHARS = 256
        bad_char = [-1] * NO_OF_CHARS
        
        # Store rightmost occurrence of each character
        for i in range(self.pattern_len):
            bad_char[ord(self.pattern[i])] = i
        
        return bad_char
    
    def _build_good_suffix_table(self):
        """
        Build good suffix shift table.
        Precomputes shift distances for good suffix rule.
        """
        pattern = self.pattern
        m = self.pattern_len
        good_suffix = [0] * (m + 1)
        
        # Initialize border positions
        border_pos = [0] * (m + 1)
        
        # Case 1: The matching suffix occurs elsewhere in the pattern
        i = m
        j = m + 1
        border_pos[i] = j
        
        while i > 0:
            while j <= m and pattern[i - 1] != pattern[j - 1]:
                if good_suffix[j] == 0:
                    good_suffix[j] = j - i
                j = border_pos[j]
            
            i -= 1
            j -= 1
            border_pos[i] = j
        
        # Case 2: Only a prefix of the matching suffix occurs
        j = border_pos[0]
        for i in range(m + 1):
            if good_suffix[i] == 0:
                good_suffix[i] = j
            if i == j:
                j = border_pos[j]
        
        return good_suffix
    
    def search(self, text):
        """
        Search for all occurrences of pattern in text.
        Returns list of starting indices where pattern is found.
        """
        n = len(text)
        m = self.pattern_len
        
        if m == 0 or m > n:
            return []
        
        occurrences = []
        shift = 0  # Current shift of pattern
        
        while shift <= n - m:
            j = m - 1  # Start from rightmost character of pattern
            
            # Keep reducing index j while characters match
            while j >= 0 and self.pattern[j] == text[shift + j]:
                j -= 1
            
            if j < 0:
                # Pattern found at position shift
                occurrences.append(shift)
                
                # Shift pattern using good suffix rule
                shift += self.good_suffix[0]
            else:
                # Mismatch occurred - calculate shift using both rules
                bad_char_shift = j - self.bad_char[ord(text[shift + j])]
                good_suffix_shift = self.good_suffix[j + 1]
                
                # Take maximum shift from both heuristics
                shift += max(1, max(bad_char_shift, good_suffix_shift))
        
        return occurrences

def boyer_moore_search(text, pattern):
    """
    Convenience function for Boyer-Moore search.
    Returns list of all starting positions where pattern appears in text.
    """
    bm = BoyerMoore(pattern)
    return bm.search(text)

# Example usage
if __name__ == "__main__":
    text = "ABAAABCDABCABCDABCDABDE"
    pattern = "ABCDAB"
    
    positions = boyer_moore_search(text, pattern)
    print(f"Text:    {text}")
    print(f"Pattern: {pattern}")
    print(f"Found at positions: {positions}")
    
    # Verify results
    for pos in positions:
        print(f"  Position {pos}: '{text[pos:pos+len(pattern)]}'")
```

**Why**: Boyer-Moore is fundamental in text editors and search engines. It can skip large sections of text, achieving sublinear time O(n/m), making it one of the fastest string search algorithms in practice.

## [Z-Algorithm for Pattern Matching]

```python
# PROBLEM: Implement the Z-algorithm for pattern matching, which computes
# the Z-array where Z[i] is the length of longest substring starting from i
# which is also a prefix of the string.
# TIME: O(n + m) SPACE: O(n + m)

def z_algorithm(s):
    """
    Compute Z-array in linear time.
    
    Z-array: For each position i, Z[i] = length of longest substring
    starting at i that matches a prefix of string s.
    
    Key insight: Use a "Z-box" [L, R] representing the interval with 
    maximum R such that S[L...R] is a prefix of S.
    """
    n = len(s)
    z = [0] * n
    z[0] = n  # First element is always the string length
    
    left, right = 0, 0  # Z-box boundaries [left, right]
    
    for i in range(1, n):
        # Case 1: i is outside current Z-box
        if i > right:
            left = right = i
            
            # Expand Z-box by comparing characters
            while right < n and s[right - left] == s[right]:
                right += 1
            
            z[i] = right - left
            right -= 1
        
        # Case 2: i is inside current Z-box
        else:
            k = i - left  # Corresponding position in prefix
            
            # If Z[k] doesn't extend beyond Z-box, use it directly
            if z[k] < right - i + 1:
                z[i] = z[k]
            
            # If Z[k] extends beyond Z-box, we need to recalculate
            else:
                left = i
                while right < n and s[right - left] == s[right]:
                    right += 1
                
                z[i] = right - left
                right -= 1
    
    return z

def z_pattern_search(text, pattern):
    """
    Use Z-algorithm for pattern matching.
    Concatenate pattern + '$' + text and compute Z-array.
    Z[i] == len(pattern) indicates pattern match at position i.
    """
    if not pattern or not text:
        return []
    
    # Create concatenated string: pattern + separator + text
    separator = '$'  # Character not in pattern or text
    concatenated = pattern + separator + text
    
    # Compute Z-array for concatenated string
    z = z_algorithm(concatenated)
    
    # Find all positions where Z[i] == pattern length
    pattern_len = len(pattern)
    positions = []
    
    for i in range(len(concatenated)):
        if z[i] == pattern_len:
            # Convert index in concatenated string to index in text
            text_position = i - pattern_len - 1
            positions.append(text_position)
    
    return positions

def z_find_all_occurrences(text, pattern):
    """
    Find all occurrences of pattern in text using Z-algorithm.
    Returns list of starting indices.
    """
    positions = z_pattern_search(text, pattern)
    return positions

def z_longest_common_prefix(strings):
    """
    Find longest common prefix among multiple strings using Z-algorithm.
    """
    if not strings:
        return ""
    
    base = strings[0]
    
    for s in strings[1:]:
        if not s:
            return ""
        
        concatenated = base + '$' + s
        z = z_algorithm(concatenated)
        
        # Find maximum Z value in the second part
        max_len = 0
        for i in range(len(base) + 1, len(concatenated)):
            max_len = max(max_len, z[i])
        
        # Update base to common prefix only
        base = base[:max_len]
        
        if not base:
            return ""
    
    return base

# Example usage
if __name__ == "__main__":
    # Example 1: Compute Z-array
    s = "aabcaabxaaaz"
    z = z_algorithm(s)
    print(f"String:  {s}")
    print(f"Z-array: {z}")
    
    # Example 2: Pattern matching
    text = "GEEKS FOR GEEKS"
    pattern = "GEEK"
    positions = z_find_all_occurrences(text, pattern)
    print(f"\nText: '{text}'")
    print(f"Pattern: '{pattern}'")
    print(f"Found at positions: {positions}")
    
    # Example 3: Longest common prefix
    strings = ["interspecies", "interstellar", "interstate"]
    lcp = z_longest_common_prefix(strings)
    print(f"\nStrings: {strings}")
    print(f"Longest common prefix: '{lcp}'")
```

**Why**: Z-algorithm provides linear-time pattern matching and is easier to implement than KMP. It's widely used in competitive programming and bioinformatics for DNA sequence analysis.

## [Huffman Coding for Data Compression]

```python
# PROBLEM: Implement Huffman coding, a lossless data compression algorithm
# that assigns variable-length codes to characters based on their frequencies.
# TIME: O(n log n) for building tree SPACE: O(n) for tree and codes

import heapq
from collections import Counter

class HuffmanNode:
    """Node for Huffman tree."""
    def __init__(self, char=None, freq=0, left=None, right=None):
        self.char = char      # Character (None for internal nodes)
        self.freq = freq      # Frequency
        self.left = left      # Left child
        self.right = right    # Right child
    
    def __lt__(self, other):
        # Comparison for min-heap (lower frequency has higher priority)
        return self.freq < other.freq
    
    def is_leaf(self):
        return self.left is None and self.right is None

class HuffmanCoding:
    """
    Huffman Coding: Lossless compression using variable-length codes.
    
    Key concepts:
    1. Build frequency table of characters
    2. Create leaf nodes for each character
    3. Build Huffman tree using min-heap
    4. Assign codes: left=0, right=1
    5. More frequent characters get shorter codes
    """
    
    def __init__(self):
        self.root = None
        self.codes = {}
        self.reverse_mapping = {}
    
    def build_frequency_table(self, text):
        """Count frequency of each character."""
        return Counter(text)
    
    def build_huffman_tree(self, text):
        """
        Build Huffman tree from text.
        Uses min-heap to always combine two lowest frequency nodes.
        """
        freq_table = self.build_frequency_table(text)
        
        # Create min-heap of leaf nodes
        heap = []
        for char, freq in freq_table.items():
            node = HuffmanNode(char=char, freq=freq)
            heapq.heappush(heap, node)
        
        # Special case: only one character
        if len(heap) == 1:
            node = heapq.heappop(heap)
            merged = HuffmanNode(freq=node.freq, left=node)
            heapq.heappush(heap, merged)
        
        # Build tree by combining lowest frequency nodes
        while len(heap) > 1:
            # Extract two minimum frequency nodes
            left = heapq.heappop(heap)
            right = heapq.heappop(heap)
            
            # Create internal node with combined frequency
            merged = HuffmanNode(
                freq=left.freq + right.freq,
                left=left,
                right=right
            )
            
            heapq.heappush(heap, merged)
        
        self.root = heap[0]
        return self.root
    
    def build_codes(self, node=None, current_code=""):
        """
        Generate Huffman codes by traversing tree.
        Left edge = 0, Right edge = 1
        """
        if node is None:
            node = self.root
            self.codes = {}
            self.reverse_mapping = {}
        
        if node.is_leaf():
            # Handle single character case
            if not current_code:
                current_code = "0"
            
            self.codes[node.char] = current_code
            self.reverse_mapping[current_code] = node.char
            return
        
        # Traverse left (add 0)
        self.build_codes(node.left, current_code + "0")
        
        # Traverse right (add 1)
        self.build_codes(node.right, current_code + "1")
    
    def encode(self, text):
        """
        Encode text using Huffman codes.
        Returns binary string representation.
        """
        if not self.codes:
            self.build_huffman_tree(text)
            self.build_codes()
        
        encoded = ""
        for char in text:
            encoded += self.codes[char]
        
        return encoded
    
    def decode(self, encoded_text):
        """
        Decode binary string using Huffman tree.
        Traverses tree according to bits until reaching leaf.
        """
        if self.root is None:
            raise ValueError("Huffman tree not built")
        
        decoded = ""
        current_node = self.root
        
        for bit in encoded_text:
            if bit == '0':
                current_node = current_node.left
            else:
                current_node = current_node.right
            
            if current_node.is_leaf():
                decoded += current_node.char
                current_node = self.root
        
        return decoded
    
    def get_compression_ratio(self, text):
        """Calculate compression ratio."""
        original_bits = len(text) * 8  # Assuming 8-bit ASCII
        encoded_bits = len(self.encode(text))
        
        return encoded_bits / original_bits
    
    def get_codes_table(self):
        """Return copy of codes dictionary."""
        return dict(sorted(self.codes.items(), key=lambda x: len(x[1])))

def compress_file(text):
    """
    Compress text using Huffman coding.
    Returns tuple: (encoded_text, huffman_obj)
    """
    huffman = HuffmanCoding()
    encoded = huffman.encode(text)
    return encoded, huffman

# Example usage
if __name__ == "__main__":
    text = "this is an example for huffman encoding"
    
    print(f"Original text: '{text}'")
    print(f"Original size: {len(text) * 8} bits\n")
    
    # Create Huffman coder
    huffman = HuffmanCoding()
    
    # Build tree and encode
    huffman.build_huffman_tree(text)
    huffman.build_codes()
    
    # Show codes
    print("Huffman Codes:")
    for char, code in huffman.get_codes_table().items():
        print(f"  '{char}': {code}")
    
    # Encode
    encoded = huffman.encode(text)
    print(f"\nEncoded: {encoded}")
    print(f"Encoded size: {len(encoded)} bits")
    
    # Decode
    decoded = huffman.decode(encoded)
    print(f"\nDecoded: '{decoded}'")
    
    # Compression ratio
    ratio = huffman.get_compression_ratio(text)
    print(f"\nCompression ratio: {ratio:.2%}")
    print(f"Space saved: {(1 - ratio) * 100:.1f}%")
    
    # Verify
    assert text == decoded, "Decoding failed!"
    print("\n✓ Encoding/Decoding verified successfully!")
```

**Why**: Huffman coding is foundational in compression (ZIP, JPEG, MP3). Understanding it reveals how variable-length codes achieve optimal compression by exploiting character frequency distributions.

## [LFU Cache Implementation]

```python
# PROBLEM: Implement LFU (Least Frequently Used) cache with O(1) get and put
# operations. When cache is full, evict the least frequently used item.
# If tie, evict least recently used among least frequent.
# TIME: O(1) for get/put SPACE: O(n)

from collections import defaultdict, OrderedDict

class LFUCache:
    """
    LFU Cache: Evicts least frequently used items.
    
    Data structures:
    1. key_to_val: key -> value mapping
    2. key_to_freq: key -> frequency mapping
    3. freq_to_keys: frequency -> OrderedDict of keys
    4. min_freq: current minimum frequency
    
    Why OrderedDict: Maintains insertion order for LRU among same frequency
    """
    
    def __init__(self, capacity):
        self.capacity = capacity
        self.size = 0
        self.min_freq = 0
        
        # Maps key to (value, frequency)
        self.key_to_val_freq = {}
        
        # Maps frequency to OrderedDict of keys (for LRU within same freq)
        self.freq_to_keys = defaultdict(OrderedDict)
    
    def get(self, key):
        """
        Get value and update frequency.
        Returns -1 if key not found.
        """
        if key not in self.key_to_val_freq:
            return -1
        
        # Get current value and frequency
        val, freq = self.key_to_val_freq[key]
        
        # Remove from current frequency list
        del self.freq_to_keys[freq][key]
        
        # Update min_freq if this was the only key at min_freq
        if not self.freq_to_keys[freq] and freq == self.min_freq:
            self.min_freq += 1
        
        # Add to next frequency list
        new_freq = freq + 1
        self.freq_to_keys[new_freq][key] = None  # Value doesn't matter
        self.key_to_val_freq[key] = (val, new_freq)
        
        return val
    
    def put(self, key, value):
        """Put key-value pair and update frequency."""
        if self.capacity <= 0:
            return
        
        # If key exists, update value and frequency
        if key in self.key_to_val_freq:
            _, freq = self.key_to_val_freq[key]
            
            # Remove from current frequency list
            del self.freq_to_keys[freq][key]
            
            # Update min_freq if needed
            if not self.freq_to_keys[freq] and freq == self.min_freq:
                self.min_freq += 1
            
            # Add to next frequency list
            new_freq = freq + 1
            self.freq_to_keys[new_freq][key] = None
            self.key_to_val_freq[key] = (value, new_freq)
            return
        
        # If cache is full, evict LFU (and LRU among ties)
        if self.size == self.capacity:
            # Get oldest key at min_freq
            lfu_key, _ = self.freq_to_keys[self.min_freq].popitem(last=False)
            del self.key_to_val_freq[lfu_key]
            self.size -= 1
        
        # Add new key with frequency 1
        self.min_freq = 1
        self.freq_to_keys[1][key] = None
        self.key_to_val_freq[key] = (value, 1)
        self.size += 1
    
    def _evict(self):
        """Evict least frequently used (and least recently used among ties)."""
        if not self.freq_to_keys[self.min_freq]:
            return None
        
        # OrderedDict popitem(last=False) removes first (oldest) item
        key, _ = self.freq_to_keys[self.min_freq].popitem(last=False)
        value, _ = self.key_to_val_freq[key]
        del self.key_to_val_freq[key]
        self.size -= 1
        
        return key
    
    def get_cache_state(self):
        """Get current cache state for debugging."""
        state = {
            'capacity': self.capacity,
            'size': self.size,
            'min_freq': self.min_freq,
            'items': {k: {'value': v, 'freq': f} 
                     for k, (v, f) in self.key_to_val_freq.items()}
        }
        return state

# Example usage
if __name__ == "__main__":
    # Create LFU cache with capacity 3
    cache = LFUCache(3)
    
    print("=== LFU Cache Demo ===\n")
    
    # Put operations
    cache.put(1, 10)
    print("put(1, 10)")
    
    cache.put(2, 20)
    print("put(2, 20)")
    
    cache.put(3, 30)
    print("put(3, 30)")
    
    print(f"\nCache state: {cache.get_cache_state()['items']}")
    
    # Access key 1 and 2 (they become more frequent)
    print(f"\nget(1) = {cache.get(1)}")  # freq: 1 -> 2
    print(f"get(2) = {cache.get(2)}")  # freq: 1 -> 2
    print(f"get(1) = {cache.get(1)}")  # freq: 2 -> 3
    
    print(f"\nCache state after accesses: {cache.get_cache_state()['items']}")
    
    # Add new item - should evict key 3 (freq=1, least recently used among freq=1)
    cache.put(4, 40)
    print("\nput(4, 40) - evicts key 3 (LFU + LRU)")
    print(f"Cache state: {cache.get_cache_state()['items']}")
    
    # Verify eviction
    print(f"\nget(3) = {cache.get(3)} (should be -1, evicted)")
    print(f"get(4) = {cache.get(4)} (should be 40)")
    
    # Test capacity 0
    print("\n=== Edge Case: Capacity 0 ===")
    cache_zero = LFUCache(0)
    cache_zero.put(1, 10)
    print(f"put(1, 10) on capacity 0")
    print(f"get(1) = {cache_zero.get(1)} (should be -1)")
```

**Why**: LFU cache is used in CDN caching, database buffer management, and web browsers. Unlike LRU, it captures actual access patterns, making it better for workloads with hot data.

## [Treap - Randomized Binary Search Tree]

```python
# PROBLEM: Implement Treap (Tree + Heap), a randomized BST that maintains
# BST property by key and heap property by random priority.
# TIME: O(log n) expected for insert/delete/search SPACE: O(n)

import random

class TreapNode:
    """Node for Treap data structure."""
    def __init__(self, key, value=None):
        self.key = key
        self.value = value
        self.priority = random.random()  # Random priority for heap property
        self.left = None
        self.right = None

class Treap:
    """
    Treap: Combines BST and Heap properties.
    
    Properties:
    1. BST property: left.key < node.key < right.key
    2. Heap property: node.priority < children.priority (min-heap)
    
    Randomization: Random priorities ensure O(log n) expected height,
    avoiding worst-case O(n) of regular BST.
    
    Operations:
    - Insert: Add as leaf, rotate up to maintain heap property
    - Delete: Rotate down to leaf, then remove
    - Search: Standard BST search
    """
    
    def __init__(self):
        self.root = None
        self.size = 0
    
    def insert(self, key, value=None):
        """Insert key-value pair maintaining BST and heap properties."""
        self.root = self._insert(self.root, key, value)
        self.size += 1
    
    def _insert(self, root, key, value):
        """Recursive insert with rotations to maintain heap property."""
        if root is None:
            return TreapNode(key, value)
        
        # BST insertion
        if key <= root.key:
            root.left = self._insert(root.left, key, value)
            
            # Rotate right if heap property violated
            if root.left and root.left.priority < root.priority:
                root = self._rotate_right(root)
        else:
            root.right = self._insert(root.right, key, value)
            
            # Rotate left if heap property violated
            if root.right and root.right.priority < root.priority:
                root = self._rotate_left(root)
        
        return root
    
    def delete(self, key):
        """Delete key from treap."""
        self.root = self._delete(self.root, key)
        self.size -= 1
    
    def _delete(self, root, key):
        """Recursive delete with rotations to maintain heap property."""
        if root is None:
            return None
        
        if key < root.key:
            root.left = self._delete(root.left, key)
        elif key > root.key:
            root.right = self._delete(root.right, key)
        else:
            # Key found - delete this node
            if root.left is None:
                return root.right
            elif root.right is None:
                return root.left
            else:
                # Both children exist - rotate smaller priority child up
                if root.left.priority < root.right.priority:
                    root = self._rotate_right(root)
                    root.right = self._delete(root.right, key)
                else:
                    root = self._rotate_left(root)
                    root.left = self._delete(root.left, key)
        
        return root
    
    def search(self, key):
        """Search for key, return value or None."""
        node = self._search(self.root, key)
        return node.value if node else None
    
    def _search(self, root, key):
        """Recursive BST search."""
        if root is None:
            return None
        
        if key == root.key:
            return root
        elif key < root.key:
            return self._search(root.left, key)
        else:
            return self._search(root.right, key)
    
    def _rotate_right(self, y):
        """Right rotation."""
        x = y.left
        T = x.right
        
        # Rotate
        x.right = y
        y.left = T
        
        return x
    
    def _rotate_left(self, x):
        """Left rotation."""
        y = x.right
        T = y.left
        
        # Rotate
        y.left = x
        x.right = T
        
        return y
    
    def inorder(self):
        """Return keys in sorted order."""
        result = []
        self._inorder(self.root, result)
        return result
    
    def _inorder(self, root, result):
        """Inorder traversal."""
        if root:
            self._inorder(root.left, result)
            result.append(root.key)
            self._inorder(root.right, result)
    
    def get_height(self):
        """Calculate tree height."""
        return self._height(self.root)
    
    def _height(self, root):
        """Recursive height calculation."""
        if root is None:
            return 0
        return 1 + max(self._height(root.left), self._height(root.right))
    
    def split(self, key):
        """
        Split treap into two: keys <= key and keys > key.
        Returns (left_treap, right_treap).
        """
        return self._split(self.root, key)
    
    def _split(self, root, key):
        """Recursive split operation."""
        if root is None:
            return (None, None)
        
        if key < root.key:
            left, right = self._split(root.left, key)
            root.left = right
            return (left, root)
        else:
            left, right = self._split(root.right, key)
            root.right = left
            return (root, right)
    
    def merge(self, other):
        """Merge two treaps (all keys in self < all keys in other)."""
        self.root = self._merge(self.root, other.root)
        self.size += other.size
    
    def _merge(self, left, right):
        """Merge two treaps maintaining heap property."""
        if left is None:
            return right
        if right is None:
            return left
        
        # Choose root with smaller priority
        if left.priority < right.priority:
            left.right = self._merge(left.right, right)
            return left
        else:
            right.left = self._merge(left, right.left)
            return right

# Example usage
if __name__ == "__main__":

    # Insert elements

        treap.insert(key)

    print(f"Inserted: {keys}")
    print(f"Sorted order: {treap.inorder()}")

    print(f"  Right (>50): {treap._inorder(right, [])}")
```

**Why**: Treaps provide balanced BST performance without complex rebalancing logic. Randomization guarantees O(log n) expected operations, making them simpler than AVL/Red-Black trees while maintaining good performance.

## [Distributed Lock with Redlock Algorithm]

```python
# PROBLEM: Implement distributed lock using Redlock algorithm across multiple
# Redis instances to achieve safety in distributed systems.
# TIME: O(n) where n = number of Redis instances SPACE: O(1)

import time
import uuid
import hashlib
from threading import Thread, Lock

class MockRedis:
    """Mock Redis instance for demonstration."""
    
    def __init__(self, name):
        self.name = name
        self.lock = Lock()
        self.data = {}
    
    def set(self, key, value, nx=True, px=None):
        """Set if not exists (NX) with expiration in milliseconds (PX)."""
        with self.lock:
            if key in self.data:
                return False
            self.data[key] = {
                'value': value,
                'expires_at': time.time() * 1000 + px if px else None
            }
            return True
    
    def get(self, key):
        """Get value if not expired."""
        with self.lock:
            if key not in self.data:
                return None
            entry = self.data[key]
            if entry['expires_at'] and time.time() * 1000 > entry['expires_at']:
                del self.data[key]
                return None
            return entry['value']
    
    def delete(self, key):
        """Delete key."""
        with self.lock:
            self.data.pop(key, None)
    
    def eval_delete_if_match(self, key, value):
        """Lua script equivalent: delete only if value matches."""
        with self.lock:
            if key in self.data and self.data[key]['value'] == value:
                del self.data[key]
                return True
            return False

class DistributedLock:
    """
    Redlock Algorithm: Distributed lock across multiple Redis instances.
    
    Safety guarantees:
    1. Mutual exclusion: Only one client can hold lock at a time
    2. Deadlock free: Locks expire automatically
    3. Fault tolerance: Works if majority of Redis instances are available
    
    Algorithm:
    1. Generate unique lock value
    2. Try to acquire lock on all instances sequentially
    3. Lock acquired if obtained on majority within validity time
    4. Release: Delete lock from all instances
    """
    
    def __init__(self, redis_instances, lock_name, ttl_ms=10000):
        """
        Initialize distributed lock.
        
        Args:
            redis_instances: List of Redis client instances
            lock_name: Name of the lock resource
            ttl_ms: Lock time-to-live in milliseconds
        """
        self.redis_instances = redis_instances
        self.lock_name = lock_name
        self.ttl_ms = ttl_ms
        self.lock_value = None
        self.quorum = (len(redis_instances) // 2) + 1
    
    def acquire(self, retry_times=3, retry_delay_ms=200):
        """
        Try to acquire the distributed lock.
        
        Returns True if lock acquired, False otherwise.
        """
        self.lock_value = str(uuid.uuid4())
        
        for attempt in range(retry_times):
            start_time = time.time()
            acquired_count = 0
            
            # Try to acquire lock on all instances
            for redis in self.redis_instances:
                try:
                    if redis.set(
                        self.lock_name,
                        self.lock_value,
                        nx=True,
                        px=self.ttl_ms
                    ):
                        acquired_count += 1
                except Exception as e:
                    print(f"Redis {redis.name} error: {e}")
            
            # Calculate time spent acquiring locks
            elapsed_ms = (time.time() - start_time) * 1000
            
            # Calculate remaining validity time
            validity_time = self.ttl_ms - elapsed_ms
            
            # Lock acquired if:
            # 1. Acquired on majority of instances (quorum)
            # 2. Validity time is positive
            if acquired_count >= self.quorum and validity_time > 0:
                return True
            
            # Failed to acquire - release any partial locks
            self._release_internal()
            
            # Wait before retry
            if attempt < retry_times - 1:
                time.sleep(retry_delay_ms / 1000)
        
        return False
    
    def release(self):
        """
        Release the distributed lock.
        
        Only releases if we own the lock (value matches).
        """
        if self.lock_value:
            self._release_internal()
            self.lock_value = None
    
    def _release_internal(self):
        """Internal release: delete lock from all instances if value matches."""
        for redis in self.redis_instances:
            try:
                redis.eval_delete_if_match(self.lock_name, self.lock_value)
            except Exception as e:
                print(f"Redis {redis.name} release error: {e}")
    
    def extend(self, extend_ms=None):
        """
        Extend lock TTL.
        
        Returns True if extended successfully, False otherwise.
        """
        if not self.lock_value:
            return False
        
        extend_ms = extend_ms or self.ttl_ms
        start_time = time.time()
        extended_count = 0
        
        for redis in self.redis_instances:
            try:
                # Check if we still own the lock
                if redis.get(self.lock_name) == self.lock_value:
                    # Extend by setting new expiration
                    if redis.set(
                        self.lock_name,
                        self.lock_value,
                        nx=False,  # Allow overwrite
                        px=extend_ms
                    ):
                        extended_count += 1
            except Exception as e:
                print(f"Redis {redis.name} extend error: {e}")
        
        elapsed_ms = (time.time() - start_time) * 1000
        validity_time = extend_ms - elapsed_ms
        
        return extended_count >= self.quorum and validity_time > 0

# Example usage
if __name__ == "__main__":

    redis_instances = [

        print("✓ Lock released")
    else:

        else:
```

## [Sudoku Solver with Backtracking]

```python
# PROBLEM: Solve a 9x9 Sudoku puzzle using backtracking algorithm with
# constraint propagation and efficient cell selection.
# TIME: O(9^(n*n)) worst case, much faster with heuristics SPACE: O(n*n)

class SudokuSolver:
    """
    Sudoku Solver: Backtracking with constraint propagation.
    
    Optimizations:
    1. Constraint propagation: Eliminate impossible values
    2. MRV heuristic: Choose cell with minimum remaining values
    3. Early termination: Return immediately when solved
    4. Bitmask for fast validity checking
    """
    
    def __init__(self, board):
        self.board = board
        self.size = 9
        self.box_size = 3
    
    def solve(self):
        """
        Solve the Sudoku puzzle using backtracking.
        Returns True if solution exists, False otherwise.
        """
        empty = self._find_empty_mrv()
        
        if empty is None:
            return True  # No empty cells = solved
        
        row, col = empty
        
        # Try values 1-9
        for num in range(1, 10):
            if self._is_valid(row, col, num):
                # Place number
                self.board[row][col] = num
                
                # Recursively solve
                if self.solve():
                    return True
                
                # Backtrack
                self.board[row][col] = 0
        
        return False  # No valid number found
    
    def _find_empty_mrv(self):
        """
        Find empty cell using Minimum Remaining Values (MRV) heuristic.
        Returns cell with fewest possible values.
        """
        min_count = 10
        best_cell = None
        
        for row in range(self.size):
            for col in range(self.size):
                if self.board[row][col] == 0:
                    # Count valid values for this cell
                    count = sum(
                        1 for num in range(1, 10)
                        if self._is_valid(row, col, num)
                    )
                    
                    # If only one valid value, use this cell immediately
                    if count == 1:
                        return (row, col)
                    
                    # Track cell with minimum remaining values
                    if count < min_count:
                        min_count = count
                        best_cell = (row, col)
        
        return best_cell
    
    def _is_valid(self, row, col, num):
        """
        Check if placing num at (row, col) is valid.
        Must not exist in same row, column, or 3x3 box.
        """
        # Check row
        if num in self.board[row]:
            return False
        
        # Check column
        if num in [self.board[i][col] for i in range(self.size)]:
            return False
        
        # Check 3x3 box
        box_row = (row // self.box_size) * self.box_size
        box_col = (col // self.box_size) * self.box_size
        
        for i in range(box_row, box_row + self.box_size):
            for j in range(box_col, box_col + self.box_size):
                if self.board[i][j] == num:
                    return False
        
        return True
    
    def get_possible_values(self, row, col):
        """Get all valid values for a cell."""
        return [
            num for num in range(1, 10)
            if self._is_valid(row, col, num)
        ]
    
    def print_board(self):
        """Print board in readable format."""
        for i in range(self.size):
            if i % 3 == 0 and i != 0:
                print("-" * 21)
            
            for j in range(self.size):
                if j % 3 == 0 and j != 0:
                    print("| ", end="")
                
                print(f"{self.board[i][j] if self.board[i][j] != 0 else '.'} ", end="")
            
            print()
    
    def is_valid_board(self):
        """Check if the current board state is valid."""
        for i in range(self.size):
            for j in range(self.size):
                if self.board[i][j] != 0:
                    num = self.board[i][j]
                    self.board[i][j] = 0
                    valid = self._is_valid(i, j, num)
                    self.board[i][j] = num
                    if not valid:
                        return False
        return True

def solve_sudoku(board):
    """
    Convenience function to solve a Sudoku board.
    Returns solved board or None if no solution exists.
    """
    solver = SudokuSolver([row[:] for row in board])  # Deep copy
    
    if not solver.is_valid_board():
        return None
    
    if solver.solve():
        return solver.board
    return None

        [5, 3, 0, 0, 7, 0, 0, 0, 0],
        [6, 0, 0, 1, 9, 5, 0, 0, 0],
        [0, 9, 8, 0, 0, 0, 0, 6, 0],
        [8, 0, 0, 0, 6, 0, 0, 0, 3],
        [4, 0, 0, 8, 0, 3, 0, 0, 1],
        [7, 0, 0, 0, 2, 0, 0, 0, 6],
        [0, 6, 0, 0, 0, 0, 2, 8, 0],
        [0, 0, 0, 4, 1, 9, 0, 0, 5],
        [0, 0, 0, 0, 8, 0, 0, 7, 9]
    ]
    
    solver = SudokuSolver([row[:] for row in puzzle])
```

**Why**: Sudoku solver demonstrates backtracking with constraint propagation - a fundamental technique for constraint satisfaction problems. MRV heuristic dramatically reduces search space, applicable to scheduling, routing, and AI problems.
