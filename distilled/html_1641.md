                    item.style.display = '';
                } else {
                    item.style.display = 'none';
                }
            });
        });
    </script>

            </tr>
        </thead>
        <tbody id="tableBody">
            <tr>
                <td>John Doe</td>
                <td>28</td>
                <td>New York</td>
            </tr>
            <tr>
                <td>Jane Smith</td>
                <td>34</td>
                <td>London</td>
            </tr>
            <tr>
                <td>Bob Johnson</td>
                <td>22</td>
                <td>Paris</td>
            </tr>
        </tbody>
    </table>
    
    <script>
        /**
         * Sorts table rows when column header is clicked
         * Demonstrates TBODY manipulation for dynamic tables
         */
        let sortDirection = [true, true, true]; // true = ascending
        
        function sortTable(columnIndex) {
            const tbody = document.getElementById('tableBody');
            const rows = Array.from(tbody.getElementsByTagName('tr'));
            
            // Sort rows based on column content
            rows.sort((a, b) => {
                const aCell = a.cells[columnIndex].textContent;
                const bCell = b.cells[columnIndex].textContent;
                
                // Check if numeric
                const aNum = parseFloat(aCell);
                const bNum = parseFloat(bCell);
                
                if (!isNaN(aNum) && !isNaN(bNum)) {
                    return sortDirection[columnIndex] ? aNum - bNum : bNum - aNum;
                }
                
                return sortDirection[columnIndex] 
                    ? aCell.localeCompare(bCell)
                    : bCell.localeCompare(aCell);
            });
            
            // Toggle direction for next click
            sortDirection[columnIndex] = !sortDirection[columnIndex];
            
            // Re-append sorted rows
            rows.forEach(row => tbody.appendChild(row));
        }
    </script>
</body>
</html>
```
**Why**: TBODY semantically groups table data, enabling proper scrolling and better DOM manipulation for dynamic tables.

         */

        content.addEventListener('input', () => {
            const text = content.value;
            
            // Create Blob from text
            const blob = new Blob([text], { type: 'text/plain' });
            
            // Create object URL
            const url = URL.createObjectURL(blob);
            
            // Update download link
            downloadLink.href = url;
        });
        
        // Generate CSV file
        function downloadCSV(data) {
            // Convert array of objects to CSV
            const headers = Object.keys(data[0]).join(',');
            const rows = data.map(row => Object.values(row).join(','));
            const csv = [headers, ...rows].join('\n');
            
            const blob = new Blob([csv], { type: 'text/csv' });
            const url = URL.createObjectURL(blob);
            
            const link = document.createElement('a');
            link.href = url;
            link.download = 'data.csv';
            link.click();
            
            // Cleanup
            URL.revokeObjectURL(url);
        }
        
        // Example usage:
        // downloadCSV([{name: 'John', age: 30}, {name: 'Jane', age: 25}]);
    </script>
</body>
</html>
```
**Why**: The download attribute enables client-side file generation without server requests, improving performance and privacy.

## 8. Arrow Functions Showcase

```html
<!-- PROBLEM: Demonstrate various arrow function patterns and use cases

<!DOCTYPE html>
<html lang="en">

    <script>
        const output = document.getElementById('output');
        
        /**
         * BASIC ARROW FUNCTION SYNTAX
         */
        
        // Single parameter - parentheses optional
        const double = x => x * 2;
        console.log(double(5)); // 10
        
        // Multiple parameters - parentheses required
        const add = (a, b) => a + b;
        console.log(add(3, 4)); // 7
        
        // No parameters - empty parentheses required
        const getRandom = () => Math.random();
        console.log(getRandom());
        
        /**
         * BLOCK BODY ARROW FUNCTIONS
         */
        
        // Multiple statements require block and return
        const calculateArea = (width, height) => {
            const area = width * height;
            return `Area: ${area} sq units`;
        };
        console.log(calculateArea(5, 10)); // "Area: 50 sq units"
        
        /**
         * ARROW FUNCTIONS IN ARRAY METHODS
         */
        
        const numbers = [1, 2, 3, 4, 5];
        
        // Map with arrow function
        const squared = numbers.map(n => n ** 2);
        console.log(squared); // [1, 4, 9, 16, 25]
        
        // Filter with arrow function
        const evens = numbers.filter(n => n % 2 === 0);
        console.log(evens); // [2, 4]
        
        // Reduce with arrow function
        const sum = numbers.reduce((acc, n) => acc + n, 0);
        console.log(sum); // 15
        
        /**
         * ARROW FUNCTIONS AND 'THIS'
         * Arrow functions inherit 'this' from enclosing scope
         */
        
        function Counter() {
            this.count = 0;
            
            // Arrow function preserves 'this'
            this.increment = () => {
                this.count++;
                console.log(this.count);
            };
            
            // Regular function loses 'this'
            this.decrement = function() {
                this.count--; // 'this' is undefined in strict mode
            };
        }
        
        const counter = new Counter();
        counter.increment(); // 1
        counter.increment(); // 2
        
        /**
         * ARROW FUNCTIONS IN OBJECT LITERALS
         */
        
        const calculator = {
            value: 0,
            add: (x) => this.value + x, // 'this' is NOT the object!
            subtract(x) { return this.value - x; } // Regular function works
        };
        
        console.log(calculator.subtract(5)); // -5
        
        /**
         * ARROW FUNCTIONS AS CALLBACKS
         */
        
        const users = [
            { name: 'Alice', age: 30 },
            { name: 'Bob', age: 25 },
            { name: 'Charlie', age: 35 }
        ];
        
        // Sort by age using arrow function
        users.sort((a, b) => a.age - b.age);
        console.log(users);
        
        /**
         * IMMEDIATELY INVOKED ARROW FUNCTIONS
         */
        
        const result = ((x) => x * x)(5);
        console.log(result); // 25
        
        // Display results
        output.innerHTML = `
            <h2>Arrow Function Results</h2>
            <p>Double 5: ${double(5)}</p>
            <p>Add 3+4: ${add(3, 4)}</p>
            <p>Squared: [${squared}]</p>
            <p>Evens: [${evens}]</p>
            <p>Sum: ${sum}</p>
        `;
    </script>
</body>
</html>
```
**Why**: Arrow functions provide concise syntax and lexical `this` binding, essential for modern JavaScript, especially in callbacks and array methods.
