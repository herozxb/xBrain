## [html_source_type_attribute]

```html
<!-- PROBLEM: Specify media file format for responsive images and videos
     TIME: O(1) SPACE: O(1) -->

<picture>
  <!-- WebP for modern browsers (smaller file size) -->
  <source srcset="image.webp" type="image/webp">
  
  <!-- AVIF for next-gen browser support -->
  <source srcset="image.avif" type="image/avif">
  
  <!-- Fallback to JPEG for older browsers -->
  <img src="image.jpg" alt="Responsive image with format fallback">
</picture>

<!-- Video with multiple format sources -->
<video controls>
  <source src="video.mp4" type="video/mp4">
  <source src="video.webm" type="video/webm">
  <source src="video.ogg" type="video/ogg">
  Your browser does not support the video tag.
</video>
```

**Why**: The `type` attribute in `<source>` elements helps browsers choose the most efficient format before downloading, saving bandwidth and improving load times. Browser will select the first supported format and skip downloading others.

// Check if browser supports AVIF
function supportsAvif() {
  const canvas = document.createElement('canvas');
  canvas.width = 1;
  canvas.height = 1;
  return canvas.toDataURL('image/avif').startsWith('data:image/avif');
}

// Check if browser supports WebP
function supportsWebp() {
  const canvas = document.createElement('canvas');
  canvas.width = 1;
  canvas.height = 1;
  return canvas.toDataURL('image/webp').startsWith('data:image/webp');

        <button role="menuitem" class="menu-item" tabindex="-1">
          Edit
        </button>
      </li>

        <button role="menuitem" class="menu-item" tabindex="-1">

          Delete (Disabled)
        </button>
      </li>
      
      <li class="menu-separator" role="separator"></li>
      
      <li role="none">
        <button role="menuitem" class="menu-item" tabindex="-1">
          Share
        </button>
      </li>
      
      <li role="none">
        <button role="menuitem" class="menu-item" tabindex="-1">
          Export
        </button>
      </li>
    </ul>

  <script>
    class AccessibleMenu {
      constructor(container) {
        this.container = container;
        this.trigger = container.querySelector('.menu-trigger');
        this.menu = container.querySelector('.menu');
        this.menuItems = Array.from(container.querySelectorAll('[role="menuitem"]'));
        this.focusedIndex = -1;

        this.bindEvents();
      }

      bindEvents() {
        // Toggle menu on trigger click
        this.trigger.addEventListener('click', () => this.toggle());

        // Keyboard navigation
        this.trigger.addEventListener('keydown', (e) => this.handleTriggerKeydown(e));
        this.menu.addEventListener('keydown', (e) => this.handleMenuKeydown(e));

        // Close on outside click
        document.addEventListener('click', (e) => {
          if (!this.container.contains(e.target)) {
            this.close();
          }
        });

        // Menu item click
        this.menuItems.forEach((item, index) => {
          item.addEventListener('click', () => {
            if (item.getAttribute('aria-disabled') !== 'true') {
              console.log('Selected:', item.textContent);
              this.close();
            }
          });
        });
      }

      toggle() {
        const isExpanded = this.trigger.getAttribute('aria-expanded') === 'true';
        isExpanded ? this.close() : this.open();
      }

      open() {
        this.menu.classList.add('visible');
        this.trigger.setAttribute('aria-expanded', 'true');
        this.focusedIndex = -1;
        this.menuItems[0]?.focus();
        this.focusedIndex = 0;
      }

      close() {
        this.menu.classList.remove('visible');
        this.trigger.setAttribute('aria-expanded', 'false');
        this.trigger.focus();
        this.focusedIndex = -1;
      }

      handleTriggerKeydown(e) {
        if (e.key === 'Enter' || e.key === ' ' || e.key === 'ArrowDown') {
          e.preventDefault();
          this.open();
        }
      }

      handleMenuKeydown(e) {
        const enabledItems = this.menuItems.filter(
          item => item.getAttribute('aria-disabled') !== 'true'
        );

        switch (e.key) {
          case 'ArrowDown':
            e.preventDefault();
            this.focusedIndex = (this.focusedIndex + 1) % enabledItems.length;
            enabledItems[this.focusedIndex].focus();
            break;

          case 'ArrowUp':
            e.preventDefault();
            this.focusedIndex = this.focusedIndex <= 0 
              ? enabledItems.length - 1 
              : this.focusedIndex - 1;
            enabledItems[this.focusedIndex].focus();
            break;

          case 'Escape':
            this.close();
            break;

          case 'Home':
            e.preventDefault();
            enabledItems[0].focus();
            this.focusedIndex = 0;
            break;

          case 'End':
            e.preventDefault();
            enabledItems[enabledItems.length - 1].focus();
            this.focusedIndex = enabledItems.length - 1;
            break;
        }
      }
    }

    // Initialize menu
    document.querySelectorAll('.menu-container').forEach(container => {
      new AccessibleMenu(container);
    });
  </script>
</body>
</html>
```

**Why**: The `role="menu"` creates a proper accessible menu pattern for screen readers. Combined with `aria-haspopup`, `aria-expanded`, and keyboard navigation, it ensures users can operate dropdown menus with keyboard alone, meeting WCAG accessibility guidelines.

    class Accordion {
      constructor(accordionElem) {
        this.accordion = accordionElem;
        this.triggers = Array.from(
          accordionElem.querySelectorAll('.accordion-trigger')
        );
        this.allowMultiple = accordionElem.hasAttribute('data-allow-multiple');
        
        this.bindEvents();
      }

      bindEvents() {
        this.triggers.forEach(trigger => {
          trigger.addEventListener('click', () => this.toggle(trigger));
          trigger.addEventListener('keydown', (e) => this.handleKeydown(e, trigger));
        });
      }

      toggle(trigger) {
        const isExpanded = trigger.getAttribute('aria-expanded') === 'true';
        const panelId = trigger.getAttribute('aria-controls');
        const panel = document.getElementById(panelId);

        if (!this.allowMultiple) {
          // Close all other panels
          this.triggers.forEach(otherTrigger => {
            if (otherTrigger !== trigger) {
              const otherPanelId = otherTrigger.getAttribute('aria-controls');
              const otherPanel = document.getElementById(otherPanelId);
              otherTrigger.setAttribute('aria-expanded', 'false');
              otherPanel.setAttribute('aria-hidden', 'true');
            }
          });
        }

        // Toggle current panel
        trigger.setAttribute('aria-expanded', !isExpanded);
        panel.setAttribute('aria-hidden', isExpanded);
      }

      handleKeydown(e, trigger) {
        const triggersArray = this.triggers;
        const index = triggersArray.indexOf(trigger);

        switch (e.key) {
          case 'ArrowDown':
            e.preventDefault();
            const nextIndex = (index + 1) % triggersArray.length;
            triggersArray[nextIndex].focus();
            break;

          case 'ArrowUp':
            e.preventDefault();
            const prevIndex = (index - 1 + triggersArray.length) % triggersArray.length;
            triggersArray[prevIndex].focus();
            break;

          case 'Home':
            e.preventDefault();
            triggersArray[0].focus();
            break;

          case 'End':
            e.preventDefault();
            triggersArray[triggersArray.length - 1].focus();
            break;
        }
      }
    }

    // Initialize accordions
    document.querySelectorAll('.accordion').forEach(accordion => {
      new Accordion(accordion);
    });

    // Disclosure button functionality
    document.querySelectorAll('.disclosure-button').forEach(button => {
      button.addEventListener('click', () => {
        const isExpanded = button.getAttribute('aria-expanded') === 'true';
        const panelId = button.getAttribute('aria-controls');
        const panel = document.getElementById(panelId);

        button.setAttribute('aria-expanded', !isExpanded);
        panel.setAttribute('aria-hidden', isExpanded);

        // Update button text
        button.textContent = isExpanded 
          ? 'Show More Information' 
          : 'Hide Information';
      });
    });
  </script>
</body>
</html>
```

**Why**: `aria-expanded` is critical for screen reader users to understand whether a collapsible section is open or closed. Combined with `aria-controls` and `aria-hidden`, it creates a complete accessible pattern for accordions, menus, and disclosure widgets per WAI-ARIA best practices.

    function setOpenGraphUrl(url) {
      // Remove existing og:url meta tag
      const existingOgUrl = document.querySelector('meta[property="og:url"]');
      if (existingOgUrl) {
        existingOgUrl.remove();
      }

      // Create and append new og:url meta tag
      const meta = document.createElement('meta');
      meta.setAttribute('property', 'og:url');
      meta.setAttribute('content', url);
      document.head.appendChild(meta);

      // Also update canonical link
      let canonical = document.querySelector('link[rel="canonical"]');
      if (canonical) {
        canonical.setAttribute('href', url);
      } else {
        canonical = document.createElement('link');
        canonical.setAttribute('rel', 'canonical');

        document.head.appendChild(canonical);
