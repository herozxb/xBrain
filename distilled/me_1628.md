## Tim Sort
```python
# PROBLEM: Hybrid stable sorting algorithm combining merge sort and insertion sort,
#          optimized for real-world data with natural runs and adaptive behavior
# TIME: O(n log n) worst case, O(n) best case (already sorted)
# SPACE: O(n)
def tim_sort(arr):
    """
    Tim Sort - Python's built-in sorting algorithm.
    Uses insertion sort for small runs, merge sort for combining runs.
    """
    if not arr or len(arr) <= 1:
        return arr
    
    n = len(arr)
    min_run = _calculate_min_run(n)
    
    # Sort individual subarrays of size min_run using insertion sort
    for start in range(0, n, min_run):
        end = min(start + min_run, n)
        _insertion_sort(arr, start, end)
    
    # Start merging from size min_run (or 1 if min_run is 1)
    size = min_run
    while size < n:
        # Pick starting point of left sub array
        for left in range(0, n, 2 * size):
            mid = min(left + size, n)
            right = min(left + 2 * size, n)
            
            # Merge subarrays arr[left..mid-1] and arr[mid..right-1]
            if mid < right:
                _merge(arr, left, mid, right)
        
        size *= 2
    
    return arr

def _calculate_min_run(n):
    """Calculate minimum run length (between 32 and 64)."""
    min_run = 32
    r = 0
    while n >= min_run:
        r |= n & 1
        n >>= 1
    return n + r

def _insertion_sort(arr, left, right):
    """Insertion sort for small subarrays."""
    for i in range(left + 1, right):
        key = arr[i]
        j = i - 1
        while j >= left and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key

def _merge(arr, left, mid, right):
    """Merge two sorted subarrays with galloping mode optimization."""
    left_arr = arr[left:mid]
    right_arr = arr[mid:right]
    
    i = j = 0
    k = left
    
    while i < len(left_arr) and j < len(right_arr):
        if left_arr[i] <= right_arr[j]:
            arr[k] = left_arr[i]
            i += 1
        else:
            arr[k] = right_arr[j]
            j += 1
        k += 1
    
    # Copy remaining elements
    while i < len(left_arr):
        arr[k] = left_arr[i]
        i += 1
        k += 1
    
    while j < len(right_arr):
        arr[k] = right_arr[j]
        j += 1
        k += 1
```
**Why**: Tim Sort is the default sorting algorithm in Python and Java. It's highly optimized for real-world data, which often contains pre-sorted sequences. It adapts to data patterns and provides stable sorting with O(n) best-case performance on already sorted data.

## Boyer-Moore String Search
```python
# PROBLEM: Efficient string matching using bad character and good suffix heuristics
# TIME: O(n/m) best case, O(n + m) average, O(nm) worst case
# SPACE: O(k) where k is alphabet size
def boyer_moore_search(text, pattern):
    """
    Boyer-Moore algorithm skips sections of text using two heuristics:
    1. Bad Character: Skip to last occurrence of mismatched character
    2. Good Suffix: Use already-matched suffix to skip ahead
    """
    if not pattern or not text or len(pattern) > len(text):
        return []
    
    n, m = len(text), len(pattern)
    result = []
    
    # Preprocess pattern for both heuristics
    bad_char = _build_bad_char_table(pattern)
    good_suffix = _build_good_suffix_table(pattern)
    
    s = 0  # Shift of pattern with respect to text
    while s <= n - m:
        j = m - 1  # Start matching from rightmost character
        
        # Keep reducing index j while characters match
        while j >= 0 and pattern[j] == text[s + j]:
            j -= 1
        
        # Pattern found
        if j < 0:
            result.append(s)
            # Shift pattern to find next occurrence
            s += good_suffix[0] if s + m < n else 1
        else:
            # Use maximum of both heuristics for maximum skip
            bad_char_shift = j - bad_char.get(text[s + j], -1)
            good_suffix_shift = good_suffix[j + 1]
            s += max(bad_char_shift, good_suffix_shift)
    
    return result

def _build_bad_char_table(pattern):
    """
    Build bad character heuristic table.
    Stores the last occurrence of each character in pattern.
    """
    bad_char = {}
    m = len(pattern)
    
    for i in range(m):
        bad_char[pattern[i]] = i
    
    return bad_char

def _build_good_suffix_table(pattern):
    """
    Build good suffix heuristic table.
    Determines how much to shift when a suffix has been matched.
    """
    m = len(pattern)
    good_suffix = [0] * (m + 1)
    
    # First pass: find border positions
    border_pos = [0] * (m + 1)
    i = m
    j = m + 1
    border_pos[i] = j
    
    while i > 0:
        while j <= m and pattern[i - 1] != pattern[j - 1]:
            if good_suffix[j] == 0:
                good_suffix[j] = j - i
            j = border_pos[j]
        
        i -= 1
        j -= 1
        border_pos[i] = j
    
    # Second pass: set shift values
    j = border_pos[0]
    for i in range(m + 1):
        if good_suffix[i] == 0:
            good_suffix[i] = j
        if i == j:
            j = border_pos[j]
    
    return good_suffix

# Example usage
if __name__ == "__main__":
    text = "ABAAABCDABCABCABCD"
    pattern = "ABC"
    matches = boyer_moore_search(text, pattern)
    print(f"Pattern found at positions: {matches}")  # [5, 8, 11]
```
**Why**: Boyer-Moore is one of the most efficient string matching algorithms in practice. It often achieves sublinear time complexity by skipping characters entirely. This makes it ideal for text editors, search engines, and DNA sequence matching where the pattern is typically much smaller than the text.

## Dinic's Max Flow Algorithm
```python
# PROBLEM: Find maximum flow in a flow network using blocking flows
# TIME: O(V²E) for general graphs, O(E√V) for bipartite matching
# SPACE: O(V + E)
from collections import deque

class DinicMaxFlow:
    """
    Dinic's algorithm uses BFS for level graph construction
    and DFS for finding blocking flows.
    """
    
    def __init__(self, num_vertices):
        self.num_vertices = num_vertices
        self.adj = [[] for _ in range(num_vertices)]
        self.capacity = {}
        self.flow = {}
    
    def add_edge(self, u, v, cap):
        """Add directed edge with capacity."""
        self.adj[u].append(v)
        self.adj[v].append(u)  # Residual edge
        self.capacity[(u, v)] = cap
        self.capacity[(v, u)] = 0
        self.flow[(u, v)] = 0
        self.flow[(v, u)] = 0
    
    def max_flow(self, source, sink):
        """Calculate maximum flow from source to sink."""
        max_flow = 0
        level = [0] * self.num_vertices
        
        # Repeat while there's an augmenting path
        while self._bfs_level(source, sink, level):
            # DFS to find blocking flow
            ptr = [0] * self.num_vertices
            while True:
                pushed = self._dfs_send_flow(source, sink, float('inf'), ptr, level)
                if pushed == 0:
                    break
                max_flow += pushed
        
        return max_flow
    
    def _bfs_level(self, source, sink, level):
        """Build level graph using BFS."""
        level[:] = [0] * self.num_vertices
        level[source] = 1
        
        queue = deque([source])
        while queue:
            u = queue.popleft()
            
            for v in self.adj[u]:
                # Only consider edges with remaining capacity
                if level[v] == 0 and self._remaining_capacity(u, v) > 0:
                    level[v] = level[u] + 1
                    queue.append(v)
        
        return level[sink] != 0
    
    def _dfs_send_flow(self, u, sink, flow_in, ptr, level):
        """Send flow using DFS with current arc optimization."""
        if u == sink:
            return flow_in
        
        # Try all edges from ptr[u] onwards
        for i in range(ptr[u], len(self.adj[u])):
            v = self.adj[u][i]
            remaining = self._remaining_capacity(u, v)
            
            # Check if edge is admissible (correct level and capacity)
            if level[v] == level[u] + 1 and remaining > 0:
                # Recursively send flow
                pushed = self._dfs_send_flow(
                    v, sink, min(flow_in, remaining), ptr, level
                )
                
                if pushed > 0:
                    # Update flow
                    self.flow[(u, v)] += pushed
                    self.flow[(v, u)] -= pushed
                    return pushed
            
            ptr[u] += 1
        
        return 0
    
    def _remaining_capacity(self, u, v):
        """Get remaining capacity of edge."""
        return self.capacity[(u, v)] - self.flow[(u, v)]
    
    def get_min_cut(self, source):
        """Find vertices reachable from source in residual graph."""
        visited = [False] * self.num_vertices
        queue = deque([source])
        visited[source] = True
        
        while queue:
            u = queue.popleft()
            for v in self.adj[u]:
                if not visited[v] and self._remaining_capacity(u, v) > 0:
                    visited[v] = True
                    queue.append(v)
        
        return visited

# Example: Network flow problem
if __name__ == "__main__":
    # Create graph with 6 vertices
    dinic = DinicMaxFlow(6)
    
    # Add edges (source=0, sink=5)
    dinic.add_edge(0, 1, 10)
    dinic.add_edge(0, 2, 10)
    dinic.add_edge(1, 2, 2)
    dinic.add_edge(1, 3, 4)
    dinic.add_edge(1, 4, 8)
    dinic.add_edge(2, 4, 9)
    dinic.add_edge(3, 5, 10)
    dinic.add_edge(4, 3, 6)
    dinic.add_edge(4, 5, 10)
    
    max_flow = dinic.max_flow(0, 5)
    print(f"Maximum flow: {max_flow}")  # Output: 19
```
**Why**: Dinic's algorithm is one of the most efficient max flow algorithms, especially for bipartite matching. It's widely used in network routing, image segmentation, airline scheduling, and resource allocation problems. The combination of level graphs and blocking flows ensures strong theoretical guarantees.

## Red-Black Tree
```python
# PROBLEM: Self-balancing BST maintaining balance through color properties
# TIME: O(log n) for insert, delete, search
# SPACE: O(n)
class RedBlackNode:
    """Node with color (RED=0, BLACK=1) for balancing."""
    __slots__ = ['key', 'value', 'color', 'left', 'right', 'parent']
    
    def __init__(self, key, value=None, color=0):
        self.key = key
        self.value = value
        self.color = color  # 0=RED, 1=BLACK
        self.left = None
        self.right = None
        self.parent = None

class RedBlackTree:
    """
    Red-Black Tree properties:
    1. Every node is either red or black
    2. Root is black
    3. No two consecutive red nodes
    4. Every path from root to leaf has same black height
    """
    
    RED = 0
    BLACK = 1
    
    def __init__(self):
        self.NIL = RedBlackNode(None, color=1)  # Sentinel node
        self.root = self.NIL
    
    def insert(self, key, value=None):
        """Insert key-value pair maintaining RB properties."""
        node = RedBlackNode(key, value, self.RED)
        node.left = node.right = self.NIL
        
        # Standard BST insert
        parent = None
        current = self.root
        
        while current != self.NIL:
            parent = current
            if key < current.key:
                current = current.left
            elif key > current.key:
                current = current.right
            else:
                current.value = value  # Update existing
                return
        
        node.parent = parent
        if parent is None:
            self.root = node
        elif key < parent.key:
            parent.left = node
        else:
            parent.right = node
        
        # Fix Red-Black violations
        self._fix_insert(node)
    
    def _fix_insert(self, node):
        """Fix violations after insertion using rotations and recoloring."""
        while node.parent and node.parent.color == self.RED:
            if node.parent == node.parent.parent.left:
                uncle = node.parent.parent.right
                
                if uncle.color == self.RED:
                    # Case 1: Uncle is red - recolor
                    node.parent.color = self.BLACK
                    uncle.color = self.BLACK
                    node.parent.parent.color = self.RED
                    node = node.parent.parent
                else:
                    if node == node.parent.right:
                        # Case 2: Left rotation needed
                        node = node.parent
                        self._rotate_left(node)
                    
                    # Case 3: Right rotation + recolor
                    node.parent.color = self.BLACK
                    node.parent.parent.color = self.RED
                    self._rotate_right(node.parent.parent)
            else:
                # Mirror cases for right side
                uncle = node.parent.parent.left
                
                if uncle.color == self.RED:
                    node.parent.color = self.BLACK
                    uncle.color = self.BLACK
                    node.parent.parent.color = self.RED
                    node = node.parent.parent
                else:
                    if node == node.parent.left:
                        node = node.parent
                        self._rotate_right(node)
                    
                    node.parent.color = self.BLACK
                    node.parent.parent.color = self.RED
                    self._rotate_left(node.parent.parent)
        
        self.root.color = self.BLACK
    
    def _rotate_left(self, x):
        """Left rotation around node x."""
        y = x.right
        x.right = y.left
        
        if y.left != self.NIL:
            y.left.parent = x
        
        y.parent = x.parent
        
        if x.parent is None:
            self.root = y
        elif x == x.parent.left:
            x.parent.left = y
        else:
            x.parent.right = y
        
        y.left = x
        x.parent = y
    
    def _rotate_right(self, x):
        """Right rotation around node x."""
        y = x.left
        x.left = y.right
        
        if y.right != self.NIL:
            y.right.parent = x
        
        y.parent = x.parent
        
        if x.parent is None:
            self.root = y
        elif x == x.parent.right:
            x.parent.right = y
        else:
            x.parent.left = y
        
        y.right = x
        x.parent = y
    
    def search(self, key):
        """Search for key in tree."""
        node = self.root
        while node != self.NIL:
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                return node.value
        return None
    
    def inorder(self):
        """In-order traversal yielding sorted keys."""
        def _inorder(node):
            if node != self.NIL:
                yield from _inorder(node.left)
                yield (node.key, node.value)
                yield from _inorder(node.right)
        
        return list(_inorder(self.root))

# Example usage
if __name__ == "__main__":
    rbt = RedBlackTree()
    
    # Insert elements
    for i in [7, 3, 18, 10, 22, 8, 11, 26]:
        rbt.insert(i, f"value_{i}")
    
    print("In-order traversal:", rbt.inorder())
    print("Search for 10:", rbt.search(10))
    print("Search for 99:", rbt.search(99))
```
**Why**: Red-Black trees are the backbone of many language standard libraries (C++ std::map, Java TreeMap). They provide guaranteed O(log n) operations while being simpler to implement than AVL trees. The color-based balancing makes them ideal for database indexes and filesystem implementations.

## Composite Pattern
```python
# PROBLEM: Compose objects into tree structures to represent part-whole hierarchies
# TIME: O(n) for operations where n is number of children
# SPACE: O(n) for tree structure
from abc import ABC, abstractmethod
from typing import List, Optional

class FileSystemComponent(ABC):
    """Abstract base class for files and directories."""
    
    @abstractmethod
    def display(self, indent: int = 0) -> None:
        """Display the component."""
        pass
    
    @abstractmethod
    def get_size(self) -> int:
        """Get size in bytes."""
        pass

class File(FileSystemComponent):
    """Leaf component - represents a file."""
    
    def __init__(self, name: str, size: int):
        self.name = name
        self.size = size
    
    def display(self, indent: int = 0) -> None:
        print("  " * indent + f"📄 {self.name} ({self.size} bytes)")
    
    def get_size(self) -> int:
        return self.size

class Directory(FileSystemComponent):
    """Composite component - represents a directory."""
    
    def __init__(self, name: str):
        self.name = name
        self.children: List[FileSystemComponent] = []
    
    def add(self, component: FileSystemComponent) -> None:
        """Add child component."""
        self.children.append(component)
    
    def remove(self, component: FileSystemComponent) -> None:
        """Remove child component."""
        self.children.remove(component)
    
    def get_child(self, index: int) -> Optional[FileSystemComponent]:
        """Get child by index."""
        return self.children[index] if index < len(self.children) else None
    
    def display(self, indent: int = 0) -> None:
        print("  " * indent + f"📁 {self.name}/")
        for child in self.children:
            child.display(indent + 1)
    
    def get_size(self) -> int:
        """Recursively calculate total size."""
        return sum(child.get_size() for child in self.children)
    
    def find_by_name(self, name: str) -> Optional[FileSystemComponent]:
        """Search for component by name."""
        for child in self.children:
            if isinstance(child, File) and child.name == name:
                return child
            elif isinstance(child, Directory):
                if child.name == name:
                    return child
                result = child.find_by_name(name)
                if result:
                    return result
        return None

# Example: Building a file system structure
if __name__ == "__main__":
    # Create root directory
    root = Directory("root")
    
    # Create subdirectories
    documents = Directory("documents")
    pictures = Directory("pictures")
    code = Directory("code")
    
    # Create files
    readme = File("README.md", 1024)
    todo = File("TODO.txt", 512)
    photo1 = File("vacation.jpg", 2048000)
    photo2 = File("family.png", 1536000)
    main_py = File("main.py", 4096)
    utils_py = File("utils.py", 2048)
    
    # Build hierarchy
    root.add(readme)
    root.add(documents)
    root.add(pictures)
    root.add(code)
    
    documents.add(todo)
    pictures.add(photo1)
    pictures.add(photo2)
    code.add(main_py)
    code.add(utils_py)
    
    # Display structure
    print("File System Structure:")
    root.display()
    
    # Calculate sizes
    print(f"\nRoot directory size: {root.get_size():,} bytes")
    print(f"Pictures directory size: {pictures.get_size():,} bytes")
    
    # Search for file
    found = root.find_by_name("main.py")
    print(f"\nFound file: {found.name if found else 'Not found'}")
```
**Why**: The Composite pattern lets clients treat individual objects and compositions uniformly. It's essential for GUI frameworks (widgets contain widgets), file systems, organization charts, and any tree-structured data. It simplifies client code by eliminating type checks and enables recursive algorithms naturally.

## Database Sharding Pattern
```python
# PROBLEM: Distribute data across multiple database instances for scalability
# TIME: O(1) for shard routing, O(log n) for range-based queries
# SPACE: O(k) where k is number of shards
import hashlib
from typing import Dict, List, Any, Optional
from dataclasses import dataclass
from enum import Enum

class ShardStrategy(Enum):
    """Sharding strategies."""
    HASH = "hash"           # Hash-based sharding
    RANGE = "range"         # Range-based sharding
    DIRECTORY = "directory" # Directory-based lookup

@dataclass
class ShardConfig:
    """Configuration for a database shard."""
    shard_id: int
    host: str
    port: int
    min_key: Optional[int] = None  # For range sharding
    max_key: Optional[int] = None

class DatabaseShardingManager:
    """
    Manages data distribution across multiple database shards.
    Supports multiple sharding strategies.
    """
    
    def __init__(self, strategy: ShardStrategy = ShardStrategy.HASH):
        self.strategy = strategy
        self.shards: Dict[int, ShardConfig] = {}
        self.directory: Dict[Any, int] = {}  # For directory-based sharding
        self.consistent_hash_ring: List[tuple] = []  # For consistent hashing
    
    def add_shard(self, config: ShardConfig) -> None:
        """Register a new shard."""
        self.shards[config.shard_id] = config
        
        if self.strategy == ShardStrategy.RANGE:
            # Keep shards sorted by range
            self.shards = dict(sorted(
                self.shards.items(),
                key=lambda x: x[1].min_key or 0
            ))
        elif self.strategy == ShardStrategy.HASH:
            self._build_hash_ring()
    
    def _build_hash_ring(self) -> None:
        """Build consistent hash ring for better distribution."""
        self.consistent_hash_ring = []
        virtual_nodes = 150  # Virtual nodes per physical node
        
        for shard_id, config in self.shards.items():
            for i in range(virtual_nodes):
                key = f"{config.host}:{config.port}:{i}"
                hash_val = int(hashlib.md5(key.encode()).hexdigest(), 16)
                self.consistent_hash_ring.append((hash_val, shard_id))
        
        self.consistent_hash_ring.sort()
    
    def get_shard(self, key: Any) -> Optional[ShardConfig]:
        """Determine which shard should handle the key."""
        if self.strategy == ShardStrategy.HASH:
            return self._hash_shard(key)
        elif self.strategy == ShardStrategy.RANGE:
            return self._range_shard(key)
        elif self.strategy == ShardStrategy.DIRECTORY:
            return self._directory_shard(key)
    
    def _hash_shard(self, key: Any) -> Optional[ShardConfig]:
        """Route using consistent hashing."""
        if not self.consistent_hash_ring:
            return None
        
        key_hash = int(hashlib.md5(str(key).encode()).hexdigest(), 16)
        
        # Find first node with hash >= key_hash
        for ring_hash, shard_id in self.consistent_hash_ring:
            if ring_hash >= key_hash:
                return self.shards[shard_id]
        
        # Wrap around to first node
        return self.shards[self.consistent_hash_ring[0][1]]
    
    def _range_shard(self, key: int) -> Optional[ShardConfig]:
        """Route based on key ranges."""
        for config in self.shards.values():
            if config.min_key is not None and config.max_key is not None:
                if config.min_key <= key < config.max_key:
                    return config
        return None
    
    def _directory_shard(self, key: Any) -> Optional[ShardConfig]:
        """Route using lookup directory."""
        shard_id = self.directory.get(key)
        return self.shards.get(shard_id) if shard_id else None
    
    def add_directory_mapping(self, key: Any, shard_id: int) -> None:
        """Add mapping for directory-based sharding."""
        self.directory[key] = shard_id
    
    def execute_query(self, key: Any, query: str) -> str:
        """Route query to appropriate shard."""
        shard = self.get_shard(key)
        if shard:
            return f"Executing '{query}' on {shard.host}:{shard.port}"
        return "No shard found for key"
    
    def get_statistics(self) -> Dict[str, Any]:
        """Get shard statistics."""
        return {
            "strategy": self.strategy.value,
            "total_shards": len(self.shards),
            "shard_ids": list(self.shards.keys()),
            "directory_size": len(self.directory) if self.strategy == ShardStrategy.DIRECTORY else 0
        }

# Example: Setting up a sharded database cluster
if __name__ == "__main__":
    # Create sharding manager with consistent hashing
    manager = DatabaseShardingManager(ShardStrategy.HASH)
    
    # Add database shards
    manager.add_shard(ShardConfig(
        shard_id=1, host="db1.example.com", port=5432
    ))
    manager.add_shard(ShardConfig(
        shard_id=2, host="db2.example.com", port=5432
    ))
    manager.add_shard(ShardConfig(
        shard_id=3, host="db3.example.com", port=5432
    ))
    
    # Route queries based on user IDs
    user_ids = [1001, 2005, 3050, 4999, 5555]
    
    print("Routing users to shards:")
    for user_id in user_ids:
        shard = manager.get_shard(user_id)
        print(f"  User {user_id} -> {shard.host}")
    
    # Execute query
    result = manager.execute_query(1001, "SELECT * FROM orders")
    print(f"\n{result}")
    
    # Statistics
    stats = manager.get_statistics()
    print(f"\nSharding Stats: {stats}")
```
**Why**: Sharding is crucial for horizontal scaling of databases. It distributes load across multiple machines, enabling systems to handle massive datasets that exceed single-machine capacity. Used by major platforms like Facebook, Twitter, and Netflix to serve billions of users. Consistent hashing minimizes data movement when adding/removing shards.

```python
# PROBLEM: Handle asynchronous streams of data with proper resource management
# TIME: O(n) for processing n items, non-blocking
# SPACE: O(1) for streaming, O(n) if collecting
import asyncio
from typing import AsyncIterator, AsyncContextManager
from contextlib import asynccontextmanager
import aiohttp

class AsyncDataStream:
    """
    Demonstrates async generators for streaming data
    and async context managers for resource cleanup.
    """
    
    def __init__(self, batch_size: int = 10):
        self.batch_size = batch_size
    
    async def fetch_paginated_data(self, url: str) -> AsyncIterator[dict]:
        """
        Async generator for paginated API responses.
        Yields one page at a time.
        """
        page = 1
        has_more = True
        
        async with aiohttp.ClientSession() as session:
            while has_more:
                params = {"page": page, "size": self.batch_size}
                
                async with session.get(url, params=params) as response:
                    data = await response.json()
                    
                    # Yield the current page
                    yield {
                        "page": page,
                        "data": data.get("items", []),
                        "total": data.get("total", 0)
                    }
                    
                    # Check if more pages exist
                    has_more = len(data.get("items", [])) == self.batch_size
                    page += 1
                    
                    # Rate limiting
                    await asyncio.sleep(0.1)
    
    async def stream_items(self, data: list) -> AsyncIterator[dict]:
        """
        Async generator that yields items one at a time.
        Useful for processing large datasets incrementally.
        """
        for item in data:
            # Simulate async processing
            await asyncio.sleep(0.01)
            yield item
    
    async def batch_stream(self, data: list) -> AsyncIterator[list]:
        """
        Async generator yielding batches of items.
        Balances memory usage with throughput.
        """
        batch = []
        for item in data:
            batch.append(item)
            if len(batch) >= self.batch_size:
                yield batch
                batch = []
        
        # Yield remaining items
        if batch:
            yield batch

@asynccontextmanager
async def managed_connection(host: str, port: int):
    """
    Async context manager for database/network connections.
    Ensures proper cleanup even on exceptions.
    """
    print(f"Connecting to {host}:{port}...")
    connection = {"host": host, "port": port, "connected": True}
    
    try:
        yield connection
    finally:
        # Cleanup - always executed
        connection["connected"] = False
        print(f"Connection to {host}:{port} closed")

class AsyncResourcePool:
    """Async context manager pool pattern."""
    
    def __init__(self, max_connections: int = 5):
        self.max_connections = max_connections
        self._pool = asyncio.Queue(maxsize=max_connections)
        self._initialized = False
    
    async def initialize(self):
        """Pre-create connection pool."""
        for i in range(self.max_connections):
            await self._pool.put(f"connection_{i}")
        self._initialized = True
    
    @asynccontextmanager
    async def acquire(self):
        """Acquire a connection from the pool."""
        if not self._initialized:
            await self.initialize()
        
        connection = await self._pool.get()
        try:
            yield connection
        finally:
            await self._pool.put(connection)
    
    async def close_all(self):
        """Close all connections."""
        while not self._pool.empty():
            await self._pool.get()

# Example: Processing data stream with proper resource management
async def process_large_dataset():
    """Demonstrate async generators with context managers."""
    
    # Simulated large dataset
    dataset = [{"id": i, "value": i * 10} for i in range(100)]
    
    stream = AsyncDataStream(batch_size=10)
    
    # Process items as they arrive
    print("Streaming items:")
    count = 0
    async for item in stream.stream_items(dataset):
        count += 1
        if count <= 5:  # Show first 5
            print(f"  Processed: {item}")
    
    print(f"\nTotal items streamed: {count}")
    
    # Process in batches
    print("\nBatch processing:")
    async for batch in stream.batch_stream(dataset):
        print(f"  Batch size: {len(batch)}, IDs: {[i['id'] for i in batch[:3]]}...")
    
    # Use managed connection
    print("\nConnection management:")
    async with managed_connection("db.example.com", 5432) as conn:
        print(f"  Using connection: {conn}")
        # Simulate work
        await asyncio.sleep(0.1)
    
    # Use connection pool
    print("\nConnection pool:")
    pool = AsyncResourcePool(max_connections=3)
    
    async def worker(worker_id: int):
        async with pool.acquire() as conn:
            print(f"  Worker {worker_id} acquired {conn}")
            await asyncio.sleep(0.1)
            return f"result_{worker_id}"
    
    # Concurrent access to pool
    results = await asyncio.gather(*[worker(i) for i in range(5)])
    print(f"  Results: {results}")
    
    await pool.close_all()

if __name__ == "__main__":
    asyncio.run(process_large_dataset())
```
**Why**: Async generators enable efficient streaming of large datasets without loading everything into memory. Combined with async context managers, they provide clean resource management for I/O-bound operations. This pattern is essential for web scraping, real-time data processing, microservices communication, and handling massive API responses.
