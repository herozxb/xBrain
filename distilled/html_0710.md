```html
  <script>
    /* 
     * PROBLEM: Control access to class properties with validation
     * TIME: O(1) SPACE: O(1)
     * 
     * Getters and setters allow controlled access to private data.
     * Use them for validation, computed properties, and encapsulation.
     */
    
    class Temperature {
      // Private field (ES2022)
      #celsius = 0;
      
      constructor(celsius = 0) {
        this.#celsius = celsius;
      }
      
      // Getter: Access computed/read-only value
      get celsius() {
        return this.#celsius;
      }
      
      // Setter: Validate before setting
      set celsius(value) {
        if (value < -273.15) {
          throw new Error('Temperature below absolute zero!');
        }
        this.#celsius = value;
      }
      
      // Computed getter (derived from #celsius)
      get fahrenheit() {
        return (this.#celsius * 9/5) + 32;
      }
      
      // Setter for fahrenheit (converts to celsius internally)
      set fahrenheit(value) {
        this.celsius = (value - 32) * 5/9;
      }
      
      // Read-only getter (no setter)
      get kelvin() {
        return this.#celsius + 273.15;
      }
      
      // String representation
      toString() {
        return `${this.#celsius.toFixed(1)}°C / ${this.fahrenheit.toFixed(1)}°F / ${this.kelvin.toFixed(2)}K`;
      }
    }
    
    // Demo usage
    const output = document.getElementById('output');
    const temp = new Temperature(25);
    
    const results = [
      '=== Temperature Class Demo ===',
      '',
      'Initial (25°C):',
      temp.toString(),
      '',
      'Set via setter (celsius = 100):',
      (() => { temp.celsius = 100; return temp.toString(); })(),
      '',
      'Set via fahrenheit setter (68°F):',
      (() => { temp.fahrenheit = 68; return temp.toString(); })(),
      '',
      'Read kelvin (computed, read-only):',
      `${temp.kelvin.toFixed(2)}K`,
      '',
      'Attempting invalid value (-300°C):',
      (() => {
        try {
          temp.celsius = -300;
        } catch (e) {
          return `Error caught: ${e.message}`;
        }
      })()
    ];
    
    output.textContent = results.join('\n');
    
    // Log class structure

      Object.getOwnPropertyNames(Temperature.prototype)
        .filter(p => Object.getOwnPropertyDescriptor(Temperature.prototype, p).get)
```

```html
        channel.port2.postMessage(null);
```

```html
  <meta charset="UTF-8">
  <title>element.innerHTML Property</title>
</head>
<body>
  <h1>element.innerHTML - Read/Write HTML Content</h1>
  
  <div id="demo-box" style="padding: 20px; border: 2px solid #333;">
    <p>Original content</p>
  </div>
  
  <pre id="output"></pre>
  
  <script>
    /* 
     * PROBLEM: Get or set HTML content of an element
     * TIME: O(n) SPACE: O(n)
     * 
     * innerHTML returns the HTML markup inside an element as a string.
     * Setting it parses the string and replaces all children.
     * WARNING: Security risk with user input (XSS) - sanitize first!
     */
    
    const output = document.getElementById('output');
    const demoBox = document.getElementById('demo-box');
    const results = [];
    
    results.push('=== innerHTML Demo ===');
    results.push('');
    
    // Reading innerHTML
    results.push('--- Reading ---');
    results.push('Original innerHTML:');
    results.push(demoBox.innerHTML);
    results.push('');
    
    // Setting innerHTML - completely replaces content
    results.push('--- Setting (Replace) ---');
    demoBox.innerHTML = '<strong>Bold</strong> and <em>italic</em> text';
    results.push('New innerHTML:');
    results.push(demoBox.innerHTML);
    results.push('');
    
    // Building HTML dynamically
    results.push('--- Dynamic HTML Building ---');
    
    const users = [
      { name: 'Alice', role: 'Admin' },
      { name: 'Bob', role: 'User' },
      { name: 'Charlie', role: 'User' }
    ];
    
    // Build table HTML
    let tableHTML = '<table border="1" style="border-collapse: collapse; margin: 10px 0;">';
    tableHTML += '<tr><th>Name</th><th>Role</th></tr>';
    
    users.forEach(user => {
      tableHTML += `<tr>
        <td>${escapeHTML(user.name)}</td>
        <td>${escapeHTML(user.role)}</td>
      </tr>`;
    });
    
    tableHTML += '</table>';
    demoBox.innerHTML = tableHTML;
    
    results.push('Generated table HTML (see above)');
    results.push('');
    
    // SECURITY: XSS Prevention
    results.push('--- Security: XSS Prevention ---');
    
    // NEVER do this with user input:
    // element.innerHTML = userInput; // DANGEROUS!
    
    // Safe alternatives:
    results.push('Safe methods:');
    results.push('1. textContent - escapes HTML');
    results.push('2. Sanitize before innerHTML');
    results.push('3. Use DOM methods instead');
    results.push('');
    
    // Helper function to escape HTML
    function escapeHTML(str) {
      const div = document.createElement('div');
      div.textContent = str;
      return div.innerHTML;
    }
    
    // Demonstrate escaping
    const maliciousInput = '<script>alert("XSS!")</script>';
    results.push('Malicious input: ' + maliciousInput);
    results.push('Escaped: ' + escapeHTML(maliciousInput));
    results.push('');
    
    // Performance consideration
    results.push('--- Performance Tips ---');
    results.push('• innerHTML re-parses and rebuilds DOM');
    results.push('• For multiple updates, build string then assign once');
    results.push('• For frequent updates, consider DOM methods');
    results.push('');
    
    // innerHTML vs textContent
    results.push('--- innerHTML vs textContent ---');
    
    const compareDiv = document.createElement('div');
    compareDiv.innerHTML = '<b>Bold</b>';
    
    results.push('With <b>Bold</b>:');
    results.push('innerHTML: ' + compareDiv.innerHTML);
    results.push('textContent: ' + compareDiv.textContent);
    
    output.textContent = results.join('\n');
    
    // Clean up for this demo
    compareDiv.remove();
  </script>
</body>
</html>
```
**Why:** Efficiently renders HTML content but requires sanitization to prevent XSS attacks.

## [js_classList_add]
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>classList.add() Method</title>
  <style>
    .highlight { background: yellow; }
    .bold { font-weight: bold; }
    .italic { font-style: italic; }
    .large { font-size: 24px; }
    .red { color: red; }
    
    .box {
      padding: 20px;
      margin: 10px 0;
      border: 2px solid #333;
      transition: all 0.3s;
    }
  </style>
</head>
<body>
  <h1>classList.add() - Add CSS Classes</h1>
  
  <div id="demo" class="box">Demo Box - Click buttons to add classes</div>
  
  <div style="margin: 10px 0;">
    <button onclick="addClass('highlight')">Add Highlight</button>
    <button onclick="addClass('bold')">Add Bold</button>
    <button onclick="addClass('italic')">Add Italic</button>
    <button onclick="addClass('large', 'red')">Add Multiple</button>
    <button onclick="resetClasses()">Reset</button>
  </div>
  
  <pre id="output"></pre>
  
  <script>
    /* 
     * PROBLEM: Add one or more CSS classes to an element
     * TIME: O(n) where n = number of classes SPACE: O(1)
     * 
     * classList.add() appends classes without affecting existing ones.
     * Much cleaner than manipulating className string directly.
     */
    
    const demo = document.getElementById('demo');
    const output = document.getElementById('output');
    
    function updateOutput() {
      const results = [];
      results.push('=== classList.add() Demo ===');
      results.push('');
      results.push('Current classes: ' + demo.className);
      results.push('classList.length: ' + demo.classList.length);
      results.push('');
      results.push('classList entries:');
      demo.classList.forEach((cls, i) => {
        results.push(`  ${i + 1}. "${cls}"`);
      });
      results.push('');
      results.push('--- Methods ---');
      results.push('add("class") - Add class');
      results.push('add("a", "b") - Add multiple');
      results.push('remove("class") - Remove class');
      results.push('toggle("class") - Toggle on/off');
      results.push('contains("class") - Check exists');
      results.push('replace("old", "new") - Swap class');
      
      output.textContent = results.join('\n');
    }
    
    function addClass(...classes) {
      // classList.add accepts multiple arguments (ES6+)
      demo.classList.add(...classes);
      updateOutput();
    }
    
    function resetClasses() {
      // Remove all custom classes, keep 'box'
      demo.className = 'box';
      updateOutput();
    }
    
    // Initial output
    updateOutput();
    
    // Comparison: className vs classList
    console.log('=== className vs classList ===');
    
    // Old way (error-prone)
    const oldWay = document.createElement('div');
    oldWay.className = 'class1';
    // Adding another requires string manipulation
    oldWay.className += ' class2'; // Need space!
    // Risk of duplicates
    oldWay.className += ' class1'; // Now has duplicate!
    console.log('Old way:', oldWay.className);
    
    // New way (classList)
    const newWay = document.createElement('div');
    newWay.classList.add('class1');
    newWay.classList.add('class2');
    newWay.classList.add('class1'); // Ignored - no duplicate!
    console.log('New way:', newWay.className);
    
    // Practical example: Form validation
    function validateInput(input) {
      input.classList.remove('valid', 'invalid');
      
      if (input.value.length >= 3) {
        input.classList.add('valid');
        // Add multiple at once
        input.classList.add('highlight', 'bold');
      } else {
        input.classList.add('invalid');
      }
    }
    
    // Batch adding from array
    const classesToAdd = ['large', 'red'];
    // Spread operator works great with classList.add
    demo.classList.add(...classesToAdd);
  </script>
</body>
</html>
```
**Why:** Adds CSS classes cleanly without string manipulation or duplicate class issues.
