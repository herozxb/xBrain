```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>ARIA Modal Dialog</title>
    <style>
        .modal-overlay {
            position: fixed;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            background: rgba(0, 0, 0, 0.5);
            display: none;
        }
        .modal-overlay.active {
            display: flex;
            justify-content: center;
            align-items: center;
        }
        .modal {
            background: white;
            padding: 2rem;
            border-radius: 8px;
            max-width: 500px;
        }
    </style>
</head>
<body>
    <!-- 
        PROBLEM: Create accessible modal dialog with proper ARIA attributes
        TIME: O(1) SPACE: O(1)
        
        ARIA modal ensures screen readers announce the dialog correctly
        and trap focus within it while open.
    -->
    
    <button id="openModal">Open Modal</button>
    
    <div class="modal-overlay" 
         id="modalOverlay"
         role="dialog" 
         aria-modal="true" 
         aria-labelledby="modalTitle"
         aria-describedby="modalDesc">
        <div class="modal">
            <h2 id="modalTitle">Confirm Action</h2>
            <p id="modalDesc">Are you sure you want to proceed?</p>
            <button id="confirmBtn">Confirm</button>
            <button id="cancelBtn">Cancel</button>
        </div>
    </div>

    <script>
        const openBtn = document.getElementById('openModal');
        const overlay = document.getElementById('modalOverlay');
        const confirmBtn = document.getElementById('confirmBtn');
        const cancelBtn = document.getElementById('cancelBtn');
        let lastFocusedElement = null;
        
        function openModal() {
            lastFocusedElement = document.activeElement;
            overlay.classList.add('active');
            confirmBtn.focus();
            document.body.style.overflow = 'hidden';
        }
        
        function closeModal() {
            overlay.classList.remove('active');
            document.body.style.overflow = '';
            if (lastFocusedElement) {
                lastFocusedElement.focus();
            }
        }
        
        // Focus trap within modal
        overlay.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') {
                closeModal();
            }
            
            if (e.key === 'Tab') {
                const focusableElements = overlay.querySelectorAll(
                    'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
                );
                const first = focusableElements[0];
                const last = focusableElements[focusableElements.length - 1];
                
                if (e.shiftKey && document.activeElement === first) {
                    e.preventDefault();
                    last.focus();
                } else if (!e.shiftKey && document.activeElement === last) {
                    e.preventDefault();
                    first.focus();
                }
            }
        });
        
        openBtn.addEventListener('click', openModal);
        cancelBtn.addEventListener('click', closeModal);
        confirmBtn.addEventListener('click', () => {
            console.log('Confirmed!');
            closeModal();
        });
    </script>
</body>
</html>
```

**Why**: ARIA modal ensures accessibility compliance, traps focus, and provides proper semantics for screen readers to announce dialogs correctly.

## js_array_at

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Array.at() Method</title>
</head>
<body>
    <div id="output"></div>

    <script>
        // 
        // PROBLEM: Access array elements from the end without length calculation
        // TIME: O(1) SPACE: O(1)
        //
        // The at() method accepts negative indices to access elements
        // from the end of the array. arr.at(-1) is cleaner than arr[arr.length - 1].
        //
        
        const items = ['apple', 'banana', 'cherry', 'date', 'elderberry'];
        const output = document.getElementById('output');
        
        function log(message) {
            output.innerHTML += `<p>${message}</p>`;
        }
        
        // Traditional way (verbose)
        const lastOld = items[items.length - 1];
        log(`Last (old way): ${lastOld}`); // elderberry
        
        // New way with at()
        const lastNew = items.at(-1);
        log(`Last (at(-1)): ${lastNew}`); // elderberry
        
        const secondLast = items.at(-2);
        log(`Second to last: ${secondLast}`); // date
        
        // Positive indices work like bracket notation
        const first = items.at(0);
        log(`First: ${first}`); // apple
        
        // Out of bounds returns undefined (not an error)
        const outOfBounds = items.at(-100);
        log(`Out of bounds: ${outOfBounds}`); // undefined
        
        // Works with array-like objects using Array.from
        const nodeList = document.querySelectorAll('script');
        const lastScript = Array.from(nodeList).at(-1);
        log(`Last script tag: ${lastScript ? 'found' : 'none'}`);
        
        // Practical use: undo/redo stacks
        const history = ['state1', 'state2', 'state3'];
        function getLastState() {
            return history.at(-1);
        }
        function getPreviousState() {
            return history.at(-2);
        }
        log(`Current state: ${getLastState()}`); // state3
        log(`Previous state: ${getPreviousState()}`); // state2
        
        // Rotating carousel
        const slides = ['slide1', 'slide2', 'slide3'];
        let currentIndex = -1;
        
        function nextSlide() {
            currentIndex = (currentIndex - 1 + slides.length) % slides.length;
            return slides.at(currentIndex);
        }
        
        log(`Next slide: ${nextSlide()}`); // slide3
        log(`Next slide: ${nextSlide()}`); // slide2
    </script>
</body>
</html>
```

**Why**: Array.at() provides cleaner syntax for accessing elements from the end of arrays, especially useful for stacks, queues, and circular data structures.

## js_object_fromEntries

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Object.fromEntries()</title>
</head>
<body>
    <div id="output"></div>

    <script>
        // 
        // PROBLEM: Convert key-value pairs (entries/Map) back to object
        // TIME: O(n) SPACE: O(n)
        //
        // Object.fromEntries() transforms a list of key-value pairs into an object.
        // It's the reverse of Object.entries().
        //
        
        const output = document.getElementById('output');
        function log(msg) {
            output.innerHTML += `<p><strong>${msg}</strong></p>`;
        }
        function logObj(label, obj) {
            output.innerHTML += `<p>${label}: <code>${JSON.stringify(obj)}</code></p>`;
        }
        
        // Convert entries back to object
        const entries = [['name', 'Alice'], ['age', 30], ['city', 'NYC']];
        const obj = Object.fromEntries(entries);
        logObj('From entries', obj); // {name: "Alice", age: 30, city: "NYC"}
        
        // Convert Map to object
        const map = new Map([['a', 1], ['b', 2], ['c', 3]]);
        const objFromMap = Object.fromEntries(map);
        logObj('From Map', objFromMap); // {a: 1, b: 2, c: 3}
        
        // Filter object properties
        const scores = { alice: 95, bob: 62, charlie: 88, dave: 45 };
        const passing = Object.fromEntries(
            Object.entries(scores).filter(([name, score]) => score >= 70)
        );
        logObj('Passing scores', passing); // {alice: 95, charlie: 88}
        
        // Transform object keys
        const user = { firstName: 'John', lastName: 'Doe', emailAddress: 'john@example.com' };
        const transformed = Object.fromEntries(
            Object.entries(user).map(([key, value]) => [
                key.replace(/([A-Z])/g, '_$1').toLowerCase(), // camelCase to snake_case
                value
            ])
        );
        logObj('Snake case keys', transformed);
        
        // Merge multiple objects
        const defaults = { theme: 'light', fontSize: 14 };
        const userPrefs = { theme: 'dark', notifications: true };
        const merged = Object.fromEntries(
            new Map([...Object.entries(defaults), ...Object.entries(userPrefs)])
        );
        logObj('Merged config', merged);
        
        // URL query string parsing
        const queryString = 'name=John&age=30&city=NYC';
        const params = Object.fromEntries(
            new URLSearchParams(queryString)
        );
        logObj('URL params', params); // {name: "John", age: "30", city: "NYC"}
        
        // Object mapping with transformation
        const prices = { apple: 1.20, banana: 0.80, cherry: 2.50 };
        const withTax = Object.fromEntries(
            Object.entries(prices).map(([item, price]) => [
                item,
                { price, withTax: (price * 1.1).toFixed(2) }
            ])
        );
        logObj('Prices with tax', withTax);
        
        // Remove null/undefined values
        const dirty = { a: 1, b: null, c: 3, d: undefined, e: 5 };
        const clean = Object.fromEntries(
            Object.entries(dirty).filter(([_, value]) => value != null)
        );
        logObj('Cleaned object', clean); // {a: 1, c: 3, e: 5}
    </script>
</body>
</html>
```

**Why**: Object.fromEntries() enables clean object transformations, Map-to-object conversion, and functional object manipulation without mutation.

## js_fetch_timeout

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Fetch with Timeout</title>
</head>
<body>
    <button id="fetchBtn">Fetch with Timeout</button>
    <div id="output"></div>

    <script>
        // 
        // PROBLEM: Implement timeout for fetch requests to avoid hanging
        // TIME: O(1) SPACE: O(1)
        //
        // Fetch doesn't have built-in timeout. Use AbortController with
        // setTimeout to cancel requests that take too long.
        //
        
        const output = document.getElementById('output');
        const fetchBtn = document.getElementById('fetchBtn');
        
        function log(message) {
            output.innerHTML += `<p>${message}</p>`;
        }
        
        // Fetch with timeout wrapper
        function fetchWithTimeout(url, options = {}, timeout = 5000) {
            return new Promise((resolve, reject) => {
                const controller = new AbortController();
                const signal = controller.signal;
                
                // Set timeout to abort
                const timeoutId = setTimeout(() => {
                    controller.abort();
                    reject(new Error(`Request timeout after ${timeout}ms`));
                }, timeout);
                
                fetch(url, { ...options, signal })
                    .then(response => {
                        clearTimeout(timeoutId);
                        resolve(response);
                    })
                    .catch(error => {
                        clearTimeout(timeoutId);
                        if (error.name === 'AbortError') {
                            reject(new Error('Request was aborted'));
                        } else {
                            reject(error);
                        }
                    });
            });
        }
        
        // Alternative: Using Promise.race
        function fetchWithRace(url, options = {}, timeout = 5000) {
            const controller = new AbortController();
            
            const timeoutPromise = new Promise((_, reject) => {
                setTimeout(() => {
                    controller.abort();
                    reject(new Error(`Timeout after ${timeout}ms`));
                }, timeout);
            });
            
            return Promise.race([
                fetch(url, { ...options, signal: controller.signal }),
                timeoutPromise
            ]);
        }
        
        // Usage
        fetchBtn.addEventListener('click', async () => {
            output.innerHTML = '';
            log('Fetching...');
            
            try {
                const response = await fetchWithTimeout(
                    'https://jsonplaceholder.typicode.com/posts/1',
                    {},
                    3000 // 3 second timeout
                );
                const data = await response.json();
                log(`Success: ${JSON.stringify(data).substring(0, 100)}...`);
            } catch (error) {
                log(`Error: ${error.message}`);
            }
        });
        
        // Fetch with retry and timeout
        async function fetchWithRetry(url, options = {}, timeout = 5000, retries = 3) {
            for (let i = 0; i < retries; i++) {
                try {
                    const response = await fetchWithTimeout(url, options, timeout);
                    if (response.ok) {
                        return response;
                    }
                } catch (error) {
                    if (i === retries - 1) throw error;
                    log(`Retry ${i + 1}/${retries}...`);
                    await new Promise(r => setTimeout(r, 1000 * (i + 1))); // Exponential backoff
                }
            }
            throw new Error('All retries failed');
        }
        
        // Production-ready fetch wrapper
        class FetchClient {
            constructor(baseURL, defaultTimeout = 5000) {
                this.baseURL = baseURL;
                this.defaultTimeout = defaultTimeout;
            }
            
            async request(endpoint, options = {}) {
                const timeout = options.timeout || this.defaultTimeout;
                const url = this.baseURL + endpoint;
                
                const controller = new AbortController();
                const timeoutId = setTimeout(() => controller.abort(), timeout);
                
                try {
                    const response = await fetch(url, {
                        ...options,
                        signal: controller.signal
                    });
                    clearTimeout(timeoutId);
                    
                    if (!response.ok) {
                        throw new Error(`HTTP ${response.status}: ${response.statusText}`);
                    }
                    
                    return await response.json();
                } catch (error) {
                    clearTimeout(timeoutId);
                    throw error;
                }
            }
        }
    </script>
</body>
</html>
```

**Why**: Timeout prevents indefinitely hanging requests, improves user experience, and allows graceful error handling with retry logic.

## js_element_createDocumentFragment

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>createDocumentFragment</title>
</head>
<body>
    <ul id="list"></ul>
    <div id="output"></div>

    <script>
        // 
        // PROBLEM: Batch DOM insertions to avoid multiple reflows
        // TIME: O(n) SPACE: O(n)
        //
        // DocumentFragment is a lightweight container that doesn't trigger
        // reflow when modified. Inserting it appends all children at once.
        //
        
        const list = document.getElementById('list');
        const output = document.getElementById('output');
        
        function log(msg) {
            output.innerHTML += `<p>${msg}</p>`;
        }
        
        // BAD: Multiple DOM insertions (causes multiple reflows)
        function addItemBad(items) {
            console.time('Bad approach');
            items.forEach(item => {
                const li = document.createElement('li');
                li.textContent = item;
                list.appendChild(li); // Triggers reflow each time
            });
            console.timeEnd('Bad approach');
        }
        
        // GOOD: Single DOM insertion with fragment
        function addItemGood(items) {
            console.time('Good approach');
            const fragment = document.createDocumentFragment();
            
            items.forEach(item => {
                const li = document.createElement('li');
                li.textContent = item;
                fragment.appendChild(li); // No reflow
            });
            
            list.appendChild(fragment); // Single reflow
            console.timeEnd('Good approach');
        }
        
        // Example: Render 1000 items efficiently
        const items = Array.from({ length: 1000 }, (_, i) => `Item ${i + 1}`);
        
        // Clear and render
        list.innerHTML = '';
        addItemGood(items);
        log(`Rendered ${items.length} items efficiently`);
        
        // Building table with fragment
        function buildTable(data) {
            const fragment = document.createDocumentFragment();
            
            data.forEach(row => {
                const tr = document.createElement('tr');
                Object.values(row).forEach(cell => {
                    const td = document.createElement('td');
                    td.textContent = cell;
                    tr.appendChild(td);
                });
                fragment.appendChild(tr);
            });
            
            return fragment;
        }
        
        const tableData = [
            { name: 'Alice', age: 30, city: 'NYC' },
            { name: 'Bob', age: 25, city: 'LA' },
            { name: 'Charlie', age: 35, city: 'SF' }
        ];
        
        const table = document.createElement('table');
        table.appendChild(buildTable(tableData));
        document.body.appendChild(table);
        
        // Template rendering with fragment
        function renderTemplate(items, templateFn) {
            const fragment = document.createDocumentFragment();
            items.forEach(item => {
                const element = templateFn(item);
                if (element instanceof Node) {
                    fragment.appendChild(element);
                }
            });
            return fragment;
        }
        
        // Usage
        const users = [
            { id: 1, name: 'John', email: 'john@example.com' },
            { id: 2, name: 'Jane', email: 'jane@example.com' }
        ];
        
        const userTemplate = (user) => {
            const div = document.createElement('div');
            div.className = 'user-card';
            div.innerHTML = `<h3>${user.name}</h3><p>${user.email}</p>`;
            return div;
        };
        
        const userContainer = document.createElement('div');
        userContainer.id = 'users';
        userContainer.appendChild(renderTemplate(users, userTemplate));
        document.body.appendChild(userContainer);
        
        log('All items rendered with single DOM insertion');
    </script>
</body>
</html>
```

**Why**: DocumentFragment batches DOM modifications into a single reflow, dramatically improving performance when inserting multiple elements.

## js_async_await_parallel

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Async Await Parallel Execution</title>
</head>
<body>
    <button id="sequentialBtn">Load Sequential</button>
    <button id="parallelBtn">Load Parallel</button>
    <div id="output"></div>

    <script>
        // 
        // PROBLEM: Execute async operations in parallel instead of sequentially
        // TIME: O(1) with parallel execution SPACE: O(n)
        //
        // Use Promise.all() with async/await to run operations concurrently,
        // reducing total wait time to the longest operation.
        //
        
        const output = document.getElementById('output');
        
        function log(msg) {
            output.innerHTML += `<p>${msg}</p>`;
        }
        
        // Simulated API calls
        async function fetchUser() {
            await new Promise(r => setTimeout(r, 1000));
            return { id: 1, name: 'John' };
        }
        
        async function fetchPosts() {
            await new Promise(r => setTimeout(r, 1500));
            return [{ id: 1, title: 'Post 1' }, { id: 2, title: 'Post 2' }];
        }
        
        async function fetchComments() {
            await new Promise(r => setTimeout(r, 800));
            return [{ id: 1, text: 'Great!' }];
        }
        
        // SEQUENTIAL: Slow, waits for each to complete
        document.getElementById('sequentialBtn').addEventListener('click', async () => {
            output.innerHTML = '';
            console.time('Sequential');
            log('Starting sequential load...');
            
            const user = await fetchUser();      // 1s
            const posts = await fetchPosts();    // 1.5s
            const comments = await fetchComments(); // 0.8s
            // Total: ~3.3s
            
            console.timeEnd('Sequential');
            log(`User: ${user.name}`);
            log(`Posts: ${posts.length}`);
            log(`Comments: ${comments.length}`);
        });
        
        // PARALLEL: Fast, all run concurrently
        document.getElementById('parallelBtn').addEventListener('click', async () => {
            output.innerHTML = '';

            const [user, posts, comments] = await Promise.all([
                fetchUser(),

            // Total: ~1.5s (max of all)

        async function fetchAllData() {
            try {
                const [user, posts, comments] = await Promise.all([
                    fetchUser(),
                    fetchPosts(),
                    fetchComments()
                ]);
                return { user, posts, comments };
            } catch (error) {
                console.error('One or more requests failed:', error);
                throw error;
            }
        }
        
        // Promise.allSettled - doesn't fail if one rejects
        async function fetchWithFallback() {
            const results = await Promise.allSettled([
                fetchUser(),
                Promise.reject('Failed'),
                fetchComments()
            ]);
            
            const successful = results
                .filter(r => r.status === 'fulfilled')
                .map(r => r.value);
            
            const failed = results
                .filter(r => r.status === 'rejected')
                .map(r => r.reason);
            
            console.log('Successful:', successful);
            console.log('Failed:', failed);
        }
        
        // Promise.race - first to complete wins
        async function fetchWithRace() {
            const cachedData = new Promise(resolve => {
                setTimeout(() => resolve({ cached: true }), 500);
            });
            
            const freshData = new Promise(resolve => {
                setTimeout(() => resolve({ cached: false }), 1000);
            });
            
            const result = await Promise.race([cachedData, freshData]);
            console.log('First result:', result); // cached: true
            return result;
        }
        
        // Sequential with dependencies
        async function fetchWithDependencies() {
            // First fetch user, then use user.id for other fetches
            const user = await fetchUser();
            
            // These can run in parallel since they depend only on user
            const [posts, comments] = await Promise.all([
                fetchPosts(),
                fetchComments()
            ]);
            
            return { user, posts, comments };
        }
        
        // Batch parallel operations
        async function processBatch(items, batchSize = 3) {
            const results = [];
            
            for (let i = 0; i < items.length; i += batchSize) {
                const batch = items.slice(i, i + batchSize);
                const batchResults = await Promise.all(
                    batch.map(item => fetchUser())
                );
                results.push(...batchResults);
            }
            
            return results;
        }
```

**Why**: Parallel async/await with Promise.all() significantly reduces total execution time for independent operations, essential for efficient data loading.

## js_localStorage_key

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>localStorage.key() Method</title>
</head>
<body>
    <button id="showKeys">Show All Keys</button>
    <button id="clearAll">Clear All</button>
    <div id="output"></div>

    <script>
        // 
        // PROBLEM: Iterate over localStorage keys without knowing them
        // TIME: O(n) SPACE: O(n)
        //
        // localStorage.key(index) returns the key name at a given index.
        // Useful for debugging, exporting data, or bulk operations.
        //
        
        const output = document.getElementById('output');
        
        function log(msg) {
            output.innerHTML += `<p>${msg}</p>`;
        }
        
        // Setup: Add some test data
        localStorage.setItem('user_theme', 'dark');
        localStorage.setItem('user_lang', 'en');
        localStorage.setItem('session_token', 'abc123');
        localStorage.setItem('cart_count', '5');
        
        // Get all keys using .key()
        function getAllKeys() {
            const keys = [];
            for (let i = 0; i < localStorage.length; i++) {
                keys.push(localStorage.key(i));
            }
            return keys;
        }
        
        document.getElementById('showKeys').addEventListener('click', () => {
            output.innerHTML = '<h3>All localStorage keys:</h3>';
            
            // Method 1: Using key() with index
            log(`Total items: ${localStorage.length}`);
            
            for (let i = 0; i < localStorage.length; i++) {
                const key = localStorage.key(i);
                const value = localStorage.getItem(key);
                log(`${i}: ${key} = ${value}`);
            }
        });
        
        // Export all localStorage data
        function exportLocalStorage() {
            const data = {};
            for (let i = 0; i < localStorage.length; i++) {
                const key = localStorage.key(i);
                data[key] = localStorage.getItem(key);
            }
            return JSON.stringify(data, null, 2);
        }
        
        // Find keys by prefix
        function getKeysByPrefix(prefix) {
            const matchingKeys = [];
            for (let i = 0; i < localStorage.length; i++) {
                const key = localStorage.key(i);
                if (key.startsWith(prefix)) {
                    matchingKeys.push(key);
                }
            }
            return matchingKeys;
        }
        
        // Usage
        log('Keys starting with "user_": ' + getKeysByPrefix('user_').join(', '));
        
        // Remove keys by pattern
        function removeKeysByPattern(pattern) {
            const regex = new RegExp(pattern);
            const keysToRemove = [];
            
            for (let i = 0; i < localStorage.length; i++) {
                const key = localStorage.key(i);
                if (regex.test(key)) {
                    keysToRemove.push(key);
                }
            }
            
            keysToRemove.forEach(key => localStorage.removeItem(key));
            return keysToRemove.length;
        }
        
        // Get storage size
        function getStorageSize() {
            let total = 0;
            for (let i = 0; i < localStorage.length; i++) {
                const key = localStorage.key(i);
                const value = localStorage.getItem(key);
                total += key.length + value.length;
            }
            return (total * 2 / 1024).toFixed(2) + ' KB'; // UTF-16 = 2 bytes per char
        }
        
        log(`Storage size: ${getStorageSize()}`);
        
        // Clear all
        document.getElementById('clearAll').addEventListener('click', () => {
            const count = localStorage.length;
            localStorage.clear();
            log(`Cleared ${count} items from localStorage`);
        });
        
        // Alternative: Object.keys(localStorage)
        // Modern browsers also support direct iteration
        function getAllKeysModern() {
            return Object.keys(localStorage);
        }
        
        log('Modern method: ' + getAllKeysModern().join(', '));
        
        // Debug helper: Display all storage
        function displayAllStorage() {
            const container = document.createElement('div');
            container.style.cssText = 'background:#f5f5f5;padding:10px;margin:10px 0;';
            
            const title = document.createElement('h3');
            title.textContent = `localStorage (${localStorage.length} items)`;
            container.appendChild(title);
            
            for (let i = 0; i < localStorage.length; i++) {
                const key = localStorage.key(i);
                const value = localStorage.getItem(key);
                const item = document.createElement('div');
                item.innerHTML = `<strong>${key}:</strong> ${value}`;
                container.appendChild(item);
            }
            
            output.appendChild(container);
        }
        
        displayAllStorage();
    </script>
</body>
</html>
```

**Why**: localStorage.key() enables iteration over stored data for debugging, export, and bulk operations when you don't know the exact key names.
