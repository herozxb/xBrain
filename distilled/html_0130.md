## html5_url_createObjectURL

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>URL.createObjectURL Demo</title>
  <style>
    body { font-family: system-ui; padding: 20px; }
    .preview { max-width: 300px; margin-top: 10px; border: 1px solid #ccc; }
  </style>
</head>
<body>
  <h2>Create Object URL from File</h2>
  <!-- 
    PROBLEM: Create a temporary blob URL from a File or Blob object
    TIME: O(1) SPACE: O(1)
  -->
  <!-- 
    WHY: Allows instant preview of uploaded files without server roundtrip.
         The URL is valid only within the current document session.
  -->

  <input type="file" id="fileInput" accept="image/*">
  <div id="previewContainer"></div>

  <script>
    // Get references to DOM elements
    const fileInput = document.getElementById('fileInput');
    const previewContainer = document.getElementById('previewContainer');

    // Listen for file selection changes
    fileInput.addEventListener('change', (event) => {
      // Get the first selected file from the FileList
      const file = event.target.files[0];
      
      // Guard clause: exit if no file selected
      if (!file) return;

      // Create a temporary blob URL for the file
      // This URL points to the file in browser memory
      // Format: blob:origin/uuid
      const objectURL = URL.createObjectURL(file);

      // Create an image element to display the preview
      const img = document.createElement('img');
      img.src = objectURL;
      img.className = 'preview';
      img.alt = 'File preview';

      // Clear any existing preview and add the new one
      previewContainer.innerHTML = '';
      previewContainer.appendChild(img);

      // Log the generated URL for debugging
      console.log('Created object URL:', objectURL);
    });
  </script>
</body>
</html>
```

**Key Points:**
- Returns a `blob:` URL string referencing the Blob/File in memory
- URL is valid until revoked or the document is unloaded
- Useful for file previews, audio/video playback, downloads

## html5_url_revokeObjectURL

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>URL.revokeObjectURL Demo</title>
  <style>
    body { font-family: system-ui; padding: 20px; }
    .status { margin-top: 10px; padding: 10px; background: #f0f0f0; }
  </style>
</head>
<body>
  <h2>Revoke Object URL - Memory Management</h2>
  <!-- 
    PROBLEM: Free memory by invalidating a previously created blob URL
    TIME: O(1) SPACE: O(1) - releases memory
  -->
  <!-- 
    WHY: Prevents memory leaks. After revoking, the blob URL becomes invalid
         and the underlying Blob can be garbage collected if no other references exist.
  -->

  <input type="file" id="fileInput" accept="image/*">
  <div id="previewContainer"></div>
  <button id="revokeBtn" disabled>Revoke URL</button>
  <div id="status" class="status"></div>

  <script>
    // Store the current object URL for later revocation
    let currentObjectURL = null;

    const fileInput = document.getElementById('fileInput');
    const previewContainer = document.getElementById('previewContainer');
    const revokeBtn = document.getElementById('revokeBtn');
    const status = document.getElementById('status');

    // Create object URL when file is selected
    fileInput.addEventListener('change', (event) => {
      const file = event.target.files[0];
      if (!file) return;

      // Revoke any existing URL first to prevent memory leaks
      if (currentObjectURL) {
        URL.revokeObjectURL(currentObjectURL);
      }

      // Create new object URL
      currentObjectURL = URL.createObjectURL(file);

      // Display the preview
      previewContainer.innerHTML = `<img src="${currentObjectURL}" class="preview" alt="Preview">`;
      
      // Enable revoke button and update status
      revokeBtn.disabled = false;
      status.textContent = `Active URL: ${currentObjectURL}`;
    });

    // Revoke the object URL when button is clicked
    revokeBtn.addEventListener('click', () => {
      if (currentObjectURL) {
        // Invalidate the blob URL - it will no longer work
        URL.revokeObjectURL(currentObjectURL);
        
        status.textContent = `Revoked: ${currentObjectURL} (image will fail to load now)`;
        
        // Clear the stored URL
        currentObjectURL = null;
        revokeBtn.disabled = true;

        // The image may still appear cached, but new requests will fail
        console.log('Object URL has been revoked');
      }
    });

    // Clean up on page unload
    window.addEventListener('beforeunload', () => {
      if (currentObjectURL) {
        URL.revokeObjectURL(currentObjectURL);
      }
    });
  </script>
</body>
</html>
```

**Key Points:**
- Call `revokeObjectURL()` when the URL is no longer needed
- Essential for long-running applications with many file uploads
- Browser automatically revokes URLs on page unload, but explicit cleanup is best practice

## html5_history_api

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>History API Demo</title>
  <style>
    body { font-family: system-ui; padding: 20px; }
    nav a { margin-right: 15px; }
    #content { margin-top: 20px; padding: 20px; border: 1px solid #ccc; }
  </style>
</head>
<body>
  <h2>History API - pushState</h2>
  <!-- 
    PROBLEM: Navigate between views without page reload, updating browser history
    TIME: O(1) SPACE: O(1)
  -->
  <!-- 
    WHY: Enables Single Page Application (SPA) navigation while maintaining
         working back/forward buttons and shareable URLs.
  -->

  <nav>
    <a href="/home" data-page="home">Home</a>
    <a href="/about" data-page="about">About</a>
    <a href="/contact" data-page="contact">Contact</a>
  </nav>

  <div id="content">Select a page above</div>

  <script>
    // Page content data
    const pages = {
      home: { title: 'Home', content: 'Welcome to our SPA!' },
      about: { title: 'About', content: 'Learn more about us.' },
      contact: { title: 'Contact', content: 'Get in touch with our team.' }
    };

    const contentDiv = document.getElementById('content');

    // Function to render page content
    function renderPage(pageName) {
      const page = pages[pageName];
      if (page) {
        contentDiv.innerHTML = `<h3>${page.title}</h3><p>${page.content}</p>`;
        document.title = page.title + ' - My SPA';
      }
    }

    // Handle navigation clicks
    document.querySelectorAll('nav a').forEach(link => {
      link.addEventListener('click', (event) => {
        // Prevent default browser navigation
        event.preventDefault();

        const pageName = link.dataset.page;
        const url = link.getAttribute('href');

        // Push new state to history stack
        // Parameters: state object, title (ignored by most browsers), URL
        history.pushState({ page: pageName }, '', url);

        // Update the view
        renderPage(pageName);
      });
    });

    // Handle browser back/forward buttons
    // This event fires when active history entry changes
    window.addEventListener('popstate', (event) => {
      // event.state contains the state object we passed to pushState
      if (event.state && event.state.page) {
        renderPage(event.state.page);
      }
    });

    // Handle initial page load
    // Check if we have a state from history (e.g., after refresh)
    if (history.state && history.state.page) {
      renderPage(history.state.page);
    }
  </script>
</body>
</html>
```

**Key Points:**
- `pushState()` adds a new entry to history stack
- `popstate` event fires on back/forward navigation
- State object is serialized; avoid functions or circular references

## js_regexp_character_classes

```html
<!DOCTYPE html>
<html lang="en">
<head>

  <!-- 
    WHY: Character classes provide concise patterns for common character sets

      '\\w': 'Word character [a-zA-Z0-9_]',
      '\\W': 'Non-word character',
      '\\s': 'Whitespace [ \\t\\n\\r\\f\\v]',
      '\\S': 'Non-whitespace',
      '\\b': 'Word boundary',
      '\\B': 'Non-word boundary'
    };

    input.addEventListener('input', (event) => {
      const text = event.target.value;
      let output = '<strong>Character Class Tests:</strong><br>';

      // Test each character class
      output += `Contains digit (\\d): ${/\d/.test(text)}<br>`;
      output += `Contains letter (\\w without \\d): ${/[a-zA-Z]/.test(text)}<br>`;
      output += `Contains whitespace (\\s): ${/\s/.test(text)}<br>`;
      output += `Only word characters (\\w): /^\\w+$/.test → ${/^\w+$/.test(text)}<br>`;
      output += `Only digits (\\d): /^\\d+$/.test → ${/^\d+$/.test(text)}<br>`;

      // Custom character set examples
      output += `<br><strong>Custom Sets:</strong><br>`;
      output += `Vowels only: /^[aeiou]+$/i.test → ${/^[aeiou]+$/i.test(text)}<br>`;
      output += `Hex color format: /^#[0-9A-Fa-f]{6}$/.test → ${/^#[0-9A-Fa-f]{6}$/.test(text)}<br>`;

      // Negated character class
      output += `<br><strong>Negated Sets [^...]:</strong><br>`;
      output += `No vowels: /^[^aeiou]+$/i.test → ${/^[^aeiou]+$/i.test(text)}<br>`;

      // Word boundary example
      output += `<br><strong>Word Boundary \\b:</strong><br>`;
      const words = text.match(/\b\w+\b/g);
      output += `Words found: ${words ? words.join(', ') : 'none'}<br>`;

      results.innerHTML = output;
    });

    // Display character class reference on load
    console.log('Character Classes Reference:');
    Object.entries(classes).forEach(([cls, desc]) => {
      console.log(`  ${cls}: ${desc}`);
    });
  </script>
</body>
</html>
```

**Key Points:**
- `.` matches any char except newline (use `[\s\S]` for truly any char)
- `\w` includes underscores, useful for identifiers
- `\b` is zero-width; matches position, not characters

## js_regexp_groups

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>RegExp Groups</title>
  <style>
    body { font-family: system-ui; padding: 20px; }
    .result { background: #f5f5f5; padding: 10px; margin-top: 10px; font-family: monospace; }
  </style>
</head>
<body>
  <h2>RegExp Groups - Capturing & Non-Capturing</h2>
  <!-- 
    PROBLEM: Group patterns for reuse, alternation, and capturing matched content
    TIME: O(n) where n = string length SPACE: O(m) where m = number of groups
  -->
  <!-- 
    WHY: Groups enable extracting specific parts of matches, applying quantifiers
         to sub-patterns, and creating complex alternations.
  -->

  <h3>Phone Number Parser</h3>
  <input type="text" id="phone" placeholder="(123) 456-7890" value="(555) 123-4567">
  <div id="phoneResult" class="result"></div>

  <h3>Named Groups Demo</h3>
  <input type="text" id="date" placeholder="YYYY-MM-DD" value="2026-02-28">
  <div id="dateResult" class="result"></div>

  <script>
    // === CAPTURING GROUPS ===
    // Parentheses () create capturing groups
    const phoneInput = document.getElementById('phone');
    const phoneResult = document.getElementById('phoneResult');

    // Pattern: (area code) exchange-subscriber
    // Groups: 1=area, 2=exchange, 3=subscriber
    const phonePattern = /\((\d{3})\)\s*(\d{3})-(\d{4})/;

    function parsePhone(text) {
      const match = text.match(phonePattern);
      if (match) {
        // match[0] is the full match
        // match[1], [2], [3] are captured groups
        return `Full: ${match[0]}\n` +
               `Area Code: ${match[1]}\n` +
               `Exchange: ${match[2]}\n` +
               `Subscriber: ${match[3]}`;
      }
      return 'No match found';
    }

    phoneInput.addEventListener('input', (e) => {
      phoneResult.textContent = parsePhone(e.target.value);
    });
    phoneResult.textContent = parsePhone(phoneInput.value);

    // === NAMED CAPTURING GROUPS ===
    // (?<name>...) assigns a name to the group
    const dateInput = document.getElementById('date');
    const dateResult = document.getElementById('dateResult');

    // Named groups for year, month, day
    const datePattern = /^(?<year>\d{4})-(?<month>\d{2})-(?<day>\d{2})$/;

    function parseDate(text) {
      const match = text.match(datePattern);
      if (match) {
        // Access via match.groups.name
        const { year, month, day } = match.groups;
        return `Full: ${match[0]}\n` +
               `Year: ${year} (groups.year)\n` +
               `Month: ${month} (groups.month)\n` +
               `Day: ${day} (groups.day)\n` +
               `Also available as: match[1]=${match[1]}`;
      }
      return 'No match - use YYYY-MM-DD format';
    }

    dateInput.addEventListener('input', (e) => {
      dateResult.textContent = parseDate(e.target.value);
    });
    dateResult.textContent = parseDate(dateInput.value);

    // === NON-CAPTURING GROUPS ===
    // (?:...) groups without capturing
    const examples = `
=== NON-CAPTURING GROUPS (?:...) ===

// With capturing: (ab)+ captures "ab" in match[1]
"ababab".match(/(ab)+/)
→ ["ababab", "ab"]

// Non-capturing: (?:ab)+ has no group index
"ababab".match(/(?:ab)+/)
→ ["ababab"]

// Useful with alternation
/(?:red|green|blue) car/.test("green car")
→ true

// Combining both types
const urlPattern = /(?:https?:\/\/)?([^/]+)/;
"https://example.com/path".match(urlPattern)
→ ["https://example.com", "example.com"]
`;

    console.log(examples);
  </script>
</body>
</html>
```

**Key Points:**
- `()` captures; `(?:)` groups without capturing
- Named groups `(?<name>)` improve readability
- Access named groups via `match.groups.name`

## js_regexp_assertions

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>RegExp Assertions</title>
  <style>
    body { font-family: system-ui; padding: 20px; }
    .result { background: #f5f5f5; padding: 10px; margin-top: 10px; font-family: monospace; }
  </style>
</head>
<body>
  <h2>RegExp Lookahead & Lookbehind Assertions</h2>
  <!-- 
    PROBLEM: Match patterns based on context without including context in match
    TIME: O(n*m) potentially, where n=string, m=pattern complexity SPACE: O(1)
  -->
  <!-- 
    WHY: Assertions allow complex conditional matching - "match X only if 
         followed/not followed by Y" without consuming Y in the result.
  -->

  <div id="demo" class="result"></div>

  <script>
    const demo = document.getElementById('demo');
    let output = '';

    // === LOOKAHEAD ASSERTIONS ===
    output += '<strong>LOOKAHEAD (ahead of current position)</strong>\n\n';

    // Positive lookahead: (?=...)
    // Match "foo" only if followed by "bar"
    const text1 = "foobar foobaz foo";
    output += `Text: "${text1}"\n`;
    output += `foo(?=bar) - "foo" followed by "bar":\n`;
    output += `  Matches: ${text1.match(/foo(?=bar)/g)}\n\n`;

    // Negative lookahead: (?!...)
    // Match "foo" only if NOT followed by "bar"
    output += `foo(?!bar) - "foo" NOT followed by "bar":\n`;
    output += `  Matches: ${text1.match(/foo(?!bar)/g)}\n\n`;

    // Practical example: password validation
    output += `<strong>Password Pattern (min 8, must have digit & uppercase):</strong>\n`;
    const passwordPattern = /^(?=.*\d)(?=.*[A-Z]).{8,}$/;
    output += `^(?=.*\\d)(?=.*[A-Z]).{8,}$\n`;
    output += `  "password1": ${passwordPattern.test('password1')}\n`;
    output += `  "Password1": ${passwordPattern.test('Password1')}\n\n`;

    // === LOOKBEHIND ASSERTIONS ===
    output += '<strong>LOOKBEHIND (behind current position)</strong>\n\n';

    // Positive lookbehind: (?<=...)
    const text2 = "Price: $100 and €200";
    output += `Text: "${text2}"\n`;
    output += `(?<=\\$)\\d+ - digits after "$":\n`;
    output += `  Matches: ${text2.match(/(?<=\$)\d+/g)}\n\n`;

    // Negative lookbehind: (?<!...)
    output += `(?<!\\$)\\d+ - digits NOT after "$":\n`;
    output += `  Matches: ${text2.match(/(?<!\$)\d+/g)}\n\n`;

    // Combined example: extract values between specific markers
    const html = "<div>Hello</div><span>World</span>";
    output += `<strong>Extract content between tags:</strong>\n`;
    output += `HTML: "${html}"\n`;
    output += `(?<=<\\w+>).+?(?=</\\w+>)\n`;
    output += `  Matches: ${html.match(/(?<=<\w+>).+?(?=<\/\w+>)/g)}\n\n`;

    // Display all output
    demo.textContent = output;

    // Console reference
    console.log(`
ASSERTION REFERENCE:
====================
Lookahead:
  (?=pattern)   Positive - match if followed by pattern
  (?!pattern)   Negative - match if NOT followed by pattern

Lookbehind:
  (?<=pattern)  Positive - match if preceded by pattern
  (?<!pattern)  Negative - match if NOT preceded by pattern

Note: Lookbehind support requires modern browsers (ES2018+)
    `);
  </script>
</body>
</html>
```

**Key Points:**
- Assertions are zero-width; they don't consume characters
- Lookbehind support is ES2018+; test for older browser compatibility
- Password patterns often use multiple lookaheads for each requirement

## js_date_setDate

```html
<!DOCTYPE html>
<html lang="en">

  <!-- 
    PROBLEM: Set the day of the month for a Date object

  -->
  <!-- 
    WHY: Useful for navigating dates by day - scheduling, calendars, 
         deadlines. Handles month/year overflow automatically.
  -->

  <label>Current date: <span id="currentDate"></span></label><br>
  <label>Set day to: <input type="number" id="dayInput" min="1" max="31" value="15"></label>

    const result = document.getElementById('result');

    // Display current date
    function displayDate() {
      currentDateEl.textContent = date.toDateString();
    }
    displayDate();

    setBtn.addEventListener('click', () => {
      const newDay = parseInt(dayInput.value, 10);
      
      // Store old date for comparison
      const oldDate = date.toDateString();
      const oldDay = date.getDate();

      // setDate() sets the day of the month (1-31)
      // Returns new timestamp in milliseconds
      const newTimestamp = date.setDate(newDay);

      // Build result output
      let output = `Original: ${oldDate} (day ${oldDay})\n`;
      output += `setDate(${newDay}): ${date.toDateString()}\n`;
      output += `Returned timestamp: ${newTimestamp}\n\n`;

      // Demonstrate overflow handling
      output += `<strong>Overflow Handling:</strong>\n`;
      const testDate = new Date(2026, 1, 28); // Feb 28, 2026
      output += `Feb 28, 2026 → setDate(31): ${testDate.toDateString()}\n`;
      // Sets to Feb 31 → automatically rolls to March 3
      // Because Feb 2026 has 28 days, day 31 = Feb 28 + 3 = Mar 3

      const testDate2 = new Date(2026, 0, 31); // Jan 31, 2026
      output += `Jan 31, 2026 → setDate(32): ${testDate2.toDateString()}\n\n`;

      // Negative values count from last day of previous month
      output += `<strong>Negative Values:</strong>\n`;
      const testDate3 = new Date(2026, 2, 15); // Mar 15, 2026
      testDate3.setDate(-1);
      output += `Mar 15 → setDate(-1): ${testDate3.toDateString()}\n`;
      // -1 = last day of Feb = Feb 28

      result.textContent = output;
      displayDate();
    });

    // Console examples
    console.log(`
setDate() BEHAVIOR:
====================
- Accepts 1-31 typically
- Values > days in month overflow to next month
- Values < 1 go to previous month
- Returns new timestamp (mutates original date)

Examples:
  new Date(2026, 0, 31).setDate(32) // Feb 1, 2026
  new Date(2026, 1, 1).setDate(0)   // Jan 31, 2026
  new Date(2026, 1, 1).setDate(-1)  // Jan 30, 2026
    `);
  </script>
</body>
</html>
```

**Key Points:**
- Mutates the Date object in place
- Returns new timestamp value
- Handles overflow automatically (day 32 → next month)

## js_date_setHours

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Date setHours() Method</title>
  <style>
    body { font-family: system-ui; padding: 20px; }
    .result { background: #f5f5f5; padding: 10px; margin-top: 10px; }
    .clock { font-size: 2em; font-family: monospace; }
  </style>
</head>
<body>
  <h2>Date.prototype.setHours()</h2>
  <!-- 
    PROBLEM: Set the hour (and optionally minutes, seconds, milliseconds) for a Date
    TIME: O(1) SPACE: O(1)
  -->
  <!-- 
    WHY: Essential for time manipulation - scheduling, alarms, time zone 
         adjustments. Can set multiple time components in one call.
  -->

  <div class="clock" id="clock"></div>
  <br>
  <button id="morningBtn">Set to 9:00 AM</button>
  <button id="noonBtn">Set to 12:00 PM</button>
  <button id="eveningBtn">Set to 6:00 PM</button>
  <button id="resetBtn">Reset to Now</button>

  <div id="info" class="result"></div>

  <script>
    let date = new Date();

    const clockEl = document.getElementById('clock');
    const infoEl = document.getElementById('info');

    // Format time as HH:MM:SS
    function formatTime(d) {
      return d.toLocaleTimeString('en-US', { hour12: false });
    }

    // Update clock display
    function updateClock() {
      clockEl.textContent = formatTime(date);
    }
    updateClock();

    // Generic function to set time
    function setTime(hours, minutes = 0, seconds = 0, ms = 0) {
      const oldTime = formatTime(date);
      
      // setHours can accept 1-4 arguments:
      // setHours(hours)
      // setHours(hours, minutes)
      // setHours(hours, minutes, seconds)
      // setHours(hours, minutes, seconds, milliseconds)
      const timestamp = date.setHours(hours, minutes, seconds, ms);

      infoEl.innerHTML = `
<strong>setHours(${hours}, ${minutes}, ${seconds}, ${ms})</strong>
Before: ${oldTime}
After: ${formatTime(date)}
Timestamp: ${timestamp}
      `;

      updateClock();
    }

    // Button handlers
    document.getElementById('morningBtn').addEventListener('click', () => {
      setTime(9, 0, 0);
    });

    document.getElementById('noonBtn').addEventListener('click', () => {
      setTime(12, 0, 0);
    });

    document.getElementById('eveningBtn').addEventListener('click', () => {
      setTime(18, 0, 0);
    });

    document.getElementById('resetBtn').addEventListener('click', () => {
      date = new Date();
      updateClock();
      infoEl.textContent = 'Reset to current time';
    });

    // Console reference
    console.log(`
setHours() SYNTAX:
===================
date.setHours(hours)
date.setHours(hours, minutes)
date.setHours(hours, minutes, seconds)
date.setHours(hours, minutes, seconds, milliseconds)

- hours: 0-23 (values > 23 increment day)
- minutes: 0-59 (values > 59 increment hour)
- seconds: 0-59
- milliseconds: 0-999

Overflow Example:
  new Date(2026, 0, 1, 22).setHours(26)  // Jan 2, 2:00 AM
  new Date(2026, 0, 1, 12).setHours(-1)  // Dec 31, 11:00 PM
    `);
  </script>
```

## js_math_sin

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Math.sin() - Sine Wave Animation</title>
  <style>
    body { font-family: system-ui; padding: 20px; }
    canvas { border: 1px solid #ccc; display: block; margin-top: 10px; }
    .controls { margin: 10px 0; }
  </style>
</head>
<body>
  <h2>Math.sin() - Sine Wave Visualization</h2>
  <!-- 
    PROBLEM: Calculate the sine of an angle (in radians)
    TIME: O(1) SPACE: O(1)
  -->
  <!-- 
    WHY: Fundamental for wave calculations, oscillations, animations,
         circular motion, and trigonometry in graphics/canvas.
  -->

  <div class="controls">
    <label>Frequency: <input type="range" id="freq" min="0.5" max="5" step="0.1" value="1"></label>
    <label>Amplitude: <input type="range" id="amp" min="10" max="100" step="5" value="50"></label>
    <button id="toggleBtn">Pause</button>
  </div>

  <canvas id="canvas" width="600" height="200"></canvas>

  <div id="info" style="margin-top: 10px; font-family: monospace;"></div>

  <script>
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');
    const freqSlider = document.getElementById('freq');
    const ampSlider = document.getElementById('amp');
    const toggleBtn = document.getElementById('toggleBtn');
    const infoEl = document.getElementById('info');

    let phase = 0;
    let running = true;
    let animationId;

    // Convert degrees to radians
    const toRadians = (deg) => deg * Math.PI / 180;

    // Convert radians to degrees
    const toDegrees = (rad) => rad * 180 / Math.PI;

    function draw() {
      const width = canvas.width;
      const height = canvas.height;
      const centerY = height / 2;
      const frequency = parseFloat(freqSlider.value);
      const amplitude = parseFloat(ampSlider.value);

      // Clear canvas
      ctx.clearRect(0, 0, width, height);

      // Draw center line
      ctx.strokeStyle = '#eee';
      ctx.beginPath();
      ctx.moveTo(0, centerY);
      ctx.lineTo(width, centerY);
      ctx.stroke();

      // Draw sine wave
      ctx.strokeStyle = '#3498db';
      ctx.lineWidth = 2;
      ctx.beginPath();

      for (let x = 0; x < width; x++) {
        // Calculate angle for this x position
        // angle = (x / width) * 2π * frequency + phase
        const angle = (x / width) * Math.PI * 2 * frequency + phase;
        
        // sin() returns -1 to 1, multiply by amplitude for visible range
        const y = centerY + Math.sin(angle) * amplitude;

        if (x === 0) {
          ctx.moveTo(x, y);
        } else {
          ctx.lineTo(x, y);
        }
      }
      ctx.stroke();

      // Draw moving point
      const pointAngle = phase;
      const pointY = centerY + Math.sin(pointAngle) * amplitude;
      ctx.fillStyle = '#e74c3c';
      ctx.beginPath();
      ctx.arc(10, pointY, 8, 0, Math.PI * 2);
      ctx.fill();

      // Update info
      infoEl.innerHTML = `
Phase: ${(phase % (Math.PI * 2)).toFixed(2)} rad (${toDegrees(phase % (Math.PI * 2)).toFixed(0)}°)
sin(${(phase % (Math.PI * 2)).toFixed(2)}) = ${Math.sin(phase).toFixed(4)}
      `;

      // Increment phase for animation
      phase += 0.05;

      if (running) {
        animationId = requestAnimationFrame(draw);
      }
    }

    toggleBtn.addEventListener('click', () => {
      running = !running;
      toggleBtn.textContent = running ? 'Pause' : 'Play';
      if (running) draw();
    });

    // Start animation
    draw();

    // Console reference
    console.log(`
Math.sin() REFERENCE:
======================
- Accepts angle in RADIANS (not degrees!)
- Returns value between -1 and 1

Key angles:
  Math.sin(0)           = 0
  Math.sin(π/6)         = 0.5
  Math.sin(π/2)         = 1
  Math.sin(π)           ≈ 0 (floating point)
  Math.sin(3π/2)        = -1
  Math.sin(2π)          ≈ 0

Converting degrees to radians:
  radians = degrees * (Math.PI / 180)
  Math.sin(90 * Math.PI / 180) = 1

Creating a wave:
  y = amplitude * Math.sin(frequency * x + phase)
    `);
  </script>
</body>
</html>
```

**Key Points:**
- Input is in radians, not degrees
- Returns value between -1 and 1
- Use `angle * Math.PI / 180` to convert degrees to radians
