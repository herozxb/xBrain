## js_array_lastIndexOf

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>JavaScript Array lastIndexOf()</title>
</head>
<body>
  <h1>JavaScript Array lastIndexOf()</h1>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Find the last occurrence of an element in array
    // TIME: O(n) SPACE: O(1)

    const log = (...args) => {
      document.getElementById('output').textContent += args.join('\n') + '\n\n';
    };

    // Basic usage
    const fruits = ['apple', 'banana', 'orange', 'banana', 'grape', 'banana'];
    
    log('Array:', JSON.stringify(fruits),
        "lastIndexOf('banana'):", fruits.lastIndexOf('banana'),  // 5
        "lastIndexOf('apple'):", fruits.lastIndexOf('apple'),    // 0
        "lastIndexOf('kiwi'):", fruits.lastIndexOf('kiwi'));     // -1

    // With fromIndex parameter - search backwards from position
    log('Searching from index 4:',
        "lastIndexOf('banana', 4):", fruits.lastIndexOf('banana', 4),  // 3
        "lastIndexOf('banana', 3):", fruits.lastIndexOf('banana', 3)); // 3

    // Practical example: Find last occurrence of a number
    const numbers = [1, 2, 3, 4, 3, 2, 1, 5, 3, 2];
    
    function findLastDuplicate(arr, value) {
      const lastIndex = arr.lastIndexOf(value);
      const firstIndex = arr.indexOf(value);
      
      if (lastIndex !== firstIndex) {
        return {
          value,
          firstIndex,
          lastIndex,
          isDuplicate: true
        };
      }
      return { value, isDuplicate: false };
    }

    log('Finding duplicates:',
        JSON.stringify(findLastDuplicate(numbers, 3), null, 2));

    // Practical: Remove last occurrence
    function removeLastOccurrence(arr, value) {
      const index = arr.lastIndexOf(value);
      if (index > -1) {
        // Create new array without the last occurrence
        return [...arr.slice(0, index), ...arr.slice(index + 1)];
      }
      return arr;
    }

    log('Remove last banana:',
        'Before:', JSON.stringify(fruits),
        'After:', JSON.stringify(removeLastOccurrence(fruits, 'banana')));

    // Count occurrences using both indexOf and lastIndexOf
    function countOccurrences(arr, value) {
      const first = arr.indexOf(value);
      const last = arr.lastIndexOf(value);
      if (first === -1) return 0;
      if (first === last) return 1;
      
      // For accurate count, use filter
      return arr.filter(x => x === value).length;
    }

    log('Count of bananas:', countOccurrences(fruits, 'banana'));

    // Using lastIndexOf to check if element appears more than once
    function hasDuplicates(arr, value) {
      return arr.indexOf(value) !== arr.lastIndexOf(value);
    }

    log('Has duplicate bananas?', hasDuplicates(fruits, 'banana'),
        'Has duplicate kiwi?', hasDuplicates(fruits, 'kiwi'));

    // Edge cases
    log('Edge cases:',
        'Empty array [].lastIndexOf(1):', [].lastIndexOf(1),  // -1
        'Negative fromIndex:',
        '[1,2,3].lastIndexOf(3, -1):', [1,2,3].lastIndexOf(3, -1),  // 2
        '[1,2,3].lastIndexOf(1, -2):', [1,2,3].lastIndexOf(1, -2)); // 0
  </script>
</body>
</html>
```
**Why**: Finds the last position of an element; useful for removing last occurrences, detecting duplicates, and searching arrays from the end.

## js_object_freeze

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>JavaScript Object.freeze()</title>
</head>
<body>
  <h1>JavaScript Object.freeze()</h1>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Make an object immutable (cannot be changed)
    // TIME: O(1) SPACE: O(1)

    const log = (...args) => {
      document.getElementById('output').textContent += args.join('\n') + '\n\n';
    };

    // Basic freeze
    const config = {
      apiUrl: 'https://api.example.com',
      timeout: 5000,
      retries: 3
    };

    Object.freeze(config);

    log('=== Basic Freeze ===',
        'Original config:', JSON.stringify(config));

    // Attempt to modify (fails silently in non-strict mode)
    config.apiUrl = 'https://hacked.com';  // Won't work
    config.timeout = 0;                     // Won't work

    log('After modification attempt:', JSON.stringify(config));

    // Attempt to add property
    config.newProp = 'test';  // Won't work
    log('After adding property:', JSON.stringify(config));

    // Attempt to delete
    delete config.retries;  // Won't work
    log('After delete attempt:', JSON.stringify(config));

    // Check if frozen
    log('Is config frozen?', Object.isFrozen(config));

    // Shallow freeze limitation
    const user = {
      name: 'John',
      preferences: {
        theme: 'dark',
        language: 'en'
      }
    };

    Object.freeze(user);

    // This won't work
    user.name = 'Jane';
    log('user.name after change:', user.name);  // Still 'John'

    // But nested object CAN be modified (shallow freeze)
    user.preferences.theme = 'light';
    log('user.preferences.theme:', user.preferences.theme);  // 'light'!

    // Deep freeze function
    function deepFreeze(obj) {
      // Get all properties
      const propNames = Object.getOwnPropertyNames(obj);
      
      // Freeze nested objects first
      for (const name of propNames) {
        const value = obj[name];
        if (value && typeof value === 'object') {
          deepFreeze(value);
        }
      }
      
      // Freeze the object itself
      return Object.freeze(obj);
    }

    // Using deep freeze
    const deepConfig = {
      api: { url: 'https://api.example.com' },
      features: { darkMode: true, notifications: true }
    };

    deepFreeze(deepConfig);

    // Now nested objects are also frozen
    deepConfig.api.url = 'hacked';  // Won't work in strict mode
    log('Deep frozen api.url:', deepConfig.api.url);

    // Practical use: Constants
    const COLORS = Object.freeze({
      PRIMARY: '#3498db',
      SECONDARY: '#2ecc71',
      DANGER: '#e74c3c',
      WARNING: '#f39c12'
    });

    log('=== Constants ===', JSON.stringify(COLORS, null, 2));

    // Use with strict mode for errors
    log('Tip: Use "use strict" to get TypeError when trying to modify frozen objects');

    // Object.freeze vs const
    log('=== freeze vs const ===',
        'const: prevents reassignment of variable',
        'freeze: prevents modification of object properties',
        'Use both for true immutability');
  </script>
</body>
</html>
```
**Why**: Creates immutable objects for configuration, constants, and security; prevents accidental or malicious modification of critical data.

## js_string_concat

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>JavaScript String concat()</title>
</head>
<body>
  <h1>JavaScript String concat()</h1>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Join multiple strings together
    // TIME: O(n) where n = total length SPACE: O(n)

    const log = (...args) => {
      document.getElementById('output').textContent += args.join('\n') + '\n\n';
    };

    // Basic concat
    const str1 = 'Hello';
    const str2 = ' ';
    const str3 = 'World';

    log('=== Basic concat ===',
        'str1:', str1,
        'str2:', str2,
        'str3:', str3,
        'concat result:', str1.concat(str2, str3));

    // Concat with multiple arguments
    const greeting = 'Hello'.concat(', ', 'JavaScript', '!');
    log('Multiple arguments:', greeting);

    // Concat non-strings (auto-converted)
    const mixed = 'Value: '.concat(42, ' ', true, ' ', null);
    log('Mixed types:', mixed);

    // Building URLs
    function buildUrl(base, ...segments) {
      let url = base;
      for (const segment of segments) {
        url = url.concat('/', segment);
      }
      return url;
    }

    log('=== Practical: Build URL ===',
        buildUrl('https://api.example.com', 'users', '123', 'profile'));

    // Chained concat
    const chained = 'a'.concat('b').concat('c').concat('d');
    log('Chained concat:', chained);

    // Compare with + operator
    const withPlus = 'Hello' + ' ' + 'World' + '!';
    const withConcat = 'Hello'.concat(' ', 'World', '!');
    log('=== + vs concat ===',
        'Using +:', withPlus,
        'Using concat:', withConcat,
        'Result is same:', withPlus === withConcat);

    // Building HTML strings
    function createLink(text, url) {
      return '<a href="'.concat(url, '">', text, '</a>');
    }

    log('=== HTML generation ===',
        createLink('Click here', 'https://example.com'));

    // Performance note
    log('=== Performance Tips ===',
        'For many concatenations, use array.join() instead',
        'For simple cases, + operator is fine',
        'Template literals are often more readable');

    // Array join vs concat (for many strings)
    const parts = ['The', 'quick', 'brown', 'fox', 'jumps'];
    const sentence = parts.join(' ');
    log('Using array.join:', sentence);

    // concat() creates new string (original unchanged)
    const original = 'Hello';
    const newString = original.concat(' World');
    log('=== Immutability ===',
        'Original after concat:', original,
        'New string:', newString);

    // Empty string concat
    log('Empty concat:', ''.concat('a', 'b', 'c'));
  </script>
```

```html
    <button onclick="removeSingle()">Remove 'username'</button>
    <button onclick="removeMultiple()">Remove Multiple</button>

  <script>
    // PROBLEM: Remove specific item from localStorage
    // TIME: O(1) SPACE: O(1)

    const log = (...args) => {
      document.getElementById('output').textContent += args.join('\n') + '\n\n';
    };

    const clearLog = () => {
      document.getElementById('output').textContent = '';
    };

    // Set demo items
    function setItems() {
      clearLog();
      localStorage.setItem('username', 'john_doe');
      localStorage.setItem('theme', 'dark');
      localStorage.setItem('language', 'en');
      localStorage.setItem('notifications', 'true');
      log('Demo items set!',
          'username: john_doe',
          'theme: dark',
          'language: en',
          'notifications: true');
    }

    // Remove single item
    function removeSingle() {
      clearLog();
      log('Before removal:',
          'username:', localStorage.getItem('username'));
      
      localStorage.removeItem('username');
      
      log('After localStorage.removeItem("username"):',
          'username:', localStorage.getItem('username'),  // null
          'Item is completely removed from storage');
    }

    // Remove multiple items
    function removeMultiple() {
      clearLog();
      const itemsToRemove = ['theme', 'language'];
      
      log('Removing:', itemsToRemove.join(', '));
      
      itemsToRemove.forEach(key => {
        localStorage.removeItem(key);
      });
      
      log('Removed!',
          'theme:', localStorage.getItem('theme'),
          'language:', localStorage.getItem('language'));
    }

    // Show all localStorage
    function showAll() {
      clearLog();
      log('=== All localStorage Items ===',
          'Length:', localStorage.length);
      
      for (let i = 0; i < localStorage.length; i++) {
        const key = localStorage.key(i);
        log(key + ':', localStorage.getItem(key));
      }
    }

    // removeItem vs setting to null/undefined
    log('=== removeItem vs null/undefined ===');
    
    localStorage.setItem('test1', 'value1');
    localStorage.setItem('test2', 'value2');
    
    // Setting to null (still exists in storage)
    localStorage.setItem('test1', null);
    log("After setItem('test1', null):",
        'test1:', localStorage.getItem('test1'),  // "null" as string!
        'Still in storage');
    
    // Using removeItem (completely removed)
    localStorage.removeItem('test2');
    log("After removeItem('test2'):",
        'test2:', localStorage.getItem('test2'),  // null (actual null)
        'Removed from storage');

    // Utility functions
    function removeIfExists(key) {
      if (localStorage.getItem(key) !== null) {
        localStorage.removeItem(key);
        return true;
      }
      return false;
    }

    function removeByPrefix(prefix) {
      const keysToRemove = [];
      for (let i = 0; i < localStorage.length; i++) {
        const key = localStorage.key(i);
        if (key.startsWith(prefix)) {
          keysToRemove.push(key);
        }
      }
      keysToRemove.forEach(key => localStorage.removeItem(key));
      return keysToRemove.length;
    }

    // Practical: Clear user session
    function clearUserSession() {
      const sessionKeys = ['token', 'userId', 'username', 'preferences'];
      sessionKeys.forEach(key => localStorage.removeItem(key));
      log('User session cleared!');
    }

        '1. removeItem(key) - removes item completely',
        '2. getItem returns null for missing keys',
        '3. Setting to null creates string "null"',
        '4. Use clear() to remove everything',
        '5. Data persists until explicitly removed');
```
