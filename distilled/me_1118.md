## [Farthest Pair of Points]

```python
# PROBLEM: Find the two points with maximum Euclidean distance in a 2D point set
# TIME: O(n log n) SPACE: O(n)
def farthest_pair_points(points):
    """
    Find the pair of points with maximum distance using rotating calipers
    on the convex hull. The farthest pair must be on the convex hull.
    
    Args:
        points: List of (x, y) coordinate tuples
    
    Returns:
        Tuple of two points and their distance
    """
    if len(points) < 2:
        return None, None, 0
    
    # Build convex hull using monotone chain
    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
    
    points = sorted(set(points))
    if len(points) <= 1:
        return points[0] if points else None, None, 0
    
    # Build lower hull
    lower = []
    for p in points:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    
    # Build upper hull
    upper = []
    for p in reversed(points):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    
    hull = lower[:-1] + upper[:-1]
    
    # Rotating calipers to find farthest pair
    def distance(p1, p2):
        return ((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)**0.5
    
    n = len(hull)
    if n == 2:
        return hull[0], hull[1], distance(hull[0], hull[1])
    
    max_dist = 0
    best_pair = (hull[0], hull[1])
    
    # Find the point farthest from edge (hull[i], hull[i+1])
    j = 1
    for i in range(n):
        # Advance j while next point is farther from current edge
        while True:
            next_j = (j + 1) % n
            # Compare distances using cross product (avoids sqrt)
            curr_dist = abs(cross(hull[i], hull[(i+1)%n], hull[j]))
            next_dist = abs(cross(hull[i], hull[(i+1)%n], hull[next_j]))
            if next_dist > curr_dist:
                j = next_j
            else:
                break
        
        d = distance(hull[i], hull[j])
        if d > max_dist:
            max_dist = d
            best_pair = (hull[i], hull[j])
    
    return best_pair[0], best_pair[1], max_dist
```
**Why**: Fundamental computational geometry problem; rotating calipers technique is elegant and optimal for diameter computation on convex polygons.

## [Graph Isomorphism - VF2 Algorithm]

```python
# PROBLEM: Determine if two graphs are structurally identical (isomorphic)
# TIME: O(n! * n) worst case SPACE: O(n)
def graph_isomorphism_vf2(adj1, adj2):
    """
    Check if two graphs are isomorphic using VF2 algorithm.
    Uses backtracking with pruning based on degree and connectivity.
    
    Args:
        adj1: Adjacency list/dict for graph 1
        adj2: Adjacency list/dict for graph 2
    
    Returns:
        Mapping dict from graph1 nodes to graph2 nodes if isomorphic, else None
    """
    n1, n2 = len(adj1), len(adj2)
    if n1 != n2:
        return None
    
    # Quick check: degree sequences must match
    deg1 = sorted(len(adj1[v]) for v in adj1)
    deg2 = sorted(len(adj2[v]) for v in adj2)
    if deg1 != deg2:
        return None
    
    nodes1, nodes2 = list(adj1.keys()), list(adj2.keys())
    mapping = {}  # node1 -> node2
    reverse_mapping = {}  # node2 -> node1
    
    def is_consistent(v1, v2):
        """Check if mapping v1->v2 is consistent with partial mapping"""
        for u1, u2 in mapping.items():
            # Check adjacency preservation
            edge1 = v2 in adj2.get(u2, []) or u2 in adj2.get(v2, [])
            edge2 = u1 in adj1.get(v1, []) or v1 in adj1.get(u1, [])
            if edge1 != edge2:
                return False
        return True
    
    def get_candidates():
        """Get candidate pairs for next mapping"""
        mapped1, mapped2 = set(mapping), set(reverse_mapping)
        
        # Find frontier nodes (adjacent to mapped nodes)
        frontier1 = set()
        for v in mapping:
            frontier1.update(adj1.get(v, []))
        frontier1 -= mapped1
        
        frontier2 = set()
        for v in reverse_mapping:
            frontier2.update(adj2.get(v, []))
        frontier2 -= mapped2
        
        if frontier1 and frontier2:
            candidates1, candidates2 = frontier1, frontier2
        else:
            # Use unmapped nodes
            candidates1 = set(nodes1) - mapped1
            candidates2 = set(nodes2) - mapped2
        
        return candidates1, candidates2
    
    def backtrack():
        if len(mapping) == n1:
            return dict(mapping)
        
        candidates1, candidates2 = get_candidates()
        
        # Pruning: check size feasibility
        if len(candidates1) != len(candidates2):
            unmapped1 = set(nodes1) - set(mapping)
            unmapped2 = set(nodes2) - set(reverse_mapping)
            if len(unmapped1) != len(unmapped2):
                return None
        
        # Try first candidate from candidates1
        if not candidates1:
            return None
        v1 = min(candidates1)  # Deterministic choice
        
        for v2 in candidates2:
            if is_consistent(v1, v2):
                mapping[v1] = v2
                reverse_mapping[v2] = v1
                
                result = backtrack()
                if result is not None:
                    return result
                
                del mapping[v1]
                del reverse_mapping[v2]
        
        return None
    
    return backtrack()
```
**Why**: Classic problem with no known polynomial solution; VF2 is practical for medium-sized graphs with effective pruning.

## [Minimum Dominating Set - Greedy Approximation]

```python
# PROBLEM: Find minimum set of vertices where every vertex is either in the set or adjacent to it
# TIME: O(n^2) SPACE: O(n)
def minimum_dominating_set(adj):
    """
    Find a dominating set using greedy approximation.
    Each step picks the vertex that dominates the most undominated vertices.
    
    Args:
        adj: Adjacency list/dict representing the graph
    
    Returns:
        Set of vertices forming a dominating set
    """
    if not adj:
        return set()
    
    nodes = set(adj.keys())
    dominated = set()  # Vertices that are dominated
    dominating_set = set()
    
    while dominated != nodes:
        # Find vertex that dominates the most new vertices
        best_vertex = None
        best_gain = -1
        
        for v in nodes:
            # Vertices that v would dominate (itself + neighbors)
            would_dominate = {v} | set(adj.get(v, []))
            # New vertices dominated by choosing v
            new_dominated = would_dominate - dominated
            gain = len(new_dominated)
            
            if gain > best_gain:
                best_gain = gain
                best_vertex = v
        
        if best_vertex is None or best_gain == 0:
            break
        
        dominating_set.add(best_vertex)
        # Mark the vertex and its neighbors as dominated
        dominated.add(best_vertex)
        dominated.update(adj.get(best_vertex, []))
    
    return dominating_set


def is_dominating_set(adj, dom_set):
    """Verify if a set is a valid dominating set"""
    nodes = set(adj.keys())
    dominated = set()
    for v in dom_set:
        dominated.add(v)
        dominated.update(adj.get(v, []))
    return dominated == nodes
```
**Why**: NP-hard problem with greedy approximation achieving ln(n) approximation ratio; fundamental in network design and facility location.

## [DSatur Graph Coloring]

```python
# PROBLEM: Color graph vertices so adjacent vertices have different colors, minimizing colors
# TIME: O(n^2) SPACE: O(n)
def dsatur_graph_coloring(adj):
    """
    Color graph using DSatur (Degree of Saturation) heuristic.
    Always color the vertex with highest saturation (most different colored neighbors).
    
    Args:
        adj: Adjacency list/dict representing the graph
    
    Returns:
        Dict mapping vertices to color numbers (0-indexed)
    """
    if not adj:
        return {}
    
    nodes = list(adj.keys())
    n = len(nodes)
    
    # Track: saturation (set of colors used by neighbors), degree
    saturation = {v: set() for v in nodes}
    degree = {v: len(adj.get(v, [])) for v in nodes}
    color = {v: -1 for v in nodes}
    
    uncolored = set(nodes)
    
    def get_saturation_degree(v):
        """Number of different colors used by neighbors"""
        return len(saturation[v])
    
    def select_next_vertex():
        """Select uncolored vertex with max saturation, break ties by degree"""
        candidates = list(uncolored)
        # Sort by saturation (desc), then degree (desc), then node name for determinism
        candidates.sort(key=lambda v: (-get_saturation_degree(v), -degree[v], v))
        return candidates[0] if candidates else None
    
    def get_available_color(v):
        """Find smallest color not used by neighbors"""
        used = saturation[v]
        c = 0
        while c in used:
            c += 1
        return c
    
    while uncolored:
        v = select_next_vertex()
        c = get_available_color(v)
        color[v] = c
        
        # Update saturation of uncolored neighbors
        for neighbor in adj.get(v, []):
            if neighbor in uncolored:
                saturation[neighbor].add(c)
        
        uncolored.remove(v)
    
    return color


def verify_coloring(adj, color):
    """Verify that no adjacent vertices have the same color"""
    for v in adj:
        for neighbor in adj[v]:
            if color.get(v) == color.get(neighbor):
                return False
    return True
```
**Why**: DSatur often produces optimal or near-optimal colorings; particularly effective for graphs with varying degree distributions.

## [Threaded Binary Tree]

```python
# PROBLEM: Implement binary tree with threads for efficient inorder traversal without stack/ recursion
# TIME: O(n) build, O(1) amortized traversal SPACE: O(1) for traversal
class ThreadedNode:
    """Node for threaded binary tree"""
    def __init__(self, val):
        self.val = val
        self.left = None
        self.right = None
        self.left_thread = False   # True if left points to inorder predecessor
        self.right_thread = False  # True if right points to inorder successor


class ThreadedBinaryTree:
    """
    Right-threaded binary tree for efficient forward traversal.
    Right thread points to inorder successor when right child is None.
    """
    
    def __init__(self):
        self.root = None
        self.dummy = ThreadedNode(None)  # Sentinel node
        self.dummy.left = self.dummy
        self.dummy.right = self.dummy
        self.dummy.left_thread = True
        self.dummy.right_thread = True
    
    def insert(self, val):
        """Insert value maintaining BST property and threads"""
        new_node = ThreadedNode(val)
        
        if self.root is None:
            self.root = new_node
            self.dummy.left = self.root
            self.root.left = self.dummy
            self.root.right = self.dummy
            self.root.left_thread = True
            self.root.right_thread = True
            return
        
        current = self.root
        while True:
            if val < current.val:
                if not current.left_thread and current.left:
                    current = current.left
                else:
                    # Insert as left child
                    new_node.left = current.left
                    new_node.left_thread = True
                    new_node.right = current
                    new_node.right_thread = True
                    current.left = new_node
                    current.left_thread = False
                    break
            else:
                if not current.right_thread and current.right:
                    current = current.right
                else:
                    # Insert as right child
                    new_node.right = current.right
                    new_node.right_thread = True
                    new_node.left = current
                    new_node.left_thread = True
                    current.right = new_node
                    current.right_thread = False
                    break
    
    def inorder_first(self):
        """Find leftmost node (first in inorder)"""
        if self.root is None:
            return None
        current = self.root
        while not current.left_thread:
            current = current.left
        return current
    
    def inorder_successor(self, node):
        """Find inorder successor using threads"""
        if node.right_thread:
            return node.right if node.right != self.dummy else None
        
        # Right child exists, go to leftmost node in right subtree
        current = node.right
        while not current.left_thread:
            current = current.left
        return current
    
    def inorder_traversal(self):
        """Iterate in order using threads - no stack needed"""
        result = []
        current = self.inorder_first()
        while current is not None:
            result.append(current.val)
            current = self.inorder_successor(current)
        return result
```
**Why**: Enables O(1) space inorder traversal; useful in memory-constrained environments and for bidirectional tree navigation.

## [Disjoint Sparse Table]

```python
# PROBLEM: Answer range queries (min, max, gcd, etc.) in O(1) with O(n log n) preprocessing
# TIME: O(n log n) build, O(1) query SPACE: O(n log n)
class DisjointSparseTable:
    """
    Disjoint sparse table for idempotent operations (min, max, gcd, AND, OR).
    Unlike regular sparse table, works for any associative operation.
    Each query is answered by exactly one precomputed segment.
    """
    
    def __init__(self, arr, func=min):
        """
        Build disjoint sparse table.
        
        Args:
            arr: Input array
            func: Associative function (min, max, gcd, etc.)
        """
        self.n = len(arr)
        self.func = func
        self.arr = arr[:]
        self.table = []
        self._build()
    
    def _build(self):
        """Build the disjoint sparse table in O(n log n)"""
        if self.n == 0:
            return
        
        self.table = [self.arr[:]]
        
        level = 1
        size = 2
        
        while size < self.n:
            row = [float('inf')] * self.n if self.func == min else [float('-inf')] * self.n
            
            # Process each segment of length 2*size
            for seg_start in range(0, self.n, size * 2):
                mid = seg_start + size
                
                if mid < self.n:
                    # Forward pass: mid to segment end
                    row[mid] = self.arr[mid]
                    for i in range(mid + 1, min(mid + size, self.n)):
                        row[i] = self.func(row[i-1], self.arr[i])
                    
                    # Backward pass: mid-1 to segment start
                    if mid - 1 >= seg_start:
                        row[mid - 1] = self.arr[mid - 1]
                        for i in range(mid - 2, seg_start - 1, -1):
                            row[i] = self.func(row[i + 1], self.arr[i])
            
            self.table.append(row)
            size *= 2
            level += 1
    
    def query(self, l, r):
        """
        Answer query on range [l, r] in O(1).
        
        Args:
            l: Left bound (inclusive)
            r: Right bound (inclusive)
        
        Returns:
            Result of func on range [l, r]
        """
        if l == r:
            return self.arr[l]
        
        # Find level where l and r are in different halves of a segment
        length = r - l + 1
        level = length.bit_length() - 1
        
        return self.func(self.table[level][l], self.table[level][r])
```
**Why**: O(1) queries for any associative operation; simpler than segment tree for static arrays with idempotent operations.

## [Implicit Treap]

```python
# PROBLEM: Maintain sequence with split, merge, insert, delete, and range queries in O(log n)
# TIME: O(log n) per operation SPACE: O(n)
import random

class ImplicitTreapNode:
    """Node for implicit treap (cartesian tree by index)"""
    def __init__(self, val):
        self.val = val
        self.priority = random.random()
        self.size = 1
        self.left = None
        self.right = None
        self.sum = val  # Example aggregate: sum of subtree
        self.lazy = 0   # Lazy propagation value


class ImplicitTreap:
    """
    Implicit treap: BST by implicit index (position), heap by priority.
    Supports split/merge, insert at position, delete range, range queries.
    """
    
    def __init__(self):
        self.root = None
    
    def _update(self, node):
        """Update size and aggregate values"""
        if node:
            node.size = 1
            node.sum = node.val
            if node.left:
                node.size += node.left.size
                node.sum += node.left.sum
            if node.right:
                node.size += node.right.size
                node.sum += node.right.sum
    
    def _push(self, node):
        """Push lazy propagation to children"""
        if node and node.lazy:
            if node.left:
                node.left.lazy += node.lazy
            if node.right:
                node.right.lazy += node.lazy
            node.lazy = 0
    
    def _size(self, node):
        return node.size if node else 0
    
    def split(self, node, k):
        """
        Split tree into two: first k elements, rest.
        Returns (left_tree, right_tree)
        """
        if not node:
            return None, None
        
        self._push(node)
        
        left_size = self._size(node.left)
        
        if k <= left_size:
            # Split in left subtree
            left, right = self.split(node.left, k)
            node.left = right
            self._update(node)
            return left, node
        else:
            # Split in right subtree
            left, right = self.split(node.right, k - left_size - 1)
            node.right = left
            self._update(node)
            return node, right
    
    def merge(self, left, right):
        """Merge two treaps where all indices in left < all in right"""
        if not left or not right:
            return left or right
        
        self._push(left)
        self._push(right)
        
        if left.priority > right.priority:
            left.right = self.merge(left.right, right)
            self._update(left)
            return left
        else:
            right.left = self.merge(left, right.left)
            self._update(right)
            return right
    
    def insert(self, pos, val):
        """Insert value at position pos (0-indexed)"""
        new_node = ImplicitTreapNode(val)
        left, right = self.split(self.root, pos)
        self.root = self.merge(self.merge(left, new_node), right)
    
    def erase(self, pos):
        """Remove element at position pos"""
        left, mid_right = self.split(self.root, pos)
        mid, right = self.split(mid_right, 1)
        self.root = self.merge(left, right)
        return mid.val if mid else None
    
    def range_query(self, l, r):
        """Query sum on range [l, r]"""
        left, mid_right = self.split(self.root, l)
        mid, right = self.split(mid_right, r - l + 1)
        result = mid.sum if mid else 0
        self.root = self.merge(left, self.merge(mid, right))
        return result
    
    def to_list(self, node=None):
        """Convert treap to list (for debugging)"""
        if node is None:
            node = self.root
        if not node:
            return []
        self._push(node)
        return self.to_list(node.left) + [node.val] + self.to_list(node.right)
```
**Why**: Combines BST flexibility with heap balance; supports dynamic sequences with splits/merges efficiently.

## [Mo's Algorithm on Trees]

```python
# PROBLEM: Answer path queries on tree offline in O((n+q) * sqrt(n))
# TIME: O((n+q) * sqrt(n)) SPACE: O(n)
from collections import defaultdict
import math

class MoTree:
    """
    Mo's algorithm for tree path queries.
    Flattens tree using Euler tour, then applies Mo's ordering.
    Supports queries like: count distinct values on path, sum on path, etc.
    """
    
    def __init__(self, n, adj, values):
        """
        Initialize Mo's algorithm on tree.
        
        Args:
            n: Number of nodes
            adj: Adjacency list
            values: Value at each node (1-indexed)
        """
        self.n = n
        self.adj = adj
        self.values = values
        self.flatten()
    
    def flatten(self):
        """Flatten tree using Euler tour (in-out times)"""
        self.in_time = [0] * (self.n + 1)
        self.out_time = [0] * (self.n + 1)
        self.flat = [0] * (2 * self.n + 1)  # Flattened array
        self.depth = [0] * (self.n + 1)
        self.parent = [0] * (self.n + 1)
        
        timer = 1
        stack = [(1, 0, True)]  # (node, parent, is_enter)
        
        while stack:
            node, par, is_enter = stack.pop()
            if is_enter:
                self.parent[node] = par
                self.in_time[node] = timer
                self.flat[timer] = node
                timer += 1
                # Push exit event first, then children (for correct order)
                stack.append((node, par, False))
                for child in self.adj.get(node, []):
                    if child != par:
                        self.depth[child] = self.depth[node] + 1
                        stack.append((child, node, True))
            else:
                self.out_time[node] = timer
                self.flat[timer] = node
                timer += 1
    
    def is_ancestor(self, u, v):
        """Check if u is ancestor of v"""
        return self.in_time[u] <= self.in_time[v] <= self.out_time[u]
    
    def get_lca(self, u, v):
        """Simple LCA using parent pointers (can be optimized with binary lifting)"""
        if self.is_ancestor(u, v):
            return u
        if self.is_ancestor(v, u):
            return v
        while not self.is_ancestor(self.parent[u], v):
            u = self.parent[u]
        return self.parent[u]
    
    def query_range(self, l, r):
        """Map query(u,v) to range in flattened array"""
        lca = self.get_lca(l, r)
        if lca == l:
            return self.in_time[l], self.in_time[r]
        return self.out_time[l], self.in_time[r]
    
    def process_queries(self, queries, add_func, remove_func, get_answer):
        """
        Process queries offline using Mo's ordering.
        
        Args:
            queries: List of (l, r, query_id)
            add_func: Called when adding a node
            remove_func: Called when removing a node  
            get_answer: Returns current answer
        """
        block_size = int(math.sqrt(2 * self.n)) + 1
        
        # Sort queries by Mo's order
        def mo_cmp(q):
            l, r, idx = q
            block = l // block_size
            return (block, r if block % 2 == 0 else -r)
        
        sorted_queries = sorted(queries, key=mo_cmp)
        
        answers = [0] * len(queries)
        visited = [False] * (self.n + 1)
        cur_l, cur_r = 1, 0
        
        def toggle(node):
            if visited[node]:
                remove_func(node)
            else:
                add_func(node)
            visited[node] = not visited[node]
        
        for l, r, idx in sorted_queries:
            # Expand/contract current range
            while cur_l > l:
                cur_l -= 1
                toggle(self.flat[cur_l])
            while cur_r < r:
                cur_r += 1
                toggle(self.flat[cur_r])
            while cur_l < l:
                toggle(self.flat[cur_l])
                cur_l += 1
            while cur_r > r:
                toggle(self.flat[cur_r])
                cur_r -= 1
            
            answers[idx] = get_answer()
        
        return answers
```
**Why**: Enables offline processing of tree path queries with complex aggregations; LCA + Mo's trick extends array Mo's to trees.

## [Priority Search Tree]

```python
# PROBLEM: Query for minimum value in a 2D range (x in [x1,x2], y in [y1,y2])
# TIME: O(n log n) build, O(log^2 n) query SPACE: O(n)
class PrioritySearchNode:
    """Node for priority search tree"""
    def __init__(self, point):
        self.point = point  # (x, y, value)
        self.left = None
        self.right = None
        self.y_max = point[1]  # Max y in subtree


class PrioritySearchTree:
    """
    Priority Search Tree: Heap on value, BST on x-coordinate.
    Efficiently queries minimum value in a 2D rectangular region.
    """
    
    def __init__(self, points):
        """
        Build priority search tree from points.
        
        Args:
            points: List of (x, y, value) tuples
        """
        self.points = sorted(points, key=lambda p: p[0])  # Sort by x
        self.root = self._build(self.points)
    
    def _build(self, points):
        """Build tree recursively"""
        if not points:
            return None
        
        # Find point with minimum value (becomes root of subtree)
        min_idx = min(range(len(points)), key=lambda i: points[i][2])
        root_point = points[min_idx]
        
        node = PrioritySearchNode(root_point)
        
        # Split remaining points by median x
        remaining = points[:min_idx] + points[min_idx+1:]
        if not remaining:
            return node
        
        mid = len(remaining) // 2
        left_points = remaining[:mid]
        right_points = remaining[mid:]
        
        node.left = self._build(left_points)
        node.right = self._build(right_points)
        
        # Update y_max
        node.y_max = root_point[1]
        if node.left:
            node.y_max = max(node.y_max, node.left.y_max)
        if node.right:
            node.y_max = max(node.y_max, node.right.y_max)
        
        return node
    
    def query_min(self, x1, x2, y1, y2):
        """
        Find minimum value among points where x in [x1,x2] and y in [y1,y2].
        
        Args:
            x1, x2: x-coordinate bounds
            y1, y2: y-coordinate bounds
        
        Returns:
            Minimum value and its point, or None if no points in range
        """
        return self._query_min(self.root, x1, x2, y1, y2)
    
    def _query_min(self, node, x1, x2, y1, y2):
        """Recursive query helper"""
        if node is None:
            return None
        
        x, y, val = node.point
        
        # Check if node's y is outside query range (prune by y_max)
        if node.y_max < y1:
            return None
        
        result = None
        
        # Check if current point is in range
        if x1 <= x <= x2 and y1 <= y <= y2:
            result = (val, node.point)
        
        # Search children
        if node.left and node.left.y_max >= y1:
            left_result = self._query_min(node.left, x1, x2, y1, y2)
            if left_result:
                if result is None or left_result[0] < result[0]:
                    result = left_result
        
        if node.right and node.right.y_max >= y1:
            right_result = self._query_min(node.right, x1, x2, y1, y2)
            if right_result:
                if result is None or right_result[0] < result[0]:
                    result = right_result
        
        return result
    
    def query_all_in_range(self, x1, x2, y1, y2):
        """Find all points in range (for reporting queries)"""
        result = []
        self._query_all(self.root, x1, x2, y1, y2, result)
        return result
    
    def _query_all(self, node, x1, x2, y1, y2, result):
        if node is None or node.y_max < y1:
            return
        
        x, y, val = node.point
        if x1 <= x <= x2 and y1 <= y <= y2:
            result.append(node.point)
        
        self._query_all(node.left, x1, x2, y1, y2, result)
        self._query_all(node.right, x1, x2, y1, y2, result)
```
**Why**: Combines benefits of heap and BST for 2D range queries; useful in computational geometry and database indexing.

## [Range Tree]

```python
# PROBLEM: Report all points in a 2D/3D rectangular query range
# TIME: O(n log^(d-1) n) build, O(log^(d-1) n + k) query SPACE: O(n log^(d-1) n)
class RangeTree1D:
    """1D range tree (balanced BST) for single dimension queries"""
    
    def __init__(self, points, key_idx=0):
        """
        Build 1D range tree.
        
        Args:
            points: List of points (tuples)
            key_idx: Index of coordinate to use for sorting
        """
        self.key_idx = key_idx
        sorted_points = sorted(points, key=lambda p: p[key_idx])
        self.root = self._build(sorted_points)
    
    def _build(self, points):
        """Build balanced BST from sorted points"""
        if not points:
            return None
        
        mid = len(points) // 2
        node = {
            'point': points[mid],
            'key': points[mid][self.key_idx],
            'left': self._build(points[:mid]),
            'right': self._build(points[mid+1:])
        }
        return node
    
    def query(self, low, high):
        """Find all points with key in [low, high]"""
        result = []
        self._query(self.root, low, high, result)
        return result
    
    def _query(self, node, low, high, result):
        if node is None:
            return
        
        if low <= node['key'] <= high:
            result.append(node['point'])
        
        if low < node['key']:
            self._query(node['left'], low, high, result)
        if high > node['key']:
            self._query(node['right'], low, high, result)


class RangeTree2D:
    """
    2D Range Tree: Primary tree on x, each node has auxiliary tree on y.
    Queries: O(log^2 n + k) where k is output size.
    """
    
    def __init__(self, points):
        """
        Build 2D range tree.
        
        Args:
            points: List of (x, y, ...) tuples
        """
        self.points = points[:]
        self.root = self._build(points)
    
    def _build(self, points):
        """Build 2D range tree with fractional cascading optimization"""
        if not points:
            return None
        
        # Sort by x
        sorted_x = sorted(points, key=lambda p: p[0])
        mid = len(sorted_x) // 2
        
        # Points in this subtree (for auxiliary tree)
        subtree_points = sorted_x
        
        node = {
            'x': sorted_x[mid][0],
            'point': sorted_x[mid],
            'left': self._build(sorted_x[:mid]),
            'right': self._build(sorted_x[mid+1:]),
            # Auxiliary tree on y for all points in subtree
            'aux_tree': RangeTree1D(subtree_points, key_idx=1)
        }
        return node
    
    def query(self, x1, x2, y1, y2):
        """
        Find all points with x in [x1, x2] and y in [y1, y2].
        
        Returns:
            List of points in the query rectangle
        """
        result = []
        self._query(self.root, x1, x2, y1, y2, result)
        return result
    
    def _query(self, node, x1, x2, y1, y2, result):
        if node is None:
            return
        
        # Find split node (where paths to x1 and x2 diverge)
        x = node['x']
        
        if x1 <= x <= x2:
            # Check current point
            point = node['point']
            if y1 <= point[1] <= y2:
                result.append(point)
            
            # In left subtree: query for all x >= x1
            self._query_left(node['left'], x1, y1, y2, result)
            # In right subtree: query for all x <= x2
            self._query_right(node['right'], x2, y1, y2, result)
        
        elif x < x1:
            # Go right
            self._query(node['right'], x1, x2, y1, y2, result)
        else:  # x > x2
            # Go left
            self._query(node['left'], x1, x2, y1, y2, result)
    
    def _query_left(self, node, x1, y1, y2, result):
        """Query nodes where we need x >= x1 (all right subtrees are included)"""
        if node is None:
            return
        
        x = node['x']
        point = node['point']
        
        if x >= x1:
            # This node and its right subtree are fully in range (x-wise)
            # Query auxiliary tree for y range
            for p in node['aux_tree'].query(y1, y2):
                if p not in result:  # Avoid duplicates
                    result.append(p)
            # Also check left subtree
            self._query_left(node['left'], x1, y1, y2, result)
        else:
            # Only check right subtree
            self._query_left(node['right'], x1, y1, y2, result)
    
    def _query_right(self, node, x2, y1, y2, result):
        """Query nodes where we need x <= x2 (all left subtrees are included)"""
        if node is None:
            return
        
        x = node['x']
        
        if x <= x2:
            # This node and its left subtree are fully in range (x-wise)
            for p in node['aux_tree'].query(y1, y2):
                if p not in result:
                    result.append(p)
            self._query_right(node['right'], x2, y1, y2, result)
        else:
            self._query_right(node['left'], x2, y1, y2, result)
```
**Why**: Optimal data structure for orthogonal range reporting in higher dimensions; foundation for many geometric algorithms.
