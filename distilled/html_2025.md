## js_event_passive_option

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Passive Event Listeners</title>
</head>
<body>
    <div id="scrollContainer" style="height: 300px; overflow-y: scroll; border: 2px solid #ccc;">
        <div style="height: 1000px; padding: 20px;">
            Scroll down to see passive event listener in action...
            <p>Check console for event timing</p>
        </div>
    </div>

    <script>
        // PROBLEM: Improve scroll performance by preventing blocking
        // TIME: O(1) SPACE: O(1)
        
        const container = document.getElementById('scrollContainer');
        
        // Passive event listener - tells browser we won't call preventDefault()
        // This allows browser to scroll immediately without waiting for handler
        container.addEventListener('scroll', (event) => {
            console.log('Scroll position:', container.scrollTop);
            // Browser can scroll immediately - no preventDefault() allowed
        }, { passive: true }); // Key: passive: true
        
        // Non-passive listener - browser must wait for handler to complete
        container.addEventListener('touchstart', (event) => {
            // Could call preventDefault() here, so browser waits
            console.log('Touch start - non-passive');
        }, { passive: false });
        
        // Why: Passive listeners improve scroll/touch performance by ~50%
        // Browser doesn't need to wait for preventDefault() check
    </script>
</body>
</html>
```

**Why:** Passive event listeners dramatically improve scrolling performance by telling the browser the event handler won't call preventDefault(). This allows the browser to start scrolling immediately instead of waiting for the JavaScript to execute, resulting in smoother user experience especially on mobile devices.

```html
<!DOCTYPE html>
<html lang="en">

    <title>IndexedDB Add Operation</title>
</head>
<body>
    <input type="text" id="taskInput" placeholder="Enter task">

    <div id="status"></div>

    <script>
        // PROBLEM: Store data persistently in browser with IndexedDB
        // TIME: O(log n) for indexing SPACE: O(n) for storage
        
        let db;
        
        // Open or create database
        const request = indexedDB.open('TaskDatabase', 1);
        
        request.onerror = (event) => {
            console.error('Database error:', event.target.error);
        };
        
        request.onsuccess = (event) => {
            db = event.target.result;
            console.log('Database opened successfully');
        };
        
        // Create object store when database is created/upgraded
        request.onupgradeneeded = (event) => {
            db = event.target.result;
            
            // Create object store with key path
            const objectStore = db.createObjectStore('tasks', {
                keyPath: 'id',
                autoIncrement: true
            });
            
            // Create index for searching by name
            objectStore.createIndex('taskName', 'taskName', { unique: false });
            
            console.log('Object store created');
        };
        
        function addTask() {
            const taskInput = document.getElementById('taskInput');
            const taskName = taskInput.value.trim();
            
            if (!taskName) {
                document.getElementById('status').textContent = 'Please enter a task';
                return;
            }
            
            // Start a readwrite transaction
            const transaction = db.transaction(['tasks'], 'readwrite');
            const objectStore = transaction.objectStore('tasks');
            
            // Add new task
            const task = {
                taskName: taskName,
                created: new Date().toISOString()
            };
            
            const request = objectStore.add(task);
            
            request.onsuccess = (event) => {
                document.getElementById('status').textContent = 
                    `Task added with ID: ${event.target.result}`;
                taskInput.value = '';
            };
            
            request.onerror = (event) => {
                document.getElementById('status').textContent = 
                    `Error adding task: ${event.target.error}`;
            };
        }
        
        // Why: IndexedDB provides large-scale, structured, persistent storage
        // Unlike localStorage (5MB limit), IndexedDB can store hundreds of MB
    </script>
</body>
</html>
```

**Why:** IndexedDB add() provides persistent, large-scale storage in the browser with indexing support. Unlike localStorage's 5MB limit and synchronous API, IndexedDB handles hundreds of megabytes asynchronously with transaction support, making it ideal for offline-capable web applications.

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Web Workers - Background Processing</title>
</head>
<body>
    <button onclick="startHeavyComputation()">Start Computation</button>

    <div id="output">Click button to start</div>
    <div>UI remains responsive: <input type="text" placeholder="Type while computing..."></div>

    <script>
        // PROBLEM: Run CPU-intensive tasks without blocking UI
        // TIME: O(n) for computation SPACE: O(1) for worker communication
        
        let worker;
        
        // Create worker from inline code using Blob
        const workerCode = `
            self.onmessage = function(event) {
                const { start, end } = event.data;
                let result = 0;
                
                // Heavy computation
                for (let i = start; i < end; i++) {
                    result += Math.sqrt(i) * Math.sin(i);
                }
                
                // Send result back to main thread
                self.postMessage({ result: result, processed: end - start });
            };
        `;
        
        const blob = new Blob([workerCode], { type: 'application/javascript' });
        const workerUrl = URL.createObjectURL(blob);
        
        function startHeavyComputation() {
            // Create new worker
            worker = new Worker(workerUrl);
            
            const output = document.getElementById('output');
            output.textContent = 'Computing...';
            
            // Listen for messages from worker
            worker.onmessage = (event) => {
                const { result, processed } = event.data;
                output.textContent = `Result: ${result.toFixed(2)} (processed ${processed} numbers)`;
                
                // Clean up worker
                worker.terminate();
                worker = null;
            };
            
            // Handle worker errors
            worker.onerror = (error) => {
                output.textContent = `Worker error: ${error.message}`;
                worker.terminate();
                worker = null;
            };
            
            // Start computation
            worker.postMessage({ start: 0, end: 1000000 });
            
            console.log('Computation started in background thread');
        }
        
        function stopWorker() {
            if (worker) {
                worker.terminate();
                worker = null;
                document.getElementById('output').textContent = 'Computation stopped';
            }
        }
        
        // Why: Web Workers run in separate threads, keeping UI responsive
        // Essential for heavy computations, large data processing, etc.
    </script>
</body>
</html>
```

**Why:** Web Workers enable true parallel processing in JavaScript by running code in background threads separate from the main UI thread. This prevents heavy computations from freezing the interface, maintaining responsiveness while performing CPU-intensive tasks like data processing, encryption, or complex calculations.
