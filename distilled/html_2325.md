```html
<!DOCTYPE html>

    <script>
        /**
         * Delete a record from IndexedDB
         * PROBLEM: Remove data from browser's IndexedDB database
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} dbName - Database name
         * @param {string} storeName - Object store name
         * @param {*} key - Key of the record to delete
         * @returns {Promise} Resolves when deletion is complete
         */
        async function deleteFromIndexedDB(dbName, storeName, key) {
            return new Promise((resolve, reject) => {
                // Open database connection
                const request = indexedDB.open(dbName, 1);
                
                request.onerror = () => reject(request.error);
                
                request.onsuccess = (event) => {
                    const db = event.target.result;
                    
                    // Start a readwrite transaction
                    const transaction = db.transaction([storeName], 'readwrite');
                    const store = transaction.objectStore(storeName);
                    
                    // Delete the record by key
                    const deleteRequest = store.delete(key);
                    
                    deleteRequest.onsuccess = () => {
                        console.log('Record deleted successfully');
                        resolve();
                    };
                    
                    deleteRequest.onerror = () => {
                        console.error('Error deleting record:', deleteRequest.error);
                        reject(deleteRequest.error);
                    };
                    
                    // Close database when transaction completes
                    transaction.oncomplete = () => db.close();
                };
            });
        }
        
        // Example usage
        async function deleteRecord() {
            try {
                await deleteFromIndexedDB('MyDatabase', 'users', 'user123');
                document.getElementById('status').textContent = 'Record deleted!';
            } catch (error) {
                document.getElementById('status').textContent = 'Error: ' + error.message;
            }
        }
    </script>
</body>
</html>
```

**Why:** IndexedDB delete operations are essential for managing browser-based storage, allowing offline applications to remove outdated or unwanted data without server roundtrips.

## 2. js_indexeddb_transaction

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>IndexedDB Transaction Management</title>
</head>
<body>
    <div id="log"></div>
    
    <script>
        /**
         * Execute multiple operations in a single IndexedDB transaction
         * PROBLEM: Perform atomic database operations with rollback support
         * TIME: O(n) where n is number of operations SPACE: O(1)
         * 
         * @param {string} dbName - Database name
         * @param {string} storeName - Object store name
         * @param {Array} operations - Array of operations to execute
         * @returns {Promise} Resolves when transaction completes
         */
        async function executeTransaction(dbName, storeName, operations) {
            return new Promise((resolve, reject) => {
                const request = indexedDB.open(dbName, 1);
                
                request.onerror = () => reject(request.error);
                
                request.onsuccess = (event) => {
                    const db = event.target.result;
                    
                    // Create a readwrite transaction
                    // All operations in this transaction are atomic
                    const transaction = db.transaction([storeName], 'readwrite');
                    const store = transaction.objectStore(storeName);
                    
                    // Execute each operation
                    operations.forEach((op) => {
                        let request;
                        
                        switch (op.type) {
                            case 'add':
                                request = store.add(op.data, op.key);
                                break;
                            case 'put':
                                request = store.put(op.data, op.key);
                                break;
                            case 'delete':
                                request = store.delete(op.key);
                                break;
                            default:
                                console.warn('Unknown operation type:', op.type);
                        }
                        
                        request.onsuccess = () => {
                            console.log(`Operation ${op.type} completed`);
                        };
                        
                        request.onerror = () => {
                            console.error(`Operation ${op.type} failed:`, request.error);
                        };
                    });
                    
                    // Transaction lifecycle events
                    transaction.oncomplete = () => {
                        console.log('Transaction completed successfully');
                        db.close();
                        resolve();
                    };
                    
                    transaction.onerror = () => {
                        console.error('Transaction failed:', transaction.error);
                        db.close();
                        reject(transaction.error);
                    };
                    
                    transaction.onabort = () => {
                        console.log('Transaction aborted');
                        db.close();
                        reject(new Error('Transaction aborted'));
                    };
                };
            });
        }
        
        // Example: Atomic update of user and log
        async function updateUserAndLog() {
            const operations = [
                { type: 'put', data: { name: 'John', age: 30 }, key: 'user1' },
                { type: 'add', data: { timestamp: Date.now(), action: 'update' }, key: `log_${Date.now()}` }
            ];
            
            try {
                await executeTransaction('MyAppDB', 'users', operations);
                document.getElementById('log').textContent = 'Transaction successful!';
            } catch (error) {
                document.getElementById('log').textContent = 'Transaction failed: ' + error.message;
            }
        }
        
        updateUserAndLog();
    </script>
</body>
</html>
```

**Why:** Transactions ensure data integrity by making multiple operations atomic - either all succeed or all fail, preventing partial updates that could corrupt data.

```html
         * Create and use a Request object for fetch operations
         * PROBLEM: Configure HTTP requests with full control over headers, method, and body
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} url - Request URL
         * @param {Object} options - Request configuration
         * @returns {Promise<Response>} Fetch response
         */
        async function makeRequest(url, options = {}) {
            // Create a Request object with configuration
            const request = new Request(url, {
                method: options.method || 'GET',
                headers: new Headers({
                    'Content-Type': 'application/json',
                    'Authorization': `Bearer ${options.token || ''}`,
                    ...options.headers
                }),
                body: options.body ? JSON.stringify(options.body) : undefined,
                mode: options.mode || 'cors', // cors, no-cors, same-origin
                credentials: options.credentials || 'same-origin', // include, same-origin, omit
                cache: options.cache || 'default', // default, no-store, reload, no-cache
                redirect: options.redirect || 'follow', // follow, error, manual
                referrer: options.referrer || 'about:client',
                referrerPolicy: options.referrerPolicy || 'no-referrer-when-downgrade'
            });
            
            // Log request details for debugging
            console.log('Request URL:', request.url);
            console.log('Request Method:', request.method);
            console.log('Request Headers:', [...request.headers.entries()]);
            
            try {
                const response = await fetch(request);
                
                if (!response.ok) {
                    throw new Error(`HTTP ${response.status}: ${response.statusText}`);
                }
                
                return response;
            } catch (error) {
                console.error('Request failed:', error);
                throw error;
            }
        }
        
        // Example: POST request with Request object
        async function createUser() {
            try {
                const response = await makeRequest('https://api.example.com/users', {
                    method: 'POST',
                    body: { name: 'Alice', email: 'alice@example.com' },

                    credentials: 'include'
                });
                
                const data = await response.json();
                document.getElementById('response').textContent = JSON.stringify(data, null, 2);
            } catch (error) {
                document.getElementById('response').textContent = 'Error: ' + error.message;
            }
        }

    </script>
</body>
</html>
```

**Why:** The Request object provides granular control over HTTP requests, enabling advanced features like cache control, CORS handling, and credential management in a reusable, testable way.

## 4. js_abortsignal

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>AbortController and AbortSignal</title>
</head>
<body>
    <button onclick="fetchData()">Fetch Data</button>
    <button onclick="cancelFetch()">Cancel</button>
    <div id="status"></div>
    
    <script>
        let controller = null;
        
        /**
         * Fetch with abort capability using AbortController
         * PROBLEM: Cancel ongoing fetch requests when no longer needed
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} url - URL to fetch
         * @param {number} timeout - Timeout in milliseconds
         * @returns {Promise} Resolves with data or rejects on abort/timeout
         */
        async function fetchWithAbort(url, timeout = 5000) {
            // Create a new AbortController for this request
            controller = new AbortController();
            const signal = controller.signal;
            
            // Optional: Add timeout
            const timeoutId = setTimeout(() => {
                controller.abort();
            }, timeout);
            
            try {
                const response = await fetch(url, {
                    signal: signal // Pass the abort signal
                });
                
                clearTimeout(timeoutId);
                
                if (!response.ok) {
                    throw new Error(`HTTP ${response.status}`);
                }
                
                const data = await response.json();
                return data;
                
            } catch (error) {
                clearTimeout(timeoutId);
                
                // Check if request was aborted
                if (error.name === 'AbortError') {
                    throw new Error('Request was cancelled');
                }
                
                throw error;
            } finally {
                controller = null;
            }
        }
        
        // Example: Fetch with cancel button
        async function fetchData() {
            const statusEl = document.getElementById('status');
            statusEl.textContent = 'Fetching...';
            
            try {
                const data = await fetchWithAbort('https://api.example.com/data', 10000);
                statusEl.textContent = 'Data: ' + JSON.stringify(data);
            } catch (error) {
                statusEl.textContent = 'Error: ' + error.message;
            }
        }
        
        // Cancel the ongoing fetch
        function cancelFetch() {
            if (controller) {
                controller.abort();
                document.getElementById('status').textContent = 'Request cancelled';
            }
        }
        
        // Advanced: Multiple requests sharing one abort signal
        async function fetchMultipleWithAbort(urls) {
            const controller = new AbortController();
            const signal = controller.signal;
            
            const promises = urls.map(url => 
                fetch(url, { signal }).then(r => r.json())
            );
            
            // Cancel all if one fails
            try {
                return await Promise.all(promises);
            } catch (error) {
                controller.abort(); // Cancel all remaining requests
                throw error;
            }
        }
    </script>
</body>
</html>
```

**Why:** AbortController enables request cancellation, improving user experience by stopping unnecessary network activity and freeing resources when users navigate away or change their mind.

```html
            element.classList.remove('focused');
            
            console.log(`Element blurred: ${element.id}`);
            
            // relatedTarget: element gaining focus (for blur event)
            if (event.relatedTarget) {
                console.log(`Next focus: ${event.relatedTarget.id}`);
            }
            
            // Validate on blur
            validateField(element);

         * Validate form field
         * @param {HTMLElement} element - Form element to validate

                    message = isValid ? '✓ Valid username' : '✗ Username must be 3+ characters';

                case 'email':
                    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
```

```html
         * 
         * @param {FormData} formData - FormData object
         * @param {string} fieldName - Name of field to check
         * @returns {boolean} True if field exists in FormData

        function hasField(formData, fieldName) {

        }
        
        /**
         * Validate form before submission
         */
        function validateForm() {
            const form = document.getElementById('profileForm');
            const formData = new FormData(form);

            if (!hasField(formData, 'fullName')) {
                errors.push('Full name is required');
            } else {
                const name = formData.get('fullName');
                if (name.trim() === '') {
                    errors.push('Full name cannot be empty');
                }
            }
            
            // Check if checkboxes are checked
            // Note: Unchecked checkboxes are NOT included in FormData
            if (!hasField(formData, 'terms')) {
                errors.push('You must agree to the terms');
            }
            
            // Optional: Check newsletter subscription
            const isSubscribed = hasField(formData, 'subscribe');
            
            // Display validation results
            if (errors.length > 0) {
                validationEl.innerHTML = `
                    <div style="color: red;">
                        <strong>Validation Errors:</strong><br>
                        ${errors.join('<br>')}
                    </div>
                `;
            } else {
                validationEl.innerHTML = `
                    <div style="color: green;">
                        <strong>✓ Form is valid!</strong><br>
                        Newsletter: ${isSubscribed ? 'Subscribed' : 'Not subscribed'}
                    </div>
                `;
            }
            
            return errors.length === 0;
        }
        
        // Advanced: Conditional field checking
        function checkConditionalFields() {
            const formData = new FormData(document.getElementById('profileForm'));
            
            // Different validation based on subscription status
            if (formData.has('subscribe')) {

                    console.log('Email provided:', formData.get('email'));
```

## 8. html5_service_worker_register

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Service Worker Registration</title>
</head>
<body>
    <h1>Service Worker Registration Demo</h1>
    <div id="status"></div>
    <button onclick="checkRegistration()">Check Registration</button>
    <button onclick="unregister()">Unregister</button>
    
    <script>
        /**
         * Register a service worker for offline capabilities
         * PROBLEM: Enable offline functionality and background processing
         * TIME: O(1) SPACE: O(1)
         * 
         * @param {string} scriptURL - Path to service worker script
         * @param {Object} options - Registration options
         * @returns {Promise<ServiceWorkerRegistration>} Registration object
         */
        async function registerServiceWorker(scriptURL = '/sw.js', options = {}) {
            // Check if service workers are supported
            if (!('serviceWorker' in navigator)) {
                console.warn('Service Workers not supported in this browser');
                return null;
            }
            
            try {
                // Register the service worker
                const registration = await navigator.serviceWorker.register(scriptURL, {
                    scope: options.scope || '/' // Controls which pages the SW controls
                });
                
                console.log('Service Worker registered successfully:', registration);
                
                // Log registration state
                if (registration.installing) {
                    console.log('Service Worker installing');
                } else if (registration.waiting) {
                    console.log('Service Worker waiting');
                } else if (registration.active) {
                    console.log('Service Worker active');
                }
                
                // Update found
                registration.onupdatefound = () => {
                    console.log('New Service Worker version available');
                    const newWorker = registration.installing;
                    
                    newWorker.onstatechange = () => {
                        console.log('Service Worker state:', newWorker.state);
                        
                        if (newWorker.state === 'installed' && navigator.serviceWorker.controller) {
                            // New content is available
                            console.log('New content available; please refresh.');
                            notifyUserOfUpdate();
                        }
                    };
                };
                
                return registration;
                
            } catch (error) {
                console.error('Service Worker registration failed:', error);
                throw error;
            }
        }
        
        /**
         * Check current registration status
         */
        async function checkRegistration() {
            const statusEl = document.getElementById('status');
            
            if (!('serviceWorker' in navigator)) {
                statusEl.textContent = 'Service Workers not supported';
                return;
            }
            
            const registration = await navigator.serviceWorker.getRegistration();
            
            if (registration) {
                statusEl.innerHTML = `
                    <strong>Service Worker Registered</strong><br>
                    Scope: ${registration.scope}<br>
                    Active: ${registration.active ? 'Yes' : 'No'}
                `;
            } else {
                statusEl.textContent = 'No Service Worker registered';
            }
        }
        
        /**
         * Unregister service worker
         */
        async function unregister() {
            const registration = await navigator.serviceWorker.getRegistration();
            
            if (registration) {
                const success = await registration.unregister();
                document.getElementById('status').textContent = 
                    success ? 'Service Worker unregistered' : 'Unregistration failed';
            }
        }
        
        /**
         * Notify user of available update
         */
        function notifyUserOfUpdate() {
            if (confirm('New version available! Reload?')) {
                window.location.reload();
            }
```

```html
         * PROBLEM: Cache essential resources during service worker installation
         * TIME: O(n) where n is number of resources SPACE: O(n) for cache storage
         * 
         * NOTE: This code goes in the service worker file (sw.js), not the main page
         */
        
        // Content for sw.js file:
        const SW_INSTALL_CODE = `
// Service Worker Install Event
const CACHE_NAME = 'my-app-cache-v1';
const RESOURCES_TO_CACHE = [
    '/',
    '/index.html',
    '/styles.css',
    '/app.js',
    '/images/logo.png',
    '/offline.html'
];

// Install event - cache essential resources
self.addEventListener('install', (event) => {
    console.log('[Service Worker] Installing...');
    
    // Wait until caching is complete
    event.waitUntil(
        caches.open(CACHE_NAME)
            .then((cache) => {
                console.log('[Service Worker] Caching app shell');
                
                // Add all resources to cache
                return cache.addAll(RESOURCES_TO_CACHE);
            })
            .then(() => {
                console.log('[Service Worker] Install complete');
                
                // Force the waiting service worker to become active
                return self.skipWaiting();
            })
            .catch((error) => {
                console.error('[Service Worker] Install failed:', error);
            })
    );
});

// Activate event - clean up old caches
self.addEventListener('activate', (event) => {
    console.log('[Service Worker] Activating...');
    
    event.waitUntil(
        caches.keys()
            .then((cacheNames) => {
                return Promise.all(
                    cacheNames.map((cacheName) => {
                        // Delete old caches
                        if (cacheName !== CACHE_NAME) {
                            console.log('[Service Worker] Deleting old cache:', cacheName);
                            return caches.delete(cacheName);
                        }
                    })
                );
            })
            .then(() => {
                console.log('[Service Worker] Claiming clients');
                
                // Take control of all pages immediately
                return self.clients.claim();
            })
    );
});

// Fetch event - serve from cache, fallback to network
self.addEventListener('fetch', (event) => {
    event.respondWith(
        caches.match(event.request)
            .then((cachedResponse) => {
                // Return cached version or fetch from network
                if (cachedResponse) {
                    return cachedResponse;
                }
                
                return fetch(event.request)
                    .catch(() => {
                        // Return offline page for navigation requests
                        if (event.request.mode === 'navigate') {
                            return caches.match('/offline.html');
                        }
                    });
            })
    );
});
`;

        /**
         * Main page code: Monitor service worker installation
         */
        async function monitorInstallation() {
            const statusEl = document.getElementById('status');
            
            if (!('serviceWorker' in navigator)) {
                statusEl.textContent = 'Service Workers not supported';
                return;
            }
            
            try {
                const registration = await navigator.serviceWorker.register('/sw.js');
                
                // Monitor installation progress
                if (registration.installing) {
                    const worker = registration.installing;
                    
                    worker.addEventListener('statechange', () => {
                        statusEl.textContent = \`Service Worker state: \${worker.state}\`;
                        
                        if (worker.state === 'installed') {
                            statusEl.textContent = 'Resources cached successfully!';
                        }
                    });
                }
                
                // Log updates
                registration.addEventListener('updatefound', () => {
                    console.log('New version installing...');
                });
                
            } catch (error) {
                statusEl.textContent = 'Registration failed: ' + error.message;
            }
        }
        
        // Display the Service Worker code for reference
        document.getElementById('status').innerHTML = `
            <h3>Service Worker Install Code (sw.js):</h3>
            <pre style="background: #f5f5f5; padding: 10px; overflow: auto; max-height: 400px;">
```
