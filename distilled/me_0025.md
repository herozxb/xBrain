```python
from typing import Optional, Callable, TypeVar

T = TypeVar('T')

class VirtualProxy:
    """Delays object creation until first use."""
    
    def __init__(self, factory: Callable[[], T]):
        self._factory = factory
        self._subject: Optional[T] = None
    
    def __getattr__(self, name: str):
        # Lazy initialization on first access
        if self._subject is None:
            self._subject = self._factory()
        return getattr(self._subject, name)

        return self._subject is not None
```

```python
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import List

class Visitor(ABC):
    @abstractmethod
    def visit_element_a(self, element: 'ElementA') -> None: pass
    
    @abstractmethod
    def visit_element_b(self, element: 'ElementB') -> None: pass


class Element(ABC):
    @abstractmethod
    def accept(self, visitor: Visitor) -> None: pass


@dataclass
class ElementA(Element):
    name: str
    value: int
    
    def accept(self, visitor: Visitor) -> None:
        visitor.visit_element_a(self)


@dataclass
class ElementB(Element):
    title: str
    items: List[str]
    
    def accept(self, visitor: Visitor) -> None:
        visitor.visit_element_b(self)


class ExportVisitor(Visitor):
    """Exports elements to JSON format."""
    
    def __init__(self):
        self._output: List[str] = []
    
    def visit_element_a(self, element: ElementA) -> None:
        self._output.append(f'{{"type":"A","name":"{element.name}","value":{element.value}}}')
    
    def visit_element_b(self, element: ElementB) -> None:
        items_str = ','.join(f'"{item}"' for item in element.items)
        self._output.append(f'{{"type":"B","title":"{element.title}","items":[{items_str}]}}')
    
    def get_result(self) -> str:
        return '[' + ','.join(self._output) + ']'


class ObjectStructure:
    """Container for elements that can be visited."""
    
    def __init__(self):
        self._elements: List[Element] = []
    
    def attach(self, element: Element) -> None:
        self._elements.append(element)
    
    def accept(self, visitor: Visitor) -> None:
        for element in self._elements:
            element.accept(visitor)
```
**Why**: Separates algorithms from object structure, enables easy addition of new operations.

## 5. Message Bus (In-Memory)

```python
# PROBLEM: Decouple message producers from consumers
# TIME: O(1) publish, O(m) for m subscribers SPACE: O(n) queued messages
from typing import Callable, Dict, List, Any
from queue import Queue
from threading import Thread, Lock
from dataclasses import dataclass
import uuid

@dataclass
class Message:
    """Message envelope with metadata."""
    id: str
    topic: str
    payload: Any
    headers: Dict[str, str]


class MessageBus:
    """In-memory pub/sub message bus with queue support."""
    
    def __init__(self):
        self._subscribers: Dict[str, List[Callable[[Message], None]]] = {}
        self._queues: Dict[str, Queue] = {}
        self._lock = Lock()
        self._running = False
    
    def subscribe(self, topic: str, handler: Callable[[Message], None]) -> None:
        """Register handler for topic."""
        with self._lock:
            if topic not in self._subscribers:
                self._subscribers[topic] = []
            self._subscribers[topic].append(handler)
    
    def publish(self, topic: str, payload: Any, headers: Dict[str, str] = None) -> str:
        """Publish message to topic, returns message ID."""
        message = Message(
            id=str(uuid.uuid4()),
            topic=topic,
            payload=payload,
            headers=headers or {}
        )
        
        with self._lock:
            # Notify subscribers
            for handler in self._subscribers.get(topic, []):
                try:
                    handler(message)
                except Exception as e:
                    print(f"Handler error: {e}")
            
            # Queue for async consumers
            if topic in self._queues:
                self._queues[topic].put(message)
        
        return message.id
    
    def create_queue(self, topic: str, maxsize: int = 1000) -> Queue:
        """Create queue for async consumption."""
        with self._lock:
            if topic not in self._queues:
                self._queues[topic] = Queue(maxsize=maxsize)
            return self._queues[topic]
    
    def consume(self, topic: str, handler: Callable[[Message], None], 
                blocking: bool = True) -> None:
        """Consume messages from queue."""
        queue = self.create_queue(topic)
        
        def worker():
            while self._running:
                try:
                    message = queue.get(timeout=1.0)
                    handler(message)
                    queue.task_done()
                except:
                    continue
        
        self._running = True
        if blocking:
            worker()
        else:
            Thread(target=worker, daemon=True).start()
    
    def stop(self) -> None:
        self._running = False


# Usage example
bus = MessageBus()

def log_orders(msg: Message) -> None:
    print(f"Order received: {msg.payload}")

bus.subscribe("orders", log_orders)
msg_id = bus.publish("orders", {"order_id": 123, "item": "Widget"})
```
**Why**: Enables loose coupling between components, supports async processing patterns.

## 6. GiST Index Implementation (Simplified)

```python
# PROBLEM: Generalized Search Tree for custom index types
# TIME: O(log n) search/insert SPACE: O(n)
from typing import List, Tuple, Optional, Protocol, Any
from dataclasses import dataclass
from abc import ABC, abstractmethod

class Indexable(Protocol):
    """Protocol for indexable objects."""
    def contains(self, other: 'Indexable') -> bool: ...
    def intersects(self, other: 'Indexable') -> bool: ...
    def union(self, other: 'Indexable') -> 'Indexable': ...


@dataclass
class Interval:
    """Example indexable type for GiST."""
    start: float
    end: float
    
    def contains(self, other: 'Interval') -> bool:
        return self.start <= other.start and self.end >= other.end
    
    def intersects(self, other: 'Interval') -> bool:
        return not (self.end < other.start or self.start > other.end)
    
    def union(self, other: 'Interval') -> 'Interval':
        return Interval(
            min(self.start, other.start),
            max(self.end, other.end)
        )


class GiSTNode:
    """GiST tree node."""
    
    def __init__(self, is_leaf: bool = True):
        self.is_leaf = is_leaf
        self.entries: List[Tuple[Any, Optional['GiSTNode']]] = []
        self.boundary: Optional[Indexable] = None
    
    def update_boundary(self) -> None:
        if not self.entries:
            return
        boundaries = [e[0] for e in self.entries]
        self.boundary = boundaries[0]
        for b in boundaries[1:]:
            self.boundary = self.boundary.union(b)


class GiST:
    """Generalized Search Tree implementation."""
    
    def __init__(self, max_entries: int = 4):
        self._root = GiSTNode(is_leaf=True)
        self._max_entries = max_entries
    
    def insert(self, key: Indexable, value: Any = None) -> None:
        """Insert key into index."""
        path = self._choose_path(self._root, key)
        node = path[-1]
        
        node.entries.append((key, value))
        node.update_boundary()
        
        # Split if necessary
        if len(node.entries) > self._max_entries:
            self._split(path)
    
    def _choose_path(self, node: GiSTNode, key: Indexable) -> List[GiSTNode]:
        """Find insertion path."""
        if node.is_leaf:
            return [node]
        
        # Find best child
        best_child = None
        min_penalty = float('inf')
        
        for entry_key, child in node.entries:
            if child is None:
                continue
            # Penalty: increase in boundary size
            if child.boundary:
                union = child.boundary.union(key)
                penalty = self._size(union) - self._size(child.boundary)
                if penalty < min_penalty:
                    min_penalty = penalty
                    best_child = child
        
        if best_child:
            return [node] + self._choose_path(best_child, key)
        return [node]
    
    def _size(self, interval: Interval) -> float:
        """Measure interval size."""
        return interval.end - interval.start
    
    def _split(self, path: List[GiSTNode]) -> None:
        """Split overflowing node."""
        node = path[-1]
        mid = len(node.entries) // 2
        
        new_node = GiSTNode(is_leaf=node.is_leaf)
        new_node.entries = node.entries[mid:]
        node.entries = node.entries[:mid]
        
        node.update_boundary()
        new_node.update_boundary()
        
        # Propagate split up
        if len(path) == 1:  # Root split
            new_root = GiSTNode(is_leaf=False)
            new_root.entries = [
                (node.boundary, node),
                (new_node.boundary, new_node)
            ]
            new_root.update_boundary()
            self._root = new_root
    
    def search(self, query: Indexable) -> List[Any]:
        """Search for entries intersecting query."""
        results = []
        self._search_node(self._root, query, results)
        return results
    
    def _search_node(self, node: GiSTNode, query: Indexable, 
                     results: List[Any]) -> None:
        """Recursive search."""
        for key, value in node.entries:
            if key.intersects(query):
                if node.is_leaf:
                    results.append(value)
                elif value:  # value is child node
                    self._search_node(value, query, results)
```
**Why**: Extensible index framework supporting custom data types and predicates.

```python
# PROBLEM: Efficient fuzzy text search using trigrams
# TIME: O(n) index build, O(m) query where m = matching trigrams SPACE: O(n)
from collections import defaultdict
from typing import Set, List, Dict

class TrigramIndex:
    """Trigram-based full-text search index."""
    
    def __init__(self):
        self._index: Dict[str, Set[int]] = defaultdict(set)
        self._documents: Dict[int, str] = {}
        self._doc_counter = 0
    
    def _tokenize(self, text: str) -> List[str]:
        """Extract trigrams from text."""
        # Normalize and pad text
        text = text.lower().strip()
        padded = f"  {text}  "  # Padding for word boundaries
        
        trigrams = []
        for i in range(len(padded) - 2):
            trigrams.append(padded[i:i+3])
        return trigrams
    
    def add_document(self, doc_id: int, text: str) -> None:
        """Index a document."""
        self._documents[doc_id] = text
        trigrams = self._tokenize(text)
        
        for trigram in trigrams:
            self._index[trigram].add(doc_id)
    
    def remove_document(self, doc_id: int) -> None:
        """Remove document from index."""
        if doc_id not in self._documents:
            return
        
        text = self._documents[doc_id]
        trigrams = self._tokenize(text)
        
        for trigram in trigrams:
            self._index[trigram].discard(doc_id)
        
        del self._documents[doc_id]
    
    def search(self, query: str, min_similarity: float = 0.3) -> List[tuple]:
        """Search for similar documents, returns (doc_id, similarity)."""
        query_trigrams = set(self._tokenize(query))
        
        if not query_trigrams:
            return []
        
        # Count matches per document
        doc_matches: Dict[int, int] = defaultdict(int)
        for trigram in query_trigrams:
            for doc_id in self._index.get(trigram, set()):
                doc_matches[doc_id] += 1
        
        # Calculate Jaccard similarity
        results = []
        for doc_id, matches in doc_matches.items():
            doc_trigrams = set(self._tokenize(self._documents[doc_id]))
            similarity = len(query_trigrams & doc_trigrams) / len(query_trigrams | doc_trigrams)
            
            if similarity >= min_similarity:
                results.append((doc_id, similarity))
        
        # Sort by similarity descending
        return sorted(results, key=lambda x: (-x[1], x[0]))
    
    def search_prefix(self, prefix: str) -> List[int]:
        """Find documents starting with prefix."""
        prefix = prefix.lower()
        prefix_trigrams = set(self._tokenize(prefix))
        
        candidates = None
        for trigram in prefix_trigrams:
            docs = self._index.get(trigram, set())
            if candidates is None:
                candidates = docs.copy()
            else:
                candidates &= docs
        
        # Filter actual matches
        results = []
        for doc_id in candidates or []:
            if self._documents[doc_id].lower().startswith(prefix):
                results.append(doc_id)
        
        return results


# Usage
index = TrigramIndex()
index.add_document(1, "hello world")
index.add_document(2, "hello there")
index.add_document(3, "goodbye world")

print(index.search("helo"))  # Fuzzy matches 1, 2
print(index.search_prefix("hel"))  # Matches 1, 2
```
**Why**: Provides fast fuzzy matching and spell-tolerant search.

## 8. Bitmap Index Implementation

```python
# PROBLEM: Efficient indexing for low-cardinality columns
# TIME: O(1) equality lookup, O(n/word_size) bitwise ops SPACE: O(n * c) for c values
from typing import Dict, List, Set
import array

class BitArray:
    """Memory-efficient bit array using integers."""
    
    def __init__(self, size: int):
        self._size = size
        self._bits: List[int] = [0] * ((size + 63) // 64)
    
    def set(self, index: int) -> None:
        """Set bit at index."""
        word, bit = divmod(index, 64)
        self._bits[word] |= (1 << bit)
    
    def clear(self, index: int) -> None:
        """Clear bit at index."""
        word, bit = divmod(index, 64)
        self._bits[word] &= ~(1 << bit)
    
    def get(self, index: int) -> bool:
        """Get bit at index."""
        word, bit = divmod(index, 64)
        return bool(self._bits[word] & (1 << bit))
    
    def and_with(self, other: 'BitArray') -> 'BitArray':
        """Bitwise AND with another bit array."""
        result = BitArray(self._size)
        for i in range(len(self._bits)):
            result._bits[i] = self._bits[i] & other._bits[i]
        return result
    
    def or_with(self, other: 'BitArray') -> 'BitArray':
        """Bitwise OR with another bit array."""
        result = BitArray(self._size)
        for i in range(len(self._bits)):
            result._bits[i] = self._bits[i] | other._bits[i]
        return result
    
    def not_(self) -> 'BitArray':
        """Bitwise NOT."""
        result = BitArray(self._size)
        for i in range(len(self._bits)):
            result._bits[i] = ~self._bits[i]
        # Clear excess bits
        excess = self._size % 64
        if excess:
            result._bits[-1] &= (1 << excess) - 1
        return result
    
    def count(self) -> int:
        """Count set bits (popcount)."""
        total = 0
        for word in self._bits:
            total += bin(word).count('1')
        return total
    
    def iter_set(self) -> List[int]:
        """Get indices of set bits."""
        result = []
        for i, word in enumerate(self._bits):
            if word:
                for bit in range(64):
                    if word & (1 << bit):
                        idx = i * 64 + bit
                        if idx < self._size:
                            result.append(idx)
        return result


class BitmapIndex:
    """Bitmap index for low-cardinality columns."""
    
    def __init__(self, capacity: int = 10000):
        self._capacity = capacity
        self._bitmaps: Dict[str, BitArray] = {}
        self._next_row = 0
    
    def add_row(self, value: str) -> int:
        """Add value and return row ID."""
        if self._next_row >= self._capacity:
            raise ValueError("Index full")
        
        if value not in self._bitmaps:
            self._bitmaps[value] = BitArray(self._capacity)
        
        self._bitmaps[value].set(self._next_row)
        return self._next_row - 1
    
    def query(self, value: str) -> List[int]:
        """Get row IDs for value."""
        if value not in self._bitmaps:
            return []
        return self._bitmaps[value].iter_set()
    
    def query_not(self, value: str) -> List[int]:
        """Get row IDs not matching value."""
        if value not in self._bitmaps:
            # All rows don't have this value
            return list(range(self._next_row))
        
        # Get all rows and subtract matching
        all_bits = BitArray(self._capacity)
        for i in range(self._next_row):
            all_bits.set(i)
        
        result = all_bits.and_with(self._bitmaps[value].not_())
        return result.iter_set()[:self._next_row]
    
    def query_and(self, value1: str, value2: str) -> List[int]:
        """Rows matching both values."""
        if value1 not in self._bitmaps or value2 not in self._bitmaps:
            return []
        result = self._bitmaps[value1].and_with(self._bitmaps[value2])
        return result.iter_set()
    
    def query_or(self, value1: str, value2: str) -> List[int]:
        """Rows matching either value."""
        bits1 = self._bitmaps.get(value1, BitArray(self._capacity))
        bits2 = self._bitmaps.get(value2, BitArray(self._capacity))
        result = bits1.or_with(bits2)
        return result.iter_set()[:self._next_row]
    
    def stats(self) -> Dict[str, int]:
        """Get index statistics."""
        return {
            'distinct_values': len(self._bitmaps),
            'rows': self._next_row,
            'total_bits': sum(b.count() for b in self._bitmaps.values())
        }
```
**Why**: Extremely efficient for analytical queries on low-cardinality columns.

## 9. Async Implementation Pattern

```python
# PROBLEM: Structure concurrent operations with async/await
# TIME: O(1) task creation, depends on workload SPACE: O(n) pending tasks
import asyncio
from typing import List, Any, Coroutine
from dataclasses import dataclass
from concurrent.futures import ThreadPoolExecutor

@dataclass
class TaskResult:
    """Wrapper for task results."""
    task_id: int
    success: bool
    value: Any = None
    error: str = None


class AsyncTaskRunner:
    """Manage and execute async tasks with concurrency control."""
    
    def __init__(self, max_concurrent: int = 10):
        self._max_concurrent = max_concurrent
        self._semaphore = asyncio.Semaphore(max_concurrent)
        self._results: List[TaskResult] = []
    
    async def run_with_limit(self, coro: Coroutine, task_id: int) -> TaskResult:
        """Execute coroutine with concurrency limit."""
        async with self._semaphore:
            try:
                result = await coro
                return TaskResult(task_id=task_id, success=True, value=result)
            except Exception as e:
                return TaskResult(task_id=task_id, success=False, error=str(e))
    
    async def run_all(self, tasks: List[Coroutine]) -> List[TaskResult]:
        """Run all tasks concurrently with limit."""
        wrapped = [
            self.run_with_limit(task, i) 
            for i, task in enumerate(tasks)
        ]
        self._results = await asyncio.gather(*wrapped)
        return self._results
    
    async def run_with_timeout(self, coro: Coroutine, 
                                timeout: float) -> Any:
        """Run with timeout."""
        try:
            return await asyncio.wait_for(coro, timeout=timeout)
        except asyncio.TimeoutError:
            raise TimeoutError("Task exceeded timeout")
    
    async def run_with_retry(self, coro_factory: Callable[[], Coroutine],
                             max_retries: int = 3,
                             delay: float = 1.0) -> Any:
        """Run with exponential backoff retry."""
        last_error = None
        for attempt in range(max_retries):
            try:
                return await coro_factory()
            except Exception as e:
                last_error = e
                if attempt < max_retries - 1:
                    await asyncio.sleep(delay * (2 ** attempt))
        raise last_error


class AsyncPipeline:
    """Pipeline for processing data through async stages."""
    
    def __init__(self):
        self._stages: List[Callable] = []
    
    def add_stage(self, processor: Callable[[Any], Coroutine]) -> 'AsyncPipeline':
        """Add processing stage."""
        self._stages.append(processor)
        return self
    
    async def process(self, data: Any) -> Any:
        """Run data through all stages."""
        result = data
        for stage in self._stages:
            result = await stage(result)
        return result
    
    async def process_batch(self, items: List[Any]) -> List[Any]:
        """Process multiple items through pipeline."""
        return await asyncio.gather(*[self.process(item) for item in items])


class AsyncProducerConsumer:
    """Classic producer-consumer with async queues."""
    
    def __init__(self, queue_size: int = 100):
        self._queue: asyncio.Queue = asyncio.Queue(maxsize=queue_size)
        self._producers: List[asyncio.Task] = []
        self._consumers: List[asyncio.Task] = []
    
    async def produce(self, items: List[Any]) -> None:
        """Producer coroutine."""
        for item in items:
            await self._queue.put(item)
            await asyncio.sleep(0.01)  # Simulate work
    
    async def consume(self, processor: Callable[[Any], Coroutine],
                      consumer_id: int) -> int:
        """Consumer coroutine."""
        processed = 0
        while True:
            try:
                item = await asyncio.wait_for(self._queue.get(), timeout=1.0)
                await processor(item)
                processed += 1
                self._queue.task_done()
            except asyncio.TimeoutError:
                break
        return processed
    
    async def run(self, items: List[Any], processor: Callable,
                  num_consumers: int = 3) -> List[int]:
        """Run producer-consumer pattern."""
        # Start consumers
        consumers = [
            asyncio.create_task(self.consume(processor, i))
            for i in range(num_consumers)
        ]
        
        # Start producer
        await self.produce(items)
        
        # Wait for completion
        await self._queue.join()
        results = await asyncio.gather(*consumers)
        return results

async def fetch_url(url: str) -> str:
```

**Why**: Enables efficient I/O-bound concurrency without thread overhead.

## 10. Duck Typing Pattern

```python
# PROBLEM: Write flexible code that works with any compatible type
# TIME: O(1) for type checking SPACE: O(1)
from typing import Protocol, runtime_checkable, Any
from dataclasses import dataclass

# Protocol-based duck typing (structural subtyping)
@runtime_checkable
class Drawable(Protocol):
    """Protocol for drawable objects."""
    def draw(self) -> str: ...


@runtime_checkable  
class Serializable(Protocol):
    """Protocol for serializable objects."""
    def to_dict(self) -> dict: ...
    def to_json(self) -> str: ...


# Duck-typed implementations - no inheritance required
@dataclass
class Circle:
    """Circle that can be drawn."""
    radius: float
    center: tuple
    
    def draw(self) -> str:
        return f"Circle(r={self.radius}, center={self.center})"


@dataclass
class Rectangle:
    """Rectangle that can be drawn."""
    width: float
    height: float
    
    def draw(self) -> str:
        return f"Rectangle({self.width}x{self.height})"


@dataclass
class Person:
    """Person with serialization."""
    name: str
    age: int
    
    def to_dict(self) -> dict:
        return {"name": self.name, "age": self.age}
    
    def to_json(self) -> str:
        import json
        return json.dumps(self.to_dict())


class DuckTypedProcessor:
    """Processor that works with any compatible type."""
    
    @staticmethod
    def draw_all(objects: list) -> list:
        """Draw any objects with draw() method."""
        results = []
        for obj in objects:
            # Duck typing: if it quacks like a drawable...
            if hasattr(obj, 'draw') and callable(obj.draw):
                results.append(obj.draw())
            else:
                results.append(f"[Cannot draw {type(obj).__name__}]")
        return results
    
    @staticmethod
    def serialize_all(objects: list) -> list:
        """Serialize any serializable objects."""
        results = []
        for obj in objects:
            if isinstance(obj, Serializable):
                results.append(obj.to_dict())
            elif hasattr(obj, '__dict__'):
                results.append(obj.__dict__)
            else:
                results.append({"value": str(obj)})
        return results


class FlexibleContainer:
    """Container that accepts any compatible type."""
    
    def __init__(self):
        self._items: list = []
    
    def add(self, item: Any) -> 'FlexibleContainer':
        """Add any item."""
        self._items.append(item)
        return self
    
    def get_drawables(self) -> list:
        """Filter and return drawable items."""
        return [item for item in self._items 
                if hasattr(item, 'draw') and callable(item.draw)]
    
    def get_serializables(self) -> list:
        """Filter and return serializable items."""
        return [item for item in self._items
                if isinstance(item, Serializable)]
    
    def process_with(self, processor_func) -> list:
        """Apply any processor to all items."""
        return [processor_func(item) for item in self._items]


# EAFP pattern (Easier to Ask Forgiveness than Permission)
class SafeAccessor:
    """Safe attribute access using EAFP."""
    
    @staticmethod
    def get_length(obj: Any) -> int:
        """Get length if object supports it."""
        try:
            return len(obj)
        except TypeError:
            try:
                return len(obj.__dict__)
            except AttributeError:
                return 0
    
    @staticmethod
    def get_value(obj: Any, key: str, default: Any = None) -> Any:
        """Get value from dict-like or object."""
        # Try dict-style access
        try:
            return obj[key]
        except (TypeError, KeyError):
            pass
        
        # Try attribute access
        try:
            return getattr(obj, key)
        except AttributeError:
            pass
        
        return default


# Usage
shapes = [Circle(5, (0, 0)), Rectangle(10, 20), "not a shape"]
processor = DuckTypedProcessor()
print(processor.draw_all(shapes))
# ['Circle(r=5, center=(0, 0))', 'Rectangle(10x20)', '[Cannot draw str]']

container = FlexibleContainer()
container.add(Circle(3, (1, 1))).add(Person("Alice", 30)).add(Rectangle(5, 5))

print(len(container.get_drawables()))  # 2
print(len(container.get_serializables()))  # 1
```
**Why**: Pythonic approach enabling polymorphism without inheritance hierarchies.
