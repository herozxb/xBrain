## js_indexeddb_put
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>IndexedDB Put Operation</title>
</head>
<body>
    <h2>IndexedDB Put - Update or Add Records</h2>
    <input type="text" id="keyInput" placeholder="Enter key">
    <input type="text" id="valueInput" placeholder="Enter value">
    <button onclick="putData()">Put Data</button>
    <p id="status"></p>

    <script>
        // PROBLEM: Update existing record or add new record to IndexedDB
        // TIME: O(1) SPACE: O(1) for operation, O(n) for storage
        
        let db;
        
        // Open or create the database
        const request = indexedDB.open('MyDatabase', 1);
        
        // Create object store on first run or version upgrade
        request.onupgradeneeded = (event) => {
            db = event.target.result;
            // Create an object store named 'store' with 'id' as key path
            if (!db.objectStoreNames.contains('store')) {
                db.createObjectStore('store', { keyPath: 'id' });
            }
        };
        
        // Successfully opened database
        request.onsuccess = (event) => {
            db = event.target.result;
            document.getElementById('status').textContent = 'Database ready';
        };
        
        // Handle errors during database opening
        request.onerror = (event) => {
            document.getElementById('status').textContent = 'Database error: ' + event.target.error;
        };
        
        /**
         * Put data into IndexedDB - updates if exists, adds if new
         * The put() method is idempotent and handles both add and update
         */
        function putData() {
            const key = document.getElementById('keyInput').value;
            const value = document.getElementById('valueInput').value;
            
            if (!key || !value) {
                document.getElementById('status').textContent = 'Please enter both key and value';
                return;
            }
            
            // Start a readwrite transaction
            const transaction = db.transaction(['store'], 'readwrite');
            const store = transaction.objectStore('store');
            
            // Create the data object with id as key
            const data = {
                id: key,
                value: value,
                timestamp: Date.now()
            };
            
            // Put operation - will update if key exists, add if new
            const putRequest = store.put(data);
            
            putRequest.onsuccess = () => {
                document.getElementById('status').textContent = 
                    `Data "${key}" saved successfully`;
            };
            
            putRequest.onerror = (event) => {
                document.getElementById('status').textContent = 
                    'Error: ' + event.target.error;
            };
        }
    </script>
</body>
</html>
```
**Why:** `put()` is the primary method for writing to IndexedDB, handling both insert and update in one operation for efficient data persistence.

    <div id="output"></div>

    <script>
        // PROBLEM: Access and iterate HTTP response headers from fetch
        // TIME: O(n) where n = number of headers SPACE: O(n)
        
        /**
         * Fetch a resource and display all response headers
         * Headers are accessed via the Headers interface which provides
         * methods like get(), has(), entries(), keys(), values()
         */
        async function fetchHeaders() {
            const output = document.getElementById('output');
            output.innerHTML = '<p>Loading...</p>';
            
            try {
                // Fetch with CORS - many headers require CORS exposure
                const response = await fetch('https://httpbin.org/headers', {
                    method: 'GET',
                    headers: {
                        'Accept': 'application/json'
                    }
                });
                
                // Check if response is ok
                if (!response.ok) {
                    throw new Error(`HTTP ${response.status}: ${response.statusText}`);
                }
                
                // Get the Headers object from response
                const headers = response.headers;
                
                // Build output string
                let html = '<h3>Response Headers:</h3><ul>';
                
                // Method 1: Iterate using entries() - returns [key, value] pairs
                for (const [key, value] of headers.entries()) {
                    html += `<li><strong>${key}:</strong> ${value}</li>`;
                }
                
                html += '</ul>';
                
                // Method 2: Get specific header by name
                const contentType = headers.get('Content-Type');
                html += `<p><em>Content-Type via get():</em> ${contentType}</p>`;
                
                // Method 3: Check if header exists
                html += `<p><em>Has 'date' header:</em> ${headers.has('date')}</p>`;
                
                output.innerHTML = html;
                
            } catch (error) {
                output.innerHTML = `<p style="color: red;">Error: ${error.message}</p>`;
            }
        }
    </script>
</body>
</html>
```
**Why:** Response headers provide critical metadata (content type, caching, CORS) needed to properly handle fetched resources.

            const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            
            if (value === '') {
                event.target.classList.remove('valid', 'invalid');
                emailError.textContent = '';
            } else if (emailRegex.test(value)) {
                event.target.classList.remove('invalid');
                event.target.classList.add('valid');
                emailError.textContent = '✓ Valid';
                emailError.style.color = 'green';
            } else {
                event.target.classList.remove('valid');
                event.target.classList.add('invalid');
                emailError.textContent = '✗ Invalid email format';
                emailError.style.color = 'red';
            }
        });
        
        // Blur event for username - minimum length validation
        usernameInput.addEventListener('blur', function(event) {
            const value = event.target.value.trim();
            
            if (value === '') {
                event.target.classList.remove('valid', 'invalid');
                usernameError.textContent = '';
            } else if (value.length >= 3) {
                event.target.classList.remove('invalid');
                event.target.classList.add('valid');
                usernameError.textContent = '✓ Valid';
                usernameError.style.color = 'green';

                event.target.classList.remove('valid');
                event.target.classList.add('invalid');

            }

    </div>

        <p>Link Zone<br><small>(Creates references)</small></p>

        // PROBLEM: Define what type of drop operation a zone accepts
        // TIME: O(1) per event SPACE: O(1)
        
        const draggables = document.querySelectorAll('.draggable');
        const dropzones = document.querySelectorAll('.dropzone');
        let draggedElement = null;
        
        // Store original parent for move operations
        let originalParent = null;
        
        // Configure draggable elements
        draggables.forEach(item => {
            item.addEventListener('dragstart', (e) => {
                draggedElement = e.target;
                originalParent = e.target.parentElement;
                e.dataTransfer.setData('text/plain', e.target.id);
                e.dataTransfer.effectAllowed = 'all';
                e.target.style.opacity = '0.5';
            });
            
            item.addEventListener('dragend', (e) => {
                e.target.style.opacity = '1';
                draggedElement = null;
            });
        });
        
        // Configure drop zones
        dropzones.forEach(zone => {
            zone.addEventListener('dragover', (e) => {
                e.preventDefault();
                // Set drop effect based on dropzone type
                const dropzoneType = zone.dataset.dropzone;
                e.dataTransfer.dropEffect = dropzoneType === 'copy' ? 'copy' : 
                                            dropzoneType === 'move' ? 'move' : 'link';
                zone.classList.add('dragover');
            });
            
            zone.addEventListener('dragleave', (e) => {
                zone.classList.remove('dragover');
            });
            
            zone.addEventListener('drop', (e) => {
                e.preventDefault();
                zone.classList.remove('dragover');
                
                const id = e.dataTransfer.getData('text/plain');
                const dropzoneType = zone.dataset.dropzone;
                const droppedItems = zone.querySelector('.dropped-items');
                
                if (dropzoneType === 'copy') {
                    // Create a copy of the element
                    const clone = document.getElementById(id).cloneNode(true);
                    clone.id = id + '-copy-' + Date.now();
                    clone.style.opacity = '1';
                    droppedItems.appendChild(clone);
                    console.log('Copied item to zone');
                    
                } else if (dropzoneType === 'move') {
                    // Move the original element
                    draggedElement.style.opacity = '1';
                    droppedItems.appendChild(draggedElement);
                    console.log('Moved item to zone');
                    
                } else if (dropzoneType === 'link') {
                    // Create a reference/link indicator
                    const link = document.createElement('span');
                    link.textContent = `[→ ${id}]`;
                    link.style.cssText = 'background:#ffc107;padding:5px;margin:2px;border-radius:3px;display:inline-block;';
                    droppedItems.appendChild(link);
                    console.log('Created link reference');
                }
            });
        });
    </script>
</body>
</html>

         * Safe URI decoding with error handling
         */

            output.innerHTML = '<h3>Decoding Test:</h3>';

                const decoded = decodeURIComponent(input);

            }
        }

    <script>
        // PROBLEM: Split string using regex patterns for flexible tokenization
        // TIME: O(n) where n = string length SPACE: O(n) for result array
        
        const output = document.getElementById('output');
        
        /**
         * RegExp.split() splits string by regex pattern
         * More powerful than string split - handles multiple delimiters
         */
        function performSplit() {
            const text = document.getElementById('inputText').value;
            const patternStr = document.getElementById('pattern').value;
            
            let html = '<h3>Results:</h3>';
            
            try {
                // Create regex from user input
                const regex = new RegExp(patternStr);
                
                // Split using the regex pattern
                const parts = text.split(regex);
                
                // Filter out empty strings (common with split)
                const filtered = parts.filter(p => p.length > 0);
                
                html += `<p><strong>Pattern:</strong> /${patternStr}/</p>`;
                html += `<p><strong>Raw split result (${parts.length} items):</strong></p>`;
                html += '<ol>';
                parts.forEach((part, i) => {
                    const display = part === '' ? '<em>(empty)</em>' : part;
                    html += `<li>${display}</li>`;
                });
                html += '</ol>';
                
                html += `<p><strong>Filtered result (${filtered.length} items):</strong></p>`;
                html += '<ol>';
                filtered.forEach(part => {
                    html += `<li>${part}</li>`;
                });
                html += '</ol>';
                
            } catch (error) {
                html += `<p style="color: red;">Invalid regex: ${error.message}</p>`;
            }
            
            output.innerHTML = html;
        }
        
        /**
         * Demonstrate various split patterns
         */
        function demonstratePatterns() {
            const examples = [
                {
                    text: 'one,two;three|four',
                    pattern: '[,;|]',
                    desc: 'Multiple single-character delimiters'
                },
                {
                    text: 'hello123world456test',
                    pattern: '\\d+',
                    desc: 'Split by numbers'
                },
                {
                    text: '  extra   spaces   here  ',
                    pattern: '\\s+',
                    desc: 'Split by whitespace'
                },
                {
                    text: '<div>content</div><span>more</span>',
                    pattern: '<[^>]+>',
                    desc: 'Split by HTML tags'
                },
                {
                    text: 'word1-ABC-word2-DEF-word3',
                    pattern: '-[A-Z]+-',
                    desc: 'Split by pattern with capturing'
                }
            ];
            
            let html = '<h3>Pattern Examples:</h3>';
            
            examples.forEach(ex => {
                const regex = new RegExp(ex.pattern);
                const parts = ex.text.split(regex).filter(p => p);
                html += `<p><strong>${ex.desc}</strong><br>`;
                html += `Text: <code>"${ex.text}"</code><br>`;
                html += `Pattern: <code>/${ex.pattern}/</code><br>`;
                html += `Result: [${parts.map(p => `"${p}"`).join(', ')}]</p>`;
            });
            
            output.innerHTML = html;
        }
        
        // Run demo on load
        demonstratePatterns();
    </script>
</body>
</html>
```
**Why:** Regex-powered split enables complex tokenization patterns impossible with simple string delimiters, essential for text parsing.

    <script>
        // PROBLEM: Control which properties are serialized and how
        // TIME: O(n) where n = object depth SPACE: O(n)
        
        const data = {
            name: 'John Doe',
            email: 'john@example.com',
            password: 'secret123',
            ssn: '123-45-6789',
            age: 30,
            createdAt: new Date(),
            method: function() { return 'test'; }
        };
        
        // ========== 1. Array Replacer (Whitelist) ==========
        const arrayReplacerDiv = document.getElementById('arrayReplacer');
        
        // Only include specific properties
        const whitelist = ['name', 'email', 'age'];
        const whitelisted = JSON.stringify(data, whitelist, 2);
        
        arrayReplacerDiv.innerHTML = `
            <p class="note">Array replacer acts as a whitelist of property names to include:</p>
            <pre>const whitelist = ['name', 'email', 'age'];

        `;
        
        // ========== 2. Function Replacer (Transform) ==========
        const functionReplacerDiv = document.getElementById('functionReplacer');
        
        /**
         * Replacer function receives (key, value) for each property
         * Return undefined to exclude the property
         * Return modified value to transform it
         */
        function secureReplacer(key, value) {
            // Hide sensitive fields
            const sensitiveFields = ['password', 'ssn', 'token', 'secret'];
            if (sensitiveFields.includes(key)) {
                return '***REDACTED***';
            }
            
            // Convert Date to ISO string
            if (value instanceof Date) {
                return value.toISOString();
            }
            
            // Skip functions
            if (typeof value === 'function') {
                return undefined;
            }
            
            return value; // Keep all other values as-is
        }
        
        const secured = JSON.stringify(data, secureReplacer, 2);
        
        functionReplacerDiv.innerHTML = `
            <p class="note">Function replacer can filter and transform values:</p>
            <pre>function secureReplacer(key, value) {
    if (['password', 'ssn'].includes(key)) return '***REDACTED***';
    if (value instanceof Date) return value.toISOString();
    if (typeof value === 'function') return undefined;
    return value;
}</pre>

                    if (seen.has(value)) {
                        return '[Circular]';

                    seen.add(value);

            if (value === undefined) return 'undefined';
            if (Number.isNaN(value)) return 'NaN';
            if (value === Infinity) return 'Infinity';
            if (value === -0) return '-0';
            return value;
        };

        examples += `<pre>${JSON.stringify(debugData, debugReplacer, 2)}</pre>`;
        
        // Example 4: Type preservation
        examples += '<h4>Type Preservation:</h4>';
        const typePreservingReplacer = (key, value) => {
            if (value !== null && typeof value === 'object') {
                return { ...value, __type: value.constructor.name };
            }
            return value;
        };
        
        const typedData = { date: new Date(), regex: /test/gi };
