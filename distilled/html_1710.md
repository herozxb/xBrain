```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Get Cookies</title>
</head>
<body>
    <button id="showCookies">Show All Cookies</button>
    <button id="getSpecific">Get 'username' Cookie</button>
    <div id="output"></div>

    <script>
        // PROBLEM: Read and parse cookies from document.cookie string
        // TIME: O(n) where n = number of cookies SPACE: O(n)
        
        const output = document.getElementById('output');

        // Function to get all cookies as an object
        function getAllCookies() {
            const cookies = document.cookie;
            
            if (!cookies) {
                return {};
            }
            
            // Parse cookie string into key-value pairs
            return cookies.split(';').reduce((acc, cookie) => {
                const [key, value] = cookie.trim().split('=');
                acc[key] = decodeURIComponent(value);
                return acc;
            }, {});
        }

        // Function to get a specific cookie by name
        function getCookie(name) {
            const cookies = getAllCookies();
            return cookies[name] || null;
        }

        // Display all cookies
        document.getElementById('showCookies').addEventListener('click', function() {
            const cookies = getAllCookies();
            const cookieList = Object.entries(cookies)
                .map(([key, value]) => `<li><strong>${key}:</strong> ${value}</li>`)
                .join('');
            
            output.innerHTML = `<h3>All Cookies:</h3><ul>${cookieList || '<li>No cookies found</li>'}</ul>`;
        });

        // Get specific cookie
        document.getElementById('getSpecific').addEventListener('click', function() {
            const username = getCookie('username');
            
            if (username) {
                output.innerHTML = `<p>Username cookie value: <strong>${username}</strong></p>`;
            } else {
                output.innerHTML = '<p style="color: red;">"username" cookie not found. Try setting one first!</p>';
            }
        });

        // Set a test cookie for demonstration

        console.log('Cookies:', getAllCookies());
    </script>

        function createDatabase() {
            return new Promise((resolve, reject) => {
                // Open or create database named 'MyAppDB' with version 1
                const request = indexedDB.open('MyAppDB', 1);
                
                // Error handling
                request.onerror = () => reject(request.error);
                request.onsuccess = () => resolve(request.result);
                
                // This event fires when database is being created or version is changing
                request.onupgradeneeded = function(event) {
                    const db = event.target.result;
                    
                    // Create object store for users
                    if (!db.objectStoreNames.contains('users')) {
                        const usersStore = db.createObjectStore('users', {
                            keyPath: 'id',      // Primary key
                            autoIncrement: true // Auto-generate IDs
                        });
                        
                        // Create indexes for efficient querying
                        usersStore.createIndex('email', 'email', { unique: true });
                        usersStore.createIndex('username', 'username', { unique: true });
                        usersStore.createIndex('age', 'age', { unique: false });
                        
                        console.log('Object store "users" created with indexes');
                    }
                    
                    // Create object store for products
                    if (!db.objectStoreNames.contains('products')) {
                        const productsStore = db.createObjectStore('products', {
                            keyPath: 'sku'  // Use SKU as primary key
                        });
                        
                        productsStore.createIndex('category', 'category', { unique: false });
                        productsStore.createIndex('price', 'price', { unique: false });
                        
                        console.log('Object store "products" created');
                    }
                    
                    // Create object store with compound key
                    if (!db.objectStoreNames.contains('orders')) {
                        const ordersStore = db.createObjectStore('orders', {
                            keyPath: ['userId', 'orderId']
                        });
                        
                        ordersStore.createIndex('status', 'status', { unique: false });
                        ordersStore.createIndex('date', 'date', { unique: false });
                    }
                    
                    output.innerHTML += '<p>Object stores created: users, products, orders</p>';

        }

        const output = document.getElementById('output');

            // Create new AbortController for each request
            abortController = new AbortController();
            const signal = abortController.signal;
            
            // Enable cancel button
            document.getElementById('cancelFetch').disabled = false;
            output.innerHTML = '<p>Fetch started... (simulating 5 second delay)</p>';
            
            try {
                // Pass signal to fetch options
                const response = await fetch('https://httpbin.org/delay/5', {
                    signal: signal  // Attach abort signal
                });
                
                const data = await response.json();
                output.innerHTML += '<p style="color: green;">Fetch completed successfully!</p>';
                console.log('Data:', data);
                
            } catch (error) {
                // Check if request was aborted
                if (error.name === 'AbortError') {
                    output.innerHTML += '<p style="color: orange;">Fetch was cancelled by user</p>';
                } else {
                    output.innerHTML += `<p style="color: red;">Fetch error: ${error.message}</p>`;
                }
            } finally {
                // Reset UI
                document.getElementById('cancelFetch').disabled = true;
                abortController = null;
            }
        });

        document.getElementById('cancelFetch').addEventListener('click', function() {
            if (abortController) {
                // Abort the fetch request
                abortController.abort();
                output.innerHTML += '<p>Cancellation requested...</p>';
            }
        });

        // AbortController can abort multiple fetch requests
        async function fetchMultipleWithAbort() {
            const controller = new AbortController();
            
            // All these requests share the same abort signal
            const requests = [
                fetch('https://httpbin.org/delay/3', { signal: controller.signal }),
                fetch('https://httpbin.org/delay/2', { signal: controller.signal }),
                fetch('https://httpbin.org/delay/1', { signal: controller.signal })
            ];
            
            // Cancel all after 1.5 seconds
            setTimeout(() => {
                controller.abort();
                console.log('All requests cancelled');
            }, 1500);
            
            try {
                const responses = await Promise.all(requests);
                console.log('All completed');
            } catch (error) {
                if (error.name === 'AbortError') {
                    console.log('Requests aborted');
                }
            }
        }

        // Check if signal is already aborted
        const signal = new AbortController().signal;
        console.log('Signal aborted:', signal.aborted); // false
    </script>
</body>
</html>
```

**Why**: `AbortController` provides a clean mechanism to cancel in-flight network requests, preventing resource waste and improving user experience when requests become obsolete due to navigation or user actions.

    <script>
        // PROBLEM: Programmatically validate form inputs using Constraint Validation API
        // TIME: O(1) SPACE: O(1)
        
        const form = document.getElementById('myForm');
        const emailInput = document.getElementById('email');
        const ageInput = document.getElementById('age');
        const passwordInput = document.getElementById('password');
        const output = document.getElementById('output');

        // Validate a single field and display custom error
        function validateField(input, errorElement) {
            const validity = input.validity;
            let errorMessage = '';
            
            // Check various validation states
            if (validity.valueMissing) {
                errorMessage = 'This field is required';
            } else if (validity.typeMismatch && input.type === 'email') {
                errorMessage = 'Please enter a valid email address';
            } else if (validity.rangeUnderflow) {
                errorMessage = `Value must be at least ${input.min}`;
            } else if (validity.rangeOverflow) {
                errorMessage = `Value must be no more than ${input.max}`;
            } else if (validity.tooShort) {
                errorMessage = `Must be at least ${input.minLength} characters`;
            } else if (validity.patternMismatch) {
                errorMessage = 'Please match the requested format';
            }
            
            // Display error message
            errorElement.textContent = errorMessage;
            
            // Check if input is valid
            if (input.checkValidity()) {
                input.setCustomValidity(''); // Clear any custom error
                return true;
            }
            
            return false;
        }

        // Add real-time validation
        emailInput.addEventListener('blur', function() {
            validateField(this, document.getElementById('emailError'));
        });

        ageInput.addEventListener('blur', function() {
            validateField(this, document.getElementById('ageError'));
        });

        passwordInput.addEventListener('input', function() {
            validateField(this, document.getElementById('passwordError'));
        });

        // Form submission with validation
        form.addEventListener('submit', function(event) {
            event.preventDefault();
            
            let isValid = true;
            
            // Validate all fields
            if (!validateField(emailInput, document.getElementById('emailError'))) {
                isValid = false;
            }
            
            if (!validateField(ageInput, document.getElementById('ageError'))) {
                isValid = false;
            }
            
            if (!validateField(passwordInput, document.getElementById('passwordError'))) {
                isValid = false;
            }
            
            // Custom validation logic
            if (passwordInput.value.includes('password')) {
                passwordInput.setCustomValidity('Password cannot contain "password"');
                validateField(passwordInput, document.getElementById('passwordError'));
                isValid = false;
            }
            
            if (isValid) {
                output.innerHTML = '<p style="color: green;">Form is valid! Submitting...</p>';
                console.log('Form data:', {
                    email: emailInput.value,
                    age: ageInput.value,
                    password: passwordInput.value
                });
            } else {
                output.innerHTML = '<p style="color: red;">Form has errors. Please fix them.</p>';
            }
        });

        // Access validity properties
        console.log('Validity properties:', {
            valueMissing: emailInput.validity.valueMissing,
            typeMismatch: emailInput.validity.typeMismatch,
            patternMismatch: emailInput.validity.patternMismatch,
            tooShort: emailInput.validity.tooShort,
            rangeUnderflow: ageInput.validity.rangeUnderflow,
            valid: emailInput.validity.valid
        });
    </script>
</body>
</html>
```

**Why**: The Constraint Validation API provides programmatic access to HTML5 form validation, enabling custom error messages, real-time validation feedback, and complex validation rules beyond basic HTML attributes.

        // PROBLEM: Handle form submission with validation and async processing
        // TIME: O(1) SPACE: O(1)
        
        const form = document.getElementById('userForm');
        const output = document.getElementById('output');

        // Modern submit event handler
        form.addEventListener('submit', async function(event) {
            // Always prevent default form submission
            event.preventDefault();
            
            // Get form data easily
            const formData = new FormData(this);
            const data = Object.fromEntries(formData.entries());
            
            console.log('Form submitted:', data);
            
            // Access the submitter (which button was clicked)
            const submitter = event.submitter;
            console.log('Submitted by:', submitter?.textContent);
            
            // Validate data
            if (!data.username || data.username.length < 2) {

                return;
            }
            
            // Show loading state
            const originalText = submitter.textContent;
            submitter.textContent = 'Submitting...';
            submitter.disabled = true;
            
            try {
                // Simulate API call
                await submitToServer(data);
                
                output.innerHTML = '<p style="color: green;">Form submitted successfully!</p>';
                
                // Reset form on success
                this.reset();
                
            } catch (error) {
                output.innerHTML = `<p style="color: red;">Error: ${error.message}</p>`;
            } finally {
                // Restore button state
                submitter.textContent = originalText;
                submitter.disabled = false;
            }
        });

        // Simulate server submission
        function submitToServer(data) {
            return new Promise((resolve, reject) => {
                setTimeout(() => {
                    // Simulate success (90% chance)
                    if (Math.random() > 0.1) {
                        resolve({ success: true, id: Date.now() });
                    } else {
                        reject(new Error('Server error'));
                    }
                }, 1000);
            });
        }

        // Alternative: Using FormData directly
        form.addEventListener('formdata', function(event) {
            // This event fires when form creates FormData
            const formData = event.formData;
            
            // Append additional data
            formData.append('timestamp', Date.now());
            formData.append('userAgent', navigator.userAgent);
            
            console.log('FormData created:', [...formData.entries()]);
        });

        // Manual form submission
        function manualSubmit() {
            // Trigger submit event programmatically
            form.dispatchEvent(new Event('submit', {
                bubbles: true,
                cancelable: true
            }));
        }

        // Check form validity before submission
        if (form.checkValidity()) {
            console.log('Form is valid');
        }
    </script>
</body>
</html>
```

**Why**: The `submit` event is the standard way to intercept form submissions, enabling client-side validation, async data processing, and progressive enhancement while maintaining form accessibility and SEO benefits.

        // TIME: O(n) where n = number of fields SPACE: O(n)
        
        const form = document.getElementById('uploadForm');
        const output = document.getElementById('output');

        form.addEventListener('submit', async function(event) {
            event.preventDefault();
            
            // Create FormData from form element
            const formData = new FormData(this);
            
            // Get individual values
            console.log('Username:', formData.get('username'));
            console.log('Email:', formData.get('email'));
            
            // Get all values for a field (e.g., multiple select)
            console.log('Tags:', formData.getAll('tags'));
            
            // Check if field exists
            console.log('Has avatar:', formData.has('avatar'));
            
            // Set/update a value
            formData.set('timestamp', Date.now());
            
            // Append additional value (doesn't overwrite)
            formData.append('tags', 'nodejs');
            
            // Delete a field
            // formData.delete('email');
            
            // Iterate over all entries
            console.log('--- All FormData entries ---');
            for (const [key, value] of formData.entries()) {
                if (value instanceof File) {
                    console.log(`${key}: File(${value.name}, ${value.size} bytes, ${value.type})`);
                } else {
                    console.log(`${key}: ${value}`);
                }
            }
            
            // Convert to object (useful for JSON APIs)
            const dataObject = Object.fromEntries(formData.entries());
            console.log('As object:', dataObject);

            try {
                const response = await fetch('https://httpbin.org/post', {
                    method: 'POST',
                    body: formData  // Browser sets Content-Type automatically with boundary
                });
                
                const result = await response.json();
                output.innerHTML = '<p style="color: green;">Upload successful!</p>';
                console.log('Server response:', result);
                
            } catch (error) {
                output.innerHTML = `<p style="color: red;">Error: ${error.message}</p>`;
