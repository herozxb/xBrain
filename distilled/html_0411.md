                    isValid = false;
                    // highlight invalid field

                alert('Please fix the validation errors');

                    this.style.borderColor = 'red';
                }
            });
        });

</html>

        // JavaScript to manage custom radio button behavior
        const radiogroup = document.querySelector('[role="radiogroup"]');
        const radios = radiogroup.querySelectorAll('[role="radio"]');
        
        // Handle radio selection
        function selectRadio(radio) {
            // Deselect all radios in group
            radios.forEach(r => {
                r.setAttribute('aria-checked', 'false');
                r.setAttribute('tabindex', '-1');
            });

            radio.setAttribute('aria-checked', 'true');

            );

            switch(e.key) {
                case 'ArrowDown':
                case 'ArrowRight':
                    newIndex = (currentIndex + 1) % radios.length;
                    e.preventDefault();
                    break;
                case 'ArrowUp':
                case 'ArrowLeft':
                    newIndex = (currentIndex - 1 + radios.length) % radios.length;
                    e.preventDefault();
                    break;
                case ' ':
                case 'Enter':
                    // Space/Enter selects focused radio
                    selectRadio(document.activeElement);
                    e.preventDefault();
                    break;
            }
            
            if (newIndex !== currentIndex) {
                selectRadio(radios[newIndex]);
            }
        });
    </script>

</html>

**Why**: ARIA radio role allows creating custom-styled radio buttons while maintaining accessibility for screen readers and keyboard navigation.

            const ogData = {};
            
            // Get all meta tags with property attribute
            const metaTags = document.querySelectorAll('meta[property^="og:"]');
            
            metaTags.forEach(meta => {
                const property = meta.getAttribute('property').replace('og:', '');
                const content = meta.getAttribute('content');
                
                // Handle multiple values (like article:tag)
                if (ogData[property]) {
                    if (Array.isArray(ogData[property])) {
                        ogData[property].push(content);
                    } else {
                        ogData[property] = [ogData[property], content];
                    }
                } else {
                    ogData[property] = content;
                }
            });
            
            return ogData;
        }
        
        // Log current Open Graph configuration
        console.log('Open Graph Data:', getOpenGraphData());
        
        // Validate required OG tags
        const requiredTags = ['og:title', 'og:description', 'og:image', 'og:url'];
        requiredTags.forEach(tag => {
            const meta = document.querySelector(`meta[property="${tag}"]`);
            if (!meta || !meta.getAttribute('content')) {
                console.warn(`Missing or empty: ${tag}`);
            } else {
                console.log(`✓ ${tag}: ${meta.getAttribute('content')}`);
            }
        });
    </script>

        const prefetchLinks = document.querySelectorAll('link[rel="prefetch"]');
        
        // Check if resource timing API is available
        if (window.performance && performance.getEntriesByType) {
            function checkPrefetchStatus() {
                const entries = performance.getEntriesByType('resource');
                
                prefetchLinks.forEach(link => {
                    const href = link.href;
                    const entry = entries.find(e => e.name === href);
                    
                    if (entry) {
                        console.log(`✓ Prefetched: ${href}`);
                        console.log(`  Duration: ${entry.duration.toFixed(2)}ms`);
                        console.log(`  Size: ${entry.transferSize} bytes`);
                    } else {
                        console.log(`⏳ Pending: ${href}`);
                    }
                });
            }

            window.addEventListener('load', () => {
                setTimeout(checkPrefetchStatus, 1000);
            });
        }
        
        // Dynamically add prefetch hints
        function addPrefetch(url, as = 'document') {
            const link = document.createElement('link');
            link.rel = 'prefetch';
            link.href = url;
            link.as = as;
            document.head.appendChild(link);
            console.log(`Added prefetch for: ${url}`);

        // Predictive prefetching based on user behavior
        let hoverTimer;

        // JavaScript to dynamically change align-self
        function changeAlignSelf(element, value) {
            // Valid values: flex-start, flex-end, center, stretch, baseline, auto
            const validValues = [
                'flex-start', 'flex-end', 'center', 
                'stretch', 'baseline', 'auto'
            ];
            
            if (validValues.includes(value)) {

<!-- PROBLEM: Wait for all promises regardless of success or failure -->
<!-- TIME: O(n) SPACE: O(n) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Promise.allSettled()</title>
    <style>
        .result { margin: 10px 0; padding: 10px; border-radius: 5px; }
        .fulfilled { background: #d4edda; color: #155724; }
        .rejected { background: #f8d7da; color: #721c24; }
    </style>
</head>
<body>
    <h2>Promise.allSettled() Demo</h2>
    <button id="runDemo">Run Demo</button>
    <div id="results"></div>
    
    <script>
        // Promise.allSettled waits for ALL promises to complete
        // Unlike Promise.all, it doesn't fail-fast
        
        // Simulate API calls with different outcomes
        function fetchUser(id) {
            return new Promise((resolve, reject) => {
                setTimeout(() => {
                    if (id === 2) {
                        reject(new Error(`User ${id} not found`));
                    } else {
                        resolve({ id, name: `User ${id}` });
                    }
                }, Math.random() * 1000);
            });
        }
        
        function fetchData(url) {
            return new Promise((resolve, reject) => {
                setTimeout(() => {
                    if (url.includes('error')) {
                        reject(new Error(`Failed to fetch ${url}`));
                    } else {
                        resolve({ url, data: 'Some data' });
                    }
                }, Math.random() * 1000);
            });
        }
        
        document.getElementById('runDemo').addEventListener('click', async () => {
            const resultsDiv = document.getElementById('results');
            resultsDiv.innerHTML = '<p>Loading...</p>';
            
            // Create multiple promises with mixed outcomes
            const promises = [
                fetchUser(1),           // Will resolve
                fetchUser(2),           // Will reject
                fetchUser(3),           // Will resolve
                fetchData('/api/data'), // Will resolve
                fetchData('/api/error') // Will reject
            ];
            
            try {
                // allSettled returns array of {status, value/reason}
                const results = await Promise.allSettled(promises);
                
                resultsDiv.innerHTML = '';
                
                results.forEach((result, index) => {
                    const div = document.createElement('div');
                    div.className = `result ${result.status}`;
                    
                    if (result.status === 'fulfilled') {
                        div.innerHTML = `
                            <strong>Promise ${index + 1} - Fulfilled</strong><br>
                            Value: ${JSON.stringify(result.value)}
                        `;
                    } else {
                        div.innerHTML = `
                            <strong>Promise ${index + 1} - Rejected</strong><br>
                            Reason: ${result.reason.message}
                        `;
                    }
                    
                    resultsDiv.appendChild(div);
                });
                
                // Process results based on status
                const successful = results
                    .filter(r => r.status === 'fulfilled')
                    .map(r => r.value);
                    
                const failed = results
                    .filter(r => r.status === 'rejected')
                    .map(r => r.reason);
                
                console.log('Successful:', successful);
                console.log('Failed:', failed);
                
            } catch (error) {
                // Note: allSettled itself rarely rejects (only on invalid input)
                console.error('Unexpected error:', error);
            }
        });
        
        // Comparison with Promise.all
        async function compareAllVsAllSettled() {
            const promises = [
                Promise.resolve(1),
                Promise.reject('Error'),
                Promise.resolve(3)
            ];
            
            console.log('--- Promise.all ---');
            try {
                const all = await Promise.all(promises);
                console.log('All resolved:', all);
            } catch (e) {
                console.log('Promise.all failed:', e); // Fails fast
            }
            
            console.log('--- Promise.allSettled ---');
            const allSettled = await Promise.allSettled(promises);
            console.log('All results:', allSettled);
            // [{status:'fulfilled',value:1},{status:'rejected',reason:'Error'},{status:'fulfilled',value:3}]
        }
    </script>

**Why**: Promise.allSettled() is crucial when you need all results regardless of failures, such as batch operations where partial success is acceptable and you need to know which succeeded/failed.

    <div class="example">
        <h3>Problem: Regular Function Loses 'this'</h3>

    <script>
        // Regular function vs Arrow function 'this' binding
        
        // Example 1: Object method context
        const obj = {
            name: 'MyObject',
            
            // Regular function: 'this' depends on how it's called
            regularMethod: function() {
                console.log('Regular method - this.name:', this.name);
                
                // Problem: setTimeout creates new context
                setTimeout(function() {
                    console.log('Inside setTimeout (regular) - this:', this);
                    // 'this' is Window object, not 'obj'
                }, 100);
            },
            
            // Arrow function: 'this' is lexically bound
            arrowMethod: function() {
                console.log('Arrow method - this.name:', this.name);
                
                // Solution: Arrow function inherits 'this'
                setTimeout(() => {
                    console.log('Inside setTimeout (arrow) - this.name:', this.name);
                    // 'this' is still 'obj'
                }, 100);
            }
        };
        
        // Example 2: Event handlers
        class ButtonHandler {
            constructor(buttonId, outputId) {
                this.button = document.getElementById(buttonId);
                this.output = document.getElementById(outputId);
                this.clickCount = 0;
            }
            
            // Regular function: 'this' becomes the button element
            bindRegular() {
                this.button.addEventListener('click', function() {
                    // 'this' is the button, not ButtonHandler instance
                    this.output.textContent = `'this' is: ${this.constructor.name}`;
                    // Error: this.output is undefined!
                });
            }
            
            // Arrow function: 'this' is ButtonHandler instance
            bindArrow() {
                this.button.addEventListener('click', () => {
                    // 'this' is ButtonHandler instance
                    this.clickCount++;
                    this.output.textContent = `Click count: ${this.clickCount} (this: ${this.constructor.name})`;
                });
            }
        }
        
        // Setup handlers
        const regularHandler = new ButtonHandler('regularBtn', 'regularOutput');
        regularHandler.bindArrow(); // Using arrow to fix the issue
        
        const arrowHandler = new ButtonHandler('arrowBtn', 'arrowOutput');
        arrowHandler.bindArrow();
        
        // Example 3: Class methods comparison
        class Counter {
            constructor() {
                this.count = 0;
            }
            
            // Regular method: can be rebound
            increment() {
                this.count++;
                return this.count;
            }
            
            // Arrow property: always bound to instance
            decrement = () => {
                this.count--;
                return this.count;
            }
        }
        
        const counter = new Counter();
        
        // Regular method loses 'this' when extracted
        const inc = counter.increment;
        // inc(); // Error: 'this' is undefined
        
        // Arrow method preserves 'this'
        const dec = counter.decrement;
        console.log('Decrement:', dec()); // Works: -1

                return this.numbers.map(n => n * this.multiplier);

        };
