```html
       controls 
       width="640" 
       height="360">
  <source src="https://secure.example.com/video.mp4" type="video/mp4">
</video>

<script>
  const video = document.querySelector('video');
  
  // Handle CORS errors
  video.addEventListener('error', (e) => {
    const error = video.error;
    let message = '';
    
    switch(error.code) {
      case MediaError.MEDIA_ERR_SRC_NOT_SUPPORTED:
        message = 'Video format not supported or CORS issue';
        break;
      case MediaError.MEDIA_ERR_NETWORK:
        message = 'Network error occurred while loading video';
        break;
      case MediaError.MEDIA_ERR_DECODE:
        message = 'Video decoding failed';
        break;
      default:
        message = 'An unknown error occurred';
    }
    
    console.error('Video Error:', message);
  });
  
  // Check CORS support
  video.addEventListener('loadedmetadata', () => {
    console.log('Video loaded successfully with CORS');
  });
  
  // Draw video to canvas (requires CORS)
  const canvas = document.createElement('canvas');
  const ctx = canvas.getContext('2d');
  
  video.addEventListener('canplay', () => {
    canvas.width = video.videoWidth;
    canvas.height = video.videoHeight;
    
    try {
      ctx.drawImage(video, 0, 0);
      console.log('Video drawn to canvas successfully');
    } catch (e) {
      console.error('Cannot draw video: CORS or tainted canvas');
    }
  });
</script>
```

**Why**: Crossorigin attribute enables proper CORS handling for videos from different origins, essential for canvas manipulation and avoiding security errors.

```html
  // Monitor preload behavior
  const audioNone = document.querySelector('audio[preload="none"]');
  const audioMetadata = document.getElementById('audio-metadata');
  const audioAuto = document.getElementById('audio-auto');
  
  // Track loading progress
  audioAuto.addEventListener('progress', (e) => {
    if (audioAuto.buffered.length > 0) {
      const buffered = audioAuto.buffered.end(0);
      const duration = audioAuto.duration;
      const percent = (buffered / duration * 100).toFixed(2);
      console.log(`Preloaded: ${percent}%`);
    }
  });
  
  // Check metadata availability
  audioMetadata.addEventListener('loadedmetadata', () => {
    console.log(`Duration: ${audioMetadata.duration}s`);
    console.log(`Ready state: ${audioMetadata.readyState}`);
  });
  
  // Lazy load audio on demand
  function loadAudioOnDemand(audioElement) {
    if (audioElement.preload === 'none') {
      audioElement.load(); // Start loading
      console.log('Audio loading started');
    }
  }
  
  // Example: Load when user hovers
  audioNone.addEventListener('mouseenter', () => {
    loadAudioOnDemand(audioNone);
  });
  
  // Network-aware preload strategy
  if (navigator.connection) {
    const connection = navigator.connection;
    
    if (connection.effectiveType === 'slow-2g' || connection.saveData) {
      // Set all audio to preload="none" on slow networks
      document.querySelectorAll('audio').forEach(audio => {
        audio.preload = 'none';
      });
```

```html
<!-- PROBLEM: Sort table rows dynamically -->
<!-- TIME: O(n log n) SPACE: O(n) -->

<table id="sortable-table">
  <thead>
    <tr>
      <th data-sort="name" data-type="string">Name ▲</th>
      <th data-sort="age" data-type="number">Age ▲</th>
      <th data-sort="salary" data-type="number">Salary ▲</th>
      <th data-sort="date" data-type="date">Join Date ▲</th>
    </tr>
  </thead>
  <tbody>
    <tr>
      <td>Alice Johnson</td>
      <td>28</td>
      <td>$75,000</td>
      <td>2023-03-15</td>
    </tr>
    <tr>
      <td>Bob Smith</td>
      <td>34</td>
      <td>$92,000</td>
      <td>2022-07-22</td>
    </tr>
    <tr>
      <td>Carol Davis</td>
      <td>25</td>
      <td>$65,000</td>
      <td>2024-01-10</td>
    </tr>
    <tr>
      <td>David Wilson</td>
      <td>31</td>
      <td>$88,000</td>
      <td>2023-11-05</td>
    </tr>
  </tbody>
</table>

<script>
  class TableSorter {
    constructor(tableId) {
      this.table = document.getElementById(tableId);
      this.headers = this.table.querySelectorAll('th[data-sort]');
      this.sortDirection = {};
      
      this.init();
    }
    
    init() {
      this.headers.forEach(header => {
        header.style.cursor = 'pointer';
        header.addEventListener('click', () => this.sortTable(header));
      });
    }
    
    sortTable(header) {
      const column = header.dataset.sort;
      const type = header.dataset.type;
      const direction = this.sortDirection[column] === 'asc' ? 'desc' : 'asc';
      
      this.sortDirection[column] = direction;
      
      const tbody = this.table.querySelector('tbody');
      const rows = Array.from(tbody.querySelectorAll('tr'));
      
      // Get column index
      const columnIndex = Array.from(header.parentNode.children)
        .indexOf(header);
      
      rows.sort((a, b) => {
        let aValue = a.children[columnIndex].textContent.trim();
        let bValue = b.children[columnIndex].textContent.trim();
        
        // Parse values based on type
        switch(type) {
          case 'number':
            aValue = parseFloat(aValue.replace(/[^0-9.-]/g, ''));
            bValue = parseFloat(bValue.replace(/[^0-9.-]/g, ''));
            break;
          case 'date':
            aValue = new Date(aValue);
            bValue = new Date(bValue);
            break;
          default:
            aValue = aValue.toLowerCase();
            bValue = bValue.toLowerCase();
        }
        
        if (aValue < bValue) return direction === 'asc' ? -1 : 1;
        if (aValue > bValue) return direction === 'asc' ? 1 : -1;
        return 0;
      });
      
      // Re-append sorted rows
      rows.forEach(row => tbody.appendChild(row));
      
      // Update header indicators
      this.updateHeaderIndicators(header, direction);
    }
    
    updateHeaderIndicators(activeHeader, direction) {
      // Clear all indicators
      this.headers.forEach(h => {
        h.textContent = h.textContent.replace(/[▲▼]/g, '');
      });
      
      // Add active indicator
      const arrow = direction === 'asc' ? ' ▲' : ' ▼';
      activeHeader.textContent = activeHeader.textContent + arrow;
    }
  }
  
  // Initialize table sorter
  const sorter = new TableSorter('sortable-table');
</script>

<style>

    width: 100%;
    border-collapse: collapse;
  }
  
  #sortable-table th,
  #sortable-table td {
    padding: 12px;
    text-align: left;
    border-bottom: 1px solid #ddd;
  }
  
  #sortable-table th {
    background-color: #4CAF50;
    color: white;
    user-select: none;
  }
  
  #sortable-table th:hover {
    background-color: #45a049;
  }

  }

```

**Why**: Client-side table sorting improves UX by allowing users to organize data without server requests, using efficient O(n log n) algorithms.

```html
<script>
  class TableFilter {
    constructor(tableId) {
      this.table = document.getElementById(tableId);
      this.rows = this.table.querySelectorAll('tbody tr');
      this.searchInput = document.getElementById('search-input');
      this.columnFilter = document.getElementById('column-filter');
      this.statusFilter = document.getElementById('status-filter');
      this.resultCount = document.getElementById('result-count');
      
      this.init();
    }
    
    init() {
      // Add event listeners
      this.searchInput.addEventListener('input', () => this.filter());
      this.columnFilter.addEventListener('change', () => this.filter());
      this.statusFilter.addEventListener('change', () => this.filter());
      
      // Initial count
      this.updateResultCount();
    }
    
    filter() {
      const searchTerm = this.searchInput.value.toLowerCase();
      const columnValue = this.columnFilter.value;
      const statusValue = this.statusFilter.value;
      
      let visibleCount = 0;
      
      this.rows.forEach(row => {
        const cells = row.querySelectorAll('td');
        let showRow = true;
        
        // Apply text search filter
        if (searchTerm) {
          let found = false;
          
          if (columnValue === 'all') {
            // Search all columns
            cells.forEach(cell => {
              if (cell.textContent.toLowerCase().includes(searchTerm)) {
                found = true;
              }
            });
          } else {
            // Search specific column
            const cell = cells[parseInt(columnValue)];
            if (cell && cell.textContent.toLowerCase().includes(searchTerm)) {
              found = true;
            }
          }
          
          showRow = showRow && found;
        }
        
        // Apply status filter
        if (statusValue) {
          const statusCell = cells[3]; // Status column
          showRow = showRow && (statusCell.textContent === statusValue);
        }
        
        // Show or hide row
        row.style.display = showRow ? '' : 'none';
        if (showRow) visibleCount++;
      });
      
      this.updateResultCount(visibleCount);
    }
    
    updateResultCount(count) {
      const total = this.rows.length;
      const visible = count !== undefined ? count : total;
      this.resultCount.textContent = `Showing ${visible} of ${total} records`;
    }
    
    // Reset all filters
    reset() {
      this.searchInput.value = '';
      this.columnFilter.value = 'all';
      this.statusFilter.value = '';
      this.filter();
    }
  }
  
  // Initialize filter
  const tableFilter = new TableFilter('filterable-table');
</script>
```

```html
      this.closeBtn.addEventListener('click', () => this.close());
      this.cancelBtn.addEventListener('click', () => this.close());
      this.confirmBtn.addEventListener('click', () => this.confirm());

        if (e.key === 'Escape' && !this.overlay.hidden) {

        }
      });
      
      // Trap focus within dialog
      this.dialog.addEventListener('keydown', (e) => {
        if (e.key === 'Tab') {
          this.trapFocus(e);
        }
      });
    }
    
    open() {
      // Store current focus
      this.previousActiveElement = document.activeElement;
      
      // Show dialog
      this.overlay.hidden = false;
      this.overlay.setAttribute('aria-hidden', 'false');
      
      // Get focusable elements
      this.focusableElements = this.dialog.querySelectorAll(
        'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
      );
      
      // Set focus to first focusable element
      this.focusableElements[0].focus();
      
      // Prevent body scroll
      document.body.style.overflow = 'hidden';
    }
    
    close() {
      // Hide dialog
      this.overlay.hidden = true;
      this.overlay.setAttribute('aria-hidden', 'true');
      
      // Restore focus
      if (this.previousActiveElement) {
        this.previousActiveElement.focus();
      }
      
      // Restore body scroll
      document.body.style.overflow = '';
    }
    
    confirm() {
      console.log('Action confirmed!');
      this.close();
    }
    
    trapFocus(e) {
      const firstFocusable = this.focusableElements[0];
      const lastFocusable = this.focusableElements[this.focusableElements.length - 1];
      
      if (e.shiftKey) {
        // Shift + Tab
        if (document.activeElement === firstFocusable) {
          e.preventDefault();
          lastFocusable.focus();
        }
      } else {
        // Tab
        if (document.activeElement === lastFocusable) {
          e.preventDefault();
          firstFocusable.focus();
        }
      }
    }
```
