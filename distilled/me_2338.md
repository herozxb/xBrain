## 1. Dependency Injection Container

```python
# PROBLEM: Manage object dependencies and lifecycle automatically
# TIME: O(1) for registration, O(n) for resolution where n = dependency depth
# SPACE: O(m) where m = number of registered services
class DIContainer:
    """Lightweight dependency injection container with singleton and transient support."""
    
    def __init__(self):
        self._services = {}      # Registered service definitions
        self._instances = {}     # Singleton instances cache
        self._resolving = set()  # Circular dependency detection
    
    def register(self, name: str, factory: callable, singleton: bool = True) -> 'DIContainer':
        """Register a service with its factory function."""
        self._services[name] = {'factory': factory, 'singleton': singleton}
        return self  # Fluent interface
    
    def resolve(self, name: str) -> object:
        """Resolve a service and its dependencies recursively."""
        if name not in self._services:
            raise KeyError(f"Service '{name}' not registered")
        
        # Check for circular dependencies
        if name in self._resolving:
            raise RuntimeError(f"Circular dependency detected: {name}")
        
        # Return cached singleton if exists
        config = self._services[name]
        if config['singleton'] and name in self._instances:
            return self._instances[name]
        
        # Resolve with cycle detection
        self._resolving.add(name)
        try:
            instance = config['factory'](self)
            if config['singleton']:
                self._instances[name] = instance
            return instance
        finally:
            self._resolving.discard(name)
    
    def __getitem__(self, name: str) -> object:
        """Allow container['service'] syntax."""
        return self.resolve(name)
```
**Why**: Enables loose coupling and testable code through inversion of control.

## 2. Event Bus Pattern

```python
# PROBLEM: Decouple event publishers from subscribers with type-safe dispatch
# TIME: O(1) for publish, O(k) for subscriber notification where k = subscribers
# SPACE: O(n) where n = total registered handlers
from collections import defaultdict
from typing import Callable, Any
from dataclasses import dataclass

@dataclass
class Event:
    """Base event class for type-safe event handling."""
    type: str
    payload: Any
    timestamp: float = 0.0

class EventBus:
    """Pub/sub event bus with priority-based handler execution."""
    
    def __init__(self):
        self._handlers: dict[str, list[tuple[int, Callable]]] = defaultdict(list)
        self._middlewares: list[Callable] = []
    
    def subscribe(self, event_type: str, handler: Callable, priority: int = 0) -> None:
        """Register handler for event type with optional priority."""
        self._handlers[event_type].append((priority, handler))
        # Keep handlers sorted by priority (descending)
        self._handlers[event_type].sort(key=lambda x: -x[0])
    
    def use(self, middleware: Callable) -> 'EventBus':
        """Add middleware that processes all events."""
        self._middlewares.append(middleware)
        return self
    
    def publish(self, event: Event) -> int:
        """Dispatch event to all matching handlers. Returns handler count."""
        # Run middlewares first
        for mw in self._middlewares:
            if mw(event) is False:  # Middleware can cancel propagation
                return 0
        
        handlers = self._handlers.get(event.type, [])
        for priority, handler in handlers:
            handler(event.payload)
        
        # Also notify wildcard subscribers
        for priority, handler in self._handlers.get('*', []):
            handler(event)
        
        return len(handlers)
    
    def unsubscribe(self, event_type: str, handler: Callable) -> bool:
        """Remove specific handler. Returns True if found."""
        handlers = self._handlers.get(event_type, [])
        original_len = len(handlers)
        self._handlers[event_type] = [(p, h) for p, h in handlers if h != handler]
        return len(self._handlers[event_type]) < original_len
```
**Why**: Decouples components through event-driven architecture; supports middleware chain.

## 3. Inverted Index Implementation

```python
# PROBLEM: Build searchable index mapping terms to document locations
# TIME: O(n * m) for indexing where n=docs, m=terms; O(k) for lookup
# SPACE: O(t * d) where t=unique terms, d=avg docs per term
from collections import defaultdict
from typing import Iterator
import re

class InvertedIndex:
    """Full-text search index with positional tracking."""
    
    def __init__(self):
        self._index: dict[str, dict[int, list[int]]] = defaultdict(lambda: defaultdict(list))
        self._docs: dict[int, str] = {}
        self._next_doc_id = 0
    
    def _tokenize(self, text: str) -> list[str]:
        """Normalize and tokenize text into searchable terms."""
        return re.findall(r'\b\w+\b', text.lower())
    
    def add(self, document: str) -> int:
        """Index a document, returning its ID."""
        doc_id = self._next_doc_id
        self._next_doc_id += 1
        self._docs[doc_id] = document
        
        for position, term in enumerate(self._tokenize(document)):
            self._index[term][doc_id].append(position)
        
        return doc_id
    
    def search(self, query: str) -> list[tuple[int, list[int]]]:
        """Find documents containing all query terms with positions."""
        terms = self._tokenize(query)
        if not terms:
            return []
        
        # Intersect document sets for all terms
        doc_sets = [set(self._index[term].keys()) for term in terms]
        matching_docs = set.intersection(*doc_sets) if doc_sets else set()
        
        results = []
        for doc_id in matching_docs:
            positions = [self._index[term][doc_id] for term in terms]
            results.append((doc_id, positions))
        
        return sorted(results, key=lambda x: x[0])
    
    def phrase_search(self, phrase: str) -> list[int]:
        """Find documents where terms appear consecutively."""
        terms = self._tokenize(phrase)
        if len(terms) < 2:
            return [d for d, _ in self.search(phrase)]
        
        results = []
        candidates = self.search(phrase)
        
        for doc_id, positions_list in candidates:
            # Check if positions form consecutive sequence
            first_positions = positions_list[0]
            for start_pos in first_positions:
                if all((start_pos + i) in positions_list[i] for i in range(1, len(terms))):
                    results.append(doc_id)
                    break
        
        return results
```
**Why**: Enables O(1) term lookup for fast full-text search with phrase detection.

## 4. Middleware Pipeline Pattern

```python
# PROBLEM: Chain multiple processors in a composable request/response flow
# TIME: O(n) for pipeline execution where n = middleware count
# SPACE: O(n) for middleware stack depth
from typing import Callable, Any, Protocol

class Context:
    """Request context passed through middleware chain."""
    def __init__(self, data: dict = None):
        self.data = data or {}
        self.state: dict = {}  # Middleware can store state here
        self._halted = False
    
    def halt(self) -> None:
        """Stop pipeline execution."""
        self._halted = True

Middleware = Callable[[Context, Callable], Any]

class Pipeline:
    """Composable middleware pipeline with early exit support."""
    
    def __init__(self):
        self._middlewares: list[Middleware] = []
    
    def use(self, middleware: Middleware) -> 'Pipeline':
        """Add middleware to the pipeline."""
        self._middlewares.append(middleware)
        return self
    
    def _compose(self, middlewares: list[Middleware], final: Callable) -> Callable:
        """Compose middlewares into a single callable (recursive for clarity)."""
        if not middlewares:
            return final
        
        head, *tail = middlewares
        return lambda ctx: (
            final(ctx) if ctx._halted 
            else head(ctx, self._compose(tail, final))
        )
    
    def execute(self, ctx: Context) -> Any:
        """Run request through middleware chain."""
        def terminal(c: Context):
            return c
        
        composed = self._compose(self._middlewares, terminal)
        return composed(ctx)
    
    def __call__(self, ctx: Context) -> Any:
        """Allow pipeline(ctx) syntax."""
        return self.execute(ctx)

# Example usage
def logging_middleware(ctx: Context, next: Callable) -> Any:
    print(f"-> Processing: {ctx.data}")
    result = next(ctx)
    print(f"<- Completed: {ctx.data}")
    return result

def auth_middleware(ctx: Context, next: Callable) -> Any:
    if not ctx.data.get('authenticated'):
        ctx.halt()
        return ctx
    return next(ctx)
```
**Why**: Enables composable, reusable processing chains; supports early exit.

## 5. Service Locator Pattern

```python
# PROBLEM: Provide global access to services without tight coupling
# TIME: O(1) for service lookup
# SPACE: O(n) where n = number of registered services
from typing import Any, TypeVar, Generic, Optional
from functools import wraps

T = TypeVar('T')

class ServiceLocator:
    """Global service registry with lazy initialization support."""
    
    _instance: 'ServiceLocator' = None
    _services: dict[type, Any] = {}
    _factories: dict[type, Callable[[], Any]] = {}
    
    def __new__(cls) -> 'ServiceLocator':
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance
    
    @classmethod
    def register(cls, interface: type, implementation: Any) -> None:
        """Register a service instance for an interface."""
        cls._services[interface] = implementation
    
    @classmethod
    def register_factory(cls, interface: type, factory: Callable[[], Any]) -> None:
        """Register a lazy factory for on-demand creation."""
        cls._factories[interface] = factory
    
    @classmethod
    def get(cls, interface: type[T]) -> T:
        """Retrieve service by interface type."""
        if interface in cls._services:
            return cls._services[interface]
        
        if interface in cls._factories:
            instance = cls._factories[interface]()
            cls._services[interface] = instance
            return instance
        
        raise KeyError(f"No service registered for {interface.__name__}")
    
    @classmethod
    def has(cls, interface: type) -> bool:
        """Check if service is registered."""
        return interface in cls._services or interface in cls._factories
    
    @classmethod
    def clear(cls) -> None:
        """Remove all registrations (useful for testing)."""
        cls._services.clear()
        cls._factories.clear()

def inject(*interfaces: type):
    """Decorator to inject services into function arguments."""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            injected = [ServiceLocator.get(iface) for iface in interfaces]
            return func(*injected, *args, **kwargs)
        return wrapper
    return decorator
```
**Why**: Provides global access point while maintaining testability through registration.

## 6. Data Access Object (DAO) Pattern

```python
# PROBLEM: Abstract data persistence operations from business logic
# TIME: O(1) for in-memory operations, O(n) for queries
# SPACE: O(n) where n = stored records
from abc import ABC, abstractmethod
from typing import Generic, TypeVar, Optional, Iterator
from dataclasses import dataclass, asdict

T = TypeVar('T')

@dataclass
class User:
    id: int
    name: str
    email: str

class Repository(ABC, Generic[T]):
    """Abstract repository interface for CRUD operations."""
    
    @abstractmethod
    def find_by_id(self, id: int) -> Optional[T]:
        pass
    
    @abstractmethod
    def find_all(self) -> Iterator[T]:
        pass
    
    @abstractmethod
    def save(self, entity: T) -> T:
        pass
    
    @abstractmethod
    def delete(self, id: int) -> bool:
        pass

class InMemoryUserRepository(Repository[User]):
    """In-memory implementation for testing/prototyping."""
    
    def __init__(self):
        self._store: dict[int, User] = {}
        self._next_id = 1
        self._indexes: dict[str, dict[str, int]] = {'email': {}}  # Secondary index
    
    def find_by_id(self, id: int) -> Optional[User]:
        return self._store.get(id)
    
    def find_by_email(self, email: str) -> Optional[User]:
        """Query by indexed field."""
        user_id = self._indexes['email'].get(email.lower())
        return self._store.get(user_id) if user_id else None
    
    def find_all(self) -> Iterator[User]:
        return iter(self._store.values())
    
    def save(self, user: User) -> User:
        if user.id is None:
            user.id = self._next_id
            self._next_id += 1
        
        self._store[user.id] = user
        self._indexes['email'][user.email.lower()] = user.id
        return user
    
    def delete(self, id: int) -> bool:
        user = self._store.pop(id, None)
        if user:
            self._indexes['email'].pop(user.email.lower(), None)
            return True
        return False

class UserService:
    """Business logic layer using DAO."""
    
    def __init__(self, repo: Repository[User]):
        self.repo = repo
    
    def get_or_create(self, email: str, name: str) -> User:
        existing = self.repo.find_by_email(email)
        if existing:
            return existing
        return self.repo.save(User(id=None, name=name, email=email))
```
**Why**: Separates persistence logic from domain; enables easy testing with mock repositories.

## 7. Protocol Pattern (Structural Subtyping)

```python
# PROBLEM: Define interfaces without inheritance using duck typing
# TIME: O(1) for type checking at runtime
# SPACE: O(0) no runtime overhead
from typing import Protocol, runtime_checkable, Any

@runtime_checkable
class Comparable(Protocol):
    """Protocol for objects that support comparison."""
    def __lt__(self, other: Any) -> bool: ...
    def __eq__(self, other: Any) -> bool: ...

@runtime_checkable  
class Serializable(Protocol):
    """Protocol for objects that can serialize to dict."""
    def to_dict(self) -> dict: ...
    @classmethod
    def from_dict(cls, data: dict) -> 'Serializable': ...

@runtime_checkable
class Observable(Protocol):
    """Protocol for objects supporting change notification."""
    def subscribe(self, callback: callable) -> None: ...
    def notify(self, event: Any) -> None: ...

def sort_and_serialize(items: list[Comparable & Serializable]) -> list[dict]:
    """Process items implementing multiple protocols."""
    return [item.to_dict() for item in sorted(items)]

# Example implementation
class Product:
    def __init__(self, name: str, price: float):
        self.name = name
        self.price = price
    
    def __lt__(self, other: 'Product') -> bool:
        return self.price < other.price
    
    def __eq__(self, other: object) -> bool:
        return isinstance(other, Product) and self.price == other.price
    
    def to_dict(self) -> dict:
        return {'name': self.name, 'price': self.price}
    
    @classmethod
    def from_dict(cls, data: dict) -> 'Product':
        return cls(data['name'], data['price'])

# Runtime protocol checking
product = Product("Widget", 9.99)
print(isinstance(product, Comparable))  # True
print(isinstance(product, Serializable))  # True
```
**Why**: Enables structural typing for flexible, composable interfaces without inheritance.

## 8. Coroutine Implementation

```python
# PROBLEM: Implement cooperative multitasking with generator-based coroutines
# TIME: O(1) per yield, O(n) total for n tasks
# SPACE: O(n) for task stack
from typing import Generator, Any
from collections import deque

Task = Generator[Any, Any, Any]

class Scheduler:
    """Simple cooperative scheduler for generator-based coroutines."""
    
    def __init__(self):
        self._ready: deque[Task] = deque()
        self._waiting: dict[Any, list[Task]] = {}
    
    def spawn(self, task: Task) -> None:
        """Add new task to ready queue."""
        self._ready.append(task)
    
    def run(self) -> None:
        """Execute tasks until all complete."""
        while self._ready:
            task = self._ready.popleft()
            try:
                event = next(task)
                self._handle_event(task, event)
            except StopIteration:
                pass  # Task completed
    
    def _handle_event(self, task: Task, event: Any) -> None:
        """Process task event and reschedule."""
        if isinstance(event, WaitEvent):
            key = event.resource
            self._waiting.setdefault(key, []).append(task)
        elif isinstance(event, SignalEvent):
            key = event.resource
            if key in self._waiting:
                self._ready.extend(self._waiting.pop(key))
        else:
            self._ready.append(task)  # Reschedule

class WaitEvent:
    def __init__(self, resource: str):
        self.resource = resource

class SignalEvent:
    def __init__(self, resource: str):
        self.resource = resource

def wait_for(resource: str) -> WaitEvent:
    return WaitEvent(resource)

def signal(resource: str) -> SignalEvent:
    return SignalEvent(resource)

# Example usage
def producer(queue_name: str, count: int) -> Task:
    for i in range(count):
        print(f"Producing item {i}")
        yield  # Cooperatively yield
    yield signal(queue_name)

def consumer(queue_name: str) -> Task:
    print("Waiting for producer...")
    yield wait_for(queue_name)
    print("Consumer done!")
```
**Why**: Demonstrates cooperative multitasking fundamentals; foundation for async/await.

## 9. Full-Text Search with TF-IDF

```python
# PROBLEM: Rank search results by term relevance using TF-IDF
# TIME: O(n) for indexing, O(q * log n) for ranked search
# SPACE: O(n * t) where n=docs, t=terms
import math
from collections import Counter, defaultdict
from typing import Optional

class TFIDFIndex:
    """Full-text search with TF-IDF ranking."""
    
    def __init__(self):
        self._docs: dict[int, dict] = {}  # doc_id -> {text, term_freq}
        self._df: Counter = Counter()      # document frequency per term
        self._next_id = 0
    
    def _tokenize(self, text: str) -> list[str]:
        return text.lower().split()
    
    def add(self, document: str) -> int:
        """Index document with term frequency calculation."""
        doc_id = self._next_id
        self._next_id += 1
        
        terms = self._tokenize(document)
        tf = Counter(terms)
        
        # Update document frequency
        for term in set(terms):
            self._df[term] += 1
        
        self._docs[doc_id] = {
            'text': document,
            'tf': tf,
            'length': len(terms)
        }
        
        return doc_id
    
    def _tfidf(self, term: str, tf: int, total_docs: int) -> float:
        """Calculate TF-IDF score for a term."""
        if tf == 0:
            return 0.0
        
        # Logarithmic TF
        log_tf = 1 + math.log10(tf)
        
        # IDF
        df = self._df.get(term, 0)
        if df == 0:
            return 0.0
        idf = math.log10(total_docs / df)
        
        return log_tf * idf
    
    def search(self, query: str, top_k: int = 10) -> list[tuple[int, float]]:
        """Return documents ranked by TF-IDF relevance."""
        query_terms = self._tokenize(query)
        total_docs = len(self._docs)
        
        if total_docs == 0:
            return []
        
        scores: dict[int, float] = defaultdict(float)
        
        for term in query_terms:
            for doc_id, doc_data in self._docs.items():
                tf = doc_data['tf'].get(term, 0)
                scores[doc_id] += self._tfidf(term, tf, total_docs)
        
        # Sort by score descending
        ranked = sorted(scores.items(), key=lambda x: -x[1])
        return ranked[:top_k]
    
    def get_document(self, doc_id: int) -> Optional[str]:
        return self._docs.get(doc_id, {}).get('text')
```
**Why**: Provides relevance-ranked search; TF-IDF is foundation of information retrieval.

## 10. Projection Pattern (CQRS Read Model)

```python
# PROBLEM: Maintain optimized read models from event stream
# TIME: O(1) for projection update, O(k) for query where k = index depth
# SPACE: O(n * p) where n = events, p = projections
from typing import Callable, Any
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum, auto

class EventType(Enum):
    USER_CREATED = auto()
    USER_UPDATED = auto()
    ORDER_PLACED = auto()
    ORDER_SHIPPED = auto()

@dataclass
class Event:
    type: EventType
    aggregate_id: str
    data: dict
    timestamp: float = field(default_factory=lambda: datetime.now().timestamp())

class Projection:
    """Base class for read model projections."""
    
    def __init__(self):
        self._handlers: dict[EventType, Callable] = {}
        self._last_sequence = 0
    
    def register(self, event_type: EventType, handler: Callable) -> None:
        self._handlers[event_type] = handler
    
    def apply(self, event: Event, sequence: int) -> None:
        if sequence <= self._last_sequence:
            return  # Skip already processed
        
        handler = self._handlers.get(event.type)
        if handler:
            handler(event.data)
        self._last_sequence = sequence

class UserSummaryProjection(Projection):
    """Maintains summary statistics for users."""
    
    def __init__(self):
        super().__init__()
        self._users: dict[str, dict] = {}
        self._order_counts: dict[str, int] = {}
        
        self.register(EventType.USER_CREATED, self._on_user_created)
        self.register(EventType.ORDER_PLACED, self._on_order_placed)
    
    def _on_user_created(self, data: dict) -> None:
        user_id = data['user_id']
        self._users[user_id] = {
            'name': data['name'],
            'email': data['email'],
            'created_at': data['timestamp']
        }
        self._order_counts[user_id] = 0
    
    def _on_order_placed(self, data: dict) -> None:
        user_id = data['user_id']
        if user_id in self._order_counts:
            self._order_counts[user_id] += 1
    
    def get_user(self, user_id: str) -> dict | None:
        if user_id not in self._users:
            return None
        return {
            **self._users[user_id],
            'order_count': self._order_counts.get(user_id, 0)
        }
    
    def get_all_users(self) -> list[dict]:
        return [
            {**u, 'order_count': self._order_counts.get(uid, 0)}
            for uid, u in self._users.items()
        ]

class ProjectionManager:
    """Coordinates multiple projections."""
    
    def __init__(self):
        self._projections: list[Projection] = []
    
    def add(self, projection: Projection) -> 'ProjectionManager':
        self._projections.append(projection)
        return self
    
    def rebuild(self, events: list[tuple[Event, int]]) -> None:
        """Rebuild all projections from event history."""
        for projection in self._projections:
            for event, seq in events:
                projection.apply(event, seq)
```
**Why**: Enables optimized read models separate from write path; supports eventual consistency.
