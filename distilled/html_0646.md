```html
PROBLEM: Tables overflow on mobile devices, causing horizontal scrolling
TIME: O(1) SPACE: O(1)
-->

        .table-wrapper {
            overflow-x: auto;

            width: 100%;
            margin: 1rem 0;
            border: 1px solid #ddd;
        }

            width: 100%;
            border-collapse: collapse;
            min-width: 600px; /* Forces scroll on small screens */

            border-bottom: 1px solid #ddd;
        }

            background-color: #f4f4f4;
            font-weight: bold;
        }
        
        /* Screen reader only label */

            width: 1px;
            height: 1px;
            padding: 0;

            overflow: hidden;

                <tr>

    <script>
        // Enhance keyboard navigation for screen readers
        document.addEventListener('DOMContentLoaded', () => {
            const wrapper = document.querySelector('.table-wrapper');
            
            // Add scroll indicator
            if (wrapper.scrollWidth > wrapper.clientWidth) {
                const indicator = document.createElement('p');
                indicator.className = 'sr-only';
                indicator.textContent = 'Table can be scrolled horizontally';
                wrapper.parentNode.insertBefore(indicator, wrapper.nextSibling);
            }
        });
    </script>
</body>
</html>
```

**Why**: Tables naturally overflow on mobile devices. A scrollable wrapper preserves data integrity while providing touch-friendly navigation, avoiding content truncation or awkward line breaks.

## html_table_pagination

```html
<!-- 
PROBLEM: Large tables are hard to navigate and slow to render
TIME: O(n) where n = items per page SPACE: O(1)
-->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Table Pagination</title>
    <style>
        table {
            width: 100%;
            border-collapse: collapse;
        }
        
        th, td {
            padding: 10px;
            border: 1px solid #ddd;
            text-align: left;
        }
        
        th {
            background-color: #4CAF50;
            color: white;
        }
        
        .pagination {
            display: flex;
            justify-content: center;
            gap: 5px;
            margin-top: 20px;
        }
        
        .pagination button {
            padding: 8px 16px;
            border: 1px solid #ddd;
            background-color: white;
            cursor: pointer;
            transition: background-color 0.3s;
        }
        
        .pagination button:hover:not(:disabled) {
            background-color: #f0f0f0;
        }
        
        .pagination button.active {
            background-color: #4CAF50;
            color: white;
            border-color: #4CAF50;
        }
        
        .pagination button:disabled {
            opacity: 0.5;
            cursor: not-allowed;
        }
        
        .page-info {
            text-align: center;
            margin-top: 10px;
            color: #666;
        }
    </style>
</head>
<body>
    <table id="dataTable">
        <thead>
            <tr>
                <th>ID</th>
                <th>Name</th>
                <th>Email</th>
                <th>Status</th>
            </tr>
        </thead>
        <tbody id="tableBody"></tbody>
    </table>
    
    <div class="page-info" id="pageInfo"></div>
    
    <div class="pagination" id="pagination" role="navigation" aria-label="Table pagination"></div>
    
    <script>
        // Sample dataset
        const data = Array.from({ length: 50 }, (_, i) => ({
            id: i + 1,
            name: `User ${i + 1}`,
            email: `user${i + 1}@example.com`,
            status: i % 3 === 0 ? 'Active' : 'Inactive'
        }));
        
        class TablePaginator {
            constructor(data, itemsPerPage = 10) {
                this.data = data;
                this.itemsPerPage = itemsPerPage;
                this.currentPage = 1;
                this.totalPages = Math.ceil(data.length / itemsPerPage);
                this.init();
            }
            
            init() {
                this.render();
                this.createPagination();
            }
            
            // Render current page data
            render() {
                const start = (this.currentPage - 1) * this.itemsPerPage;
                const end = start + this.itemsPerPage;
                const pageData = this.data.slice(start, end);
                
                const tbody = document.getElementById('tableBody');
                tbody.innerHTML = pageData.map(item => `
                    <tr>
                        <td>${item.id}</td>
                        <td>${item.name}</td>
                        <td>${item.email}</td>
                        <td>${item.status}</td>
                    </tr>
                `).join('');
                
                // Update page info
                document.getElementById('pageInfo').textContent = 
                    `Showing ${start + 1}-${Math.min(end, this.data.length)} of ${this.data.length} items`;
            }
            
            // Create pagination buttons
            createPagination() {
                const container = document.getElementById('pagination');
                container.innerHTML = '';
                
                // Previous button
                const prevBtn = this.createButton('← Previous', () => this.goToPage(this.currentPage - 1));
                prevBtn.disabled = this.currentPage === 1;
                container.appendChild(prevBtn);
                
                // Page numbers
                for (let i = 1; i <= this.totalPages; i++) {
                    const btn = this.createButton(i.toString(), () => this.goToPage(i));
                    if (i === this.currentPage) btn.classList.add('active');
                    container.appendChild(btn);
                }
                
                // Next button
                const nextBtn = this.createButton('Next →', () => this.goToPage(this.currentPage + 1));
                nextBtn.disabled = this.currentPage === this.totalPages;
                container.appendChild(nextBtn);
            }
            
            createButton(text, handler) {
                const btn = document.createElement('button');
                btn.textContent = text;
                btn.addEventListener('click', handler);
                return btn;
            }
            
            goToPage(page) {
                if (page < 1 || page > this.totalPages) return;
                this.currentPage = page;
                this.render();
                this.createPagination();
            }
        }
        
        // Initialize
        const paginator = new TablePaginator(data, 10);
    </script>
</body>
</html>
```

**Why**: Pagination dramatically improves UX for large datasets by reducing render time, scroll fatigue, and cognitive load. It also enables easier data navigation and bookmarking.

```html
        // Screen readers can jump directly to contentinfo landmark
        document.querySelector('footer[role="contentinfo"]').addEventListener('keydown', (e) => {
            // Allow keyboard navigation within footer
            if (e.key === 'Tab') {
                const focusableElements = e.currentTarget.querySelectorAll('a, button');
                const firstElement = focusableElements[0];
                const lastElement = focusableElements[focusableElements.length - 1];
                
                if (e.shiftKey && document.activeElement === firstElement) {
                    e.preventDefault();
```

**Why**: The `contentinfo` role explicitly marks footer content as a landmark region, enabling screen reader users to quickly navigate to copyright, privacy policy, and contact information without tabbing through the entire page.

```html
    <script>
        // Tab functionality
        const tabs = document.querySelectorAll('.tab-button');
        const panels = document.querySelectorAll('.tab-panel');
        
        tabs.forEach(tab => {
            tab.addEventListener('click', () => {
                // Deselect all tabs
                tabs.forEach(t => t.setAttribute('aria-selected', 'false'));
                // Select clicked tab
                tab.setAttribute('aria-selected', 'true');
                
                // Hide all panels
                panels.forEach(p => p.setAttribute('aria-hidden', 'true'));
                // Show corresponding panel
                const panel = document.getElementById(tab.getAttribute('aria-controls'));
                panel.setAttribute('aria-hidden', 'false');
            });
            
            // Keyboard navigation
            tab.addEventListener('keydown', (e) => {
                const index = Array.from(tabs).indexOf(tab);
                if (e.key === 'ArrowRight' && index < tabs.length - 1) {
                    tabs[index + 1].focus();
                } else if (e.key === 'ArrowLeft' && index > 0) {
                    tabs[index - 1].focus();
                }
            });
        });
        
        // Listbox functionality
        const options = document.querySelectorAll('.listbox-option');
        const status = document.getElementById('status');
        
        options.forEach(option => {
            option.addEventListener('click', () => {
                // Deselect all options
                options.forEach(o => o.setAttribute('aria-selected', 'false'));
                // Select clicked option
                option.setAttribute('aria-selected', 'true');
                
                // Update status
                status.textContent = `Current selection: ${option.textContent}`;
            });
            
            option.addEventListener('keydown', (e) => {
                if (e.key === 'Enter' || e.key === ' ') {
                    e.preventDefault();
                    option.click();
                }
            });
        });
    </script>
</body>
</html>
```

**Why**: `aria-selected` communicates selection state to assistive technologies. Essential for tabs, listboxes, and grid components where visual selection cues (colors, borders) aren't accessible to screen reader users.
