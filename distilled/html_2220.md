## js_indexeddb_get

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>IndexedDB Get Operation</title>
</head>
<body>
    <div id="result"></div>
    <script>
        // PROBLEM: Retrieve a specific record from IndexedDB object store
        // TIME: O(1) average case for key lookup, SPACE: O(1) for single record
        
        // Open or create database
        const request = indexedDB.open('MyDatabase', 1);
        
        // Create object store on database creation/upgrade
        request.onupgradeneeded = (event) => {
            const db = event.target.result;
            // Create object store with keyPath
            if (!db.objectStoreNames.contains('users')) {
                db.createObjectStore('users', { keyPath: 'id' });
            }
        };
        
        request.onsuccess = (event) => {
            const db = event.target.result;
            
            // Start a read-only transaction
            const transaction = db.transaction(['users'], 'readonly');
            const store = transaction.objectStore('users');
            
            // Get record by key
            const getRequest = store.get(1);
            
            getRequest.onsuccess = (event) => {
                const user = event.target.result;
                if (user) {
                    document.getElementById('result').innerHTML = 
                        `User: ${user.name} - ${user.email}`;
                } else {
                    document.getElementById('result').innerHTML = 'User not found';
                }
            };
            
            getRequest.onerror = (event) => {
                console.error('Error fetching record:', event.target.error);
            };
        };
        
        request.onerror = (event) => {
            console.error('Database error:', event.target.error);
        };
    </script>
</body>
</html>
```

**Why**: IndexedDB's `get()` method provides efficient key-based retrieval from browser storage, essential for offline-first applications and client-side data persistence.

## js_response_formData

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Response.formData() API</title>
</head>
<body>
    <button id="fetchBtn">Fetch Form Data</button>
    <div id="output"></div>
    
    <script>
        // PROBLEM: Extract form-encoded data from a Response object
        // TIME: O(n) where n is the number of form fields, SPACE: O(n) for FormData storage
        
        document.getElementById('fetchBtn').addEventListener('click', async () => {
            try {
                // Fetch data from server that returns form-encoded content
                const response = await fetch('https://httpbin.org/post', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/x-www-form-urlencoded',
                    },
                    body: 'username=john&email=john@example.com&age=30'
                });
                
                // Check if response is OK
                if (!response.ok) {
                    throw new Error(`HTTP error! status: ${response.status}`);
                }
                
                // Extract FormData from response
                // Note: Only works if server returns multipart/form-data or application/x-www-form-urlencoded
                const formData = await response.formData();
                
                // Iterate through FormData entries
                let output = '<h3>Form Data Received:</h3><ul>';
                for (const [key, value] of formData.entries()) {
                    output += `<li><strong>${key}:</strong> ${value}</li>`;
                }
                output += '</ul>';
                
                document.getElementById('output').innerHTML = output;
                
            } catch (error) {
                console.error('Error:', error);
                document.getElementById('output').innerHTML = 
                    `<p style="color: red;">Error: ${error.message}</p>`;
            }
        });
    </script>
</body>
</html>
```

**Why**: The `response.formData()` method enables parsing form-encoded server responses directly into FormData objects, useful for handling form submissions and multipart data in web applications.

```html
                selectedIndex: event.target.selectedIndex

        // Change event for number input
        // Fires when value changes and element loses focus
        document.getElementById('quantityInput').addEventListener('change', (event) => {
            const quantity = event.target.value;
            feedback.innerHTML = `Quantity updated to: <strong>${quantity}</strong>`;
            
            console.log('Quantity changed:', quantity);
        });
        
        // Change event for checkbox
        // Fires immediately when checkbox state changes
        document.getElementById('termsCheckbox').addEventListener('change', (event) => {
            const isChecked = event.target.checked;
            feedback.innerHTML = `Terms ${isChecked ? 'accepted' : 'not accepted'}`;
            
            console.log('Checkbox changed:', isChecked);
        });
        
        // Note: For real-time input tracking, use 'input' event instead
        // Change event fires on blur (for text inputs) or immediately (for checkboxes/radios)
    </script>
```

**Why**: The `change` event provides reliable detection of committed value changes in form elements, firing when user completes their input, making it ideal for form validation and state management.

## html5_service_workers

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Service Worker Registration</title>
    <style>
        .status { padding: 15px; margin: 10px 0; border-radius: 5px; }
        .success { background: #d4edda; color: #155724; }
        .error { background: #f8d7da; color: #721c24; }
        .info { background: #d1ecf1; color: #0c5460; }
    </style>
</head>
<body>
    <h2>Service Worker Demo</h2>
    <div id="status"></div>
    <button id="unregister">Unregister Service Worker</button>
    
    <script>
        // PROBLEM: Register and manage service workers for offline caching and background processing
        // TIME: O(1) for registration, SPACE: O(n) for cached resources
        
        const statusDiv = document.getElementById('status');
        
        // Check if service workers are supported
        if ('serviceWorker' in navigator) {
            // Register service worker
            navigator.serviceWorker.register('/sw.js', { scope: '/' })
                .then((registration) => {
                    console.log('Service Worker registered:', registration);
                    showStatus('Service Worker registered successfully!', 'success');
                    
                    // Check if there's an update available
                    registration.addEventListener('updatefound', () => {
                        const newWorker = registration.installing;
                        showStatus('New Service Worker installing...', 'info');
                        
                        newWorker.addEventListener('statechange', () => {
                            if (newWorker.state === 'installed' && navigator.serviceWorker.controller) {
                                showStatus('New version available! Refresh to update.', 'info');
                            }
                        });
                    });
                    
                    // Check registration state
                    if (registration.installing) {
                        console.log('Service Worker installing');
                    } else if (registration.waiting) {
                        console.log('Service Worker waiting');
                    } else if (registration.active) {
                        console.log('Service Worker active');
                    }
                })
                .catch((error) => {
                    console.error('Service Worker registration failed:', error);
                    showStatus(`Registration failed: ${error.message}`, 'error');
                });
            
            // Listen for controlling service worker changes
            navigator.serviceWorker.addEventListener('controllerchange', () => {
                console.log('Controller changed, reloading...');
                window.location.reload();
            });
            
            // Unregister button handler
            document.getElementById('unregister').addEventListener('click', async () => {
                const registration = await navigator.serviceWorker.getRegistration();
                if (registration) {
                    registration.unregister()
                        .then(() => {
                            showStatus('Service Worker unregistered', 'info');
                        })
                        .catch((error) => {
                            showStatus(`Unregister failed: ${error.message}`, 'error');
                        });
                }
            });
            
        } else {
            showStatus('Service Workers not supported in this browser', 'error');
        }
        
        function showStatus(message, type) {
            statusDiv.innerHTML = `<div class="status ${type}">${message}</div>`;
        }
    </script>
</body>
</html>
```

**Why**: Service workers enable offline functionality, push notifications, and background sync, making them essential for Progressive Web Apps (PWAs) and improved user experience.

```html
        // PROBLEM: Handle errors when values are outside the allowable range
        // TIME: O(1) for error creation, SPACE: O(1) for error object
        
        const output = document.getElementById('output');
        
        // Example 1: Array length must be finite and non-negative
        function demonstrateArrayLength() {
            try {
                // This will throw RangeError
                const arr = new Array(-1);
            } catch (error) {
                if (error instanceof RangeError) {
                    showError('Array Length Error', error.message);
                }
            }
        }
        
        // Example 2: Number toFixed() digits out of range
        function demonstrateToFixed() {
            try {
                const num = 3.14159;
                // toFixed accepts 0-100, this will throw RangeError
                const result = num.toFixed(101);
            } catch (error) {
                if (error instanceof RangeError) {
                    showError('toFixed Range Error', error.message);
                }
            }
        }
        
        // Example 3: Number precision out of range
        function demonstratePrecision() {
            try {
                const num = 123.456;
                // toPrecision accepts 1-100
                const result = num.toPrecision(0);
            } catch (error) {
                if (error instanceof RangeError) {
                    showError('Precision Range Error', error.message);
                }
            }
        }
        
        // Example 4: String repeat count out of range
        function demonstrateRepeat() {
            try {
                const str = 'hello';
                // Negative count throws RangeError
                const result = str.repeat(-1);
            } catch (error) {
                if (error instanceof RangeError) {
                    showError('Repeat Count Error', error.message);
                }
            }
        }
        
        // Example 5: Stack overflow (recursion depth)
        let recursionCount = 0;
        function infiniteRecursion() {
            recursionCount++;
            if (recursionCount > 10000) {
                // Prevent actual crash
                showError('Recursion Prevented', 'Would cause stack overflow');
                return;
            }
            try {
                return infiniteRecursion();
            } catch (error) {
                if (error instanceof RangeError) {
                    showError('Stack Overflow', 'Maximum call stack size exceeded');
                }
            }
        }
        
        // Example 6: Proper validation to prevent RangeError
        function safeArrayCreation(length) {
            if (!Number.isInteger(length) || length < 0 || length > Number.MAX_SAFE_INTEGER) {
                throw new RangeError(`Invalid array length: ${length}`);
            }
            return new Array(length);
        }
        
        function testSafeArray() {
            try {
                const arr = safeArrayCreation(5);
                showError('Safe Array Creation', `Created array with ${arr.length} elements`, true);
            } catch (error) {
                showError('Error', error.message);
            }
        }

        demonstrateRepeat();

        // - Invalid array length (negative or too large)
        // - toFixed() digits (0-100)
        // - toPrecision() digits (1-100)  
        // - String repeat count (non-negative)
        // - Stack overflow from infinite recursion
        // - Date values outside valid range

</html>
```

**Why**: `RangeError` indicates values outside acceptable bounds, helping developers identify and handle numeric range violations, invalid array lengths, and stack overflow conditions.

## js_regexp_search

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>RegExp search() Method</title>
    <style>
        .result { padding: 10px; margin: 10px 0; background: #f5f5f5; }
        .match { background: #d4edda; }
        .no-match { background: #f8d7da; }
    </style>
</head>
<body>
    <h2>String.prototype.search() with RegExp</h2>
    <input type="text" id="inputText" value="The quick brown fox jumps over the lazy dog" size="50">
    <br><br>
    <input type="text" id="pattern" value="fox" placeholder="Enter pattern">
    <button id="searchBtn">Search</button>
    <div id="output"></div>
    
    <script>
        // PROBLEM: Find the first match position of a pattern in a string
        // TIME: O(n*m) worst case where n=string length, m=pattern length
        // SPACE: O(1) - only returns index
        
        document.getElementById('searchBtn').addEventListener('click', () => {
            const text = document.getElementById('inputText').value;
            const pattern = document.getElementById('pattern').value;
            const output = document.getElementById('output');
            
            // Create RegExp from user input
            const regex = new RegExp(pattern);
            
            // search() returns index of first match or -1 if not found
            const index = text.search(regex);
            
            let html = '<div class="result ';
            
            if (index !== -1) {
                html += 'match">';
                html += `<strong>Pattern:</strong> /${pattern}/<br>`;
                html += `<strong>Found at index:</strong> ${index}<br>`;
                html += `<strong>Matched text:</strong> "${text.substring(index, index + pattern.length)}"<br>`;
                
                // Show context around match
                const start = Math.max(0, index - 10);
                const end = Math.min(text.length, index + pattern.length + 10);
                const context = text.substring(start, end);
                html += `<strong>Context:</strong> ...${context}...`;
            } else {
                html += 'no-match">';
                html += `<strong>Pattern:</strong> /${pattern}/<br>`;
                html += `<strong>Result:</strong> No match found`;
            }
            
            html += '</div>';
            output.innerHTML = html;
        });
        
        // Examples demonstrating search() behavior
        const examples = [
            { text: 'Hello World', pattern: /world/i, desc: 'Case-insensitive search' },
            { text: 'user@example.com', pattern: /@/, desc: 'Find @ symbol' },
            { text: '123-456-7890', pattern: /\d{3}/, desc: 'Find 3 digits' },
            { text: 'foo bar baz', pattern: /bar/, desc: 'Simple word search' },
            { text: 'no match here', pattern: /xyz/, desc: 'No match returns -1' }
        ];
        
        // Display examples
        let examplesHtml = '<h3>Examples:</h3>';
        examples.forEach(ex => {
            const index = ex.text.search(ex.pattern);
            examplesHtml += `<div class="result ${index !== -1 ? 'match' : 'no-match'}">`;
            examplesHtml += `<strong>${ex.desc}</strong><br>`;
            examplesHtml += `Text: "${ex.text}"<br>`;
            examplesHtml += `Pattern: ${ex.pattern}<br>`;
            examplesHtml += `Result: ${index}`;
            examplesHtml += '</div>';
        });
        
        output.innerHTML += examplesHtml;
        
        // Note: search() vs match() vs exec()
        // - search(): Returns index of first match (like indexOf but with regex)
        // - match(): Returns array of matches or null
        // - exec(): Returns detailed match information with groups
        // - indexOf(): String search without regex support
    </script>
</body>
</html>
```

**Why**: The `search()` method efficiently locates the first occurrence of a pattern in a string, returning the index position, making it ideal for quick pattern existence checks and position finding.
