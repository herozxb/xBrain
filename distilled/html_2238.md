## IndexedDB Put Operation

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>IndexedDB Put Operation</title>
</head>
<body>
    <h2>IndexedDB Put (Add/Update Data)</h2>
    <input type="text" id="keyInput" placeholder="Enter key">
    <input type="text" id="valueInput" placeholder="Enter value">
    <button onclick="putData()">Save Data</button>
    <p id="status"></p>

    <script>
        // PROBLEM: Store or update data in IndexedDB
        // TIME: O(1) SPACE: O(n) where n is stored data size
        
        let db = null;
        
        // Open or create the database
        const request = indexedDB.open('MyDatabase', 1);
        
        // Create object store when database is first created or version changes
        request.onupgradeneeded = (event) => {
            db = event.target.result;
            // Create an object store named 'store' with 'id' as the key path
            if (!db.objectStoreNames.contains('store')) {
                db.createObjectStore('store', { keyPath: 'id' });
            }
        };
        
        // Successfully opened database
        request.onsuccess = (event) => {
            db = event.target.result;
            document.getElementById('status').textContent = 'Database ready!';
        };
        
        // Error handling
        request.onerror = (event) => {
            document.getElementById('status').textContent = 'Database error!';
            console.error('Database error:', event.target.error);
        };
        
        /**
         * Put data into IndexedDB (add new or update existing)
         * Uses 'put' which will insert or update based on key
         */
        function putData() {
            const key = document.getElementById('keyInput').value.trim();
            const value = document.getElementById('valueInput').value.trim();
            
            if (!key || !value) {
                document.getElementById('status').textContent = 'Please enter both key and value!';
                return;
            }
            
            // Start a readwrite transaction
            const transaction = db.transaction(['store'], 'readwrite');
            const store = transaction.objectStore('store');
            
            // Create data object with id as key
            const data = {
                id: key,
                value: value,
                timestamp: Date.now()
            };
            
            // Put will add if key doesn't exist, or update if it does
            const putRequest = store.put(data);
            
            putRequest.onsuccess = () => {
                document.getElementById('status').textContent = `Data saved/updated: ${key} = ${value}`;
            };
            
            putRequest.onerror = (event) => {
                document.getElementById('status').textContent = 'Failed to save data!';
                console.error('Put error:', event.target.error);
            };
        }
    </script>
</body>
</html>
```

**Why**: IndexedDB's `put()` method is crucial for offline-first applications, enabling persistent storage that survives browser restarts. It automatically handles both insert and update operations.

## Response Headers

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Fetch Response Headers</title>

<body>
    <h2>Reading Response Headers</h2>
    <button onclick="fetchHeaders()">Fetch Headers</button>
    <div id="output"></div>

    <script>
        // PROBLEM: Access HTTP response headers from Fetch API
        // TIME: O(n) where n is number of headers SPACE: O(n)
        
        /**
         * Fetch a URL and display all response headers
         * Headers are accessed via the Headers interface
         */
        async function fetchHeaders() {
            const output = document.getElementById('output');
            output.innerHTML = '<p>Loading...</p>';
            
            try {
                // Make a fetch request to a public API
                const response = await fetch('https://httpbin.org/headers', {
                    method: 'GET'
                });
                
                // Check if response is ok (status 200-299)
                if (!response.ok) {
                    throw new Error(`HTTP error! status: ${response.status}`);
                }
                
                // Get the Headers object from response
                const headers = response.headers;
                
                // Build output string
                let html = '<h3>Response Headers:</h3><ul>';
                
                // Method 1: Iterate using entries() iterator
                for (const [name, value] of headers.entries()) {
                    html += `<li><strong>${name}:</strong> ${value}</li>`;
                }
                
                html += '</ul>';
                
                // Method 2: Access specific headers directly
                html += '<h3>Specific Headers:</h3>';
                html += `<p>Content-Type: ${headers.get('content-type') || 'N/A'}</p>`;
                html += `<p>Date: ${headers.get('date') || 'N/A'}</p>`;
                
                // Note: Some headers may not be accessible due to CORS
                html += '<p><em>Note: CORS may restrict access to some headers</em></p>';
                
                output.innerHTML = html;
                
            } catch (error) {
                output.innerHTML = `<p style="color: red;">Error: ${error.message}</p>`;
                console.error('Fetch error:', error);
            }
        }
    </script>
</body>
</html>
```

**Why**: Response headers contain crucial metadata like content type, caching directives, and authentication tokens. Understanding how to read them is essential for proper API integration and debugging.

```html
                emailError.textContent = '✓ Valid';
                emailError.style.color = 'green';
                this.classList.remove('error');
                this.classList.add('valid');
            } else {
                emailError.textContent = '✗ Invalid email format';
                emailError.style.color = 'red';
                this.classList.remove('valid');
                this.classList.add('error');
            }
        });
        
        /**
         * Validate age field on blur
         * Demonstrates numeric validation
         */
        ageInput.addEventListener('blur', function() {
            const value = this.value.trim();
            const age = parseInt(value, 10);
            
            if (value === '') {
                ageError.textContent = '';
                this.classList.remove('error', 'valid');
            } else if (isNaN(age) || age < 0 || age > 150) {
                ageError.textContent = '✗ Enter a valid age (0-150)';
                ageError.style.color = 'red';
                this.classList.remove('valid');
                this.classList.add('error');
            } else {
                ageError.textContent = '✓ Valid';
                ageError.style.color = 'green';
                this.classList.remove('error');
                this.classList.add('valid');
            }
        });
```

**Why**: The blur event is ideal for form validation because it triggers after the user finishes editing, avoiding interruptions during typing. It provides a better UX than validating on every keystroke.

## History PushState

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>History PushState API</title>
    <style>
        .nav-btn { padding: 10px 20px; margin: 5px; cursor: pointer; }
        #content { padding: 20px; border: 1px solid #ccc; margin-top: 20px; }
    </style>
</head>
<body>
    <h2>History API - PushState</h2>
    <p>Navigate without page reload:</p>
    <button class="nav-btn" onclick="navigateTo('home')">Home</button>
    <button class="nav-btn" onclick="navigateTo('about')">About</button>
    <button class="nav-btn" onclick="navigateTo('contact')">Contact</button>
    <button class="nav-btn" onclick="navigateTo('products')">Products</button>
    
    <div id="content">
        <p>Select a page to navigate</p>
    </div>

    <script>
        // PROBLEM: Update browser URL and history without page reload
        // TIME: O(1) SPACE: O(1)
        
        const content = document.getElementById('content');
        
        // Page content data
        const pages = {
            home: { title: 'Home', html: '<h3>Welcome Home</h3><p>This is the home page.</p>' },
            about: { title: 'About Us', html: '<h3>About Us</h3><p>Learn more about our company.</p>' },
            contact: { title: 'Contact', html: '<h3>Contact Us</h3><p>Email: contact@example.com</p>' },
            products: { title: 'Products', html: '<h3>Our Products</h3><ul><li>Product A</li><li>Product B</li></ul>' }
        };
        
        /**
         * Navigate to a page using pushState
         * @param {string} page - Page identifier
         */
        function navigateTo(page) {
            // Check if page exists
            if (!pages[page]) return;
            
            // Update content
            content.innerHTML = pages[page].html;
            
            // Push new state to history
            // pushState(state object, title (ignored by most browsers), URL)
            history.pushState(
                { page: page },           // State object - can be any serializable data
                pages[page].title,        // Title (mostly ignored by browsers)
                `/spa/${page}`            // URL to display in address bar
            );
            
            // Update document title
            document.title = pages[page].title + ' - My SPA';
        }
        
        /**
         * Handle browser back/forward buttons
         * popstate fires when active history entry changes
         */
        window.addEventListener('popstate', (event) => {
            // event.state contains the state object we pushed
            if (event.state && event.state.page) {
                const page = event.state.page;
                content.innerHTML = pages[page].html;
                document.title = pages[page].title + ' - My SPA';
            } else {
                // No state - show default
                content.innerHTML = '<p>Select a page to navigate</p>';
            }
        });
        
        /**
         * Handle initial page load with history state
         */
        window.addEventListener('load', () => {
            // Replace current state to ensure we have a state object
            history.replaceState(
                { page: 'home' },
                'Home - My SPA',
                '/spa/home'
            );
        });
    </script>
</body>
</html>
```

**Why**: pushState enables Single Page Applications (SPAs) by allowing URL changes without page reloads. Combined with popstate, it creates seamless navigation that works with browser history.

```html
    <script>
        // PROBLEM: Implement dynamic theme switching
        // TIME: O(1) SPACE: O(1)
        
        /**
         * Toggle between light and dark themes
         * Uses data-theme attribute on document element
         */
        function toggleTheme() {
            const currentTheme = document.documentElement.getAttribute('data-theme');
            const newTheme = currentTheme === 'dark' ? 'light' : 'dark';
            
            // Set the new theme
            document.documentElement.setAttribute('data-theme', newTheme);
            
            // Persist preference in localStorage
            localStorage.setItem('theme', newTheme);
        }
        
        /**
         * Load saved theme preference on page load
         */
        function loadTheme() {
            const savedTheme = localStorage.getItem('theme');
            
            if (savedTheme) {
                document.documentElement.setAttribute('data-theme', savedTheme);
            } else {
                // Check system preference
                if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) {
                    document.documentElement.setAttribute('data-theme', 'dark');
                }
            }
        }
        
        // Listen for system theme changes
        window.matchMedia('(prefers-color-scheme: dark)').addEventListener('change', (e) => {
            if (!localStorage.getItem('theme')) {
                document.documentElement.setAttribute('data-theme', e.matches ? 'dark' : 'light');
            }
        });
        
        // Load theme on page load
        loadTheme();
    </script>
</body>
</html>
```

**Why**: CSS custom properties (variables) enable powerful theming systems. Combined with data attributes and localStorage, they provide persistent, accessible theme switching that respects user preferences.

```html
    <script>
        // PROBLEM: Handle URI encoding/decoding errors properly
        // TIME: O(n) where n is string length SPACE: O(n)
        
        /**
         * Encode a URI component with error handling
         * URIError thrown when string contains malformed UTF-8
         */
        function encodeString() {
            const input = document.getElementById('encodeInput').value;
            const output = document.getElementById('output');
            
            try {
                // encodeURIComponent encodes all special characters
                const encoded = encodeURIComponent(input);
                output.innerHTML = `
                    <p><strong>Original:</strong> ${input}</p>
                    <p><strong>Encoded:</strong> ${encoded}</p>
                    <p><em>Safe for URL parameters</em></p>
                `;
                output.style.color = 'green';
                
            } catch (error) {
                // Handle URIError specifically
                if (error instanceof URIError) {
                    output.innerHTML = `<p style="color: red;">URIError: ${error.message}</p>`;
                    console.error('URI encoding error:', error);
                } else {
                    throw error; // Re-throw non-URI errors
                }
            }
        }
        
        /**
         * Decode a URI component with error handling
         * URIError thrown when input has invalid escape sequences
         */
        function decodeString() {
            const input = document.getElementById('decodeInput').value;
            const output = document.getElementById('output');
            
            try {
                // decodeURIComponent may throw URIError for malformed input
                const decoded = decodeURIComponent(input);
                output.innerHTML = `
                    <p><strong>Encoded:</strong> ${input}</p>
                    <p><strong>Decoded:</strong> ${decoded}</p>
                `;
                output.style.color = 'green';
                
            } catch (error) {
                if (error instanceof URIError) {
                    output.innerHTML = `
                        <p style="color: red;"><strong>URIError:</strong> ${error.message}</p>
                        <p style="color: red;">The input contains invalid percent-encoded sequences.</p>
                        <p><em>Example invalid: "%E0%A4%A" (incomplete sequence)</em></p>
                    `;
                    console.error('URI decoding error:', error);
                } else {
                    throw error;
                }
            }
        }
        
        /**
         * Demonstrate when URIError occurs
         */
        function demonstrateURIError() {
            // These will throw URIError:
            const invalidInputs = ['%', '%E0%A4', '%%'];
            
            invalidInputs.forEach(str => {
                try {
                    decodeURIComponent(str);
                } catch (e) {
                    console.log(`"${str}" causes URIError: ${e.message}`);
                }
            });
        }
        
        // Run demonstration
        demonstrateURIError();
    </script>
</body>
</html>
```

**Why**: URIError handling is critical for web applications that process URLs. Invalid escape sequences can crash applications without proper error handling, especially in user-generated content scenarios.

## RegExp Split

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>RegExp Split</title>
</head>
<body>
    <h2>String Split with RegExp</h2>
    
    <div>
        <label>Text to split:</label><br>
        <textarea id="inputText" rows="3" cols="50">apple, banana; cherry|orange   grape</textarea>
    </div>
    
    <div style="margin-top: 10px;">
        <label>Split pattern (RegExp):</label>
        <input type="text" id="patternInput" value="[,;|\s]+" size="30">
    </div>
    
    <div style="margin: 10px 0;">
        <button onclick="splitText()">Split</button>
        <button onclick="splitWithCapture()">Split with Capture Groups</button>
        <button onclick="splitWithLimit()">Split with Limit</button>
    </div>
    
    <div id="output" style="margin-top: 20px; padding: 10px; background: #f5f5f5;"></div>

    <script>
        // PROBLEM: Split strings using regular expressions
        // TIME: O(n) where n is string length SPACE: O(n)
        
        /**
         * Basic split using RegExp
         * Splits on any matching pattern
         */
        function splitText() {
            const text = document.getElementById('inputText').value;
            const pattern = document.getElementById('patternInput').value;
            const output = document.getElementById('output');
            
            try {
                // Create RegExp from string pattern
                const regex = new RegExp(pattern, 'g');
                
                // Split using the pattern
                const parts = text.split(regex);
                
                // Display results
                output.innerHTML = `
                    <p><strong>Pattern:</strong> /${pattern}/g</p>
                    <p><strong>Parts (${parts.length}):</strong></p>
                    <ol>${parts.map(p => `<li>"${p}"</li>`).join('')}</ol>
                `;
                
            } catch (error) {
                output.innerHTML = `<p style="color: red;">Error: ${error.message}</p>`;
            }
        }
        
        /**
         * Split with capturing groups
         * Captured groups are included in the result
         */
        function splitWithCapture() {
            const text = document.getElementById('inputText').value;
            const output = document.getElementById('output');
            
            // Pattern with capture group: splits on separators but keeps them
            const regex = /([,;|\s]+)/;
            
            const parts = text.split(regex);
            
            output.innerHTML = `
                <p><strong>Split with capture group /([,;|\\s]+)/:</strong></p>
                <p><em>Captured separators are included in results</em></p>
                <ol>${parts.map((p, i) => 
                    `<li>"${p}" ${i % 2 === 1 ? '(separator)' : '(text)'}</li>`
                ).join('')}</ol>
            `;
        }
        
        /**
         * Split with limit parameter
         * Limits number of splits
         */
```
