## js_array_fill
```html

    // PROBLEM: Populate array with a value (mutates original)
    // TIME: O(n) SPACE: O(1)

    // Basic usage: fill entire array
    const arr1 = [1, 2, 3, 4, 5];
    arr1.fill(0);
    console.log('Fill with 0:', arr1); // [0, 0, 0, 0, 0]
    
    // Fill with start index
    const arr2 = [1, 2, 3, 4, 5];
    arr2.fill(0, 2); // Start at index 2
    console.log('Fill from index 2:', arr2); // [1, 2, 0, 0, 0]
    
    // Fill with start and end index (end not included)
    const arr3 = [1, 2, 3, 4, 5];
    arr3.fill(0, 1, 4); // Indices 1, 2, 3
    console.log('Fill indices 1-3:', arr3); // [1, 0, 0, 0, 5]
    
    // Create array of specific size and fill
    const zeros = new Array(5).fill(0);
    console.log('New array filled:', zeros); // [0, 0, 0, 0, 0]
    
    // ⚠️ Warning: Object references!
    const objects = new Array(3).fill({ value: 0 });
    objects[0].value = 99;
    console.log('Same object reference!', objects);
    // [{value: 99}, {value: 99}, {value: 99}]
    
    // Solution: Use map for distinct objects
    const distinctObjects = new Array(3).fill(null).map(() => ({ value: 0 }));

  <pre id="output"></pre>

  <script>
    // PROBLEM: Create repeated string patterns without loops
    // TIME: O(n) SPACE: O(n)

    // Basic usage
    const star = '*'.repeat(5);
    console.log(star); // '*****'
    
    const hello = 'Hello '.repeat(3);
    console.log(hello); // 'Hello Hello Hello '
    
    // Zero repeats returns empty string
    console.log('test'.repeat(0)); // ''
    
    // RangeError for negative or Infinity
    // 'test'.repeat(-1);  // RangeError
    // 'test'.repeat(Infinity); // RangeError
    
    // Practical examples
    
    // 1. Text divider
    const divider = '─'.repeat(50);
    console.log(divider);
    
    // 2. Indentation
    function indent(text, levels) {
      return '  '.repeat(levels) + text;
    }
    console.log(indent('Nested item', 3));
    
    // 3. Progress bar
    function progressBar(percent, width = 20) {
      const filled = Math.round(width * percent / 100);
      const empty = width - filled;
      return '[' + '█'.repeat(filled) + '░'.repeat(empty) + '] ' + percent + '%';
    }
    
    // 4. ASCII art
    function christmasTree(height) {
      let tree = '';
      for (let i = 1; i <= height; i++) {
        const spaces = ' '.repeat(height - i);
        const leaves = '▲'.repeat(i);
        tree += spaces + leaves + '\n';
      }
      tree += ' '.repeat(height - 1) + '|||\n';
      return tree;
    }
    
    // 5. Table formatting
    function formatRow(cols, width) {
      return '| ' + cols.map(c => c.padEnd(width)).join(' | ') + ' |';
    }
    
    function tableSeparator(count, width) {
      return '+-' + '-'.repeat(width).repeat(count).split('').join('-') + '-+';
    }
    
    // Display
    document.getElementById('output').textContent = `
Basic: '*'.repeat(5) = "${star}"

Divider:
${divider}

Indentation:
${indent('Level 1', 1)}
${indent('Level 2', 2)}
${indent('Level 3', 3)}

Progress Bars:
${progressBar(25)}
${progressBar(50)}
${progressBar(75)}
${progressBar(100)}

ASCII Art:
${christmasTree(5)}

Table:
${formatRow(['Name', 'Age', 'City'], 10)}
${tableSeparator(3, 10)}
${formatRow(['Alice', '25', 'NYC'], 10)}
${formatRow(['Bob', '30', 'LA'], 10)}
    `.trim();
  </script>
</body>
</html>
```
**Why**: `repeat()` generates repeated patterns without loops—perfect for ASCII art, progress bars, text dividers, indentation, and table formatting. Cleaner and more readable than `for` loops or `Array(n).join()`.
