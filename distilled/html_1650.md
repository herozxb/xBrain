```html
    // PROBLEM: Handle unified pointer input (mouse, touch, pen)
    // TIME: O(1) SPACE: O(1)

    const area = document.getElementById('pointer-area');
    const info = document.getElementById('info');

    // Pointer event types: pointerdown, pointermove, pointerup, 
    // pointerenter, pointerleave, pointercancel

    area.addEventListener('pointerdown', (e) => {
      // e.pointerType: 'mouse', 'touch', 'pen'
      // e.isPrimary: true for primary pointer
      // e.pressure: 0-1 pressure level
      // e.width, e.height: contact geometry
      
      area.style.transform = 'scale(0.98)';
      info.textContent = `Pointer DOWN | Type: ${e.pointerType} | ` +
        `Pressure: ${e.pressure.toFixed(2)} | ID: ${e.pointerId}`;
    });

    area.addEventListener('pointermove', (e) => {
      const rect = area.getBoundingClientRect();
      const x = e.clientX - rect.left;
      const y = e.clientY - rect.top;
      
      info.textContent = `Position: (${x.toFixed(0)}, ${y.toFixed(0)}) | ` +
        `Type: ${e.pointerType} | Pressure: ${e.pressure.toFixed(2)}`;
    });

    area.addEventListener('pointerup', (e) => {
      area.style.transform = 'scale(1)';
      info.textContent = `Pointer UP | Type: ${e.pointerType}`;
    });

    area.addEventListener('pointerenter', () => {
      area.style.boxShadow = '0 0 20px rgba(102, 126, 234, 0.6)';
    });

    area.addEventListener('pointerleave', () => {
      area.style.boxShadow = 'none';
      area.style.transform = 'scale(1)';
      info.textContent = 'Move pointer over the area above';
    });

    // Capture pointer for reliable tracking
    area.addEventListener('pointerdown', (e) => {
      area.setPointerCapture(e.pointerId);
    });

    area.addEventListener('pointerup', (e) => {
      area.releasePointerCapture(e.pointerId);
    });
  </script>
</body>
</html>
```

```html
      <p>Line 4: deltaX handles horizontal scroll.</p>
      <p>Line 5: event.deltaMode indicates unit (pixel, line, page).</p>
      <p>Line 6: Prevent default to override browser scroll.</p>
      <p>Line 7: Combine with passive: false for custom scroll.</p>
      <p>Line 8: Great for custom scrollbars, zoom, carousels.</p>
      <p>Line 9: Remember accessibility - provide alternatives.</p>
      <p>Line 10: End of scrollable content.</p>
    </div>

  </div>

  <script>
    // PROBLEM: Handle mouse wheel and trackpad scroll events
    // TIME: O(1) SPACE: O(1)

    const container = document.getElementById('scroll-container');
    const content = document.getElementById('scrollable-content');
    const indicator = document.getElementById('indicator');

    let currentOffset = 0;
    const maxOffset = content.offsetHeight - container.offsetHeight;

    // Use passive: false to allow preventDefault()
    container.addEventListener('wheel', (e) => {
      // e.deltaY: vertical scroll amount
      // e.deltaX: horizontal scroll amount
      // e.deltaMode: 0=pixels, 1=lines, 2=pages
      // e.ctrlKey: true for pinch-zoom on trackpad

      e.preventDefault(); // Prevent page scroll

      // Normalize scroll speed across browsers
      const scrollMultiplier = e.deltaMode === 1 ? 20 : 1;
      const delta = e.deltaY * scrollMultiplier;

      // Update offset with bounds checking
      currentOffset = Math.max(0, Math.min(maxOffset, currentOffset + delta));

      // Apply transform for smooth scrolling
      content.style.transform = `translateY(-${currentOffset}px)`;

      // Update indicator
      const percentage = maxOffset > 0 ? (currentOffset / maxOffset) * 100 : 0;
      indicator.textContent = `${percentage.toFixed(0)}%`;

      // Detect horizontal scroll (shift+wheel or trackpad)
      if (Math.abs(e.deltaX) > Math.abs(e.deltaY)) {
        console.log('Horizontal scroll detected:', e.deltaX);
      }

      // Detect pinch-zoom (ctrlKey + wheel on trackpad)
      if (e.ctrlKey) {
        console.log('Pinch-zoom gesture detected');
      }
    }, { passive: false });
  </script>

</html>
```

**Why**: Wheel events enable custom scroll behaviors, zoom functionality, and interactive UI controls beyond default browser scrolling.

```html
      const links = nav.querySelectorAll('a');
      
      // Keyboard navigation support
      nav.addEventListener('keydown', (e) => {
        const focused = document.activeElement;
        const linksArray = Array.from(links);
        const currentIndex = linksArray.indexOf(focused);

        if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
          e.preventDefault();
          const nextIndex = (currentIndex + 1) % linksArray.length;
          linksArray[nextIndex].focus();
        }
        
        if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
          e.preventDefault();
          const prevIndex = (currentIndex - 1 + linksArray.length) % linksArray.length;
          linksArray[prevIndex].focus();
        }
      });

      links.forEach(link => {
        link.addEventListener('click', (e) => {
          console.log(`Navigation: ${link.textContent}`);
```

```html
      console.log(`  - Copyright: ${hasCopyright}`);
      console.log(`  - Author info: ${!!hasAuthor}`);
      console.log(`  - Links: ${hasLinks}`);
    });

    // Page footer automatically has role="contentinfo" in HTML5
    // It's a landmark for screen readers
    // Can be used per article/section for local footer info
  </script>
```

**Why**: The footer element semantically marks concluding content, serving as a landmark for screen readers and organizing document structure.

## 9. html5_main_element

```html

  </style>

<body>
  <!-- Skip link for keyboard users -->
  <a href="#main-content" class="skip-link">Skip to main content</a>
  
  <header>

  </header>

    <a href="/about">About</a>

    <p>The <code>&lt;main&gt;</code> element represents the dominant content 
    of the <code>&lt;body&gt;</code>. Important rules:</p>
    
    <ul>
      <li><strong>One per page:</strong> Only one main element per document</li>
      <li><strong>No nesting:</strong> Cannot be inside header, nav, article, aside, or footer</li>
      <li><strong>Excludes sidebars:</strong> Contains only page-specific content</li>
    </ul>
    
    <section>
      <h3>Benefits</h3>
      <ul>
        <li>Screen readers can skip directly to main content</li>
        <li>Improves document outline and semantics</li>
        <li>Helps search engines identify primary content</li>
      </ul>
    </section>
  </main>

  <footer>
    <p>© 2026 My Website</p>
  </footer>

  <script>
    // PROBLEM: Demonstrate main element usage and accessibility
    // TIME: O(1) SPACE: O(1)

    // Get the main element
    const mainElement = document.querySelector('main');

    // Main element is a landmark
    // Screen readers announce it as "main content" or "main landmark"
    console.log('Main element found:', mainElement.id);

    // Focus management for skip link
    document.querySelector('.skip-link').addEventListener('click', (e) => {
      e.preventDefault();
      mainElement.tabIndex = -1; // Make programmatically focusable
      mainElement.focus();
    });

    // Validate: should only be one main element
    const mainElements = document.querySelectorAll('main');
    if (mainElements.length > 1) {
      console.warn('Warning: Multiple main elements detected. Only one allowed.');
    }

    // Check main is not nested in excluded elements
    const excludedParents = mainElement.closest('header, nav, article, aside, footer');
    if (excludedParents) {
      console.error('Error: main element should not be nested in:', excludedParents.tagName);
    }

    // Accessibility: role="main" is implicit in HTML5
    // mainElement.setAttribute('role', 'main'); // Not needed in HTML5
  </script>
</body>
</html>
```

**Why**: The main element identifies the primary content area, enabling screen reader users to skip navigation and access content directly.

```html
    }

  </style>
</head>
<body>
  <h1>HTML5 Figure Element</h1>
  
  <p>The <code>&lt;figure&gt;</code> element represents self-contained content, 
  like illustrations, diagrams, photos, or code listings.</p>

  <!-- Figure with image -->
  <figure>
    <img src="https://via.placeholder.com/600x300?text=Sample+Image" 
         alt="A sample placeholder image demonstrating figure usage"
         width="600" height="300">
    <figcaption>Figure 1: A demonstration of the figure element with an image.</figcaption>
  </figure>

  <!-- Figure with code -->
  <figure class="code">
    <figcaption>Example: Creating a figure element</figcaption>
    <pre><code>&lt;figure&gt;
  &lt;img src="diagram.png" alt="System architecture"&gt;
  &lt;figcaption&gt;System Architecture Overview&lt;/figcaption&gt;
&lt;/figure&gt;</code></pre>
  </figure>

  <!-- Figure with quote -->
  <figure class="quote">
    <blockquote>
      "The figure element provides a way to associate content with its caption, 
      creating a semantic unit that can be positioned independently."
    </blockquote>
    <figcaption>— HTML5 Specification</figcaption>
  </figure>

  <!-- Figure with multiple images -->
  <figure>
    <div style="display: flex; gap: 10px;">
      <img src="https://via.placeholder.com/150?text=1" alt="Thumbnail 1">
      <img src="https://via.placeholder.com/150?text=2" alt="Thumbnail 2">
      <img src="https://via.placeholder.com/150?text=3" alt="Thumbnail 3">
    </div>
    <figcaption>Figure 2: Multiple images can share one caption.</figcaption>
  </figure>

  <script>
    // PROBLEM: Demonstrate figure element with various content types
    // TIME: O(n) SPACE: O(1) where n = number of figures

    const figures = document.querySelectorAll('figure');

    figures.forEach((figure, index) => {
      // Get caption
      const caption = figure.querySelector('figcaption');
      const captionText = caption?.textContent || 'No caption';
      
      // Determine figure content type
      const hasImage = figure.querySelector('img');
      const hasCode = figure.querySelector('pre, code');
      const hasQuote = figure.querySelector('blockquote');
      
      let type = 'other';
      if (hasImage) type = 'image';
      else if (hasCode) type = 'code';
      else if (hasQuote) type = 'quote';
      
      console.log(`Figure ${index + 1}: [${type}] "${captionText.substring(0, 50)}..."`);
      
      // Add keyboard navigation for accessibility
      figure.tabIndex = 0;
      figure.addEventListener('keydown', (e) => {
        if (e.key === 'Enter') {
          // Could open lightbox, copy code, etc.
          console.log(`Figure ${index + 1} activated`);
        }
      });
    });

    // Figures are self-contained - can be moved without losing context
    // Figcaption provides accessible description
    // Screen readers announce figure boundaries and captions
  </script>
</body>
</html>
```

**Why**: The figure element semantically groups self-contained content with its caption, improving content organization and accessibility for diagrams, images, code, and quotes.
