## html_link_rel_author

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <!--
        PROBLEM: Link to author information page
        TIME: O(1) SPACE: O(1)
    -->
    <meta charset="UTF-8">
    <title>Link Rel Author Example</title>
    <!-- Link to the author's profile or about page -->
    <link rel="author" href="https://example.com/author/john-doe">
</head>
<body>
    <article>
        <h1>Article Title</h1>
        <p>Written by <a href="https://example.com/author/john-doe" rel="author">John Doe</a></p>
        <p>Article content goes here...</p>
    </article>
    
    <script>
        // JavaScript to demonstrate accessing author link information
        document.addEventListener('DOMContentLoaded', function() {
            // Get all links with rel="author"
            const authorLinks = document.querySelectorAll('a[rel="author"], link[rel="author"]');
            
            authorLinks.forEach(link => {
                console.log('Author link found:', link.href);
            });
        });
    </script>
</body>
</html>
```

**Why**: The `rel="author"` attribute semantically indicates a link to the author's profile, improving SEO and content attribution.

## html_link_rel_help

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <!--
        PROBLEM: Provide help documentation link
        TIME: O(1) SPACE: O(1)
    -->
    <meta charset="UTF-8">
    <title>Link Rel Help Example</title>
    <!-- Link to help documentation for this page -->
    <link rel="help" href="https://example.com/help/faq">
</head>
<body>
    <form id="userForm">
        <label for="username">Username:</label>
        <input type="text" id="username" name="username" required>
        
        <!-- Help link specific to this form -->
        <a href="https://example.com/help/username-rules" rel="help">?</a>
        
        <button type="submit">Submit</button>
    </form>
    
    <script>
        // JavaScript to handle help link clicks
        document.addEventListener('DOMContentLoaded', function() {
            const helpLinks = document.querySelectorAll('[rel="help"]');
            
            helpLinks.forEach(link => {
                link.addEventListener('click', function(e) {
                    // Open help in new window for better UX
                    e.preventDefault();
                    const helpWindow = window.open(this.href, 'help', 'width=600,height=400');
                    helpWindow.focus();
                });
            });
        });
    </script>
</body>
</html>
```

**Why**: The `rel="help"` attribute links to contextual help documentation, improving user experience and accessibility.

## css_grid_gap

```html
<!DOCTYPE html>
<html lang="en">

    <!--
        PROBLEM: Create consistent spacing between grid items
        TIME: O(1) SPACE: O(1)

    <style>
        /* Grid container with uniform gap */
        .grid-container {
            display: grid;
            grid-template-columns: repeat(3, 1fr);
            /* Sets both row and column gaps to 20px */
            gap: 20px;
            padding: 20px;
            background-color: #f0f0f0;
        }
        
        .grid-item {
            background-color: #4CAF50;
            color: white;
            padding: 20px;

        }
    </style>
</head>
<body>
    <h1>Grid Gap Example</h1>
    <div class="grid-container">
        <div class="grid-item">Item 1</div>
        <div class="grid-item">Item 2</div>
        <div class="grid-item">Item 3</div>
        <div class="grid-item">Item 4</div>
        <div class="grid-item">Item 5</div>
        <div class="grid-item">Item 6</div>
    </div>
    
    <script>
        // JavaScript to dynamically adjust gap
        const container = document.querySelector('.grid-container');
        
        function changeGap(newGap) {
            container.style.gap = newGap + 'px';
            console.log('Gap changed to:', newGap + 'px');
        }
        
        // Example: change gap every 2 seconds
        let currentGap = 20;
        setInterval(() => {
            currentGap = currentGap === 20 ? 40 : 20;
            changeGap(currentGap);
        }, 2000);
    </script>

```

**Why**: The `gap` property provides clean, consistent spacing between grid items without margins, simplifying layout management.

## css_grid_row_gap

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <!--
        PROBLEM: Create vertical spacing between grid rows
        TIME: O(1) SPACE: O(1)
    -->
    <meta charset="UTF-8">
    <title>CSS Grid Row Gap Example</title>
    <style>
        /* Grid with row-specific spacing */
        .grid-container {
            display: grid;
            grid-template-columns: repeat(3, 1fr);
            /* Only vertical spacing between rows */
            row-gap: 30px;
            column-gap: 10px;
            padding: 20px;
        }
        
        .grid-item {
            background: linear-gradient(135deg, #667eea 0%, #764ba2 100%);
            color: white;
            padding: 15px;
            text-align: center;
        }
    </style>
</head>
<body>
    <h1>Grid Row Gap Example</h1>
    <div class="grid-container">
        <div class="grid-item">Row 1, Col 1</div>
        <div class="grid-item">Row 1, Col 2</div>
        <div class="grid-item">Row 1, Col 3</div>
        <div class="grid-item">Row 2, Col 1</div>
        <div class="grid-item">Row 2, Col 2</div>
        <div class="grid-item">Row 2, Col 3</div>
    </div>
    
    <script>
        // JavaScript to calculate total row gaps
        document.addEventListener('DOMContentLoaded', function() {
            const container = document.querySelector('.grid-container');
            const computedStyle = window.getComputedStyle(container);
            const rowGap = computedStyle.rowGap;
            
            console.log('Row gap:', rowGap);
            
            // Count rows (3 items per row)
            const items = container.children.length;
            const rows = Math.ceil(items / 3);
            const totalRowGap = (rows - 1) * parseInt(rowGap);
            
            console.log('Total vertical spacing:', totalRowGap + 'px');
        });
    </script>
</body>
</html>
```

**Why**: `row-gap` allows independent control of vertical spacing, useful for creating vertically emphasized layouts.

```html
            <p>Description of feature 2</p>
        </div>
        <div class="card">
            <h3>Feature 3</h3>
            <p>Description of feature 3</p>
        </div>
        <div class="card">
            <h3>Feature 4</h3>
            <p>Description of feature 4</p>
        </div>
    </div>
    
    <script>
        // JavaScript to dynamically adjust column gap based on screen width
        function adjustColumnGap() {
            const container = document.querySelector('.grid-container');
            const screenWidth = window.innerWidth;
            
            if (screenWidth < 768) {
                container.style.columnGap = '20px';
            } else if (screenWidth < 1024) {
                container.style.columnGap = '30px';
            } else {
                container.style.columnGap = '40px';
            }
        }
        
        window.addEventListener('resize', adjustColumnGap);
        adjustColumnGap(); // Initial call
    </script>
</body>
</html>
```

**Why**: `column-gap` enables responsive horizontal spacing control, improving readability across different screen sizes.

## css_color_hsl

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <!--
        PROBLEM: Create colors using HSL (Hue, Saturation, Lightness) model
        TIME: O(1) SPACE: O(1)
    -->
    <meta charset="UTF-8">
    <title>CSS HSL Color Example</title>
    <style>
        /* HSL: hsl(hue, saturation%, lightness%)
           hue: 0-360 (color wheel)
           saturation: 0% (gray) - 100% (full color)
           lightness: 0% (black) - 100% (white)
        */
        
        .color-palette {
            display: flex;
            gap: 10px;
            padding: 20px;
        }
        
        .color-box {
            width: 100px;
            height: 100px;
            display: flex;
            align-items: center;
            justify-content: center;
            color: white;
            font-weight: bold;
            border-radius: 8px;
            text-shadow: 1px 1px 2px rgba(0,0,0,0.5);
        }
        
        /* Primary colors using HSL */
        .red { background-color: hsl(0, 100%, 50%); }
        .green { background-color: hsl(120, 100%, 50%); }
        .blue { background-color: hsl(240, 100%, 50%); }
        
        /* Variations by adjusting lightness */
        .light-blue { background-color: hsl(240, 100%, 70%); }
        .dark-blue { background-color: hsl(240, 100%, 30%); }
        
        /* Variations by adjusting saturation */
        .vivid-purple { background-color: hsl(270, 100%, 50%); }
        .muted-purple { background-color: hsl(270, 50%, 50%); }
    </style>
</head>
<body>
    <h1>HSL Color Model</h1>
    
    <h2>Primary Colors</h2>
    <div class="color-palette">
        <div class="color-box red">Red<br>0°</div>
        <div class="color-box green">Green<br>120°</div>
        <div class="color-box blue">Blue<br>240°</div>
    </div>
    
    <h2>Lightness Variations</h2>
    <div class="color-palette">
        <div class="color-box dark-blue">Dark<br>30%</div>
        <div class="color-box blue">Normal<br>50%</div>
        <div class="color-box light-blue">Light<br>70%</div>
    </div>
    
    <h2>Saturation Variations</h2>
    <div class="color-palette">
        <div class="color-box vivid-purple">Vivid<br>100%</div>
        <div class="color-box muted-purple">Muted<br>50%</div>
    </div>
    
    <script>
        // JavaScript to generate HSL color palette
        function generateHSLPalette(hue, steps = 10) {
            const palette = [];
            for (let i = 0; i < steps; i++) {
                const lightness = 10 + (i * 8); // 10% to 90%
                palette.push(`hsl(${hue}, 100%, ${lightness}%)`);
            }
            return palette;
        }
        
        // Create dynamic color palette
        const container = document.createElement('div');
        container.style.display = 'flex';
        container.style.gap = '5px';
        container.style.padding = '20px';
        
        const bluePalette = generateHSLPalette(210, 10);
        bluePalette.forEach((color, index) => {
            const box = document.createElement('div');
            box.style.width = '50px';
            box.style.height = '50px';
            box.style.backgroundColor = color;
            box.style.borderRadius = '5px';
            box.title = color;
            container.appendChild(box);
        });
        
        document.body.appendChild(container);
        console.log('Generated palette:', bluePalette);
    </script>
</body>
</html>
```

**Why**: HSL provides intuitive color manipulation through hue, saturation, and lightness, making it easier to create color schemes and variations than RGB.

## css_color_hsla

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <!--
        PROBLEM: Create colors with alpha transparency using HSLA
        TIME: O(1) SPACE: O(1)
    -->
    <meta charset="UTF-8">
    <title>CSS HSLA Color Example</title>
    <style>
        /* HSLA: hsla(hue, saturation%, lightness%, alpha)
           alpha: 0 (transparent) - 1 (opaque)
        */
        
        .demo-container {
            position: relative;
            width: 100%;
            height: 300px;
            background: linear-gradient(45deg, #ccc 25%, transparent 25%),
                        linear-gradient(-45deg, #ccc 25%, transparent 25%),
                        linear-gradient(45deg, transparent 75%, #ccc 75%),
                        linear-gradient(-45deg, transparent 75%, #ccc 75%);
            background-size: 20px 20px;
            background-position: 0 0, 0 10px, 10px -10px, -10px 0px;
            background-color: white;
        }
        
        .overlay-box {
            position: absolute;
            width: 150px;
            height: 150px;
            border-radius: 10px;
            display: flex;
            align-items: center;
            justify-content: center;
            color: white;
            font-weight: bold;
            text-shadow: 1px 1px 2px rgba(0,0,0,0.3);
        }
        
        /* Different alpha levels */
        .alpha-1 { 
            background-color: hsla(0, 100%, 50%, 1); 
            top: 20px; 
            left: 20px; 
        }
        .alpha-07 { 
            background-color: hsla(0, 100%, 50%, 0.7); 
            top: 20px; 
            left: 100px; 
        }
        .alpha-04 { 
            background-color: hsla(0, 100%, 50%, 0.4); 
            top: 20px; 
            left: 180px; 
        }
        .alpha-01 { 
            background-color: hsla(0, 100%, 50%, 0.1); 
            top: 20px; 
            left: 260px; 
        }
        
        /* Glassmorphism effect */
        .glass-card {
            background-color: hsla(240, 50%, 50%, 0.3);
            backdrop-filter: blur(10px);
            border: 1px solid hsla(240, 50%, 50%, 0.5);
            padding: 20px;
            border-radius: 15px;
            margin: 20px;
            color: white;
        }
        
        /* Gradient overlay */
        .gradient-overlay {
            position: absolute;
            top: 0;
            right: 0;
            width: 200px;
            height: 200px;
            background: linear-gradient(
                135deg,
                hsla(120, 100%, 50%, 0.8) 0%,
                hsla(240, 100%, 50%, 0.8) 100%
            );
            border-radius: 10px;
        }
    </style>
</head>
<body>
    <h1>HSLA Color with Alpha Transparency</h1>
    
    <h2>Alpha Transparency Levels</h2>
    <div class="demo-container">
        <div class="overlay-box alpha-1">Alpha: 1.0</div>
        <div class="overlay-box alpha-07">Alpha: 0.7</div>
        <div class="overlay-box alpha-04">Alpha: 0.4</div>
        <div class="overlay-box alpha-01">Alpha: 0.1</div>
        <div class="gradient-overlay">Gradient</div>
    </div>
    
    <div class="glass-card">
        <h2>Glassmorphism Card</h2>
        <p>This card uses HSLA for semi-transparent background with blur effect.</p>
    </div>
    
    <script>
        // JavaScript to dynamically adjust alpha on hover
        document.querySelectorAll('.overlay-box').forEach(box => {
            box.addEventListener('mouseenter', function() {
                const currentBg = this.style.backgroundColor || 
                                  window.getComputedStyle(this).backgroundColor;
                this.dataset.originalBg = currentBg;
                this.style.backgroundColor = currentBg.replace(/[\d.]+\)$/, '0.9)');
            });
            
            box.addEventListener('mouseleave', function() {
                this.style.backgroundColor = this.dataset.originalBg;
            });
        });
        
        // Function to generate HSLA color string
        function hsla(h, s, l, a) {
            return `hsla(${h}, ${s}%, ${l}%, ${a})`;
        }
        
        // Example usage
        console.log('Semi-transparent red:', hsla(0, 100, 50, 0.5));
        console.log('Light semi-transparent blue:', hsla(240, 100, 70, 0.6));
    </script>
</body>
</html>
```

**Why**: HSLA enables transparency control through the alpha channel, essential for overlays, glassmorphism effects, and layered designs while maintaining HSL's intuitive color manipulation.
