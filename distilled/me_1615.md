## 1. Maximum Sum Subarray of Size K

```python
# PROBLEM: Find the maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)
def max_sum_subarray_k(nums, k):
    """
    Sliding window approach to find maximum sum subarray of fixed size k.
    Window slides one position at a time, adding new element and removing old.
    """
    # Edge case: array smaller than window
    if len(nums) < k:
        return -1
    
    # Calculate sum of first window
    window_sum = sum(nums[:k])
    max_sum = window_sum
    
    # Slide window: subtract leftmost, add rightmost
    for i in range(k, len(nums)):
        window_sum = window_sum - nums[i - k] + nums[i]
        max_sum = max(max_sum, window_sum)
    
    return max_sum
```
**Why**: Classic sliding window pattern. Avoids recalculating entire window sum each time - just update incrementally. O(n) vs O(nk) brute force.

## 2. Three Sum

```python
# PROBLEM: Find all unique triplets that sum to zero
# TIME: O(n²) SPACE: O(1) excluding output
def three_sum(nums):
    """
    Two pointers technique on sorted array.
    Fix one element, then use two pointers to find complementary pairs.
    """
    nums.sort()  # Required for two-pointer approach
    result = []
    n = len(nums)
    
    for i in range(n - 2):
        # Skip duplicates for first element
        if i > 0 and nums[i] == nums[i - 1]:
            continue
        
        # Two pointers: one after fixed element, one at end
        left, right = i + 1, n - 1
        
        while left < right:
            total = nums[i] + nums[left] + nums[right]
            
            if total == 0:
                result.append([nums[i], nums[left], nums[right]])
                # Skip duplicates for second element
                while left < right and nums[left] == nums[left + 1]:
                    left += 1
                # Skip duplicates for third element
                while left < right and nums[right] == nums[right - 1]:
                    right -= 1
                left += 1
                right -= 1
            elif total < 0:
                left += 1  # Need larger sum
            else:
                right -= 1  # Need smaller sum
    
    return result
```
**Why**: Two pointers on sorted array enables efficient pair-finding. Sorting enables skipping duplicates easily.

## 3. N-Queens

```python
# PROBLEM: Place n queens on n×n board so no two attack each other
# TIME: O(n!) SPACE: O(n²) for board
def solve_n_queens(n):
    """
    Backtracking with constraint checking.
    Place queens row by row, backtrack when conflict detected.
    """
    def is_safe(board, row, col):
        """Check if queen at (row, col) is safe from all previous queens."""
        # Check column (above current row)
        for i in range(row):
            if board[i][col] == 'Q':
                return False
        
        # Check upper-left diagonal
        i, j = row - 1, col - 1
        while i >= 0 and j >= 0:
            if board[i][j] == 'Q':
                return False
            i -= 1
            j -= 1
        
        # Check upper-right diagonal
        i, j = row - 1, col + 1
        while i >= 0 and j < n:
            if board[i][j] == 'Q':
                return False
            i -= 1
            j += 1
        
        return True
    
    def backtrack(board, row):
        """Try placing queen in each column of current row."""
        if row == n:
            # All queens placed successfully
            solutions.append([''.join(r) for r in board])
            return
        
        for col in range(n):
            if is_safe(board, row, col):
                # Place queen
                board[row][col] = 'Q'
                # Recurse to next row
                backtrack(board, row + 1)
                # Remove queen (backtrack)
                board[row][col] = '.'
    
    solutions = []
    board = [['.' for _ in range(n)] for _ in range(n)]
    backtrack(board, 0)
    return solutions
```
**Why**: Classic backtracking. Prunes invalid branches early. Constraint propagation would optimize further.

```python
# PROBLEM: Count connected islands of '1's in a 2D grid
# TIME: O(m×n) SPACE: O(m×n) worst case for recursion stack
class UnionFind:
    """Union-Find (Disjoint Set Union) data structure."""
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.count = 0  # Number of connected components
    
    def find(self, x):
        """Find root with path compression."""
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        """Union by rank - attach smaller tree to larger."""
        root_x, root_y = self.find(x), self.find(y)
        if root_x == root_y:
            return  # Already connected
        
        # Union by rank
        if self.rank[root_x] < self.rank[root_y]:
            self.parent[root_x] = root_y
        elif self.rank[root_x] > self.rank[root_y]:
            self.parent[root_y] = root_x
        else:
            self.parent[root_y] = root_x
            self.rank[root_x] += 1
        
        self.count -= 1  # Merged two components

def num_islands(grid):
    """
    Use Union-Find to merge adjacent land cells.
    Each land cell starts as its own component, then union with neighbors.
    """
    if not grid or not grid[0]:
        return 0
    
    rows, cols = len(grid), len(grid[0])
    uf = UnionFind(rows * cols)
    
    # Count initial land cells
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == '1':
                uf.count += 1
    
    # Directions: right and down (to avoid duplicate unions)
    directions = [(0, 1), (1, 0)]
    
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == '1':
                # Convert 2D to 1D index
                idx = r * cols + c
                for dr, dc in directions:
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] == '1':
                        neighbor_idx = nr * cols + nc
                        uf.union(idx, neighbor_idx)
    
    return uf.count
```
**Why**: Union-Find excels at dynamic connectivity. Path compression + union by rank gives near-constant amortized operations.

## 5. Longest Common Subsequence

```python
# PROBLEM: Find length of longest subsequence common to two strings
# TIME: O(m×n) SPACE: O(min(m,n)) with space optimization
def longest_common_subsequence(text1, text2):
    """
    Dynamic programming with space optimization.
    dp[i][j] = LCS length of text1[:i] and text2[:j]
    Only need previous row, so use two 1D arrays.
    """
    # Ensure text2 is shorter for space optimization
    if len(text1) < len(text2):
        text1, text2 = text2, text1
    
    m, n = len(text1), len(text2)
    # prev and curr represent consecutive rows of DP table
    prev = [0] * (n + 1)
    
    for i in range(1, m + 1):
        curr = [0] * (n + 1)
        for j in range(1, n + 1):
            if text1[i - 1] == text2[j - 1]:
                # Characters match: extend previous LCS
                curr[j] = prev[j - 1] + 1
            else:
                # Take max of excluding one character from either string
                curr[j] = max(prev[j], curr[j - 1])
        prev = curr
    
    return prev[n]
```
**Why**: DP classic. Space optimization reduces from O(mn) to O(min(m,n)). Recurrence: match extends, mismatch takes max of subproblems.

## 6. Dijkstra's Shortest Path

```python
# PROBLEM: Find shortest path from source to all nodes in weighted graph
# TIME: O((V + E) log V) SPACE: O(V)
import heapq

def dijkstra(n, edges, source):
    """
    Dijkstra's algorithm using min-heap.
    Greedily always process the node with smallest known distance.
    Works only with non-negative edge weights.
    """
    # Build adjacency list
    graph = {i: [] for i in range(n)}
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))  # Remove for directed graph
    
    # Distance array initialized to infinity
    dist = [float('inf')] * n
    dist[source] = 0
    
    # Min-heap: (distance, node)
    heap = [(0, source)]
    
    while heap:
        d, u = heapq.heappop(heap)
        
        # Skip if we've already found a better path
        if d > dist[u]:
            continue
        
        # Relax all edges from current node
        for v, weight in graph[u]:
            new_dist = dist[u] + weight
            if new_dist < dist[v]:
                dist[v] = new_dist
                heapq.heappush(heap, (new_dist, v))
    
    return dist
```
**Why**: Min-heap ensures we always process the closest unvisited node. Optimal for non-negative weights. Greedy choice is provably correct.

## 7. Lowest Common Ancestor in Binary Tree

```python
# PROBLEM: Find lowest common ancestor of two nodes in binary tree
# TIME: O(n) SPACE: O(h) where h is tree height
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def lowest_common_ancestor(root, p, q):
    """
    Recursive approach: LCA is where paths to p and q diverge.
    If one node is ancestor of other, that node is LCA.
    """
    # Base case: reached end or found one of the nodes
    if root is None or root == p or root == q:
        return root
    
    # Recurse on both subtrees
    left = lowest_common_ancestor(root.left, p, q)
    right = lowest_common_ancestor(root.right, p, q)
    
    # If both sides returned non-null, current node is LCA
    if left and right:
        return root
    
    # Otherwise, return whichever side found a node
    return left if left else right
```
**Why**: Post-order traversal naturally finds where paths diverge. Clean recursion handles all cases including when one node is ancestor of other.

## 8. Top K Frequent Elements

```python
# PROBLEM: Find k most frequent elements in array
# TIME: O(n log k) SPACE: O(n)
import heapq
from collections import Counter

def top_k_frequent(nums, k):
    """
    Use min-heap of size k to track top k elements.
    Counter for frequencies, then heap maintains k largest.
    """
    # Count frequencies: O(n)
    freq = Counter(nums)
    
    # Min-heap of size k: O(n log k)
    # Store (frequency, element) pairs
    heap = []
    
    for num, count in freq.items():
        heapq.heappush(heap, (count, num))
        # Keep only top k by removing smallest
        if len(heap) > k:
            heapq.heappop(heap)
    
    # Extract elements from heap
    return [num for count, num in heap]
```
**Why**: Min-heap of size k avoids sorting all elements. O(n log k) vs O(n log n) for full sort. Alternative: bucket sort for O(n) when frequencies bounded.

## 9. Word Dictionary with Wildcard

```python
# PROBLEM: Add words and search with '.' wildcard support
# TIME: O(m) add, O(26^w × m) search where w = number of wildcards
# SPACE: O(total characters added)
class TrieNode:
    """Trie node with children map and end-of-word flag."""
    def __init__(self):
        self.children = {}  # char -> TrieNode
        self.is_word = False

class WordDictionary:
    """
    Trie structure for efficient prefix matching.
    Wildcard search requires recursive backtracking.
    """
    def __init__(self):
        self.root = TrieNode()
    
    def add_word(self, word):
        """Insert word into trie."""
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_word = True
    
    def search(self, word):
        """Search with '.' wildcard support."""
        def dfs(node, index):
            # Base case: reached end of word
            if index == len(word):
                return node.is_word
            
            char = word[index]
            
            if char == '.':
                # Wildcard: try all children
                for child in node.children.values():
                    if dfs(child, index + 1):
                        return True
                return False
            else:
                # Regular character: must match exactly
                if char not in node.children:
                    return False
                return dfs(node.children[char], index + 1)
        
        return dfs(self.root, 0)
```
**Why**: Trie provides O(m) prefix lookup. Wildcard forces exponential worst case but practical for sparse tries. Space-efficient for common prefixes.

## 10. LRU Cache

```python
# PROBLEM: Implement cache with O(1) get/put and LRU eviction
# TIME: O(1) for get and put SPACE: O(capacity)
from collections import OrderedDict

class LRUCache:
    """
    Ordered dictionary maintains insertion order.
    Move to end on access to implement LRU policy.
    """
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = OrderedDict()  # Maintains access order
    
    def get(self, key):
        """Get value and mark as recently used."""
        if key not in self.cache:
            return -1
        
        # Move to end (most recently used position)
        self.cache.move_to_end(key)
        return self.cache[key]
    
    def put(self, key, value):
        """Insert/update and evict LRU if over capacity."""
        if key in self.cache:
            # Update existing: move to end
            self.cache.move_to_end(key)
        else:
            # New key: check capacity
            if len(self.cache) >= self.capacity:
                # Evict least recently used (first item)
                self.cache.popitem(last=False)
        
        self.cache[key] = value
```
**Why**: OrderedDict combines hash table + doubly linked list internally. O(1) access and order maintenance. Alternative: implement doubly linked list manually for interview settings.
