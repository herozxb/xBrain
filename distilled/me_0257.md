## [Divide and Conquer - Maximum Subarray]
```python
# PROBLEM: Find the contiguous subarray with maximum sum using divide and conquer
# TIME: O(n log n) SPACE: O(log n) - recursion stack
def max_subarray_divide_conquer(nums, left=0, right=None):
    # Initialize right boundary on first call
    if right is None:
        right = len(nums) - 1
    
    # Base case: single element is its own max subarray
    if left == right:
        return nums[left], left, right
    
    # Divide: split array into two halves
    mid = (left + right) // 2
    
    # Conquer: recursively find max in left and right halves
    left_max, left_start, left_end = max_subarray_divide_conquer(nums, left, mid)
    right_max, right_start, right_end = max_subarray_divide_conquer(nums, mid + 1, right)
    
    # Cross-sum: find max crossing the midpoint
    # Left side from midpoint
    left_sum = float('-inf')
    current_sum = 0
    cross_left = mid
    for i in range(mid, left - 1, -1):
        current_sum += nums[i]
        if current_sum > left_sum:
            left_sum = current_sum
            cross_left = i
    
    # Right side from midpoint + 1
    right_sum = float('-inf')
    current_sum = 0
    cross_right = mid + 1
    for i in range(mid + 1, right + 1):
        current_sum += nums[i]
        if current_sum > right_sum:
            right_sum = current_sum
            cross_right = i
    
    cross_max = left_sum + right_sum
    
    # Combine: return the maximum of three candidates
    if left_max >= right_max and left_max >= cross_max:
        return left_max, left_start, left_end
    elif right_max >= left_max and right_max >= cross_max:
        return right_max, right_start, right_end
    else:
        return cross_max, cross_left, cross_right

# Example usage
arr = [-2, 1, -3, 4, -1, 2, 1, -5, 4]
max_sum, start, end = max_subarray_divide_conquer(arr)
print(f"Max sum: {max_sum}, Subarray: {arr[start:end+1]}")
```
**Why**: Demonstrates classic divide and conquer - split problem, solve subproblems, combine results. Useful for parallelization and understanding recursive decomposition.

## [AVL Tree - Self-Balancing BST]
```python
# PROBLEM: Implement AVL tree with automatic balancing
# TIME: O(log n) for insert/delete/search SPACE: O(n)
class AVLNode:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None
        self.height = 1  # Height of node for balance factor

class AVLTree:
    def get_height(self, node):
        # Return height or 0 if node is None
        return node.height if node else 0
    
    def get_balance(self, node):
        # Balance factor = left_height - right_height
        # Positive: left-heavy, Negative: right-heavy
        return self.get_height(node.left) - self.get_height(node.right) if node else 0
    
    def update_height(self, node):
        # Height = 1 + max of children heights
        node.height = 1 + max(self.get_height(node.left), self.get_height(node.right))
    
    def rotate_right(self, y):
        # Right rotation for left-heavy tree
        # y becomes child of x
        x = y.left
        T2 = x.right
        
        # Perform rotation
        x.right = y
        y.left = T2
        
        # Update heights (y first, then x)
        self.update_height(y)
        self.update_height(x)
        
        return x  # New root
    
    def rotate_left(self, x):
        # Left rotation for right-heavy tree
        # x becomes child of y
        y = x.right
        T2 = y.left
        
        # Perform rotation
        y.left = x
        x.right = T2
        
        # Update heights (x first, then y)
        self.update_height(x)
        self.update_height(y)
        
        return y  # New root
    
    def insert(self, node, key):
        # Standard BST insertion
        if not node:
            return AVLNode(key)
        
        if key < node.key:
            node.left = self.insert(node.left, key)
        elif key > node.key:
            node.right = self.insert(node.right, key)
        else:
            return node  # Duplicate keys not allowed
        
        # Update height of current node
        self.update_height(node)
        
        # Get balance factor to check if unbalanced
        balance = self.get_balance(node)
        
        # Case 1: Left Left (single right rotation)
        if balance > 1 and key < node.left.key:
            return self.rotate_right(node)
        
        # Case 2: Right Right (single left rotation)
        if balance < -1 and key > node.right.key:
            return self.rotate_left(node)
        
        # Case 3: Left Right (left rotation then right rotation)
        if balance > 1 and key > node.left.key:
            node.left = self.rotate_left(node.left)
            return self.rotate_right(node)
        
        # Case 4: Right Left (right rotation then left rotation)
        if balance < -1 and key < node.right.key:
            node.right = self.rotate_right(node.right)
            return self.rotate_left(node)
        
        return node  # Already balanced
    
    def inorder(self, node):
        # In-order traversal to verify BST property
        if node:
            self.inorder(node.left)
            print(f"{node.key}(h={node.height}, b={self.get_balance(node)})", end=" ")
            self.inorder(node.right)

# Example usage
avl = AVLTree()
root = None
for key in [10, 20, 30, 40, 50, 25]:
    root = avl.insert(root, key)
    print(f"\nAfter inserting {key}:")
    avl.inorder(root)
```
**Why**: AVL trees guarantee O(log n) operations by maintaining balance. Critical for applications requiring guaranteed performance (databases, memory allocators).

## [Chain of Responsibility - Request Processing]
```python
# PROBLEM: Process requests through a chain of handlers
# TIME: O(n) where n is chain length SPACE: O(1) per handler
from abc import ABC, abstractmethod

class Handler(ABC):
    def __init__(self):
        self.next_handler = None  # Next handler in chain
    
    def set_next(self, handler):
        # Set the next handler and return it for chaining
        self.next_handler = handler
        return handler
    
    @abstractmethod
    def handle(self, request):
        # Each handler must implement this
        pass
    
    def pass_to_next(self, request):
        # Pass request to next handler if exists
        if self.next_handler:
            return self.next_handler.handle(request)
        return None

class AuthHandler(Handler):
    def handle(self, request):
        # Check if user is authenticated
        if not request.get('authenticated', False):
            return "❌ Authentication failed"
        print("✓ Authentication passed")
        return self.pass_to_next(request)

class RateLimitHandler(Handler):
    def __init__(self, max_requests=5):
        super().__init__()
        self.request_count = {}
        self.max_requests = max_requests
    
    def handle(self, request):
        # Check rate limiting
        user_id = request.get('user_id', 'unknown')
        count = self.request_count.get(user_id, 0)
        
        if count >= self.max_requests:
            return f"❌ Rate limit exceeded for user {user_id}"
        
        self.request_count[user_id] = count + 1
        print(f"✓ Rate limit check passed ({count + 1}/{self.max_requests})")
        return self.pass_to_next(request)

class ValidationHandler(Handler):
    def handle(self, request):
        # Validate request data
        required_fields = ['user_id', 'action']
        for field in required_fields:
            if field not in request:
                return f"❌ Missing required field: {field}"
        
        print(f"✓ Validation passed for action: {request['action']}")
        return self.pass_to_next(request)

class LogHandler(Handler):
    def handle(self, request):
        # Log the request
        print(f"📝 Logging request from user {request.get('user_id')}")
        return self.pass_to_next(request) or "✅ Request processed successfully"

# Build the chain
auth = AuthHandler()
rate_limit = RateLimitHandler(max_requests=3)
validation = ValidationHandler()
logger = LogHandler()

# Chain: auth -> rate_limit -> validation -> logger
auth.set_next(rate_limit).set_next(validation).set_next(logger)

# Test the chain
requests = [
    {'user_id': 'user1', 'authenticated': True, 'action': 'read'},
    {'user_id': 'user1', 'authenticated': True, 'action': 'write'},
    {'user_id': 'user1', 'authenticated': True, 'action': 'delete'},
    {'user_id': 'user1', 'authenticated': True, 'action': 'update'},  # Rate limited
    {'user_id': 'user2', 'authenticated': False, 'action': 'read'},   # Auth failed
]

for i, req in enumerate(requests, 1):
    print(f"\n--- Request {i} ---")
    result = auth.handle(req)
    print(f"Result: {result}")
```
**Why**: Decouples senders from receivers. Each handler focuses on single responsibility. Easy to add/remove handlers without changing client code.

## [Skiplist - Probabilistic Data Structure]
```python
# PROBLEM: Implement skiplist for O(log n) average search/insert
# TIME: O(log n) average, O(n) worst SPACE: O(n log n) average
import random

class SkipNode:
    def __init__(self, key=None, level=0):
        self.key = key
        self.forward = [None] * level  # Pointers to next nodes at each level

class SkipList:
    def __init__(self, max_level=16, p=0.5):
        self.max_level = max_level  # Maximum height of skiplist
        self.p = p  # Probability for promoting to higher level
        self.level = 1  # Current highest level
        self.head = SkipNode(level=self.max_level)  # Head node with max level
    
    def random_level(self):
        # Randomly determine node height using coin flips
        level = 1
        while random.random() < self.p and level < self.max_level:
            level += 1
        return level
    
    def insert(self, key):
        # Array to track update path
        update = [None] * self.max_level
        current = self.head
        
        # Find position to insert (top-down, left-to-right)
        for i in range(self.level - 1, -1, -1):
            while current.forward[i] and current.forward[i].key < key:
                current = current.forward[i]
            update[i] = current  # Track the node before insertion point
        
        # Create new node with random level
        new_level = self.random_level()
        new_node = SkipNode(key, new_level)
        
        # If new level exceeds current, initialize new levels
        if new_level > self.level:
            for i in range(self.level, new_level):
                update[i] = self.head
            self.level = new_level
        
        # Insert node by updating forward pointers
        for i in range(new_level):
            new_node.forward[i] = update[i].forward[i]
            update[i].forward[i] = new_node
    
    def search(self, key):
        current = self.head
        
        # Search from top level down
        for i in range(self.level - 1, -1, -1):
            while current.forward[i] and current.forward[i].key < key:
                current = current.forward[i]
        
        # Check if we found the key at level 0
        current = current.forward[0]
        if current and current.key == key:
            return True
        return False
    
    def display(self):
        # Print skiplist level by level
        for i in range(self.level - 1, -1, -1):
            current = self.head.forward[i]
            print(f"Level {i}: ", end=" ")
            while current:
                print(f"{current.key}", end=" -> ")
                current = current.forward[i]
            print("END")

# Example usage
sl = SkipList()
keys = [3, 6, 7, 9, 12, 19, 17, 26, 21, 25]

print("Inserting keys:", keys)
for key in keys:
    sl.insert(key)

print("\nSkiplist structure:")
sl.display()

print("\nSearch results:")
for test_key in [6, 15, 19, 100]:
    found = sl.search(test_key)
    print(f"Key {test_key}: {'Found ✓' if found else 'Not found ✗'}")
```
**Why**: Alternative to balanced trees with simpler implementation. Used in Redis ZSET, LevelDB for efficient range queries. Probabilistic balancing avoids complex rotations.

## [API Gateway Pattern - Request Routing]
```python
# PROBLEM: Implement API gateway for routing and request aggregation
# TIME: O(1) for routing O(n) for aggregation SPACE: O(m) where m = services
from dataclasses import dataclass
from typing import Dict, Callable, Any, List
import time
from collections import defaultdict

@dataclass
class Request:
    path: str
    method: str
    headers: dict
    body: dict
    user_id: str = None

@dataclass
class Response:
    status: int
    body: dict
    headers: dict = None

class ServiceRegistry:
    def __init__(self):
        self.services: Dict[str, str] = {}  # service_name -> base_url
    
    def register(self, name: str, url: str):
        self.services[name] = url
        print(f"✓ Registered service: {name} at {url}")
    
    def get(self, name: str) -> str:
        return self.services.get(name)

class APIGateway:
    def __init__(self):
        self.routes: Dict[str, Callable] = {}  # path_pattern -> handler
        self.service_registry = ServiceRegistry()
        self.rate_limiter = defaultdict(list)  # user_id -> [timestamps]
        self.middleware: List[Callable] = []
    
    def register_service(self, name: str, url: str):
        # Register backend service
        self.service_registry.register(name, url)
    
    def route(self, path_pattern: str):
        # Decorator for registering routes
        def decorator(handler):
            self.routes[path_pattern] = handler
            return handler
        return decorator
    
    def add_middleware(self, middleware_func):
        # Add middleware to processing chain
        self.middleware.append(middleware_func)
    
    def check_rate_limit(self, user_id: str, limit: int = 100, window: int = 60):
        # Sliding window rate limiting
        now = time.time()
        timestamps = self.rate_limiter[user_id]
        
        # Remove old timestamps outside window
        self.rate_limiter[user_id] = [t for t in timestamps if now - t < window]
        
        if len(self.rate_limiter[user_id]) >= limit:
            return False
        
        self.rate_limiter[user_id].append(now)
        return True
    
    def match_route(self, path: str) -> tuple:
        # Simple prefix matching (can be extended with regex)
        for pattern, handler in self.routes.items():
            if path.startswith(pattern):
                return handler, path[len(pattern):]
        return None, None
    
    def aggregate(self, requests: List[Request]) -> Response:
        # Aggregate multiple service calls into single response
        results = []
        for req in requests:
            handler, _ = self.match_route(req.path)
            if handler:
                results.append(handler(req))
        
        return Response(
            status=200,
            body={'aggregated': [r.body for r in results]}
        )
    
    def handle(self, request: Request) -> Response:
        # Main request handling with middleware chain
        try:
            # Apply middleware
            for mw in self.middleware:
                request = mw(request)
                if request is None:
                    return Response(status=400, body={'error': 'Middleware rejected'})
            
            # Rate limiting
            if request.user_id and not self.check_rate_limit(request.user_id):
                return Response(status=429, body={'error': 'Rate limit exceeded'})
            
            # Route matching
            handler, params = self.match_route(request.path)
            if not handler:
                return Response(status=404, body={'error': 'Not found'})
            
            # Execute handler
            return handler(request)
        
        except Exception as e:
            return Response(status=500, body={'error': str(e)})

# Example usage
gateway = APIGateway()

# Register services
gateway.register_service('user-service', 'http://users:8001')
gateway.register_service('order-service', 'http://orders:8002')

# Define routes
@gateway.route('/api/users')
def handle_users(request: Request) -> Response:
    return Response(status=200, body={'users': ['alice', 'bob']})

@gateway.route('/api/orders')
def handle_orders(request: Request) -> Response:
    return Response(status=200, body={'orders': [{'id': 1, 'item': 'widget'}]})

# Add authentication middleware
def auth_middleware(request: Request):
    token = request.headers.get('Authorization')
    if token == 'valid-token':
        request.user_id = 'user123'
    return request

gateway.add_middleware(auth_middleware)

# Test requests
requests = [
    Request(path='/api/users', method='GET', 
            headers={'Authorization': 'valid-token'}, body={}),
    Request(path='/api/orders', method='GET',
            headers={'Authorization': 'valid-token'}, body={}),
    Request(path='/api/unknown', method='GET',
            headers={'Authorization': 'valid-token'}, body={}),
]

for req in requests:
    response = gateway.handle(req)
    print(f"{req.path} -> {response.status}: {response.body}")
```
**Why**: Single entry point for microservices. Handles cross-cutting concerns (auth, rate limiting, logging). Enables request aggregation and protocol translation.

## [Database Sharding - Horizontal Partitioning]
```python
# PROBLEM: Implement sharding strategy for horizontal scaling
# TIME: O(1) for shard lookup SPACE: O(k) where k = number of shards
import hashlib
from dataclasses import dataclass
from typing import Dict, List, Any, Optional

@dataclass
class ShardConfig:
    shard_id: int
    connection_string: str
    range_start: int
    range_end: int

class HashSharding:
    """Hash-based sharding - distributes data using hash function"""
    
    def __init__(self, num_shards: int):
        self.num_shards = num_shards
        self.shards: Dict[int, List] = {i: [] for i in range(num_shards)}
    
    def get_shard(self, key: str) -> int:
        # Hash key and map to shard
        hash_value = int(hashlib.md5(key.encode()).hexdigest(), 16)
        return hash_value % self.num_shards
    
    def insert(self, key: str, value: Any):
        shard_id = self.get_shard(key)
        self.shards[shard_id].append({'key': key, 'value': value})
        print(f"Inserted '{key}' into shard {shard_id}")
    
    def get(self, key: str) -> Optional[Any]:
        shard_id = self.get_shard(key)
        for item in self.shards[shard_id]:
            if item['key'] == key:
                return item['value']
        return None
    
    def get_distribution(self) -> Dict[int, int]:
        return {sid: len(items) for sid, items in self.shards.items()}

class RangeSharding:
    """Range-based sharding - distributes data by key ranges"""
    
    def __init__(self, shard_ranges: List[tuple]):
        # shard_ranges: [(shard_id, start, end), ...]
        self.shards: Dict[int, ShardConfig] = {}
        self.shard_data: Dict[int, List] = {}
        
        for shard_id, start, end, conn_str in shard_ranges:
            self.shards[shard_id] = ShardConfig(
                shard_id=shard_id,
                connection_string=conn_str,
                range_start=start,
                range_end=end
            )
            self.shard_data[shard_id] = []
    
    def get_shard(self, key: int) -> Optional[int]:
        # Find shard by checking ranges
        for shard_id, config in self.shards.items():
            if config.range_start <= key <= config.range_end:
                return shard_id
        return None
    
    def insert(self, key: int, value: Any):
        shard_id = self.get_shard(key)
        if shard_id is None:
            print(f"❌ No shard found for key {key}")
            return
        
        self.shard_data[shard_id].append({'key': key, 'value': value})
        config = self.shards[shard_id]
        print(f"Inserted {key} into shard {shard_id} (range: {config.range_start}-{config.range_end})")
    
    def get_distribution(self) -> Dict[int, int]:
        return {sid: len(items) for sid, items in self.shard_data.items()}

class ConsistentHashing:
    """Consistent hashing - minimizes redistribution on shard changes"""
    
    def __init__(self, num_replicas: int = 150):
        self.num_replicas = num_replicas
        self.ring: Dict[int, str] = {}  # hash position -> node
        self.sorted_keys: List[int] = []
        self.nodes: set = set()
    
    def _hash(self, key: str) -> int:
        return int(hashlib.md5(key.encode()).hexdigest(), 16)
    
    def add_node(self, node: str):
        # Add multiple replicas to ring for even distribution
        self.nodes.add(node)
        for i in range(self.num_replicas):
            replica_key = f"{node}:replica:{i}"
            position = self._hash(replica_key)
            self.ring[position] = node
            self.sorted_keys.append(position)
        
        self.sorted_keys.sort()
        print(f"Added node '{node}' with {self.num_replicas} replicas")
    
    def remove_node(self, node: str):
        # Remove all replicas of node
        self.nodes.discard(node)
        for i in range(self.num_replicas):
            replica_key = f"{node}:replica:{i}"
            position = self._hash(replica_key)
            self.ring.pop(position, None)
            self.sorted_keys.remove(position)
        
        print(f"Removed node '{node}'")
    
    def get_node(self, key: str) -> str:
        # Find the first node clockwise from key's hash
        if not self.ring:
            return None
        
        position = self._hash(key)
        
        # Binary search for first node >= position
        for hash_val in self.sorted_keys:
            if hash_val >= position:
                return self.ring[hash_val]
        
        # Wrap around to first node
        return self.ring[self.sorted_keys[0]]

# Example usage
print("=== Hash-based Sharding ===")
hash_shard = HashSharding(num_shards=4)
for user in ['alice', 'bob', 'charlie', 'david', 'eve']:
    hash_shard.insert(user, f"data_{user}")

print(f"Distribution: {hash_shard.get_distribution()}\n")

print("=== Range-based Sharding ===")
range_shard = RangeSharding([
    (0, 0, 999, 'shard0.db'),
    (1, 1000, 1999, 'shard1.db'),
    (2, 2000, 2999, 'shard2.db'),
])
for order_id in [150, 1500, 2500, 3500]:
    range_shard.insert(order_id, f"order_{order_id}")

print(f"Distribution: {range_shard.get_distribution()}\n")

print("=== Consistent Hashing ===")
ch = ConsistentHashing(num_replicas=100)
ch.add_node('node1')
ch.add_node('node2')
ch.add_node('node3')

keys = ['user:100', 'user:200', 'user:300', 'user:400']
for key in keys:
    node = ch.get_node(key)
    print(f"Key '{key}' -> {node}")

print("\nAfter adding node4:")
ch.add_node('node4')
for key in keys:
    node = ch.get_node(key)
    print(f"Key '{key}' -> {node}")
```
**Why**: Enables horizontal scaling of databases. Distributes load across multiple servers. Consistent hashing minimizes data movement when adding/removing shards. Critical for high-traffic applications.
