## 6. Link Preload Resource
```html
<!-- PROBLEM: Prioritize loading of critical resources -->
<!-- TIME: O(1) SPACE: O(1) -->

Why: Preloading critical resources (fonts, CSS, JS) improves page load
performance by starting downloads early in the critical rendering path.
-->

<!DOCTYPE html>
<html lang="en">

  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">

    rel="preload" 
    href="styles.css" 
    as="style"
    onload="this.onload=null;this.rel='stylesheet'">
  
  <!-- Preload critical JavaScript -->
  <link 
    rel="preload" 
    href="app.js" 
    as="script">
  
  <!-- Preload web fonts -->

    as="font"
```

## 8. JavaScript Array CopyWithin
```javascript
// PROBLEM: Copy elements within an array to other positions
// TIME: O(n) SPACE: O(1)

/**
 * Why: copyWithin() is a memory-efficient way to shift array elements
 * without creating a new array. Useful for buffer manipulation and
 * implementing circular buffers.
 */

// Basic usage: Copy from position 0 to position 3
const arr1 = [1, 2, 3, 4, 5];
console.log(arr1.copyWithin(3, 0)); // [1, 2, 3, 1, 2]

// Copy with end boundary (exclusive)
const arr2 = [1, 2, 3, 4, 5];
console.log(arr2.copyWithin(0, 3, 4)); // [4, 2, 3, 4, 5]
// Copies from index 3 to 4 (just the 4) to position 0

// Negative indices work from the end
const arr3 = [1, 2, 3, 4, 5];
console.log(arr3.copyWithin(0, -2)); // [4, 5, 3, 4, 5]
// Copies last 2 elements to the beginning

// Practical example: Circular buffer implementation
class CircularBuffer {
  constructor(size) {
    this.buffer = new Array(size);
    this.size = size;
    this.count = 0;
  }
  
  add(item) {
    if (this.count < this.size) {
      this.buffer[this.count] = item;
      this.count++;
    } else {
      // Shift elements left using copyWithin
      this.buffer.copyWithin(0, 1);
      this.buffer[this.size - 1] = item;
    }
  }
  
  getItems() {
    return this.buffer.slice(0, this.count);
  }
}

const buffer = new CircularBuffer(3);
buffer.add(1); // [1, empty, empty]
buffer.add(2); // [1, 2, empty]
buffer.add(3); // [1, 2, 3]
buffer.add(4); // [2, 3, 4] - copyWithin shifts left
console.log(buffer.getItems()); // [2, 3, 4]
```

## 9. JavaScript Try-Catch-Finally
```javascript
// PROBLEM: Handle errors gracefully and ensure cleanup code runs
// TIME: O(1) SPACE: O(1)

/**
 * Why: try-catch-finally ensures code can recover from errors while
 * guaranteeing cleanup operations (closing connections, releasing resources)
 * always execute regardless of success or failure.
 */

function processFile(filename) {
  let fileHandle = null;
  
  try {
    // Attempt to open and process file
    console.log(`Opening ${filename}...`);
    fileHandle = openFileSync(filename); // Simulated sync operation
    
    const data = fileHandle.read();
    const processed = processData(data);
    
    return { success: true, data: processed };
    
  } catch (error) {
    // Handle specific error types
    if (error.code === 'ENOENT') {
      console.error('File not found:', filename);
      return { success: false, error: 'File not found' };
    } else if (error.code === 'EACCES') {
      console.error('Permission denied:', filename);
      return { success: false, error: 'Permission denied' };
    } else {
      console.error('Unexpected error:', error.message);
      return { success: false, error: 'Processing failed' };
    }
    
  } finally {
    // Always runs - cleanup resources
    if (fileHandle) {
      console.log('Closing file handle...');
      fileHandle.close();
    }
    console.log('Cleanup complete');
  }
}

// Practical example: Database transaction
async function transferFunds(fromId, toId, amount) {
  const transaction = await db.beginTransaction();
  
  try {
    // Deduct from sender
    await transaction.query(
      'UPDATE accounts SET balance = balance - ? WHERE id = ?',
      [amount, fromId]
    );
    
    // Add to receiver
    await transaction.query(
      'UPDATE accounts SET balance = balance + ? WHERE id = ?',
      [amount, toId]
    );
    
    // Commit transaction
    await transaction.commit();
    return { success: true };
    
  } catch (error) {
    // Rollback on any error
    await transaction.rollback();
    console.error('Transfer failed:', error);
    return { success: false, error: error.message };
    
  } finally {
    // Always release connection back to pool
    transaction.release();
    console.log('Database connection released');
  }
}
```

## 10. JavaScript Map Constructor
```javascript
// PROBLEM: Create key-value collections with any type of key
// TIME: O(1) for creation, O(1) for get/set SPACE: O(n)

/**
 * Why: Map provides better key-value storage than Objects:
 * - Keys can be any type (objects, functions, primitives)
 * - Preserves insertion order
 * - Has .size property
 * - Better performance for frequent additions/removals
 */

// Create Map from constructor
const map1 = new Map();
map1.set('name', 'Alice');
map1.set('age', 30);
console.log(map1.get('name')); // 'Alice'
console.log(map1.size); // 2

// Create Map from iterable (array of [key, value] pairs)
const map2 = new Map([
  ['a', 1],
  ['b', 2],
  ['c', 3]
]);
console.log(map2.get('b')); // 2

// Map with object keys (not possible with plain objects)
const userMap = new Map();
const user1 = { id: 1, name: 'Alice' };
const user2 = { id: 2, name: 'Bob' };

userMap.set(user1, { permissions: ['read', 'write'] });
userMap.set(user2, { permissions: ['read'] });

console.log(userMap.get(user1)); // { permissions: ['read', 'write'] }

// Practical example: Memoization cache using Map
function memoize(fn) {
  const cache = new Map();
  
  return function(...args) {
    // Create cache key from arguments
    const key = JSON.stringify(args);
    
    if (cache.has(key)) {
      console.log('Cache hit for', args);
      return cache.get(key);
    }
    
    console.log('Cache miss for', args);
    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };
}

// Expensive calculation
function fibonacci(n) {
  if (n <= 1) return n;
  return fibonacci(n - 1) + fibonacci(n - 2);
}

const fastFib = memoize(fibonacci);
console.log(fastFib(40)); // Calculates once
console.log(fastFib(40)); // Returns from cache

// Practical example: Event emitter using Map
class EventEmitter {
  constructor() {
    this.events = new Map();
  }
  
  on(event, callback) {
    if (!this.events.has(event)) {
      this.events.set(event, new Set());
    }
    this.events.get(event).add(callback);
  }
  
  emit(event, ...args) {
    if (this.events.has(event)) {
      this.events.get(event).forEach(callback => {
        callback.apply(this, args);
      });
    }
  }
  
  off(event, callback) {
    if (this.events.has(event)) {
      this.events.get(event).delete(callback);
    }
  }
}

const emitter = new EventEmitter();
emitter.on('userCreated', (user) => {
  console.log('New user:', user.name);
});
emitter.emit('userCreated', { name: 'Alice', email: 'alice@example.com' });
```
