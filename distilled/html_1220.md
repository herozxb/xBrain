         * @param {string} name - Cookie name
         * @param {string} value - Cookie value
         * @param {number} days - Days until expiration
         * @param {string} path - Cookie path
         */
        function setCookie(name, value, days = 7, path = '/') {

            const expires = new Date();

<!-- PROBLEM: Control redirect behavior in fetch requests -->
<!-- TIME: O(1) SPACE: O(1) -->

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Fetch Redirect Control</title>
</head>
<body>
    <button id="fetchBtn">Fetch with Redirect</button>
    <pre id="result"></pre>

    <script>
        /**
         * Fetch with redirect control
         * @param {string} url - URL to fetch
         * @param {string} redirectMode - 'follow', 'error', or 'manual'
         */
        async function fetchWithRedirect(url, redirectMode = 'follow') {
            try {
                const response = await fetch(url, {
                    method: 'GET',
                    redirect: redirectMode // 'follow' | 'error' | 'manual'
                });
                
                // Check if response was redirected
                if (response.redirected) {
                    console.log('Request was redirected');
                    console.log('Final URL:', response.url);
                }
                
                // For manual redirect, inspect the redirect response
                if (redirectMode === 'manual' && response.type === 'opaqueredirect') {
                    console.log('Manual redirect - response is opaque');
                    return { status: 'redirect', opaque: true };
                }
                
                const data = await response.text();
                console.log('Response status:', response.status);
                return data;
                
            } catch (error) {
                console.error('Fetch error:', error);
                throw error;
            }
        }
        
        // Usage examples
        document.getElementById('fetchBtn').addEventListener('click', async () => {
            const result = document.getElementById('result');
            
            // Follow redirects (default)
            result.textContent = 'Fetching with redirect: follow...\n';
            const data1 = await fetchWithRedirect('https://httpstat.us/302', 'follow');
            result.textContent += `Result: ${data1}\n\n`;
            
            // Manual redirect handling
            result.textContent += 'Fetching with redirect: manual...\n';
            const data2 = await fetchWithRedirect('https://httpstat.us/302', 'manual');
            result.textContent += `Result: ${JSON.stringify(data2)}`;
        });
    </script>
</body>
</html>
```

**Why:** Controlling redirect behavior gives you fine-grained control over HTTP redirects, useful for API clients, authentication flows, and tracking redirect chains.

        /**
         * Build and send FormData with files
         * @param {FileList} files - Files to upload
         */
        async function uploadFiles(files) {

            const formData = new FormData();
            
            // Append text fields
            formData.append('username', 'john_doe');
            formData.append('email', 'john@example.com');
            
            // Append multiple files
            for (let i = 0; i < files.length; i++) {
                // Can use same key for multiple files
                formData.append('files', files[i]);
                console.log(`Appended file: ${files[i].name}`);
            }
            
            // Append additional metadata
            formData.append('uploadDate', new Date().toISOString());
            
            // Log FormData contents (for debugging)
            for (let [key, value] of formData.entries()) {
                console.log(key, value);
            }
            
            // Send to server
            try {
                const response = await fetch('/api/upload', {
                    method: 'POST',
                    body: formData
                    // Don't set Content-Type header - browser sets it with boundary
                });
                
                return await response.json();
            } catch (error) {
                console.error('Upload error:', error);
                throw error;
            }
        }
        
        // Usage
        document.getElementById('uploadBtn').addEventListener('click', async () => {
            const files = document.getElementById('fileInput').files;
            const result = document.getElementById('result');
            
            if (files.length === 0) {
                result.textContent = 'Please select files first';
                return;
            }
            
            result.textContent = 'Uploading files...\n';
            
            // For demo, just show what would be sent
            const formData = new FormData();
            for (let file of files) {
                formData.append('files', file);
            }
            
            for (let [key, value] of formData.entries()) {
                result.textContent += `${key}: ${value.name || value}\n`;
            }
        });
    </script>
</body>

        const dropzone = document.getElementById('dropzone');
        const result = document.getElementById('result');
        
        // Add dragstart event to draggable elements
        draggables.forEach(draggable => {
            draggable.addEventListener('dragstart', (event) => {
                // Set data in the dataTransfer object
                // Can set multiple data types
                event.dataTransfer.setData('text/plain', draggable.id);
                event.dataTransfer.setData('text/uri-list', `#${draggable.id}`);
                event.dataTransfer.setData('application/json', JSON.stringify({
                    id: draggable.id,
                    content: draggable.textContent,
                    timestamp: Date.now()
                }));
                
                // Set drag image (optional)
                event.dataTransfer.effectAllowed = 'move';
                
                console.log('Drag started:', draggable.id);
            });
        });
        
        // Dropzone events
        dropzone.addEventListener('dragover', (event) => {
            event.preventDefault(); // Allow dropping
            event.dataTransfer.dropEffect = 'move';
            dropzone.classList.add('drag-over');
        });
        
        dropzone.addEventListener('dragleave', () => {
            dropzone.classList.remove('drag-over');
        });
        
        dropzone.addEventListener('drop', (event) => {
            event.preventDefault();
            dropzone.classList.remove('drag-over');
            
            // Get the data that was set during dragstart
            const id = event.dataTransfer.getData('text/plain');
            const jsonData = event.dataTransfer.getData('application/json');
            const parsedData = JSON.parse(jsonData);
            
            // Display the dropped data
            result.textContent = `Dropped: ${id}\nData: ${jsonData}`;
            
            // Move the element
            const draggedElement = document.getElementById(id);
            dropzone.appendChild(draggedElement);
            
            console.log('Dropped data:', parsedData);
        });
    </script>
</body>

            // Get computed styles

--secondary-color: ${getCSSProperty('--secondary-color')}
--spacing: ${spacing}
--border-radius: ${borderRadius}

Computed box color: ${getCSSProperty('background-color', box)}`;

        /**
         * Extract patterns from text using regex match

            // String.match returns array of matches or null
            const matches = text.match(pattern);

                return matches;
            }
            
            console.log('No matches found');
            return [];
        }
        
        // Common regex patterns
        const patterns = {

            words: /\b\w{4,}\b/g,  // Words with 4+ letters
            numbers: /\d+/g,
            urls: /https?:\/\/[^\s]+/g
        };
        
        // Usage
        document.getElementById('extractBtn').addEventListener('click', () => {
            const text = document.getElementById('textInput').value;
            const result = document.getElementById('result');
            
            // Extract emails
            const emails = extractMatches(text, patterns.email);
            
            // Extract phone numbers
            const phones = extractMatches(text, patterns.phone);
            
            // Display results
            result.textContent = `Text: "${text}"

Extracted Data:
--------------
Emails: ${emails.length ? emails.join(', ') : 'None'}
Phones: ${phones.length ? phones.join(', ') : 'None'}

All Words (4+ chars): ${extractMatches(text, patterns.words).join(', ')}

Regex Pattern Details:
- Email: ${patterns.email.source}
- Phone: ${patterns.phone.source}`;
        });
        
        // Advanced: Match with groups
        function matchWithGroups(text, pattern) {
            // Without 'g' flag, returns detailed match info with groups
            const match = text.match(pattern);
            
            if (match) {
                console.log('Full match:', match[0]);
                console.log('Groups:', match.slice(1));
                console.log('Named groups:', match.groups);
                return match;
            }
            return null;
        }
        
        // Example with named groups
        const datePattern = /(?<year>\d{4})-(?<month>\d{2})-(?<day>\d{2})/;
        const dateText = 'Today is 2026-02-27';
        const dateMatch = matchWithGroups(dateText, datePattern);
        
        console.log('Date match with groups:', dateMatch?.groups);
    </script>

</html>
```

**Why:** String.match with regex is powerful for data extraction, validation, and text processing. It's essential for parsing emails, URLs, phone numbers, and custom patterns.

## js_date_parse

```html
<!-- PROBLEM: Parse a date string into milliseconds since epoch -->
<!-- TIME: O(1) SPACE: O(1) -->

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Date.parse</title>
</head>
<body>
    <input type="text" id="dateInput" placeholder="Enter date string" value="2026-02-27">
    <button id="parseBtn">Parse Date</button>
    <pre id="result"></pre>

    <script>
        /**
         * Parse date string to timestamp
         * @param {string} dateString - Date string to parse
         */
        function parseDate(dateString) {
            // Date.parse returns milliseconds since Jan 1, 1970
            const timestamp = Date.parse(dateString);
            
            if (isNaN(timestamp)) {
                return { error: 'Invalid date format' };
            }
            
            // Create Date object from timestamp
            const date = new Date(timestamp);
            
            return {
                input: dateString,
                timestamp: timestamp,
                isoString: date.toISOString(),
                localString: date.toLocaleString(),
                utcString: date.toUTCString(),
                dateObject: date
            };
        }
        
        // Supported date formats
        const formats = [
            '2026-02-27',                      // ISO date
            '2026-02-27T14:30:00',             // ISO datetime
            '2026-02-27T14:30:00Z',            // ISO UTC
            '2026-02-27T14:30:00+08:00',       // ISO with timezone
            'Feb 27 2026',                     // Short month
            'February 27, 2026',               // Long month
            '02/27/2026',                      // US format
            '27-02-2026',                      // Day first
            '2026/02/27 14:30:00'              // Slash format
        ];
        
        console.log('Date.parse format examples:');
        formats.forEach(format => {
            const ts = Date.parse(format);
            console.log(`  "${format}" → ${ts} (${ts ? 'valid' : 'INVALID'})`);
        });
        
        // Usage
        document.getElementById('parseBtn').addEventListener('click', () => {
            const input = document.getElementById('dateInput').value;
            const result = parseDate(input);
            const output = document.getElementById('result');
            
            if (result.error) {
                output.textContent = result.error;
                return;
            }
            
            output.textContent = `Date Parsing Results:
====================
Input: "${result.input}"

Timestamp (ms): ${result.timestamp}
ISO String: ${result.isoString}
Local String: ${result.localString}
UTC String: ${result.utcString}

Time Calculations:
-----------------
Seconds since epoch: ${Math.floor(result.timestamp / 1000)}
Minutes since epoch: ${Math.floor(result.timestamp / 60000)}
Hours since epoch: ${Math.floor(result.timestamp / 3600000)}
Days since epoch: ${Math.floor(result.timestamp / 86400000)}

Supported Formats:
${formats.map(f => `- ${f}`).join('\n')}`;
        });
        
        // Auto-parse on load
        document.getElementById('parseBtn').click();
    </script>
</body>
</html>
```

**Why:** Date.parse converts date strings to timestamps, enabling date calculations, comparisons, and storage. Understanding supported formats prevents parsing errors in applications.
