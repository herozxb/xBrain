```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>removeEventListener Demo</title>
    <style>
        .box { width: 100px; height: 100px; background: #3498db; margin: 20px; }
        .box.active { background: #e74c3c; }
    </style>
</head>
<body>
    <!-- PROBLEM: Remove attached event listener from element
         TIME: O(1) SPACE: O(1) -->
    
    <div id="box" class="box">Click Me</div>
    <button id="toggle">Disable Clicks</button>
    <p id="count">Clicks: 0</p>

    <script>
        // Why: removeEventListener cleans up memory and prevents unwanted triggers
        // Must pass the SAME function reference used in addEventListener
        
        let clickCount = 0;
        const box = document.getElementById('box');
        const toggleBtn = document.getElementById('toggle');
        const countDisplay = document.getElementById('count');
        
        // Named function required for removal (anonymous functions can't be removed)
        function handleClick() {
            clickCount++;
            countDisplay.textContent = `Clicks: ${clickCount}`;
            box.classList.toggle('active');
        }
        
        // Add listener
        box.addEventListener('click', handleClick);
        
        let isEnabled = true;
        
        toggleBtn.addEventListener('click', () => {
            if (isEnabled) {
                // Remove the listener - stops click handling
                box.removeEventListener('click', handleClick);
                toggleBtn.textContent = 'Enable Clicks';
                box.style.opacity = '0.5';
            } else {
                // Re-add the listener
                box.addEventListener('click', handleClick);
                toggleBtn.textContent = 'Disable Clicks';
                box.style.opacity = '1';
            }
            isEnabled = !isEnabled;
        });
    </script>
</body>
</html>
```

## js_sessionStorage_getItem

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>sessionStorage.getItem Demo</title>
</head>
<body>
    <!-- PROBLEM: Retrieve data from session storage
         TIME: O(1) SPACE: O(1) -->
    
    <h2>Session Storage Demo</h2>
    <input type="text" id="username" placeholder="Enter username">
    <button onclick="saveSession()">Save to Session</button>
    <button onclick="loadSession()">Load from Session</button>
    <button onclick="clearSession()">Clear Session</button>
    <p id="display"></p>

    <script>
        // Why: sessionStorage persists data only for the current browser session
        // Data is cleared when tab/window closes - perfect for temporary state
        
        const usernameKey = 'temp_username';
        
        function saveSession() {
            const username = document.getElementById('username').value;
            
            if (username) {
                // Store value (automatically converts to string)
                sessionStorage.setItem(usernameKey, username);
                document.getElementById('display').textContent = 
                    `Saved: "${username}" (tab-specific)`;
            }
        }
        
        function loadSession() {
            // Retrieve value (returns null if key doesn't exist)
            const savedUsername = sessionStorage.getItem(usernameKey);
            
            if (savedUsername !== null) {
                document.getElementById('username').value = savedUsername;
                document.getElementById('display').textContent = 
                    `Loaded: "${savedUsername}"`;
            } else {
                document.getElementById('display').textContent = 
                    'No saved session data found';
            }
        }
        
        function clearSession() {
            sessionStorage.removeItem(usernameKey);
            document.getElementById('username').value = '';
            document.getElementById('display').textContent = 'Session cleared';
        }
        
        // Auto-load on page refresh (session survives refresh)
        window.addEventListener('DOMContentLoaded', loadSession);
    </script>
</body>
</html>
```

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Fetch with Body Demo</title>
</head>
<body>
    <!-- PROBLEM: Send data to server using fetch with request body
         TIME: O(1) for fetch setup SPACE: O(n) for data -->
    
    <h2>POST Data with Fetch</h2>
    <input type="text" id="title" placeholder="Title">
    <input type="text" id="body" placeholder="Body content">
    <button onclick="postData()">Send POST Request</button>
    <pre id="result"></pre>

    <script>
        // Why: The body option sends data to server in POST/PUT/PATCH requests
        // Must set Content-Type header and stringify JSON data
        
        async function postData() {
            const title = document.getElementById('title').value;
            const bodyContent = document.getElementById('body').value;
            
            // Prepare data object
            const data = {
                title: title,
                body: bodyContent,
                userId: 1
            };
            
            try {
                // Fetch with body - sending data to server
                const response = await fetch('https://jsonplaceholder.typicode.com/posts', {
                    method: 'POST',
                    headers: {
                        // Tell server we're sending JSON
                        'Content-Type': 'application/json'
                    },
                    // Body must be a string (JSON.stringify for objects)
                    body: JSON.stringify(data)
                });
                
                if (!response.ok) {
                    throw new Error(`HTTP error! status: ${response.status}`);
                }
                
                const result = await response.json();
                document.getElementById('result').textContent = 
                    JSON.stringify(result, null, 2);
                
            } catch (error) {
                document.getElementById('result').textContent = 
                    `Error: ${error.message}`;
            }
        }
    </script>
</body>
</html>
```

```html
        form.addEventListener('submit', (e) => {
```

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>watchPosition Demo</title>
    <style>
        #map { width: 100%; height: 300px; background: #ecf0f1; display: flex; align-items: center; justify-content: center; }
        .coords { font-family: monospace; }
    </style>
</head>
<body>
    <!-- PROBLEM: Continuously track user's location
         TIME: O(1) per update SPACE: O(1) -->
    
    <h2>Live Location Tracking</h2>
    <div id="map">
        <div class="coords" id="location">Click Start to track location</div>
    </div>
    <p>Accuracy: <span id="accuracy">-</span> meters</p>
    <p>Updates: <span id="updates">0</span></p>
    <button id="startBtn" onclick="startTracking()">Start Tracking</button>
    <button id="stopBtn" onclick="stopTracking()" disabled>Stop Tracking</button>

    <script>
        // Why: watchPosition provides continuous location updates
        // Returns a watchId for stopping the tracker
        
        let watchId = null;
        let updateCount = 0;
        
        const locationDisplay = document.getElementById('location');
        const accuracyDisplay = document.getElementById('accuracy');
        const updatesDisplay = document.getElementById('updates');
        const startBtn = document.getElementById('startBtn');
        const stopBtn = document.getElementById('stopBtn');
        
        function startTracking() {
            if (!navigator.geolocation) {
                locationDisplay.textContent = 'Geolocation not supported';
                return;
            }
            
            locationDisplay.textContent = 'Getting location...';
            
            // Start watching position - updates automatically
            watchId = navigator.geolocation.watchPosition(
                // Success callback
                (position) => {
                    const { latitude, longitude, accuracy } = position.coords;
                    const timestamp = new Date(position.timestamp).toLocaleTimeString();
                    
                    updateCount++;
                    locationDisplay.innerHTML = `
                        Lat: ${latitude.toFixed(6)}<br>
                        Lon: ${longitude.toFixed(6)}<br>
                        Updated: ${timestamp}
                    `;
                    accuracyDisplay.textContent = accuracy.toFixed(1);
                    updatesDisplay.textContent = updateCount;
                },
                // Error callback
                (error) => {
                    locationDisplay.textContent = `Error: ${error.message}`;
                },
                // Options
                {
                    enableHighAccuracy: true,
                    timeout: 10000,
                    maximumAge: 0
                }
            );
            
            startBtn.disabled = true;
            stopBtn.disabled = false;
        }
        
        function stopTracking() {
            if (watchId !== null) {
                // Stop watching using the watchId
                navigator.geolocation.clearWatch(watchId);
                watchId = null;
                locationDisplay.textContent += '<br>(Tracking stopped)';
            }
            
            startBtn.disabled = false;
            stopBtn.disabled = true;
        }
    </script>
</body>
</html>
```

```html
        .sibling {
            /* Inherits from :root directly (not affected by .parent) */
            font-size: var(--text-size); /* Gets 16px from :root */
```

```html
        // Why: Different error types help identify specific issues

            const output = document.getElementById('output');
            output.innerHTML = '';
            
            // 1. ReferenceError - accessing undefined variable
            try {
                console.log(undefinedVariable);
            } catch (e) {
                output.innerHTML += createErrorBox('ReferenceError', e);
            }
            
            // 2. TypeError - wrong type operation
            try {
                null.toUpperCase();
            } catch (e) {
                output.innerHTML += createErrorBox('TypeError', e);
            }
            
            // 3. SyntaxError - invalid syntax (using eval for demo)
            try {
                eval('function(');
            } catch (e) {
                output.innerHTML += createErrorBox('SyntaxError', e);
            }
            
            // 4. RangeError - value out of range
            try {
                const arr = new Array(-1);
            } catch (e) {
                output.innerHTML += createErrorBox('RangeError', e);
            }
            
            // 5. URIError - malformed URI
            try {
                decodeURIComponent('%');
            } catch (e) {
                output.innerHTML += createErrorBox('URIError', e);
            }
        }
        
        function createErrorBox(type, error) {
            return `
                <div class="error-box" style="background: #fee; border-left: 4px solid #e74c3c;">
                    <strong>${type}</strong>
                    <pre>Name: ${error.name}</pre>
                    <pre>Message: ${error.message}</pre>
                </div>
            `;
        }
        
        // Custom error class
        class ValidationError extends Error {
            constructor(message, field) {
                super(message);
                this.name = 'ValidationError';
                this.field = field;
            }
        }
        
        function customError() {
            const output = document.getElementById('output');
            
            try {
                // Throw custom error
                throw new ValidationError('Email is required', 'email');

                    <div class="error-box" style="background: #fef3cd; border-left: 4px solid #ffc107;">
                        <strong>Custom Error</strong>
                        <pre>Name: ${e.name}</pre>
                        <pre>Message: ${e.message}</pre>
                        <pre>Field: ${e.field}</pre>
                        <pre>Stack: ${e.stack?.split('\n')[0]}</pre>
                    </div>
                `;
            }
        }
    </script>
```

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>RegExp Constructor Demo</title>
    <style>
        .match { background: #d4edda; padding: 2px 6px; border-radius: 3px; }
        input, button { margin: 5px; }
    </style>
</head>
<body>
    <!-- PROBLEM: Create regex patterns dynamically with RegExp constructor
         TIME: O(n) for compilation SPACE: O(m) for pattern -->
    
    <h2>RegExp Constructor</h2>
    
    <div>
        <input type="text" id="pattern" placeholder="Pattern (e.g., \\d+)">
        <input type="text" id="flags" placeholder="Flags (e.g., gi)" value="gi">
        <button onclick="buildRegex()">Build & Test</button>
    </div>
    
    <div>
        <textarea id="testText" rows="4" cols="50" placeholder="Text to test...">Hello 123, World 456! Email: test@example.com</textarea>
    </div>
    
    <button onclick="testPattern()">Test</button>
    <button onclick="matchPattern()">Match</button>
    <button onclick="replacePattern()">Replace</button>
    
    <div id="output"></div>

    <script>
        // Why: RegExp() allows dynamic pattern creation from strings
        // Useful when pattern comes from user input or variables
        
        let currentRegex = null;
        
        function buildRegex() {
            const patternStr = document.getElementById('pattern').value;
            const flagsStr = document.getElementById('flags').value;
            
            try {
                // Constructor: new RegExp(pattern, flags)
                currentRegex = new RegExp(patternStr, flagsStr);
                
                document.getElementById('output').innerHTML = `
                    <p><strong>Created RegExp:</strong></p>
                    <code>/${currentRegex.source}/${currentRegex.flags}</code>
                    <p>Global: ${currentRegex.global}</p>
                    <p>Case insensitive: ${currentRegex.ignoreCase}</p>
                    <p>Multiline: ${currentRegex.multiline}</p>
                `;
            } catch (e) {
                document.getElementById('output').innerHTML = 
                    `<p style="color: red;">Invalid pattern: ${e.message}</p>`;
            }
        }
        
        function testPattern() {
            if (!currentRegex) buildRegex();
            const text = document.getElementById('testText').value;
            
            // test() returns true/false
            const result = currentRegex.test(text);
            
            document.getElementById('output').innerHTML += `
                <p><strong>Test result:</strong> ${result ? '✅ Match found' : '❌ No match'}</p>
            `;
        }
        
        function matchPattern() {
            if (!currentRegex) buildRegex();
            const text = document.getElementById('testText').value;
            
            // match() returns array of matches or null
            const matches = text.match(currentRegex);
            
            if (matches) {
                document.getElementById('output').innerHTML += `
                    <p><strong>Matches (${matches.length}):</strong></p>
                    <ul>${matches.map(m => `<li>"${m}"</li>`).join('')}</ul>
                `;
            } else {
                document.getElementById('output').innerHTML += 
                    `<p>No matches found</p>`;
            }
        }
        
        function replacePattern() {
            if (!currentRegex) buildRegex();
            const text = document.getElementById('testText').value;
            const replacement = prompt('Replacement text:', '[REDACTED]');
            
            // replace() returns new string with replacements
            const result = text.replace(currentRegex, replacement);
            
            document.getElementById('output').innerHTML += `
                <p><strong>Replaced:</strong></p>
                <p>${result}</p>
            `;
        }
        
        // Example: Dynamic pattern building
        function searchWord(word) {
            // Build regex that matches whole word (case insensitive)
            const regex = new RegExp(`\\b${word}\\b`, 'gi');
            return regex;
        }
    </script>
</body>
</html>
```
