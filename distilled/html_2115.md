```html
<!-- PROBLEM: Create responsive images that adapt to screen sizes -->
<!-- TIME: O(1) SPACE: O(1) -->
<style>
  .responsive-img {
    max-width: 100%;
    height: auto;
    display: block;
  }
</style>

<img 
  src="image-800.jpg"
  srcset="image-400.jpg 400w,
          image-800.jpg 800w,
          image-1200.jpg 1200w"
  sizes="(max-width: 600px) 400px,
         (max-width: 1000px) 800px,
         1200px"
  alt="Responsive example"
  class="responsive-img"
  loading="lazy"
>

<script>
  // Check if native lazy loading is supported
  if ('loading' in HTMLImageElement.prototype) {
    console.log('Native lazy loading supported');
  } else {
    // Fallback: Intersection Observer
    const images = document.querySelectorAll('img[loading="lazy"]');
    const imageObserver = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          const img = entry.target;
          img.src = img.dataset.src;
          imageObserver.unobserve(img);
        }
      });
    });
    images.forEach(img => imageObserver.observe(img));
  }
</script>
```

**Why**: Responsive images optimize bandwidth by serving appropriate image sizes for different devices, improving page load performance and user experience.

## js_inheritance

```html
<!-- PROBLEM: Implement class inheritance in JavaScript -->
<!-- TIME: O(1) SPACE: O(1) -->
<script>
  // ES6 Class Inheritance
  class Animal {
    constructor(name) {
      this.name = name;
    }
    
    speak() {
      console.log(`${this.name} makes a sound`);
    }
    
    static create(species) {
      return new Animal(species);
    }
  }
  
  class Dog extends Animal {
    constructor(name, breed) {
      super(name); // Call parent constructor
      this.breed = breed;
    }
    
    speak() {
      console.log(`${this.name} barks`);
    }
    
    fetch() {
      console.log(`${this.name} fetches the ball`);
    }
  }
  
  // Usage
  const dog = new Dog('Max', 'Golden Retriever');
  dog.speak(); // "Max barks"
  dog.fetch(); // "Max fetches the ball"
  
  console.log(dog instanceof Dog); // true
  console.log(dog instanceof Animal); // true
  
  // Prototype-based Inheritance (ES5)
  function Cat(name, color) {
    Animal.call(this, name);
    this.color = color;
  }
  
  Cat.prototype = Object.create(Animal.prototype);
  Cat.prototype.constructor = Cat;
  
  Cat.prototype.speak = function() {
    console.log(`${this.name} meows`);
  };
  
  // Multiple Inheritance using Mixins
  const Swimmer = {
    swim() {
      console.log(`${this.name} swims`);
    }
  };
  
  const Flyer = {
    fly() {
      console.log(`${this.name} flies`);
    }
  };
  
  class Duck extends Animal {
    constructor(name) {
      super(name);
      Object.assign(this, Swimmer, Flyer);
    }
    
    speak() {
      console.log(`${this.name} quacks`);
    }
  }
  
  const duck = new Duck('Donald');
  duck.speak(); // "Donald quacks"
  duck.swim(); // "Donald swims"
  duck.fly(); // "Donald flies"
</script>
```

**Why**: Inheritance enables code reuse, establishes hierarchical relationships between classes, and supports polymorphism for flexible and maintainable code architecture.

```html
<script type="module">
  // Named imports
  import { add, subtract, PI } from './math.js';

  }

  console.log(add(5, 3)); // 8

  const calc = new Calculator();
  calc.compute();
  
  utils.formatDate(new Date());
</script>

<!-- math.js -->
<script>
  // Named exports
  export const PI = 3.14159;
  
  export function add(a, b) {
    return a + b;
  }
  
  export function subtract(a, b) {
    return a - b;
  }
  
  // Default export (one per module)
  export default class Calculator {
    compute() {
      console.log('Computing...');
    }
  }
  
  // Re-export
  export { add as sum } from './math.js';
  export * from './constants.js';
  
  // Module pattern (IIFE)
  const Module = (function() {
    let privateVar = 0;
    
    function privateMethod() {
      return privateVar;
    }
    
    return {
      publicMethod() {
        return privateMethod();
      },
      increment() {
        privateVar++;
      }
    };
  })();
  
  // CommonJS style (Node.js)
  module.exports = {
    add: (a, b) => a + b,
    subtract: (a, b) => a - b
  };
</script>
```

**Why**: Modules provide encapsulation, prevent global namespace pollution, enable code reuse, and support dependency management for scalable applications.

## dom_replace_child

```html
<!-- PROBLEM: Replace an existing DOM element with a new one -->
<!-- TIME: O(1) SPACE: O(1) -->
<div id="container">
  <p id="old-element">Old content</p>
</div>

<button id="replace-btn">Replace Element</button>

<script>
  // Method 1: replaceChild()
  function replaceWithNew() {
    const container = document.getElementById('container');
    const oldElement = document.getElementById('old-element');
    
    // Create new element
    const newElement = document.createElement('div');
    newElement.id = 'new-element';
    newElement.textContent = 'New content';
    newElement.style.color = 'blue';
    
    // Replace old with new
    container.replaceChild(newElement, oldElement);
  }
  
  // Method 2: replaceWith() - Modern approach
  function replaceWithModern() {
    const oldElement = document.getElementById('old-element');
    
    const newElement = document.createElement('div');
    newElement.textContent = 'Modern replacement';
    
    oldElement.replaceWith(newElement);
  }
  
  // Method 3: Replace multiple elements
  function replaceMultiple() {
    const container = document.getElementById('container');
    const oldElement = document.getElementById('old-element');
    
    // Create document fragment with multiple children
    const fragment = document.createDocumentFragment();
    
    ['Item 1', 'Item 2', 'Item 3'].forEach(text => {
      const li = document.createElement('li');
      li.textContent = text;
      fragment.appendChild(li);
    });
  
    const newList = document.createElement('ul');
    newList.appendChild(fragment);
    
    container.replaceChild(newList, oldElement);
  }
  
  // Replace with cloned element
  function replaceWithClone() {
    const oldElement = document.getElementById('old-element');
    const clone = oldElement.cloneNode(true); // Deep clone
    clone.textContent = 'Cloned and modified';
    clone.style.backgroundColor = 'yellow';
    
    oldElement.replaceWith(clone);
  }
  
  document.getElementById('replace-btn').addEventListener('click', replaceWithNew);
</script>
```

**Why**: replaceChild() and replaceWith() enable dynamic content updates while maintaining DOM structure, useful for real-time updates and state changes.

## dom_clone_node

```html
<!-- PROBLEM: Clone DOM elements with or without children -->
<!-- TIME: O(n) SPACE: O(n) where n is number of nodes -->
<div id="original">
  <h3>Original Element</h3>
  <p>This is some content</p>
  <ul>
    <li>Item 1</li>
    <li>Item 2</li>
  </ul>
</div>

<button id="clone-shallow">Clone Shallow</button>
<button id="clone-deep">Clone Deep</button>
<div id="clone-container"></div>

<script>
  // Shallow clone (only the element, no children)
  function shallowClone() {
    const original = document.getElementById('original');
    const cloneContainer = document.getElementById('clone-container');
    
    const shallowClone = original.cloneNode(false); // Shallow
    shallowClone.id = 'clone-shallow';
    shallowClone.style.border = '2px solid red';
    shallowClone.textContent = 'Shallow clone (no children)';
    
    cloneContainer.appendChild(shallowClone);
  }
  
  // Deep clone (element + all children)
  function deepClone() {
    const original = document.getElementById('original');
    const cloneContainer = document.getElementById('clone-container');
    
    const deepClone = original.cloneNode(true); // Deep
    deepClone.id = 'clone-deep';
    deepClone.style.border = '2px solid blue';
    
    // Note: Event listeners are NOT cloned
    // Note: IDs must be unique - change them
    
    // Update IDs in cloned tree
    deepClone.querySelectorAll('[id]').forEach(el => {
      el.id = el.id + '-clone';
    });
    
    cloneContainer.appendChild(deepClone);
  }
  
  // Clone with event listeners
  function cloneWithEvents() {
    const original = document.getElementById('original');
    
    // Add event listener to original
    original.addEventListener('click', function handleClick() {
      console.log('Clicked!');
    });
    
    // Clone doesn't have event listeners
    const clone = original.cloneNode(true);
    
    // Must re-attach events manually
    clone.addEventListener('click', function handleClick() {
      console.log('Clone clicked!');
    });
    
    document.body.appendChild(clone);
  }
  
  // Clone template content
  function cloneTemplate() {
    const template = document.createElement('template');
    template.innerHTML = `
      <div class="card">
        <h4>Template Card</h4>
        <p>Content here</p>
      </div>
    `;
    
    // Clone template content
    const clone = template.content.cloneNode(true);
    document.body.appendChild(clone);
  }
  
  // Performance: Clone vs Create
  function performanceTest() {
    console.time('Clone');
    for (let i = 0; i < 1000; i++) {
      const original = document.getElementById('original');
      const clone = original.cloneNode(true);
    }
    console.timeEnd('Clone');
    
    console.time('Create');
    for (let i = 0; i < 1000; i++) {
      const div = document.createElement('div');
      div.innerHTML = '<h3>Created</h3><p>Content</p>';
    }
    console.timeEnd('Create');
  }
  
  document.getElementById('clone-shallow').addEventListener('click', shallowClone);
  document.getElementById('clone-deep').addEventListener('click', deepClone);
</script>
```

**Why**: Cloning enables efficient element duplication for lists, templates, and dynamic UI components, avoiding expensive re-rendering of complex structures.
