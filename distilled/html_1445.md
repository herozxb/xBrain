<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">

</head>

    <!-- PROBLEM: Get mouse cursor X position relative to screen -->
    <!-- TIME: O(1) SPACE: O(1) -->
    <div id="display" style="width:100%;height:100vh;background:#f0f0f0;">
        Move your mouse anywhere on this page
    </div>

    <script>
        // screenX returns the horizontal coordinate of the mouse
        // relative to the entire screen (monitor), not the browser window
        const display = document.getElementById('display');

            display.innerHTML = `
                <p>Screen X: ${screenX}px (from left edge of monitor)</p>
                <p>Client X: ${clientX}px (from left edge of viewport)</p>
            `;
        });
    </script>
</body>
</html>
```
**Why:** Use screenX when you need absolute screen coordinates, such as positioning popup windows or multi-monitor detection.

            counter++;
            updateDisplay();
        });
        
        clearBtn.addEventListener('click', () => {
            sessionStorage.clear();
            counter = 0;
            updateDisplay();
        });
        
        updateDisplay();
    </script>

    <!-- PROBLEM: Listen for storage changes from other tabs -->
    <!-- TIME: O(1) SPACE: O(1) -->

    <script>
        // StorageEvent fires in OTHER tabs/windows when
        // localStorage/sessionStorage changes
        // Does NOT fire in the tab that made the change!
        
        const log = document.getElementById('log');
        const keyInput = document.getElementById('key');
        const valueInput = document.getElementById('value');
        const setBtn = document.getElementById('set');
        
        // Listen for storage events from other tabs
        window.addEventListener('storage', (event) => {
            console.log('Storage event received:', event);
            
            // Event properties:
            // - key: the changed key (null if clear())
            // - oldValue: previous value (null if new)
            // - newValue: new value (null if removed)
            // - url: page URL that made the change
            // - storageArea: localStorage or sessionStorage
            
            log.innerHTML += `
                <div style="margin:5px 0;padding:5px;background:#f0f0f0;">
                    <strong>Key:</strong> ${event.key}<br>
                    <strong>Old:</strong> ${event.oldValue}<br>
                    <strong>New:</strong> ${event.newValue}<br>
                    <strong>URL:</strong> ${event.url}
                </div>
            `;
        });
        
        setBtn.addEventListener('click', () => {
            localStorage.setItem(keyInput.value, valueInput.value);
            log.innerHTML += `<p style="color:blue;">Set in this tab: ${keyInput.value}</p>`;
        });
    </script>
</body>
</html>
```
**Why:** Essential for cross-tab communication - sync state across browser tabs, detect logouts, or coordinate user actions.

## fetch_referrer
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Fetch referrer Policy</title>
</head>
<body>
    <!-- PROBLEM: Control referrer header in fetch requests -->
    <!-- TIME: O(1) SPACE: O(1) -->
    <button id="fetch1">Fetch with default referrer</button>
    <button id="fetch2">Fetch with no-referrer</button>
    <button id="fetch3">Fetch with origin only</button>
    <div id="result"></div>

    <script>
        const result = document.getElementById('result');
        
        // referrerPolicy controls what referrer info is sent
        // Available policies:
        // - '' (default): full URL as referrer
        // - 'no-referrer': no referrer header
        // - 'origin': only send origin (domain)
        // - 'same-origin': send only for same-origin requests
        // - 'strict-origin': origin only, no referrer on downgrade
        // - 'no-referrer-when-downgrade': full URL, no referrer on https->http
        
        document.getElementById('fetch1').addEventListener('click', async () => {
            // Default: sends full page URL as referrer
            const response = await fetch('https://httpbin.org/headers', {
                // referrerPolicy defaults to 'no-referrer-when-downgrade'
            });
            const data = await response.json();
            result.textContent = JSON.stringify(data.headers.Referer, null, 2);
        });
        
        document.getElementById('fetch2').addEventListener('click', async () => {
            // No referrer sent at all
            const response = await fetch('https://httpbin.org/headers', {
                referrerPolicy: 'no-referrer'
            });
            const data = await response.json();
            result.textContent = 'No Referer header sent';
        });
        
        document.getElementById('fetch3').addEventListener('click', async () => {
            // Only send origin (domain), not full path
            const response = await fetch('https://httpbin.org/headers', {
                referrerPolicy: 'origin'
            });
            const data = await response.json();
            result.textContent = JSON.stringify(data.headers.Referer, null, 2);
        });
    </script>
</body>
</html>
```
**Why:** Protect user privacy by controlling referrer exposure, or comply with API requirements for specific referrer handling.

        // with the full response body as a string
        // Useful for: plain text, CSV, XML, HTML responses
        
        document.getElementById('fetch').addEventListener('click', async () => {
            try {
                const response = await fetch('https://httpbin.org/html');
                
                // Check if request was successful
                if (!response.ok) {
                    throw new Error(`HTTP error: ${response.status}`);
                }
                
                // Get response body as text
                const text = await response.text();
                
                // Now 'text' is a plain string
                output.textContent = text;
                
                // You can then process it:
                // - Parse as HTML: new DOMParser().parseFromString(text, 'text/html')
                // - Parse as XML: new DOMParser().parseFromString(text, 'text/xml')
                // - Process as CSV lines: text.split('\n')
                
                console.log('Response length:', text.length, 'characters');
                
            } catch (error) {
                output.textContent = `Error: ${error.message}`;
            }
        });
    </script>
</body>
</html>
```
**Why:** Use text() when you need raw text content, want to manually parse formats, or work with non-JSON APIs.

            e.preventDefault();
            
            // Only check validity of inputs that willValidate
            const validInputs = Array.from(inputs).filter(inp => inp.willValidate);

        });
    </script>

            formData.append('tags', 'tag1');
            formData.append('tags', 'tag2');
            formData.delete('tags');
            
            result += '\nAfter adding and deleting tags:\n';
            result += '  (tags completely removed)\n';
            
            output.textContent = result;
        });
    </script>
</body>
</html>
```

## dataTransfer_getData
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>DataTransfer.getData()</title>
    <style>
        .draggable { width: 100px; height: 50px; background: #4CAF50; margin: 10px; cursor: move; }
        .dropzone { width: 200px; height: 100px; border: 2px dashed #333; margin: 10px; }
    </style>
</head>
<body>
    <!-- PROBLEM: Retrieve data set during drag operation -->
    <!-- TIME: O(1) SPACE: O(1) -->
    <div class="draggable" draggable="true" id="item1">Drag Item 1</div>
    <div class="draggable" draggable="true" id="item2">Drag Item 2</div>
    <div class="dropzone" id="dropzone">Drop here</div>
    <p id="message"></p>

    <script>
        const draggables = document.querySelectorAll('.draggable');
        const dropzone = document.getElementById('dropzone');
        const message = document.getElementById('message');
        
        // On dragstart, we set data using setData()
        draggables.forEach(item => {
            item.addEventListener('dragstart', (e) => {
                // Set data with format and value
                e.dataTransfer.setData('text/plain', item.id);
                e.dataTransfer.setData('application/json', JSON.stringify({
                    id: item.id,
                    text: item.textContent
                }));
                
                // Set visual feedback
                e.dataTransfer.effectAllowed = 'move';
            });
        });
        
        dropzone.addEventListener('dragover', (e) => {
            e.preventDefault(); // Allow drop
            e.dataTransfer.dropEffect = 'move';
        });
        
        // On drop, we retrieve data using getData()
        dropzone.addEventListener('drop', (e) => {
            e.preventDefault();
            
            // Get data by format type
            const plainId = e.dataTransfer.getData('text/plain');
            const jsonData = e.dataTransfer.getData('application/json');
            
            // Parse JSON data
            const data = JSON.parse(jsonData);
            
            message.textContent = `Dropped: ${data.text} (ID: ${data.id})`;
            dropzone.style.background = '#a5d6a7';
            
            // Note: getData() only works in drop handler
            // Returns empty string if called outside drop event
            // or if format doesn't exist
        });
    </script>
</body>
</html>
```
**Why:** Use getData() to transfer custom data during drag-and-drop operations, enabling rich interaction between draggable elements and drop targets.

## css_setProperty
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>CSSStyleDeclaration.setProperty()</title>
    <style>
        :root {
            --primary-color: #4CAF50;
            --spacing: 20px;
        }
        .box {
            background: var(--primary-color);
            padding: var(--spacing);
            margin: 10px;
        }
    </style>
</head>
<body>
    <!-- PROBLEM: Set CSS property value via JavaScript -->
    <!-- TIME: O(1) SPACE: O(1) -->
    <div class="box" id="box">Styled Box</div>
    <input id="color" type="color" value="#4CAF50">
    <input id="spacing" type="range" min="0" max="50" value="20">
    <button id="important">Set with !important</button>

    <script>
        const box = document.getElementById('box');
        const colorInput = document.getElementById('color');
        const spacingInput = document.getElementById('spacing');
        
        // setProperty(propertyName, value, priority)
        // - propertyName: CSS property name (with or without --)
        // - value: the value to set
        // - priority: optional 'important' flag
        
        colorInput.addEventListener('input', (e) => {
            // Method 1: Set CSS variable on :root
            document.documentElement.style.setProperty(
                '--primary-color', 
                e.target.value
            );
            
            // This affects all elements using var(--primary-color)
        });
        
        spacingInput.addEventListener('input', (e) => {
            // Method 2: Set variable directly on element
            box.style.setProperty('--spacing', e.target.value + 'px');
        });
        
        document.getElementById('important').addEventListener('click', () => {
            // Set with !important priority
            box.style.setProperty(
                'background', 
                '#ff5722', 
                'important'
            );
            // Equivalent to: background: #ff5722 !important;
        });
        
        // You can also set regular CSS properties
        // box.style.setProperty('font-size', '24px');
        // Same as: box.style.fontSize = '24px';
        
        // But setProperty is required for:
        // 1. CSS variables (--custom-properties)
        // 2. Setting !important
        // 3. Properties with hyphens (avoids camelCase)
    </script>
</body>
</html>
```
**Why:** Use setProperty() to dynamically change CSS variables, set !important rules, or manipulate any CSS property programmatically.
