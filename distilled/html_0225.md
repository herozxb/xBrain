                let count = start;  // Private variable in closure memory
                
                return {
                    increment: function() {
                        count++;
                        return count;
                    },
                    decrement: function() {
                        count--;
                        return count;
                    },
                    getCount: function() {
                        return count;
                    }
                };
            }

         * 2. Variables stay in memory as long as the closure exists

         * 4. Can lead to memory leaks if closures hold references unnecessarily
