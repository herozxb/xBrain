## 1. Bridge Pattern Variants

```python
# PROBLEM: Decouple abstraction from implementation to allow both to vary independently
# TIME: O(1) SPACE: O(n) for object composition
from abc import ABC, abstractmethod

class DrawAPI(ABC):
    """Implementation interface for drawing operations"""
    @abstractmethod
    def draw_circle(self, radius: int, x: int, y: int) -> None:
        pass

class RedCircle(DrawAPI):
    """Concrete implementation - draws red circles"""
    def draw_circle(self, radius: int, x: int, y: int) -> None:
        print(f"Drawing Circle[ color: red, radius: {radius}, x: {x}, y: {y}]")

class GreenCircle(DrawAPI):
    """Concrete implementation - draws green circles"""
    def draw_circle(self, radius: int, x: int, y: int) -> None:
        print(f"Drawing Circle[ color: green, radius: {radius}, x: {x}, y: {y}]")

class Shape(ABC):
    """Abstraction - uses composition to delegate to implementation"""
    def __init__(self, draw_api: DrawAPI):
        self._draw_api = draw_api
    
    @abstractmethod
    def draw(self) -> None:
        pass

class Circle(Shape):
    """Refined abstraction - extends Shape interface"""
    def __init__(self, radius: int, x: int, y: int, draw_api: DrawAPI):
        super().__init__(draw_api)
        self._radius = radius
        self._x = x
        self._y = y
    
    def draw(self) -> None:
        # Delegates to implementation object
        self._draw_api.draw_circle(self._radius, self._x, self._y)

# Usage example
if __name__ == "__main__":
    red_circle = Circle(100, 10, 20, RedCircle())
    green_circle = Circle(50, 5, 10, GreenCircle())
    
    red_circle.draw()
    green_circle.draw()
```
**Why**: Enables runtime binding of implementations, promotes loose coupling, and allows independent extension of both abstraction and implementation hierarchies.

## 2. Composite Pattern Variants

```python
# PROBLEM: Compose objects into tree structures to represent part-whole hierarchies
# TIME: O(n) for traversal SPACE: O(h) where h is tree height
from abc import ABC, abstractmethod
from typing import List

class Component(ABC):
    """Common interface for all components in the tree"""
    @abstractmethod
    def operation(self) -> str:
        pass
    
    @abstractmethod
    def add(self, component: 'Component') -> None:
        pass
    
    @abstractmethod
    def remove(self, component: 'Component') -> None:
        pass
    
    @abstractmethod
    def get_child(self, index: int) -> 'Component':
        pass

class Leaf(Component):
    """Represents leaf objects with no children"""
    def operation(self) -> str:
        return "Leaf"
    
    def add(self, component: Component) -> None:
        raise NotImplementedError("Cannot add to a leaf")
    
    def remove(self, component: Component) -> None:
        raise NotImplementedError("Cannot remove from a leaf")
    
    def get_child(self, index: int) -> Component:
        raise NotImplementedError("Leaf has no children")

class Composite(Component):
    """Stores child components and implements child-related operations"""
    def __init__(self):
        self._children: List[Component] = []
    
    def operation(self) -> str:
        # Recursively collect results from all children
        results = []
        for child in self._children:
            results.append(child.operation())
        return f"Branch({', '.join(results)})"
    
    def add(self, component: Component) -> None:
        self._children.append(component)
    
    def remove(self, component: Component) -> None:
        self._children.remove(component)
    
    def get_child(self, index: int) -> Component:
        return self._children[index]

# Usage example
if __name__ == "__main__":
    root = Composite()
    branch1 = Composite()
    branch2 = Composite()
    
    leaf1 = Leaf()
    leaf2 = Leaf()
    leaf3 = Leaf()
    
    branch1.add(leaf1)
    branch1.add(leaf2)
    branch2.add(leaf3)
    
    root.add(branch1)
    root.add(branch2)
    
    print(root.operation())  # Output: Branch(Branch(Leaf, Leaf), Branch(Leaf))
```
**Why**: Clients treat individual objects and compositions uniformly, simplifies client code, and makes it easier to add new component types.

## 3. Decorator Pattern Variants

```python
# PROBLEM: Add responsibilities to objects dynamically without inheritance
# TIME: O(1) SPACE: O(d) where d is decorator depth
from abc import ABC, abstractmethod

class Coffee(ABC):
    """Base component interface"""
    @abstractmethod
    def cost(self) -> float:
        pass
    
    @abstractmethod
    def description(self) -> str:
        pass

class SimpleCoffee(Coffee):
    """Concrete component"""
    def cost(self) -> float:
        return 2.0
    
    def description(self) -> str:
        return "Simple coffee"

class CoffeeDecorator(Coffee):
    """Base decorator class - wraps component"""
    def __init__(self, coffee: Coffee):
        self._coffee = coffee
    
    def cost(self) -> float:
        return self._coffee.cost()
    
    def description(self) -> str:
        return self._coffee.description()

class MilkDecorator(CoffeeDecorator):
    """Concrete decorator - adds milk"""
    def cost(self) -> float:
        return super().cost() + 0.5
    
    def description(self) -> str:
        return f"{super().description()}, with milk"

class SugarDecorator(CoffeeDecorator):
    """Concrete decorator - adds sugar"""
    def cost(self) -> float:
        return super().cost() + 0.2
    
    def description(self) -> str:
        return f"{super().description()}, with sugar"

class WhipDecorator(CoffeeDecorator):
    """Concrete decorator - adds whip"""
    def cost(self) -> float:
        return super().cost() + 0.7
    
    def description(self) -> str:
        return f"{super().description()}, with whip"

# Usage example
if __name__ == "__main__":
    # Create base coffee and decorate it
    coffee = SimpleCoffee()
    coffee_with_milk = MilkDecorator(coffee)
    coffee_with_milk_sugar = SugarDecorator(coffee_with_milk)
    
    print(coffee_with_milk_sugar.description())
    print(f"Cost: ${coffee_with_milk_sugar.cost()}")
    
    # Stack multiple decorators
    fancy_coffee = WhipDecorator(MilkDecorator(SugarDecorator(SimpleCoffee())))
    print(fancy_coffee.description())
```
**Why**: More flexible than static inheritance, avoids feature-laden classes high in hierarchy, and supports runtime addition/removal of responsibilities.

## 4. Anti-Corruption Layer (ACL)

```python
# PROBLEM: Protect domain model from external system's data model inconsistencies
# TIME: O(1) for translation SPACE: O(n) for cached mappings
from dataclasses import dataclass
from typing import Dict, Optional
from datetime import datetime

@dataclass
class OrderDTO:
    """Data Transfer Object from legacy system"""
    order_id: str
    customer_name: str
    items_data: str  # JSON string from legacy system
    created_timestamp: int  # Unix timestamp
    status_code: int  # Legacy status codes

@dataclass
class Order:
    """Clean domain model"""
    id: str
    customer_name: str
    items: list
    created_at: datetime
    status: str

class OrderTranslator:
    """ACL component that translates between models"""
    
    STATUS_MAPPING = {
        0: "pending",
        1: "processing",
        2: "shipped",
        3: "delivered",
        4: "cancelled"
    }
    
    @staticmethod
    def to_domain(dto: OrderDTO) -> Order:
        """Translate legacy DTO to domain model"""
        # Parse items from JSON string
        items = eval(dto.items_data) if dto.items_data else []
        
        # Convert Unix timestamp to datetime
        created_at = datetime.fromtimestamp(dto.created_timestamp)
        
        # Map status code to readable status
        status = OrderTranslator.STATUS_MAPPING.get(
            dto.status_code, "unknown"
        )
        
        return Order(
            id=dto.order_id,
            customer_name=dto.customer_name,
            items=items,
            created_at=created_at,
            status=status
        )
    
    @staticmethod
    def to_dto(order: Order) -> OrderDTO:
        """Translate domain model to legacy DTO"""
        # Reverse mapping
        status_code = next(
            (k for k, v in OrderTranslator.STATUS_MAPPING.items() 
             if v == order.status), 0
        )
        
        return OrderDTO(
            order_id=order.id,
            customer_name=order.customer_name,
            items_data=str(order.items),
            created_timestamp=int(order.created_at.timestamp()),
            status_code=status_code
        )

class LegacyOrderService:
    """Facade for legacy system operations"""
    
    def __init__(self):
        self._translator = OrderTranslator()
    
    def get_order(self, order_id: str) -> Order:
        """Fetch from legacy system and translate to domain model"""
        # Simulate legacy system call
        legacy_dto = OrderDTO(
            order_id=order_id,
            customer_name="John Doe",
            items_data='["item1", "item2"]',
            created_timestamp=1672531200,
            status_code=1
        )
        
        # Translate through ACL
        return self._translator.to_domain(legacy_dto)

# Usage example
if __name__ == "__main__":
    service = LegacyOrderService()
    order = service.get_order("12345")
    
    print(f"Order ID: {order.id}")
    print(f"Customer: {order.customer_name}")
    print(f"Items: {order.items}")
    print(f"Created: {order.created_at}")
    print(f"Status: {order.status}")
```
**Why**: Maintains domain model integrity, isolates bounded contexts, and enables gradual migration from legacy systems without corrupting business logic.

## 5. Lost Update Prevention

```python
# PROBLEM: Prevent concurrent updates from overwriting each other's changes
# TIME: O(1) for check SPACE: O(1) for version storage
from dataclasses import dataclass
from typing import Optional
import threading

@dataclass
class Account:
    """Entity with optimistic locking support"""
    id: str
    balance: float
    version: int = 0  # Version counter for optimistic locking

class OptimisticLockError(Exception):
    """Raised when version mismatch detected"""
    pass

class AccountRepository:
    """Repository with optimistic locking enforcement"""
    
    def __init__(self):
        self._accounts: dict[str, Account] = {}
        self._lock = threading.Lock()
    
    def get_by_id(self, account_id: str) -> Optional[Account]:
        """Retrieve account (includes version)"""
        return self._accounts.get(account_id)
    
    def save(self, account: Account, expected_version: int) -> None:
        """
        Save account with version check.
        Raises OptimisticLockError if version mismatch.
        """
        with self._lock:
            existing = self._accounts.get(account.id)
            
            # First save - create new
            if existing is None:
                account.version = 1
                self._accounts[account.id] = account
                return
            
            # Check version before update
            if existing.version != expected_version:
                raise OptimisticLockError(
                    f"Version mismatch: expected {expected_version}, "
                    f"but current is {existing.version}"
                )
            
            # Increment version on successful update
            account.version = existing.version + 1
            self._accounts[account.id] = account

class TransactionService:
    """Service demonstrating lost update prevention"""
    
    def __init__(self, repository: AccountRepository):
        self._repository = repository
    
    def transfer(
        self, 
        from_id: str, 
        to_id: str, 
        amount: float,
        from_version: int,
        to_version: int
    ) -> None:
        """Transfer with optimistic locking"""
        # Load accounts with current versions
        from_account = self._repository.get_by_id(from_id)
        to_account = self._repository.get_by_id(to_id)
        
        if not from_account or not to_account:
            raise ValueError("Account not found")
        
        # Perform business logic
        if from_account.balance < amount:
            raise ValueError("Insufficient funds")
        
        from_account.balance -= amount
        to_account.balance += amount
        
        # Save with version check - will fail if concurrent update occurred
        try:
            self._repository.save(from_account, from_version)
            self._repository.save(to_account, to_version)
        except OptimisticLockError as e:
            # Retry logic or propagate error
            raise Exception(
                f"Transfer failed due to concurrent modification: {e}"
            )

# Usage example
if __name__ == "__main__":
    repo = AccountRepository()
    
    # Initial setup
    acc1 = Account(id="acc1", balance=1000.0)
    acc2 = Account(id="acc2", balance=500.0)
    repo.save(acc1, 0)
    repo.save(acc2, 0)
    
    # Simulate concurrent access
    service = TransactionService(repo)
    
    # Both operations read same version
    acc1_v1 = repo.get_by_id("acc1")
    acc1_v2 = repo.get_by_id("acc1")  # Same version
    
    # First update succeeds
    service.transfer("acc1", "acc2", 100.0, 
                     from_version=1, to_version=1)
    
    # Second update fails - version changed
    try:
        service.transfer("acc1", "acc2", 50.0,
                        from_version=1, to_version=1)  # Old version
    except Exception as e:
        print(f"Expected error: {e}")
```
**Why**: Detects concurrent modifications without database locks, scales better than pessimistic locking, and provides clear conflict resolution points.

## 6. Nonrepeatable Read Prevention

```python
# PROBLEM: Prevent reading different values for same row within one transaction
# TIME: O(1) SPACE: O(m) where m is number of locked rows
from dataclasses import dataclass, field
from typing import Dict, Optional
from datetime import datetime
import threading
import time

@dataclass
class Transaction:
    """Transaction context with isolation level"""
    id: str
    start_time: datetime = field(default_factory=datetime.now)
    locks_held: set = field(default_set_factory=set)

@dataclass
class Product:
    """Entity with version tracking"""
    id: str
    name: str
    price: float
    version: int = 0

class PessimisticLockManager:
    """Manages row-level locks for repeatable reads"""
    
    def __init__(self):
        self._row_locks: Dict[str, threading.Lock] = {}
        self._global_lock = threading.Lock()
    
    def acquire_read_lock(self, row_id: str, txn: Transaction) -> bool:
        """
        Acquire shared lock for reading.
        Multiple readers allowed, no writers while locks held.
        """
        with self._global_lock:
            if row_id not in self._row_locks:
                self._row_locks[row_id] = threading.Lock()
            
            lock = self._row_locks[row_id]
            # Try non-blocking acquire
            if lock.acquire(blocking=False):
                txn.locks_held.add(row_id)
                return True
            return False
    
    def release_locks(self, txn: Transaction) -> None:
        """Release all locks held by transaction"""
        with self._global_lock:
            for row_id in txn.locks_held:
                if row_id in self._row_locks:
                    self._row_locks[row_id].release()
            txn.locks_held.clear()

class ProductRepository:
    """Repository with repeatable read isolation"""
    
    def __init__(self):
        self._products: Dict[str, Product] = {}
        self._lock_manager = PessimisticLockManager()
    
    def get_with_lock(
        self, 
        product_id: str, 
        txn: Transaction
    ) -> Optional[Product]:
        """
        Get product and hold lock for transaction duration.
        Ensures same value read multiple times.
        """
        # Acquire lock to prevent concurrent modifications
        if not self._lock_manager.acquire_read_lock(product_id, txn):
            raise Exception(
                f"Could not acquire lock for product {product_id}"
            )
        
        # Return current state - guaranteed not to change
        return self._products.get(product_id)
    
    def update_price(
        self, 
        product_id: str, 
        new_price: float,
        txn: Optional[Transaction] = None
    ) -> None:
        """Update price - blocked if lock held"""
        if txn and product_id in txn.locks_held:
            # Transaction holds lock - allow update
            product = self._products[product_id]
            product.price = new_price
            product.version += 1
        elif product_id in self._products:
            # Check if locked by another transaction
            if not self._lock_manager.acquire_read_lock(product_id, txn or Transaction(id="temp")):
                raise Exception(f"Product {product_id} is locked")
            # Release immediately for non-transactional update
            self._products[product_id].price = new_price

class PriceAnalyzer:
    """Demonstrates nonrepeatable read prevention"""
    
    def __init__(self, repo: ProductRepository):
        self._repo = repo
    
    def analyze_price_change(self, product_id: str) -> float:
        """
        Read price twice within transaction.
        Without protection, second read could see different value.
        """
        txn = Transaction(id=f"analyze_{product_id}")
        
        try:
            # First read - acquires lock
            product = self._repo.get_with_lock(product_id, txn)
            if not product:
                raise ValueError("Product not found")
            
            price1 = product.price
            print(f"First read: {price1}")
            
            # Simulate some processing time
            time.sleep(0.1)
            
            # Second read - sees same value due to lock
            product = self._repo._products[product_id]
            price2 = product.price
            print(f"Second read: {price2}")
            
            # Guaranteed to be same value
            assert price1 == price2, "Nonrepeatable read detected!"
            
            return price2
            
        finally:
            # Release locks
            self._repo._lock_manager.release_locks(txn)

# Usage example
if __name__ == "__main__":
    repo = ProductRepository()
    
    # Setup
    repo._products["p1"] = Product(
        id="p1", name="Widget", price=10.0
    )
    
    analyzer = PriceAnalyzer(repo)
    
    # This guarantees consistent reads
    price = analyzer.analyze_price_change("p1")
    print(f"Final price: {price}")
```
**Why**: Ensures data consistency within transactions, prevents phantom reads, and maintains isolation guarantees required by many business operations.

## 7. Abstract Method Patterns

```python
# PROBLEM: Define interface contract that subclasses must implement
# TIME: O(1) SPACE: O(1)
from abc import ABC, abstractmethod
from typing import List, Optional

class DataProcessor(ABC):
    """Abstract base class defining processing contract"""
    
    @abstractmethod
    def load_data(self, source: str) -> List[dict]:
        """Must be implemented by all processors"""
        pass
    
    @abstractmethod
    def process(self, data: List[dict]) -> List[dict]:
        """Transformation logic - subclass specific"""
        pass
    
    @abstractmethod
    def validate(self, data: List[dict]) -> bool:
        """Data validation rules"""
        pass
    
    @abstractmethod
    def save(self, data: List[dict], destination: str) -> bool:
        """Persistence mechanism"""
        pass
    
    # Template method - common algorithm
    def execute(self, source: str, destination: str) -> bool:
        """Orchestrates the processing pipeline"""
        data = self.load_data(source)
        
        if not self.validate(data):
            raise ValueError("Data validation failed")
        
        processed = self.process(data)
        
        if not self.validate(processed):
            raise ValueError("Processed data validation failed")
        
        return self.save(processed, destination)

class CSVProcessor(DataProcessor):
    """Concrete processor for CSV files"""
    
    def load_data(self, source: str) -> List[dict]:
        # Implementation for CSV loading
        print(f"Loading CSV from {source}")
        return [{"id": 1, "value": "test"}]
    
    def process(self, data: List[dict]) -> List[dict]:
        # CSV-specific processing
        print(f"Processing {len(data)} CSV records")
        return [{"id": item["id"], "processed": True} for item in data]
    
    def validate(self, data: List[dict]) -> bool:
        # CSV validation rules
        required_keys = {"id", "value"}
        return all(required_keys.issubset(item.keys()) for item in data)
    
    def save(self, data: List[dict], destination: str) -> bool:
        # CSV persistence
        print(f"Saving {len(data)} records to {destination}")
        return True

class JSONProcessor(DataProcessor):
    """Concrete processor for JSON files"""
    
    def load_data(self, source: str) -> List[dict]:
        # Implementation for JSON loading
        print(f"Loading JSON from {source}")
        return [{"uuid": "abc", "data": "value"}]
    
    def process(self, data: List[dict]) -> List[dict]:
        # JSON-specific processing
        print(f"Processing {len(data)} JSON records")
        return [{"uuid": item["uuid"], "processed": True} for item in data]
    
    def validate(self, data: List[dict]) -> bool:
        # JSON validation rules
        required_keys = {"uuid", "data"}
        return all(required_keys.issubset(item.keys()) for item in data)
    
    def save(self, data: List[dict], destination: str) -> bool:
        # JSON persistence
        print(f"Saving {len(data)} records to {destination}")
        return True

# Usage example
if __name__ == "__main__":
    # Cannot instantiate abstract class
    # processor = DataProcessor()  # TypeError
    
    # Use concrete implementations
    csv_processor = CSVProcessor()
    csv_processor.execute("data.csv", "output.csv")
    
    json_processor = JSONProcessor()
    json_processor.execute("data.json", "output.json")
```
**Why**: Enforces implementation contracts, provides compile-time safety, enables polymorphism, and documents expected interface clearly.

## 8. Interface Patterns

```python
# PROBLEM: Define type contracts for structural typing and duck typing
# TIME: O(1) SPACE: O(1)
from typing import Protocol, runtime_checkable, List
from dataclasses import dataclass

@runtime_checkable
class Serializable(Protocol):
    """Protocol for objects that can be serialized"""
    
    def to_dict(self) -> dict:
        """Convert object to dictionary"""
        ...
    
    @classmethod
    def from_dict(cls, data: dict) -> 'Serializable':
        """Create object from dictionary"""
        ...

@runtime_checkable
class Comparable(Protocol):
    """Protocol for comparable objects"""
    
    def __lt__(self, other: 'Comparable') -> bool:
        ...
    
    def __gt__(self, other: 'Comparable') -> bool:
        ...
    
    def __eq__(self, other: object) -> bool:
        ...

@runtime_checkable
class Repository(Protocol):
    """Protocol for repository pattern"""
    
    def get(self, id: str) -> object:
        ...
    
    def save(self, entity: object) -> None:
        ...
    
    def delete(self, id: str) -> None:
        ...
    
    def list(self) -> List[object]:
        ...

@dataclass
class User:
    """Domain model implementing Serializable"""
    id: str
    name: str
    email: str
    
    def to_dict(self) -> dict:
        return {
            "id": self.id,
            "name": self.name,
            "email": self.email
        }
    
    @classmethod
    def from_dict(cls, data: dict) -> 'User':
        return cls(
            id=data["id"],
            name=data["name"],
            email=data["email"]
        )

class InMemoryUserRepository:
    """Repository implementation"""
    
    def __init__(self):
        self._storage: dict[str, User] = {}
    
    def get(self, id: str) -> User:
        return self._storage.get(id)
    
    def save(self, entity: User) -> None:
        self._storage[entity.id] = entity
    
    def delete(self, id: str) -> None:
        self._storage.pop(id, None)
    
    def list(self) -> List[User]:
        return list(self._storage.values())

def serialize_to_json(obj: Serializable) -> str:
    """Function accepting any Serializable object"""
    import json
    return json.dumps(obj.to_dict())

def find_max(items: List[Comparable]) -> Comparable:
    """Function accepting any Comparable objects"""
    if not items:
        raise ValueError("Empty list")
    
    max_item = items[0]
    for item in items[1:]:
        if item > max_item:
            max_item = item
    return max_item

def process_with_repo(repo: Repository, entity: object) -> None:
    """Function accepting any Repository implementation"""
    repo.save(entity)
    print(f"Saved entity: {repo.get(entity.id)}")

# Usage example
if __name__ == "__main__":
    # Protocol-based polymorphism
    user = User(id="1", name="Alice", email="alice@example.com")
    
    # Check protocol conformance at runtime
    print(f"Is Serializable: {isinstance(user, Serializable)}")
    
    # Use with generic serializer
    json_str = serialize_to_json(user)
    print(f"Serialized: {json_str}")
    
    # Use repository
    repo = InMemoryUserRepository()
    process_with_repo(repo, user)
    
    # User implements protocols implicitly
    # No explicit inheritance needed
```
**Why**: Enables structural typing, supports duck typing with type safety, allows protocol composition, and provides flexible interface definitions without inheritance.

## 9. Context Mapping (DDD)

```python
# PROBLEM: Define relationships and translations between bounded contexts
# TIME: O(1) for mapping SPACE: O(n) for context stores
from dataclasses import dataclass
from typing import Dict, Optional, Callable
from enum import Enum

class ContextRelationship(Enum):
    """Types of context relationships"""
    SHARED_KERNEL = "shared_kernel"
    CUSTOMER_SUPPLIER = "customer_supplier"
    CONFORMIST = "conformist"
    ANTICORRUPTION_LAYER = "anticorruption_layer"
    SEPARATE_WAYS = "separate_ways"
    OPEN_HOST_SERVICE = "open_host_service"

@dataclass
class BoundedContext:
    """Represents a bounded context in DDD"""
    name: str
    namespace: str
    models: Dict[str, type]

@dataclass
class ContextMap:
    """Maps relationships between bounded contexts"""
    upstream: str
    downstream: str
    relationship: ContextRelationship
    translator: Optional[Callable] = None

class ContextMappingService:
    """Manages context boundaries and translations"""
    
    def __init__(self):
        self._contexts: Dict[str, BoundedContext] = {}
        self._mappings: Dict[str, ContextMap] = {}
    
    def register_context(self, context: BoundedContext) -> None:
        """Register a bounded context"""
        self._contexts[context.name] = context
    
    def add_mapping(
        self, 
        upstream: str,
        downstream: str,
        relationship: ContextRelationship,
        translator: Optional[Callable] = None
    ) -> None:
        """Define relationship between contexts"""
        key = f"{upstream}->{downstream}"
        self._mappings[key] = ContextMap(
            upstream=upstream,
            downstream=downstream,
            relationship=relationship,
            translator=translator
        )
    
    def translate(
        self, 
        model: object,
        from_context: str,
        to_context: str
    ) -> object:
        """Translate model between contexts"""
        key = f"{from_context}->{to_context}"
        mapping = self._mappings.get(key)
        
        if not mapping:
            raise ValueError(
                f"No mapping found: {from_context} -> {to_context}"
            )
        
        if mapping.translator:
            return mapping.translator(model)
        
        # Default: conformist (pass-through)
        return model
    
    def get_relationship(
        self, 
        upstream: str, 
        downstream: str
    ) -> Optional[ContextRelationship]:
        """Get relationship type between contexts"""
        key = f"{upstream}->{downstream}"
        mapping = self._mappings.get(key)
        return mapping.relationship if mapping else None
```

## 10. Write Skew Prevention

```python
# PROBLEM: Prevent concurrent transactions from creating inconsistent state
# TIME: O(n) for range checks SPACE: O(r) for range locks
from dataclasses import dataclass
from typing import Dict, List, Set, Optional
import threading
from datetime import datetime

@dataclass
class Doctor:
    """Entity in scheduling system"""
    id: str
    name: str
    on_call: bool = False

@dataclass
class Shift:
    """Represents a scheduled shift"""
    id: str
    doctor_id: str
    date: datetime
    time_slot: str

class RangeLock:
    """Lock for range queries"""
    
    def __init__(self):
        self._lock = threading.Lock()
        self._readers = 0
        self._writers = 0
    
    def acquire_read(self) -> None:
        with self._lock:
            while self._writers > 0:
                pass  # Spin wait
            self._readers += 1
    
    def release_read(self) -> None:
        with self._lock:
            self._readers -= 1
    
    def acquire_write(self) -> None:
        with self._lock:
            while self._writers > 0 or self._readers > 0:
                pass  # Spin wait
            self._writers += 1
    
    def release_write(self) -> None:
        with self._lock:
            self._writers -= 1

class ScheduleRepository:
    """Repository with serializable isolation"""
    
    def __init__(self):
        self._doctors: Dict[str, Doctor] = {}
        self._shifts: Dict[str, List[Shift]] = {}
        self._range_locks: Dict[str, RangeLock] = {}
        self._global_lock = threading.Lock()
    
    def get_on_call_doctors(self) -> List[Doctor]:
        """Query with predicate (needs range lock)"""
        lock = self._get_range_lock("on_call")
        lock.acquire_read()
        
        try:
            return [d for d in self._doctors.values() if d.on_call]
        finally:
            lock.release_read()
    
    def update_doctor_status(
        self, 
        doctor_id: str, 
        on_call: bool
    ) -> None:
        """
        Update with predicate check.
        Prevents write skew by locking the range.
        """
        lock = self._get_range_lock("on_call")
        lock.acquire_write()
        
        try:
            doctor = self._doctors.get(doctor_id)
            if not doctor:
                raise ValueError("Doctor not found")
            
            # Business rule: at least one doctor must be on call
            current_on_call = [
                d for d in self._doctors.values() if d.on_call
            ]
            
            if not on_call and len(current_on_call) <= 1:
                raise ValueError(
                    "Cannot remove last on-call doctor"
                )
            
            doctor.on_call = on_call
            
        finally:
            lock.release_write()
    
    def _get_range_lock(self, predicate: str) -> RangeLock:
        with self._global_lock:
            if predicate not in self._range_locks:
                self._range_locks[predicate] = RangeLock()
            return self._range_locks[predicate]

class SchedulingService:
    """Service demonstrating write skew prevention"""
    
    def __init__(self, repo: ScheduleRepository):
        self._repo = repo
    
    def request_time_off(self, doctor_id: str) -> bool:
        """
        Attempt to go off call.
        Without protection, two doctors could both
        successfully go off call simultaneously.
        """
        try:
            # Check current on-call status
            on_call = self._repo.get_on_call_doctors()
            print(f"Current on-call doctors: {len(on_call)}")
            
            # Attempt to go off call
            self._repo.update_doctor_status(doctor_id, False)
            print(f"Doctor {doctor_id} successfully went off call")
            return True
            
        except ValueError as e:
            print(f"Request denied: {e}")
            return False

# Usage example
if __name__ == "__main__":
    repo = ScheduleRepository()
    
    # Setup: two doctors on call
    repo._doctors["d1"] = Doctor(id="d1", name="Dr. Alice", on_call=True)
    repo._doctors["d2"] = Doctor(id="d2", name="Dr. Bob", on_call=True)
    
    service = SchedulingService(repo)
    
    # First request succeeds
    service.request_time_off("d1")
    
    # Second request fails - would leave no one on call
    service.request_time_off("d2")
    
    # Verify state
    on_call = repo.get_on_call_doctors()
    print(f"Doctors on call: {[d.name for d in on_call]}")
```
**Why**: Prevents subtle concurrency bugs, ensures data integrity under concurrent access, and provides serializable isolation without full table locks.
