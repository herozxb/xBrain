  <script>
    // Show page content and update URL without adding new history entry
    // PROBLEM: Navigate SPA without polluting browser history
    // TIME: O(1) SPACE: O(1)
    function showPage(pageName) {
      const content = document.getElementById('content');
      
      // Define page content
      const pages = {
        home: '<h2>Welcome Home</h2><p>This is the home page content.</p>',
        about: '<h2>About Us</h2><p>Learn more about our company.</p>',
        contact: '<h2>Contact</h2><p>Email: contact@example.com</p>'
      };
      
      // Update content
      content.innerHTML = pages[pageName];
      
      // Replace current history entry (no new entry added)
      // This overwrites the current history state
      const state = { page: pageName };
      const title = pageName.charAt(0).toUpperCase() + pageName.slice(1);
      const url = `#${pageName}`;
      
      // replaceState: replace current entry instead of adding new one
      window.history.replaceState(state, title, url);
    }
    
    // Handle initial page load with hash
    window.addEventListener('DOMContentLoaded', () => {
      const hash = window.location.hash.slice(1) || 'home';
      showPage(hash);
    });
  </script>
</body>
</html>
```

**Why**: Use replaceState to update URL in SPAs without creating extra history entries, keeping back button behavior clean.

    <div class="step" data-step="1">Step 1: Introduction</div>
    <div class="step" data-step="2">Step 2: Details</div>

    <button onclick="navigate(2)">Step 2</button>
    <button onclick="navigate(3)">Step 3</button>

  <p id="info">Current step: 1</p>

  <script>
    // Handle browser navigation (back/forward buttons)
    // PROBLEM: Sync UI state with browser history navigation
    // TIME: O(1) SPACE: O(1)
    
    let currentStep = 1;
    
    // Navigate to a specific step
    function navigate(step) {
      currentStep = step;
      updateUI();
      
      // Add new history entry
      const state = { step: step };
      const title = `Step ${step}`;
      const url = `#step-${step}`;
      
      window.history.pushState(state, title, url);
    }
    
    // Update UI based on current step
    function updateUI() {
      // Update step indicators
      document.querySelectorAll('.step').forEach(el => {
        el.classList.remove('active');
        if (parseInt(el.dataset.step) === currentStep) {
          el.classList.add('active');
        }
      });
      
      // Update info text
      document.getElementById('info').textContent = `Current step: ${currentStep}`;
    }
    
    // Listen for popstate event (back/forward button clicks)
    // This fires when user navigates through history
    window.addEventListener('popstate', (event) => {
      // event.state contains the state object we passed to pushState
      if (event.state && event.state.step) {
        currentStep = event.state.step;
        updateUI();
      } else {
        // No state, default to step 1
        currentStep = 1;
        updateUI();
      }
    });
    
    // Handle initial page load
    window.addEventListener('DOMContentLoaded', () => {
      // Check if there's a hash in URL
      const hash = window.location.hash;
      const match = hash.match(/step-(\d+)/);
      
      if (match) {
        currentStep = parseInt(match[1]);
      }
      
      // Replace state for initial load
      window.history.replaceState({ step: currentStep }, '', `#step-${currentStep}`);
      updateUI();
    });
  </script>
</body>
</html>
```

**Why**: popstate event enables responsive SPAs that react to browser navigation, maintaining consistent state between UI and history.

    // PROBLEM: Navigate sections based on URL hash without page reload
    // TIME: O(1) SPACE: O(1)
    
    // Show section based on current hash
    function showSection() {
      // Get hash from URL (remove # symbol)
      const hash = window.location.hash.slice(1) || 'home';
      
      // Hide all sections
      document.querySelectorAll('section').forEach(section => {
        section.classList.remove('active');
      });
      
      // Show target section
      const targetSection = document.getElementById(hash);
      if (targetSection) {
        targetSection.classList.add('active');
      }
      
      // Update nav active state
      document.querySelectorAll('nav a').forEach(link => {
        link.classList.remove('active');
        if (link.getAttribute('href') === `#${hash}`) {
          link.classList.add('active');
        }
      });
    }
    
    // Listen for hashchange event
    // Fires when URL hash changes (link click or manual URL change)
    window.addEventListener('hashchange', showSection);
    
    // Handle initial page load
    window.addEventListener('DOMContentLoaded', () => {
      // Set default hash if none exists
      if (!window.location.hash) {
        window.location.hash = 'home';
      }
      showSection();
    });
  </script>

    let timerDate = new Date(0, 0, 0, 0, 0, 0); // Start at 00:00:00
    let timerInterval = null;
    let isRunning = false;
    
    // Update timer display
    function updateDisplay() {
      const hours = String(timerDate.getHours()).padStart(2, '0');
      const minutes = String(timerDate.getMinutes()).padStart(2, '0');

      timerInterval = setInterval(() => {
        // Get current seconds
        const currentSeconds = timerDate.getSeconds();

        // setSeconds handles overflow automatically (>59 increments minutes)
        timerDate.setSeconds(currentSeconds + 1);
        
        updateDisplay();

    function stopTimer() {
      if (!isRunning) return;
      
      isRunning = false;
      clearInterval(timerInterval);
      document.getElementById('status').textContent = 'Timer stopped';

    function resetTimer() {

    updateDisplay();

  <script>
    // Demonstrate Math.cos() for wave animation
    // PROBLEM: Calculate horizontal position in circular motion/wave
    // TIME: O(n) SPACE: O(1) where n = number of points
    
    const canvas = document.getElementById('waveCanvas');
    const ctx = canvas.getContext('2d');
    const width = canvas.width;
    const height = canvas.height;
    
    let animationId = null;
    let offset = 0;
    
    // Draw a sine wave using Math.cos (shifted sine)
    function drawWave(phase) {
      ctx.clearRect(0, 0, width, height);
      
      // Draw grid
      ctx.strokeStyle = '#ddd';
      ctx.lineWidth = 1;
      
      // Horizontal center line
      ctx.beginPath();
      ctx.moveTo(0, height / 2);
      ctx.lineTo(width, height / 2);
      ctx.stroke();
      
      // Draw cosine wave
      ctx.strokeStyle = '#007bff';
      ctx.lineWidth = 3;
      ctx.beginPath();
      
      const amplitude = height / 3;
      const frequency = 0.02; // How many cycles per pixel
      
      for (let x = 0; x < width; x++) {
        // Math.cos returns value between -1 and 1
        // Angle = (x * frequency) + phase (in radians)
        const angle = (x * frequency) + phase;
        const cosineValue = Math.cos(angle);
        
        // Convert to Y coordinate (0 to height)
        const y = (height / 2) - (cosineValue * amplitude);
        
        if (x === 0) {
          ctx.moveTo(x, y);
        } else {
          ctx.lineTo(x, y);
        }
      }
      
      ctx.stroke();
      
      // Draw moving point on wave
      const pointX = width / 2;
      const pointAngle = (pointX * frequency) + phase;
      const pointY = (height / 2) - (Math.cos(pointAngle) * amplitude);
      
      ctx.fillStyle = '#dc3545';
      ctx.beginPath();
      ctx.arc(pointX, pointY, 10, 0, Math.PI * 2);
      ctx.fill();
      
      // Display current cosine value
      const cosValue = Math.cos(pointAngle);
      document.getElementById('output').innerHTML = `
        Point angle: ${(pointAngle % (2 * Math.PI)).toFixed(2)} radians<br>
        Math.cos(angle) = ${cosValue.toFixed(4)}<br>
        Y position: ${pointY.toFixed(2)}<br><br>
        Note: Math.cos() takes radians, not degrees<br>
        To convert: radians = degrees × (π / 180)
      `;
    }
    
    // Animate the wave
    function animateWave() {
      if (animationId) {
        cancelAnimationFrame(animationId);
      }
      
      function animate() {
        offset += 0.05;
        drawWave(offset);
        animationId = requestAnimationFrame(animate);
      }
      
      animate();
    }
    
    // Draw static wave
    function drawStatic() {
      if (animationId) {
        cancelAnimationFrame(animationId);
        animationId = null;
      }
      drawWave(0);
    }
    
    // Clear canvas
    function clearCanvas() {
      if (animationId) {
        cancelAnimationFrame(animationId);
        animationId = null;
      }
      ctx.clearRect(0, 0, width, height);
    }
    
    // Initialize with static wave
    drawStatic();
  </script>
