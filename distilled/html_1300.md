```html
        /**
         * Handle click events with detail tracking
         * # PROBLEM: Detect single vs double vs triple clicks using event.detail
         * # TIME: O(1) SPACE: O(1)
         */
        function setupClickTracking(elementId, logId) {
            const element = document.getElementById(elementId);
            const log = document.getElementById(logId);
            
            if (!element || !log) return;

            /**
             * Get click detail information
             * # PROBLEM: Retrieve the number of consecutive clicks
             * # TIME: O(1) SPACE: O(1)
             */
            function getClickDetail(event) {
                // event.detail returns:
                // 1 for single click
                // 2 for double click
                // 3 for triple click
                // etc.
                return event.detail;
            }

            element.addEventListener('click', (e) => {
                const detail = getClickDetail(e);
                const clickType = 
                    detail === 1 ? 'Single Click' :
                    detail === 2 ? 'Double Click' :
                    detail === 3 ? 'Triple Click' :
                    `${detail}x Click`;
                
                const entry = document.createElement('div');
                entry.className = 'log-entry';
                entry.innerHTML = `
                    <strong>${clickType}</strong> (detail: ${detail})<br>
                    <small>Time: ${new Date().toLocaleTimeString()}</small>
                `;
                log.insertBefore(entry, log.firstChild);
                
                // Keep only last 10 entries
                while (log.children.length > 10) {
                    log.removeChild(log.lastChild);
                }
            });

            // Alternative: listen for dblclick event
            element.addEventListener('dblclick', (e) => {
                console.log('Double-click event fired');
                console.log('Detail:', e.detail); // Will be 2
            });
        }

        setupClickTracking('click-area', 'log');
    </script>
</body>
</html>
```

**Why:** The event.detail property is useful for detecting multiple clicks without setting up manual timing logic, enabling natural single/double/triple click interactions and distinguishing between them for different actions.

```html
                return event.clientY;

                // Get position relative to the draw area
                const rect = area.getBoundingClientRect();

                if (e.buttons === 1) { // Left mouse button pressed
```

```html
        /**
         * Handle keyboard events using key property
         * # PROBLEM: Detect which key was pressed using the key property
         * # TIME: O(1) SPACE: O(1)
         */
        function setupKeyboardTracking(inputId) {
            const input = document.getElementById(inputId);
            const lastKeyDisplay = document.getElementById('last-key');
            const keyHistory = document.getElementById('key-history');
            const keyHistoryArray = [];

            /**
             * Get key value from keyboard event
             * # PROBLEM: Retrieve the string value of the pressed key
             * # TIME: O(1) SPACE: O(1)
             */
            function getKey(event) {
                // event.key returns:
                // - Character for printable keys ("a", "5", "@")
                // - Named values for special keys ("Enter", "Escape", "ArrowUp")
                // - " " for space
                // - "Control", "Alt", "Shift", "Meta" for modifier keys
                return event.key;
            }

                const key = getKey(e);

                lastKeyDisplay.textContent = key;
                
                // Add to history (keep last 10)
                keyHistoryArray.unshift(key);
                if (keyHistoryArray.length > 10) {
                    keyHistoryArray.pop();
                }
                keyHistory.textContent = keyHistoryArray.join(' → ');

                // Handle keyboard shortcuts
                handleShortcuts(e);
            });

        /**
         * Handle keyboard shortcuts
         * # PROBLEM: Implement common keyboard shortcuts
         * # TIME: O(1) SPACE: O(1)
         */
        function handleShortcuts(event) {
            const key = event.key;
            
            // Ctrl/Cmd + S = Save
            if ((event.ctrlKey || event.metaKey) && key === 's') {
                event.preventDefault();
                alert('Save triggered! (Ctrl+S)');
            }
            
            // Ctrl/Cmd + Z = Undo
            else if ((event.ctrlKey || event.metaKey) && key === 'z') {
                event.preventDefault();
                alert('Undo triggered! (Ctrl+Z)');
            }
            
            // Escape = Clear
            else if (key === 'Escape') {
                document.getElementById('input-area').value = '';
                alert('Cleared! (Escape)');
            }
        }

        setupKeyboardTracking('input-area');

        // Global keyboard event listener example
        document.addEventListener('keydown', (e) => {
            // Detect arrow keys
            if (e.key.startsWith('Arrow')) {
                console.log(`Arrow key pressed: ${e.key}`);
            }
        });
    </script>
```

```html
        function getSessionStorageKey(index) {
            // sessionStorage.key(index) returns:
            // - The name of the key at the given index
            // - null if the index is out of range
            // - Order is not guaranteed to be consistent
            return sessionStorage.key(index);
        }

        /**
         * Get all keys from sessionStorage
         * # PROBLEM: Retrieve all key names from sessionStorage
         * # TIME: O(n) SPACE: O(n) where n is number of items

        function getAllSessionStorageKeys() {
            const keys = [];
            for (let i = 0; i < sessionStorage.length; i++) {
                const key = getSessionStorageKey(i);
                if (key !== null) {
                    keys.push(key);
                }
            }
```
