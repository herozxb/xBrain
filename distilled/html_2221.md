            const ogUrl = document.querySelector('meta[property="og:url"]');
            return ogUrl ? ogUrl.content : null;
        }
        
        // Dynamic og:url for single-page applications
        function updateOgUrl(newUrl) {
            let ogUrl = document.querySelector('meta[property="og:url"]');
            if (ogUrl) {
                ogUrl.content = newUrl;
            } else {
                ogUrl = document.createElement('meta');
                ogUrl.setAttribute('property', 'og:url');
                ogUrl.content = newUrl;
                document.head.appendChild(ogUrl);
            }

## html_meta_canonical

```html
<!-- PROBLEM: Prevent duplicate content issues by specifying the preferred URL -->
<!-- TIME: O(1) SPACE: O(1) -->

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Canonical URL Meta Tag</title>
    
    <!-- 
        rel="canonical": Tells search engines which version of a page to index
        Critical for SEO to avoid duplicate content penalties
        
        Common scenarios:
        1. Multiple URLs lead to same content
        2. Query parameters don't change content
        3. HTTP vs HTTPS versions
        4. WWW vs non-WWW versions
        5. Mobile vs desktop versions
    -->
    
    <!-- Example 1: Standard canonical URL -->
    <link rel="canonical" href="https://example.com/products/widget">
    
    <!-- Example 2: With query parameters (ignore tracking params) -->
    <!-- If URL is example.com/products/widget?utm_source=twitter -->
    <!-- Canonical should still be: -->
    <!-- <link rel="canonical" href="https://example.com/products/widget"> -->
    
    <!-- Example 3: Pagination -->
    <!-- Page 2 of products -->
    <!-- <link rel="canonical" href="https://example.com/products?page=2"> -->
    
    <!-- Also include og:url matching canonical -->
    <meta property="og:url" content="https://example.com/products/widget">
    
    <!-- HTTP header alternative (server-side) -->
    <!-- Link: <https://example.com/products/widget>; rel="canonical" -->
</head>
<body>
    <h1>Product: Widget</h1>
    <p>This page can be accessed via multiple URLs:</p>
    
    <ul>
        <li>https://example.com/products/widget (canonical)</li>
        <li>https://www.example.com/products/widget</li>
        <li>https://example.com/products/widget?ref=newsletter</li>
        <li>https://example.com/products/widget&utm_source=google</li>
    </ul>
    
    <p>The canonical tag ensures search engines know which version to index.</p>
    
    <script>
        // Function to check if current URL matches canonical
        function checkCanonical() {
            const canonical = document.querySelector('link[rel="canonical"]');
            if (canonical) {
                const canonicalUrl = canonical.href;
                const currentUrl = window.location.href;
                
                // Remove query params for comparison
                const currentBase = currentUrl.split('?')[0];
                const canonicalBase = canonicalUrl.split('?')[0];
                
                if (currentBase !== canonicalBase) {
                    console.warn('Current URL does not match canonical URL');
                    console.log('Current:', currentUrl);
                    console.log('Canonical:', canonicalUrl);
                    
                    // Optional: Redirect to canonical (use carefully!)
                    // window.location.href = canonicalUrl;
                }
            }
        }
        
        // Run on page load
        checkCanonical();
        
        // Function to set canonical dynamically
        function setCanonicalUrl(url) {
            let canonical = document.querySelector('link[rel="canonical"]');
            if (canonical) {
                canonical.href = url;
            } else {
                canonical = document.createElement('link');
                canonical.rel = 'canonical';
                canonical.href = url;
                document.head.appendChild(canonical);
            }
            console.log('Canonical URL set to:', url);
        }
        
        // Example: Set canonical for current page
        // setCanonicalUrl(window.location.origin + window.location.pathname);
    </script>
</body>
</html>
```

**Why**: The canonical tag prevents SEO duplicate content penalties by telling search engines which URL version to index. It consolidates ranking signals and ensures consistent indexing across URL variations.

        // Function to create download link dynamically
        function downloadFile(url, filename) {
            const link = document.createElement('a');
            link.href = url;
            link.download = filename || url.split('/').pop();
            
            // For programmatic downloads, need to add to DOM temporarily
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }
        
        // Function to download generated content
        function downloadTextFile(content, filename) {
            // Create blob from text content
            const blob = new Blob([content], { type: 'text/plain' });
            const url = URL.createObjectURL(blob);
            
            const link = document.createElement('a');
            link.href = url;
            link.download = filename || 'download.txt';
            
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            
            // Clean up object URL
            URL.revokeObjectURL(url);
        }

        function downloadJSON(data, filename) {
            const jsonString = JSON.stringify(data, null, 2);
            const blob = new Blob([jsonString], { type: 'application/json' });
            const url = URL.createObjectURL(blob);
            
            const link = document.createElement('a');
            link.href = url;
            link.download = filename || 'data.json';
            
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            
            URL.revokeObjectURL(url);
