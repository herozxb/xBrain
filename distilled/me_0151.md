## Two Sum
```python
# PROBLEM: Find two indices whose values sum to target
# TIME: O(n) SPACE: O(n)

def two_sum(nums: list, target: int) -> list:
    """
    Find two indices whose values add up to target.
    
    Args:
        nums: List of integers
        target: Target sum value
    
    Returns:
        List of two indices, or empty list if not found
    """
    seen = {}  # value -> index mapping
    
    for i, num in enumerate(nums):
        complement = target - num
        
        if complement in seen:
            return [seen[complement], i]
        
        seen[num] = i
    
    return []
```

**Why**: Hash map enables O(1) complement lookup, making overall solution O(n).

## Binary Search
```python
# PROBLEM: Find target in sorted array
# TIME: O(log n) SPACE: O(1)

def binary_search(arr: list, target: int) -> int:
    """
    Find target value in sorted array using binary search.
    
    Args:
        arr: Sorted list of integers
        target: Value to find
    
    Returns:
        Index of target, or -1 if not found
    """
    left, right = 0, len(arr) - 1
    
    while left <= right:
        mid = (left + right) // 2
        
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    
    return -1
```

**Why**: Halves search space each iteration, achieving O(log n) time complexity.

## Maximum Subarray
```python
# PROBLEM: Find contiguous subarray with largest sum
# TIME: O(n) SPACE: O(1)

def max_subarray(nums: list) -> int:
    """
    Find maximum sum of any contiguous subarray.
    
    Args:
        nums: List of integers (can include negatives)
    
    Returns:
        Maximum subarray sum
    """
    if not nums:
        return 0
    
    max_sum = curr_sum = nums[0]
    
    for num in nums[1:]:
        # Either start new subarray or extend current
        curr_sum = max(num, curr_sum + num)
        max_sum = max(max_sum, curr_sum)
    
    return max_sum
```

**Why**: Kadane's algorithm tracks best ending at each position in O(n) single pass.

## Valid Parentheses
```python
# PROBLEM: Check if brackets are properly matched
# TIME: O(n) SPACE: O(n)

def is_valid(s: str) -> bool:
    """
    Validate bracket string has proper matching.
    
    Args:
        s: String containing only ()[]{}
    
    Returns:
        True if valid, False otherwise
    """
    stack = []
    pairs = {')': '(', '}': '{', ']': '['}
    
    for char in s:
        if char in '({[':
            stack.append(char)
        elif char in ')}]':
            if not stack or stack.pop() != pairs[char]:
                return False
    
    return len(stack) == 0
```

**Why**: Stack naturally handles nested structures, matching opening with closing brackets.

## Merge Two Sorted Lists
```python
# PROBLEM: Merge two sorted linked lists into one
# TIME: O(n + m) SPACE: O(1)

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def merge_two_lists(l1: ListNode, l2: ListNode) -> ListNode:
    """
    Merge two sorted linked lists.
    
    Args:
        l1: Head of first sorted list
        l2: Head of second sorted list
    
    Returns:
        Head of merged sorted list
    """
    dummy = ListNode(0)
    curr = dummy
    
    while l1 and l2:
        if l1.val <= l2.val:
            curr.next = l1
            l1 = l1.next
        else:
            curr.next = l2
            l2 = l2.next
        curr = curr.next
    
    # Append remaining nodes
    curr.next = l1 if l1 else l2
    
    return dummy.next
```

**Why**: Two-pointer merge is optimal for sorted lists, O(n+m) time with O(1) extra space.

## Best Time to Buy/Sell Stock
```python
# PROBLEM: Find maximum profit from one transaction
# TIME: O(n) SPACE: O(1)

def max_profit(prices: list) -> int:
    """
    Find maximum profit from buying and selling once.
    
    Args:
        prices: List of daily stock prices
    
    Returns:
        Maximum profit (0 if no profit possible)
    """
    if not prices:
        return 0
    
    min_price = float('inf')
    max_profit = 0
    
    for price in prices:
        min_price = min(min_price, price)
        max_profit = max(max_profit, price - min_price)
    
    return max_profit
```

**Why**: Track minimum price seen so far, calculate profit at each step in single pass.

## Reverse Linked List
```python
# PROBLEM: Reverse a singly linked list
# TIME: O(n) SPACE: O(1)

def reverse_list(head: ListNode) -> ListNode:
    """
    Reverse linked list in place.
    
    Args:
        head: Head of linked list
    
    Returns:
        New head of reversed list
    """
    prev = None
    curr = head
    
    while curr:
        next_temp = curr.next
        curr.next = prev
        prev = curr
        curr = next_temp
    
    return prev
```

**Why**: Iterative reversal with three pointers is O(n) time and O(1) space.

## Climbing Stairs
```python
# PROBLEM: Count ways to climb n stairs (1 or 2 steps)
# TIME: O(n) SPACE: O(1)

def climb_stairs(n: int) -> int:
    """
    Count distinct ways to climb n stairs.
    
    Args:
        n: Number of stairs
    
    Returns:
        Number of distinct ways
    """
    if n <= 2:
        return n
    
    prev, curr = 1, 2
    
    for _ in range(3, n + 1):
        prev, curr = curr, prev + curr
    
    return curr
```

**Why**: Fibonacci pattern - each step is sum of ways to reach previous two steps.

## Contains Duplicate
```python
# PROBLEM: Check if array contains duplicates
# TIME: O(n) SPACE: O(n)

def contains_duplicate(nums: list) -> bool:
    """
    Check if any value appears at least twice.
    
    Args:
        nums: List of integers
    
    Returns:
        True if duplicate exists, False otherwise
    """
    seen = set()
    
    for num in nums:
        if num in seen:
            return True
        seen.add(num)
    
    return False
```

**Why**: Hash set provides O(1) membership test, making overall O(n).

## Maximum Depth of Binary Tree
```python
# PROBLEM: Find maximum depth of binary tree
# TIME: O(n) SPACE: O(h) where h is height

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def max_depth(root: TreeNode) -> int:
    """
    Find maximum depth of binary tree.
    
    Args:
        root: Root node of tree
    
    Returns:
        Maximum depth (0 if empty)
    """
    if not root:
        return 0
    
    return 1 + max(max_depth(root.left), max_depth(root.right))
```

**Why**: Recursive DFS naturally computes depth by finding max of subtrees.
