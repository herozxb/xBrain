## 1. Dispatcher View Pattern

```python
# PROBLEM: Centralized request routing with view dispatching
# TIME: O(n) for route matching, O(1) for dispatch SPACE: O(n) for routes
class DispatcherView:
    """Dispatches requests to appropriate view handlers based on routing."""
    
    def __init__(self):
        # Store route-to-handler mappings
        self.routes = {}
    
    def register(self, path, handler):
        """Register a view handler for a specific path."""
        self.routes[path] = handler
    
    def dispatch(self, request):
        """Dispatch request to matching view handler."""
        path = request.get('path', '/')
        handler = self.routes.get(path)
        
        if handler:
            return handler(request)
        else:
            return {'status': 404, 'message': 'View not found'}
```

**Why**: Separates routing logic from view handlers, enabling clean MVC architecture and easy route management.

## 2. Conformist Pattern (DDD)

```python
# PROBLEM: Integrating with external bounded context without translation
# TIME: O(1) SPACE: O(1)
class ConformistAdapter:
    """Conforms to external model without anti-corruption layer."""
    
    def __init__(self, external_service):
        self.external = external_service
    
    def fetch_data(self, resource_id):
        """Fetch data using external model directly."""
        # Conform to external schema - no translation
        return self.external.get(resource_id)
    
    def send_command(self, command):
        """Send command in external format."""
        # Accept external model as-is
        return self.external.execute(command)
```

**Why**: Simplifies integration when upstream model is stable and compatible with local model.

## 3. Repositories Pattern (DDD)

```python
# PROBLEM: Abstract data persistence and retrieval logic
# TIME: O(1) for simple ops, O(n) for queries SPACE: O(n) for storage
class Repository:
    """Domain-driven repository pattern for aggregate persistence."""
    
    def __init__(self):
        self.storage = {}
    
    def add(self, aggregate):
        """Store aggregate root."""
        self.storage[aggregate.id] = aggregate
    
    def get_by_id(self, aggregate_id):
        """Retrieve aggregate by identity."""
        return self.storage.get(aggregate_id)
    
    def get_by_specification(self, spec):
        """Query aggregates by specification."""
        return [agg for agg in self.storage.values() if spec.is_satisfied_by(agg)]
    
    def remove(self, aggregate_id):
        """Delete aggregate from repository."""
        if aggregate_id in self.storage:
            del self.storage[aggregate_id]
```

**Why**: Encapsulates persistence logic, keeping domain model focused on business rules.

## 4. Value Objects Pattern (DDD)

```python
# PROBLEM: Model domain concepts with identity based on attributes
# TIME: O(1) for comparison SPACE: O(n) for attributes
class ValueObject:
    """Immutable value object with equality based on attributes."""
    
    def __init__(self, **attributes):
        self._attributes = attributes
    
    def __eq__(self, other):
        """Value equality based on attributes, not identity."""
        if not isinstance(other, ValueObject):
            return False
        return self._attributes == other._attributes
    
    def __hash__(self):
        """Hash based on attribute values for set/dict use."""
        return hash(tuple(sorted(self._attributes.items())))
    
    def get(self, key):
        """Access attribute value."""
        return self._attributes.get(key)
```

**Why**: Enables immutable domain modeling with semantic equality comparison.

## 5. Pipeline Pattern

```python
# PROBLEM: Chain multiple processing steps in sequence
# TIME: O(n * m) where n=steps, m=data size SPACE: O(m) for intermediate
class Pipeline:
    """Composable pipeline of processing stages."""
    
    def __init__(self):
        self.steps = []
    
    def add_step(self, processor):
        """Add processing step to pipeline."""
        self.steps.append(processor)
        return self  # Enable fluent chaining
    
    def execute(self, data):
        """Run data through all pipeline steps."""
        result = data
        for step in self.steps:
            result = step.process(result)
        return result

class PipelineStep:
    """Base class for pipeline processing steps."""
    
    def process(self, data):
        """Override in subclasses for specific processing."""
        raise NotImplementedError
```

**Why**: Enables flexible data processing with composable, reusable transformation steps.

## 6. Composite Index Implementation

```python
# PROBLEM: Multi-column indexing for efficient querying
# TIME: O(log n) for lookup SPACE: O(n)
class CompositeIndex:
    """B-tree based composite index for multiple columns."""
    
    def __init__(self, columns):
        self.columns = columns  # List of column names
        self.index = {}  # Sorted dictionary simulation
    
    def insert(self, record):
        """Insert record into composite index."""
        # Build composite key from column values
        key = tuple(record[col] for col in self.columns)
        if key not in self.index:
            self.index[key] = []
        self.index[key].append(record)
    
    def search(self, values):
        """Search records matching composite key values."""
        key = tuple(values)
        return self.index.get(key, [])
    
    def range_search(self, start_values, end_values):
        """Search records within composite key range."""
        results = []
        start_key = tuple(start_values)
        end_key = tuple(end_values)
        
        for key in sorted(self.index.keys()):
            if start_key <= key <= end_key:
                results.extend(self.index[key])
        
        return results
```

**Why**: Accelerates multi-column queries by organizing data in sorted composite key order.

## 7. Columnar Storage Implementation

```python
# PROBLEM: Store data by columns for analytical workloads
# TIME: O(n) for column scan, O(1) for value access SPACE: O(n)
class ColumnarStorage:
    """Column-oriented storage for analytical queries."""
    
    def __init__(self, schema):
        self.schema = schema  # List of column names
        self.columns = {col: [] for col in schema}
        self.row_count = 0
    
    def insert(self, row):
        """Insert row, storing values by column."""
        for col in self.schema:
            self.columns[col].append(row.get(col))
        self.row_count += 1
    
    def get_column(self, column_name):
        """Fetch entire column for aggregate operations."""
        return self.columns[column_name]
    
    def aggregate(self, column_name, operation):
        """Perform aggregate operation on column."""
        col_data = self.columns[column_name]
        
        if operation == 'sum':
            return sum(col_data)
        elif operation == 'avg':
            return sum(col_data) / len(col_data) if col_data else 0
        elif operation == 'min':
            return min(col_data) if col_data else None
        elif operation == 'max':
            return max(col_data) if col_data else None
        
        return None
    
    def get_row(self, row_index):
        """Reconstruct row from columnar storage."""
        if 0 <= row_index < self.row_count:
            return {col: self.columns[col][row_index] for col in self.schema}
        return None
```

**Why**: Optimizes analytical queries that scan few columns across many rows.

## 8. Run-Length Encoding

```python
# PROBLEM: Compress consecutive repeated values
# TIME: O(n) for encode/decode SPACE: O(k) where k=unique runs
class RunLengthEncoder:
    """Compress sequences with run-length encoding."""
    
    @staticmethod
    def encode(data):
        """Encode data by replacing runs with (value, count) pairs."""
        if not data:
            return []
        
        encoded = []
        current_value = data[0]
        count = 1
        
        for i in range(1, len(data)):
            if data[i] == current_value:
                count += 1
            else:
                encoded.append((current_value, count))
                current_value = data[i]
                count = 1
        
        # Add last run
        encoded.append((current_value, count))
        return encoded
    
    @staticmethod
    def decode(encoded):
        """Decode run-length encoded data back to original."""
        decoded = []
        
        for value, count in encoded:
            decoded.extend([value] * count)
        
        return decoded
```

**Why**: Efficient compression for data with many consecutive repeated values.

## 9. Factory Pattern

```python
# PROBLEM: Create objects without specifying exact class
# TIME: O(1) SPACE: O(k) for registered types
class Factory:
    """Factory for creating objects by type identifier."""
    
    def __init__(self):
        self._creators = {}
    
    def register(self, type_name, creator_func):
        """Register creator function for a type."""
        self._creators[type_name] = creator_func
    
    def create(self, type_name, **kwargs):
        """Create object of specified type with parameters."""
        creator = self._creators.get(type_name)
        
        if not creator:
            raise ValueError(f"Unknown type: {type_name}")
        
        return creator(**kwargs)

# Example usage
class ProductFactory(Factory):
    """Specialized factory for product creation."""
    
    def __init__(self):
        super().__init__()
        # Register default product types
        self.register('standard', lambda **kw: StandardProduct(**kw))
        self.register('premium', lambda **kw: PremiumProduct(**kw))

class StandardProduct:
    def __init__(self, name, price):
        self.name = name
        self.price = price

class PremiumProduct:
    def __init__(self, name, price, warranty):
        self.name = name
        self.price = price
        self.warranty = warranty
```

**Why**: Decouples object creation from usage, enabling flexible instantiation.

## 10. Builder Pattern

```python
# PROBLEM: Construct complex objects step-by-step
# TIME: O(1) per step, O(n) for final build SPACE: O(n) for attributes
class QueryBuilder:
    """Builder for constructing complex database queries."""
    
    def __init__(self):
        self._table = None
        self._columns = []
        self._conditions = []
        self._joins = []
        self._order_by = None
        self._limit = None
    
    def select(self, *columns):
        """Specify columns to select."""
        self._columns.extend(columns)
        return self
    
    def from_table(self, table_name):
        """Specify table to query."""
        self._table = table_name
        return self
    
    def where(self, condition):
        """Add WHERE condition."""
        self._conditions.append(condition)
        return self
    
    def join(self, table, on_clause):
        """Add JOIN clause."""
        self._joins.append(f"JOIN {table} ON {on_clause}")
        return self
    
    def order_by(self, column, desc=False):
        """Set ORDER BY clause."""
        direction = "DESC" if desc else "ASC"
        self._order_by = f"{column} {direction}"
        return self
    
    def limit(self, count):
        """Set LIMIT clause."""
        self._limit = count
        return self
    
    def build(self):
        """Construct final SQL query string."""
        if not self._table:
            raise ValueError("Table name required")
        
        # Build SELECT clause
        columns = ", ".join(self._columns) if self._columns else "*"
        sql = f"SELECT {columns} FROM {self._table}"
        
        # Add JOINs
        for join in self._joins:
            sql += f" {join}"
        
        # Add WHERE clause
        if self._conditions:
            sql += " WHERE " + " AND ".join(self._conditions)
        
        # Add ORDER BY
        if self._order_by:
            sql += f" ORDER BY {self._order_by}"
        
        # Add LIMIT
        if self._limit:
            sql += f" LIMIT {self._limit}"
        
        return sql

# Example usage
query = (QueryBuilder()
         .select("id", "name", "email")
         .from_table("users")
         .where("active = 1")
         .where("created_at > '2025-01-01'")
         .order_by("created_at", desc=True)
         .limit(100)
         .build())
```

**Why**: Enables fluent, readable construction of complex objects with optional parameters.
