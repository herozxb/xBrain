## html_input_time

```html
<!-- Problem: Create time input with validation -->
<!-- Time: O(1) Space: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Time Input Handler</title>
</head>
<body>
    <label for="meeting-time">Select meeting time:</label>
    <input type="time" id="meeting-time" min="09:00" max="18:00" required>
    <button onclick="validateTime()">Check Time</button>
    <p id="result"></p>

    <script>
        // Validate time is within business hours (9 AM - 6 PM)
        function validateTime() {
            const timeInput = document.getElementById('meeting-time');
            const result = document.getElementById('result');
            const timeValue = timeInput.value;
            
            if (!timeValue) {
                result.textContent = 'Please select a time';
                result.style.color = 'red';
                return false;
            }
            
            const [hours, minutes] = timeValue.split(':').map(Number);
            const minHour = 9, maxHour = 18;
            
            // Check if time is within business hours
            if (hours >= minHour && hours <= maxHour) {
                if (hours === maxHour && minutes > 0) {
                    result.textContent = 'Time must be before 6:00 PM';
                    result.style.color = 'red';
                    return false;
                }
                result.textContent = `Valid time: ${timeValue}`;
                result.style.color = 'green';
                return true;
            } else {
                result.textContent = 'Please select time between 9:00 AM and 6:00 PM';
                result.style.color = 'red';
                return false;
            }
        }
        
        // Format time to 12-hour format with AM/PM
        function formatTime24to12(time24) {
            const [hours, minutes] = time24.split(':').map(Number);
            const period = hours >= 12 ? 'PM' : 'AM';
            const hours12 = hours % 12 || 12;
            return `${hours12}:${minutes.toString().padStart(2, '0')} ${period}`;
        }
    </script>
</body>
</html>
```

**Why**: Time inputs provide native time selection with built-in validation, making it easy to collect standardized time data without parsing errors.

```html
        </tbody>

    <script>
        // Calculate total from table cells
        function calculateTableTotal() {
            const cells = document.querySelectorAll('td');
            let total = 0;
            
            cells.forEach(cell => {
                const text = cell.textContent;
                if (text.includes('$')) {
                    const amount = parseFloat(text.replace(/[$,]/g, ''));
                    if (!isNaN(amount)) {
                        total += amount;
                    }
                }
            });
            
            console.log(`Total Sales: $${total.toLocaleString()}`);
            return total;
        }
        
        // Merge cells dynamically
        function mergeCells(row, startCol, span) {
            const table = document.querySelector('table');
            const targetRow = table.rows[row];
            
            if (targetRow && targetRow.cells[startCol]) {
                targetRow.cells[startCol].colSpan = span;
                // Remove merged cells
                for (let i = 1; i < span; i++) {
                    if (targetRow.cells[startCol + 1]) {
                        targetRow.deleteCell(startCol + 1);
                    }
                }
            }
        }
        
        calculateTableTotal();
    </script>
</body>
</html>
```

**Why**: Rowspan and colspan enable complex table layouts for reports and dashboards, allowing cells to span multiple rows or columns for better data organization.

```html
<!-- Problem: Object-oriented programming with ES6 classes -->
<!-- Time: O(1) Space: O(n) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>JavaScript Classes</title>
</head>

    <div id="output"></div>

    <script>
        // Base class
        class Animal {
            constructor(name, age) {
                this.name = name;
                this.age = age;
            }
            
            // Method
            speak() {
                return `${this.name} makes a sound.`;
            }
            
            // Getter
            get info() {
                return `${this.name} is ${this.age} years old`;
            }
            
            // Setter
            set updateAge(newAge) {
                if (newAge > 0) {
                    this.age = newAge;
                }
            }
            
            // Static method
            static create(name, age) {
                return new Animal(name, age);
            }
        }
        
        // Derived class (inheritance)
        class Dog extends Animal {
            constructor(name, age, breed) {
                super(name, age); // Call parent constructor
                this.breed = breed;
            }
            
            // Override parent method
            speak() {
                return `${this.name} barks!`;
            }
            
            // Additional method
            fetch() {
                return `${this.name} is fetching the ball.`;
            }
        }
        
        // Usage
        const output = document.getElementById('output');
        
        const genericAnimal = new Animal('Generic', 5);
        const myDog = new Dog('Buddy', 3, 'Golden Retriever');
        
        output.innerHTML = `
            <h3>Animal Class</h3>
            <p>${genericAnimal.speak()}</p>
            <p>${genericAnimal.info}</p>
            
            <h3>Dog Class (extends Animal)</h3>
            <p>${myDog.speak()}</p>
            <p>Breed: ${myDog.breed}</p>
            <p>${myDog.fetch()}</p>
        `;
        
        // Static method usage
        const staticAnimal = Animal.create('Static', 2);
        console.log('Static creation:', staticAnimal.info);
        
        // Setter usage
        myDog.updateAge = 4;
        console.log('Updated age:', myDog.age);
    </script>
</body>
</html>
```

**Why**: ES6 classes provide clean syntax for OOP in JavaScript, supporting inheritance, encapsulation, static methods, getters/setters, and method overriding.

```html
                list.appendChild(li);
                input.value = '';
            }
        }

        function clearAll() {
            // Method 1: Loop and remove (slower)
            // while (list.firstChild) {
            //     list.removeChild(list.firstChild);
            // }
            
            // Method 2: innerHTML (faster but less safe)
            // list.innerHTML = '';
            
            // Method 3: remove() method (modern)
            while (list.firstChild) {
                list.firstChild.remove();
            }
            
            console.log('All items cleared');
        }
        
        // Remove last item
        function removeLast() {
            const lastItem = list.lastElementChild;
            if (lastItem) {
                list.removeChild(lastItem);
                // Or: lastItem.remove();
            }
        }
```

```html
        const log = document.getElementById('log');
        let eventCount = 0;
        
        function addLog(message) {
            eventCount++;
            const entry = document.createElement('div');
            entry.textContent = `${eventCount}. ${message}`;
            log.insertBefore(entry, log.firstChild);
            
            // Keep only last 20 entries
            while (log.children.length > 20) {
                log.removeChild(log.lastChild);
            }
        }
        
        // Box 1: mouseover/mouseout (bubbles from children)
        const box1 = document.getElementById('box1');
        
        box1.addEventListener('mouseover', (e) => {
            box1.style.background = '#2ecc71';
            addLog('mouseover: Box 1');
        });
        
        box1.addEventListener('mouseout', (e) => {
            box1.style.background = '#3498db';
            addLog('mouseout: Box 1');
        });
        
        // Box 2: mouseenter/mouseleave (does NOT bubble)
        const box2 = document.getElementById('box2');
        
        box2.addEventListener('mouseenter', (e) => {
            box2.style.background = '#9b59b6';
            addLog('mouseenter: Box 2');
        });
        
        box2.addEventListener('mouseleave', (e) => {
            box2.style.background = '#3498db';
            addLog('mouseleave: Box 2');
        });
        
        // Difference: hover over nested element
        // mouseover/out will trigger when entering/leaving nested child
        // mouseenter/leave only triggers once for the entire parent
    </script>
</body>
</html>
```

**Why**: mouseout/mouseover bubble to parent elements while mouseleave/mouseenter don't. Understanding this difference is crucial for hover effects, dropdowns, and tooltips.

```html
        const datetimeInput = document.getElementById('event-datetime');
        const eventNameInput = document.getElementById('event-name');
        const result = document.getElementById('result');
        
        // Set current datetime
        function setCurrentTime() {
            const now = new Date();
            // Format: YYYY-MM-DDTHH:MM
            const year = now.getFullYear();
            const month = String(now.getMonth() + 1).padStart(2, '0');
            const day = String(now.getDate()).padStart(2, '0');
            const hours = String(now.getHours()).padStart(2, '0');
            const minutes = String(now.getMinutes()).padStart(2, '0');
            
            datetimeInput.value = `${year}-${month}-${day}T${hours}:${minutes}`;
        }
        
        // Schedule event
        function scheduleEvent() {
            const datetime = datetimeInput.value;
            const eventName = eventNameInput.value.trim();
            
            if (!datetime) {
                result.innerHTML = '<span style="color:red">Please select a date and time</span>';
                return;
            }
            
            if (!eventName) {
                result.innerHTML = '<span style="color:red">Please enter an event name</span>';
                return;
            }
            
            const eventDate = new Date(datetime);
            const now = new Date();
            
            if (eventDate < now) {
                result.innerHTML = '<span style="color:red">Cannot schedule events in the past</span>';
                return;
            }
            
            // Format display
            const options = { 
                weekday: 'long', 
                year: 'numeric', 
                month: 'long', 
                day: 'numeric',
                hour: '2-digit',
                minute: '2-digit'
            };
            
            result.innerHTML = `
                <strong>Event Scheduled!</strong><br>
                Event: ${eventName}<br>
                Date: ${eventDate.toLocaleDateString('en-US', options)}<br>
                Timezone: ${Intl.DateTimeFormat().resolvedOptions().timeZone}<br>
                Unix Timestamp: ${eventDate.getTime()}
            `;
            
            console.log('Event scheduled:', {
                name: eventName,
                date: eventDate,
                iso: eventDate.toISOString()
            });
        }
        
        // Initialize with current time
        setCurrentTime();
    </script>

</html>
```
