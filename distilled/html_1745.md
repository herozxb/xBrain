## html_table_pagination

```html
<!-- Paginated table for large datasets -->
<table id="dataTable">
  <thead>
    <tr><th>ID</th><th>Name</th><th>Status</th></tr>
  </thead>
  <tbody id="tableBody"></tbody>
</table>

<div id="pagination"></div>

<script>
const data = Array.from({length: 100}, (_, i) => ({
  id: i + 1,
  name: `Item ${i + 1}`,
  status: i % 2 ? 'Active' : 'Inactive'
}));

const rowsPerPage = 10;
let currentPage = 1;

// Render page - TIME: O(n) where n=rowsPerPage, SPACE: O(n)
function renderTable(page) {
  const start = (page - 1) * rowsPerPage;
  const end = start + rowsPerPage;
  const pageData = data.slice(start, end);
  
  const tbody = document.getElementById('tableBody');
  tbody.innerHTML = pageData.map(row => `
    <tr>
      <td>${row.id}</td>
      <td>${row.name}</td>
      <td>${row.status}</td>
    </tr>
  `).join('');
}

// Render pagination controls
function renderPagination() {
  const totalPages = Math.ceil(data.length / rowsPerPage);
  const pagination = document.getElementById('pagination');
  
  pagination.innerHTML = Array.from({length: totalPages}, (_, i) => `
    <button class="page-btn" data-page="${i + 1}">${i + 1}</button>
  `).join('');
  
  pagination.addEventListener('click', (e) => {
    if (e.target.classList.contains('page-btn')) {
      currentPage = parseInt(e.target.dataset.page);
      renderTable(currentPage);
    }
  });
}

renderTable(currentPage);
renderPagination();
</script>
```

**Why**: Breaks large datasets into digestible chunks, improving performance and user experience.

## html_bookmark_links

```html
<!-- Anchor links that navigate to sections within the same page -->
<nav>
  <a href="#section1">Go to Section 1</a>
  <a href="#section2">Go to Section 2</a>
  <a href="#section3">Go to Section 3</a>
</nav>

<section id="section1">
  <h2>Section 1</h2>
  <p>Content for section 1...</p>
</section>

<section id="section2">
  <h2>Section 2</h2>
  <p>Content for section 2...</p>
</section>

<section id="section3">
  <h2>Section 3</h2>
  <p>Content for section 3...</p>
</section>

<script>
// Smooth scroll to bookmark - TIME: O(1) SPACE: O(1)
document.querySelectorAll('a[href^="#"]').forEach(anchor => {
  anchor.addEventListener('click', function(e) {
    e.preventDefault();
    const target = document.querySelector(this.getAttribute('href'));
    target?.scrollIntoView({ behavior: 'smooth' });
  });
});
</script>
```

**Why**: Enables quick navigation within long documents without JavaScript routing.

## js_control_flow

```html
<!-- JavaScript control flow examples -->
<script>
// Conditional statements - TIME: O(1) SPACE: O(1)
const score = 85;
let grade;

if (score >= 90) {
  grade = 'A';
} else if (score >= 80) {
  grade = 'B';
} else if (score >= 70) {
  grade = 'C';
} else {
  grade = 'F';
}

console.log(`Grade: ${grade}`); // "Grade: B"

// Switch statement
const day = 'Monday';
switch (day) {
  case 'Monday':
  case 'Tuesday':
    console.log('Start of week');
    break;
  case 'Wednesday':
    console.log('Midweek');
    break;
  default:
    console.log('End of week');
}

// Ternary operator
const isAdult = age >= 18 ? 'Yes' : 'No';

// Loops - TIME: O(n) SPACE: O(1)
for (let i = 0; i < 5; i++) {
  console.log(i); // 0, 1, 2, 3, 4
}

const items = ['a', 'b', 'c'];
for (const item of items) {
  console.log(item); // 'a', 'b', 'c'
}

for (const index in items) {
  console.log(index); // '0', '1', '2'
}

// While loop
let count = 0;
while (count < 3) {
  console.log(count);
  count++;
}
</script>
```

**Why**: Control flow structures are the building blocks of program logic, enabling conditional execution and iteration.

## event_dblclick

```html
<!-- Double-click event for actions requiring deliberate user intent -->
<div id="dblclickDemo" style="padding: 20px; background: #f0f0f0;">
  Double-click me to edit
</div>

<script>
const demo = document.getElementById('dblclickDemo');

// dblclick handler - TIME: O(1) SPACE: O(1)
demo.addEventListener('dblclick', function(e) {
  const currentText = this.textContent.trim();
  
  // Create input for editing
  const input = document.createElement('input');
  input.type = 'text';
  input.value = currentText;
  input.style.width = '100%';
  
  // Replace text with input
  this.innerHTML = '';
  this.appendChild(input);
  input.focus();
  input.select();
  
  // Save on blur or enter
  const saveEdit = () => {
    this.textContent = input.value || currentText;
  };
  
  input.addEventListener('blur', saveEdit);
  input.addEventListener('keydown', (e) => {
    if (e.key === 'Enter') {
      saveEdit();
    }
  });
});

// Configuration: detect double-click timing
let lastClick = 0;
document.addEventListener('click', (e) => {
  const now = Date.now();
  if (now - lastClick < 300) {
    console.log('Potential double-click detected');
  }
  lastClick = now;
});
</script>
```

**Why**: Double-click distinguishes casual clicks from intentional actions, preventing accidental triggers.

## event_preventdefault

```html
<!-- preventDefault stops the default browser behavior -->
<form id="myForm">
  <input type="text" name="query" required>
  <button type="submit">Search</button>
</form>

<a id="myLink" href="https://example.com">Visit Example</a>

<script>
const form = document.getElementById('myForm');
const link = document.getElementById('myLink');

// Prevent form submission - TIME: O(1) SPACE: O(1)
form.addEventListener('submit', function(e) {
  e.preventDefault(); // Stop form from submitting to server
  
  const formData = new FormData(this);
  const query = formData.get('query');
  
  // Handle submission with JavaScript instead
  console.log('Searching for:', query);
  
  // Could use fetch() to submit via AJAX
  // fetch('/api/search', { method: 'POST', body: formData });
});

// Prevent link navigation
link.addEventListener('click', function(e) {
  e.preventDefault(); // Stop browser from following link
  
  // Custom behavior: open in new tab or show preview
  console.log('Link clicked, but navigation prevented');
  window.open(this.href, '_blank'); // Manual control
});

// Prevent context menu
document.addEventListener('contextmenu', (e) => {
  e.preventDefault();
  console.log('Right-click menu blocked');
});

// Prevent keyboard shortcuts
document.addEventListener('keydown', (e) => {
  if (e.ctrlKey && e.key === 's') {
    e.preventDefault(); // Prevent browser's save dialog
    console.log('Custom save triggered');
  }
});
</script>
```

**Why**: Essential for SPAs and custom interactions where default browser behaviors would interfere with application logic.
