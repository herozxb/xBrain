## 1. html_aria_atomic

```html
<!-- 
  PROBLEM: How to announce dynamic content updates as a complete unit
  TIME: O(1) SPACE: O(1)
-->
<!-- 
  Why: aria-atomic tells screen readers whether to announce the entire 
  content or just the changed portion. Essential for live regions that 
  need to be read as a whole (e.g., "5 items selected" not just "5").
-->

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>ARIA Atomic Live Region</title>
</head>
<body>
  <h2>Selection Counter (Atomic Announcement)</h2>
  
  <!-- 
    aria-live="polite": Announces when user is idle
    aria-atomic="true": Reads entire content, not just changes
    aria-relevant="additions": Announces added content
  -->
  <div 
    id="selection-status"
    role="status"
    aria-live="polite"
    aria-atomic="true"
    aria-relevant="additions"
    style="padding: 10px; background: #f0f0f0;"
  >
    0 items selected
  </div>

  <button onclick="selectItem()">Select Item</button>
  <button onclick="clearSelection()">Clear Selection</button>

  <script>
    let selectedCount = 0;
    const statusEl = document.getElementById('selection-status');

    function selectItem() {
      selectedCount++;
      // Screen reader announces: "3 items selected" (full phrase)
      statusEl.textContent = `${selectedCount} item${selectedCount !== 1 ? 's' : ''} selected`;
    }

    function clearSelection() {
      selectedCount = 0;
      // Screen reader announces: "0 items selected" (full phrase)
      statusEl.textContent = '0 items selected';
    }
  </script>
</body>
</html>
```

## 2. html_aria_relevant

```html
<!-- 
  PROBLEM: How to control what types of changes trigger announcements
  TIME: O(1) SPACE: O(1)
-->
<!-- 
  Why: aria-relevant specifies what changes to announce:
  - "additions": New nodes added
  - "removals": Nodes removed
  - "text": Text content changed
  - "all": Everything (default for some roles)
-->

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>ARIA Relevant Demo</title>
</head>
<body>
  <h2>Task Manager with Smart Announcements</h2>
  
  <!-- 
    aria-relevant="additions text": Announces new tasks AND text changes
    Note: "removals" is often skipped to avoid noise
  -->
  <div 
    id="task-list"
    role="log"
    aria-live="polite"
    aria-atomic="false"
    aria-relevant="additions text"
    style="border: 1px solid #ccc; padding: 10px; min-height: 100px;"
  >
    <div>Task list updates will be announced here...</div>
  </div>

  <input type="text" id="task-input" placeholder="Enter task">
  <button onclick="addTask()">Add Task</button>

  <script>
    const taskList = document.getElementById('task-list');
    const taskInput = document.getElementById('task-input');
    let taskCount = 0;

    function addTask() {
      const taskText = taskInput.value.trim();
      if (!taskText) return;

      taskCount++;
      const task = document.createElement('div');
      task.id = `task-${taskCount}`;
      task.innerHTML = `
        <span class="task-text">${taskText}</span>
        <button onclick="completeTask('${task.id}')">Complete</button>
      `;

      // Only the NEW task is announced (aria-atomic="false")
      taskList.appendChild(task);
      taskInput.value = '';
    }

    function completeTask(taskId) {
      const taskEl = document.getElementById(taskId);
      const textEl = taskEl.querySelector('.task-text');
      
      // Text change triggers announcement (aria-relevant="text")
      textEl.style.textDecoration = 'line-through';
      textEl.textContent += ' ✓';
    }
  </script>
</body>
</html>
```

## 3. html_aria_haspopup

```html
<!-- 
  PROBLEM: How to indicate an element triggers a popup menu/dialog
  TIME: O(1) SPACE: O(1)
-->
<!-- 
  Why: aria-haspopup tells assistive tech that activating this element
  opens a menu, dialog, listbox, tree, or grid. Improves keyboard 
  navigation expectations.
-->

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>ARIA HasPopup Menu</title>
  <style>
    .menu { display: none; position: absolute; background: white; 
            border: 1px solid #ccc; list-style: none; padding: 0; margin: 0; }
    .menu.visible { display: block; }
    .menu li { padding: 8px 16px; cursor: pointer; }
    .menu li:hover { background: #f0f0f0; }
  </style>
</head>
<body>
  <h2>Dropdown Menu with ARIA</h2>
  
  <div style="position: relative;">
    <button 
      id="menu-trigger"
      aria-haspopup="menu"
      aria-expanded="false"
      aria-controls="dropdown-menu"
      onclick="toggleMenu()"
    >
      Options ▼
    </button>

    <ul 
      id="dropdown-menu"
      role="menu"
      aria-labelledby="menu-trigger"
      class="menu"
    >
      <li role="menuitem" onclick="selectOption('Edit')">Edit</li>
      <li role="menuitem" onclick="selectOption('Duplicate')">Duplicate</li>
      <li role="menuitem" onclick="selectOption('Delete')">Delete</li>
    </ul>
  </div>

  <p id="selection-result"></p>

  <script>
    const trigger = document.getElementById('menu-trigger');
    const menu = document.getElementById('dropdown-menu');

    function toggleMenu() {
      const isOpen = menu.classList.contains('visible');
      
      if (isOpen) {
        menu.classList.remove('visible');
        trigger.setAttribute('aria-expanded', 'false');
      } else {
        menu.classList.add('visible');
        trigger.setAttribute('aria-expanded', 'true');
        // Focus first menu item for keyboard users
        menu.querySelector('[role="menuitem"]').focus();
      }
    }

    function selectOption(option) {
      document.getElementById('selection-result').textContent = 
        `Selected: ${option}`;
      toggleMenu();
    }

    // Close menu when clicking outside
    document.addEventListener('click', (e) => {
      if (!trigger.contains(e.target) && !menu.contains(e.target)) {
        menu.classList.remove('visible');
        trigger.setAttribute('aria-expanded', 'false');
      }
    });

    // Keyboard navigation
    trigger.addEventListener('keydown', (e) => {
      if (e.key === 'ArrowDown' && !menu.classList.contains('visible')) {
        toggleMenu();
      }
    });
  </script>
</body>
</html>
```

```html
<!-- 
  PROBLEM: How to indicate the next page in a paginated series
  TIME: O(1) SPACE: O(1)
-->
<!-- 
  Why: rel="next" helps search engines understand pagination and 
  consolidate link equity. Improves SEO for multi-page content.
  Also used by browsers for prefetching.
-->

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Blog - Page 1 of 10</title>
  
  <!-- Points to the next page in the series -->
  <link rel="next" href="/blog?page=2">
  
  <!-- Not the first page, so no rel="first" needed here -->
  <!-- On page 2+, you'd also have: <link rel="prev" href="/blog?page=1"> -->
  
  <!-- Optional: Prefetch next page for faster navigation -->
  <link rel="prefetch" href="/blog?page=2">
  
  <style>
    .pagination { display: flex; gap: 8px; margin-top: 20px; }
    .pagination a, .pagination span {
      padding: 8px 12px; border: 1px solid #ccc; text-decoration: none;
    }
    .pagination .current { background: #007bff; color: white; border-color: #007bff; }
  </style>
</head>
<body>
  <h1>Blog Posts - Page 1</h1>
  
  <main>
    <article><h2>Post 1</h2><p>Excerpt...</p></article>
    <article><h2>Post 2</h2><p>Excerpt...</p></article>
    <article><h2>Post 3</h2><p>Excerpt...</p></article>
  </main>
  
  <!-- Visual pagination with semantic links -->
  <nav class="pagination" aria-label="Blog pagination">
    <span class="current" aria-current="page">1</span>
    <a href="/blog?page=2" rel="next">2</a>
    <a href="/blog?page=3">3</a>
    <a href="/blog?page=4">4</a>
    <a href="/blog?page=2" rel="next" aria-label="Next page">Next →</a>
    <a href="/blog?page=10">Last</a>
  </nav>
  
  <script>
    // JavaScript can use rel="next" link for dynamic loading
    function loadNextPage() {
      const nextLink = document.querySelector('link[rel="next"]');
      if (nextLink) {
        fetch(nextLink.href)
          .then(res => res.text())
          .then(html => {
            // Append new posts to current page
            console.log('Prefetched next page:', nextLink.href);
          });
      }
    }
    
    // Keyboard navigation: Arrow Right = Next Page
    document.addEventListener('keydown', (e) => {
      if (e.key === 'ArrowRight') {
        const nextLink = document.querySelector('a[rel="next"]');
        if (nextLink) window.location = nextLink.href;
      }
    });
  </script>
</body>
</html>
```

```html
    const suggestionsEl = document.getElementById('suggestions');
    
    // Simulated search endpoint (would match OpenSearch URL template)
    async function searchDocs(query) {
      // In production, this would be: /search?q=${query}&format=json
      return [
        { title: 'Getting Started', url: '/docs/getting-started' },
        { title: 'API Reference', url: '/docs/api' },
        { title: 'Configuration', url: '/docs/config' }
      ].filter(item => 
        item.title.toLowerCase().includes(query.toLowerCase())
      );

    searchInput.addEventListener('input', (e) => {

          <div class="search-result">
            <a href="${r.url}">${r.title}</a>
          </div>
        `).join('');
      }, 300);
    });
  </script>
```

## 8. html_anchor_hreflang

```html
<!-- 
  PROBLEM: How to specify the language of a linked document
  TIME: O(1) SPACE: O(1)
-->
<!-- 
  Why: hreflang indicates the language of the target URL. Used by:
  - Search engines to show correct language version
  - Screen readers to announce language change
  - Browsers to understand content language
-->

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Multi-Language Documentation</title>
  
  <!-- 
    hreflang in <head>: Tells search engines about alternate language versions
    x-default: The default/fallback version (usually English or neutral)
  -->
  <link rel="alternate" hreflang="en" href="https://example.com/en/docs">
  <link rel="alternate" hreflang="es" href="https://example.com/es/docs">
  <link rel="alternate" hreflang="fr" href="https://example.com/fr/docs">
  <link rel="alternate" hreflang="zh-CN" href="https://example.com/zh/docs">
  <link rel="alternate" hreflang="x-default" href="https://example.com/docs">
  
  <style>
    .language-switcher { 
      display: flex; gap: 10px; padding: 10px; 
      background: #f5f5f5; border-radius: 4px;
    }
    .language-switcher a { 
      padding: 5px 10px; text-decoration: none; 
      border: 1px solid #ccc; border-radius: 3px;
    }
    .language-switcher a:hover { background: #e0e0e0; }
    .language-switcher .current { 
      background: #007bff; color: white; border-color: #007bff; 
    }
  </style>
</head>
<body>
  <h1>Documentation</h1>
  
  <!-- Language switcher with hreflang on links -->
  <nav class="language-switcher" aria-label="Language selection">
    <span>Language:</span>
    <a href="/en/docs" hreflang="en" lang="en">English</a>
    <a href="/es/docs" hreflang="es" lang="es">Español</a>
    <a href="/fr/docs" hreflang="fr" lang="fr">Français</a>
    <a href="/zh/docs" hreflang="zh-CN" lang="zh-CN">中文</a>
  </nav>
  
  <article>
    <h2>Welcome to the Documentation</h2>
    <p>This page is available in multiple languages.</p>
    
    <!-- 
      Cross-language reference with hreflang.
      Screen reader will announce: "French link: En savoir plus"
    -->
    <p>
      For more details, see the 
      <a href="/fr/docs/advanced" hreflang="fr" lang="fr">
        guide avancé (French)
      </a>.
    </p>
  </article>
  
  <script>
    // Detect and suggest user's preferred language
    function suggestLanguage() {
      const userLang = navigator.language || navigator.userLanguage;
      const currentPath = window.location.pathname;
      
      // Map browser language to site language
      const langMap = {
        'en': '/en/docs',
        'es': '/es/docs',
        'fr': '/fr/docs',
        'zh': '/zh/docs',
        'zh-CN': '/zh/docs',
        'zh-TW': '/zh/docs'
      };
      
      const preferredLang = userLang.split('-')[0];
      const suggestedPath = langMap[preferredLang];
      
      // If user is on default and prefers a different language
      if (suggestedPath && currentPath === '/docs') {
        const notice = document.createElement('div');
        notice.style.cssText = 'padding: 15px; background: #fff3cd; margin: 10px 0;';
        notice.innerHTML = `
          This page is also available in ${preferredLang.toUpperCase()}. 
          <a href="${suggestedPath}" hreflang="${preferredLang}">Switch language</a>
        `;
        document.body.insertBefore(notice, document.body.firstChild);
      }
    }
    
    // Only run once, store preference
    if (!localStorage.getItem('langSelected')) {
      suggestLanguage();
    }
    
    // Store preference when user clicks a language link
    document.querySelectorAll('a[hreflang]').forEach(link => {
      link.addEventListener('click', () => {
        localStorage.setItem('langSelected', link.getAttribute('hreflang'));
      });
    });
  </script>
</body>
</html>
```

```html
    function checkReferrer() {
      const referrer = document.referrer;
      if (!referrer) {

      try {

        const currentUrl = new URL(window.location.href);
```
