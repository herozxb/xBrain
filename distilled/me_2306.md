## 1. B-Tree

```python
# PROBLEM: Implement a B-tree data structure for efficient disk-based storage
# TIME: O(log n) for search/insert/delete SPACE: O(n)
class BTreeNode:
    """Node in a B-tree containing keys and children pointers."""
    
    def __init__(self, leaf: bool = False):
        self.keys: list = []          # List of keys in the node
        self.children: list = []       # List of child pointers
        self.leaf: bool = leaf         # True if node is a leaf

class BTree:
    """B-tree implementation with configurable minimum degree."""
    
    def __init__(self, min_degree: int = 3):
        self.root = BTreeNode(leaf=True)
        self.t = min_degree  # Minimum degree (defines range of keys)
    
    def search(self, node: BTreeNode, key: int) -> tuple:
        """Search for a key in the B-tree.
        
        Returns tuple of (node, index) if found, else (None, -1).
        """
        i = 0
        # Find first key >= search key
        while i < len(node.keys) and key > node.keys[i]:
            i += 1
        
        # Key found at this node
        if i < len(node.keys) and key == node.keys[i]:
            return (node, i)
        
        # Key not found and this is a leaf
        if node.leaf:
            return (None, -1)
        
        # Recurse to appropriate child
        return self.search(node.children[i], key)
    
    def insert(self, key: int) -> None:
        """Insert a key into the B-tree."""
        root = self.root
        
        # If root is full, tree grows in height
        if len(root.keys) == 2 * self.t - 1:
            new_root = BTreeNode()
            new_root.children.append(self.root)
            self._split_child(new_root, 0)
            self.root = new_root
            self._insert_non_full(new_root, key)
        else:
            self._insert_non_full(root, key)
    
    def _split_child(self, parent: BTreeNode, index: int) -> None:
        """Split the child at given index."""
        t = self.t
        full_child = parent.children[index]
        new_child = BTreeNode(leaf=full_child.leaf)
        
        # Move median key up to parent
        parent.keys.insert(index, full_child.keys[t - 1])
        parent.children.insert(index + 1, new_child)
        
        # Split keys between old and new child
        new_child.keys = full_child.keys[t:]
        full_child.keys = full_child.keys[:t - 1]
        
        # Split children if not leaf
        if not full_child.leaf:
            new_child.children = full_child.children[t:]
            full_child.children = full_child.children[:t]
    
    def _insert_non_full(self, node: BTreeNode, key: int) -> None:
        """Insert key into a node that is not full."""
        i = len(node.keys) - 1
        
        if node.leaf:
            # Insert key in correct position
            node.keys.append(0)
            while i >= 0 and key < node.keys[i]:
                node.keys[i + 1] = node.keys[i]
                i -= 1
            node.keys[i + 1] = key
        else:
            # Find child to insert into
            while i >= 0 and key < node.keys[i]:
                i -= 1
            i += 1
            
            # Split child if full
            if len(node.children[i].keys) == 2 * self.t - 1:
                self._split_child(node, i)
                if key > node.keys[i]:
                    i += 1
            self._insert_non_full(node.children[i], key)
```
**Why**: B-trees maintain sorted data with efficient insertion/deletion, widely used in databases and filesystems for minimizing disk I/O operations.

## 2. CQRS Pattern

```python
# PROBLEM: Separate read and write operations for scalable systems
# TIME: O(1) for command, O(n) for query SPACE: O(n)
from dataclasses import dataclass
from typing import Dict, List, Any, Callable
from datetime import datetime
import asyncio

@dataclass
class Command:
    """Base command class for write operations."""
    timestamp: datetime = None
    
    def __post_init__(self):
        self.timestamp = self.timestamp or datetime.now()

@dataclass
class Query:
    """Base query class for read operations."""
    pass

class CommandHandler:
    """Handles command dispatch and execution."""
    
    def __init__(self):
        self.handlers: Dict[type, Callable] = {}
        self.event_store: List[Dict] = []
    
    def register(self, command_type: type, handler: Callable):
        """Register a handler for a specific command type."""
        self.handlers[command_type] = handler
    
    async def dispatch(self, command: Command) -> Any:
        """Dispatch command to appropriate handler."""
        handler = self.handlers.get(type(command))
        if not handler:
            raise ValueError(f"No handler for {type(command).__name__}")
        
        result = await handler(command)
        self.event_store.append({
            'command': type(command).__name__,
            'timestamp': command.timestamp,
            'result': result
        })
        return result

class QueryHandler:
    """Handles query dispatch with separate read model."""
    
    def __init__(self):
        self.handlers: Dict[type, Callable] = {}
        self.read_model: Dict[str, Any] = {}
    
    def register(self, query_type: type, handler: Callable):
        """Register a handler for a specific query type."""
        self.handlers[query_type] = handler
    
    async def ask(self, query: Query) -> Any:
        """Execute query against read model."""
        handler = self.handlers.get(type(query))
        if not handler:
            raise ValueError(f"No handler for {type(query).__name__}")
        return await handler(query, self.read_model)

class CQRSMediator:
    """Mediator coordinating command and query sides."""
    
    def __init__(self):
        self.command_bus = CommandHandler()
        self.query_bus = QueryHandler()
    
    def send(self, command: Command) -> Any:
        """Send command to write side."""
        return asyncio.run(self.command_bus.dispatch(command))
    
    def ask(self, query: Query) -> Any:
        """Query the read side."""
        return asyncio.run(self.query_bus.ask(query))

# Example: User creation command and query
@dataclass
class CreateUserCommand(Command):
    user_id: str
    name: str
    email: str

@dataclass
class GetUserQuery(Query):
    user_id: str

class UserService:
    """Service demonstrating CQRS separation."""
    
    def __init__(self, mediator: CQRSMediator):
        self.mediator = mediator
        self._setup_handlers()
    
    def _setup_handlers(self):
        async def handle_create(cmd: CreateUserCommand):
            # Write to write model (database)
            print(f"Creating user: {cmd.user_id}")
            return {'id': cmd.user_id, 'name': cmd.name}
        
        async def handle_get(query: GetUserQuery, read_model):
            # Read from read model (optimized view)
            return read_model.get(query.user_id, {'error': 'Not found'})
        
        self.mediator.command_bus.register(CreateUserCommand, handle_create)
        self.mediator.query_bus.register(GetUserQuery, handle_get)
```
**Why**: CQRS optimizes read/write performance independently, enables different scaling strategies, and simplifies complex business logic by separating concerns.

## 3. Saga Pattern (Distributed)

```python
# PROBLEM: Manage distributed transactions across multiple services
# TIME: O(n) for n steps SPACE: O(n) for compensation log
from dataclasses import dataclass
from typing import List, Callable, Any, Optional
from enum import Enum
import asyncio

class SagaStep:
    """Represents a single step in a saga with compensation."""
    
    def __init__(self, 
                 action: Callable,
                 compensation: Callable,
                 name: str = ""):
        self.action = action
        self.compensation = compensation
        self.name = name or action.__name__

class SagaState(Enum):
    """Possible states of a saga execution."""
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    COMPENSATING = "compensating"
    COMPENSATED = "compensated"
    FAILED = "failed"

@dataclass
class SagaResult:
    """Result of saga execution."""
    state: SagaState
    completed_steps: List[str]
    failed_step: Optional[str] = None
    error: Optional[str] = None

class DistributedSaga:
    """Orchestrates distributed transactions with compensation."""
    
    def __init__(self, saga_id: str):
        self.saga_id = saga_id
        self.steps: List[SagaStep] = []
        self.completed: List[SagaStep] = []
        self.state = SagaState.PENDING
    
    def add_step(self, step: SagaStep) -> 'DistributedSaga':
        """Add a step to the saga chain."""
        self.steps.append(step)
        return self
    
    async def execute(self) -> SagaResult:
        """Execute saga with automatic compensation on failure."""
        self.state = SagaState.RUNNING
        
        for step in self.steps:
            try:
                await step.action()
                self.completed.append(step)
                print(f"✓ Completed: {step.name}")
            except Exception as e:
                print(f"✗ Failed: {step.name} - {e}")
                return await self._compensate(step.name, str(e))
        
        self.state = SagaState.COMPLETED
        return SagaResult(
            state=self.state,
            completed_steps=[s.name for s in self.completed]
        )
    
    async def _compensate(self, failed_step: str, error: str) -> SagaResult:
        """Execute compensating transactions in reverse order."""
        self.state = SagaState.COMPENSATING
        print(f"\n⟲ Starting compensation from: {failed_step}")
        
        compensation_errors = []
        
        # Reverse through completed steps
        for step in reversed(self.completed):
            try:
                await step.compensation()
                print(f"  ↩ Compensated: {step.name}")
            except Exception as e:
                compensation_errors.append(f"{step.name}: {e}")
                print(f"  ⚠ Compensation failed: {step.name} - {e}")
        
        self.state = SagaState.COMPENSATED if not compensation_errors else SagaState.FAILED
        
        return SagaResult(
            state=self.state,
            completed_steps=[s.name for s in self.completed],
            failed_step=failed_step,
            error=error
        )

# Example: E-commerce order saga
async def create_order():
    await asyncio.sleep(0.1)  # Simulate API call
    print("  → Order created in database")

async def cancel_order():
    await asyncio.sleep(0.1)
    print("  → Order cancelled")

async def reserve_inventory():
    await asyncio.sleep(0.1)
    print("  → Inventory reserved")
    # Simulate failure
    raise Exception("Out of stock!")

async def release_inventory():
    await asyncio.sleep(0.1)
    print("  → Inventory released")

async def charge_payment():
    await asyncio.sleep(0.1)
    print("  → Payment charged")

async def refund_payment():
    await asyncio.sleep(0.1)
    print("  → Payment refunded")
```
**Why**: Sagas ensure data consistency across distributed systems without two-phase commit, providing fault tolerance through compensating transactions.

## 4. Outbox Pattern

```python
# PROBLEM: Reliably publish events in distributed systems
# TIME: O(1) for write SPACE: O(n) for outbox storage
from dataclasses import dataclass, asdict
from typing import List, Dict, Any, Optional
from datetime import datetime
from enum import Enum
import json
import asyncio

class EventStatus(Enum):
    PENDING = "pending"
    PUBLISHED = "published"
    FAILED = "failed"

@dataclass
class OutboxMessage:
    """Message stored in outbox before publishing."""
    id: str
    aggregate_type: str
    aggregate_id: str
    event_type: str
    payload: Dict[str, Any]
    status: EventStatus
    created_at: datetime
    published_at: Optional[datetime] = None
    retry_count: int = 0
    max_retries: int = 3

class Outbox:
    """Stores events to be reliably published."""
    
    def __init__(self):
        self.messages: List[OutboxMessage] = []
        self.message_handlers: Dict[str, Callable] = {}
    
    def register_handler(self, event_type: str, handler: Callable):
        """Register handler for specific event type."""
        self.message_handlers[event_type] = handler
    
    async def save(self, message: OutboxMessage) -> None:
        """Atomically save message with business data."""
        # In real implementation, this would be in same transaction
        self.messages.append(message)
        print(f"📝 Saved to outbox: {message.event_type} [{message.id}]")
    
    async def mark_published(self, message_id: str) -> None:
        """Mark message as successfully published."""
        for msg in self.messages:
            if msg.id == message_id:
                msg.status = EventStatus.PUBLISHED
                msg.published_at = datetime.now()
                print(f"✓ Marked published: {message_id}")
                break

class OutboxPublisher:
    """Background process that publishes outbox messages."""
    
    def __init__(self, outbox: Outbox):
        self.outbox = outbox
        self.running = False
    
    async def start(self, interval_seconds: int = 5):
        """Start polling and publishing messages."""
        self.running = True
        while self.running:
            await self._process_pending()
            await asyncio.sleep(interval_seconds)
    
    def stop(self):
        """Stop the publisher."""
        self.running = False
    
    async def _process_pending(self):
        """Process all pending messages."""
        pending = [m for m in self.outbox.messages 
                   if m.status == EventStatus.PENDING]
        
        for message in pending:
            await self._publish_message(message)
    
    async def _publish_message(self, message: OutboxMessage):
        """Attempt to publish a single message."""
        handler = self.outbox.message_handlers.get(message.event_type)
        
        if not handler:
            print(f"⚠ No handler for: {message.event_type}")
            message.status = EventStatus.FAILED
            return
        
        try:
            await handler(message.payload)
            await self.outbox.mark_published(message.id)
        except Exception as e:
            message.retry_count += 1
            if message.retry_count >= message.max_retries:
                message.status = EventStatus.FAILED
                print(f"✗ Failed after {message.retry_count} retries: {message.id}")
            else:
                print(f"↻ Retry {message.retry_count}/{message.max_retries}: {message.id}")

class OrderService:
    """Example service using outbox pattern."""
    
    def __init__(self, outbox: Outbox):
        self.outbox = outbox
        self.orders: Dict[str, Dict] = {}
    
    async def create_order(self, order_id: str, items: List[Dict]) -> None:
        """Create order and event atomically."""
        # 1. Save business entity
        self.orders[order_id] = {
            'id': order_id,
            'items': items,
            'status': 'created',
            'created_at': datetime.now().isoformat()
        }
        
        # 2. Create outbox message in same transaction
        message = OutboxMessage(
            id=f"msg-{order_id}",
            aggregate_type="Order",
            aggregate_id=order_id,
            event_type="OrderCreated",
            payload={'order_id': order_id, 'items': items},
            status=EventStatus.PENDING,
            created_at=datetime.now()
        )
        
        # Both saved atomically (simulated)
        await self.outbox.save(message)

# Message broker simulation
async def publish_to_kafka(payload: Dict):
    """Simulate publishing to message broker."""
    await asyncio.sleep(0.1)
    print(f"📤 Published to Kafka: {payload['order_id']}")
```
**Why**: The outbox pattern ensures reliable event publishing with at-least-once delivery guarantees, solving the dual-write problem in distributed systems.

## 5. Distributed Tracing

```python
# PROBLEM: Track requests across distributed microservices
# TIME: O(1) per span SPACE: O(n) for trace storage
from dataclasses import dataclass, field
from typing import Dict, List, Optional
from datetime import datetime
import uuid
import time

@dataclass
class SpanContext:
    """Context identifying a trace span."""
    trace_id: str
    span_id: str
    parent_span_id: Optional[str] = None
    
    @classmethod
    def new_root(cls) -> 'SpanContext':
        """Create new root span context."""
        return cls(
            trace_id=uuid.uuid4().hex[:16],
            span_id=uuid.uuid4().hex[:8]
        )
    
    def new_child(self) -> 'SpanContext':
        """Create child span context."""
        return SpanContext(
            trace_id=self.trace_id,
            span_id=uuid.uuid4().hex[:8],
            parent_span_id=self.span_id
        )

@dataclass
class Span:
    """Represents a unit of work in distributed tracing."""
    context: SpanContext
    operation_name: str
    start_time: float = field(default_factory=time.time)
    end_time: Optional[float] = None
    tags: Dict[str, str] = field(default_factory=dict)
    logs: List[Dict] = field(default_factory=list)
    
    def finish(self):
        """Mark span as completed."""
        self.end_time = time.time()
    
    def duration_ms(self) -> float:
        """Calculate span duration in milliseconds."""
        if self.end_time:
            return (self.end_time - self.start_time) * 1000
        return 0
    
    def log(self, message: str, **kwargs):
        """Add log entry to span."""
        self.logs.append({
            'timestamp': datetime.now().isoformat(),
            'message': message,
            **kwargs
        })
    
    def set_tag(self, key: str, value: str):
        """Add tag to span."""
        self.tags[key] = value

class Tracer:
    """Distributed tracer managing spans and contexts."""
    
    def __init__(self, service_name: str):
        self.service_name = service_name
        self.spans: List[Span] = []
        self._active_span: Optional[Span] = None
    
    def start_span(self, 
                   operation_name: str,
                   parent: Optional[Span] = None) -> Span:
        """Start a new span, optionally as child of another."""
        if parent:
            context = parent.context.new_child()
        elif self._active_span:
            context = self._active_span.context.new_child()
        else:
            context = SpanContext.new_root()
        
        span = Span(
            context=context,
            operation_name=operation_name
        )
        span.set_tag('service', self.service_name)
        
        self.spans.append(span)
        self._active_span = span
        
        print(f"🔍 Started span: {operation_name} [{context.span_id}]")
        return span
    
    def finish_span(self, span: Span):
        """Complete a span and log duration."""
        span.finish()
        duration = span.duration_ms()
        
        print(f"✓ Finished span: {span.operation_name} "
              f"(trace={span.context.trace_id}, "
              f"duration={duration:.2f}ms)")
    
    def get_trace(self, trace_id: str) -> List[Span]:
        """Get all spans for a trace."""
        return [s for s in self.spans 
                if s.context.trace_id == trace_id]
    
    def export_trace(self, trace_id: str) -> Dict:
        """Export trace data for analysis."""
        spans = self.get_trace(trace_id)
        
        return {
            'trace_id': trace_id,
            'service': self.service_name,
            'spans': [
                {
                    'span_id': s.context.span_id,
                    'parent_id': s.context.parent_span_id,
                    'operation': s.operation_name,
                    'duration_ms': s.duration_ms(),
                    'tags': s.tags,
                    'logs': s.logs
                }
                for s in spans
            ],
            'total_spans': len(spans),
            'total_duration_ms': sum(s.duration_ms() for s in spans)
        }

class TracingMiddleware:
    """Middleware for automatic request tracing."""
    
    def __init__(self, tracer: Tracer):
        self.tracer = tracer
    
    def before_request(self, request: Dict) -> Span:
        """Start span before processing request."""
        span = self.tracer.start_span(
            operation_name=f"{request.get('method', 'GET')} "
                          f"{request.get('path', '/')}"
        )
        span.set_tag('http.method', request.get('method', 'GET'))
        span.set_tag('http.url', request.get('url', ''))
        return span
    
    def after_request(self, span: Span, response: Dict):
        """Finish span after response."""
        span.set_tag('http.status_code', str(response.get('status', 200)))
        self.tracer.finish_span(span)
```
**Why**: Distributed tracing provides end-to-end visibility across microservices, enabling performance analysis, debugging, and understanding of request flows.

## 6. API Composition Pattern

```python
# PROBLEM: Aggregate data from multiple microservices efficiently
# TIME: O(max(service_times)) parallel SPACE: O(n) for response
from dataclasses import dataclass
from typing import Dict, List, Any, Callable
import asyncio
from datetime import datetime

@dataclass
class ServiceResponse:
    """Response from a single service call."""
    service_name: str
    data: Any
    duration_ms: float
    error: Optional[str] = None

class ServiceClient:
    """Client for calling a microservice."""
    
    def __init__(self, name: str, endpoint: str):
        self.name = name
        self.endpoint = endpoint
    
    async def fetch(self, resource: str, params: Dict = None) -> Any:
        """Fetch data from service (simulated)."""
        start = datetime.now()
        await asyncio.sleep(0.1)  # Simulate network delay
        
        # Simulated responses
        data = {f"{self.name}_data": params or {}}
        
        duration = (datetime.now() - start).total_seconds() * 1000
        return ServiceResponse(
            service_name=self.name,
            data=data,
            duration_ms=duration
        )

class APIComposer:
    """Composes responses from multiple services."""
    
    def __init__(self):
        self.services: Dict[str, ServiceClient] = {}
    
    def register(self, service: ServiceClient):
        """Register a service client."""
        self.services[service.name] = service
    
    async def fetch_parallel(self, 
                            requests: List[Dict]) -> Dict[str, ServiceResponse]:
        """Fetch from multiple services in parallel."""
        tasks = []
        
        for req in requests:
            service = self.services.get(req['service'])
            if service:
                tasks.append(
                    service.fetch(req['resource'], req.get('params'))
                )
        
        responses = await asyncio.gather(*tasks, return_exceptions=True)
        
        result = {}
        for i, resp in enumerate(responses):
            req = requests[i]
            if isinstance(resp, Exception):
                result[req['service']] = ServiceResponse(
                    service_name=req['service'],
                    data=None,
                    duration_ms=0,
                    error=str(resp)
                )
            else:
                result[req['service']] = resp
        
        return result
    
    async def fetch_with_fallback(self,
                                  primary: str,
                                  fallbacks: List[str],
                                  resource: str) -> ServiceResponse:
        """Fetch with fallback services on failure."""
        all_services = [primary] + fallbacks
        
        for service_name in all_services:
            service = self.services.get(service_name)
            if not service:
                continue
            
            try:
                response = await service.fetch(resource)
                if not response.error:
                    return response
            except Exception as e:
                print(f"⚠ {service_name} failed: {e}")
                continue
        
        return ServiceResponse(
            service_name='none',
            data=None,
            duration_ms=0,
            error='All services failed'
        )

class ProductAggregator:
    """Example aggregator combining product data from services."""
    
    def __init__(self, composer: APIComposer):
        self.composer = composer
    
    async def get_product_page(self, product_id: str) -> Dict:
        """Aggregate product page data from multiple services."""
        # Define parallel requests
        requests = [
            {'service': 'product', 'resource': f'/products/{product_id}'},
            {'service': 'inventory', 'resource': f'/inventory/{product_id}'},
            {'service': 'reviews', 'resource': f'/reviews/{product_id}'},
            {'service': 'pricing', 'resource': f'/pricing/{product_id}'},
        ]
        
        # Fetch all in parallel
        responses = await self.composer.fetch_parallel(requests)
        
        # Combine into unified response
        result = {
            'product_id': product_id,
            'timestamp': datetime.now().isoformat(),
            'details': responses.get('product', {}).data,
            'inventory': responses.get('inventory', {}).data,
            'reviews': responses.get('reviews', {}).data,
            'pricing': responses.get('pricing', {}).data,
            'performance': {
                name: resp.duration_ms 
                for name, resp in responses.items()
            }
        }
        
        return result
    
    async def get_product_with_fallback(
        self, product_id: str) -> ServiceResponse:
        """Get product data with caching fallback."""
        return await self.composer.fetch_with_fallback(
            primary='product',
            fallbacks=['product-cache', 'product-readonly'],
            resource=f'/products/{product_id}'
        )
```
**Why**: API composition reduces client complexity, enables parallel fetching for better performance, and provides a unified interface over distributed services.

## 7. MVCC Internals (Multi-Version Concurrency Control)

```python
# PROBLEM: Implement concurrent read/write access without blocking
# TIME: O(1) for reads SPACE: O(n*m) for n rows with m versions
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Any
from datetime import datetime
import threading

@dataclass
class Version:
    """Single version of a data row."""
    data: Any
    created_at: int          # Transaction ID that created this version
    expired_at: Optional[int] = None  # Transaction ID that expired this
    xid: int = 0             # Version's transaction ID

class MVCCRow:
    """Row storing multiple versions for MVCC."""
    
    def __init__(self):
        self.versions: List[Version] = []
        self.lock = threading.Lock()
    
    def read(self, tx_id: int) -> Optional[Any]:
        """Read most recent version visible to transaction."""
        with self.lock:
            # Find newest version visible to this transaction
            for version in reversed(self.versions):
                if self._is_visible(version, tx_id):
                    return version.data
        return None
    
    def _is_visible(self, version: Version, tx_id: int) -> bool:
        """Check if version is visible to transaction."""
        # Version created by committed transaction before our snapshot
        if version.created_at <= tx_id:
            # Not expired, or expired by transaction after our snapshot
            if version.expired_at is None or version.expired_at > tx_id:
                return True
        return False
    
    def write(self, tx_id: int, data: Any) -> bool:
        """Create new version of row."""
        with self.lock:
            # Expire current latest visible version
            for version in reversed(self.versions):
                if version.expired_at is None:
                    version.expired_at = tx_id
                    break
            
            # Create new version
            new_version = Version(
                data=data,
                created_at=tx_id
            )
            self.versions.append(new_version)
            return True

class MVCCStore:
    """MVCC-based key-value store."""
    
    def __init__(self):
        self.rows: Dict[str, MVCCRow] = {}
        self.lock = threading.Lock()
        self._next_tx_id = 1
    
    def begin_transaction(self) -> int:
        """Start new transaction, returning snapshot ID."""
        with self.lock:
            tx_id = self._next_tx_id
            self._next_tx_id += 1
            return tx_id
    
    def get(self, key: str, tx_id: int) -> Optional[Any]:
        """Read value visible to transaction."""
        row = self.rows.get(key)
        if row:
            return row.read(tx_id)
        return None
    
    def set(self, key: str, value: Any, tx_id: int):
        """Write new version of key."""
        with self.lock:
            if key not in self.rows:
                self.rows[key] = MVCCRow()
            self.rows[key].write(tx_id, value)
    
    def vacuum(self, oldest_active_tx: int):
        """Remove versions no longer visible to any transaction."""
        with self.lock:
            for key, row in self.rows.items():
                with row.lock:
                    # Keep only versions visible to oldest active transaction
                    row.versions = [
                        v for v in row.versions
                        if v.expired_at is None or 
                           v.expired_at > oldest_active_tx
                    ]

class MVCCExample:
    """Demonstrates MVCC concurrent access."""
    
    def __init__(self):
        self.store = MVCCStore()
    
    def demonstrate(self):
        """Show MVCC behavior with concurrent transactions."""
        # Transaction 1: Initial write
        tx1 = self.store.begin_transaction()
        self.store.set('balance', 100, tx1)
        print(f"TX{tx1}: Set balance = 100")
        
        # Transaction 2: Read (sees 100)
        tx2 = self.store.begin_transaction()
        val = self.store.get('balance', tx2)
        print(f"TX{tx2}: Read balance = {val}")
        
        # Transaction 3: Update (creates new version)
        tx3 = self.store.begin_transaction()
        self.store.set('balance', 200, tx3)
        print(f"TX{tx3}: Set balance = 200")
        
        # Transaction 2 still sees old value (snapshot isolation)
        val = self.store.get('balance', tx2)
        print(f"TX{tx2}: Still sees balance = {val}")
        
        # New transaction sees latest committed
        tx4 = self.store.begin_transaction()
        val = self.store.get('balance', tx4)
        print(f"TX{tx4}: Sees balance = {val}")
```
**Why**: MVCC provides snapshot isolation without read locks, allowing concurrent readers and writers without blocking, fundamental to PostgreSQL and other databases.

## 8. Bloom Filter Index

```python
# PROBLEM: Probabilistic set membership with space efficiency
# TIME: O(k) for k hash functions SPACE: O(m) bits
import mmh3  # MurmurHash3
from typing import List, Tuple
import math

class BloomFilter:
    """Space-efficient probabilistic data structure for set membership."""
    
    def __init__(self, 
                 expected_items: int,
                 false_positive_rate: float = 0.01):
        """Initialize bloom filter with optimal parameters."""
        # Calculate optimal size and number of hash functions
        self.m = self._calculate_size(expected_items, false_positive_rate)
        self.k = self._calculate_hashes(self.m, expected_items)
        
        # Initialize bit array
        self.bits = [False] * self.m
        self.items_count = 0
    
    def _calculate_size(self, n: int, p: float) -> int:
        """Calculate optimal bit array size.
        
        m = -n * ln(p) / (ln(2)^2)
        """
        return int(-n * math.log(p) / (math.log(2) ** 2))
    
    def _calculate_hashes(self, m: int, n: int) -> int:
        """Calculate optimal number of hash functions.
        
        k = (m/n) * ln(2)
        """
        return int((m / n) * math.log(2))
    
    def _hashes(self, item: str) -> List[int]:
        """Generate k hash values for an item using double hashing."""
        # Use MurmurHash3 for better distribution
        h1 = mmh3.hash(str(item), 0) % self.m
        h2 = mmh3.hash(str(item), h1) % self.m
        
        positions = []
        for i in range(self.k):
            # Double hashing: (h1 + i * h2) mod m
            pos = (h1 + i * h2) % self.m
            positions.append(pos)
        
        return positions
    
    def add(self, item: str) -> None:
        """Add item to bloom filter."""
        for pos in self._hashes(item):
            self.bits[pos] = True
        self.items_count += 1
    
    def might_contain(self, item: str) -> bool:
        """Check if item might be in set (may have false positives)."""
        return all(self.bits[pos] for pos in self._hashes(item))
    
    def false_positive_probability(self) -> float:
        """Calculate current false positive probability."""
        if self.items_count == 0:
            return 0.0
        
        # P = (1 - e^(-kn/m))^k
        return (1 - math.exp(-self.k * self.items_count / self.m)) ** self.k

class BloomFilterIndex:
    """Bloom filter used as database index for membership queries."""
    
    def __init__(self, table_name: str, column: str):
        self.table_name = table_name
        self.column = column
        self.filter = None
        self.min_value = None
        self.max_value = None
    
    def build(self, values: List[str], fp_rate: float = 0.01):
        """Build bloom filter index from column values."""
        self.filter = BloomFilter(len(values), fp_rate)
        
        for val in values:
            self.filter.add(val)
            # Track min/max for range queries
            if self.min_value is None or val < self.min_value:
                self.min_value = val
            if self.max_value is None or val > self.max_value:
                self.max_value = val
        
        print(f"Built index for {self.table_name}.{self.column}:")
        print(f"  - {len(values)} items")
        print(f"  - {self.filter.m} bits ({self.filter.m / 8 / 1024:.2f} KB)")
        print(f"  - {self.filter.k} hash functions")
        print(f"  - FP rate: {self.filter.false_positive_probability():.4f}")
    
    def might_contain(self, value: str) -> Tuple[bool, str]:
        """Check if value might exist in indexed column."""
        if self.filter is None:
            return True, "NO_INDEX"
        
        # Quick range check first
        if self.min_value and value < self.min_value:
            return False, "BELOW_MIN"
        if self.max_value and value > self.max_value:
            return False, "ABOVE_MAX"
        
        # Bloom filter check
        if self.filter.might_contain(value):
            return True, "MIGHT_EXIST"
        else:
            return False, "DEFINITELY_NOT_EXISTS"

class QueryOptimizer:
    """Uses bloom filter indexes for query optimization."""
    
    def __init__(self):
        self.indexes: Dict[str, BloomFilterIndex] = {}
    
    def create_index(self, table: str, column: str, values: List[str]):
        """Create bloom filter index on table column."""
        key = f"{table}.{column}"
        index = BloomFilterIndex(table, column)
        index.build(values)
        self.indexes[key] = index
    
    def optimize_query(self, table: str, column: str, value: str) -> str:
        """Optimize query using bloom filter index."""
        key = f"{table}.{column}"
        index = self.indexes.get(key)
        
        if not index:
            return "FULL_TABLE_SCAN"
        
        might_exist, reason = index.might_contain(value)
        
        if not might_exist:
            return f"SKIP_SCAN ({reason})"
        
        return "INDEX_SCAN"
```
**Why**: Bloom filters provide space-efficient membership testing with tunable false positive rates, used in databases for index skipping, cache filtering, and join optimization.

## 9. Object Pool Pattern

```python
# PROBLEM: Reuse expensive objects to improve performance
# TIME: O(1) for acquire/release SPACE: O(n) for pool size
from dataclasses import dataclass
from typing import Generic, TypeVar, Optional, List
from threading import Lock
import time

T = TypeVar('T')

@dataclass
class PooledObject:
    """Wrapper for objects in the pool."""
    obj: T
    in_use: bool = False
    last_used: float = 0
    use_count: int = 0

class ObjectPool(Generic[T]):
    """Thread-safe pool for reusable objects."""
    
    def __init__(self, 
                 factory: callable,
                 max_size: int = 10,
                 validate: callable = None):
        self.factory = factory
        self.max_size = max_size
        self.validate = validate or (lambda x: True)
        
        self.pool: List[PooledObject] = []
        self.lock = Lock()
        self._created = 0
        self._borrowed = 0
        self._returned = 0
    
    def acquire(self, timeout: float = 5.0) -> Optional[T]:
        """Get an object from the pool."""
        start = time.time()
        
        while time.time() - start < timeout:
            with self.lock:
                # Try to find available valid object
                for pooled in self.pool:
                    if not pooled.in_use and self.validate(pooled.obj):
                        pooled.in_use = True
                        pooled.last_used = time.time()
                        pooled.use_count += 1
                        self._borrowed += 1
                        return pooled.obj
                
                # Create new object if under limit
                if len(self.pool) < self.max_size:
                    obj = self.factory()
                    pooled = PooledObject(
                        obj=obj,
                        in_use=True,
                        last_used=time.time(),
                        use_count=1
                    )
                    self.pool.append(pooled)
                    self._created += 1
                    self._borrowed += 1
                    return obj
            
            # Wait briefly before retry
            time.sleep(0.01)
        
        return None  # Timeout
    
    def release(self, obj: T) -> None:
        """Return object to the pool."""
        with self.lock:
            for pooled in self.pool:
                if pooled.obj is obj:
                    pooled.in_use = False
                    self._returned += 1
                    return
        
        # Object not from this pool - dispose it
        self._dispose(obj)
    
    def _dispose(self, obj: T) -> None:
        """Dispose of an object."""
        if hasattr(obj, 'close'):
            obj.close()
    
    def evict_idle(self, max_idle_seconds: float = 300) -> int:
        """Remove idle objects from pool."""
        now = time.time()
        evicted = 0
        
        with self.lock:
            remaining = []
            for pooled in self.pool:
                if not pooled.in_use:
                    idle_time = now - pooled.last_used
                    if idle_time > max_idle_seconds:
                        self._dispose(pooled.obj)
                        evicted += 1
                        continue
                remaining.append(pooled)
            self.pool = remaining
        
        return evicted
    
    def stats(self) -> dict:
        """Get pool statistics."""
        with self.lock:
            available = sum(1 for p in self.pool if not p.in_use)
            return {
                'total_created': self._created,
                'total_borrowed': self._borrowed,
                'total_returned': self._returned,
                'pool_size': len(self.pool),
                'available': available,
                'in_use': len(self.pool) - available
            }

class DatabaseConnection:
    """Example expensive object to pool."""
    
    def __init__(self, conn_string: str):
        self.conn_string = conn_string
        self.connected = False
        self._connect()
    
    def _connect(self):
        """Simulate expensive connection."""
        time.sleep(0.1)  # Connection overhead
        self.connected = True
        print(f"✓ Connected: {self.conn_string}")
    
    def execute(self, query: str) -> str:
        """Execute query on connection."""
        if not self.connected:
            raise Exception("Not connected")
        return f"Result of: {query}"
    
    def close(self):
        """Close connection."""
        self.connected = False
        print(f"✗ Disconnected: {self.conn_string}")

class ConnectionPool(ObjectPool[DatabaseConnection]):
    """Specialized pool for database connections."""
    
    def __init__(self, conn_string: str, max_size: int = 10):
        super().__init__(
            factory=lambda: DatabaseConnection(conn_string),
            max_size=max_size,
            validate=lambda c: c.connected
        )
        self.conn_string = conn_string
    
    def get_connection(self) -> Optional[DatabaseConnection]:
        """Acquire database connection."""
        return self.acquire()
    
    def return_connection(self, conn: DatabaseConnection):
        """Release database connection."""
        self.release(conn)

# Context manager for automatic release
class PooledConnection:
    """Context manager for pooled connections."""
    
    def __init__(self, pool: ConnectionPool):
        self.pool = pool
        self.conn = None
    
    def __enter__(self) -> DatabaseConnection:
        self.conn = self.pool.get_connection()
        return self.conn
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.conn:
            self.pool.return_connection(self.conn)
        return False
```
**Why**: Object pools avoid expensive object creation/destruction, manage resource limits, and provide thread-safe access to shared resources like database connections.

# PROBLEM: Understand memory visibility and ordering in concurrent code
# TIME: O(1) per operation SPACE: O(1) for atomic variables
import threading
from threading import Barrier, Lock
import time

class MemoryOrderingDemo:
    """Demonstrates memory ordering issues and solutions."""
    
    def __init__(self):
        self.x = 0
        self.y = 0
        self.flag = False
        self.lock = Lock()
    
    def demonstrate_reordering(self):
        """Show compiler/CPU instruction reordering."""
        print("=== Demonstrating Memory Reordering ===")
        
        results = {'r1': 0, 'r2': 0}
        
        def thread1():
            self.x = 1
            results['r1'] = self.y
        
        def thread2():
            self.y = 1
            results['r2'] = self.x
        
        # Reset
        self.x = 0
        self.y = 0
        
        t1 = threading.Thread(target=thread1)
        t2 = threading.Thread(target=thread2)
        
        t1.start()
        t2.start()
        t1.join()
        t2.join()
        
        # Possible results: (1,1), (1,0), (0,1)
        # In rare cases: (0,0) due to reordering
        print(f"Results: r1={results['r1']}, r2={results['r2']}")

class SequentialBarrier:
    """Full memory barrier ensuring sequential consistency."""
    
    def __init__(self):
        self.lock = Lock()
    
    def acquire(self):
        """Acquire barrier - prevents reordering of reads after."""
        self.lock.acquire()
    
    def release(self):
        """Release barrier - prevents reordering of writes before."""
        self.lock.release()

class SpinLock:
    """Lock using memory ordering guarantees."""
    
    def __init__(self):
        self._locked = False
        self._lock = threading.Lock()
    
    def acquire(self):
        """Acquire spin lock with acquire semantics."""
        while True:
            with self._lock:
                if not self._locked:
                    self._locked = True
                    return  # Acquire barrier
            # Spin wait
            time.sleep(0.0001)
    
    def release(self):
        """Release spin lock with release semantics."""
        with self._lock:
            self._locked = False  # Release barrier

class DoubleCheckedLocking:
    """Thread-safe initialization with memory ordering."""
    
    def __init__(self):
        self._instance = None
        self._lock = Lock()
    
    def get_instance(self, factory):
        """Get singleton with double-checked locking.
        
        Requires:
        1. Volatile read of _instance
        2. Memory barrier before publishing
        """
        # First check (without lock)
        if self._instance is None:
            with self._lock:
                # Second check (with lock)
                if self._instance is None:
                    instance = factory()
                    # Memory barrier ensures instance fully constructed
                    # before assignment to _instance
                    self._instance = instance
        return self._instance

class ProducerConsumerPattern:
    """Producer-consumer with proper memory ordering."""
    
    def __init__(self, buffer_size: int = 10):
        self.buffer = []
        self.buffer_size = buffer_size
        self.lock = Lock()
        self.not_empty = threading.Condition(self.lock)
        self.not_full = threading.Condition(self.lock)
    
    def produce(self, item):
        """Produce item with release semantics."""
        with self.not_full:
            while len(self.buffer) >= self.buffer_size:
                self.not_full.wait()
            
            # Critical section - modification
            self.buffer.append(item)
            
            # Release: signal consumer (memory barrier)
            self.not_empty.notify()
    
    def consume(self):
        """Consume item with acquire semantics."""
        with self.not_empty:
            while len(self.buffer) == 0:
                self.not_empty.wait()
            
            # Critical section - modification
            item = self.buffer.pop(0)
            
            # Release: signal producer
            self.not_full.notify()
            
        return item

class HappensBeforeExample:
    """Demonstrates happens-before relationships."""
    
    def __init__(self):
        self.shared_data = 0
        self.ready = False
        self.lock = Lock()
    
    def writer_thread(self):
        """Write data then set flag (happens-before)."""
        with self.lock:
            self.shared_data = 42
            # Lock release creates happens-before with next acquire
            self.ready = True
    
    def reader_thread(self):
        """Check flag then read data (synchronizes-with)."""
        while not self.ready:
            time.sleep(0.001)
        
        with self.lock:
            # Lock acquire synchronizes-with writer's release
            # Therefore sees shared_data = 42 guaranteed
            print(f"Read data: {self.shared_data}")
    
    def run(self):
        """Run threads demonstrating happens-before."""
        # Reset state
        self.shared_data = 0
        self.ready = False
        
        t1 = threading.Thread(target=self.writer_thread)
        t2 = threading.Thread(target=self.reader_thread)
        
        t2.start()  # Start reader first
        t1.start()  # Then writer
        
        t1.join()
        t2.join()

class DataRacePrevention:
    """Shows how to prevent data races."""
    
    def __init__(self):
        self.counter = 0
        self.lock = Lock()
        self.atomic_counter = 0
    
    def unsafe_increment(self):
        """UNSAFE: Data race on counter."""
        self.counter += 1  # Not atomic!
    
    def safe_increment(self):
        """SAFE: Protected by lock."""
        with self.lock:
            self.counter += 1
    
    def atomic_increment(self):
        """SIMULATE: Atomic increment (Python GIL makes this safe anyway)."""
        # In languages without GIL, would use atomic operations
        with self.lock:
            self.atomic_counter += 1
    
    def demonstrate_race(self):
        """Show difference between safe and unsafe access."""
        # Unsafe version
        self.counter = 0
        threads = [
            threading.Thread(target=self.unsafe_increment)
            for _ in range(1000)
        ]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        print(f"Unsafe result: {self.counter} (expected 1000)")
        
        # Safe version
        self.counter = 0
