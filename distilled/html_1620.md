## 4. html_aria_role_tablist

```html
<!-- PROBLEM: Create accessible tab navigation -->
<!-- TIME: O(1) SPACE: O(1) -->
<div class="tabs">
  <div role="tablist" aria-label="Settings tabs">
    <button 
      role="tab" 
      id="tab-general"
      aria-selected="true"
      aria-controls="panel-general"
      tabindex="0">
      General
    </button>
    <button 
      role="tab" 
      id="tab-security"
      aria-selected="false"
      aria-controls="panel-security"
      tabindex="-1">
      Security
    </button>
    <button 
      role="tab" 
      id="tab-privacy"
      aria-selected="false"
      aria-controls="panel-privacy"
      tabindex="-1">
      Privacy
    </button>
  </div>

  <div 
    role="tabpanel" 
    id="panel-general"
    aria-labelledby="tab-general"
    tabindex="0">
    <h2>General Settings</h2>
    <p>Configure your general preferences here.</p>
  </div>

  <div 
    role="tabpanel" 
    id="panel-security"
    aria-labelledby="tab-security"
    tabindex="0"
    hidden>
    <h2>Security Settings</h2>
    <p>Manage security options.</p>
  </div>

  <div 
    role="tabpanel" 
    id="panel-privacy"
    aria-labelledby="tab-privacy"
    tabindex="0"
    hidden>
    <h2>Privacy Settings</h2>
    <p>Control your privacy settings.</p>
  </div>
</div>

<script>
const tabs = document.querySelectorAll('[role="tab"]');
const panels = document.querySelectorAll('[role="tabpanel"]');

tabs.forEach(tab => {
  tab.addEventListener('click', () => {
    // Deselect all tabs
    tabs.forEach(t => {
      t.setAttribute('aria-selected', 'false');
      t.setAttribute('tabindex', '-1');
    });
    
    // Hide all panels
    panels.forEach(p => p.hidden = true);
    
    // Select clicked tab
    tab.setAttribute('aria-selected', 'true');
    tab.setAttribute('tabindex', '0');
    
    // Show corresponding panel
    const panelId = tab.getAttribute('aria-controls');
    document.getElementById(panelId).hidden = false;
  });

  // Keyboard navigation
  tab.addEventListener('keydown', (e) => {
    const tabList = Array.from(tabs);
    const index = tabList.indexOf(tab);
    
    if (e.key === 'ArrowRight') {
      const nextTab = tabList[(index + 1) % tabList.length];
      nextTab.focus();
      nextTab.click();
    } else if (e.key === 'ArrowLeft') {
      const prevTab = tabList[(index - 1 + tabList.length) % tabList.length];
      prevTab.focus();
      prevTab.click();
    }
  });
});
</script>
```
**Why**: `role="tablist"` creates a semantic tab interface that screen readers can navigate. Combined with `role="tab"`, `role="tabpanel"`, and proper ARIA attributes, it provides full keyboard navigation and screen reader support for tabbed interfaces.

## 6. css_flexbox_flex_shrink

```html

<!-- TIME: O(1) SPACE: O(1) -->

      flex-basis: 200px;
      background: lightgreen;
      padding: 20px;
      margin: 5px;
    }
    
    .item-shrink-2 {
      flex-shrink: 2; /* Shrinks twice as much */

      display: flex;
      width: 400px;

    }

    <div class="item-no-shrink">Item 2</div>

Setting to 0 prevents shrinking, useful for fixed-width sidebars. Higher values make items shrink more relative to others, enabling precise layout control.

## 8. js_promise_any

```html
<!-- PROBLEM: Get the first successfully resolved promise -->
<!-- TIME: O(n) SPACE: O(n) -->
<!DOCTYPE html>
<html lang="en">
<head>
  <title>Promise.any() Examples</title>
</head>
<body>
  <div id="output"></div>
  
  <script>
    const output = document.getElementById('output');
    
    function log(message) {
      output.innerHTML += `<p>${message}</p>`;
    }

    // Example 1: Basic usage - returns first resolved
    const promise1 = new Promise((resolve, reject) => {
      setTimeout(() => reject('Promise 1 failed'), 1000);
    });
    
    const promise2 = new Promise((resolve, reject) => {
      setTimeout(() => resolve('Promise 2 succeeded'), 500);
    });
    
    const promise3 = new Promise((resolve, reject) => {
      setTimeout(() => resolve('Promise 3 succeeded'), 800);
    });

    Promise.any([promise1, promise2, promise3])
      .then(result => log(`First resolved: ${result}`))
      .catch(error => log(`All failed: ${error}`));
    // Output: "First resolved: Promise 2 succeeded"

    // Example 2: Fetching from multiple APIs - use first response
    function fetchFromMirror(url) {
      return fetch(url).then(response => {
        if (!response.ok) throw new Error('Request failed');
        return response.json();
      });
    }

    const apiUrls = [
      'https://api.mirror1.com/data',
      'https://api.mirror2.com/data',
      'https://api.mirror3.com/data'
    ];

    const fetchPromises = apiUrls.map(url => fetchFromMirror(url));
    
    Promise.any(fetchPromises)
      .then(data => {
        log('Got data from fastest mirror');
        console.log(data);
      })
      .catch(error => {
        log('All mirrors failed: ' + error);
      });

    // Example 3: All promises reject - AggregateError
    const failingPromises = [
      Promise.reject('Error 1'),
      Promise.reject('Error 2'),
      Promise.reject('Error 3')
    ];

    Promise.any(failingPromises)
      .then(result => log(`Success: ${result}`))
      .catch(error => {
        // error is an AggregateError
        log(`All promises rejected. Errors:`);
        error.errors.forEach((err, i) => {
          log(`  ${i + 1}: ${err}`);
        });
      });

    // Example 4: Practical use - image loading
    function loadImage(src) {
      return new Promise((resolve, reject) => {
        const img = new Image();
        img.onload = () => resolve(img);
        img.onerror = () => reject(`Failed to load: ${src}`);
        img.src = src;
      });
    }

    const imageSources = [
      'image-high.jpg',    // Might fail
      'image-medium.jpg',  // Might fail
      'image-low.jpg'      // Fallback
    ];

    const imagePromises = imageSources.map(src => loadImage(src));
    
    Promise.any(imagePromises)
      .then(img => {
        document.body.appendChild(img);
        log('Image loaded successfully');
      })
      .catch(error => {
        log('Failed to load any image version');
      });

    // Example 5: Timeout race with Promise.any
    function withTimeout(promise, ms) {
      const timeout = new Promise((_, reject) => {
        setTimeout(() => reject(`Timeout after ${ms}ms`), ms);
      });
      return Promise.any([promise, timeout]);
    }

    const slowOperation = new Promise(resolve => {
      setTimeout(() => resolve('Slow operation done'), 3000);
    });

    withTimeout(slowOperation, 2000)
      .then(result => log(result))
      .catch(error => log(`Operation timed out: ${error}`));
  </script>
</body>
</html>
```
**Why**: `Promise.any()` waits for the first promise to fulfill (resolve), opposite of `Promise.race()` which settles with the first rejection or resolution. Perfect for redundancy patterns, trying multiple sources simultaneously, and using whichever responds first. Throws `AggregateError` if all promises reject.

## 9. js_array_pop

```html
<!-- PROBLEM: Remove and return the last element from an array -->
<!-- TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
  <title>Array.pop() Examples</title>
</head>
<body>
  <div id="output"></div>
  
  <script>
    const output = document.getElementById('output');
    
    function log(message) {
      output.innerHTML += `<p>${message}</p>`;
    }

    // Example 1: Basic usage
    const fruits = ['apple', 'banana', 'orange', 'grape'];
    const lastFruit = fruits.pop();
    
    log(`Removed: ${lastFruit}`); // "grape"
    log(`Remaining: ${fruits}`);  // "apple,banana,orange"
    log(`Length: ${fruits.length}`); // 3

    // Example 2: Implementing a stack (LIFO)
    class Stack {
      constructor() {
        this.items = [];
      }
      
      push(item) {
        this.items.push(item);
      }
      
      pop() {
        return this.items.pop();
      }
      
      peek() {
        return this.items[this.items.length - 1];
      }
      
      isEmpty() {
        return this.items.length === 0;
      }
    }

    const stack = new Stack();
    stack.push('First');
    stack.push('Second');
    stack.push('Third');
    
    log(`Top: ${stack.peek()}`); // "Third"
    log(`Popped: ${stack.pop()}`); // "Third"
    log(`Popped: ${stack.pop()}`); // "Second"

    // Example 3: Undo functionality
    const actions = [];
    const undoStack = [];

    function doAction(action) {
      actions.push(action);
      undoStack.push(action);
      log(`Did: ${action}`);
    }

    function undo() {
      const action = undoStack.pop();
      if (action) {
        const index = actions.indexOf(action);
        if (index > -1) actions.splice(index, 1);
        log(`Undid: ${action}`);
      } else {
        log('Nothing to undo');
      }
    }

    doAction('Type A');
    doAction('Type B');
    doAction('Type C');
    undo();
    undo();

    // Example 4: Processing queue (using pop with reverse)
    const tasks = ['Task 1', 'Task 2', 'Task 3', 'Task 4'];
    
    // Pop until empty
    while (tasks.length > 0) {
      const task = tasks.pop();
      log(`Processing: ${task}`);
    }
    log('All tasks completed');

    // Example 5: Empty array returns undefined
    const emptyArray = [];
    const result = emptyArray.pop();
    
    log(`Popped from empty: ${result}`); // "undefined"
    log(`Length still 0: ${emptyArray.length}`); // 0

    // Example 6: Browser history simulation
    const browserHistory = [
      'home.html',
      'about.html',
      'contact.html',
      'products.html'
    ];

    let currentPage = browserHistory[browserHistory.length - 1];
    log(`Current page: ${currentPage}`);

    function goBack() {
      if (browserHistory.length > 1) {
        browserHistory.pop();
        currentPage = browserHistory[browserHistory.length - 1];
        log(`Went back to: ${currentPage}`);
      } else {
        log('Cannot go back further');
      }
    }

    goBack();
    goBack();
    goBack();
    goBack();

    // Example 7: Array-like objects
    function logArguments() {
      // Convert arguments to array
      const args = Array.from(arguments);
      
      while (args.length > 0) {
        log(`Arg: ${args.pop()}`);
      }
    }

    logArguments('a', 'b', 'c');

    // Example 8: Performance - pop is O(1)
    const largeArray = Array(100000).fill(null).map((_, i) => i);
    
    console.time('pop');
    for (let i = 0; i < 1000; i++) {
      largeArray.pop();
    }
    console.timeEnd('pop');
    
    log(`Popped 1000 items from large array`);
  </script>
</body>
</html>
```
**Why**: `Array.pop()` removes and returns the last element in O(1) time, making it ideal for stack data structures and LIFO (Last In First Out) operations. It mutates the original array, decreases `length`, and returns `undefined` for empty arrays. Perfect for undo systems, history navigation, and processing arrays from end to start.

      if (e.key === 'ArrowRight') posX += step;
      if (e.key === 'ArrowLeft') posX -= step;
      if (e.key === 'ArrowDown') posY += step;
      if (e.key === 'ArrowUp') posY -= step;
      
      box.style.transform = `translate(${posX}px, ${posY}px)`;
    });

    // Parallax scrolling effect
    const parallax = document.getElementById('parallax');
    
    window.addEventListener('scroll', () => {
      const scrolled = window.pageYOffset;
      parallax.style.transform = `translateY(${scrolled * 0.5}px)`;
    });

    // Get current transform value
    function getTranslateX(element) {
      const style = window.getComputedStyle(element);
      const matrix = new DOMMatrix(style.transform);
      return matrix.m41; // translateX value
    }

    // Smooth translation with GSAP-like easing
    function smoothTranslate(element, targetX, targetY, duration = 1000) {
      const start = performance.now();
      const startX = getTranslateX(element) || 0;
      
      function animate(currentTime) {
        const elapsed = currentTime - start;
        const progress = Math.min(elapsed / duration, 1);
        
        // Ease out quad
        const eased = 1 - (1 - progress) * (1 - progress);
        
        const currentX = startX + (targetX - startX) * eased;
        element.style.transform = `translate(${currentX}px, ${targetY}px)`;
        
        if (progress < 1) {
          requestAnimationFrame(animate);
        }
      }
      
      requestAnimationFrame(animate);
    }

    // Usage: smoothTranslate(box, 200, 100);
  </script>
</body>
</html>
```
**Why**: `transform: translate()` moves elements without affecting document flow or triggering reflows, making it performant for animations. It's GPU-accelerated and works with percentages relative to the element's own size. Perfect for centering, hover effects, parallax scrolling, and drag-and-drop interfaces. Use `translate3d()` to force hardware acceleration.
