```python
    def compute_cell(self, site_idx):
        """Compute Voronoi cell for a specific site"""
        sx, sy = self.points[site_idx]
        edges = []
        
        # Compute half-planes for all neighbors
        for i, (px, py) in enumerate(self.points):
            if i == site_idx:
                continue
            # Perpendicular bisector
            mid_x = (sx + px) / 2
            mid_y = (sy + py) / 2
            dx = px - sx
            dy = py - sy
            # Store edge as (point, direction)
            edges.append(((mid_x, mid_y), (-dy, dx)))
        
        return self._clip_cell(edges)
    
    def _clip_cell(self, edges, bounds=(-1000, -1000, 1000, 1000)):
        """Clip cell to bounding box using Sutherland-Hodgman"""
        polygon = [
            (bounds[0], bounds[1]),
            (bounds[2], bounds[1]),
            (bounds[2], bounds[3]),
            (bounds[0], bounds[3])
        ]
        
        for (px, py), (dx, dy) in edges:
            new_polygon = []
            for i in range(len(polygon)):
                x1, y1 = polygon[i]
                x2, y2 = polygon[(i + 1) % len(polygon)]
                
                # Side of line: positive = inside
                v1 = (x1 - px) * dx + (y1 - py) * dy
                v2 = (x2 - px) * dx + (y2 - py) * dy
                
                if v1 >= 0:
                    new_polygon.append((x1, y1))
                if v1 * v2 < 0:
                    # Line segment crosses edge
                    t = v1 / (v1 - v2)
                    ix = x1 + t * (x2 - x1)
                    iy = y1 + t * (y2 - y1)
                    new_polygon.append((ix, iy))
            
            polygon = new_polygon
            if len(polygon) < 3:
                return []
        
        return polygon
```

**Why**: Fortune's algorithm efficiently constructs Voronoi diagrams using a sweep line with O(n log n) complexity; used in computational geometry, path planning, and spatial analysis.

### Minkowski Sum

```python
# PROBLEM: Compute Minkowski sum of two convex polygons
# TIME: O(n + m) SPACE: O(n + m)
def minkowski_sum(polygon1, polygon2):
    """
    Compute Minkowski sum of two convex polygons.
    Each polygon is list of vertices in counterclockwise order.
    """
    if not polygon1 or not polygon2:
        return []
    
    # Sort vertices by angle from centroid
    def sort_by_angle(poly):
        cx = sum(p[0] for p in poly) / len(poly)
        cy = sum(p[1] for p in poly) / len(poly)
        return sorted(poly, key=lambda p: math.atan2(p[1]-cy, p[0]-cx))
    
    p1 = sort_by_angle(polygon1)
    p2 = sort_by_angle(polygon2)
    
    # Compute edges sorted by angle
    def get_edges(poly):
        edges = []
        n = len(poly)
        for i in range(n):
            p1 = poly[i]
            p2 = poly[(i + 1) % n]
            edges.append((p2[0] - p1[0], p2[1] - p1[1]))
        return edges
    
    edges1 = get_edges(p1)
    edges2 = get_edges(p2)
    
    # Merge edges by angle
    result = []
    i, j = 0, 0
    n, m = len(edges1), len(edges2)
    
    # Start from origin
    x, y = 0, 0
    result.append((x, y))
    
    # Merge sorted edges
    while i < n or j < m:
        if i == n:
            ex, ey = edges2[j]
            j += 1
        elif j == m:
            ex, ey = edges1[i]
            i += 1
        else:
            e1x, e1y = edges1[i]
            e2x, e2y = edges2[j]
            angle1 = math.atan2(e1y, e1x)
            angle2 = math.atan2(e2y, e2x)
            
            if angle1 < angle2:
                ex, ey = e1x, e1y
                i += 1
            else:
                ex, ey = e2x, e2y
                j += 1
        
        x += ex
        y += ey
        result.append((x, y))
    
    # Remove last duplicate point
    result.pop()
    return result

def minkowski_difference(polygon1, polygon2):
    """Minkowski difference (used for collision detection)"""
    negated_p2 = [(-p[0], -p[1]) for p in polygon2]
    return minkowski_sum(polygon1, negated_p2)

def contains_origin(polygon):
    """Check if polygon contains origin (for collision detection)"""
    n = len(polygon)
    if n < 3:
        return False
    
    # Ray casting algorithm
    x, y = 0, 0
    count = 0
    
    for i in range(n):
        x1, y1 = polygon[i]
        x2, y2 = polygon[(i + 1) % n]
        
        if ((y1 > y) != (y2 > y)):
            x_intersect = x1 + (y - y1) / (y2 - y1) * (x2 - x1)
            if x < x_intersect:
                count += 1
    
    return count % 2 == 1
```

**Why**: Minkowski sum computes the convolution of two shapes; essential for collision detection, motion planning, and computational geometry operations.

### Subgraph Isomorphism

```python
# PROBLEM: Find if pattern graph exists as subgraph in target graph
# TIME: O(n! / (n-m)!) SPACE: O(V)
class SubgraphIsomorphism:
    def __init__(self, pattern, target):
        self.pattern = pattern
        self.target = target
        self.m = len(pattern)
        self.n = len(target)
        self.mapping = {}
        self.used = set()
        
    def find_isomorphism(self):
        """Find subgraph isomorphism using backtracking with pruning"""
        if self.m > self.n:
            return None
        
        # Order pattern vertices by degree (most constrained first)
        order = sorted(range(self.m), 
                      key=lambda v: len(self.pattern[v]), 
                      reverse=True)
        
        if self._search(order, 0):
            return self.mapping
        return None
    
    def _search(self, order, idx):
        """Backtracking search with constraint propagation"""
        if idx == self.m:
            return True
        
        u = order[idx]
        
        for v in range(self.n):
            if v in self.used:
                continue
            
            # Check if v is compatible with u
            if not self._is_compatible(u, v):
                continue
            
            # Try assignment
            self.mapping[u] = v
            self.used.add(v)
            
            if self._search(order, idx + 1):
                return True
            
            # Backtrack
            del self.mapping[u]
            self.used.remove(v)
        
        return False
    
    def _is_compatible(self, u, v):
        """Check if pattern vertex u can map to target vertex v"""
        # Degree check
        if len(self.pattern[u]) > len(self.target[v]):
            return False
        
        # Check already mapped neighbors
        for neighbor_u in self.pattern[u]:
            if neighbor_u in self.mapping:
                neighbor_v = self.mapping[neighbor_u]
                if neighbor_v not in self.target[v]:
                    return False
        
        return True
    
    def count_isomorphisms(self):
        """Count all distinct subgraph isomorphisms"""
        order = sorted(range(self.m), 
                      key=lambda v: len(self.pattern[v]), 
                      reverse=True)
        self.count = 0
        self._count_search(order, 0)
        return self.count
    
    def _count_search(self, order, idx):
        """Count all valid mappings"""
        if idx == self.m:
            self.count += 1
            return
        
        u = order[idx]
        
        for v in range(self.n):
            if v in self.used:
                continue
            
            if not self._is_compatible(u, v):
                continue
            
            self.mapping[u] = v
            self.used.add(v)
            self._count_search(order, idx + 1)
            del self.mapping[u]
            self.used.remove(v)
```

**Why**: Subgraph isomorphism finds pattern graphs in larger graphs; critical in pattern matching, chemical structure search, and social network analysis.

```python
# PROBLEM: Self-adjusting BST with amortized O(log n) operations
# TIME: O(log n) amortized SPACE: O(n)
class SplayNode:
    def __init__(self, key, value=None):
        self.key = key
        self.value = value
        self.left = None
        self.right = None
        self.parent = None
        self.size = 1  # Subtree size
        self.sum = key  # Subtree sum for range queries

class SplayTree:
    def __init__(self):
        self.root = None
    
    def _rotate_left(self, x):
        """Left rotation"""
        y = x.right
        x.right = y.left
        if y.left:
            y.left.parent = x
        y.parent = x.parent
        if not x.parent:
            self.root = y
        elif x == x.parent.left:
            x.parent.left = y
        else:
            x.parent.right = y
        y.left = x
        x.parent = y
        self._update(x)
        self._update(y)
    
    def _rotate_right(self, x):
        """Right rotation"""
        y = x.left
        x.left = y.right
        if y.right:
            y.right.parent = x
        y.parent = x.parent
        if not x.parent:
            self.root = y
        elif x == x.parent.right:
            x.parent.right = y
        else:
            x.parent.left = y
        y.right = x
        x.parent = y
        self._update(x)
        self._update(y)
    
    def _splay(self, x):
        """Splay operation - bring x to root"""
        while x.parent:
            if not x.parent.parent:
                # Zig step
                if x == x.parent.left:
                    self._rotate_right(x.parent)
                else:
                    self._rotate_left(x.parent)
            elif x == x.parent.left and x.parent == x.parent.parent.left:
                # Zig-zig step
                self._rotate_right(x.parent.parent)
                self._rotate_right(x.parent)
            elif x == x.parent.right and x.parent == x.parent.parent.right:
                # Zig-zig step
                self._rotate_left(x.parent.parent)
                self._rotate_left(x.parent)
            elif x == x.parent.right and x.parent == x.parent.parent.left:
                # Zig-zag step
                self._rotate_left(x.parent)
                self._rotate_right(x.parent)
            else:
                # Zig-zag step
                self._rotate_right(x.parent)
                self._rotate_left(x.parent)
    
    def _update(self, node):
        """Update node statistics"""
        if node:
            node.size = 1
            node.sum = node.key
            if node.left:
                node.size += node.left.size
                node.sum += node.left.sum
            if node.right:
                node.size += node.right.size
                node.sum += node.right.sum
    
    def insert(self, key, value=None):
        """Insert key-value pair"""
        if not self.root:
            self.root = SplayNode(key, value)
            return
        
        node = self._find(key)
        if node.key == key:
            node.value = value
            return
        
        new_node = SplayNode(key, value)
        if key < node.key:
            new_node.left = node.left
            new_node.right = node
            node.left = None
        else:
            new_node.right = node.right
            new_node.left = node
            node.right = None
        
        self.root = new_node
        self._update(node)
        self._update(new_node)
    
    def search(self, key):
        """Search for key"""
        if not self.root:
            return None
        
        node = self._find(key)
        self._splay(node)
        
        if node.key == key:
            return node.value
        return None
    
    def _find(self, key):
        """Find node with key or closest node"""
        node = self.root
        while True:
            if key < node.key and node.left:
                node = node.left
            elif key > node.key and node.right:
                node = node.right
            else:
                return node
    
    def delete(self, key):
        """Delete key"""
        if not self.root:
            return False
        
        node = self._find(key)
        self._splay(node)
        
        if node.key != key:
            return False
        
        # Merge left and right subtrees
        if not node.left:
            self.root = node.right
            if self.root:
                self.root.parent = None
        elif not node.right:
            self.root = node.left
            if self.root:
                self.root.parent = None
        else:
            left_max = node.left
            while left_max.right:
                left_max = left_max.right
            self._splay(left_max)
            left_max.right = node.right
            left_max.right.parent = left_max
            self.root = left_max
            self._update(left_max)
        
        return True
    
    def range_sum(self, lo, hi):
        """Compute sum of keys in range [lo, hi]"""
        # Split into three parts: < lo, [lo, hi], > hi
        self._split(lo)
        left_tree = self.root.left
        
        # Further split the right part
        self._splay(self._find(hi))
        right_tree = self.root.right
        
        # The middle part contains [lo, hi]
        result = self.root.sum if self.root else 0
        if left_tree:
            result -= left_tree.sum
        if right_tree:
            result -= right_tree.sum
        
        return result

class LinkCutTree(SplayTree):
    """Link-Cut Tree for dynamic tree operations"""
    
    def __init__(self, n):
        super().__init__()
        self.nodes = [SplayNode(i) for i in range(n)]
        self.root = self.nodes[0]
    
    def link(self, u, v):
        """Link tree u to tree v"""
        self._expose(u)
        self._expose(v)
        if self._find_root(u) != v:
            self.nodes[u].parent = self.nodes[v]
    
    def cut(self, u):
        """Cut edge from u to its parent"""
        self._expose(u)
        if self.nodes[u].left:
            self.nodes[u].left.parent = None
            self.nodes[u].left = None
    
    def find_root(self, u):
        """Find root of tree containing u"""
        self._expose(u)
        node = self.nodes[u]
        while node.left:
            node = node.left
        self._splay(node)
        return node.key
    
    def _expose(self, v):
        """Make preferred path from root to v"""
        node = self.nodes[v]
        self._splay(node)
        node.right = None
        self._update(node)
        
        while node.parent:
            parent = node.parent
            self._splay(parent)
            parent.right = node
            self._update(parent)
            self._splay(node)
```

**Why**: Splay trees are self-adjusting BSTs with amortized O(log n) operations and excellent locality; Link-Cut trees enable dynamic tree operations for network flow and LCA queries.

```python
# PROBLEM: Spatial indexing for 3D points with efficient range queries
# TIME: O(log n) search, O(n) build SPACE: O(n)
class OctreeNode:
    def __init__(self, center, size):
        self.center = center  # (x, y, z)
        self.size = size      # Half-width of cube
        self.children = [None] * 8  # 8 octants
        self.points = []      # Points in this node (leaf only)
        self.is_leaf = True
        self.capacity = 8     # Max points before split
        self.depth = 0

class Octree:
    def __init__(self, center, size, max_depth=10):
        self.root = OctreeNode(center, size)
        self.max_depth = max_depth
    
    def insert(self, point, data=None):
        """Insert point into octree"""
        self._insert(self.root, point, data, 0)
    
    def _insert(self, node, point, data, depth):
        """Recursive insertion"""
        if node.is_leaf:
            if len(node.points) < node.capacity or depth >= self.max_depth:
                node.points.append((point, data))
            else:
                # Split node
                self._split(node, depth)
                # Re-insert existing points
                for p, d in node.points:
                    octant = self._get_octant(node.center, p)
                    self._insert(node.children[octant], p, d, depth + 1)
                node.points = []
                # Insert new point
                octant = self._get_octant(node.center, point)
                self._insert(node.children[octant], point, data, depth + 1)
        else:
            octant = self._get_octant(node.center, point)
            self._insert(node.children[octant], point, data, depth + 1)
    
    def _split(self, node, depth):
        """Split node into 8 children"""
        size = node.size / 2
        x, y, z = node.center
        
        # Define 8 octant centers
        offsets = [
            (-size, -size, -size),
            (size, -size, -size),
            (-size, size, -size),
            (size, size, -size),
            (-size, -size, size),
            (size, -size, size),
            (-size, size, size),
            (size, size, size),
        ]
        
        for i, (dx, dy, dz) in enumerate(offsets):
            child_center = (x + dx, y + dy, z + dz)
            node.children[i] = OctreeNode(child_center, size)
            node.children[i].depth = depth + 1
        
        node.is_leaf = False
    
    def _get_octant(self, center, point):
        """Determine which octant point belongs to"""
        x, y, z = center
        px, py, pz = point
        
        octant = 0
        if px >= x: octant |= 1
        if py >= y: octant |= 2
        if pz >= z: octant |= 4
        return octant
    
    def search_radius(self, center, radius):
        """Find all points within radius of center"""
        results = []
        self._search_radius(self.root, center, radius, results)
        return results
    
    def _search_radius(self, node, center, radius, results):
        """Recursive radius search"""
        if not self._intersects_sphere(node.center, node.size, center, radius):
            return
        
        if node.is_leaf:
            for point, data in node.points:
                dist = self._distance(point, center)
                if dist <= radius:
                    results.append((point, data, dist))
        else:
            for child in node.children:
                if child:
                    self._search_radius(child, center, radius, results)
    
    def _intersects_sphere(self, box_center, size, sphere_center, radius):
        """Check if box intersects sphere"""
        # Compute closest point on box to sphere center
        closest = []
        for i in range(3):
            closest.append(max(box_center[i] - size,
                             min(sphere_center[i], box_center[i] + size)))
        
        dist_sq = sum((c - s)**2 for c, s in zip(closest, sphere_center))
        return dist_sq <= radius ** 2
    
    def _distance(self, p1, p2):
        """Euclidean distance between two points"""
        return sum((a - b)**2 for a, b in zip(p1, p2)) ** 0.5
    
    def nearest_neighbors(self, point, k):
        """Find k nearest neighbors"""
        import heapq
        heap = []  # Min heap of (-distance, point, data)
        
        self._knn_search(self.root, point, k, heap)
        
        # Convert to sorted list
        results = []
        while heap:
            dist, pt, data = heapq.heappop(heap)
            results.append((pt, data, -dist))
        
        return results[::-1]
    
    def _knn_search(self, node, point, k, heap):
        """K-nearest neighbor search"""
        if node.is_leaf:
            for pt, data in node.points:
                dist = self._distance(point, pt)
                if len(heap) < k:
                    heapq.heappush(heap, (-dist, pt, data))
                elif dist < -heap[0][0]:
                    heapq.heapreplace(heap, (-dist, pt, data))
        else:
            # Sort children by distance to query point
            child_list = [(c, self._distance(point, c.center))
                         for c in node.children if c]
            child_list.sort(key=lambda x: x[1])
            
            for child, _ in child_list:
                if len(heap) >= k:
                    # Prune if minimum possible distance is larger
                    min_dist = max(0, self._distance(point, child.center) - child.size)
                    if min_dist >= -heap[0][0]:
                        continue
                
                self._knn_search(child, point, k, heap)
```

**Why**: Octrees provide efficient spatial indexing for 3D data with O(log n) search time; used in 3D graphics, collision detection, point cloud processing, and spatial databases.

```python
# PROBLEM: Index multi-dimensional rectangles with efficient range queries
# TIME: O(log n) search/insert, O(n) build SPACE: O(n)
class Rectangle:
    def __init__(self, min_coords, max_coords):
        self.min = tuple(min_coords)  # Lower-left corner
        self.max = tuple(max_coords)  # Upper-right corner
        self.dim = len(min_coords)
    
    def area(self):
        """Compute rectangle area/volume"""
        result = 1.0
        for i in range(self.dim):
            result *= self.max[i] - self.min[i]
        return result
    
    def intersects(self, other):
        """Check if two rectangles intersect"""
        for i in range(self.dim):
            if self.max[i] < other.min[i] or self.min[i] > other.max[i]:
                return False
        return True
    
    def contains(self, other):
        """Check if this rectangle contains other"""
        for i in range(self.dim):
            if self.min[i] > other.min[i] or self.max[i] < other.max[i]:
                return False
        return True
    
    @staticmethod
    def union(rects):
        """Compute bounding rectangle of multiple rectangles"""
        if not rects:
            return None
        
        dim = rects[0].dim
        min_coords = [float('inf')] * dim
        max_coords = [float('-inf')] * dim
        
        for rect in rects:
            for i in range(dim):
                min_coords[i] = min(min_coords[i], rect.min[i])
                max_coords[i] = max(max_coords[i], rect.max[i])
        
        return Rectangle(min_coords, max_coords)
    
    def enlargement(self, other):
        """Compute area enlargement needed to include other"""
        union_rect = Rectangle.union([self, other])
        return union_rect.area() - self.area()

class RTree:
    def __init__(self, max_entries=4, dim=2):
        self.root = RTreeNode(max_entries=max_entries)
        self.max_entries = max_entries
        self.dim = dim
    
    def insert(self, rect, data):
        """Insert rectangle with data"""
        leaf = self._choose_leaf(self.root, rect)
        leaf.entries.append((rect, data))
        
        if len(leaf.entries) > self.max_entries:
            self._split_node(leaf)
        
        self._adjust_mbrs(leaf)
    
    def _choose_leaf(self, node, rect):
        """Choose leaf node for insertion"""
        if node.is_leaf:
            return node
        
        # Choose child with minimum enlargement
        best_child = None
        best_enlargement = float('inf')
        best_area = float('inf')
        
        for child_rect, child_node in node.entries:
            enlargement = child_rect.enlargement(rect)
            area = child_rect.area()
            
            if (enlargement < best_enlargement or
                (enlargement == best_enlargement and area < best_area)):
                best_enlargement = enlargement
                best_area = area
                best_child = child_node
        
        return self._choose_leaf(best_child, rect)
    
    def _split_node(self, node):
        """Split node using quadratic algorithm"""
        # Pick two seeds with maximum waste
        seed1, seed2 = self._pick_seeds(node.entries)
        
        # Create two new nodes
        node1 = RTreeNode(is_leaf=node.is_leaf, max_entries=self.max_entries)
        node2 = RTreeNode(is_leaf=node.is_leaf, max_entries=self.max_entries)
        
        node1.entries.append(seed1)
        node2.entries.append(seed2)
        
        remaining = [e for e in node.entries if e != seed1 and e != seed2]
        
        # Distribute entries
        for entry in remaining:
            rect, _ = entry
            mbr1 = Rectangle.union([e[0] for e in node1.entries] + [rect])
            mbr2 = Rectangle.union([e[0] for e in node2.entries] + [rect])
            
            area1 = mbr1.area()
            area2 = mbr2.area()
            
            if (area1 < area2 or
                (area1 == area2 and len(node1.entries) < len(node2.entries))):
                node1.entries.append(entry)
            else:
                node2.entries.append(entry)
        
        # Update parent
        if node == self.root:
            self.root = RTreeNode(is_leaf=False, max_entries=self.max_entries)
            self.root.entries = [
                (Rectangle.union([e[0] for e in node1.entries]), node1),
                (Rectangle.union([e[0] for e in node2.entries]), node2)
            ]
        else:
            # Update parent's entry for this node
            pass
        
        node.entries = node1.entries
        node.mbr = Rectangle.union([e[0] for e in node.entries])
    
    def _pick_seeds(self, entries):
        """Pick two entries with maximum combined area"""
        max_waste = -1
        seed1, seed2 = None, None
        
        for i in range(len(entries)):
            for j in range(i + 1, len(entries)):
                rect1 = entries[i][0]
                rect2 = entries[j][0]
                union = Rectangle.union([rect1, rect2])
                waste = union.area() - rect1.area() - rect2.area()
                
                if waste > max_waste:
                    max_waste = waste
                    seed1, seed2 = entries[i], entries[j]
        
        return seed1, seed2
    
    def _adjust_mbrs(self, node):
        """Adjust MBRs up the tree"""
        while node:
            node.mbr = Rectangle.union([e[0] for e in node.entries])
            # Move to parent (simplified)
            break
    
    def search(self, rect):
        """Find all rectangles intersecting query rectangle"""
        results = []
        self._search(self.root, rect, results)
        return results
    
    def _search(self, node, rect, results):
        """Recursive search"""
        if node.is_leaf:
            for entry_rect, data in node.entries:
                if rect.intersects(entry_rect):
                    results.append((entry_rect, data))
        else:
            for entry_rect, child_node in node.entries:
                if rect.intersects(entry_rect):
                    self._search(child_node, rect, results)
```

**Why**: R-trees efficiently index spatial data with rectangle hierarchies; used in GIS systems, spatial databases, and location-based services for fast range queries.

```python
# PROBLEM: Optimized R-tree with better split strategies
# TIME: O(log n) amortized SPACE: O(n)
class RStarTree:
    """R* tree with improved insertion and split algorithms"""
    def __init__(self, max_entries=4):
        self.root = RTreeNode(is_leaf=True, max_entries=max_entries)
        self.max_entries = max_entries
    
    def insert(self, rect, data):
        """Insert with forced reinsertion for better clustering"""
        path = []
        leaf = self._choose_leaf(self.root, rect, path)
        
        leaf.entries.append((rect, data))
        
        if len(leaf.entries) > self.max_entries:
            # First try forced reinsertion
            self._forced_reinsert(leaf, path)
        
        self._adjust_tree(path)
    
    def _choose_leaf(self, node, rect, path):
        """Choose leaf minimizing overlap enlargement"""
        path.append(node)
        
        if node.is_leaf:
            return node
        
        best_child = None
        best_cost = float('inf')
        
        for child_rect, child_node in node.entries:
            overlap_enlarge = self._overlap_enlargement(child_rect, rect, node)
            area_enlarge = child_rect.enlargement(rect)
            
            cost = (overlap_enlarge, area_enlarge, child_rect.area())
            
            if cost < best_cost:
                best_cost = cost
                best_child = child_node
        
        return self._choose_leaf(best_child, rect, path)
    
    def _overlap_enlargement(self, rect, new_rect, node):
        """Calculate overlap enlargement"""
        old_overlap = 0
        new_overlap = 0
        
        for other_rect, _ in node.entries:
            if other_rect == rect:
                continue
            old_overlap += self._compute_overlap(rect, other_rect)
            
            enlarged = Rectangle.union([rect, new_rect])
            new_overlap += self._compute_overlap(enlarged, other_rect)
        
        return new_overlap - old_overlap
    
    def _compute_overlap(self, r1, r2):
        """Compute overlap area between two rectangles"""
        if not r1.intersects(r2):
            return 0
        
        area = 1.0
        for i in range(r1.dim):
            overlap = min(r1.max[i], r2.max[i]) - max(r1.min[i], r2.min[i])
            area *= max(0, overlap)
        
        return area
    
    def _forced_reinsert(self, node, path):
        """Reinsert portion of entries to reduce overlap"""
        if not path:
            return
        
        # Sort by distance to center
        entries = node.entries
        center = self._compute_center(entries)
        
        entries.sort(key=lambda e: self._distance_to_center(e[0], center))
        
        # Remove top 30%
        reinsert = entries[int(len(entries) * 0.3):]
        node.entries = entries[:int(len(entries) * 0.3)]
        
        # Reinsert removed entries
        for rect, data in reinsert:
            self.insert(rect, data)
    
    def _compute_center(self, entries):
        """Compute center of all entries"""
        if not entries:
            return tuple([0] * 2)
        
        dim = entries[0][0].dim
        center = [0] * dim
        
        for rect, _ in entries:
            for i in range(dim):
                center[i] += (rect.min[i] + rect.max[i]) / 2
        
        return tuple(c / len(entries) for c in center)
    
    def _distance_to_center(self, rect, center):
        """Compute distance from rectangle center to given center"""
        sq_dist = 0
        for i in range(rect.dim):
            diff = (rect.min[i] + rect.max[i]) / 2 - center[i]
            sq_dist += diff ** 2
        return sq_dist
    
    def _adjust_tree(self, path):
        """Adjust MBRs up the tree"""
        for node in reversed(path):
            if node.entries:
                node.mbr = Rectangle.union([e[0] for e in node.entries])
```

**Why**: R* trees use sophisticated insertion strategies (forced reinsertion) to minimize overlap; provides 10-20% better query performance than standard R-trees.

```python
# PROBLEM: Metric space indexing for arbitrary distance functions
# TIME: O(log n) SPACE: O(n)
class MTreeNode:
    def __init__(self, is_leaf=True):
        self.is_leaf = is_leaf
        self.entries = []
        self.parent = None

class MTreeEntry:
    def __init__(self, obj=None, radius=0, child=None):

        self.radius = radius    # Covering radius
        self.child = child      # Child node (None for leaf)
        self.distance_to_parent = 0

class MTree:
    """M-tree for metric space indexing"""
    def __init__(self, distance_func):
        self.root = MTreeNode(is_leaf=True)
        self.distance = distance_func
        self.max_entries = 10
        self.min_entries = 2
    
    def insert(self, obj):
        """Insert object into M-tree"""
        if not self.root.entries:
            self.root.entries.append(MTreeEntry(obj=obj, radius=0))
            return
        
        self._insert_recursive(self.root, obj, None)
    
    def _insert_recursive(self, node, obj, parent_obj):
        """Recursive insertion"""
        if node.is_leaf:
            # Add to leaf
            entry = MTreeEntry(obj=obj, radius=0)
            entry.distance_to_parent = self.distance(obj, parent_obj) if parent_obj else 0
            node.entries.append(entry)
            
            if len(node.entries) > self.max_entries:
                self._split(node)
        else:
            # Choose best subtree
            best_entry = self._choose_subtree(node, obj)
            
            # Update covering radius if needed
            dist = self.distance(obj, best_entry.obj)
            if dist > best_entry.radius:
                best_entry.radius = dist
            
            self._insert_recursive(best_entry.child, obj, best_entry.obj)
    
    def _choose_subtree(self, node, obj):
        """Choose best subtree for insertion"""
        # Prefer no radius enlargement
        best = None
        best_dist = float('inf')
        
        for entry in node.entries:
            dist = self.distance(obj, entry.obj)
            
            if dist <= entry.radius:
                # Object fits in existing radius
                if best is None or dist < best_dist:
                    best = entry
                    best_dist = dist
        
        if best:
            return best
        
        # Choose entry with minimum radius enlargement
        min_enlarge = float('inf')
        for entry in node.entries:
            dist = self.distance(obj, entry.obj)
            enlarge = max(0, dist - entry.radius)
            
            if enlarge < min_enlarge:
                min_enlarge = enlarge
                best = entry
        
        return best
    
    def _split(self, node):
        """Split node using generalized hyperplane partitioning"""
        # Choose two promotion objects
        p1, p2 = self._choose_promotions(node.entries)
        
        # Create two new nodes
        left_node = MTreeNode(is_leaf=node.is_leaf)
        right_node = MTreeNode(is_leaf=node.is_leaf)
        
        # Partition entries
        for entry in node.entries:
            d1 = self.distance(entry.obj, p1.obj)
            d2 = self.distance(entry.obj, p2.obj)
            
            if d1 < d2:
                left_node.entries.append(entry)
            else:
                right_node.entries.append(entry)
        
        # Update routing objects
        p1.child = left_node
        p2.child = right_node
        
        # Update radii
        self._update_radius(p1, left_node)
        self._update_radius(p2, right_node)
        
        # Update parent
        if node == self.root:
            self.root = MTreeNode(is_leaf=False)
            self.root.entries = [p1, p2]
        else:
            # Replace old entry with new ones
            pass
    
    def _choose_promotions(self, entries):
        """Choose two objects to promote as routing objects"""
        # mM-RAD strategy: choose pair minimizing sum of covering radii
        best_pair = None
        best_sum = float('inf')
        
        for i in range(len(entries)):
            for j in range(i + 1, len(entries)):
                e1, e2 = entries[i], entries[j]
                
                # Compute covering radii
                r1, r2 = 0, 0
                for entry in entries:
                    d1 = self.distance(entry.obj, e1.obj)
                    d2 = self.distance(entry.obj, e2.obj)
                    r1 = max(r1, d1)
                    r2 = max(r2, d2)
                
                if r1 + r2 < best_sum:
                    best_sum = r1 + r2
                    best_pair = (
                        MTreeEntry(obj=e1.obj, radius=r1),
                        MTreeEntry(obj=e2.obj, radius=r2)
                    )
        
        return best_pair

        """Update covering radius"""
        max_dist = 0
        for child_entry in node.entries:
            dist = self.distance(entry.obj, child_entry.obj)
            max_dist = max(max_dist, dist + child_entry.radius)
        entry.radius = max_dist

        """Find all objects within radius of query object"""
        results = []
        self._range_search(self.root, query_obj, radius, 0, results)
        return results
    
    def _range_search(self, node, query_obj, radius, dist_to_parent, results):
        """Recursive range search"""
        for entry in node.entries:
            # Pruning using triangle inequality
            if abs(dist_to_parent - entry.distance_to_parent) > radius + entry.radius:
                continue
            
            dist = self.distance(query_obj, entry.obj)
            
            if node.is_leaf:
                if dist <= radius:
                    results.append(entry.obj)
            else:
                if dist <= radius + entry.radius:
                    self._range_search(entry.child, query_obj, radius, dist, results)
    
    def knn_query(self, query_obj, k):
        """K-nearest neighbor search"""
        import heapq
        
        heap = []  # (distance, node, dist_to_parent)
        results = []
        
        heapq.heappush(heap, (0, self.root, 0))
        
        while heap and len(results) < k:
            dist, node, dist_to_parent = heapq.heappop(heap)
            
            if node.is_leaf:
                for entry in node.entries:
                    actual_dist = self.distance(query_obj, entry.obj)
                    results.append((actual_dist, entry.obj))
                
                results.sort()
                results = results[:k]
            else:
                for entry in node.entries:
                    lower_bound = abs(dist_to_parent - entry.distance_to_parent)
                    
                    if lower_bound <= entry.radius:
                        heapq.heappush(heap, (lower_bound, entry.child,
                                            self.distance(query_obj, entry.obj)))
        
        return results[:k]
```

**Why**: M-trees index metric spaces (any distance function satisfying triangle inequality); used for similarity search, multimedia retrieval, and DNA sequence matching.
