```html
    <script>
        // PROBLEM: Remove whitespace from both ends of a string
        // TIME: O(n) SPACE: O(n)
        
        /**
         * Removes leading and trailing whitespace from a string.
         * The trim() method does not modify the original string.
         * 
         * @returns {string} A new string with whitespace removed from both ends
         */
        function trimText() {
            const input = document.getElementById('inputText');
            const result = document.getElementById('result');
            
            // Get the raw input value (may have leading/trailing spaces)
            const rawValue = input.value;
            
            // Apply trim() to remove whitespace from both ends
            const trimmedValue = rawValue.trim();
            
            // Display the result with visual indicators
            result.innerHTML = `
                <strong>Original:</strong> "${rawValue}" (${rawValue.length} chars)<br>
                <strong>Trimmed:</strong> "${trimmedValue}" (${trimmedValue.length} chars)
            `;
            
            // Why: Essential for sanitizing user input before validation or storage
            // Prevents issues with accidental spaces in form submissions
        }
        
        // Example usage:
        const testString = "   Hello World   ";
        console.log(testString.trim()); // "Hello World"
        console.log(testString); // "   Hello World   " (original unchanged)
        
        // trim() removes: spaces, tabs (\t), newlines (\n), carriage returns (\r)
        const multiline = "\n\n  Text  \n\n";
        console.log(multiline.trim()); // "Text"
    </script>
</body>

```
**Why:** Essential for sanitizing user input, preventing validation issues caused by accidental whitespace in form submissions.

```html
        // trimStart() is the preferred name per ECMAScript specification
```

```html
                </pre>
```

```html
4. innerText triggers reflow (slower)

            `;

            // Use textContent for performance or when hidden text matters
        }
        
        // Practical example: Copy visible text to clipboard

            const container = document.getElementById('container');
            const visibleText = container.innerText;
            
            try {

            } catch (err) {
                console.error('Copy failed:', err);
            }
        }
```

```html
        // PROBLEM: Get or replace an element including the element itself

        /**
         * outerHTML includes the element's own tag, unlike innerHTML.
         * Setting outerHTML completely replaces the element in the DOM.
         */
        function getOuter() {
            const card = document.getElementById('card1');

            // Get the complete HTML including the element itself

            `;
            
            // Why: Useful for serializing elements for storage or transfer
        }
        
        function replaceElement() {
            const card = document.getElementById('card1');
            
            // CAUTION: After this, the original element is gone!
            // Reference to 'card' becomes orphaned
            card.outerHTML = `
                <div id="card1" class="card" style="background: #d4edda;">
                    <h3>🔄 Replaced Card</h3>
                    <p>The entire element was replaced!</p>
                    <small>outerHTML replacement</small>
                </div>
            `;
            
            // The 'card' variable still holds reference to OLD element
            // which is no longer in the DOM
            console.log('Old reference still works:', card.tagName);
            console.log('But not in DOM:', document.contains(card)); // false
            
            // Why: Be careful - existing references become detached
        }
        
        function resetCard() {
            const output = document.getElementById('output');
            output.innerHTML = '';
            
            // Re-create original
            const container = output.previousElementSibling;
            if (!document.getElementById('card1')) {
                container.insertAdjacentHTML('beforebegin', `
                    <div id="card1" class="card">
                        <h3>Original Card</h3>
                        <p>Some content here</p>
                    </div>
                `);
            }
        }
        
        // Utility to escape HTML for display

            return str.replace(/&/g, '&amp;')
                      .replace(/</g, '&lt;')
                      .replace(/>/g, '&gt;');
        }

        // Never use with untrusted input!

    </script>
```

### js_element_createTextNode

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document createTextNode() Method</title>
    <style>
        .message { padding: 10px; margin: 5px 0; background: #f5f5f5; }
        .safe { border-left: 4px solid green; }
    </style>
</head>
<body>
    <h2>createTextNode - XSS-Safe Text Creation</h2>
    
    <input type="text" id="userInput" placeholder="Enter text (try HTML tags)" style="width: 300px;">
    <button onclick="addWithTextNode()">Add (Safe - createTextNode)</button>
    <button onclick="addWithInnerHTML()">Add (Unsafe - innerHTML)</button>
    <button onclick="clearMessages()">Clear</button>
    
    <div id="messages"></div>

    <script>
        // PROBLEM: Create a text node safely without HTML parsing
        // TIME: O(n) SPACE: O(n)
        
        /**
         * createTextNode creates a pure text node that cannot execute scripts.
         * This is the XSS-safe way to add user content to the DOM.
         */
        function addWithTextNode() {
            const input = document.getElementById('userInput');
            const messages = document.getElementById('messages');
            const value = input.value;
            
            // Create container for the message
            const div = document.createElement('div');
            div.className = 'message safe';
            
            // Create text node - completely safe from XSS
            const textNode = document.createTextNode(value);
            
            // Create label span
            const label = document.createElement('strong');
            label.textContent = 'Safe: ';
            
            // Append to DOM
            div.appendChild(label);
            div.appendChild(textNode);
            messages.appendChild(div);
            
            // Even if value is "<script>alert('xss')</script>",
            // it displays as literal text, not executed!
            
            // Why: createTextNode is inherently XSS-safe
            // The text is never parsed as HTML
        }
        
        function addWithInnerHTML() {
            const input = document.getElementById('userInput');
            const messages = document.getElementById('messages');
            const value = input.value;
            
            // DANGEROUS - can execute scripts in the HTML!
            const div = document.createElement('div');
            div.className = 'message';
            div.style.borderLeft = '4px solid red';
            
            // This is vulnerable to XSS attacks
            div.innerHTML = '<strong>Unsafe:</strong> ' + value;
            
            messages.appendChild(div);
            
            // If value contains <img src=x onerror=alert(1)>,
            // the script WILL execute!
        }
        
        function clearMessages() {
            document.getElementById('messages').innerHTML = '';
        }
        
        // Demonstration of text node properties
        const textNode = document.createTextNode('Hello World');
        console.log('Node type:', textNode.nodeType); // 3 (TEXT_NODE)
        console.log('Node name:', textNode.nodeName); // '#text'
        console.log('Node value:', textNode.nodeValue); // 'Hello World'
        
        // Text nodes can be split and merged
        const longText = document.createTextNode('OneTwoThree');
        longText.splitText(3); // Splits into 'One' and 'TwoThree'
        
        // Creating multiple text nodes is fine
        // They may be normalized (merged) later with element.normalize()
    </script>
</body>
</html>
```
**Why:** XSS-safe by design; text is never parsed as HTML. Essential for secure user input handling.

```html
        .bad { background: #ffcdd2; }
    </style>
</head>
<body>
    <h2>DocumentFragment - Batch DOM Updates</h2>
    
    <button onclick="addWithoutFragment()">Add 100 items (BAD - 100 reflows)</button>
    <button onclick="addWithFragment()">Add 100 items (GOOD - 1 reflow)</button>
    <button onclick="clearList()">Clear</button>
    
    <p id="timing"></p>
    <ul id="list"></ul>

    <script>
        // PROBLEM: Add multiple elements efficiently without multiple reflows
        // TIME: O(n) for n items, but only O(1) DOM reflows
        
        /**
         * DocumentFragment is a lightweight container for DOM nodes.
         * When appended to the DOM, the fragment disappears and only
         * its children are inserted - causing only ONE reflow.
         */
        
        // BAD APPROACH: Multiple DOM insertions
        function addWithoutFragment() {
            const list = document.getElementById('list');
            const timing = document.getElementById('timing');
            list.innerHTML = '';
            
            const start = performance.now();
            
            // Each appendChild triggers a reflow!
            for (let i = 0; i < 100; i++) {
                const li = document.createElement('li');
                li.className = 'item bad';
                li.textContent = `Item ${i + 1} - Multiple reflows`;
                list.appendChild(li); // Triggers reflow EVERY time
            }
            
            const duration = performance.now() - start;
```
