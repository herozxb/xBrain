## 1. Disjoint Set with Path Compression and Union by Rank

```python
# PROBLEM: Efficiently manage a collection of disjoint sets with union and find operations
# TIME: O(α(n)) amortized per operation SPACE: O(n)
class DisjointSet:
    """Disjoint Set Union (DSU) with path compression and union by rank.
    
    Provides near-constant time operations for dynamic connectivity problems.
    """
    
    def __init__(self, n: int):
        # Each element starts as its own parent (singleton set)
        self.parent = list(range(n))
        # Rank approximates tree height for balanced merging
        self.rank = [0] * n
        # Track number of disjoint sets
        self.num_sets = n
    
    def find(self, x: int) -> int:
        """Find the representative (root) of the set containing x.
        Uses path compression to flatten the tree structure.
        """
        if self.parent[x] != x:
            # Path compression: point directly to root
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x: int, y: int) -> bool:
        """Merge the sets containing x and y.
        Returns True if merge happened, False if already in same set.
        Uses union by rank to keep trees shallow.
        """
        root_x = self.find(x)
        root_y = self.find(y)
        
        # Already in the same set
        if root_x == root_y:
            return False
        
        # Union by rank: attach smaller tree under larger
        if self.rank[root_x] < self.rank[root_y]:
            root_x, root_y = root_y, root_x
        
        self.parent[root_y] = root_x
        if self.rank[root_x] == self.rank[root_y]:
            self.rank[root_x] += 1
        
        self.num_sets -= 1
        return True
    
    def connected(self, x: int, y: int) -> bool:
        """Check if x and y are in the same set."""
        return self.find(x) == self.find(y)
    
    def count_sets(self) -> int:
        """Return number of disjoint sets."""
        return self.num_sets
```
**Why**: DSU with path compression and union by rank achieves near-constant amortized time, making it ideal for Kruskal's MST, dynamic connectivity, and network problems.

## 2. Hash Table with Linear Probing

```python
# PROBLEM: Implement a hash table using open addressing with linear probing
# TIME: O(1) average, O(n) worst case SPACE: O(n)
class LinearProbingHashTable:
    """Hash table using linear probing for collision resolution.
    
    When collision occurs, probe next slot sequentially until empty slot found.
    Simple but suffers from primary clustering.
    """
    
    def __init__(self, capacity: int = 16):
        self.capacity = capacity
        self.size = 0
        # None means empty, special DELETED marker for tombstones
        self.DELETED = object()
        self.keys = [None] * capacity
        self.values = [None] * capacity
    
    def _hash(self, key) -> int:
        """Compute hash index for key."""
        return hash(key) % self.capacity
    
    def _probe(self, key) -> int:
        """Find the appropriate slot for key using linear probing.
        Returns index where key exists or first empty/deleted slot.
        """
        idx = self._hash(key)
        first_deleted = -1
        
        for _ in range(self.capacity):
            if self.keys[idx] is None:
                # Empty slot found
                return first_deleted if first_deleted != -1 else idx
            elif self.keys[idx] is self.DELETED:
                # Remember first deleted slot
                if first_deleted == -1:
                    first_deleted = idx
            elif self.keys[idx] == key:
                # Key found
                return idx
            # Linear probe: move to next slot
            idx = (idx + 1) % self.capacity
        
        # Table full, return first deleted slot if exists
        return first_deleted if first_deleted != -1 else -1
    
    def put(self, key, value) -> None:
        """Insert or update key-value pair."""
        if self.size >= self.capacity * 0.7:
            self._resize()
        
        idx = self._probe(key)
        if idx == -1:
            raise RuntimeError("Hash table is full")
        
        if self.keys[idx] is None or self.keys[idx] is self.DELETED:
            self.size += 1
        
        self.keys[idx] = key
        self.values[idx] = value
    
    def get(self, key, default=None):
        """Retrieve value for key, return default if not found."""
        idx = self._hash(key)
        
        for _ in range(self.capacity):
            if self.keys[idx] is None:
                return default
            if self.keys[idx] == key:
                return self.values[idx]
            idx = (idx + 1) % self.capacity
        
        return default
    
    def remove(self, key) -> bool:
        """Remove key-value pair, return True if removed."""
        idx = self._hash(key)
        
        for _ in range(self.capacity):
            if self.keys[idx] is None:
                return False
            if self.keys[idx] == key:
                # Use tombstone to maintain probe chain
                self.keys[idx] = self.DELETED
                self.values[idx] = None
                self.size -= 1
                return True
            idx = (idx + 1) % self.capacity
        
        return False
    
    def _resize(self) -> None:
        """Double capacity and rehash all entries."""
        old_keys = self.keys
        old_values = self.values
        old_capacity = self.capacity
        
        self.capacity *= 2
        self.keys = [None] * self.capacity
        self.values = [None] * self.capacity
        self.size = 0
        
        for i in range(old_capacity):
            if old_keys[i] is not None and old_keys[i] is not self.DELETED:
                self.put(old_keys[i], old_values[i])
```
**Why**: Linear probing offers cache-friendly access patterns and simple implementation. Understanding tombstones and clustering helps when designing hash tables for specific workloads.

## 3. Top K Frequent Elements

```python
# PROBLEM: Find the k most frequent elements in an array
# TIME: O(n log k) using heap, O(n) using quickselect SPACE: O(n)
import heapq
from collections import Counter
from typing import List

def top_k_frequent_heap(nums: List[int], k: int) -> List[int]:
    """Find k most frequent elements using min-heap.
    
    Build frequency map, then maintain heap of size k with top elements.
    """
    # Count frequencies
    freq = Counter(nums)
    
    # Use min-heap to track top k elements
    # Heap stores (frequency, element) tuples
    heap = []
    
    for num, count in freq.items():
        # Push current element
        heapq.heappush(heap, (count, num))
        # Keep only k largest by frequency
        if len(heap) > k:
            heapq.heappop(heap)
    
    # Extract elements from heap
    return [num for count, num in heap]


def top_k_frequent_bucket(nums: List[int], k: int) -> List[int]:
    """Find k most frequent elements using bucket sort.
    
    O(n) time when frequencies are bounded by n.
    """
    freq = Counter(nums)
    
    # Create buckets: index = frequency, value = list of elements
    # Maximum frequency is n (all elements same)
    buckets = [[] for _ in range(len(nums) + 1)]
    
    for num, count in freq.items():
        buckets[count].append(num)
    
    # Collect from highest frequency buckets
    result = []
    for i in range(len(buckets) - 1, 0, -1):
        result.extend(buckets[i])
        if len(result) >= k:
            return result[:k]
    
    return result


def top_k_frequent_quickselect(nums: List[int], k: int) -> List[int]:
    """Find k most frequent elements using quickselect.
    
    Average O(n) time, worst case O(n^2).
    """
    freq = Counter(nums)
    unique = list(freq.keys())
    
    def partition(left: int, right: int, pivot_idx: int) -> int:
        """Partition around pivot frequency, return final pivot position."""
        pivot_freq = freq[unique[pivot_idx]]
        # Move pivot to end
        unique[pivot_idx], unique[right] = unique[right], unique[pivot_idx]
        
        store_idx = left
        for i in range(left, right):
            if freq[unique[i]] > pivot_freq:  # Descending order
                unique[store_idx], unique[i] = unique[i], unique[store_idx]
                store_idx += 1
        
        # Move pivot to final position
        unique[right], unique[store_idx] = unique[store_idx], unique[right]
        return store_idx
    
    def select(left: int, right: int, k_smallest: int) -> None:
        """Select k_smallest elements with highest frequency."""
        if left >= right:
            return
        
        pivot_idx = partition(left, right, (left + right) // 2)
        
        if k_smallest == pivot_idx:
            return
        elif k_smallest < pivot_idx:
            select(left, pivot_idx - 1, k_smallest)
        else:
            select(pivot_idx + 1, right, k_smallest)
    
    # We want top k, so k_smallest = k - 1 (0-indexed)
    select(0, len(unique) - 1, k - 1)
    return unique[:k]
```
**Why**: Top K frequent is a classic problem with multiple approaches. Bucket sort gives O(n) for bounded frequencies, heap gives O(n log k) for arbitrary data, and quickselect offers average O(n).

## 4. Specification Pattern

```python
# PROBLEM: Encapsulate business rules for flexible and composable filtering
# TIME: O(n) for filtering SPACE: O(1) per specification
from abc import ABC, abstractmethod
from typing import TypeVar, Generic, List, Callable

T = TypeVar('T')

class Specification(ABC, Generic[T]):
    """Base specification interface for composable business rules."""
    
    @abstractmethod
    def is_satisfied_by(self, candidate: T) -> bool:
        """Check if candidate satisfies this specification."""
        pass
    
    def and_(self, other: 'Specification[T]') -> 'Specification[T]':
        """Combine specifications with logical AND."""
        return AndSpecification(self, other)
    
    def or_(self, other: 'Specification[T]') -> 'Specification[T]':
        """Combine specifications with logical OR."""
        return OrSpecification(self, other)
    
    def not_(self) -> 'Specification[T]':
        """Negate this specification."""
        return NotSpecification(self)


class AndSpecification(Specification[T]):
    """Specification that requires both conditions to be true."""
    
    def __init__(self, left: Specification[T], right: Specification[T]):
        self.left = left
        self.right = right
    
    def is_satisfied_by(self, candidate: T) -> bool:
        return self.left.is_satisfied_by(candidate) and self.right.is_satisfied_by(candidate)


class OrSpecification(Specification[T]):
    """Specification that requires either condition to be true."""
    
    def __init__(self, left: Specification[T], right: Specification[T]):
        self.left = left
        self.right = right
    
    def is_satisfied_by(self, candidate: T) -> bool:
        return self.left.is_satisfied_by(candidate) or self.right.is_satisfied_by(candidate)


class NotSpecification(Specification[T]):
    """Specification that negates another specification."""
    
    def __init__(self, spec: Specification[T]):
        self.spec = spec
    
    def is_satisfied_by(self, candidate: T) -> bool:
        return not self.spec.is_satisfied_by(candidate)


# Example: E-commerce product filtering
class Product:
    def __init__(self, name: str, price: float, category: str, in_stock: bool):
        self.name = name
        self.price = price
        self.category = category
        self.in_stock = in_stock


class PriceRangeSpecification(Specification[Product]):
    """Product price within specified range."""
    
    def __init__(self, min_price: float, max_price: float):
        self.min_price = min_price
        self.max_price = max_price
    
    def is_satisfied_by(self, product: Product) -> bool:
        return self.min_price <= product.price <= self.max_price


class InStockSpecification(Specification[Product]):
    """Product must be in stock."""
    
    def is_satisfied_by(self, product: Product) -> bool:
        return product.in_stock


class CategorySpecification(Specification[Product]):
    """Product must be in specified category."""
    
    def __init__(self, category: str):
        self.category = category
    
    def is_satisfied_by(self, product: Product) -> bool:
        return product.category == self.category


class ProductFilter:
    """Filter products using specifications."""
    
    @staticmethod
    def filter(products: List[Product], spec: Specification[Product]) -> List[Product]:
        return [p for p in products if spec.is_satisfied_by(p)]


# Usage example
if __name__ == "__main__":
    products = [
        Product("Laptop", 999.99, "Electronics", True),
        Product("Mouse", 29.99, "Electronics", True),
        Product("Desk", 199.99, "Furniture", False),
    ]
    
    # Complex specification: electronics in stock under $100
    spec = (
        CategorySpecification("Electronics")
        .and_(InStockSpecification())
        .and_(PriceRangeSpecification(0, 100))
    )
    
    filtered = ProductFilter.filter(products, spec)
    # Returns: [Mouse]
```
**Why**: Specification pattern enables composable, reusable business rules. Great for complex filtering logic, validation, and domain rules that need to be combined in various ways.

## 5. Fallback Pattern (Circuit Breaker with Fallback)

```python
# PROBLEM: Provide graceful degradation when services fail
# TIME: O(1) for state checks SPACE: O(1)
import time
from enum import Enum
from typing import Callable, TypeVar, Optional
from functools import wraps

T = TypeVar('T')

class CircuitState(Enum):
    CLOSED = "closed"      # Normal operation
    OPEN = "open"          # Failing, reject all calls
    HALF_OPEN = "half_open"  # Testing if recovered


class CircuitBreakerWithFallback:
    """Circuit breaker with automatic fallback on failures.
    
    Prevents cascading failures by failing fast and providing fallbacks.
    """
    
    def __init__(
        self,
        failure_threshold: int = 5,
        recovery_timeout: float = 30.0,
        success_threshold: int = 3
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.success_threshold = success_threshold
        
        self.failure_count = 0
        self.success_count = 0
        self.state = CircuitState.CLOSED
        self.last_failure_time = 0.0
    
    def _should_allow_request(self) -> bool:
        """Check if request should be allowed through."""
        if self.state == CircuitState.CLOSED:
            return True
        
        if self.state == CircuitState.OPEN:
            # Check if recovery timeout has passed
            if time.time() - self.last_failure_time >= self.recovery_timeout:
                self.state = CircuitState.HALF_OPEN
                self.success_count = 0
                return True
            return False
        
        # HALF_OPEN: allow limited requests
        return True
    
    def _record_success(self) -> None:
        """Record successful call."""
        self.failure_count = 0
        
        if self.state == CircuitState.HALF_OPEN:
            self.success_count += 1
            if self.success_count >= self.success_threshold:
                self.state = CircuitState.CLOSED
    
    def _record_failure(self) -> None:
        """Record failed call."""
        self.failure_count += 1
        self.last_failure_time = time.time()
        
        if self.state == CircuitState.HALF_OPEN:
            self.state = CircuitState.OPEN
        elif self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
    
    def call(
        self,
        primary: Callable[[], T],
        fallback: Optional[Callable[[], T]] = None,
        fallback_on_exception: tuple = (Exception,)
    ) -> T:
        """Execute primary function with fallback on failure.
        
        Args:
            primary: Main function to try
            fallback: Backup function if primary fails
            fallback_on_exception: Exception types that trigger fallback
        """
        if not self._should_allow_request():
            if fallback:
                return fallback()
            raise Exception("Circuit breaker is open")
        
        try:
            result = primary()
            self._record_success()
            return result
        except fallback_on_exception as e:
            self._record_failure()
            if fallback:
                return fallback()
            raise


def with_fallback(
    fallback_func: Callable = None,
    failure_threshold: int = 5,
    recovery_timeout: float = 30.0
):
    """Decorator to add circuit breaker with fallback to any function."""
    
    circuit_breaker = CircuitBreakerWithFallback(
        failure_threshold=failure_threshold,
        recovery_timeout=recovery_timeout
    )
    
    def decorator(func: Callable[..., T]) -> Callable[..., T]:
        @wraps(func)
        def wrapper(*args, **kwargs) -> T:
            return circuit_breaker.call(
                primary=lambda: func(*args, **kwargs),
                fallback=lambda: fallback_func(*args, **kwargs) if fallback_func else None
            )
        return wrapper
    
    return decorator


# Example usage
class PaymentService:
    """Example service with primary and fallback payment processors."""
    
    def __init__(self):
        self.circuit = CircuitBreakerWithFallback(
            failure_threshold=3,
            recovery_timeout=60.0
        )
    
    def process_payment(self, amount: float) -> dict:
        """Process payment with fallback to backup processor."""
        return self.circuit.call(
            primary=lambda: self._primary_processor(amount),
            fallback=lambda: self._backup_processor(amount)
        )
    
    def _primary_processor(self, amount: float) -> dict:
        """Primary payment gateway (may fail)."""
        # Simulate occasional failures
        import random
        if random.random() < 0.3:  # 30% failure rate
            raise ConnectionError("Primary gateway unavailable")
        return {"processor": "primary", "amount": amount, "status": "success"}
    
    def _backup_processor(self, amount: float) -> dict:
        """Backup payment gateway."""
        return {"processor": "backup", "amount": amount, "status": "success"}
```
**Why**: Fallback pattern combined with circuit breaker provides resilience. When services fail, users get degraded but functional behavior instead of complete outages.

## 6. Least Connections Load Balancer

```python
# PROBLEM: Distribute requests to servers based on current connection count
# TIME: O(n) for server selection, O(1) for updates SPACE: O(n)
import time
from typing import List, Optional
from dataclasses import dataclass, field
import threading

@dataclass
class Server:
    """Represents a backend server."""
    name: str
    address: str
    weight: int = 1
    connections: int = 0
    healthy: bool = True
    last_health_check: float = field(default_factory=time.time)


class LeastConnectionsLoadBalancer:
    """Load balancer using least connections algorithm.
    
    Routes new requests to the server with fewest active connections.
    Supports weighted servers and health checking.
    """
    
    def __init__(self, health_check_interval: float = 30.0):
        self.servers: List[Server] = []
        self.health_check_interval = health_check_interval
        self._lock = threading.Lock()
    
    def add_server(self, name: str, address: str, weight: int = 1) -> None:
        """Add a new server to the pool."""
        with self._lock:
            self.servers.append(Server(
                name=name,
                address=address,
                weight=weight
            ))
    
    def remove_server(self, name: str) -> bool:
        """Remove a server from the pool."""
        with self._lock:
            for i, server in enumerate(self.servers):
                if server.name == name:
                    self.servers.pop(i)
                    return True
            return False
    
    def select_server(self) -> Optional[Server]:
        """Select server with least connections (weighted).
        
        Uses weighted least connections: connections / weight
        to account for server capacity differences.
        """
        with self._lock:
            # Filter healthy servers
            healthy_servers = [s for s in self.servers if s.healthy]
            
            if not healthy_servers:
                return None
            
            # Find server with minimum weighted connection ratio
            best_server = min(
                healthy_servers,
                key=lambda s: s.connections / s.weight
            )
            
            # Increment connection count
            best_server.connections += 1
            return best_server
    
    def release_connection(self, server_name: str) -> None:
        """Decrement connection count when request completes."""
        with self._lock:
            for server in self.servers:
                if server.name == server_name:
                    server.connections = max(0, server.connections - 1)
                    break
    
    def mark_unhealthy(self, server_name: str) -> None:
        """Mark server as unhealthy (won't receive new connections)."""
        with self._lock:
            for server in self.servers:
                if server.name == server_name:
                    server.healthy = False
                    server.last_health_check = time.time()
                    break
    
    def mark_healthy(self, server_name: str) -> None:
        """Mark server as healthy again."""
        with self._lock:
            for server in self.servers:
                if server.name == server_name:
                    server.healthy = True
                    server.last_health_check = time.time()
                    break
    
    def get_stats(self) -> List[dict]:
        """Get statistics for all servers."""
        with self._lock:
            return [
                {
                    "name": s.name,
                    "address": s.address,
                    "connections": s.connections,
                    "weight": s.weight,
                    "healthy": s.healthy,
                    "load_ratio": s.connections / s.weight if s.weight > 0 else float('inf')
                }
                for s in self.servers
            ]


class WeightedLeastConnectionsLB(LeastConnectionsLoadBalancer):
    """Extended load balancer with smooth weighted round-robin fallback."""
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._current_weights = {}  # Track smooth weights
    
    def select_server_smooth(self) -> Optional[Server]:
        """Select server using smooth weighted least connections.
        
        Combines least connections with smooth weighted selection
        for better distribution when connections are equal.
        """
        with self._lock:
            healthy_servers = [s for s in self.servers if s.healthy]
            
            if not healthy_servers:
                return None
            
            # Calculate effective weight considering connections
            for s in healthy_servers:
                if s.name not in self._current_weights:
                    self._current_weights[s.name] = 0
            
            # Increase current weight by static weight
            for s in healthy_servers:
                self._current_weights[s.name] += s.weight
            
            # Select server with highest current weight minus connections penalty
            best_server = max(
                healthy_servers,
                key=lambda s: self._current_weights[s.name] - s.connections
            )
            
            # Reduce selected server's current weight by total weight sum
            total_weight = sum(s.weight for s in healthy_servers)
            self._current_weights[best_server.name] -= total_weight
            
            best_server.connections += 1
            return best_server
```
**Why**: Least connections load balancing adapts to varying request processing times. Combined with weights, it optimally distributes load across heterogeneous server pools.

## 7. Graceful Degradation with Feature Toggles

```python
# PROBLEM: Implement graceful degradation when dependencies are unavailable
# TIME: O(1) for feature checks SPACE: O(n) for feature states
from enum import Enum
from typing import Dict, Callable, Any, Optional
from dataclasses import dataclass
import time

class FeatureState(Enum):
    ENABLED = "enabled"
    DISABLED = "disabled"
    DEGRADED = "degraded"  # Partial functionality


@dataclass
class FeatureConfig:
    """Configuration for a degradable feature."""
    name: str
    state: FeatureState = FeatureState.ENABLED
    degradation_level: int = 0  # 0 = full, higher = more degraded
    fallback_handler: Optional[Callable] = None
    last_state_change: float = 0.0
    error_count: int = 0
    error_threshold: int = 5


class GracefulDegradationManager:
    """Manages feature states and graceful degradation.
    
    Automatically degrades features based on error rates and
    provides fallback implementations.
    """
    
    def __init__(self):
        self.features: Dict[str, FeatureConfig] = {}
        self._global_degradation_level = 0
    
    def register_feature(
        self,
        name: str,
        fallback_handler: Callable = None,
        error_threshold: int = 5
    ) -> None:
        """Register a feature that supports degradation."""
        self.features[name] = FeatureConfig(
            name=name,
            fallback_handler=fallback_handler,
            error_threshold=error_threshold
        )
    
    def is_enabled(self, feature_name: str) -> bool:
        """Check if feature is available (not disabled)."""
        if feature_name not in self.features:
            return False
        return self.features[feature_name].state != FeatureState.DISABLED
    
    def get_degradation_level(self, feature_name: str) -> int:
        """Get current degradation level for feature."""
        if feature_name not in self.features:
            return float('inf')
        return self.features[feature_name].degradation_level
    
    def execute_with_fallback(
        self,
        feature_name: str,
        primary_func: Callable,
        *args,
        **kwargs
    ) -> Any:
        """Execute function with automatic fallback on failure."""
        config = self.features.get(feature_name)
        
        if not config:
            raise ValueError(f"Unknown feature: {feature_name}")
        
        # Feature disabled - use fallback immediately
        if config.state == FeatureState.DISABLED:
            if config.fallback_handler:
                return config.fallback_handler(*args, **kwargs)
            raise Exception(f"Feature {feature_name} is disabled")
        
        # Feature degraded - optionally use fallback
        if config.state == FeatureState.DEGRADED and config.fallback_handler:
            return config.fallback_handler(*args, **kwargs)
        
        # Try primary function
        try:
            result = primary_func(*args, **kwargs)
            # Success - reset error count
            config.error_count = 0
            return result
        except Exception as e:
            config.error_count += 1
            config.last_state_change = time.time()
            
            # Auto-degrade if threshold reached
            if config.error_count >= config.error_threshold:
                self._degrade_feature(feature_name)
            
            # Use fallback if available
            if config.fallback_handler:
                return config.fallback_handler(*args, **kwargs)
            raise
    
    def _degrade_feature(self, feature_name: str) -> None:
        """Increase degradation level for feature."""
        config = self.features[feature_name]
        config.degradation_level += 1
        config.state = FeatureState.DEGRADED
    
    def enable_feature(self, feature_name: str) -> None:
```

## 8. Database Read Committed Isolation Level Simulation

```python
# PROBLEM: Simulate READ COMMITTED isolation level behavior
# TIME: O(1) for locks, O(n) for version chains SPACE: O(n) for versions
import threading
from typing import Dict, Any, Optional
from dataclasses import dataclass, field
from time import time
from enum import Enum

class LockType(Enum):
    SHARED = "shared"    # Read lock
    EXCLUSIVE = "exclusive"  # Write lock


@dataclass
class RowVersion:
    """Represents a version of a row for MVCC."""
    data: Dict[str, Any]
    commit_time: float
    created_by_tx: int


@dataclass
class Lock:
    """Represents a lock on a row."""
    lock_type: LockType
    transaction_id: int
    acquired_at: float = field(default_factory=time)


class ReadCommittedMVCC:
    """Simulates READ COMMITTED isolation using MVCC.
    
    READ COMMITTED guarantees:
    1. Only committed data is visible
    2. No dirty reads
    3. Non-repeatable reads possible (same query may see different data)
    """
    
    def __init__(self):
        # Row ID -> list of versions (newest first)
        self.versions: Dict[int, list[RowVersion]] = {}
        # Row ID -> lock
        self.locks: Dict[int, Lock] = {}
        # Current committed timestamp
        self.current_time = 0.0
        self._lock = threading.Lock()
        self._next_tx_id = 1
    
    def begin_transaction(self) -> int:
        """Start a new transaction."""
        with self._lock:
            tx_id = self._next_tx_id
            self._next_tx_id += 1
            return tx_id
    
    def read(self, tx_id: int, row_id: int) -> Optional[Dict[str, Any]]:
        """Read a row - sees only committed versions.
        
        In READ COMMITTED, each statement sees the latest committed version
        at the time it executes (not transaction start).
        """
        with self._lock:
            if row_id not in self.versions or not self.versions[row_id]:
                return None
            
            # Get the most recent committed version
            for version in self.versions[row_id]:
                # READ COMMITTED: see any committed version
                # (not just those before transaction start)
                if version.commit_time < float('inf'):
                    return version.data.copy()
            
            return None
    
    def write(
        self,
        tx_id: int,
        row_id: int,
        data: Dict[str, Any]
    ) -> bool:
        """Write to a row - creates uncommitted version."""
        with self._lock:
            # Check for conflicting locks
            if row_id in self.locks:
                existing = self.locks[row_id]
                if existing.transaction_id != tx_id:
                    if existing.lock_type == LockType.EXCLUSIVE:
                        # Blocked by exclusive lock
                        return False
            
            # Acquire exclusive lock
            self.locks[row_id] = Lock(
                lock_type=LockType.EXCLUSIVE,
                transaction_id=tx_id
            )
            
            # Create new uncommitted version
            new_version = RowVersion(
                data=data.copy(),
                commit_time=float('inf'),  # Uncommitted
                created_by_tx=tx_id
            )
            
            if row_id not in self.versions:
                self.versions[row_id] = []
            
            self.versions[row_id].insert(0, new_version)
            return True
    
    def commit(self, tx_id: int) -> bool:
        """Commit transaction - make all its changes visible."""
        with self._lock:
            commit_time = time()
            committed_count = 0
            
            for row_id, version_list in self.versions.items():
                for version in version_list:
                    if version.created_by_tx == tx_id and version.commit_time == float('inf'):
                        version.commit_time = commit_time
                        committed_count += 1
            
            # Release locks held by this transaction
            to_release = [
                row_id for row_id, lock in self.locks.items()
                if lock.transaction_id == tx_id
            ]
            for row_id in to_release:
                del self.locks[row_id]
            
            return committed_count > 0
    
    def rollback(self, tx_id: int) -> None:
        """Rollback transaction - discard uncommitted changes."""
        with self._lock:
            # Remove uncommitted versions
            for row_id in list(self.versions.keys()):
                self.versions[row_id] = [
                    v for v in self.versions[row_id]
                    if not (v.created_by_tx == tx_id and v.commit_time == float('inf'))
                ]
            
            # Release locks
            to_release = [
                row_id for row_id, lock in self.locks.items()
                if lock.transaction_id == tx_id
            ]
            for row_id in to_release:
                del self.locks[row_id]
    
    def demonstrate_non_repeatable_read(self) -> None:
        """Demonstrate non-repeatable read in READ COMMITTED."""
        print("=== Demonstrating Non-Repeatable Read ===")
        
        # Setup: insert initial row
        tx1 = self.begin_transaction()
        self.write(tx1, 1, {"value": 100})
        self.commit(tx1)
        
        # Transaction 2 reads the row
        tx2 = self.begin_transaction()
        v1 = self.read(tx2, 1)
        print(f"First read: {v1}")  # {value: 100}
        
        # Transaction 3 modifies and commits
        tx3 = self.begin_transaction()
        self.write(tx3, 1, {"value": 200})
        self.commit(tx3)
        
        # Transaction 2 reads again - sees different value!
        v2 = self.read(tx2, 1)
        print(f"Second read: {v2}")  # {value: 200}
        
        print("Non-repeatable read occurred: same tx saw different values")
```
**Why**: Understanding isolation levels is crucial for database design. READ COMMITTED prevents dirty reads but allows non-repeatable reads, balancing consistency with concurrency.

## 9. Serializable Isolation Level with Two-Phase Locking

```python
# PROBLEM: Implement SERIALIZABLE isolation using strict two-phase locking
# TIME: O(1) for lock operations SPACE: O(n) for lock table
import threading
from typing import Dict, Set, Optional
from dataclasses import dataclass, field
from enum import Enum
from collections import defaultdict

class LockMode(Enum):
    SHARED = "S"      # Read lock
    EXCLUSIVE = "X"   # Write lock


@dataclass
class LockEntry:
    """Lock entry tracking holders and waiters."""
    shared_holders: Set[int] = field(default_factory=set)
    exclusive_holder: Optional[int] = None
    waiters: list = field(default_factory=list)


class Serializable2PL:
    """Serializable isolation using Strict Two-Phase Locking (2PL).
    
    Rules of Strict 2PL:
    1. Acquire shared lock before reading
    2. Acquire exclusive lock before writing
    3. Hold all locks until transaction ends
    4. Upgrading locks requires care to prevent deadlock
    
    Guarantees:
    - No dirty reads
    - No non-repeatable reads
    - No phantom reads
    - Serializability
    """
    
    def __init__(self):
        self.lock_table: Dict[str, LockEntry] = defaultdict(LockEntry)
        self.tx_locks: Dict[int, Set[str]] = defaultdict(set)  # tx -> resources
        self._lock = threading.Lock()
        self._next_tx_id = 1
    
    def begin_transaction(self) -> int:
        """Start new transaction."""
        with self._lock:
            tx_id = self._next_tx_id
            self._next_tx_id += 1
            return tx_id
    
    def acquire_shared_lock(self, tx_id: int, resource: str) -> bool:
        """Acquire shared (read) lock on resource."""
        with self._lock:
            entry = self.lock_table[resource]
            
            # Already hold exclusive lock
            if entry.exclusive_holder == tx_id:
                return True
            
            # Check for conflict with exclusive lock
            if entry.exclusive_holder is not None:
                # Wait or fail (simplified: fail)
                return False
            
            # Check for conflicting exclusive waiters
            for waiter_id, mode in entry.waiters:
                if mode == LockMode.EXCLUSIVE and waiter_id != tx_id:
                    return False  # Prevent writer starvation
            
            # Acquire shared lock
            entry.shared_holders.add(tx_id)
            self.tx_locks[tx_id].add(resource)
            return True
    
    def acquire_exclusive_lock(self, tx_id: int, resource: str) -> bool:
        """Acquire exclusive (write) lock on resource."""
        with self._lock:
            entry = self.lock_table[resource]
            
            # Already hold exclusive lock
            if entry.exclusive_holder == tx_id:
                return True
            
            # Upgrade from shared to exclusive
            if tx_id in entry.shared_holders:
                # Can upgrade only if only holder
                if len(entry.shared_holders) == 1:
                    entry.shared_holders.remove(tx_id)
                    entry.exclusive_holder = tx_id
                    return True
                return False  # Other shared holders exist
            
            # Check for conflicts
            if entry.exclusive_holder is not None:
                return False
            if entry.shared_holders:
                return False
            
            # Acquire exclusive lock
            entry.exclusive_holder = tx_id
            self.tx_locks[tx_id].add(resource)
            return True
    
    def release_all_locks(self, tx_id: int) -> None:
        """Release all locks held by transaction (at commit/abort)."""
        with self._lock:
            for resource in self.tx_locks[tx_id]:
                entry = self.lock_table[resource]
                
                if entry.exclusive_holder == tx_id:
                    entry.exclusive_holder = None
                elif tx_id in entry.shared_holders:
                    entry.shared_holders.discard(tx_id)
                
                # Wake up waiters (simplified)
                if entry.waiters:
                    waiter_id, mode = entry.waiters.pop(0)
                    if mode == LockMode.EXCLUSIVE:
                        entry.exclusive_holder = waiter_id
                    else:
                        entry.shared_holders.add(waiter_id)
            
            del self.tx_locks[tx_id]


class SerializableDatabase:
    """Database with serializable isolation using 2PL."""
    
    def __init__(self):
        self.lock_manager = Serializable2PL()
        self.data: Dict[str, any] = {}
    
    def read(self, tx_id: int, key: str) -> Optional[any]:
        """Read value with shared lock."""
        if self.lock_manager.acquire_shared_lock(tx_id, key):
            return self.data.get(key)
        raise Exception(f"Could not acquire lock for read: {key}")
    
    def write(self, tx_id: int, key: str, value: any) -> None:
        """Write value with exclusive lock."""
        if self.lock_manager.acquire_exclusive_lock(tx_id, key):
            self.data[key] = value
            return
        raise Exception(f"Could not acquire lock for write: {key}")
    
    def commit(self, tx_id: int) -> None:
        """Commit transaction - release all locks."""
        self.lock_manager.release_all_locks(tx_id)
    
    def rollback(self, tx_id: int) -> None:
        """Rollback transaction - release all locks."""
        self.lock_manager.release_all_locks(tx_id)
    
    def demonstrate_serializability(self) -> None:
        """Show how serializable prevents non-repeatable reads."""
        print("=== Serializable: No Non-Repeatable Reads ===")
        
        # Setup
        tx0 = self.lock_manager.begin_transaction()
        self.write(tx0, "balance", 100)
        self.commit(tx0)
        
        # Transaction 1 reads
        tx1 = self.lock_manager.begin_transaction()
        v1 = self.read(tx1, "balance")
        print(f"TX1 first read: {v1}")  # 100
        
        # Transaction 2 tries to modify (will block)
        tx2 = self.lock_manager.begin_transaction()
        can_write = self.lock_manager.acquire_exclusive_lock(tx2, "balance")
        print(f"TX2 can write while TX1 holds shared: {can_write}")  # False
        
        # Transaction 1 reads again - same value
        v2 = self.read(tx1, "balance")
        print(f"TX1 second read: {v2}")  # 100 (same!)
        
        self.commit(tx1)
        print("TX1 committed, locks released")
        
        # Now TX2 can proceed
        can_write = self.lock_manager.acquire_exclusive_lock(tx2, "balance")
        print(f"TX2 can now write: {can_write}")  # True
        self.commit(tx2)
```
**Why**: Serializable is the strongest isolation level, guaranteeing that concurrent transactions appear sequential. 2PL is a classic implementation technique but can cause deadlocks and reduced concurrency.

```python
# PROBLEM: Maintain materialized views efficiently when base tables change
# TIME: O(n) for view refresh SPACE: O(n) for stored view
from typing import Dict, List, Callable, Any, Set
from dataclasses import dataclass, field
from enum import Enum
import time

class RefreshType(Enum):
    FULL = "full"        # Recompute entire view
    INCREMENTAL = "incremental"  # Apply only changes


@dataclass
class TableDelta:
    """Represents changes to a base table."""
    inserts: List[Dict] = field(default_factory=list)
    deletes: List[Dict] = field(default_factory=list)
    updates: List[tuple] = field(default_factory=list)  # (old, new)


@dataclass
class MaterializedView:
    """Represents a materialized view with metadata."""
    name: str
    query: Callable  # Function to compute view
    base_tables: Set[str]
    data: List[Dict] = field(default_factory=list)
    last_refresh: float = 0.0
    refresh_type: RefreshType = RefreshType.INCREMENTAL
    dirty: bool = True


class MaterializedViewManager:
    """Manages materialized views and incremental maintenance.
    
    Supports:
    - Full refresh (recompute everything)
    - Incremental refresh (apply deltas)
    - Automatic refresh on base table changes
    """
    
    def __init__(self):
        self.views: Dict[str, MaterializedView] = {}
        self.base_tables: Dict[str, List[Dict]] = {}
        self.pending_deltas: Dict[str, TableDelta] = {}
    
    def create_table(self, name: str, data: List[Dict]) -> None:
        """Create a base table."""
        self.base_tables[name] = data
        self.pending_deltas[name] = TableDelta()
    
    def create_view(
        self,
        name: str,
        query: Callable,
        base_tables: Set[str],
        refresh_type: RefreshType = RefreshType.INCREMENTAL
    ) -> None:
        """Create a materialized view."""
        self.views[name] = MaterializedView(
            name=name,
            query=query,
            base_tables=base_tables,
            refresh_type=refresh_type
        )
    
    def insert(self, table: str, row: Dict) -> None:
        """Insert row into base table."""
        self.base_tables[table].append(row)
        self.pending_deltas[table].inserts.append(row)
        self._mark_dependent_views_dirty(table)
    
    def delete(self, table: str, condition: Callable) -> int:
        """Delete rows matching condition."""
        to_delete = [r for r in self.base_tables[table] if condition(r)]
        for row in to_delete:
            self.base_tables[table].remove(row)
            self.pending_deltas[table].deletes.append(row)
        self._mark_dependent_views_dirty(table)
        return len(to_delete)
    
    def update(self, table: str, condition: Callable, updates: Dict) -> int:
        """Update rows matching condition."""
        count = 0
        for row in self.base_tables[table]:
            if condition(row):
                old = row.copy()
                row.update(updates)
                self.pending_deltas[table].updates.append((old, row.copy()))
                count += 1
        self._mark_dependent_views_dirty(table)
        return count
    
    def _mark_dependent_views_dirty(self, table: str) -> None:
        """Mark all views depending on this table as dirty."""
        for view in self.views.values():
            if table in view.base_tables:
                view.dirty = True
    
    def refresh_view(self, name: str) -> None:
        """Refresh a materialized view."""
        view = self.views.get(name)
        if not view:
            return
        
        if view.refresh_type == RefreshType.FULL:
            self._full_refresh(view)
        else:
            self._incremental_refresh(view)
        
        view.dirty = False
        view.last_refresh = time.time()
    
    def _full_refresh(self, view: MaterializedView) -> None:
        """Recompute entire view from base tables."""
        # Gather data from all base tables
        base_data = {t: self.base_tables[t] for t in view.base_tables}
        view.data = view.query(base_data)

        """Apply incremental changes to view."""
        if not view.data:
            # First time, do full refresh
            self._full_refresh(view)

                # Re-query for new row if it affects view
                # Simplified: just recalculate affected portion
            
            # Apply inserts
            for row in delta.inserts:
                # Check if row would appear in view
                # Simplified: recalculate and merge
                pass
        
        # For complex views, may need partial recomputation
        # This is a simplified approach
        self._full_refresh(view)
    
    def _matches_key(self, view_row: Dict, base_row: Dict) -> bool:
        """Check if view row corresponds to base row."""
        # Simplified key matching
        return view_row.get('id') == base_row.get('id')
```
