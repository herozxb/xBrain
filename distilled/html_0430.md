```html
      // Each await waits for the previous to complete

      const comments = await fetchComments(posts[0].id);
      console.log('Comments fetched:', comments);
      
      return { user, posts, comments };
    }
    
    async function fetchUser() {
      await delay(1000);
      return { id: 1, name: 'Alice' };
```

```html
    // 
    // Problem: Find index of last element matching a condition
    // Time: O(n) - Iterates from end to start
    // Space: O(1) - No additional storage
    
    const numbers = [1, 2, 3, 4, 5, 4, 3, 2, 1];
    
    // Find last index where value is greater than 3
    const lastIndex = numbers.findLastIndex(num => num > 3);
    console.log(lastIndex); // 5 (last occurrence of 4)
    
    // Find last even number index
    const lastEvenIndex = numbers.findLastIndex(num => num % 2 === 0);
    console.log(lastEvenIndex); // 7 (last occurrence of 2)
    
    // Returns -1 if not found
    const notFound = numbers.findLastIndex(num => num > 10);
    console.log(notFound); // -1
    
    // Real-world example: Find last active user
    const users = [
      { id: 1, name: 'Alice', active: false },
      { id: 2, name: 'Bob', active: true },
      { id: 3, name: 'Charlie', active: true },
      { id: 4, name: 'Dave', active: false }
    ];
    
    const lastActiveIndex = users.findLastIndex(user => user.active);
    console.log('Last active user at index:', lastActiveIndex); // 2
  </script>
</body>
</html>
```

**Why**: Efficiently searches arrays from end to find last matching element, avoiding manual reverse iteration.

```html
    // Time: O(1) - Single operation
    // Space: O(1) - No memory change
    
    const user = {
      name: 'Alice',
      age: 30
    };
    
    // Prevent adding new properties
    Object.preventExtensions(user);
    
    // Existing properties can still be modified
    user.name = 'Bob'; // ✓ Works
    console.log(user.name); // Bob
    
    // Adding new properties silently fails (or throws in strict mode)
    user.email = 'bob@example.com'; // ✗ Fails silently
    console.log(user.email); // undefined
    
    // Check if object is extensible
    console.log(Object.isExtensible(user)); // false
    
    // Delete still works
    delete user.age; // ✓ Works
    console.log(user); // { name: 'Bob' }
    
    // Use case: Lock configuration object
    const config = {
      apiUrl: 'https://api.example.com',
      timeout: 5000
    };
    Object.preventExtensions(config);
  </script>
</body>
</html>
```

**Why**: Protects objects from accidental property additions while allowing modifications to existing properties.

## 9. js_string_at

```html
  <title>String.at() Method</title>

  <script>
    // 
    // Problem: Access string characters using positive or negative indices
    // Time: O(1) - Direct index access
    // Space: O(1) - Returns single character
    
    const text = 'Hello World';
    
    // Positive indices (same as charAt)
    console.log(text.at(0)); // 'H'
    console.log(text.at(4)); // 'o'
    
    // Negative indices (from end)
    console.log(text.at(-1)); // 'd' (last character)
    console.log(text.at(-2)); // 'l' (second to last)
    console.log(text.at(-5)); // 'W'
    
    // Out of bounds returns undefined (not empty string)
    console.log(text.at(100)); // undefined
    console.log(text.at(-100)); // undefined
    
    // Compare with traditional methods
    console.log(text.charAt(text.length - 1)); // 'd' (old way)
    console.log(text.at(-1)); // 'd' (new way - cleaner)
    
    // Works with Unicode
    const emoji = '😀👍';
    console.log(emoji.at(0)); // '😀'
    console.log(emoji.at(-1)); // '👍'
  </script>
</body>
</html>
```

**Why**: Provides intuitive negative indexing for strings, making end-of-string access cleaner and more readable.

## 10. js_document_getElementsByClassName

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>getElementsByClassName</title>
</head>
<body>
  <div class="item">First</div>
  <div class="item highlight">Second</div>
  <div class="item">Third</div>
  
  <script>
    // 
    // Problem: Select multiple elements by class name
    // Time: O(n) - Where n is total DOM elements
    // Space: O(m) - Where m is matching elements
    
    // Returns live HTMLCollection
    const items = document.getElementsByClassName('item');
    console.log(items.length); // 3
    
    // Iterate over collection
    for (let item of items) {
      console.log(item.textContent);
    }
    
    // Live collection updates automatically
    const newDiv = document.createElement('div');
    newDiv.className = 'item';
    newDiv.textContent = 'Fourth';
    document.body.appendChild(newDiv);
    
    console.log(items.length); // 4 (auto-updated!)
    
    // Multiple classes (space-separated)
    const highlighted = document.getElementsByClassName('item highlight');
    console.log(highlighted.length); // 1
    
    // Access by index
    console.log(items[0].textContent); // 'First'
    
    // Convert to array for array methods
    const itemsArray = Array.from(items);
    itemsArray.forEach(item => {
      item.style.color = 'blue';
    });
  </script>
</body>
</html>
```

**Why**: Returns a live collection that automatically updates when DOM changes, efficient for dynamic content.
