```html
    <!-- PROBLEM: Create accessible table with proper headers -->
    <!-- TIME: O(n) SPACE: O(n) -->
    
    <table>
        <thead>
            <tr>
                <th scope="col">Name</th>
                <th scope="col">Age</th>
                <th scope="col">City</th>
            </tr>
        </thead>
        <tbody>
            <tr>
                <td>Alice</td>
                <td>28</td>
                <td>New York</td>

                <td>34</td>

            </tr>
        </tbody>
    </table>
    
    <script>
        // Sort table by column
        function sortTable(columnIndex) {
            const table = document.querySelector('table');
            const rows = Array.from(table.querySelectorAll('tbody tr'));
            
            rows.sort((a, b) => {
                const aVal = a.cells[columnIndex].textContent;
                const bVal = b.cells[columnIndex].textContent;
                return aVal.localeCompare(bVal, undefined, {numeric: true});
            });
            
            const tbody = table.querySelector('tbody');
            rows.forEach(row => tbody.appendChild(row));
        }
        
        // Add click handlers to headers for sorting
        document.querySelectorAll('th').forEach((th, index) => {
            th.style.cursor = 'pointer';
            th.addEventListener('click', () => sortTable(index));
        });
    </script>
</body>
</html>
```
**Why**: Table headers (`<th>`) with `scope` attribute improve accessibility. Screen readers use headers to describe cell context. Use `scope="col"` for column headers and `scope="row"` for row headers.

```html
    <!-- PROBLEM: Control element transparency -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <h3>Opacity Levels</h3>
    <div class="box opacity-full">1.0</div>
    <div class="box opacity-half">0.5</div>
    <div class="box opacity-light">0.2</div>
    <div class="box opacity-zero">0.0 (invisible)</div>

    <div class="box fade-hover">Hover me</div>
    
    <script>
        // Programmatically change opacity
        function setOpacity(element, value) {
            element.style.opacity = Math.max(0, Math.min(1, value));
        }
        
        // Fade in animation
        function fadeIn(element, duration = 1000) {
            element.style.opacity = 0;
            element.style.display = 'block';
            
            const start = performance.now();
            const animate = (currentTime) => {
                const elapsed = currentTime - start;
                const progress = Math.min(elapsed / duration, 1);
                element.style.opacity = progress;
                
                if (progress < 1) {
                    requestAnimationFrame(animate);
                }
            };
            requestAnimationFrame(animate);
        }
        
        // Fade out animation
        function fadeOut(element, duration = 1000) {
            const start = performance.now();
            const animate = (currentTime) => {
                const elapsed = currentTime - start;
                const progress = Math.min(elapsed / duration, 1);
                element.style.opacity = 1 - progress;
                
                if (progress < 1) {
                    requestAnimationFrame(animate);
                } else {
                    element.style.display = 'none';
                }
            };
            requestAnimationFrame(animate);
        }
    </script>
</body>
</html>
```
**Why**: `opacity` controls element transparency from 0 (invisible) to 1 (opaque). Affects entire element including children. Use `rgba()` for background-only transparency. Great for fade effects and overlays.

## js_async_await
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>JavaScript Async Await</title>
</head>
<body>
    <!-- PROBLEM: Handle asynchronous operations with clean syntax -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <div id="output"></div>
    <button onclick="fetchData()">Fetch Data</button>
    
    <script>
        // Simulate API call
        function fakeApiCall(delay, data) {
            return new Promise((resolve) => {
                setTimeout(() => resolve(data), delay);
            });
        }
        
        // Async function example
        async function fetchData() {
            const output = document.getElementById('output');
            output.innerHTML = 'Loading...';
            
            try {
                // Sequential async calls
                const user = await fakeApiCall(1000, { id: 1, name: 'Alice' });
                output.innerHTML += `<br>User: ${user.name}`;
                
                const posts = await fakeApiCall(1000, [
                    { title: 'Post 1' },
                    { title: 'Post 2' }
                ]);
                output.innerHTML += `<br>Posts: ${posts.length}`;
                
                // Parallel async calls
                const [comments, likes] = await Promise.all([
                    fakeApiCall(500, ['comment1', 'comment2']),
                    fakeApiCall(500, 42)
                ]);
                output.innerHTML += `<br>Comments: ${comments.length}, Likes: ${likes}`;
                
            } catch (error) {
                output.innerHTML = `Error: ${error.message}`;
            }
        }
        
        // Async arrow function
        const fetchJson = async (url) => {
            const response = await fetch(url);
            if (!response.ok) {
                throw new Error(`HTTP error! status: ${response.status}`);
            }
            return await response.json();
        };
        
        // Top-level await (ES2022)
        // const data = await fetchJson('/api/data');
        
        // Async class method
        class DataService {
            async getData(id) {
                const cached = this.cache.get(id);
                if (cached) return cached;
                
                const data = await fetchJson(`/api/data/${id}`);
                this.cache.set(id, data);
                return data;
            }
            
            cache = new Map();
        }
        
        // Error handling patterns
        async function robustFetch(url) {
            try {
                const data = await fetchJson(url);
                return { success: true, data };
            } catch (error) {
                return { success: false, error: error.message };
            }
        }
    </script>
</body>
</html>
```
**Why**: `async/await` provides synchronous-looking syntax for asynchronous code. `async` functions return Promises, `await` pauses execution until Promise resolves. Use `try/catch` for error handling. Cleaner than `.then()` chains.

```html
        }
        
        // Security: Be careful with user input!
        function unsafeExample(userInput) {
            // DANGEROUS: XSS vulnerability
            // container.innerHTML = userInput;
            
            // SAFE: Sanitize first
            const sanitized = userInput
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;');
            container.innerHTML = sanitized;
        }
        
        // Performance: For repeated updates, use DocumentFragment
        function efficientList(items) {
            const fragment = document.createDocumentFragment();
            items.forEach(item => {
                const li = document.createElement('li');
                li.textContent = item;
                fragment.appendChild(li);
            });
            
            const ul = document.createElement('ul');
            ul.appendChild(fragment);
            container.innerHTML = '';
            container.appendChild(ul);
        }
    </script>

```
**Why**: `innerHTML` gets/sets HTML markup inside an element. Fast for bulk updates but destroys existing elements and event listeners. Security risk with unsanitized user input (XSS). Use `textContent` for plain text, DOM methods for complex updates.

## event_mouseenter
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Event MouseEnter</title>
    <style>
        .box {
            width: 200px;
            height: 200px;
            background: #4CAF50;
            margin: 20px;
            padding: 20px;
            transition: background 0.3s;
        }
        
        .box.active {
            background: #ff5722;
        }
        
        .nested {
            width: 100px;
            height: 100px;
            background: #2196F3;
        }
    </style>
</head>
<body>
    <!-- PROBLEM: Detect when mouse enters element (no bubbling) -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <div class="box" id="parent">
        Parent (mouseenter/mouseleave)
        <div class="nested" id="child">Child</div>
    </div>
    
    <div class="box" id="parent2">
        Parent (mouseover/mouseout)
        <div class="nested" id="child2">Child</div>
    </div>
    
    <p id="log"></p>
    
    <script>
        const log = document.getElementById('log');
        
        // mouseenter: does NOT bubble to parent
        const parent = document.getElementById('parent');
        const child = document.getElementById('child');
        
        parent.addEventListener('mouseenter', () => {
            parent.classList.add('active');
            log.textContent = 'Mouse entered parent';
        });
        
        parent.addEventListener('mouseleave', () => {
            parent.classList.remove('active');
            log.textContent = 'Mouse left parent';
        });
        
        // child doesn't trigger parent events with mouseenter
        child.addEventListener('mouseenter', () => {
            log.textContent = 'Mouse entered child (parent still active)';
        });
        
        // COMPARISON: mouseover DOES bubble
        const parent2 = document.getElementById('parent2');
        
        parent2.addEventListener('mouseover', (e) => {
            log.textContent = `Mouseover on ${e.target.id || 'parent2'}`;
        });
        
        parent2.addEventListener('mouseout', (e) => {
            log.textContent = `Mouseout from ${e.target.id || 'parent2'}`;
        });
        
        // Hover effect helper
        function addHoverEffect(element, enterClass, leaveClass) {
            element.addEventListener('mouseenter', () => {
                element.classList.add(enterClass);
                element.classList.remove(leaveClass);
            });
            element.addEventListener('mouseleave', () => {
                element.classList.remove(enterClass);
                element.classList.add(leaveClass);
            });
        }
        
        // Tooltip example
        function showTooltip(element, text) {
            const tooltip = document.createElement('div');
            tooltip.className = 'tooltip';
            tooltip.textContent = text;
            tooltip.style.cssText = `
                position: absolute;
                background: black;
                color: white;
                padding: 5px 10px;
                border-radius: 4px;
                font-size: 12px;
            `;
            
            element.addEventListener('mouseenter', (e) => {
                document.body.appendChild(tooltip);
                tooltip.style.left = e.pageX + 10 + 'px';
                tooltip.style.top = e.pageY + 10 + 'px';
            });
            
            element.addEventListener('mouseleave', () => {
                tooltip.remove();
            });
        }
    </script>
</body>
</html>
```
**Why**: `mouseenter` fires when mouse enters element, but unlike `mouseover`, it doesn't bubble to ancestors. Paired with `mouseleave`. Perfect for hover effects on specific elements without triggering parent handlers. More efficient than `mouseover` for nested elements.

```html
            return {
                value: inputElement.value,
                type: inputElement.type,
                name: inputElement.name,
                checked: inputElement.checked,
                files: inputElement.files
            };
        }
    </script>

```
**Why**: The `input` event fires on every character change, selection, cut, or paste. Works on `<input>`, `<textarea>`, and `<select>`. Better than `keyup` for tracking value changes. Combine with debounce for search/autocomplete. Real-time validation.
