## Facade Patterns

```python
# PROBLEM: Simplify complex subsystem interfaces into a unified API
# TIME: O(1) for facade creation SPACE: O(1) additional overhead

class CPU:
    """CPU subsystem component"""
    def start(self) -> None:
        print("CPU: Initializing processor cores")
    
    def execute(self) -> None:
        print("CPU: Executing instructions")

class Memory:
    """Memory subsystem component"""
    def load(self) -> None:
        print("Memory: Loading data into RAM")
    
    def clean(self) -> None:
        print("Memory: Clearing cache")

class HardDrive:
    """Hard drive subsystem component"""
    def read(self) -> None:
        print("HardDrive: Reading data from disk")
    
    def write(self) -> None:
        print("HardDrive: Writing data to disk")

class ComputerFacade:
    """
    Facade that provides a simplified interface to the complex
    computer subsystems (CPU, Memory, HardDrive)
    """
    def __init__(self):
        self.cpu = CPU()
        self.memory = Memory()
        self.hard_drive = HardDrive()
    
    def start_computer(self) -> None:
        """Single interface to start the entire system"""
        print("=== Starting Computer ===")
        self.cpu.start()
        self.memory.load()
        self.hard_drive.read()
        self.cpu.execute()
        print("=== Computer Ready ===")
    
    def shutdown_computer(self) -> None:
        """Single interface to shutdown the system"""
        print("=== Shutting Down ===")
        self.memory.clean()
        print("=== Computer Off ===")

# Usage
if __name__ == "__main__":
    computer = ComputerFacade()
    computer.start_computer()
    # Output:
    # === Starting Computer ===
    # CPU: Initializing processor cores
    # Memory: Loading data into RAM
    # HardDrive: Reading data from disk
    # CPU: Executing instructions
    # === Computer Ready ===
```

**Why**: Facade pattern hides complexity by providing a single entry point to a complex subsystem, making the system easier to use and reducing coupling between client code and internal components.

## Memento Patterns

```python
# PROBLEM: Capture and restore object state without exposing implementation
# TIME: O(n) where n is state size SPACE: O(n) for storing mementos

from typing import Any, List
from dataclasses import dataclass
from datetime import datetime

@dataclass
class EditorMemento:
    """Memento object that stores editor state"""
    content: str
    cursor_position: int
    timestamp: datetime

class TextEditor:
    """Originator - the object whose state we want to save"""
    
    def __init__(self):
        self._content: str = ""
        self._cursor_position: int = 0
    
    def write(self, text: str) -> None:
        """Write text at cursor position"""
        self._content = (
            self._content[:self._cursor_position] + 
            text + 
            self._content[self._cursor_position:]
        )
        self._cursor_position += len(text)
    
    def move_cursor(self, position: int) -> None:
        """Move cursor to specified position"""
        self._cursor_position = max(0, min(position, len(self._content)))
    
    def delete(self, length: int = 1) -> None:
        """Delete characters from cursor position"""
        end = self._cursor_position + length
        self._content = (
            self._content[:self._cursor_position] + 
            self._content[end:]
        )
    
    def create_memento(self) -> EditorMemento:
        """Save current state to memento"""
        return EditorMemento(
            content=self._content,
            cursor_position=self._cursor_position,
            timestamp=datetime.now()
        )
    
    def restore_from_memento(self, memento: EditorMemento) -> None:
        """Restore state from memento"""
        self._content = memento.content
        self._cursor_position = memento.cursor_position
    
    def __str__(self) -> str:
        return f"Content: '{self._content}' | Cursor: {self._cursor_position}"

class EditorHistory:
    """Caretaker - manages the history of mementos"""
    
    def __init__(self, max_size: int = 10):
        self._history: List[EditorMemento] = []
        self._max_size = max_size
    
    def save(self, memento: EditorMemento) -> None:
        """Save a memento to history"""
        if len(self._history) >= self._max_size:
            self._history.pop(0)  # Remove oldest
        self._history.append(memento)
    
    def undo(self) -> EditorMemento | None:
        """Get the most recent memento"""
        if self._history:
            return self._history.pop()
        return None
    
    def get_history_count(self) -> int:
        """Return number of saved states"""
        return len(self._history)

# Usage
if __name__ == "__main__":
    editor = TextEditor()
    history = EditorHistory(max_size=5)
    
    # Initial write and save
    editor.write("Hello")
    history.save(editor.create_memento())
    print(editor)  # Content: 'Hello' | Cursor: 5
    
    # More changes
    editor.write(" World")
    history.save(editor.create_memento())
    print(editor)  # Content: 'Hello World' | Cursor: 11
    
    # More changes
    editor.write("!")
    print(editor)  # Content: 'Hello World!' | Cursor: 12
    
    # Undo to last saved state
    memento = history.undo()
    if memento:
        editor.restore_from_memento(memento)
    print(editor)  # Content: 'Hello World' | Cursor: 11
```

**Why**: Memento pattern enables undo/redo functionality by capturing object state without violating encapsulation, allowing state restoration at any time while keeping the originator's implementation private.

## Visitor Patterns

```python
# PROBLEM: Add operations to object structures without modifying them
# TIME: O(n) where n is number of elements SPACE: O(1) for visitor operations

from abc import ABC, abstractmethod
from typing import List

# Element interfaces
class DocumentElement(ABC):
    """Abstract element that accepts visitors"""
    
    @abstractmethod
    def accept(self, visitor: 'DocumentVisitor') -> None:
        """Accept a visitor to operate on this element"""
        pass

# Concrete elements
class Paragraph(DocumentElement):
    """Paragraph element containing text"""
    
    def __init__(self, text: str):
        self.text = text
    
    def accept(self, visitor: 'DocumentVisitor') -> None:
        visitor.visit_paragraph(self)

class Image(DocumentElement):
    """Image element with source path"""
    
    def __init__(self, src: str, alt: str = ""):
        self.src = src
        self.alt = alt
    
    def accept(self, visitor: 'DocumentVisitor') -> None:
        visitor.visit_image(self)

class Table(DocumentElement):
    """Table element with rows and columns"""
    
    def __init__(self, data: List[List[str]]):
        self.data = data
        self.rows = len(data)
        self.cols = len(data[0]) if data else 0
    
    def accept(self, visitor: 'DocumentVisitor') -> None:
        visitor.visit_table(self)

# Visitor interface
class DocumentVisitor(ABC):
    """Abstract visitor interface"""
    
    @abstractmethod
    def visit_paragraph(self, paragraph: Paragraph) -> None:
        pass
    
    @abstractmethod
    def visit_image(self, image: Image) -> None:
        pass
    
    @abstractmethod
    def visit_table(self, table: Table) -> None:
        pass

# Concrete visitors
class HTMLExportVisitor(DocumentVisitor):
    """Visitor that exports document elements to HTML"""
    
    def __init__(self):
        self.html_output: List[str] = []
    
    def visit_paragraph(self, paragraph: Paragraph) -> None:
        self.html_output.append(f"<p>{paragraph.text}</p>")
    
    def visit_image(self, image: Image) -> None:
        self.html_output.append(
            f'<img src="{image.src}" alt="{image.alt}">'
        )
    
    def visit_table(self, table: Table) -> None:
        rows = []
        for row in table.data:
            cells = "".join(f"<td>{cell}</td>" for cell in row)
            rows.append(f"<tr>{cells}</tr>")
        self.html_output.append(
            f"<table>{''.join(rows)}</table>"
        )
    
    def get_html(self) -> str:
        return "\n".join(self.html_output)

class WordCountVisitor(DocumentVisitor):
    """Visitor that counts words across all elements"""
    
    def __init__(self):
        self.word_count: int = 0
    
    def visit_paragraph(self, paragraph: Paragraph) -> None:
        self.word_count += len(paragraph.text.split())
    
    def visit_image(self, image: Image) -> None:
        # Count alt text words
        if image.alt:
            self.word_count += len(image.alt.split())
    
    def visit_table(self, table: Table) -> None:
        for row in table.data:
            for cell in row:
                self.word_count += len(cell.split())
    
    def get_count(self) -> int:
        return self.word_count

class MarkdownExportVisitor(DocumentVisitor):
    """Visitor that exports document elements to Markdown"""
    
    def __init__(self):
        self.markdown_output: List[str] = []
    
    def visit_paragraph(self, paragraph: Paragraph) -> None:
        self.markdown_output.append(paragraph.text)
    
    def visit_image(self, image: Image) -> None:
        self.markdown_output.append(f"![{image.alt}]({image.src})")
    
    def visit_table(self, table: Table) -> None:
        if not table.data:
            return
        
        # Header row
        header = "| " + " | ".join(table.data[0]) + " |"
        separator = "| " + " | ".join(["---"] * table.cols) + " |"
        
        # Data rows
        rows = []
        for row in table.data[1:]:
            rows.append("| " + " | ".join(row) + " |")
        
        self.markdown_output.append(
            "\n".join([header, separator] + rows)
        )
    
    def get_markdown(self) -> str:
        return "\n\n".join(self.markdown_output)

# Document structure
class Document:
    """Object structure containing elements"""
    
    def __init__(self):
        self.elements: List[DocumentElement] = []
    
    def add_element(self, element: DocumentElement) -> None:
        self.elements.append(element)
    
    def accept(self, visitor: DocumentVisitor) -> None:
        """Accept visitor for all elements"""
        for element in self.elements:
            element.accept(visitor)

# Usage
if __name__ == "__main__":
    # Create document with various elements
    doc = Document()
    doc.add_element(Paragraph("Welcome to the Document"))
    doc.add_element(Paragraph("This is a second paragraph."))
    doc.add_element(Image("photo.jpg", "A beautiful sunset"))
    doc.add_element(Table([
        ["Name", "Age", "City"],
        ["Alice", "30", "NYC"],
        ["Bob", "25", "LA"]
    ]))
    
    # Export to HTML
    html_visitor = HTMLExportVisitor()
    doc.accept(html_visitor)
    print("=== HTML Export ===")
    print(html_visitor.get_html())
    
    # Count words
    word_visitor = WordCountVisitor()
    doc.accept(word_visitor)
    print(f"\n=== Word Count: {word_visitor.get_count()} ===")
    
    # Export to Markdown
    md_visitor = MarkdownExportVisitor()
    doc.accept(md_visitor)
    print("\n=== Markdown Export ===")
    print(md_visitor.get_markdown())
```

**Why**: Visitor pattern separates algorithms from object structures, allowing new operations to be added without modifying existing element classes. It centralizes related behavior and makes it easy to add cross-cutting operations like export, validation, or transformation.
