## Tim Sort

```python
# PROBLEM: Hybrid stable sorting algorithm combining merge sort and insertion sort
# TIME: O(n log n) worst, O(n) best SPACE: O(n)
def tim_sort(arr):
    """
    Tim Sort is Python's built-in sorting algorithm.
    Uses insertion sort for small runs and merge sort for combining.
    """
    if not arr or len(arr) <= 1:
        return arr
    
    MIN_RUN = 32  # Minimum run size for insertion sort
    
    def insertion_sort(arr, left, right):
        """Sort small subarrays with insertion sort"""
        for i in range(left + 1, right + 1):
            key = arr[i]
            j = i - 1
            # Shift elements greater than key to the right
            while j >= left and arr[j] > key:
                arr[j + 1] = arr[j]
                j -= 1
            arr[j + 1] = key
    
    def merge(arr, left, mid, right):
        """Merge two sorted subarrays"""
        left_arr = arr[left:mid + 1]
        right_arr = arr[mid + 1:right + 1]
        
        i = j = 0
        k = left
        
        # Merge in sorted order
        while i < len(left_arr) and j < len(right_arr):
            if left_arr[i] <= right_arr[j]:
                arr[k] = left_arr[i]
                i += 1
            else:
                arr[k] = right_arr[j]
                j += 1
            k += 1
        
        # Copy remaining elements
        while i < len(left_arr):
            arr[k] = left_arr[i]
            i += 1
            k += 1
        
        while j < len(right_arr):
            arr[k] = right_arr[j]
            j += 1
            k += 1
    
    n = len(arr)
    
    # Sort individual subarrays of size MIN_RUN
    for start in range(0, n, MIN_RUN):
        end = min(start + MIN_RUN - 1, n - 1)
        insertion_sort(arr, start, end)
    
    # Start merging from size MIN_RUN
    size = MIN_RUN
    while size < n:
        # Pick starting point of left sub array
        for left in range(0, n, 2 * size):
            mid = min(n - 1, left + size - 1)
            right = min((left + 2 * size - 1), (n - 1))
            
            # Merge subarrays arr[left...mid] & arr[mid+1...right]
            if mid < right:
                merge(arr, left, mid, right)
        
        size = 2 * size
    
    return arr
```

**Why**: Tim Sort is the default sorting algorithm in Python and Java. It's adaptive, stable, and performs well on real-world data with patterns.

## Kosaraju's Algorithm

```python
# PROBLEM: Find all strongly connected components (SCCs) in a directed graph
# TIME: O(V + E) SPACE: O(V + E)
from collections import defaultdict

def kosaraju_scc(graph):
    """
    Kosaraju's algorithm finds SCCs using two DFS passes.
    First pass on original graph, second pass on transposed graph.
    """
    if not graph:
        return []
    
    # Build adjacency list
    adj = defaultdict(list)
    vertices = set()
    for u, v in graph:
        adj[u].append(v)
        vertices.add(u)
        vertices.add(v)
    
    visited = set()
    finish_order = []  # Vertices in order of finish times
    
    def dfs_first(u):
        """First DFS to record finish times"""
        visited.add(u)
        for v in adj[u]:
            if v not in visited:
                dfs_first(v)
        finish_order.append(u)
    
    # Step 1: Fill finish_order with vertices in order of finish times
    for v in vertices:
        if v not in visited:
            dfs_first(v)
    
    # Step 2: Create transposed graph (reverse all edges)
    adj_transpose = defaultdict(list)
    for u in vertices:
        for v in adj[u]:
            adj_transpose[v].append(u)
    
    # Step 3: Do DFS on transposed graph in reverse finish order
    visited.clear()
    sccs = []
    
    def dfs_second(u, component):
        """Second DFS to collect SCC members"""
        visited.add(u)
        component.append(u)
        for v in adj_transpose[u]:
            if v not in visited:
                dfs_second(v, component)
    
    # Process vertices in reverse finish order
    for u in reversed(finish_order):
        if u not in visited:
            component = []
            dfs_second(u, component)
            sccs.append(component)
    
    return sccs
```

**Why**: SCCs are fundamental in graph analysis - used in compiler optimization, social network analysis, and detecting cycles in dependency graphs.

## Dinic's Algorithm

```python
# PROBLEM: Find maximum flow in a flow network
# TIME: O(V²E) general, O(E√V) for unit capacities SPACE: O(V + E)
from collections import defaultdict, deque

def dinic_max_flow(graph, source, sink):
    """
    Dinic's algorithm for maximum flow.
    Uses BFS for level graph and DFS for blocking flow.
    """
    # Build adjacency list with capacities
    adj = defaultdict(list)
    capacity = defaultdict(int)
    
    for u, v, cap in graph:
        adj[u].append(v)
        adj[v].append(u)  # Add reverse edge
        capacity[(u, v)] += cap
        capacity[(v, u)] += 0  # Reverse edge starts with 0
    
    level = {}
    
    def bfs():
        """Build level graph using BFS"""
        level.clear()
        level[source] = 0
        queue = deque([source])
        
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if v not in level and capacity[(u, v)] > 0:
                    level[v] = level[u] + 1
                    queue.append(v)
        
        return sink in level
    
    def dfs(u, flow):
        """Find blocking flow using DFS"""
        if u == sink:
            return flow
        
        for v in adj[u]:
            if level.get(v, -1) == level[u] + 1 and capacity[(u, v)] > 0:
                min_flow = min(flow, capacity[(u, v)])
                result = dfs(v, min_flow)
                
                if result > 0:
                    # Update residual capacities
                    capacity[(u, v)] -= result
                    capacity[(v, u)] += result
                    return result
        
        return 0
    
    max_flow = 0
    
    # While there's a path from source to sink
    while bfs():
        # Find all augmenting paths in level graph
        while True:
            flow = dfs(source, float('inf'))
            if flow == 0:
                break
            max_flow += flow
    
    return max_flow
```

**Why**: Dinic's is faster than Ford-Fulkerson for most cases. Used in network flow problems, bipartite matching, and resource allocation.

## Boyer-Moore Algorithm

```python
# PROBLEM: Efficient string matching algorithm
# TIME: O(n/m) best, O(n + m) worst SPACE: O(k) where k = alphabet size
def boyer_moore_search(text, pattern):
    """
    Boyer-Moore string matching algorithm.
    Scans from right to left, uses bad character and good suffix rules.
    """
    if not pattern or not text or len(pattern) > len(text):
        return []
    
    def build_bad_char_table(pattern):
        """
        Build bad character table.
        Maps each character to its rightmost position in pattern.
        """
        table = {}
        for i in range(len(pattern) - 1):
            table[pattern[i]] = len(pattern) - 1 - i
        return table
    
    def build_good_suffix_table(pattern):
        """
        Build good suffix table for additional shift optimization.
        Simplified version using border computation.
        """
        m = len(pattern)
        table = [0] * (m + 1)
        
        # Case 1: matching suffix occurs elsewhere in pattern
        i = m
        j = m + 1
        border = [0] * (m + 1)
        border[i] = j
        
        while i > 0:
            while j <= m and pattern[i - 1] != pattern[j - 1]:
                if table[j] == 0:
                    table[j] = j - i
                j = border[j]
            i -= 1
            j -= 1
            border[i] = j
        
        # Case 2: prefix of pattern matches suffix of good suffix
        j = border[0]
        for i in range(m + 1):
            if table[i] == 0:
                table[i] = j
            if i == j:
                j = border[j]
        
        return table
    
    n = len(text)
    m = len(pattern)
    bad_char = build_bad_char_table(pattern)
    good_suffix = build_good_suffix_table(pattern)
    
    matches = []
    s = 0  # Shift of pattern with respect to text
    
    while s <= n - m:
        j = m - 1
        
        # Compare from right to left
        while j >= 0 and pattern[j] == text[s + j]:
            j -= 1
        
        if j < 0:
            # Pattern found
            matches.append(s)
            # Shift by good suffix rule
            s += good_suffix[0]
        else:
            # Mismatch: shift by max of bad char and good suffix
            bc_shift = bad_char.get(text[s + j], m)
            gs_shift = good_suffix[j + 1]
            s += max(bc_shift, gs_shift)
    
    return matches
```

**Why**: Boyer-Moore is the fastest string search in practice for typical text. Used in text editors, grep, and search engines.

## Red-Black Tree

```python
# PROBLEM: Self-balancing BST with guaranteed O(log n) operations
# TIME: O(log n) for insert/delete/search SPACE: O(n)
class RBNode:
    def __init__(self, key, color='RED'):
        self.key = key
        self.color = color  # 'RED' or 'BLACK'
        self.left = None
        self.right = None
        self.parent = None

class RedBlackTree:
    def __init__(self):
        self.NIL = RBNode(None, 'BLACK')  # Sentinel node
        self.root = self.NIL
    
    def insert(self, key):
        """Insert key maintaining red-black properties"""
        node = RBNode(key)
        node.left = self.NIL
        node.right = self.NIL
        
        parent = None
        current = self.root
        
        # Find insertion point
        while current != self.NIL:
            parent = current
            if node.key < current.key:
                current = current.left
            else:
                current = current.right
        
        node.parent = parent
        
        if parent is None:
            self.root = node
        elif node.key < parent.key:
            parent.left = node
        else:
            parent.right = node
        
        node.color = 'RED'
        self._fix_insert(node)
    
    def _fix_insert(self, node):
        """Fix red-black violations after insertion"""
        while node.parent and node.parent.color == 'RED':
            if node.parent == node.parent.parent.left:
                uncle = node.parent.parent.right
                
                if uncle.color == 'RED':
                    # Case 1: Uncle is red - recolor
                    node.parent.color = 'BLACK'
                    uncle.color = 'BLACK'
                    node.parent.parent.color = 'RED'
                    node = node.parent.parent
                else:
                    if node == node.parent.right:
                        # Case 2: Left rotation needed
                        node = node.parent
                        self._left_rotate(node)
                    
                    # Case 3: Right rotation + recolor
                    node.parent.color = 'BLACK'
                    node.parent.parent.color = 'RED'
                    self._right_rotate(node.parent.parent)
            else:
                # Mirror cases for right side
                uncle = node.parent.parent.left
                
                if uncle.color == 'RED':
                    node.parent.color = 'BLACK'
                    uncle.color = 'BLACK'
                    node.parent.parent.color = 'RED'
                    node = node.parent.parent
                else:
                    if node == node.parent.left:
                        node = node.parent
                        self._right_rotate(node)
                    
                    node.parent.color = 'BLACK'
                    node.parent.parent.color = 'RED'
                    self._left_rotate(node.parent.parent)
        
        self.root.color = 'BLACK'
    
    def _left_rotate(self, x):
        """Left rotation around node x"""
        y = x.right
        x.right = y.left
        
        if y.left != self.NIL:
            y.left.parent = x
        
        y.parent = x.parent
        
        if x.parent is None:
            self.root = y
        elif x == x.parent.left:
            x.parent.left = y
        else:
            x.parent.right = y
        
        y.left = x
        x.parent = y
    
    def _right_rotate(self, y):
        """Right rotation around node y"""
        x = y.left
        y.left = x.right
        
        if x.right != self.NIL:
            x.right.parent = y
        
        x.parent = y.parent
        
        if y.parent is None:
            self.root = x
        elif y == y.parent.right:
            y.parent.right = x
        else:
            y.parent.left = x
        
        x.right = y
        y.parent = x
    
    def search(self, key):
        """Search for key in tree"""
        node = self.root
        while node != self.NIL and key != node.key:
            if key < node.key:
                node = node.left
            else:
                node = node.right
        return node if node != self.NIL else None
```

**Why**: Red-Black trees guarantee O(log n) operations and are used in C++ STL map, Java TreeMap, and Linux kernel for memory management.

## B-Tree

```python
# PROBLEM: Self-balancing tree optimized for disk storage
# TIME: O(log n) for insert/delete/search SPACE: O(n)
class BTreeNode:
    def __init__(self, leaf=False):
        self.keys = []
        self.children = []
        self.leaf = leaf

class BTree:
    def __init__(self, min_degree):
        self.min_degree = min_degree  # Minimum degree (t)
        self.root = BTreeNode(leaf=True)
    
    def search(self, key, node=None):
        """Search for key in B-tree"""
        if node is None:
            node = self.root
        
        i = 0
        while i < len(node.keys) and key > node.keys[i]:
            i += 1
        
        if i < len(node.keys) and key == node.keys[i]:
            return (node, i)  # Found
        
        if node.leaf:
            return None  # Not found
        
        # Search in appropriate child
        return self.search(key, node.children[i])
    
    def insert(self, key):
        """Insert key into B-tree"""
        root = self.root
        
        # If root is full, split it
        if len(root.keys) == (2 * self.min_degree - 1):
            new_root = BTreeNode()
            new_root.children.append(self.root)
            self.root = new_root
            self._split_child(new_root, 0)
            self._insert_non_full(new_root, key)
        else:
            self._insert_non_full(root, key)
    
    def _split_child(self, parent, index):
        """Split child at index"""
        min_deg = self.min_degree
        full_child = parent.children[index]
        new_child = BTreeNode(leaf=full_child.leaf)
        
        # Move median key to parent
        parent.keys.insert(index, full_child.keys[min_deg - 1])
        parent.children.insert(index + 1, new_child)
        
        # Split keys
        new_child.keys = full_child.keys[min_deg:]
        full_child.keys = full_child.keys[:min_deg - 1]
        
        # Split children if not leaf
        if not full_child.leaf:
            new_child.children = full_child.children[min_deg:]
            full_child.children = full_child.children[:min_deg]
    
    def _insert_non_full(self, node, key):
        """Insert into non-full node"""
        if node.leaf:
            # Insert key in sorted position
            i = len(node.keys) - 1
            node.keys.append(None)
            while i >= 0 and key < node.keys[i]:
                node.keys[i + 1] = node.keys[i]
                i -= 1
            node.keys[i + 1] = key
        else:
            # Find child to insert into
            i = len(node.keys) - 1
            while i >= 0 and key < node.keys[i]:
                i -= 1
            i += 1
            
            # Split child if full
            if len(node.children[i].keys) == (2 * self.min_degree - 1):
                self._split_child(node, i)
                if key > node.keys[i]:
                    i += 1
            
            self._insert_non_full(node.children[i], key)
```

**Why**: B-Trees minimize disk I/O by storing multiple keys per node. Used in databases (MySQL, PostgreSQL) and file systems (HFS+, NTFS).

## Flyweight Pattern

```python
# PROBLEM: Minimize memory usage by sharing common state between objects
# TIME: O(1) for object retrieval SPACE: O(n) for shared objects
class Flyweight:
    """Shared state (intrinsic)"""
    def __init__(self, shared_state):
        self.shared_state = shared_state
    
    def operation(self, unique_state):
        """Operation using both shared and unique state"""
        s = f"Flyweight: Shared ({self.shared_state}) and Unique ({unique_state})"
        return s

class FlyweightFactory:
    """Factory that manages flyweight objects"""
    def __init__(self):
        self._flyweights = {}
    
    def get_flyweight(self, shared_state):
        """Get existing flyweight or create new one"""
        key = str(shared_state)
        
        if key not in self._flyweights:
            print(f"FlyweightFactory: Creating new flyweight for {shared_state}")
            self._flyweights[key] = Flyweight(shared_state)
        else:
            print(f"FlyweightFactory: Reusing existing flyweight for {shared_state}")
        
        return self._flyweights[key]
    
    def list_flyweights(self):
        """List all flyweights"""
        return [key for key in self._flyweights.keys()]

class Context:
    """Context with unique state (extrinsic)"""
    def __init__(self, flyweight, unique_state):
        self.flyweight = flyweight
        self.unique_state = unique_state
    
    def operation(self):
        """Perform operation"""
        return self.flyweight.operation(self.unique_state)

# Example usage: Text editor with character formatting
class CharacterFlyweight:
    """Shared character formatting (font, size, color)"""
    def __init__(self, font, size, color):
        self.font = font
        self.size = size
        self.color = color
    
    def render(self, char, position):
        """Render character at position with formatting"""
        return f"Char '{char}' at {position}: {self.font} {self.size}px {self.color}"

class CharacterFlyweightFactory:
    def __init__(self):
        self._flyweights = {}
    
    def get_format(self, font, size, color):
        """Get or create flyweight for formatting"""
        key = f"{font}:{size}:{color}"
        if key not in self._flyweights:
            self._flyweights[key] = CharacterFlyweight(font, size, color)
        return self._flyweights[key]
    
    def total_flyweights(self):
        return len(self._flyweights)
```

**Why**: Flyweight dramatically reduces memory when many objects share common state. Used in text editors, game engines (particle systems), and GUI frameworks.

## Visitor Pattern

```python
# PROBLEM: Add operations to object structures without modifying classes
# TIME: O(n) for visiting all elements SPACE: O(1)
from abc import ABC, abstractmethod

# Visitor interface
class Visitor(ABC):
    @abstractmethod
    def visit_concrete_element_a(self, element):
        pass
    
    @abstractmethod
    def visit_concrete_element_b(self, element):
        pass

# Element interface
class Element(ABC):
    @abstractmethod
    def accept(self, visitor):
        pass

# Concrete elements
class ConcreteElementA(Element):
    def __init__(self, data):
        self.data = data
    
    def accept(self, visitor):
        visitor.visit_concrete_element_a(self)
    
    def exclusive_method_a(self):
        return f"Element A: {self.data}"

class ConcreteElementB(Element):
    def __init__(self, value):
        self.value = value
    
    def accept(self, visitor):
        visitor.visit_concrete_element_b(self)
    
    def exclusive_method_b(self):
        return f"Element B: {self.value}"

# Concrete visitors
class ConcreteVisitor1(Visitor):
    def __init__(self):
        self.result = []
    
    def visit_concrete_element_a(self, element):
        self.result.append(f"Visitor1 processing {element.exclusive_method_a()}")
    
    def visit_concrete_element_b(self, element):
        self.result.append(f"Visitor1 processing {element.exclusive_method_b()}")

class ConcreteVisitor2(Visitor):
    def __init__(self):
        self.count = 0
    
    def visit_concrete_element_a(self, element):
        self.count += len(str(element.data))
    
    def visit_concrete_element_b(self, element):
        self.count += len(str(element.value))

# Object structure
class ObjectStructure:
    def __init__(self):
        self.elements = []
    
    def add_element(self, element):
        self.elements.append(element)
    
    def accept(self, visitor):
        for element in self.elements:
            element.accept(visitor)

# Example: Abstract Syntax Tree visitor
class ASTNode(ABC):
    @abstractmethod
    def accept(self, visitor):
        pass

class NumberNode(ASTNode):
    def __init__(self, value):
        self.value = value
    
    def accept(self, visitor):
        return visitor.visit_number(self)

class AddNode(ASTNode):
    def __init__(self, left, right):
        self.left = left
        self.right = right
    
    def accept(self, visitor):
        return visitor.visit_add(self)

class ASTEvaluator(Visitor):
    """Visitor that evaluates AST"""
    def visit_number(self, node):
        return node.value
    
    def visit_add(self, node):
        return node.left.accept(self) + node.right.accept(self)

class ASTPrinter(Visitor):
    """Visitor that prints AST"""
    def visit_number(self, node):
        return str(node.value)
    
    def visit_add(self, node):
        left = node.left.accept(self)
        right = node.right.accept(self)
        return f"({left} + {right})"
```

**Why**: Visitor separates algorithms from object structures. Used in compilers (AST processing), document processing, and when operations vary but structures are stable.

## Raft Consensus Algorithm

```python
# PROBLEM: Distributed consensus for replicated state machines
# TIME: O(n) for leader election, O(n) for log replication SPACE: O(n log m)
import random
import time
from enum import Enum
from collections import defaultdict

class State(Enum):
    FOLLOWER = "follower"
    CANDIDATE = "candidate"
    LEADER = "leader"

class LogEntry:
    def __init__(self, term, command):
        self.term = term
        self.command = command

class RaftNode:
    def __init__(self, node_id, peers):
        self.node_id = node_id
        self.peers = peers  # Other nodes in cluster
        
        # Persistent state
        self.current_term = 0
        self.voted_for = None
        self.log = [LogEntry(0, None)]  # Log entries
        
        # Volatile state
        self.commit_index = 0
        self.last_applied = 0
        self.state = State.FOLLOWER
        
        # Leader state (reinitialized after election)
        self.next_index = defaultdict(lambda: len(self.log))
        self.match_index = defaultdict(int)
        
        # Timing
        self.last_heartbeat = time.time()
        self.election_timeout = self._random_timeout()
    
    def _random_timeout(self):
        """Generate random election timeout (150-300ms)"""
        return random.uniform(0.15, 0.3)
    
    def request_vote(self, term, candidate_id, last_log_index, last_log_term):
        """Handle vote request from candidate"""
        if term < self.current_term:
            return (self.current_term, False)
        
        if term > self.current_term:
            self.current_term = term
            self.state = State.FOLLOWER
            self.voted_for = None
        
        # Check if candidate's log is at least as up-to-date
        my_last_term = self.log[-1].term if self.log else 0
        log_ok = (last_log_term > my_last_term or 
                  (last_log_term == my_last_term and last_log_index >= len(self.log) - 1))
        
        if (self.voted_for is None or self.voted_for == candidate_id) and log_ok:
            self.voted_for = candidate_id
            self.last_heartbeat = time.time()
            return (self.current_term, True)
        
        return (self.current_term, False)
    
    def append_entries(self, term, leader_id, prev_log_index, prev_log_term, entries, leader_commit):
        """Handle append entries from leader"""
        if term < self.current_term:
            return (self.current_term, False)
        
        if term > self.current_term:
            self.current_term = term
            self.state = State.FOLLOWER
            self.voted_for = None
        
        self.last_heartbeat = time.time()
        
        # Check if log contains entry at prev_log_index with matching term
        if prev_log_index >= len(self.log) or self.log[prev_log_index].term != prev_log_term:
            return (self.current_term, False)
        
        # Append new entries
        if entries:
            for i, entry in enumerate(entries):
                log_index = prev_log_index + 1 + i
                if log_index < len(self.log):
                    if self.log[log_index].term != entry.term:
                        # Delete conflicting entries
                        self.log = self.log[:log_index]
                        self.log.append(entry)
                else:
                    self.log.append(entry)
        
        # Update commit index
        if leader_commit > self.commit_index:
            self.commit_index = min(leader_commit, len(self.log) - 1)
        
        return (self.current_term, True)
    
    def start_election(self):
        """Start leader election"""
        self.state = State.CANDIDATE
        self.current_term += 1
        self.voted_for = self.node_id
        self.last_heartbeat = time.time()
        
        votes = 1  # Vote for self
        last_log_index = len(self.log) - 1
        last_log_term = self.log[-1].term if self.log else 0
        
        # Request votes from peers (simplified)
```

```python
# TIME: O(n) where n = number of Redis instances SPACE: O(1)
import time
import uuid

from threading import Thread

class DistributedLock:
    """
    Redlock-inspired distributed lock implementation.
    Uses multiple Redis instances for fault tolerance.
    """
    def __init__(self, redis_clients, retry_count=3, retry_delay=200):
        """
        redis_clients: list of Redis client instances
        retry_count: number of retries to acquire lock
        retry_delay: delay between retries in ms
        """
        self.redis_clients = redis_clients
        self.retry_count = retry_count
        self.retry_delay = retry_delay / 1000.0  # Convert to seconds
        self.quorum = len(redis_clients) // 2 + 1
    
    def acquire_lock(self, resource, ttl, timeout=10):
        """
        Acquire distributed lock on resource.
        
        resource: name of resource to lock
        ttl: lock timeout in milliseconds
        timeout: max time to try acquiring lock in seconds
        
        Returns: lock token if successful, None otherwise
        """
        lock_token = str(uuid.uuid4())
        start_time = time.time()
        
        for attempt in range(self.retry_count):
            # Try to acquire lock on all instances
            acquired_count = 0
            
            for client in self.redis_clients:
                try:
                    # SET key value NX PX ttl
                    # NX: only set if not exists
                    # PX: set with expiration in ms
                    if self._acquire_on_instance(client, resource, lock_token, ttl):
                        acquired_count += 1
                except Exception as e:
                    print(f"Error acquiring lock on instance: {e}")
            
            # Calculate time elapsed
            elapsed = (time.time() - start_time) * 1000
            
            # Check if we got quorum and there's still time left
            if acquired_count >= self.quorum and elapsed < ttl:
                # Successfully acquired lock
                return lock_token
            else:
                # Failed, release locks on all instances
                self._release_lock_internal(resource, lock_token)
            
            # Check if we've exceeded timeout
            if time.time() - start_time > timeout:
                break
            
            # Wait before retry
            time.sleep(self.retry_delay)
        
        return None
    
    def release_lock(self, resource, lock_token):
        """
        Release distributed lock.
        Uses Lua script to ensure atomic release.
        """
        self._release_lock_internal(resource, lock_token)
    
    def _acquire_on_instance(self, client, resource, lock_token, ttl):
        """Try to acquire lock on single Redis instance"""
        key = f"lock:{resource}"
        
        # SET key value NX PX ttl
        # In practice, would use: client.set(key, lock_token, nx=True, px=ttl)
        # Simplified version:

            # Simulate Redis SET NX PX

            return result is not None
        except:
            return False

        """Release lock on all instances if token matches"""
        key = f"lock:{resource}"
        
        # Lua script for atomic release
        lua_script = """
        if redis.call("get", KEYS[1]) == ARGV[1] then

        else
            return 0
```
