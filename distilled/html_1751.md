## 1. css_pseudo_element_first_line

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>CSS ::first-line Pseudo-Element</title>
  <style>
    /* PROBLEM: Style the first line of text differently */
    /* TIME: O(1) SPACE: O(1) */
    
    .paragraph {
      font-family: Georgia, serif;
      font-size: 16px;
      line-height: 1.6;
      color: #333;
      max-width: 600px;
    }
    
    /* ::first-line pseudo-element targets only the first line */
    .paragraph::first-line {
      font-weight: bold;
      font-size: 1.2em;
      color: #2c3e50;
      text-transform: uppercase;
      letter-spacing: 1px;
    }
    
    /* Works with responsive layouts - adapts to line breaks */
    .article::first-line {
      background: linear-gradient(90deg, #3498db, #9b59b6);
      -webkit-background-clip: text;
      -webkit-text-fill-color: transparent;
      background-clip: text;
    }
  </style>
</head>
<body>
  <!-- 
    Why: ::first-line allows styling only the first rendered line of text.
    Useful for creating magazine-style typography without extra markup.
    Only certain CSS properties apply: font, color, background, word-spacing,
    letter-spacing, text-decoration, text-transform, line-height.
  -->
  
  <p class="paragraph">
    This is the first line that will be styled differently. The styling 
    automatically adjusts based on the container width. Any text that wraps 
    to subsequent lines will have normal styling.
  </p>
  
  <p class="article">
    Gradient text on the first line creates visual hierarchy. This technique
    works across modern browsers and adapts responsively to different screen sizes.
  </p>
</body>
</html>
```

## 3. css_margin_collapsing

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>CSS Margin Collapsing</title>
  <style>
    /* PROBLEM: Adjacent vertical margins collapse into single margin */
    /* TIME: O(1) SPACE: O(1) */
    
    .container {
      border: 2px dashed #ccc;
      padding: 10px;
      margin-bottom: 20px;
    }
    
    /* Margin collapsing example */
    .box-top {
      background: #3498db;
      margin-bottom: 30px; /* Will collapse with below */
      padding: 20px;
      color: white;
    }
    
    .box-bottom {
      background: #e74c3c;
      margin-top: 20px; /* Collapses to max(30px, 20px) = 30px */
      padding: 20px;
      color: white;
    }
    
    /* Preventing collapse with BFC */
    .bfc-container {
      overflow: hidden; /* Creates Block Formatting Context */
    }
    
    .no-collapse-top {
      background: #27ae60;
      margin-bottom: 30px;
      padding: 20px;
      color: white;
    }
    
    .no-collapse-bottom {
      background: #f39c12;
      margin-top: 20px; /* Won't collapse due to BFC */
      padding: 20px;
      color: white;
    }
    
    /* Parent-child margin collapse */
    .parent {
      background: #ecf0f1;
      /* No border/padding = child margin collapses through */
    }
    
    .child {
      background: #9b59b6;
      margin-top: 40px; /* Collapses with parent unless separated */
      padding: 15px;
      color: white;
    }
    
    .parent-fixed {
      background: #ecf0f1;
      padding-top: 1px; /* Prevents collapse */
    }
  </style>
</head>
<body>
  <!--
    Why: Understanding margin collapsing prevents layout bugs.
    
    Rules:
    1. Only vertical margins collapse (top/bottom)
    2. Adjacent siblings: larger margin wins
    3. Parent-child: margins collapse through empty parents
    4. Block Formatting Context (BFC) prevents collapse
  -->
  
  <h2>Collapsed Margins (30px between boxes)</h2>
  <div class="container">
    <div class="box-top">Box with margin-bottom: 30px</div>
    <div class="box-bottom">Box with margin-top: 20px</div>
    <p>Actual gap: max(30, 20) = 30px (not 50px)</p>
  </div>
  
  <h2>Prevented with BFC (50px between boxes)</h2>
  <div class="container bfc-container">
    <div class="no-collapse-top">Box with margin-bottom: 30px</div>
    <div class="no-collapse-bottom">Box with margin-top: 20px</div>
    <p>Actual gap: 30 + 20 = 50px</p>
  </div>
  
  <h2>Parent-Child Collapse</h2>
  <div class="parent">
    <div class="child">Child margin escapes parent</div>
  </div>
</body>
</html>
```

## 6. js_await_keyword

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>JavaScript await Keyword</title>
</head>
<body>
  <div id="output"></div>

  <script>
    // PROBLEM: Handle asynchronous operations with cleaner syntax
    // TIME: O(async) SPACE: O(1)
    
    const output = document.getElementById('output');
    
    // Simulate async API call
    function fetchUser(id) {
      return new Promise((resolve) => {
        setTimeout(() => {
          resolve({ id, name: `User ${id}`, email: `user${id}@example.com` });
        }, 1000);
      });
    }
    
    function fetchPosts(userId) {
      return new Promise((resolve) => {
        setTimeout(() => {
          resolve([
            { id: 1, title: 'First Post', userId },
            { id: 2, title: 'Second Post', userId }
          ]);
        }, 800);
      });
    }
    
    // Using await - must be inside async function
    async function loadUserData() {
      output.innerHTML = '<p>Loading user...</p>';
      
      try {
        // await pauses execution until Promise resolves
        const user = await fetchUser(1);
        output.innerHTML += `<p>User: ${user.name} (${user.email})</p>`;
        
        output.innerHTML += '<p>Loading posts...</p>';
        const posts = await fetchPosts(user.id);
        
        output.innerHTML += '<p>Posts:</p><ul>';
        posts.forEach(post => {
          output.innerHTML += `<li>${post.title}</li>`;
        });
        output.innerHTML += '</ul>';
        
        return { user, posts };
      } catch (error) {
        output.innerHTML += `<p style="color: red;">Error: ${error.message}</p>`;
      }
    }
    
    // Parallel execution with Promise.all
    async function loadAllDataParallel() {
      output.innerHTML += '<h3>Parallel Loading:</h3>';
      
      const startTime = Date.now();
      
      // Both requests run simultaneously
      const [users, posts] = await Promise.all([
        fetchUser(2),
        fetchPosts(2)
      ]);
      
      const elapsed = Date.now() - startTime;
      output.innerHTML += `<p>Loaded in ${elapsed}ms (parallel)</p>`;
    }
    
    // Execute
    loadUserData().then(() => {
      loadAllDataParallel();
    });
    
    /*
    Why: await makes async code look synchronous, improving readability.
    - Must be used inside async function
    - Pauses execution until Promise settles
    - For parallel work, use Promise.all with await
    - Try/catch works naturally for error handling
    */
  </script>
</body>
</html>
```

```html
      const removed = numbers.splice(index, 1);
      log('\nAfter removing 30: ' + JSON.stringify(numbers));
    }
    
    // Strict equality comparison

    - Uses strict equality (===)
    - Second parameter: start search position
    - For objects, use findIndex() instead

  </script>
```

## 8. js_object_fromEntries

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>JavaScript Object.fromEntries()</title>
</head>
<body>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Convert key-value pairs array to object
    // TIME: O(n) SPACE: O(n)
    
    const log = (msg) => document.getElementById('output').textContent += msg + '\n';
    
    // Basic usage: array of entries to object
    const entries = [
      ['name', 'Alice'],
      ['age', 30],
      ['city', 'New York']
    ];
    
    const obj = Object.fromEntries(entries);
    log('From entries: ' + JSON.stringify(obj));
    // { name: 'Alice', age: 30, city: 'New York' }
    
    // Round-trip with Object.entries()
    const original = { a: 1, b: 2, c: 3 };
    const entriesArr = Object.entries(original);
    const reconstructed = Object.fromEntries(entriesArr);
    log('\nRound-trip: ' + JSON.stringify(reconstructed));
    
    // Transform object: filter entries
    const scores = { alice: 85, bob: 62, charlie: 91, dave: 45 };
    const passing = Object.fromEntries(
      Object.entries(scores).filter(([name, score]) => score >= 70)
    );
    log('\nPassing scores: ' + JSON.stringify(passing));
    
    // Transform object: map values
    const prices = { apple: 100, banana: 50, orange: 75 };
    const withTax = Object.fromEntries(
      Object.entries(prices).map(([item, price]) => [item, price * 1.1])
    );
    log('\nPrices with tax: ' + JSON.stringify(withTax));
    
    // Convert Map to Object
    const map = new Map([
      ['x', 10],
      ['y', 20],
      ['z', 30]
    ]);
    const fromMap = Object.fromEntries(map);
    log('\nFrom Map: ' + JSON.stringify(fromMap));
    
    // URL query string parsing
    const queryString = 'name=John&age=25&city=Boston';
    const params = Object.fromEntries(
      new URLSearchParams(queryString)
    );
    log('\nParsed query: ' + JSON.stringify(params));
    
    // Swap keys and values
    const roles = { admin: 'A', user: 'U', guest: 'G' };
    const swapped = Object.fromEntries(
      Object.entries(roles).map(([k, v]) => [v, k])
    );
    log('\nSwapped: ' + JSON.stringify(swapped));
    
    /*
    Why: fromEntries() is inverse of Object.entries(), useful for:
    - Transforming objects (filter, map entries)
    - Converting Map to Object
    - Parsing query strings
    - Object manipulation pipelines
    */
  </script>
</body>
</html>
```

## 9. js_string_charCodeAt

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>JavaScript String.charCodeAt()</title>
</head>
<body>
  <pre id="output"></pre>

  <script>
    // PROBLEM: Get Unicode code unit of character at position
    // TIME: O(1) SPACE: O(1)
    
    const log = (msg) => document.getElementById('output').textContent += msg + '\n';
    
    const str = 'Hello';
    
    // Basic usage
    log('String: ' + str);
    log('charCodeAt(0): ' + str.charCodeAt(0));  // 72 ('H')
    log('charCodeAt(1): ' + str.charCodeAt(1));  // 101 ('e')
    log('charCodeAt(4): ' + str.charCodeAt(4));  // 111 ('o')
    log('charCodeAt(5): ' + str.charCodeAt(5));  // NaN (out of range)
    
    // Character code reference
    log('\nCharacter codes:');
    log("'A' = " + 'A'.charCodeAt(0));  // 65
    log("'a' = " + 'a'.charCodeAt(0));  // 97
    log("'0' = " + '0'.charCodeAt(0));  // 48
    log("' ' = " + ' '.charCodeAt(0));  // 32
    
    // Practical use: check if character is uppercase
    function isUpperCase(char) {
      const code = char.charCodeAt(0);
      return code >= 65 && code <= 90;  // A-Z range
    }
    
    log('\nCase check:');
    log("'A' is upper: " + isUpperCase('A'));  // true
    log("'a' is upper: " + isUpperCase('a'));  // false
    
    // Shift cipher (Caesar cipher)
    function caesarShift(text, shift) {
      return text.split('').map(char => {
        const code = char.charCodeAt(0);
        // Only shift letters
        if (code >= 65 && code <= 90) {  // Uppercase
          return String.fromCharCode(((code - 65 + shift) % 26 + 26) % 26 + 65);
        }
        if (code >= 97 && code <= 122) {  // Lowercase
          return String.fromCharCode(((code - 97 + shift) % 26 + 26) % 26 + 97);
        }
        return char;
      }).join('');
    }
    
    log('\nCaesar cipher (shift 3):');
    const encrypted = caesarShift('Hello World', 3);
    log('Encrypted: ' + encrypted);
    log('Decrypted: ' + caesarShift(encrypted, -3));
    
    // Note: charCodeAt returns UTF-16 code unit
    // For Unicode code points > 0xFFFF, use codePointAt()
    const emoji = '😀';
    log('\nEmoji handling:');
    log('charCodeAt(0): ' + emoji.charCodeAt(0));  // Surrogate pair first half
    log('codePointAt(0): ' + emoji.codePointAt(0));  // Full code point (128512)
    
    /*
    Why: charCodeAt() accesses character encoding for:
    - Character validation (uppercase, digit, etc.)
    - Encoding/decoding operations
    - Simple cryptography
    - String algorithms
    
    For full Unicode support, prefer codePointAt()
    */
  </script>
</body>
</html>
```
