## 1. css_pseudo_class_only_of_type

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>:only-of-type Pseudo-class</title>
    <style>
        /* Style only paragraph that is the only paragraph in its parent */
        p:only-of-type {
            background-color: #e3f2fd;
            padding: 15px;
            border-left: 4px solid #2196F3;
        }
        
        /* Style only span that is the only span in its parent */
        span:only-of-type {
            color: #e91e63;
            font-weight: bold;
        }
        
        /* Container styling for demo */
        .container {
            margin: 20px;
            padding: 15px;
            border: 1px solid #ccc;
            border-radius: 8px;
        }
    </style>
</head>
<body>
    <!-- Problem: Select and style elements that are the only child of their type -->
    <!-- Time: O(1) Space: O(1) -->
    
    <!-- Why: :only-of-type targets elements with no siblings of the same type -->
    
    <div class="container">
        <p>This paragraph is the only &lt;p&gt; in this div, so it gets styled.</p>
        <span>Only span here</span>
    </div>
    
    <div class="container">
        <p>First paragraph</p>
        <p>Second paragraph - neither gets styled (siblings exist)</p>
    </div>
    
    <div class="container">
        <p>Only paragraph</p>
        <span>First span</span>
        <span>Second span - spans won't be styled</span>
    </div>
</body>
</html>
```

**Why**: `:only-of-type` selects elements that have no siblings of the same element type within their parent, useful for styling unique content blocks.

## 6. js_destructuring_arrays

```html
<!DOCTYPE html>
<html lang="en">
<head>

    <title>Array Destructuring</title>

    <h1>Array Destructuring Examples</h1>
    <pre id="output"></pre>
    
    <script>
        // Problem: Extract values from arrays into variables
        // Time: O(n) where n is array length Space: O(n)
        
        // Why: Array destructuring provides clean syntax to unpack array values
        
        const log = (...args) => {
            document.getElementById('output').textContent += args.join(' ') + '\n';
        };
        
        // Basic destructuring
        const colors = ['red', 'green', 'blue'];
        const [first, second, third] = colors;

        // Skipping elements
        const numbers = [1, 2, 3, 4, 5];
        const [a, , c, , e] = numbers;
        log('Skipping:', a, c, e); // 1 3 5
        
        // Default values
        const items = ['apple'];
        const [fruit1, fruit2 = 'banana'] = items;
        log('Default values:', fruit1, fruit2); // apple banana
        
        // Rest operator
        const scores = [95, 88, 76, 92, 85];
        const [top, ...rest] = scores;
        log('Rest operator:', top, rest); // 95 [88, 76, 92, 85]
        
        // Swapping variables
        let x = 10, y = 20;
        log('Before swap:', x, y);
        [x, y] = [y, x];
        log('After swap:', x, y); // 20 10
        
        // Nested array destructuring
        const nested = [1, [2, 3], 4];
        const [p, [q, r], s] = nested;
        log('Nested:', p, q, r, s); // 1 2 3 4
        
        // Function returning array
        function getCoordinates() {
            return [100, 200];
        }
        const [xPos, yPos] = getCoordinates();
        log('Coordinates:', xPos, yPos); // 100 200
        
        // Destructuring in function parameters
        function sum([a, b, c]) {
            return a + b + c;
        }
        log('Sum:', sum([1, 2, 3])); // 6
        
        // Iterating with destructuring
        const points = [[1, 2], [3, 4], [5, 6]];
        log('Points:');
        for (const [x, y] of points) {
            log(`  (${x}, ${y})`);
        }
        
        // Destructuring with length
        const arr = ['a', 'b', 'c'];
        const { length, 0: firstItem, [arr.length - 1]: lastItem } = arr;
        log('Array as object:', length, firstItem, lastItem); // 3 a c
    </script>
</body>
</html>
```

**Why**: Array destructuring provides concise syntax to extract values into variables, supports defaults, rest operator, nesting, and variable swapping.

## 7. js_callback_hell

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Callback Hell Example</title>
</head>
<body>
    <h1>Callback Hell vs Solutions</h1>
    <pre id="output"></pre>
    
    <script>
        // Problem: Deeply nested callbacks create unreadable "pyramid of doom"
        // Time: O(1) for each callback Space: O(n) for call stack depth
        
        // Why: Callback hell occurs when async operations depend on each other
        
        const log = (...args) => {
            document.getElementById('output').textContent += args.join(' ') + '\n';
        };
        
        // Simulated async functions
        function getUser(id, callback) {
            setTimeout(() => callback(null, { id, name: 'John' }), 100);
        }
        
        function getPosts(userId, callback) {
            setTimeout(() => callback(null, [{ id: 1, title: 'Post 1' }]), 100);
        }
        
        function getComments(postId, callback) {
            setTimeout(() => callback(null, [{ id: 1, text: 'Comment 1' }]), 100);
        }
        
        // ========== CALLBACK HELL (Anti-pattern) ==========
        log('=== Callback Hell (Avoid This) ===');
        
        getUser(1, (err, user) => {
            if (err) {
                log('Error:', err);
                return;
            }
            log('User:', user.name);
            
            getPosts(user.id, (err, posts) => {
                if (err) {
                    log('Error:', err);
                    return;
                }
                log('Posts:', posts.length);
                
                getComments(posts[0].id, (err, comments) => {
                    if (err) {
                        log('Error:', err);
                        return;
                    }
                    log('Comments:', comments.length);
                    log('Done! (3 levels deep)');
                });
            });
        });
        
        // ========== PROMISES (Better) ==========
        function getUserPromise(id) {
            return new Promise((resolve) => {
                setTimeout(() => resolve({ id, name: 'Jane' }), 100);
            });
        }
        
        function getPostsPromise(userId) {
            return new Promise((resolve) => {
                setTimeout(() => resolve([{ id: 1, title: 'Post 1' }]), 100);
            });
        }
        
        function getCommentsPromise(postId) {
            return new Promise((resolve) => {
                setTimeout(() => resolve([{ id: 1, text: 'Comment 1' }]), 100);
            });
        }
        
        setTimeout(() => {
            log('\n=== Promises (Better) ===');
            
            getUserPromise(1)
                .then(user => {
                    log('User:', user.name);
                    return getPostsPromise(user.id);
                })
                .then(posts => {
                    log('Posts:', posts.length);
                    return getCommentsPromise(posts[0].id);
                })
                .then(comments => {
                    log('Comments:', comments.length);
                    log('Done! (Flat chain)');
                })
                .catch(err => log('Error:', err));
        }, 500);
        
        // ========== ASYNC/AWAIT (Best) ==========
        setTimeout(async () => {
            log('\n=== Async/Await (Best) ===');
            
            try {
                const user = await getUserPromise(1);
                log('User:', user.name);
                
                const posts = await getPostsPromise(user.id);
                log('Posts:', posts.length);
                
                const comments = await getCommentsPromise(posts[0].id);
                log('Comments:', comments.length);
                log('Done! (Linear code)');
            } catch (err) {
                log('Error:', err);
            }
        }, 1000);
    </script>
</body>
</html>
```

**Why**: Callback hell creates deeply nested, hard-to-read code. Solutions include named functions, Promises, and async/await for flat, readable async code.

```html
        // Problem: Extract a portion of an array without modifying the original

        // Why: slice() returns a shallow copy of a portion of an array
        
        const log = (...args) => {

        };

        // Basic slice - from index 1 to end

        // ['banana', 'cherry', 'date', 'elderberry']
        
        // Slice with start and end (end is exclusive)
        log('slice(1, 4):', fruits.slice(1, 4));
        // ['banana', 'cherry', 'date']
        
        // Negative indices - count from end
        log('slice(-3):', fruits.slice(-3));
        // ['cherry', 'date', 'elderberry']
        
        log('slice(-3, -1):', fruits.slice(-3, -1));
        // ['cherry', 'date']
        
        // No arguments - shallow copy entire array

        log('slice() copy:', copy);
        log('Same reference?', fruits === copy); // false
        
        // Converting array-like objects to arrays
        function exampleFunction() {
            // arguments is array-like, not a real array
            const args = Array.prototype.slice.call(arguments);
            log('Arguments as array:', args);
            log('Has map method?', typeof args.map === 'function'); // true
        }
        exampleFunction('a', 'b', 'c');
        
        // Practical use: pagination
        const items = Array.from({ length: 25 }, (_, i) => `Item ${i + 1}`);
        const pageSize = 5;
        
        log('\nPagination example:');
        for (let page = 0; page < 3; page++) {
            const start = page * pageSize;
            const pageItems = items.slice(start, start + pageSize);
            log(`Page ${page + 1}:`, pageItems.join(', '));
        }
        
        // Practical use: getting first N elements
        log('\nFirst 3:', fruits.slice(0, 3));
        
        // Practical use: getting last N elements
        log('Last 2:', fruits.slice(-2));
        
        // Slice with empty arrays
        const empty = [];
        log('Empty slice():', empty.slice());
        
        // Out of bounds handling
        log('slice(10, 20):', fruits.slice(10, 20)); // []
        log('slice(-100, 2):', fruits.slice(-100, 2)); // ['apple', 'banana']
    </script>

```

**Why**: `slice()` extracts array portions without modifying the original, supports negative indices, and is useful for pagination, copying, and array-like conversion.

## 9. js_object_values

```html

    <title>Object.values()</title>

<body>
    <h1>Object.values() Examples</h1>
    <pre id="output"></pre>
    
    <script>
        // Problem: Get all values from an object as an array
        // Time: O(n) where n is number of properties Space: O(n)
        
        // Why: Object.values() returns an array of own enumerable property values
        
        const log = (...args) => {
            document.getElementById('output').textContent += args.join(' ') + '\n';
        };
        
        // Basic usage
        const person = {
            name: 'Alice',
            age: 30,
            city: 'New York'
        };
        
        log('Object:', person);
        log('Values:', Object.values(person));
        // ['Alice', 30, 'New York']
        
        // Iterating over values
        log('\nIterating:');
        Object.values(person).forEach((value, index) => {
            log(`  ${index}: ${value}`);
        });
        
        // Sum values
        const scores = { math: 95, science: 88, history: 92 };
        const total = Object.values(scores).reduce((sum, score) => sum + score, 0);
```
