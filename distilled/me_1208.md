# Python Code Collection - Suffix Array

## Suffix Array
```python
# PROBLEM: Build a suffix array for efficient string pattern matching and substring operations
# TIME: O(n log n) for construction, O(m log n) for pattern search where m is pattern length
# SPACE: O(n) for storing suffix indices and ranks
def build_suffix_array(text):
    """
    Build suffix array using prefix doubling algorithm.
    A suffix array is a sorted array of all suffixes of a string.
    It enables efficient string operations like pattern matching,
    longest common substring, and string similarity analysis.
    
    Args:
        text: Input string to build suffix array for
        
    Returns:
        List of starting indices of suffixes in sorted order
    """
    n = len(text)
    if n == 0:
        return []
    
    # Initialize: sort by first character
    # sa[i] = starting index of i-th smallest suffix
    sa = list(range(n))
    
    # rank[i] = rank of suffix starting at index i
    # Initially just the ASCII value of the character
    rank = [ord(text[i]) for i in range(n)]
    
    # Temporary rank array for comparison
    temp_rank = [0] * n
    
    # k is the length of prefixes being compared
    k = 1
    
    while k < n:
        # Sort suffixes based on first k characters
        # Using rank for first half and rank of (i + k) for second half
        sa.sort(key=lambda i: (
            rank[i],
            rank[i + k] if i + k < n else -1
        ))
        
        # Re-rank based on sorted order
        temp_rank[sa[0]] = 0
        for i in range(1, n):
            # Get ranks of current and previous suffix
            curr = (rank[sa[i]], rank[sa[i] + k] if sa[i] + k < n else -1)
            prev = (rank[sa[i-1]], rank[sa[i-1] + k] if sa[i-1] + k < n else -1)
            
            # Same rank if first k chars are same, otherwise increment
            temp_rank[sa[i]] = temp_rank[sa[i-1]] + (1 if curr > prev else 0)
        
        # Copy temp ranks to main rank array
        rank = temp_rank[:]
        
        # Double the comparison length
        k *= 2
    
    return sa

def pattern_search_suffix_array(text, pattern, sa):
    """
    Search for pattern using binary search on suffix array.
    This is much faster than naive search for multiple queries.
    
    Args:
        text: Original text
        pattern: Pattern to search for
        sa: Pre-built suffix array
        
    Returns:
        List of starting positions where pattern occurs
    """
    n = len(text)
    m = len(pattern)
    
    if m == 0 or n == 0 or m > n:
        return []
    
    # Binary search for lower bound (first occurrence)
    left, right = 0, n - 1
    first = -1
    
    while left <= right:
        mid = (left + right) // 2
        # Get suffix starting at sa[mid]
        suffix = text[sa[mid]:sa[mid]+m]
        
        if suffix == pattern:
            first = mid
            right = mid - 1  # Continue searching left
        elif suffix < pattern:
            left = mid + 1
        else:
            right = mid - 1
    
    if first == -1:
        return []  # Pattern not found
    
    # Binary search for upper bound (last occurrence)
    left, right = first, n - 1
    last = first
    
    while left <= right:
        mid = (left + right) // 2
        suffix = text[sa[mid]:sa[mid]+m]
        
        if suffix == pattern:
            last = mid
            left = mid + 1  # Continue searching right
        elif suffix < pattern:
            left = mid + 1
        else:
            right = mid - 1
    
    # Collect all positions where pattern occurs
    positions = [sa[i] for i in range(first, last + 1)]
    return sorted(positions)

def build_lcp_array(text, sa):
    """
    Build Longest Common Prefix array from suffix array.
    LCP[i] = length of longest common prefix between 
    suffixes at sa[i] and sa[i-1].
    
    Args:
        text: Original text
        sa: Suffix array
        
    Returns:
        LCP array where LCP[0] = 0 (no previous suffix)
    """
    n = len(text)
    if n == 0:
        return []
    
    lcp = [0] * n
    
    # Build inverse suffix array
    # inv_sa[i] = position of suffix starting at i in the suffix array
    inv_sa = [0] * n
    for i in range(n):
        inv_sa[sa[i]] = i
    
    # Build LCP array using Kasai's algorithm
    k = 0
    for i in range(n):
        # If this is the first suffix in sorted order, LCP is 0
        if inv_sa[i] == 0:
            k = 0
            continue
        
        # Get the previous suffix in sorted order
        j = sa[inv_sa[i] - 1]
        
        # Compare characters starting from position k
        while i + k < n and j + k < n and text[i + k] == text[j + k]:
            k += 1
        
        # Store LCP value
        lcp[inv_sa[i]] = k
        
        # Next suffix will have at least k-1 common characters
        if k > 0:
            k -= 1
    
    return lcp

def longest_repeated_substring(text, sa=None, lcp=None):
    """
    Find the longest substring that appears at least twice.
    Uses suffix array and LCP array for efficient computation.
    
    Args:
        text: Input text
        sa: Optional pre-computed suffix array
        lcp: Optional pre-computed LCP array
        
    Returns:
        The longest repeated substring (empty if none exists)
    """
    if sa is None:
        sa = build_suffix_array(text)
    if lcp is None:
        lcp = build_lcp_array(text, sa)
    
    if len(lcp) == 0:
        return ""
    
    # Find maximum LCP value
    max_lcp = max(lcp)
    
    if max_lcp == 0:
        return ""  # No repeated substring
    
    # Find the position of maximum LCP
    idx = lcp.index(max_lcp)
    
    # Extract the substring
    start = sa[idx]
    return text[start:start + max_lcp]

# Example usage
if __name__ == "__main__":
    # Example 1: Build and search
    text = "banana"
    sa = build_suffix_array(text)
    print(f"Text: {text}")
    print(f"Suffix Array: {sa}")
    print("Suffixes in sorted order:")
    for i, idx in enumerate(sa):
        print(f"  {i}: sa[{idx}] = '{text[idx:]}'")
    
    # Example 2: Pattern search
    pattern = "ana"
    positions = pattern_search_suffix_array(text, pattern, sa)
    print(f"\nPattern '{pattern}' found at positions: {positions}")
    
    # Example 3: LCP array
    lcp = build_lcp_array(text, sa)
    print(f"\nLCP Array: {lcp}")
    
    # Example 4: Longest repeated substring
    lrs = longest_repeated_substring(text, sa, lcp)
    print(f"Longest repeated substring: '{lrs}'")
    
    # Example 5: Multiple pattern searches
    text2 = "abracadabra"
    sa2 = build_suffix_array(text2)
    patterns = ["abra", "cad", "ra"]
    print(f"\nText: {text2}")
    for pat in patterns:
        pos = pattern_search_suffix_array(text2, pat, sa2)
        print(f"Pattern '{pat}': {pos}")
```

**Why**: Suffix arrays are fundamental data structures in string processing with applications in bioinformatics (genome analysis), text indexing, data compression (BWT), and search engines. They enable O(m log n) pattern matching vs O(mn) naive search, making them essential for large-scale text processing. Understanding suffix arrays deepens knowledge of string algorithms, binary search applications, and the connection between sorting and pattern matching—core concepts that appear in system design interviews and competitive programming.
