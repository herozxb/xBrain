## Flyweight Pattern Variants

```python
# PROBLEM: Minimize memory usage by sharing common parts of state
# TIME: O(1) for flyweight access SPACE: O(n) for shared states
class Flyweight:
    """Shared object that stores intrinsic state."""
    
    def __init__(self, shared_state: tuple):
        self._shared_state = shared_state
    
    def operation(self, unique_state: str) -> None:
        """Execute operation combining shared and unique state."""
        s = ', '.join(self._shared_state)
        print(f"Flyweight: Displaying shared ({s}) and unique ({unique_state}) state.")


class FlyweightFactory:
    """Factory that manages flyweight objects."""
    
    _flyweights: dict = {}
    
    def __init__(self, initial_flyweights: list):
        for state in initial_flyweights:
            self._flyweights[self._get_key(state)] = Flyweight(state)
    
    @staticmethod
    def _get_key(state: tuple) -> str:
        """Generate unique key for flyweight state."""
        return '_'.join(sorted(state))
    
    def get_flyweight(self, shared_state: tuple) -> Flyweight:
        """Get existing flyweight or create new one."""
        key = self._get_key(shared_state)
        if key not in self._flyweights:
            print("FlyweightFactory: Creating new flyweight.")
            self._flyweights[key] = Flyweight(shared_state)
        else:
            print("FlyweightFactory: Reusing existing flyweight.")
        return self._flyweights[key]
    
    def list_flyweights(self) -> None:
        """Display all cached flyweights."""
        count = len(self._flyweights)
        print(f"FlyweightFactory: {count} flyweights cached.")
        print('\n'.join(self._flyweights.keys()))


# Context class with extrinsic state
class Context:
    """Contains extrinsic state and reference to flyweight."""
    
    def __init__(self, flyweight: Flyweight, unique_state: str):
        self._flyweight = flyweight
        self._unique_state = unique_state
    
    def operation(self) -> None:
        """Delegate to flyweight with unique state."""
        self._flyweight.operation(self._unique_state)
```

**Why**: Reduces memory footprint when many similar objects exist by sharing intrinsic state. Useful for text editors, games, and caching scenarios.

## State Pattern Variants

```python
# PROBLEM: Allow object behavior to change based on internal state
# TIME: O(1) for state transitions SPACE: O(n) for state objects
from abc import ABC, abstractmethod


class State(ABC):
    """Abstract base state defining interface."""
    
    @abstractmethod
    def handle(self, context: 'Context') -> None:
        """Handle request in current state."""
        pass


class ConcreteStateA(State):
    """Initial state implementation."""
    
    def handle(self, context: 'Context') -> None:
        print("State A handling request.")
        print("Transitioning to State B.")
        context.state = ConcreteStateB()


class ConcreteStateB(State):
    """Alternative state implementation."""
    
    def handle(self, context: 'Context') -> None:
        print("State B handling request.")
        print("Transitioning to State A.")
        context.state = ConcreteStateA()


class Context:
    """Maintains current state and delegates requests."""
    
    def __init__(self, state: State):
        self._state = state
    
    @property
    def state(self) -> State:
        return self._state
    
    @state.setter
    def state(self, state: State) -> None:
        self._state = state
        print(f"Context: State changed to {state.__class__.__name__}")
    
    def request(self) -> None:
        """Handle request via current state."""
        self._state.handle(self)


# Hierarchical State Machine Variant
class HierarchicalState(State):
    """State that can have parent states."""
    
    def __init__(self, parent: State = None):
        self._parent = parent
    
    def handle(self, context: 'Context') -> None:
        """Handle or delegate to parent if not handled."""
        if not self._handle_local(context) and self._parent:
            self._parent.handle(context)
    
    def _handle_local(self, context: 'Context') -> bool:
        """Override in subclasses. Return True if handled."""
        return False
```

**Why**: Eliminates large conditionals, encapsulates state-specific behavior, and makes state transitions explicit. Ideal for workflows, game states, and protocol handling.

## Strategy Pattern Variants

```python
# PROBLEM: Define family of algorithms, make them interchangeable
# TIME: O(1) for strategy swap SPACE: O(1) per strategy
from abc import ABC, abstractmethod
from typing import List


class SortStrategy(ABC):
    """Abstract sorting strategy."""
    
    @abstractmethod
    def sort(self, data: List[int]) -> List[int]:
        """Sort the data."""
        pass


class BubbleSort(SortStrategy):
    """Bubble sort implementation."""
    
    def sort(self, data: List[int]) -> List[int]:
        n = len(data)
        arr = data.copy()
        for i in range(n):
            for j in range(0, n - i - 1):
                if arr[j] > arr[j + 1]:
                    arr[j], arr[j + 1] = arr[j + 1], arr[j]
        return arr


class QuickSort(SortStrategy):
    """Quick sort implementation."""
    
    def sort(self, data: List[int]) -> List[int]:
        if len(data) <= 1:
            return data
        pivot = data[len(data) // 2]
        left = [x for x in data if x < pivot]
        middle = [x for x in data if x == pivot]
        right = [x for x in data if x > pivot]
        return self.sort(left) + middle + self.sort(right)


class Context:
    """Context that uses a strategy."""
    
    def __init__(self, strategy: SortStrategy):
        self._strategy = strategy
    
    @property
    def strategy(self) -> SortStrategy:
        return self._strategy
    
    @strategy.setter
    def strategy(self, strategy: SortStrategy) -> None:
        self._strategy = strategy
    
    def do_business_logic(self, data: List[int]) -> List[int]:
        """Execute strategy on data."""
        print(f"Context: Sorting data using {self._strategy.__class__.__name__}")
        return self._strategy.sort(data)


# Functional Strategy Variant
def create_strategy(mode: str) -> callable:
    """Factory returning strategy function."""
    
    def bubble(data: list) -> list:
        arr = data.copy()
        for i in range(len(arr)):
            for j in range(len(arr) - i - 1):
                if arr[j] > arr[j + 1]:
                    arr[j], arr[j + 1] = arr[j + 1], arr[j]
        return arr
    
    def quick(data: list) -> list:
        if len(data) <= 1:
            return data
        pivot = data[len(data) // 2]
        return quick([x for x in data if x < pivot]) + \
               [x for x in data if x == pivot] + \
               quick([x for x in data if x > pivot])
    
    strategies = {'bubble': bubble, 'quick': quick}
    return strategies.get(mode, quick)
```

**Why**: Enables runtime algorithm selection, isolates algorithm implementations, and follows open/closed principle. Used in payment processors, compression utilities, and routing algorithms.

## Adapter Pattern for Microservices

```python
# PROBLEM: Make incompatible service interfaces work together
# TIME: O(1) overhead SPACE: O(1) per adapter
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Dict, Any


@dataclass
class User:
    """Standard user model."""
    id: str
    name: str
    email: str


class UserService(ABC):
    """Target interface expected by client."""
    
    @abstractmethod
    def get_user(self, user_id: str) -> User:
        """Fetch user by ID."""
        pass
    
    @abstractmethod
    def create_user(self, name: str, email: str) -> User:
        """Create new user."""
        pass


class LegacyUserService:
    """Adaptee: legacy service with incompatible interface."""
    
    def fetch_user_data(self, identifier: int) -> Dict[str, Any]:
        """Returns user as dict with different field names."""
        return {
            'user_id': identifier,
            'full_name': 'Legacy User',
            'email_address': 'legacy@example.com'
        }
    
    def create_new_user(self, data: Dict[str, Any]) -> int:
        """Creates user and returns numeric ID."""
        return 12345


class LegacyUserAdapter(UserService):
    """Adapter making legacy service compatible."""
    
    def __init__(self, legacy_service: LegacyUserService):
        self._legacy = legacy_service
    
    def get_user(self, user_id: str) -> User:
        """Adapt legacy response to User model."""
        data = self._legacy.fetch_user_data(int(user_id))
        return User(
            id=str(data['user_id']),
            name=data['full_name'],
            email=data['email_address']
        )
    
    def create_user(self, name: str, email: str) -> User:
        """Adapt creation to return User model."""
        numeric_id = self._legacy.create_new_user({
            'full_name': name,
            'email_address': email
        })
        return User(id=str(numeric_id), name=name, email=email)


# Async Microservice Adapter Variant
class AsyncServiceAdapter:
    """Adapter for async microservices."""
    
    def __init__(self, base_url: str):
        self.base_url = base_url
    
    async def call_service(self, endpoint: str, data: dict) -> dict:
        """Make async HTTP call to microservice."""
        import aiohttp
        async with aiohttp.ClientSession() as session:
            async with session.post(
                f"{self.base_url}/{endpoint}",
                json=data
            ) as response:
                return await response.json()
    
    async def get_user(self, user_id: str) -> User:
        """Adapt HTTP response to User model."""
        data = await self.call_service('users/get', {'id': user_id})
        return User(**data)
```

**Why**: Enables integration of legacy systems, third-party services, and different API versions without modifying client code. Essential for microservices architecture.

## Aggregates DDD Pattern

```python
# PROBLEM: Ensure consistency boundaries and invariants
# TIME: O(1) for aggregate root access SPACE: O(n) for entities
from dataclasses import dataclass, field
from typing import List, Optional
from datetime import datetime
import uuid


@dataclass
class OrderItem:
    """Entity within Order aggregate."""
    product_id: str
    quantity: int
    price: float
    
    def calculate_total(self) -> float:
        return self.quantity * self.price
    
    def update_quantity(self, new_quantity: int) -> None:
        if new_quantity <= 0:
            raise ValueError("Quantity must be positive")
        self.quantity = new_quantity


@dataclass
class Order:
    """Aggregate root managing order items."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    customer_id: str = ""
    items: List[OrderItem] = field(default_factory=list)
    status: str = "pending"
    created_at: datetime = field(default_factory=datetime.now)
    
    def add_item(self, product_id: str, quantity: int, price: float) -> None:
        """Add item maintaining business rules."""
        if self.status != "pending":
            raise ValueError("Cannot modify confirmed order")
        if quantity <= 0:
            raise ValueError("Quantity must be positive")
        
        # Check if item exists
        for item in self.items:
            if item.product_id == product_id:
                item.update_quantity(item.quantity + quantity)
                return
        
        self.items.append(OrderItem(product_id, quantity, price))
    
    def remove_item(self, product_id: str) -> None:
        """Remove item with validation."""
        if self.status != "pending":
            raise ValueError("Cannot modify confirmed order")
        self.items = [i for i in self.items if i.product_id != product_id]
    
    def calculate_total(self) -> float:
        """Calculate order total."""
        return sum(item.calculate_total() for item in self.items)
    
    def confirm(self) -> None:
        """Confirm order with invariant check."""
        if not self.items:
            raise ValueError("Cannot confirm empty order")
        if self.status != "pending":
            raise ValueError("Order already confirmed")
        self.status = "confirmed"


# Repository for Aggregate
class OrderRepository:
    """Repository managing Order aggregates persistence."""
    
    def __init__(self):
        self._orders: dict = {}
    
    def save(self, order: Order) -> None:
        """Persist aggregate through root only."""
        self._orders[order.id] = order
    
    def find_by_id(self, order_id: str) -> Optional[Order]:
        """Load aggregate by root ID."""
        return self._orders.get(order_id)
    
    def find_by_customer(self, customer_id: str) -> List[Order]:
        """Query by customer ID."""
        return [o for o in self._orders.values() if o.customer_id == customer_id]
```

**Why**: Maintains consistency boundaries, protects invariants, and controls access through aggregate root. Core to Domain-Driven Design for complex business domains.

## Read Model Pattern (CQRS)

```python
# PROBLEM: Separate read-optimized queries from write models
# TIME: O(1) for simple reads SPACE: O(n) for projections
from dataclasses import dataclass
from typing import List, Dict, Optional
from datetime import datetime
from collections import defaultdict


@dataclass
class OrderEvent:
    """Base event for event sourcing."""
    event_type: str
    order_id: str
    timestamp: datetime
    data: dict


@dataclass
class OrderReadModel:
    """Denormalized read-optimized model."""
    order_id: str
    customer_name: str
    total_amount: float
    item_count: int
    status: str
    last_updated: datetime


class ReadModelRepository:
    """Repository for read models."""
    
    def __init__(self):
        self._models: Dict[str, OrderReadModel] = {}
        self._by_customer: Dict[str, List[str]] = defaultdict(list)
        self._by_status: Dict[str, List[str]] = defaultdict(list)
    
    def save(self, model: OrderReadModel) -> None:
        """Save and update indexes."""
        self._models[model.order_id] = model
        self._by_customer[model.customer_name].append(model.order_id)
        self._by_status[model.status].append(model.order_id)
    
    def find_by_id(self, order_id: str) -> Optional[OrderReadModel]:
        """Fast lookup by ID."""
        return self._models.get(order_id)
    
    def find_by_customer(self, customer: str) -> List[OrderReadModel]:
        """Query via denormalized index."""
        ids = self._by_customer.get(customer, [])
        return [self._models[i] for i in ids]
    
    def find_by_status(self, status: str) -> List[OrderReadModel]:
        """Query via status index."""
        ids = self._by_status.get(status, [])
        return [self._models[i] for i in ids]


class OrderProjector:
    """Projects events into read models."""
    
    def __init__(self, repository: ReadModelRepository):
        self.repository = repository
    
    def apply(self, event: OrderEvent) -> None:
        """Apply event to update read model."""
        handlers = {
            'order_created': self._handle_created,
            'item_added': self._handle_item_added,
            'order_confirmed': self._handle_confirmed
        }
        handler = handlers.get(event.event_type)
        if handler:
            handler(event)
    
    def _handle_created(self, event: OrderEvent) -> None:
        """Handle order creation."""
        model = OrderReadModel(
            order_id=event.order_id,
            customer_name=event.data['customer_name'],
            total_amount=0.0,
            item_count=0,
            status='pending',
            last_updated=event.timestamp
        )
        self.repository.save(model)
    
    def _handle_item_added(self, event: OrderEvent) -> None:
        """Handle item addition."""
        model = self.repository.find_by_id(event.order_id)
        if model:
            model.total_amount += event.data['price'] * event.data['quantity']
            model.item_count += event.data['quantity']
            model.last_updated = event.timestamp
            self.repository.save(model)
    
    def _handle_confirmed(self, event: OrderEvent) -> None:
        """Handle order confirmation."""
        model = self.repository.find_by_id(event.order_id)
        if model:
            model.status = 'confirmed'
            model.last_updated = event.timestamp
            self.repository.save(model)
```

**Why**: Optimizes read performance with denormalized data, separates concerns from write side, and enables complex queries without affecting write model. Essential for high-read systems.

## GIN Index Implementation

```python
# PROBLEM: Efficient inverted index for composite values
# TIME: O(log n) for search SPACE: O(n) for index entries
from collections import defaultdict
from typing import List, Set, Dict, Any
import json


class GINIndex:
    """Generalized Inverted Index for multi-element values."""
    
    def __init__(self):
        # Inverted index: element -> set of row IDs
        self._inverted: Dict[Any, Set[int]] = defaultdict(set)
        # Forward index: row ID -> original value
        self._forward: Dict[int, Any] = {}
        self._next_id: int = 0
    
    def _extract_elements(self, value: Any) -> List[Any]:
        """Extract indexable elements from value."""
        if isinstance(value, (list, set, tuple)):
            return list(value)
        elif isinstance(value, dict):
            return list(value.keys()) + list(value.values())
        elif isinstance(value, str):
            # Tokenize text into words
            return value.lower().split()
        return [value]
    
    def insert(self, value: Any) -> int:
        """Insert value and return row ID."""
        row_id = self._next_id
        self._next_id += 1
        self._forward[row_id] = value
        
        # Add to inverted index
        for element in self._extract_elements(value):
            self._inverted[element].add(row_id)
        
        return row_id
    
    def delete(self, row_id: int) -> bool:
        """Delete value by row ID."""
        if row_id not in self._forward:
            return False
        
        value = self._forward[row_id]
        for element in self._extract_elements(value):
            self._inverted[element].discard(row_id)
            if not self._inverted[element]:
                del self._inverted[element]
        
        del self._forward[row_id]
        return True
    
    def search_contains(self, query: Any) -> List[int]:
        """Find rows containing query element."""
        elements = self._extract_elements(query)
        if not elements:
            return []
        
        # Get rows that contain ANY of the elements
        result = self._inverted.get(elements[0], set()).copy()
        for element in elements[1:]:
            result |= self._inverted.get(element, set())
        
        return sorted(list(result))
    
    def search_contains_all(self, query: Any) -> List[int]:
        """Find rows containing ALL query elements."""
        elements = self._extract_elements(query)
        if not elements:
            return sorted(list(self._forward.keys()))
        
        # Get rows that contain ALL elements
        result = self._inverted.get(elements[0], set()).copy()
        for element in elements[1:]:
            result &= self._inverted.get(element, set())
        
        return sorted(list(result))
    
    def get_value(self, row_id: int) -> Any:
        """Retrieve original value by row ID."""
        return self._forward.get(row_id)


# Example: JSON document indexing
class JSONGINIndex(GINIndex):
    """GIN index specialized for JSON documents."""
    
    def _extract_elements(self, value: Any) -> List[Any]:
        """Extract all paths and values from JSON."""
        elements = []
        
        def extract(obj, path=''):
            if isinstance(obj, dict):
                for k, v in obj.items():
                    new_path = f"{path}.{k}" if path else k
                    elements.append(new_path)  # Index path
                    elements.append((new_path, v))  # Index path-value pair
                    extract(v, new_path)
            elif isinstance(obj, list):
                for i, item in enumerate(obj):
                    new_path = f"{path}[{i}]"
                    extract(item, new_path)
            else:
                elements.append((path, obj))
        
        if isinstance(value, str):
            try:
                value = json.loads(value)
            except json.JSONDecodeError:
                pass
        
        extract(value)
        return elements
```

**Why**: Efficiently indexes composite values like arrays, JSON, and full-text. PostgreSQL uses GIN for jsonb and array types. Enables fast contains queries on complex data.

## Partial Index Implementation

```python
# PROBLEM: Index subset of rows matching a condition
# TIME: O(log n) for indexed subset SPACE: O(m) where m < n
from typing import Dict, List, Any, Callable, Optional
from dataclasses import dataclass


@dataclass
class Row:
    """Sample row with multiple columns."""
    id: int
    status: str
    amount: float
    category: str
    deleted: bool = False


class PartialIndex:
    """Index that only includes rows matching a predicate."""
    
    def __init__(self, predicate: Callable[[Row], bool], key_func: Callable[[Row], Any]):
        """
        Args:
            predicate: Function returning True for rows to index
            key_func: Function extracting index key from row
        """
        self._predicate = predicate
        self._key_func = key_func
        self._index: Dict[Any, List[int]] = {}  # key -> row IDs
        self._rows: Dict[int, Row] = {}  # id -> row
    
    def insert(self, row: Row) -> None:
        """Insert row if it matches predicate."""
        self._rows[row.id] = row
        if self._predicate(row):
            key = self._key_func(row)
            if key not in self._index:
                self._index[key] = []
            self._index[key].append(row.id)
    
    def delete(self, row_id: int) -> bool:
        """Remove row from index and storage."""
        if row_id not in self._rows:
            return False
        
        row = self._rows[row_id]
        if self._predicate(row):
            key = self._key_func(row)
            if key in self._index:
                self._index[key].remove(row_id)
                if not self._index[key]:
                    del self._index[key]
        
        del self._rows[row_id]
        return True
    
    def update(self, row: Row) -> bool:
        """Update row and re-evaluate predicate."""
        if row.id not in self._rows:
            return False
        
        old_row = self._rows[row.id]
        
        # Remove from old index if was indexed
        if self._predicate(old_row):
            old_key = self._key_func(old_row)
            if old_key in self._index:
                self._index[old_key].remove(row.id)
        
        # Add to new index if matches predicate
        self._rows[row.id] = row
        if self._predicate(row):
            new_key = self._key_func(row)
            if new_key not in self._index:
                self._index[new_key] = []
            self._index[new_key].append(row.id)
        
        return True
    
    def search(self, key: Any) -> List[Row]:
        """Find rows by key (only indexed rows returned)."""
        row_ids = self._index.get(key, [])
        return [self._rows[rid] for rid in row_ids]
    
    def get_all_indexed(self) -> List[Row]:
        """Get all rows in index."""
        result = []
        for row_ids in self._index.values():
            result.extend(self._rows[rid] for rid in row_ids)
        return result
    
    def size(self) -> int:
        """Return number of indexed rows."""
        return sum(len(ids) for ids in self._index.values())


# Example usage: Index only active orders
def create_active_orders_index() -> PartialIndex:
    """Create partial index for non-deleted orders."""
    return PartialIndex(
        predicate=lambda row: not row.deleted and row.status != 'cancelled',
        key_func=lambda row: row.category
    )


# Example: Composite partial index
class CompositePartialIndex:
    """Partial index with multiple key columns."""
    
    def __init__(
        self,
        predicate: Callable[[Row], bool],
        key_funcs: List[Callable[[Row], Any]]
    ):
        self._predicate = predicate
        self._key_funcs = key_funcs
        self._index: Dict[tuple, List[int]] = {}
        self._rows: Dict[int, Row] = {}
    
    def insert(self, row: Row) -> None:
        """Insert row if matching predicate."""
        self._rows[row.id] = row
        if self._predicate(row):
            key = tuple(f(row) for f in self._key_funcs)
            if key not in self._index:
                self._index[key] = []
            self._index[key].append(row.id)
    
    def search(self, *keys) -> List[Row]:
        """Search by composite key."""
        key = tuple(keys)
        row_ids = self._index.get(key, [])
        return [self._rows[rid] for rid in row_ids]
```

**Why**: Reduces index size and maintenance overhead by excluding irrelevant rows. PostgreSQL partial indexes improve write performance for selective queries.

## Inheritance Patterns

```python
# PROBLEM: Share behavior through class hierarchies
# TIME: O(1) method lookup SPACE: O(1) per instance
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import List, Optional


# Single Inheritance
class Animal:
    """Base class demonstrating single inheritance."""
    
    def __init__(self, name: str):
        self.name = name
    
    def speak(self) -> str:
        return "Some sound"
    
    def move(self) -> str:
        return "Moving"


class Dog(Animal):
    """Derived class overriding behavior."""
    
    def speak(self) -> str:
        return "Woof!"
    
    def fetch(self) -> str:
        return f"{self.name} is fetching"


# Multi-level Inheritance
class Mammal(Animal):
    """Intermediate class in hierarchy."""
    
    def __init__(self, name: str, fur_color: str):
        super().__init__(name)
        self.fur_color = fur_color
    
    def nurse(self) -> str:
        return f"{self.name} is nursing young"


class Cat(Mammal):
    """Deep inheritance example."""
    
    def speak(self) -> str:
        return "Meow!"
    
    def climb(self) -> str:
        return f"{self.name} is climbing"


# Abstract Base Class Pattern
class Shape(ABC):
    """Abstract class defining interface."""
    
    @abstractmethod
    def area(self) -> float:
        pass
    
    @abstractmethod
    def perimeter(self) -> float:
        pass
    
    def describe(self) -> str:
        return f"Area: {self.area()}, Perimeter: {self.perimeter()}"


class Rectangle(Shape):
    """Concrete implementation of Shape."""
    
    def __init__(self, width: float, height: float):
        self.width = width
        self.height = height
    
    def area(self) -> float:
        return self.width * self.height
    
    def perimeter(self) -> float:
        return 2 * (self.width + self.height)


# Mixin Pattern (Alternative to multiple inheritance)
class TimestampMixin:
    """Mixin adding timestamp functionality."""
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._created_at = self._get_timestamp()
    
    def _get_timestamp(self) -> str:
        from datetime import datetime
        return datetime.now().isoformat()
    
    @property
    def created_at(self) -> str:
        return self._created_at


class SerializableMixin:
    """Mixin adding serialization."""
    
    def to_dict(self) -> dict:
        return self.__dict__.copy()
    
    @classmethod
    def from_dict(cls, data: dict):
        return cls(**data)


class Document(TimestampMixin, SerializableMixin):
    """Class using mixins."""
    
    def __init__(self, title: str, content: str):
        self.title = title
        self.content = content
        super().__init__()


# Composition over Inheritance
class Engine:
    """Component class."""
    
    def start(self) -> str:
        return "Engine started"
    
    def stop(self) -> str:
        return "Engine stopped"


class Car:
    """Class using composition instead of inheritance."""
    
    def __init__(self, model: str):
        self.model = model
        self.engine = Engine()  # Composition
    
    def start(self) -> str:
        return f"{self.model}: {self.engine.start()}"
```

**Why**: Enables code reuse and polymorphism. Single inheritance keeps hierarchies simple. ABCs define contracts. Mixins add orthogonal functionality. Composition provides flexible alternative to deep inheritance.

## Encapsulation Patterns

```python
# PROBLEM: Hide internal state and expose controlled interface
# TIME: O(1) for access SPACE: O(1) per instance
from typing import Optional
from dataclasses import dataclass


# Private attributes with property accessors
class BankAccount:
    """Encapsulated account with controlled access."""
    
    def __init__(self, owner: str, initial_balance: float = 0):
        self._owner = owner  # Protected (convention)
        self.__balance = initial_balance  # Private (name mangling)
        self._transaction_history = []
    
    @property
    def balance(self) -> float:
        """Read-only access to balance."""
        return self.__balance
    
    @property
    def owner(self) -> str:
        """Read-only access to owner."""
        return self._owner
    
    def deposit(self, amount: float) -> None:
        """Controlled mutation with validation."""
        if amount <= 0:
            raise ValueError("Deposit must be positive")
        self.__balance += amount
        self._transaction_history.append(('deposit', amount))
    
    def withdraw(self, amount: float) -> bool:
        """Controlled mutation with business rules."""
        if amount <= 0:
            raise ValueError("Withdrawal must be positive")
        if amount > self.__balance:
            return False
        self.__balance -= amount
        self._transaction_history.append(('withdraw', amount))
        return True
    
    def get_statement(self) -> list:
        """Return copy of transaction history."""
        return self._transaction_history.copy()


# Descriptor-based encapsulation
class ValidatedAttribute:
    """Descriptor with validation."""
    
    def __init__(self, name: str, validator: callable):
        self.name = name
        self.validator = validator
        self._storage_name = f'_validated_{name}'
    
    def __get__(self, instance, owner):
        if instance is None:
            return self
        return getattr(instance, self._storage_name, None)
    
    def __set__(self, instance, value):
        if not self.validator(value):
            raise ValueError(f"Invalid value for {self.name}")
        setattr(instance, self._storage_name, value)


def positive_number(value) -> bool:
    return isinstance(value, (int, float)) and value > 0


class Product:
    """Using descriptor for validation."""
    
    price = ValidatedAttribute('price', positive_number)
    quantity = ValidatedAttribute('quantity', positive_number)
    
    def __init__(self, name: str, price: float, quantity: int):
        self.name = name
        self.price = price
        self.quantity = quantity
    
    @property
    def total_value(self) -> float:
        return self.price * self.quantity


# Slots for memory efficiency and encapsulation
class Point:
    """Immutable point using slots."""
    
    __slots__ = ['_x', '_y']
    
    def __init__(self, x: float, y: float):
        self._x = x
        self._y = y
    
    @property
    def x(self) -> float:
        return self._x
    
    @property
    def y(self) -> float:
        return self._y
    
    def __repr__(self) -> str:
        return f"Point({self._x}, {self._y})"
    
    def distance_from_origin(self) -> float:
        return (self._x ** 2 + self._y ** 2) ** 0.5


# Context manager encapsulation
class DatabaseConnection:
    """Encapsulated resource with context manager."""
    
    def __init__(self, connection_string: str):
        self._connection_string = connection_string
        self._connection = None
    
    def __enter__(self):
        """Acquire resource."""
        print(f"Connecting to {self._connection_string}")
        self._connection = {}  # Simulated connection
        return self
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        """Release resource."""
        print("Closing connection")
        self._connection = None
        return False  # Don't suppress exceptions
    
    def execute(self, query: str) -> list:
        """Controlled resource access."""
        if not self._connection:
            raise RuntimeError("No active connection")
        print(f"Executing: {query}")
        return []
```

**Why**: Protects internal state from external modification, validates inputs, maintains invariants, and provides clean API. Python uses conventions (single/double underscore) rather than strict access control.
