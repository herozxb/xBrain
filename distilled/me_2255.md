## Target Sum

```python
# PROBLEM: Given array of positive integers and target S, count ways to assign 
# + or - to each element so sum equals target S.
# TIME: O(n * sum) SPACE: O(sum)
def find_target_sum_ways(nums: list[int], target: int) -> int:
    """
    Uses subset sum DP: partition into two subsets P and N where sum(P) - sum(N) = target.
    Therefore sum(P) = (total + target) / 2. Count subsets with this sum.
    """
    total = sum(nums)
    
    # Check if valid partition possible
    if (total + target) % 2 != 0 or total + target < 0:
        return 0
    
    subset_sum = (total + target) // 2
    
    # dp[i] = number of ways to sum to i
    dp = [0] * (subset_sum + 1)
    dp[0] = 1  # One way to make sum 0: choose nothing
    
    for num in nums:
        # Traverse backwards to avoid using same element twice
        for s in range(subset_sum, num - 1, -1):
            dp[s] += dp[s - num]
    
    return dp[subset_sum]
```
**Why**: Classic subset sum variation using DP with space optimization. Key insight is transforming the sign assignment problem into a partition problem.

## Maximal Square

```python
# PROBLEM: Given binary matrix, find largest square containing only 1s and return its area.
# TIME: O(m * n) SPACE: O(n) with optimization
def maximal_square(matrix: list[list[str]]) -> int:
    """
    DP where dp[j] = side length of largest square ending at current row, column j.
    A square ending at (i,j) is limited by minimum of top, left, and top-left squares.
    """
    if not matrix or not matrix[0]:
        return 0
    
    m, n = len(matrix), len(matrix[0])
    # dp[j] represents max square side ending at column j of previous row
    dp = [0] * (n + 1)
    max_side = 0
    prev = 0  # Store dp[j-1] from previous iteration (top-left)
    
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            temp = dp[j]  # Save before overwrite
            if matrix[i - 1][j - 1] == '1':
                # dp[j] (top), dp[j-1] (left), prev (top-left)
                dp[j] = min(dp[j], dp[j - 1], prev) + 1
                max_side = max(max_side, dp[j])
            else:
                dp[j] = 0
            prev = temp
        prev = 0  # Reset for next row
    
    return max_side * max_side
```
**Why**: Elegant 1D DP optimization of 2D problem. Each cell depends only on three neighbors, enabling O(n) space instead of O(mn).

## Sliding Window Median

```python
# PROBLEM: Find median of each sliding window of size k in an array.
# TIME: O(n * log k) SPACE: O(k)
from heapq import heappush, heappop
from collections import defaultdict

def median_sliding_window(nums: list[int], k: int) -> list[float]:
    """
    Uses two heaps: max heap for smaller half, min heap for larger half.
    Lazy deletion with counter to handle removals efficiently.
    """
    # max_heap stores negatives (Python has min heap only)
    max_heap = []  # smaller half
    min_heap = []  # larger half
    delayed = defaultdict(int)  # count of elements to be removed
    
    def prune(heap: list, is_max_heap: bool) -> None:
        """Remove top elements that are marked for deletion."""
        while heap:
            num = -heap[0] if is_max_heap else heap[0]
            if (is_max_heap and delayed[num] > 0) or (not is_max_heap and delayed[num] > 0):
                delayed[num] -= 1
                if delayed[num] == 0:
                    del delayed[num]
                heappop(heap)
            else:
                break
    
    def get_median() -> float:
        """Return median based on window size."""
        if k % 2 == 1:
            return float(-max_heap[0])
        return (-max_heap[0] + min_heap[0]) / 2.0
    
    # Balance heaps: max_heap size = min_heap size or min_heap size + 1
    def balance() -> None:
        while len(max_heap) > len(min_heap) + 1:
            heappush(min_heap, -heappop(max_heap))
            prune(max_heap, True)
        while len(max_heap) < len(min_heap):
            heappush(max_heap, -heappop(min_heap))
            prune(min_heap, False)
    
    # Initialize first window
    for i in range(k):
        if len(max_heap) == 0 or nums[i] <= -max_heap[0]:
            heappush(max_heap, -nums[i])
        else:
            heappush(min_heap, nums[i])
    balance()
    
    result = [get_median()]
    
    # Slide window
    for i in range(k, len(nums)):
        out_num = nums[i - k]  # Element leaving window
        in_num = nums[i]       # Element entering window
        delayed[out_num] += 1
        
        # Remove outgoing element conceptually
        if out_num <= -max_heap[0]:
            # Was in max_heap
            if out_num == -max_heap[0]:
                prune(max_heap, True)
            balance()
        else:
            # Was in min_heap
            if out_num == min_heap[0]:
                prune(min_heap, False)
            balance()
        
        # Insert new element
        if in_num <= -max_heap[0]:
            heappush(max_heap, -in_num)
        else:
            heappush(min_heap, in_num)
        balance()
        
        prune(max_heap, True)
        prune(min_heap, False)
        result.append(get_median())
    
    return result
```
**Why**: Demonstrates lazy deletion technique for maintaining a median data structure with deletions. Essential for streaming/sliding window scenarios.

## Permutations II

```python
# PROBLEM: Generate all unique permutations of array that may contain duplicates.
# TIME: O(n! * n) SPACE: O(n) recursion depth
def permute_unique(nums: list[int]) -> list[list[int]]:
    """
    Sort first, then skip duplicates by checking if same as previous and previous
    hasn't been used (meaning we're at a new branch where this duplicate would create
    the same permutation).
    """
    nums.sort()
    n = len(nums)
    used = [False] * n
    result = []
    current = []
    
    def backtrack() -> None:
        if len(current) == n:
            result.append(current[:])
            return
        
        for i in range(n):
            if used[i]:
                continue
            # Skip duplicate: same as previous AND previous not used
            # If previous not used, we'd create same permutation as using previous first
            if i > 0 and nums[i] == nums[i - 1] and not used[i - 1]:
                continue
            
            used[i] = True
            current.append(nums[i])
            backtrack()
            current.pop()
            used[i] = False
    
    backtrack()
    return result
```
**Why**: Key insight for duplicate handling in backtracking: sort + skip when duplicate and previous unused. Prevents generating identical permutations.

## Generate Parentheses

```python
# PROBLEM: Generate all valid parentheses strings with n pairs.
# TIME: O(4^n / sqrt(n)) SPACE: O(n) recursion depth
def generate_parenthesis(n: int) -> list[str]:
    """
    Backtrack with constraints: can add '(' if count < n, can add ')' if count > open.
    This guarantees validity without needing to validate each string.
    """
    result = []
    
    def backtrack(current: list[str], open_count: int, close_count: int) -> None:
        # Base case: used all pairs
        if len(current) == 2 * n:
            result.append(''.join(current))
            return
        
        # Add '(' if we haven't used all opening brackets
        if open_count < n:
            current.append('(')
            backtrack(current, open_count + 1, close_count)
            current.pop()
        
        # Add ')' only if it won't make invalid (more closes than opens)
        if close_count < open_count:
            current.append(')')
            backtrack(current, open_count, close_count + 1)
            current.pop()
    
    backtrack([], 0, 0)
    return result
```
**Why**: Classic constraint-based backtracking. Never generates invalid strings, avoiding wasteful validation. Pattern applies to many generation problems.

## Word Search

```python
# PROBLEM: Find if word exists in 2D board by connecting adjacent cells (no reuse).
# TIME: O(m * n * 4^L) where L = word length SPACE: O(L) recursion depth
def exist(board: list[list[str]], word: str) -> bool:
    """
    DFS from each cell that matches first character. Mark visited by temporarily
    changing cell value to avoid revisiting.
    """
    if not board or not board[0]:
        return False
    
    m, n = len(board), len(board[0])
    
    def dfs(row: int, col: int, index: int) -> bool:
        """Search for word[index:] starting from (row, col)."""
        if index == len(word):
            return True  # Found complete word
        
        if row < 0 or row >= m or col < 0 or col >= n:
            return False  # Out of bounds
        
        if board[row][col] != word[index]:
            return False  # Character mismatch
        
        # Mark as visited
        temp = board[row][col]
        board[row][col] = '#'
        
        # Explore all 4 directions
        found = (dfs(row + 1, col, index + 1) or
                 dfs(row - 1, col, index + 1) or
                 dfs(row, col + 1, index + 1) or
                 dfs(row, col - 1, index + 1))
        
        # Restore cell
        board[row][col] = temp
        return found
    
    # Try starting from each cell
    for i in range(m):
        for j in range(n):
            if dfs(i, j, 0):
                return True
    
    return False
```
**Why**: In-place marking technique avoids extra visited set. Common pattern for grid DFS where space efficiency matters.

## Subsets II

```python
# PROBLEM: Generate all unique subsets of array that may contain duplicates.
# TIME: O(2^n * n) SPACE: O(n) recursion depth
def subsets_with_dup(nums: list[int]) -> list[list[int]]:
    """
    Sort array, then at each position decide to include or exclude.
    Skip duplicates at the same decision level to avoid duplicate subsets.
    """
    nums.sort()
    result = []
    
    def backtrack(start: int, current: list[int]) -> None:
        result.append(current[:])
        
        for i in range(start, len(nums)):
            # Skip duplicates at same level
            if i > start and nums[i] == nums[i - 1]:
                continue
            
            current.append(nums[i])
            backtrack(i + 1, current)  # Move to next index
            current.pop()
    
    backtrack(0, [])
    return result
```
**Why**: Subset generation pattern: each call adds current subset, then iterates through remaining elements. Duplicate check `i > start` ensures we skip only at same level.

## Combinations

```python
# PROBLEM: Generate all combinations of k numbers chosen from 1 to n.
# TIME: O(C(n,k) * k) SPACE: O(k) recursion depth
def combine(n: int, k: int) -> list[list[int]]:
    """
    Build combinations by choosing each number as start, then recursively
    choosing from numbers after it.
    """
    result = []
    
    def backtrack(start: int, current: list[int]) -> None:
        # Pruning: if remaining numbers < needed, stop early
        if len(current) + (n - start + 1) < k:
            return
        
        if len(current) == k:
            result.append(current[:])
            return
        
        for i in range(start, n + 1):
            current.append(i)
            backtrack(i + 1, current)
            current.pop()
    
    backtrack(1, [])
    return result
```
**Why**: Clean combination generation with early pruning. `i + 1` ensures strictly increasing order, preventing duplicates naturally.

## Restore IP Addresses

```python
# PROBLEM: Generate all valid IP addresses from string of digits.
# TIME: O(3^4 * n) = O(n) SPACE: O(n)
def restore_ip_addresses(s: str) -> list[str]:
    """
    IP has exactly 4 segments, each 0-255, no leading zeros unless segment is '0'.
    Backtrack by placing dots at valid positions.
    """
    result = []
    n = len(s)
    
    # Early termination: IP must be 4-12 digits
    if n < 4 or n > 12:
        return result
    
    def is_valid(segment: str) -> bool:
        """Check if segment is valid IP octet."""
        if not segment:
            return False
        # No leading zeros unless segment is "0"
        if len(segment) > 1 and segment[0] == '0':
            return False
        # Must be 0-255
        return 0 <= int(segment) <= 255
    
    def backtrack(start: int, parts: list[str]) -> None:
        """Build IP by adding one segment at a time."""
        # If we have 4 parts and used all characters
        if len(parts) == 4:
            if start == n:
                result.append('.'.join(parts))
            return
        
        # Try segments of length 1, 2, 3
        for length in range(1, 4):
            if start + length > n:
                break
            
            segment = s[start:start + length]
            if is_valid(segment):
                parts.append(segment)
                backtrack(start + length, parts)
                parts.pop()
    
    backtrack(0, [])
    return result
```
**Why**: Demonstrates segmentation backtracking. Fixed output structure (4 parts) with variable input consumption. Valid check before recursion prevents invalid branches.

## Stone Game

```python
# PROBLEM: Two players take stones from ends of array, trying to maximize sum.
# Return True if first player can win (or tie).
# TIME: O(n^2) SPACE: O(n^2)
def stone_game(piles: list[int]) -> bool:
    """
    DP where dp[i][j] = max advantage current player can have over opponent
    with piles[i:j+1]. Player can take left or right, then faces disadvantage.
    """
    n = len(piles)
    # dp[i][j] = max(piles[i] - dp[i+1][j], piles[j] - dp[i][j-1])
    # Subtract because opponent will then have the advantage
    dp = [[0] * n for _ in range(n)]
    
    # Base case: single pile
    for i in range(n):
        dp[i][i] = piles[i]
    
    # Fill by diagonal (increasing length)
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            # Take left: gain piles[i], then face opponent with [i+1, j]
            take_left = piles[i] - dp[i + 1][j]
            # Take right: gain piles[j], then face opponent with [i, j-1]
            take_right = piles[j] - dp[i][j - 1]
            dp[i][j] = max(take_left, take_right)
    
    return dp[0][n - 1] > 0  # First player wins if advantage > 0


# Space-optimized version: O(n) space
def stone_game_optimized(piles: list[int]) -> bool:
    """
    Since we only need previous diagonal, use 1D array.
    """
    n = len(piles)
    dp = piles[:]  # Start with base case
    
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            dp[i] = max(piles[i] - dp[i + 1], piles[j] - dp[i])
    
    return dp[0] > 0
```
**Why**: Minimax DP pattern. Store relative advantage, not absolute scores. Optimal play assumes opponent also plays optimally (minimize our advantage = maximize theirs).
