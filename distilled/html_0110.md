## [html5_web_storage_events]
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Web Storage Events Demo</title>
</head>
<body>
  <h2>Storage Event Listener</h2>
  <p>Open this page in two tabs. Changes in one tab will notify the other.</p>
  
  <input type="text" id="dataInput" placeholder="Enter data...">
  <button onclick="saveData()">Save to Storage</button>
  
  <div id="log"></div>

  <script>
    // PROBLEM: Sync state across browser tabs using storage events
    // TIME: O(1) SPACE: O(1)
    
    // Save data to localStorage - triggers storage event in other tabs
    function saveData() {
      const input = document.getElementById('dataInput');
      const timestamp = new Date().toLocaleTimeString();
      
      //setItem triggers 'storage' event in OTHER tabs/windows (not current)
      localStorage.setItem('sharedData', JSON.stringify({
        value: input.value,
        time: timestamp
      }));
      
      logMessage(`Saved: "${input.value}" at ${timestamp}`);
    }
    
    // Listen for storage changes from other tabs
    // The storage event fires when localStorage is modified in another document
    window.addEventListener('storage', (event) => {
      // event.key - name of the changed key
      // event.oldValue - previous value (null if new key)
      // event.newValue - new value (null if key removed)
      // event.url - URL of the document that made the change
      // event.storageArea - reference to localStorage or sessionStorage
      
      if (event.key === 'sharedData') {
        const data = JSON.parse(event.newValue);
        logMessage(`Received from another tab: "${data.value}" at ${data.time}`);
      }
    });
    
    // Helper to display messages
    function logMessage(msg) {
      const log = document.getElementById('log');
      const entry = document.createElement('p');
      entry.textContent = `[${new Date().toLocaleTimeString()}] ${msg}`;
      log.prepend(entry);
    }
  </script>
</body>
</html>
```
**Why:** Enables real-time data synchronization between browser tabs/windows without polling. Essential for multi-tab applications like dashboards or collaborative tools.

```html
<!DOCTYPE html>
<html lang="en">
<head>

<body>

  <button onclick="createImageBlob()">Create Image Blob</button>

    // PROBLEM: Create and manipulate binary large objects in browser
    // TIME: O(n) SPACE: O(n) where n is blob size
    
    // Blob constructor: new Blob(array, options)
    // - array: Array of ArrayBuffer, Blob, String, etc.
    // - options: { type: MIMEtype, endings: 'transparent'|'native' }
    
    function createTextBlob() {
      // Create a simple text blob with MIME type
      const textParts = ['Hello, ', 'Blob ', 'World!'];
      const blob = new Blob(textParts, { type: 'text/plain' });
      
      displayBlobInfo(blob, 'Text Blob');
      
      // Read blob content using FileReader
      const reader = new FileReader();
      reader.onload = () => console.log('Content:', reader.result);
      reader.readAsText(blob);
    }
    
    function createJsonBlob() {
      const data = { 
        name: 'Blob Demo', 
        timestamp: Date.now(),
        items: [1, 2, 3]
      };
      
      // Create JSON blob - useful for file downloads or API uploads
      const blob = new Blob([JSON.stringify(data, null, 2)], {
        type: 'application/json'
      });
      
      displayBlobInfo(blob, 'JSON Blob');
      
      // Create downloadable link from blob
      const url = URL.createObjectURL(blob);
      const link = document.createElement('a');
      link.href = url;
      link.download = 'data.json';
      link.textContent = ' Download JSON';
      document.getElementById('output').appendChild(link);
    }
    
    function createImageBlob() {
      // Create a canvas and draw something
      const canvas = document.createElement('canvas');
      canvas.width = 100;
      canvas.height = 100;
      const ctx = canvas.getContext('2d');
      
      // Draw a gradient
      const gradient = ctx.createLinearGradient(0, 0, 100, 100);
      gradient.addColorStop(0, 'red');
      gradient.addColorStop(1, 'blue');
      ctx.fillStyle = gradient;
      ctx.fillRect(0, 0, 100, 100);
      
      // Convert canvas to blob (asynchronous!)
      canvas.toBlob((blob) => {
        displayBlobInfo(blob, 'Image Blob');
        
        // Create image preview from blob
        const img = new Image();
        img.src = URL.createObjectURL(blob);
        img.style.border = '2px solid black';
        document.getElementById('output').appendChild(img);
      }, 'image/png');
    }
    
    function displayBlobInfo(blob, name) {
      const output = document.getElementById('output');
      const info = document.createElement('div');
      info.innerHTML = `
        <h3>${name}</h3>
        <ul>
          <li>Size: ${blob.size} bytes</li>
          <li>Type: ${blob.type || 'unknown'}</li>
        </ul>
      `;
      output.appendChild(info);
      
      // Blob methods:
      // - blob.slice(start, end, type) - create smaller blob
      // - blob.text() - returns Promise<string>
      // - blob.arrayBuffer() - returns Promise<ArrayBuffer>
      // - blob.stream() - returns ReadableStream
    }
```

```html
<!DOCTYPE html>
<html lang="en">
<head>

<body>
  <h2>File Input & File Object</h2>

  <script>
    // PROBLEM: Access and read user-selected files in browser
    // TIME: O(n) SPACE: O(n) where n is file size
    
    // File objects inherit from Blob and add:
    // - name: filename
    // - lastModified: timestamp
    // - size: bytes
    // - type: MIME type
    
    document.getElementById('fileInput').addEventListener('change', (event) => {
      // event.target.files is a FileList (array-like, not actual Array)
      const fileList = event.target.files;
      
      const output = document.getElementById('fileInfo');
      output.innerHTML = '';
      
      // Convert FileList to Array for easier iteration
      Array.from(fileList).forEach((file, index) => {
        displayFileInfo(file, index);
      });
    });
    
    function displayFileInfo(file, index) {
      const output = document.getElementById('fileInfo');
      
      // File inherits from Blob, so all Blob methods work
      const info = document.createElement('div');
      info.innerHTML = `
        <h3>File ${index + 1}: ${file.name}</h3>
        <ul>
          <li>Type: ${file.type || 'unknown'}</li>
          <li>Size: ${formatBytes(file.size)}</li>
          <li>Last Modified: ${new Date(file.lastModified).toLocaleString()}</li>
        </ul>
      `;
      
      // Add preview button based on file type
      if (file.type.startsWith('image/')) {
        const previewBtn = document.createElement('button');
        previewBtn.textContent = 'Preview Image';
        previewBtn.onclick = () => previewImage(file, info);
        info.appendChild(previewBtn);
      } else if (file.type.startsWith('text/')) {
        const readBtn = document.createElement('button');
        readBtn.textContent = 'Read Text';
        readBtn.onclick = () => readTextFile(file, info);
        info.appendChild(readBtn);
      }
      
      output.appendChild(info);
    }
    
    // Preview image file using URL.createObjectURL
    function previewImage(file, container) {
      // Create temporary URL for the file
      const url = URL.createObjectURL(file);
      
      const img = document.createElement('img');
      img.src = url;
      img.style.maxWidth = '200px';
      img.onload = () => {
        // Revoke URL after load to free memory
        URL.revokeObjectURL(url);
      };
      
      container.appendChild(img);
    }
    
    // Read text file using blob.text() Promise
    async function readTextFile(file, container) {
      try {
        // Modern approach: use blob.text() Promise
        const content = await file.text();
        
        const pre = document.createElement('pre');
        pre.textContent = content.substring(0, 500); // First 500 chars
        pre.style.background = '#f0f0f0';
        pre.style.padding = '10px';
        pre.style.overflow = 'auto';
        pre.style.maxHeight = '200px';
        
        container.appendChild(pre);
      } catch (error) {
        console.error('Error reading file:', error);
      }
    }
    
    // Format bytes to human-readable string
    function formatBytes(bytes) {
      if (bytes === 0) return '0 Bytes';
      const k = 1024;
      const sizes = ['Bytes', 'KB', 'MB', 'GB'];
      const i = Math.floor(Math.log(bytes) / Math.log(k));
      return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
    }
  </script>

</html>
```
**Why:** The File API bridges user files and web applications, enabling upload previews, client-side validation, and file processing before server submission.

## [html5_filereader_api]
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>FileReader API Demo</title>
</head>
<body>
  <h2>FileReader - Asynchronous File Reading</h2>
  <input type="file" id="fileInput">
  <div id="output"></div>
  <progress id="progress" value="0" max="100"></progress>

  <script>
    // PROBLEM: Read file contents asynchronously with progress tracking
    // TIME: O(n) SPACE: O(n) where n is file size
    
    // FileReader is an asynchronous API for reading Blob/File contents
    // Events: onloadstart, onprogress, onload, onerror, onloadend, onabort
    
    document.getElementById('fileInput').addEventListener('change', (event) => {
      const file = event.target.files[0];
      if (!file) return;
      
      readFileWithProgress(file);
    });
    
    function readFileWithProgress(file) {
      const reader = new FileReader();
      const progress = document.getElementById('progress');
      const output = document.getElementById('output');
      
      // Called when reading starts
      reader.onloadstart = () => {
        output.innerHTML = '<p>Reading started...</p>';
        progress.value = 0;
      };
      
      // Called periodically during read (for progress bars)
      reader.onprogress = (event) => {
        if (event.lengthComputable) {
          const percentLoaded = (event.loaded / event.total) * 100;
          progress.value = percentLoaded;
          console.log(`Loaded: ${percentLoaded.toFixed(1)}%`);
        }
      };
      
      // Called when read completes successfully
      reader.onload = (event) => {
        output.innerHTML += '<p>Read complete!</p>';
        
        // event.target.result contains the file data
        const result = event.target.result;
        
        // Display based on read method used
        if (typeof result === 'string') {
          displayTextResult(result, output);
        } else if (result instanceof ArrayBuffer) {
          displayArrayBufferInfo(result, output);
        }
      };
      
      // Called if an error occurs
      reader.onerror = (event) => {
        // Error codes: reader.error.NOT_FOUND_ERR, SECURITY_ERR, etc.
        output.innerHTML += `<p style="color:red">Error: ${reader.error}</p>`;
      };
      
      // Called when read completes (success or failure)
      reader.onloadend = () => {
        output.innerHTML += '<p>Read operation finished.</p>';
      };
      
      // Called if read is aborted
      reader.onabort = () => {
        output.innerHTML += '<p style="color:orange">Read aborted!</p>';
      };
      
      // Choose appropriate read method based on file type
      if (file.type.startsWith('text/')) {
        reader.readAsText(file); // Result: string
      } else {
        reader.readAsArrayBuffer(file); // Result: ArrayBuffer
      }
      
      // Other read methods:
      // reader.readAsDataURL(file) - Result: base64 data URL string
      // reader.readAsBinaryString(file) - Result: raw binary string (deprecated)
    }
    
    function displayTextResult(text, container) {
      const pre = document.createElement('pre');
      pre.textContent = text.substring(0, 1000);
      pre.style.background = '#f5f5f5';
      pre.style.padding = '10px';
      pre.style.maxHeight = '300px';
      pre.style.overflow = 'auto';
      container.appendChild(pre);
    }
    
    function displayArrayBufferInfo(buffer, container) {
      // ArrayBuffer is raw binary data - create typed array to read
      const bytes = new Uint8Array(buffer);
      
      const info = document.createElement('div');
      info.innerHTML = `
        <p>ArrayBuffer Size: ${buffer.byteLength} bytes</p>
        <p>First 16 bytes (hex): ${Array.from(bytes.slice(0, 16))
          .map(b => b.toString(16).padStart(2, '0'))
          .join(' ')}</p>
      `;
      container.appendChild(info);
    }
    
    // Example: Abort a read operation
    // reader.abort(); // Cancels the read, triggers onabort
  </script>
</body>
</html>
```
**Why:** FileReader provides asynchronous file reading with progress events, essential for handling large files without blocking the UI thread.

## [html5_filereader_readAsText]
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>FileReader.readAsText() Demo</title>
</head>
<body>
  <h2>Reading Text Files with Encoding Support</h2>
  <input type="file" id="fileInput" accept=".txt,.csv,.json,.md">
  <select id="encoding">
    <option value="UTF-8">UTF-8</option>
    <option value="ISO-8859-1">ISO-8859-1</option>
    <option value="UTF-16">UTF-16</option>
  </select>
  <div id="output"></div>

  <script>
    // PROBLEM: Read text files with proper character encoding
    // TIME: O(n) SPACE: O(n) where n is file size
    
    // readAsText(blob, encoding) - encoding defaults to UTF-8
    
    document.getElementById('fileInput').addEventListener('change', async (event) => {
      const file = event.target.files[0];
      if (!file) return;
      
      const encoding = document.getElementById('encoding').value;
      
      // Method 1: Traditional callback approach
      readWithCallbacks(file, encoding);
      
      // Method 2: Modern Promise wrapper
      const content = await readFileAsText(file, encoding);
      console.log('Promise result:', content.substring(0, 100));
    });
    
    // Traditional callback-based reading
    function readWithCallbacks(file, encoding) {
      const reader = new FileReader();
      
      reader.onload = (event) => {
        const text = event.target.result;
        displayTextContent(text, file.name, encoding);
      };
      
      reader.onerror = () => {
        console.error('Failed to read file:', reader.error);
      };
      
      // readAsText(file, encoding) - encoding is optional
      reader.readAsText(file, encoding);
    }
    
    // Modern Promise-based wrapper for cleaner async code
    function readFileAsText(file, encoding = 'UTF-8') {
      return new Promise((resolve, reject) => {
        const reader = new FileReader();
        
        reader.onload = (event) => resolve(event.target.result);
        reader.onerror = (error) => reject(error);
        
        reader.readAsText(file, encoding);
      });
    }
    
    function displayTextContent(text, filename, encoding) {
      const output = document.getElementById('output');
      
      // Analyze the text
      const lines = text.split('\n');
      const words = text.split(/\s+/).filter(w => w.length > 0);
      const chars = text.length;
      
      output.innerHTML = `
        <h3>File: ${filename}</h3>
        <p>Encoding: ${encoding}</p>
        <div style="display: flex; gap: 20px;">
          <div>Lines: <strong>${lines.length}</strong></div>
          <div>Words: <strong>${words.length}</strong></div>
          <div>Characters: <strong>${chars}</strong></div>
        </div>
        <h4>Content Preview:</h4>
        <pre style="background: #f0f0f0; padding: 15px; overflow: auto; max-height: 400px;">${escapeHtml(text)}</pre>
      `;
    }
    
    // Escape HTML to prevent XSS when displaying user content
    function escapeHtml(text) {
      const div = document.createElement('div');
      div.textContent = text;
      return div.innerHTML;
    }
    
    // Example: Reading CSV and parsing
    async function parseCSV(file) {
      const text = await readFileAsText(file);
      const lines = text.trim().split('\n');
      
      return lines.map(line => {
        // Simple CSV parsing (doesn't handle quoted fields with commas)
        return line.split(',').map(cell => cell.trim());
      });
    }
  </script>
</body>
</html>
```
**Why:** readAsText is the primary method for reading text-based files (CSV, JSON, logs, configs) with proper encoding support for international characters.
