```html
  <!-- Track element for subtitles -->
  <track

<script>
/**
 * Creates a video element with multiple track options
 * @param {string} videoSrc - Source URL of the video
 * @param {Array<{kind: string, src: string, srclang: string, label: string}>} tracks - Track configurations
 * @returns {HTMLVideoElement} Configured video element
 */
function createVideoWithTracks(videoSrc, tracks) {
  const video = document.createElement('video');
  video.controls = true;
  video.src = videoSrc;
  
  tracks.forEach((trackConfig, index) => {
    const track = document.createElement('track');
    track.kind = trackConfig.kind; // subtitles, captions, chapters, descriptions, metadata
    track.src = trackConfig.src;
    track.srclang = trackConfig.srclang;
    track.label = trackConfig.label;
    if (index === 0) track.default = true;
    video.appendChild(track);
  });
  
  return video;
}
</script>
```
**Why**: Essential for video accessibility - provides subtitles, captions, and chapter navigation for hearing-impaired users and multi-language support.

```html
/**
 * Creates a data element linking human-readable content with machine-readable value
 * @param {string} displayText - Human-readable text
 * @param {string|number} value - Machine-readable value
 * @returns {HTMLDataElement} Configured data element
 */
function createDataElement(displayText, value) {
  const dataEl = document.createElement('data');
  dataEl.value = value;
  dataEl.textContent = displayText;
  return dataEl;

 * Extracts all data values from a container
 * @param {HTMLElement} container - Container with data elements
 * @returns {Array<{text: string, value: string}>} Array of text-value pairs

function extractDataValues(container) {
```

```html
<script>
/**
 * Creates a meter element for displaying measurements
 * @param {Object} config - Meter configuration
 * @param {number} config.value - Current value
 * @param {number} [config.min=0] - Minimum value
 * @param {number} [config.max=1] - Maximum value
 * @param {number} [config.low] - Low threshold
 * @param {number} [config.high] - High threshold
 * @param {number} [config.optimum] - Optimum value
 * @param {string} [config.label] - Fallback text
 * @returns {HTMLMeterElement} Configured meter element
 */
function createMeter({ value, min = 0, max = 1, low, high, optimum, label }) {
  const meter = document.createElement('meter');
  meter.value = value;
  meter.min = min;
  meter.max = max;
  
  if (low !== undefined) meter.low = low;
  if (high !== undefined) meter.high = high;
  if (optimum !== undefined) meter.optimum = optimum;
  if (label) meter.textContent = label;

/**
 * Updates meter dynamically based on data
 * @param {HTMLMeterElement} meter - Meter element to update
 * @param {number} value - New value
 */
function updateMeter(meter, value) {
  meter.value = Math.max(meter.min, Math.min(meter.max, value));
}
</script>
```
**Why**: Semantically represents gauges and measurements (disk usage, scores, temperatures) with built-in accessibility support.

```html
/**
 * Creates a radio button group
 * @param {string} name - Radio group name (required for mutual exclusion)
 * @param {Array<{value: string, label: string, checked?: boolean}>} options - Radio options
 * @returns {HTMLFieldSetElement} Fieldset with radio buttons
 */
function createRadioGroup(name, options) {
  const fieldset = document.createElement('fieldset');
  const legend = document.createElement('legend');
  legend.textContent = name;
  fieldset.appendChild(legend);
  
  options.forEach(option => {
    const label = document.createElement('label');
    const radio = document.createElement('input');
    radio.type = 'radio';
    radio.name = name;
    radio.value = option.value;
    if (option.checked) radio.checked = true;
    
    label.appendChild(radio);
    label.appendChild(document.createTextNode(' ' + option.label));
    fieldset.appendChild(label);
    fieldset.appendChild(document.createElement('br'));
  });
  
  return fieldset;
}

/**
 * Gets the selected value from a radio group
 * @param {string} name - Radio group name
 * @returns {string|null} Selected value or null
 */
function getSelectedRadio(name) {
  const selected = document.querySelector(`input[type="radio"][name="${name}"]:checked`);
  return selected ? selected.value : null;
}

/**
 * Sets the selected radio by value
 * @param {string} name - Radio group name
 * @param {string} value - Value to select
 */
function setSelectedRadio(name, value) {
  const radio = document.querySelector(`input[type="radio"][name="${name}"][value="${value}"]`);
  if (radio) radio.checked = true;

</script>
```

## 7. html_thead_element
```html
<!-- PROBLEM: Group table header content for structure and printing
     TIME: O(1) SPACE: O(1) -->
<table id="dataTable">
  <thead>
    <tr>
      <th>ID</th>
      <th>Product</th>
      <th>Price</th>
    </tr>
  </thead>
  <tbody>
    <tr><td>1</td><td>Widget</td><td>$10</td></tr>
    <tr><td>2</td><td>Gadget</td><td>$20</td></tr>
  </tbody>
  <tfoot>
    <tr><td colspan="2">Total:</td><td>$30</td></tr>
  </tfoot>
</table>

<script>
/**
 * Creates a table with proper thead structure
 * @param {Array<string>} headers - Header texts
 * @param {Array<Array<string>>} rows - Data rows
 * @returns {HTMLTableElement} Complete table
 */
function createStructuredTable(headers, rows) {
  const table = document.createElement('table');
  
  // Create thead
  const thead = document.createElement('thead');
  const headerRow = document.createElement('tr');
  headers.forEach(text => {
    const th = document.createElement('th');
    th.textContent = text;
    headerRow.appendChild(th);
  });
  thead.appendChild(headerRow);
  table.appendChild(thead);
  
  // Create tbody
  const tbody = document.createElement('tbody');
  rows.forEach(rowData => {
    const row = document.createElement('tr');
    rowData.forEach(cellText => {
      const td = document.createElement('td');
      td.textContent = cellText;
      row.appendChild(td);
    });
    tbody.appendChild(row);
  });
  table.appendChild(tbody);
  
  return table;
}

/**
 * Sorts table by column using thead headers
 * @param {HTMLTableElement} table - Table to sort
 * @param {number} columnIndex - Column to sort by
 * @param {boolean} [ascending=true] - Sort direction
 */
function sortTableByColumn(table, columnIndex, ascending = true) {
  const tbody = table.querySelector('tbody');
  const rows = Array.from(tbody.querySelectorAll('tr'));
  
  rows.sort((a, b) => {
    const aVal = a.cells[columnIndex].textContent;
    const bVal = b.cells[columnIndex].textContent;
    return ascending ? aVal.localeCompare(bVal) : bVal.localeCompare(aVal);
  });
  
  rows.forEach(row => tbody.appendChild(row));
}
</script>
```
**Why**: Provides semantic table structure, enables repeat headers when printing, and improves accessibility for screen readers.

## 8. html_dt_element
```html
<!-- PROBLEM: Create description/definition list with term-value pairs
     TIME: O(1) SPACE: O(1) -->
<dl>
  <dt>HTML</dt>
  <dd>HyperText Markup Language - the standard markup language for documents.</dd>
  
  <dt>CSS</dt>
  <dd>Cascading Style Sheets - used for styling HTML elements.</dd>
  
  <dt>JavaScript</dt>
  <dd>A programming language that enables interactive web pages.</dd>
  <dd>Also used in server-side development (Node.js).</dd>
</dl>

<script>
/**
 * Creates a definition/description list entry
 * @param {string} term - The term being defined (dt)
 * @param {string|string[]} descriptions - One or more descriptions (dd)
 * @returns {DocumentFragment} Fragment with dt and dd elements
 */
function createDefinitionEntry(term, descriptions) {
  const fragment = document.createDocumentFragment();
  
  // Create term (dt)
  const dt = document.createElement('dt');
  dt.textContent = term;
  fragment.appendChild(dt);
  
  // Create descriptions (dd) - can have multiple per term
  const descArray = Array.isArray(descriptions) ? descriptions : [descriptions];
  descArray.forEach(desc => {
    const dd = document.createElement('dd');
    dd.textContent = desc;
    fragment.appendChild(dd);
  });
  
  return fragment;
}

/**
 * Creates a complete definition list from data
 * @param {Object.<string, string|string[]>} definitions - Term-description pairs
 * @returns {HTMLDListElement} Complete dl element
 */
function createDefinitionList(definitions) {
  const dl = document.createElement('dl');
  
  Object.entries(definitions).forEach(([term, descriptions]) => {
    dl.appendChild(createDefinitionEntry(term, descriptions));
  });
  
  return dl;
}

// Usage example
const techDefinitions = createDefinitionList({
  'API': 'Application Programming Interface - allows software components to communicate.',
  'DOM': [
    'Document Object Model - represents HTML as a tree structure.',
    'Enables JavaScript to access and modify HTML elements.'
  ]
});
</script>
```
**Why**: Semantically marks up terms and definitions, perfect for glossaries, metadata, and key-value pair displays.

## 9. html_image_decoding
```html
<!-- PROBLEM: Control image decoding behavior for performance
     TIME: O(1) SPACE: O(1) -->
<img 
  src="large-image.jpg" 
  decoding="async" 
  alt="Decorative landscape"
  loading="lazy">

<img 
  src="hero-image.jpg" 
  decoding="sync" 
  alt="Hero banner"
  loading="eager">

<img 
  src="icon.svg" 
  decoding="auto" 
  alt="Site icon">

<script>
/**
 * Creates an img element with optimized decoding settings
 * @param {Object} config - Image configuration
 * @param {string} config.src - Image source URL
 * @param {string} config.alt - Alt text (required for accessibility)
 * @param {string} [config.decoding='auto'] - Decoding: 'sync', 'async', 'auto'
 * @param {string} [config.loading='lazy'] - Loading: 'lazy', 'eager'
 * @param {number} [config.width] - Image width
 * @param {number} [config.height] - Image height
 * @returns {HTMLImageElement} Configured image element
 */
function createOptimizedImage({ src, alt, decoding = 'auto', loading = 'lazy', width, height }) {
  const img = document.createElement('img');
  img.src = src;
  img.alt = alt;
  img.decoding = decoding;
  img.loading = loading;
  
  if (width) img.width = width;
  if (height) img.height = height;
  
  return img;
}

/**
 * Decoding strategy helper
 * @param {HTMLImageElement} img - Image element
 * @returns {Promise<HTMLImageElement>} Promise resolving when decoded
 */
function decodeImage(img) {
  if (img.decode) {
    return img.decode().then(() => img);
  }
  return Promise.resolve(img);
}

/**
 * Batch decode multiple images
 * @param {HTMLImageElement[]} images - Images to decode
 * @returns {Promise<HTMLImageElement[]>} All decoded images
 */
async function decodeImages(images) {
  return Promise.all(images.map(decodeImage));
}

// Usage: async decoding prevents main thread blocking
const heroImage = createOptimizedImage({
  src: 'hero.webp',
  alt: 'Homepage hero',
  decoding: 'async',
  loading: 'eager'
});
</script>
```
**Why**: Optimizes image loading by controlling when/how images decode - async prevents blocking the main thread on large images.

```html
<script>
/**
 * Sets multiple CSS properties with various value types
 * @param {HTMLElement} element - Target element
 * @param {Object.<string, string|number>} styles - Style properties and values
 */
function setStyles(element, styles) {
  Object.entries(styles).forEach(([prop, value]) => {
    element.style[prop] = value;
  });
}

/**
 * Creates a CSS value with calculation
 * @param {string} expression - Calc expression
 * @returns {string} CSS calc() value
 */
function calc(expression) {
  return `calc(${expression})`;
}

/**
 * Creates a CSS color from RGB values
 * @param {number} r - Red (0-255)
 * @param {number} g - Green (0-255)
 * @param {number} b - Blue (0-255)
 * @param {number} [a=1] - Alpha (0-1)
 * @returns {string} CSS color value
 */
function rgba(r, g, b, a = 1) {
  return a < 1 ? `rgba(${r}, ${g}, ${b}, ${a})` : `rgb(${r}, ${g}, ${b})`;
}

/**
 * Creates a CSS clamp value for responsive sizing
 * @param {string} min - Minimum value
 * @param {string} preferred - Preferred value
 * @param {string} max - Maximum value
 * @returns {string} CSS clamp() value

function clamp(min, preferred, max) {
  return `clamp(${min}, ${preferred}, ${max})`;
}
```

**Why**: Understanding CSS value types (lengths, colors, functions, keywords) enables precise, responsive, and maintainable styling.
