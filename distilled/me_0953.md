```python
# PROBLEM: Organize domain logic into cohesive modules with clear boundaries
# TIME: O(1) SPACE: O(1) for module structure
class DomainModule:
    """Base class for domain-driven design modules with bounded context."""
    
    def __init__(self, name: str, context: str):
        self.name = name
        self.context = context
        self._aggregates = []
        self._services = []
        self._repositories = []
    
    def register_aggregate(self, aggregate_class):
        """Register an aggregate root within this module."""
        self._aggregates.append(aggregate_class)
        return aggregate_class
    
    def register_service(self, service_class):
        """Register a domain service."""
        self._services.append(service_class)
        return service_class
    
    def register_repository(self, repository_class):
        """Register a repository interface."""
        self._repositories.append(repository_class)
        return repository_class
    
    def get_dependencies(self):
        """Return module dependencies for dependency injection."""
        return {
            "aggregates": self._aggregates,
            "services": self._services,
            "repositories": self._repositories
        }

# Example: Order module in e-commerce
order_module = DomainModule("Order", "OrderManagement")

@order_module.register_aggregate
class Order:
    def __init__(self, order_id: str, customer_id: str):
        self.order_id = order_id
        self.customer_id = customer_id
        self.items = []
        self.status = "pending"

@order_module.register_service
class OrderService:
    def calculate_total(self, order: Order) -> float:
        return sum(item['price'] * item['quantity'] for item in order.items)
```
**Why**: Enables clean separation of domain concerns with explicit module boundaries, supporting scalable architecture and team autonomy.

## Big Picture Event Storming
```python
# PROBLEM: Model domain events and their relationships at a high level
# TIME: O(n) SPACE: O(n) for event storage
from dataclasses import dataclass
from typing import List, Dict, Set
from datetime import datetime

@dataclass
class DomainEvent:
    """Represents a domain event in event storming."""
    name: str
    timestamp: datetime
    metadata: dict

class BigPictureEventStorming:
    """Visualize and model domain events at system level."""
    
    def __init__(self):
        self.events: List[DomainEvent] = []
        self.command_to_events: Dict[str, List[str]] = {}
        self.event_to_policy: Dict[str, List[str]] = {}
        self.aggregates: Dict[str, Set[str]] = {}
    
    def add_command(self, command_name: str, triggers_events: List[str]):
        """Add a command that triggers specific events."""
        self.command_to_events[command_name] = triggers_events
    
    def add_event(self, event_name: str, triggers_policies: List[str]):
        """Add an event that triggers policies (reactive handlers)."""
        self.event_to_policy[event_name] = triggers_policies
    
    def assign_to_aggregate(self, aggregate: str, events: List[str]):
        """Assign events to an aggregate (consistency boundary)."""
        if aggregate not in self.aggregates:
            self.aggregates[aggregate] = set()
        self.aggregates[aggregate].update(events)
    
    def trace_flow(self, start_command: str) -> List[str]:
        """Trace the flow from command through events to policies."""
        flow = [f"Command: {start_command}"]
        for event in self.command_to_events.get(start_command, []):
            flow.append(f"  → Event: {event}")
            for policy in self.event_to_policy.get(event, []):
                flow.append(f"    → Policy: {policy}")
        return flow

# Usage
storming = BigPictureEventStorming()
storming.add_command("PlaceOrder", ["OrderPlaced", "InventoryReserved"])
storming.add_event("OrderPlaced", ["SendConfirmationEmail", "UpdateCustomerStats"])
storming.assign_to_aggregate("Order", ["OrderPlaced"])
```
**Why**: Provides a collaborative modeling technique to discover domain events, commands, and policies, enabling better understanding of business processes.

## Process Level Event Storming
```python
# PROBLEM: Model detailed business processes with event sequences
# TIME: O(n) SPACE: O(n) for process definition
from typing import List, Optional, Callable
from enum import Enum

class ProcessState(Enum):
    """State of a process instance."""
    STARTED = "started"
    COMPLETED = "completed"
    FAILED = "failed"

class ProcessLevelEventStorming:
    """Model business processes as sequences of events."""
    
    def __init__(self, process_name: str):
        self.process_name = process_name
        self.steps: List[dict] = []
        self.current_state = ProcessState.STARTED
    
    def add_step(self, event: str, actor: str, 
                 preconditions: Optional[List[str]] = None,
                 postconditions: Optional[List[str]] = None):
        """Add a process step with event, actor, and conditions."""
        self.steps.append({
            "event": event,
            "actor": actor,
            "preconditions": preconditions or [],
            "postconditions": postconditions or []
        })
    
    def validate_process(self) -> bool:
        """Validate that all preconditions are met by earlier postconditions."""
        available_facts = set()
        for step in self.steps:
            # Check preconditions
            missing = set(step["preconditions"]) - available_facts
            if missing:
                return False
            # Add postconditions
            available_facts.update(step["postconditions"])
        return True
    
    def get_sequence(self) -> List[str]:
        """Return the event sequence for this process."""
        return [step["event"] for step in self.steps]

# Example: Order fulfillment process
process = ProcessLevelEventStorming("OrderFulfillment")
process.add_step(
    event="PaymentReceived",
    actor="PaymentService",
    preconditions=["OrderPlaced"],
    postconditions=["PaymentVerified", "InventoryReserved"]
)
process.add_step(
    event="OrderShipped",
    actor="WarehouseService",
    preconditions=["PaymentVerified", "InventoryReserved"],
    postconditions=["ShipmentCreated", "OrderCompleted"]
)
```
**Why**: Enables detailed modeling of business processes with explicit preconditions and postconditions, ensuring process integrity and auditability.

## Software Design Event Storming
```python
# PROBLEM: Design software components from event storming models
# TIME: O(1) SPACE: O(n) for component registry
from typing import Dict, List, Type

class SoftwareDesignEventStorming:
    """Transform event storming models into software components."""
    
    def __init__(self):
        self.components: Dict[str, dict] = {}
        self.bounded_contexts: Dict[str, List[str]] = {}
    
    def define_bounded_context(self, context: str, description: str):
        """Define a bounded context in the domain."""
        self.bounded_contexts[context] = []
        self.components[context] = {
            "type": "bounded_context",
            "description": description,
            "aggregates": [],
            "services": [],
            "policies": []
        }
    
    def add_aggregate(self, context: str, name: str, 
                      handles_commands: List[str], emits_events: List[str]):
        """Add an aggregate to a bounded context."""
        self.components[context]["aggregates"].append({
            "name": name,
            "handles_commands": handles_commands,
            "emits_events": emits_events
        })
        self.bounded_contexts[context].append(name)
    
    def add_policy(self, context: str, name: str, 
                   subscribes_to: List[str], triggers: List[str]):
        """Add a policy (event handler) to a bounded context."""
        self.components[context]["policies"].append({
            "name": name,
            "subscribes_to": subscribes_to,
            "triggers": triggers
        })
    
    def generate_code_structure(self) -> Dict[str, List[str]]:
        """Generate suggested code structure from the design."""
        structure = {}
        for context, data in self.components.items():
            structure[context] = [
                f"aggregates/{agg['name'].lower()}.py"
                for agg in data["aggregates"]
            ] + [
                f"policies/{policy['name'].lower()}.py"
                for policy in data["policies"]
            ]
        return structure

# Example
design = SoftwareDesignEventStorming()
design.define_bounded_context("Ordering", "Handle customer orders")
design.add_aggregate("Ordering", "Order", 
                     handles_commands=["PlaceOrder", "CancelOrder"],
                     emits_events=["OrderPlaced", "OrderCancelled"])
design.add_policy("Ordering", "NotificationPolicy",
                  subscribes_to=["OrderPlaced"],
                  triggers=["SendEmail"])
```
**Why**: Bridges the gap between domain modeling and implementation, ensuring software design aligns with business processes.

## Interceptor Pattern HTTP
```python
# PROBLEM: Intercept and transform HTTP requests/responses for cross-cutting concerns
# TIME: O(1) SPACE: O(1) per interceptor
from typing import Callable, Optional, Dict, Any
from functools import wraps

class HTTPInterceptor:
    """Base class for HTTP request/response interceptors."""
    
    def pre_process(self, request: Dict[str, Any]) -> Dict[str, Any]:
        """Process request before sending (can modify or reject)."""
        return request
    
    def post_process(self, response: Dict[str, Any]) -> Dict[str, Any]:
        """Process response after receiving (can modify or handle errors)."""
        return response
    
    def handle_error(self, error: Exception) -> Optional[Dict[str, Any]]:
        """Handle errors during request processing."""
        raise error

class LoggingInterceptor(HTTPInterceptor):
    """Log all HTTP requests and responses."""
    
    def pre_process(self, request: Dict[str, Any]) -> Dict[str, Any]:
        print(f"[REQUEST] {request.get('method')} {request.get('url')}")
        return request
    
    def post_process(self, response: Dict[str, Any]) -> Dict[str, Any]:
        print(f"[RESPONSE] Status: {response.get('status')}")
        return response

class AuthenticationInterceptor(HTTPInterceptor):
    """Add authentication headers to requests."""
    
    def __init__(self, token: str):
        self.token = token
    
    def pre_process(self, request: Dict[str, Any]) -> Dict[str, Any]:
        headers = request.setdefault("headers", {})
        headers["Authorization"] = f"Bearer {self.token}"
        return request

class InterceptorChain:
    """Chain multiple interceptors together."""
    
    def __init__(self):
        self.interceptors: List[HTTPInterceptor] = []
    
    def add_interceptor(self, interceptor: HTTPInterceptor):
        """Add an interceptor to the chain."""
        self.interceptors.append(interceptor)
    
    def execute(self, request: Dict[str, Any], 
                http_client: Callable) -> Dict[str, Any]:
        """Execute the full interceptor chain."""
        # Pre-processing
        processed_request = request
        for interceptor in self.interceptors:
            processed_request = interceptor.pre_process(processed_request)
        
        # Execute HTTP request
        response = http_client(processed_request)
        
        # Post-processing
        processed_response = response
        for interceptor in reversed(self.interceptors):
            processed_response = interceptor.post_process(processed_response)
        
        return processed_response

# Usage
chain = InterceptorChain()
chain.add_interceptor(LoggingInterceptor())
chain.add_interceptor(AuthenticationInterceptor("secret-token"))
```
**Why**: Enables reusable cross-cutting concerns (logging, auth, retries) without polluting business logic, following separation of concerns principle.

## Hybrid Storage Implementation
```python
# PROBLEM: Combine hot and cold storage for optimal performance and cost
# TIME: O(1) for hot, O(log n) for cold SPACE: O(n) for cold storage
from typing import Any, Optional
from datetime import datetime, timedelta
import json

class HybridStorage:
    """Combine in-memory hot storage with persistent cold storage."""
    
    def __init__(self, hot_ttl_seconds: int = 3600, max_hot_size: int = 1000):
        self.hot_storage: dict = {}  # Fast in-memory cache
        self.hot_timestamps: dict = {}  # Track access times
        self.cold_storage: dict = {}  # Persistent storage (could be DB)
        self.hot_ttl = hot_ttl_seconds
        self.max_hot_size = max_hot_size
    
    def get(self, key: str) -> Optional[Any]:
        """Retrieve item, checking hot storage first."""
        # Check hot storage
        if key in self.hot_storage:
            if self._is_hot_valid(key):
                self.hot_timestamps[key] = datetime.now()
                return self.hot_storage[key]
            else:
                # Expired, remove from hot
                del self.hot_storage[key]
                del self.hot_timestamps[key]
        
        # Check cold storage
        if key in self.cold_storage:
            value = self.cold_storage[key]
            # Promote to hot storage
            self._promote_to_hot(key, value)
            return value
        
        return None
    
    def set(self, key: str, value: Any, persistent: bool = True):
        """Store item in hot and optionally cold storage."""
        # Always update hot storage
        self.hot_storage[key] = value
        self.hot_timestamps[key] = datetime.now()
        
        # Evict if over capacity
        if len(self.hot_storage) > self.max_hot_size:
            self._evict_coldest()
        
        # Optionally persist to cold storage
        if persistent:
            self.cold_storage[key] = value
    
    def _is_hot_valid(self, key: str) -> bool:
        """Check if hot storage entry is still valid."""
        timestamp = self.hot_timestamps.get(key)
        if not timestamp:
            return False
        return datetime.now() - timestamp < timedelta(seconds=self.hot_ttl)
    
    def _promote_to_hot(self, key: str, value: Any):
        """Move item from cold to hot storage."""
        if len(self.hot_storage) >= self.max_hot_size:
            self._evict_coldest()
        self.hot_storage[key] = value
        self.hot_timestamps[key] = datetime.now()
    
    def _evict_coldest(self):
        """Remove least recently accessed item from hot storage."""
        if not self.hot_timestamps:
            return
        coldest_key = min(self.hot_timestamps, key=self.hot_timestamps.get)
        del self.hot_storage[coldest_key]
        del self.hot_timestamps[coldest_key]

# Usage
storage = HybridStorage(hot_ttl_seconds=300, max_hot_size=100)
storage.set("user:123", {"name": "Alice", "email": "alice@example.com"})
user = storage.get("user:123")  # From hot storage
```
**Why**: Balances performance (hot storage) with cost efficiency (cold storage), automatically managing data lifecycle based on access patterns.

## Constant Patterns
```python
# PROBLEM: Manage constants with type safety and grouping
# TIME: O(1) SPACE: O(1)
from enum import Enum, auto
from typing import TypeVar, Generic, Type

T = TypeVar('T')

class ConstantGroup(Generic[T]):
    """Type-safe constant group with validation."""
    
    def __init__(self, name: str):
        self.name = name
        self._constants: dict = {}
    
    def define(self, key: str, value: T, description: str = "") -> T:
        """Define a constant with metadata."""
        self._constants[key] = {
            "value": value,
            "description": description
        }
        return value
    
    def get(self, key: str) -> T:
        """Retrieve a constant value."""
        return self._constants[key]["value"]
    
    def describe(self, key: str) -> str:
        """Get constant description."""
        return self._constants[key].get("description", "")
    
    def all_values(self) -> dict:
        """Return all constant values."""
        return {k: v["value"] for k, v in self._constants.items()}

# Example: HTTP Status Codes
HTTP_STATUS = ConstantGroup[int]("HTTPStatus")
HTTP_STATUS.define("OK", 200, "Request successful")
```

## Prototype Patterns
```python
# PROBLEM: Clone objects efficiently without depending on their classes
# TIME: O(1) for shallow, O(n) for deep SPACE: O(1) or O(n) for deep
from copy import deepcopy
from typing import Any, Dict
import json

class Prototype:
    """Base class for objects that support cloning."""
    
    def clone(self, deep: bool = True) -> 'Prototype':
        """Create a copy of this object."""
        if deep:
            return deepcopy(self)
        else:
            # Shallow copy
            new_obj = self.__class__.__new__(self.__class__)
            new_obj.__dict__.update(self.__dict__)
            return new_obj
    
    def clone_with_modifications(self, **kwargs) -> 'Prototype':
        """Clone and modify specific attributes."""
        cloned = self.clone(deep=True)
        for key, value in kwargs.items():
            setattr(cloned, key, value)
        return cloned

class PrototypeRegistry:
    """Registry of pre-initialized prototypes."""
    
    def __init__(self):
        self._prototypes: Dict[str, Prototype] = {}
    
    def register(self, name: str, prototype: Prototype):
        """Register a prototype under a name."""
        self._prototypes[name] = prototype
    
    def unregister(self, name: str):
        """Remove a prototype from registry."""
        del self._prototypes[name]
    
    def create(self, name: str, **modifications) -> Prototype:
        """Clone a registered prototype with optional modifications."""
        prototype = self._prototypes.get(name)
        if not prototype:
            raise ValueError(f"Prototype '{name}' not found")
        return prototype.clone_with_modifications(**modifications)

# Example: Document prototype
class Document(Prototype):
    def __init__(self, title: str, content: str, author: str):
        self.title = title
        self.content = content
        self.author = author
        self.metadata = {"created": "2026-02-22", "version": 1}
    
    def __repr__(self):
        return f"Document(title='{self.title}', author='{self.author}')"

# Usage
registry = PrototypeRegistry()
template = Document("Template", "Default content", "System")
registry.register("report_template", template)

# Create new documents from template
report1 = registry.create("report_template", title="Q1 Report", author="Alice")
report2 = registry.create("report_template", title="Q2 Report", content="Custom content")
```
**Why**: Enables efficient object creation when initialization is expensive, and allows dynamic object customization without subclassing.

## Bridge Patterns
```python
# PROBLEM: Decouple abstraction from implementation to vary independently
# TIME: O(1) SPACE: O(1) per abstraction-implementation pair
from abc import ABC, abstractmethod

# Implementation interface
class Renderer(ABC):
    """Implementation interface for rendering shapes."""
    
    @abstractmethod
    def render_circle(self, radius: float):
        pass
    
    @abstractmethod
    def render_rectangle(self, width: float, height: float):
        pass

# Concrete implementations
class VectorRenderer(Renderer):
    """Render shapes as vectors (SVG, etc.)."""
    
    def render_circle(self, radius: float):
        return f"Drawing circle with radius {radius} as vector"
    
    def render_rectangle(self, width: float, height: float):
        return f"Drawing rectangle {width}x{height} as vector"

class RasterRenderer(Renderer):
    """Render shapes as raster (pixels)."""
    
    def render_circle(self, radius: float):
        return f"Drawing circle with radius {radius} as pixels"
    
    def render_rectangle(self, width: float, height: float):
        return f"Drawing rectangle {width}x{height} as pixels"

# Abstraction
class Shape:
    """Abstraction that uses a renderer implementation."""
    
    def __init__(self, renderer: Renderer):
        self.renderer = renderer
    
    def draw(self) -> str:
        pass

class Circle(Shape):
    """Refined abstraction for circles."""
    
    def __init__(self, renderer: Renderer, radius: float):
        super().__init__(renderer)
        self.radius = radius
    
    def draw(self) -> str:
        return self.renderer.render_circle(self.radius)

class Rectangle(Shape):
    """Refined abstraction for rectangles."""
    
    def __init__(self, renderer: Renderer, width: float, height: float):
        super().__init__(renderer)
        self.width = width
        self.height = height
    
    def draw(self) -> str:
        return self.renderer.render_rectangle(self.width, self.height)

# Usage: Abstraction and implementation can vary independently
vector_renderer = VectorRenderer()
raster_renderer = RasterRenderer()

circle1 = Circle(vector_renderer, 5.0)
circle2 = Circle(raster_renderer, 5.0)
rect1 = Rectangle(vector_renderer, 10.0, 20.0)

print(circle1.draw())  # Vector rendering
print(circle2.draw())  # Raster rendering
```
**Why**: Separates abstraction from implementation, allowing both to evolve independently and enabling platform-independent interfaces.

## Composite Patterns
```python
# PROBLEM: Treat individual objects and compositions uniformly
# TIME: O(n) for tree traversal SPACE: O(n) for tree structure
from abc import ABC, abstractmethod
from typing import List

class Component(ABC):
    """Base component interface for all objects in composition."""
    
    @abstractmethod
    def operation(self) -> str:
        """Perform operation on component."""
        pass
    
    @abstractmethod
    def get_size(self) -> int:
        """Get size/cost of component."""
        pass

class Leaf(Component):
    """Leaf node with no children."""
    
    def __init__(self, name: str, value: int = 1):
        self.name = name
        self.value = value
    
    def operation(self) -> str:
        return f"Leaf: {self.name}"
    
    def get_size(self) -> int:
        return self.value

class Composite(Component):
    """Composite node that can contain children."""
    
    def __init__(self, name: str):
        self.name = name
        self.children: List[Component] = []
    
    def add(self, component: Component):
        """Add a child component."""
        self.children.append(component)
    
    def remove(self, component: Component):
        """Remove a child component."""
        self.children.remove(component)
    
    def operation(self) -> str:
        """Perform operation on self and all children."""
        results = [f"Composite: {self.name}"]
        for child in self.children:
            results.append(f"  {child.operation()}")
        return "\n".join(results)
    
    def get_size(self) -> int:
        """Get total size of all children."""
        return sum(child.get_size() for child in self.children)

# Example: File system
class File(Leaf):
    """File in file system."""
    pass

class Directory(Composite):
    """Directory containing files and subdirectories."""
    pass

# Usage
root = Directory("root")
docs = Directory("documents")
images = Directory("images")

root.add(docs)
root.add(images)

docs.add(File("report.pdf", 100))
docs.add(File("notes.txt", 10))
images.add(File("photo.jpg", 500))

print(root.operation())
# Total size: 610
print(f"Total size: {root.get_size()}")
```
**Why**: Enables treating individual objects and compositions uniformly, simplifying client code and supporting recursive tree structures.
