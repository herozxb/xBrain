## 1. Element.getBoundingClientRect()

```html
# PROBLEM: Get the size and position of an element relative to the viewport
# TIME: O(1) SPACE: O(1)

// getBoundingClientRect returns a DOMRect object with position and dimension info
// Properties: top, left, right, bottom, width, height, x, y
// All values are relative to the viewport (not the document)

const element = document.getElementById('myElement');
const rect = element.getBoundingClientRect();

console.log('Top:', rect.top);      // Distance from top of viewport
console.log('Left:', rect.left);    // Distance from left of viewport
console.log('Width:', rect.width);  // Element width (including padding/border)
console.log('Height:', rect.height); // Element height (including padding/border)

// Check if element is in viewport
function isInViewport(element) {
    const rect = element.getBoundingClientRect();
    return (
        rect.top >= 0 &&
        rect.left >= 0 &&
        rect.bottom <= (window.innerHeight || document.documentElement.clientHeight) &&
        rect.right <= (window.innerWidth || document.documentElement.clientWidth)
    );
}
```

**Why**: Essential for scroll-based animations, lazy loading images, collision detection, and positioning tooltips/popovers relative to elements.

## 2. Element.getClientRects()

```html
# PROBLEM: Get a collection of DOMRect objects for each CSS border box of an element
# TIME: O(n) SPACE: O(n) where n = number of client rects

// getClientRects returns a DOMRectList (array-like) of rectangles
// Useful for inline elements that span multiple lines

const element = document.getElementById('inlineElement');
const rects = element.getClientRects();

console.log('Number of boxes:', rects.length);

// Iterate through each rectangle (e.g., each line of an inline element)
for (let i = 0; i < rects.length; i++) {
    const rect = rects[i];
    console.log(`Box ${i}:`, rect.top, rect.left, rect.width, rect.height);
}

// Practical use: Get the bounding box of each line in a text span
function getLineRects(textElement) {
    const rects = textElement.getClientRects();
    const lines = [];
    for (const rect of rects) {
        lines.push({
            top: rect.top,
            left: rect.left,
            width: rect.width,
            height: rect.height
        });
    }
    return lines;
}
```

**Why**: Critical for inline elements spanning multiple lines; enables per-line highlighting, precise text selection UIs, and multi-line tooltip positioning.

## 3. Window.getComputedStyle()

```html
# PROBLEM: Get the computed CSS values of an element after all styles are applied
# TIME: O(1) SPACE: O(1)

// getComputedStyle returns a CSSStyleDeclaration object
// Contains final computed values after CSS cascade and inheritance

const element = document.getElementById('myElement');
const computedStyle = window.getComputedStyle(element);

// Get computed values (returned as strings)
console.log('Color:', computedStyle.color);           // rgb(255, 0, 0)
console.log('Font size:', computedStyle.fontSize);    // 16px
console.log('Display:', computedStyle.display);       // block
console.log('Margin:', computedStyle.marginTop);      // 10px

// Get computed value of CSS variable
const rootStyle = window.getComputedStyle(document.documentElement);
const primaryColor = rootStyle.getPropertyValue('--primary-color');

// Get pseudo-element computed styles
const beforeStyle = window.getComputedStyle(element, '::before');
console.log('Before content:', beforeStyle.content);

// Convert pixel string to number
function getPxValue(style, property) {
    return parseFloat(style[property]) || 0;
}
```

**Why**: Necessary when reading actual rendered styles (including defaults, inheritance, and browser-computed values); style property only shows inline styles.

## 4. Window.innerWidth / innerHeight

```html
# PROBLEM: Get the viewport dimensions (visible browser area)
# TIME: O(1) SPACE: O(1)

// innerWidth/innerHeight include scrollbar if present
// These represent the browser viewport, not the full document

console.log('Viewport width:', window.innerWidth);
console.log('Viewport height:', window.innerHeight);

// Common pattern: Responsive behavior
function handleResize() {
    const width = window.innerWidth;
    const height = window.innerHeight;
    
    if (width < 768) {
        console.log('Mobile view');
    } else if (width < 1024) {
        console.log('Tablet view');
    } else {
        console.log('Desktop view');
    }
}

// Listen for viewport size changes
window.addEventListener('resize', handleResize);

// Get viewport dimensions excluding scrollbars (CSS viewport)
function getViewportSize() {
    return {
        width: Math.max(document.documentElement.clientWidth || 0, window.innerWidth || 0),
        height: Math.max(document.documentElement.clientHeight || 0, window.innerHeight || 0)
    };
}
```

**Why**: Fundamental for responsive design, canvas sizing, modal positioning, and any layout calculations that depend on available viewport space.

## 5. Mouse Event - pageX / pageY

```html
# PROBLEM: Get mouse coordinates relative to the entire document (including scrolled area)
# TIME: O(1) SPACE: O(1)

// pageX/pageY include scroll offset
// Useful when you need position relative to document, not viewport

document.addEventListener('click', function(event) {
    console.log('Document X:', event.pageX);  // From left of document
    console.log('Document Y:', event.pageY);  // From top of document
});

// Draw at click position on canvas (handles scroll)
const canvas = document.getElementById('canvas');
canvas.addEventListener('click', function(event) {
    const rect = canvas.getBoundingClientRect();
    const x = event.pageX - rect.left - window.scrollX;
    const y = event.pageY - rect.top - window.scrollY;
    
    const ctx = canvas.getContext('2d');
    ctx.fillRect(x, y, 5, 5);
});

// Convert between coordinate systems
function pageToClient(pageX, pageY) {
    return {
        clientX: pageX - window.scrollX,
        clientY: pageY - window.scrollY
    };
}
```

**Why**: Ideal for drawing applications, drag-and-drop with scrolling, and any interaction where absolute document position matters regardless of scroll state.

## 6. Mouse Event - screenX / screenY

```html
# PROBLEM: Get mouse coordinates relative to the user's physical screen
# TIME: O(1) SPACE: O(1)

// screenX/screenY are relative to the monitor, not the browser
// Useful for multi-monitor detection and absolute positioning

document.addEventListener('mousemove', function(event) {
    console.log('Screen X:', event.screenX);  // From left of screen
    console.log('Screen Y:', event.screenY);  // From top of screen
});

// Detect which monitor the mouse is on (approximation)
function getMonitorInfo(event) {
    const screenX = event.screenX;
    const screenWidth = window.screen.width;
    
    // Approximate monitor number (for multi-monitor setups)
    const monitorNumber = Math.floor(screenX / screenWidth) + 1;
    return monitorNumber;
}

// Position popup window near mouse
document.getElementById('openPopup').addEventListener('click', function(event) {
    const popupWidth = 300;
    const popupHeight = 200;
    
    window.open(
        'popup.html',
        'popup',
        `width=${popupWidth},height=${popupHeight},` +
        `left=${event.screenX + 10},top=${event.screenY + 10}`
    );
});
```

**Why**: Useful for multi-monitor applications, positioning native windows/popups near cursor, and analytics tracking of user's screen position.

## 7. Keyboard Event - keyCode (Deprecated)

```html
# PROBLEM: Identify which key was pressed using numeric code
# TIME: O(1) SPACE: O(1)

// ⚠️ keyCode is DEPRECATED - use event.key or event.code instead
// Included for legacy code maintenance

document.addEventListener('keydown', function(event) {
    // Legacy approach (deprecated but still works)
    console.log('Key code:', event.keyCode);  // Numeric code
    
    // Modern recommended approach
    console.log('Key:', event.key);     // Character: 'a', 'Enter', 'ArrowUp'
    console.log('Code:', event.code);   // Physical key: 'KeyA', 'Enter', 'ArrowUp'
});

// Common key codes (for reference when maintaining legacy code)
const KEY_CODES = {
    ENTER: 13,
    ESCAPE: 27,
    SPACE: 32,
    ARROW_LEFT: 37,
    ARROW_UP: 38,
    ARROW_RIGHT: 39,
    ARROW_DOWN: 40,
    A: 65,
    Z: 90
};

// Migration example: Old vs New
document.addEventListener('keydown', function(event) {
    // OLD (deprecated)
    if (event.keyCode === 13) { /* Enter pressed */ }
    
    // NEW (recommended)
    if (event.key === 'Enter') { /* Enter pressed */ }
    if (event.code === 'Enter') { /* Physical Enter key */ }
});
```

**Why**: Still encountered in legacy codebases; understanding keyCode helps maintain and migrate older applications to modern key/Code properties.

## 8. Focus Event - relatedTarget

```html
# PROBLEM: Identify the element losing or receiving focus during focus/blur events
# TIME: O(1) SPACE: O(1)

// relatedTarget points to the "other" element in focus transitions
// For focus: relatedTarget = element losing focus
// For blur: relatedTarget = element gaining focus

const input1 = document.getElementById('input1');
const input2 = document.getElementById('input2');

input1.addEventListener('focus', function(event) {
    console.log('Coming from:', event.relatedTarget); // Element that lost focus
});

input1.addEventListener('blur', function(event) {
    console.log('Going to:', event.relatedTarget); // Element that will gain focus
});

// Practical use: Validate only when leaving to another field
const emailInput = document.getElementById('email');
emailInput.addEventListener('blur', function(event) {
    // Only validate if focus went to another form field
    if (event.relatedTarget && event.relatedTarget.tagName === 'INPUT') {
        validateEmail(this.value);
    }
});

// Focus tracking for accessibility
document.addEventListener('focusin', function(event) {
    console.log('Focus entered:', event.target);
    console.log('From:', event.relatedTarget);
});
```

**Why**: Enables smart validation (skip if clicking submit), focus management in complex forms, and building accessible custom components with proper focus tracking.

## 9. Input Event - data

```html
# PROBLEM: Get the inserted text during an input event
# TIME: O(1) SPACE: O(1)

// event.data contains the characters that were inserted
// Useful for real-time input filtering and formatting

const input = document.getElementById('myInput');

input.addEventListener('beforeinput', function(event) {
    console.log('About to insert:', event.data);  // Text to be inserted
    console.log('Input type:', event.inputType);  // 'insertText', 'deleteContent', etc.
    
    // Prevent non-numeric input
    if (event.inputType === 'insertText' && !/^\d*$/.test(event.data)) {
        event.preventDefault();
    }
});

input.addEventListener('input', function(event) {
    // Note: 'input' event may not have event.data in all browsers
    // Use 'beforeinput' for the data being inserted
    console.log('New value:', this.value);
});

// Real-time character filtering
const lettersOnly = document.getElementById('lettersOnly');
lettersOnly.addEventListener('beforeinput', function(event) {
    if (event.inputType === 'insertText' && !/^[a-zA-Z\s]$/.test(event.data)) {
        event.preventDefault();
    }
});

// Live input type detection
function handleInput(event) {
    switch (event.inputType) {
        case 'insertText':
            console.log('User typed:', event.data);
            break;
        case 'deleteContentBackward':
            console.log('User pressed backspace');
            break;
        case 'insertFromPaste':
            console.log('User pasted content');
            break;
    }
}
```

**Why**: Perfect for real-time input validation, masking, auto-formatting (phone numbers, credit cards), and character-level input control.

## 10. Event Delegation

```html
# PROBLEM: Handle events on multiple elements efficiently using a single listener
# TIME: O(1) SPACE: O(1)

// Instead of attaching listeners to each element, use delegation
// Attach one listener to a parent and use event.target

// BAD: Multiple listeners (inefficient)
document.querySelectorAll('.btn').forEach(btn => {
    btn.addEventListener('click', handleClick);
});

// GOOD: Single listener (efficient)
document.getElementById('buttonContainer').addEventListener('click', function(event) {
    // Check if clicked element is a button
    if (event.target.matches('.btn')) {
        console.log('Button clicked:', event.target.textContent);
    }
    
    // Handle nested elements (click on span inside button)
    const button = event.target.closest('.btn');
    if (button) {
        console.log('Button action:', button.dataset.action);
    }
});

// Dynamic content works automatically
const list = document.getElementById('dynamicList');
list.addEventListener('click', function(event) {
    const item = event.target.closest('li');
    if (item) {
        console.log('List item:', item.textContent);
        // Works for items added after listener was attached
    }
});

// Delegation for keyboard events
document.getElementById('menu').addEventListener('keydown', function(event) {
    const item = event.target.closest('.menu-item');
    if (!item) return;
    
    switch (event.key) {
        case 'ArrowDown':
            item.nextElementSibling?.focus();
            break;
        case 'ArrowUp':
            item.previousElementSibling?.focus();
            break;
    }
});
```

**Why**: Reduces memory usage (one listener vs many), automatically handles dynamically added elements, and simplifies event management for large lists or grids.
