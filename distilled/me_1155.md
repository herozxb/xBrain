## HTML Doctype Declaration Parser

```python
# PROBLEM: Parse and validate HTML doctype declarations
# TIME: O(n) SPACE: O(1)
def parse_doctype(html_string: str) -> dict:
    """
    Parse HTML doctype declaration and extract version information.
    Handles HTML5, HTML4, and XHTML doctypes.
    """
    import re
    
    # Pattern to match doctype declaration
    doctype_pattern = r'<!DOCTYPE\s+(\w+)(?:\s+(PUBLIC|SYSTEM))?\s*(?:"([^"]+)")?\s*(?:"([^"]+)")?\s*>'
    
    match = re.search(doctype_pattern, html_string, re.IGNORECASE)
    
    if not match:
        return {"valid": False, "error": "No doctype found"}
    
    doctype = match.group(1).upper()
    system_type = match.group(2)  # PUBLIC or SYSTEM
    public_id = match.group(3)    # Public identifier
    system_id = match.group(4)    # System identifier
    
    # Determine HTML version
    version = "unknown"
    if doctype == "HTML" and not system_type:
        version = "HTML5"
    elif public_id:
        if "HTML 4.01" in public_id:
            version = "HTML 4.01"
        elif "XHTML 1.0" in public_id:
            version = "XHTML 1.0"
        elif "XHTML 1.1" in public_id:
            version = "XHTML 1.1"
    
    return {
        "valid": True,
        "doctype": doctype,
        "version": version,
        "system_type": system_type,
        "public_id": public_id,
        "system_id": system_id
    }
```
**Why**: Validates and extracts HTML version info from doctype; essential for HTML parsers and validators to determine parsing mode.

## HTML Meta Tags Generator

```python
# PROBLEM: Generate SEO-optimized meta tags dynamically
# TIME: O(1) SPACE: O(1)
def generate_meta_tags(
    title: str,
    description: str,
    keywords: list[str],
    author: str = "",
    viewport: bool = True,
    charset: str = "UTF-8",
    og_tags: bool = True,
    twitter_cards: bool = True
) -> str:
    """
    Generate comprehensive HTML meta tags for SEO and social sharing.
    Returns formatted HTML string with all necessary meta tags.
    """
    tags = []
    
    # Charset
    tags.append(f'<meta charset="{charset}">')
    
    # Viewport
    if viewport:
        tags.append('<meta name="viewport" content="width=device-width, initial-scale=1.0">')
    
    # Basic SEO
    tags.append(f'<meta name="description" content="{description}">')
    tags.append(f'<meta name="keywords" content="{", ".join(keywords)}">')
    if author:
        tags.append(f'<meta name="author" content="{author}">')
    
    # Open Graph (Facebook, LinkedIn)
    if og_tags:
        tags.append(f'<meta property="og:title" content="{title}">')
        tags.append(f'<meta property="og:description" content="{description}">')
        tags.append('<meta property="og:type" content="website">')
    
    # Twitter Cards
    if twitter_cards:
        tags.append('<meta name="twitter:card" content="summary_large_image">')
        tags.append(f'<meta name="twitter:title" content="{title}">')
        tags.append(f'<meta name="twitter:description" content="{description}">')
    
    return "\n".join(tags)
```
**Why**: Automates SEO best practices; ensures consistent meta tag generation across pages with proper social media optimization.

## HTML Form Validator

```python
# PROBLEM: Validate HTML form input data server-side
# TIME: O(n) where n = number of fields SPACE: O(n)
def validate_form(
    form_data: dict,
    rules: dict
) -> tuple[bool, dict]:
    """
    Validate form data against rules.
    Rules format: {"field_name": {"type": "email", "required": True, "min": 6}}
    Returns (is_valid, errors_dict).
    """
    import re
    from datetime import datetime
    
    errors = {}
    
    for field, field_rules in rules.items():
        value = form_data.get(field, "")
        
        # Required check
        if field_rules.get("required", False) and not value:
            errors[field] = f"{field} is required"
            continue
        
        # Skip other validations if empty and not required
        if not value:
            continue
        
        # Type validations
        field_type = field_rules.get("type", "text")
        
        if field_type == "email":
            if not re.match(r'^[\w\.-]+@[\w\.-]+\.\w+$', value):
                errors[field] = "Invalid email format"
        
        elif field_type == "url":
            if not re.match(r'^https?://[\w\.-]+', value):
                errors[field] = "Invalid URL format"
        
        elif field_type == "number":
            try:
                num = float(value)
                if "min" in field_rules and num < field_rules["min"]:
                    errors[field] = f"Must be at least {field_rules['min']}"
                if "max" in field_rules and num > field_rules["max"]:
                    errors[field] = f"Must be at most {field_rules['max']}"
            except ValueError:
                errors[field] = "Must be a number"
        
        elif field_type == "date":
            try:
                datetime.strptime(value, "%Y-%m-%d")
            except ValueError:
                errors[field] = "Invalid date format (use YYYY-MM-DD)"
        
        elif field_type == "phone":
            if not re.match(r'^\+?[\d\s-]{10,}$', value):
                errors[field] = "Invalid phone number"
        
        # String length validations
        if field_type in ["text", "password", "textarea"]:
            if "min" in field_rules and len(value) < field_rules["min"]:
                errors[field] = f"Must be at least {field_rules['min']} characters"
            if "max" in field_rules and len(value) > field_rules["max"]:
                errors[field] = f"Must be at most {field_rules['max']} characters"
        
        # Pattern validation
        if "pattern" in field_rules:
            if not re.match(field_rules["pattern"], value):
                errors[field] = field_rules.get("pattern_message", "Invalid format")
    
    return len(errors) == 0, errors
```
**Why**: Server-side validation is security-critical; provides reusable validation logic matching HTML5 form validation attributes.

## HTML Table Generator

```python
# PROBLEM: Generate accessible HTML tables from structured data
# TIME: O(n*m) where n=rows, m=cols SPACE: O(n*m)
def generate_html_table(
    data: list[list],
    headers: list[str],
    caption: str = "",
    responsive: bool = True,
    sortable: bool = False
) -> str:
    """
    Generate semantic, accessible HTML table from 2D data.
    Includes proper ARIA attributes and optional responsive wrapper.
    """
    lines = []
    
    # Responsive wrapper
    if responsive:
        lines.append('<div class="table-responsive" role="region" aria-label="Data table">')
    
    # Table element
    table_id = f"table-{id(data)}"
    sortable_attr = ' data-sortable="true"' if sortable else ""
    lines.append(f'<table id="{table_id}"{sortable_attr}>')
    
    # Caption for accessibility
    if caption:
        lines.append(f'<caption>{caption}</caption>')
    
    # Header row
    lines.append('<thead>')
    lines.append('<tr>')
    for header in headers:
        lines.append(f'<th scope="col">{header}</th>')
    lines.append('</tr>')
    lines.append('</thead>')
    
    # Body rows
    lines.append('<tbody>')
    for row_idx, row in enumerate(data):
        lines.append('<tr>')
        for cell in row:
            # Use td for data cells
            lines.append(f'<td>{cell}</td>')
        lines.append('</tr>')
    lines.append('</tbody>')
    
    lines.append('</table>')
    
    if responsive:
        lines.append('</div>')
    
    return '\n'.join(lines)
```
**Why**: Ensures accessible, semantic HTML tables; automated generation reduces errors and improves consistency across web applications.

## HTML List Tree Builder

```python
# PROBLEM: Convert nested data structure to HTML nested lists
# TIME: O(n) where n = total nodes SPACE: O(d) where d = depth
def build_nested_list(
    data: list[dict],
    ordered: bool = False,
    id_field: str = "id",
    label_field: str = "label",
    children_field: str = "children"
) -> str:
    """
    Convert hierarchical data to nested HTML ul/ol list.
    Data format: [{"id": 1, "label": "Item", "children": [...]}]
    """
    tag = "ol" if ordered else "ul"
    lines = []
    
    def render_items(items: list[dict], depth: int = 0) -> None:
        indent = "  " * depth
        lines.append(f'{indent}<{tag}>')
        
        for item in items:
            label = item.get(label_field, "")
            children = item.get(children_field, [])
            
            if children:
                lines.append(f'{indent}  <li>{label}')
                render_items(children, depth + 2)
                lines.append(f'{indent}  </li>')
            else:
                lines.append(f'{indent}  <li>{label}</li>')
        
        lines.append(f'{indent}</{tag}>')
    
    render_items(data)
    return '\n'.join(lines)
```

## HTML Image Responsive Srcset Generator

```python
# PROBLEM: Generate responsive image srcset for different screen sizes
# TIME: O(n) where n = number of breakpoints SPACE: O(n)
def generate_responsive_image(
    base_url: str,
    widths: list[int],
    sizes: str = "100vw",
    alt: str = "",
    lazy: bool = True,
    webp_fallback: bool = True
) -> str:
    """
    Generate responsive <img> tag with srcset and sizes.
    Automatically handles WebP format with fallback.
    """
    # Generate srcset
    srcset_entries = []
    for width in widths:
        # Assuming image service follows pattern: image.jpg?w=800
        srcset_entries.append(f"{base_url}?w={width} {width}w")
    
    srcset = ", ".join(srcset_entries)
    
    # Default to largest image
    default_width = max(widths)
    default_src = f"{base_url}?w={default_width}"
    
    # Build attributes
    attrs = [
        f'src="{default_src}"',
        f'srcset="{srcset}"',
        f'sizes="{sizes}"',
        f'alt="{alt}"'
    ]
    
    if lazy:
        attrs.append('loading="lazy"')
    
    if webp_fallback:
        # Use picture element for format fallback
        webp_srcset = srcset.replace(".jpg", ".webp").replace(".png", ".webp")
        return f'''<picture>
  <source type="image/webp" srcset="{webp_srcset}" sizes="{sizes}">
  <img {" ".join(attrs)}>
</picture>'''
    
    return f'<img {" ".join(attrs)}>'
```
**Why**: Optimizes image delivery across devices; reduces bandwidth on mobile while serving high-res images on desktop; implements modern lazy loading.

## HTML Link Attribute Builder

```python
# PROBLEM: Generate secure HTML anchor tags with proper attributes
# TIME: O(1) SPACE: O(1)
def build_anchor_tag(
    href: str,
    text: str,
    target: str = "_self",
    rel: list[str] = None,
    download: bool = False,
    download_name: str = None,
    title: str = "",
    aria_label: str = ""
) -> str:
    """
    Generate secure anchor tags with automatic rel attribute handling.
    Automatically adds security attributes for external links.
    """
    if rel is None:
        rel = []
    
    attrs = [f'href="{href}"']
    
    # Handle target
    if target != "_self":
        attrs.append(f'target="{target}"')
        
        # Security: add noopener and noreferrer for target="_blank"
        if target == "_blank":
            if "noopener" not in rel:
                rel.append("noopener")
            if "noreferrer" not in rel:
                rel.append("noreferrer")
    
    # Add rel if not empty
    if rel:
        attrs.append(f'rel="{" ".join(rel)}"')
    
    # Download attribute
    if download:
        attrs.append("download")
        if download_name:
            attrs[-1] = f'download="{download_name}"'
    
    # Accessibility
    if title:
        attrs.append(f'title="{title}"')
    if aria_label:
        attrs.append(f'aria-label="{aria_label}"')
    
    return f'<a {" ".join(attrs)}>{text}</a>'


# Security-focused link checker
def is_safe_redirect(url: str, allowed_domains: list[str] = None) -> bool:
    """Check if URL is safe to redirect to (prevents open redirect attacks)."""
    from urllib.parse import urlparse
    
    if allowed_domains is None:
        return True
    
    parsed = urlparse(url)
    domain = parsed.netloc.lower()
    
    # Block javascript: and data: protocols
    if parsed.scheme in ["javascript", "data", "vbscript"]:
        return False
    
    # Check against allowed domains
    return any(domain == allowed or domain.endswith(f".{allowed}") 
               for allowed in allowed_domains)
```
**Why**: Prevents security vulnerabilities like tabnabbing (noopener/noreferrer) and open redirects; automates secure link generation.

## CSS Box Model Calculator

```python
# PROBLEM: Calculate element dimensions including padding/border/margin
# TIME: O(1) SPACE: O(1)
def calculate_box_dimensions(
    width: float,
    height: float,
    padding: tuple[int, int, int, int] = (0, 0, 0, 0),  # top, right, bottom, left
    border: tuple[int, int, int, int] = (0, 0, 0, 0),
    margin: tuple[int, int, int, int] = (0, 0, 0, 0),
    box_sizing: str = "content-box"
) -> dict:
    """
    Calculate total element dimensions based on CSS box model.
    Returns content, padding, border, and total dimensions.
    """
    p_top, p_right, p_bottom, p_left = padding
    b_top, b_right, b_bottom, b_left = border
    m_top, m_right, m_bottom, m_left = margin
    
    # Content box dimensions
    content_width = width
    content_height = height
    
    # In border-box, width includes padding and border
    if box_sizing == "border-box":
        content_width = width - p_left - p_right - b_left - b_right
        content_height = height - p_top - p_bottom - b_top - b_bottom
    
    # Padding box (content + padding)
    padding_width = content_width + p_left + p_right
    padding_height = content_height + p_top + p_bottom
    
    # Border box (padding + border)
    border_width = padding_width + b_left + b_right
    border_height = padding_height + b_top + b_bottom
    
    # Total space (border + margin)
    total_width = border_width + m_left + m_right
    total_height = border_height + m_top + m_bottom
    
    return {
        "content": {"width": content_width, "height": content_height},
        "padding_box": {"width": padding_width, "height": padding_height},
        "border_box": {"width": border_width, "height": border_height},
        "total": {"width": total_width, "height": total_height},
        "box_sizing": box_sizing
    }
```
**Why**: Essential for layout calculations and debugging; clarifies difference between content-box and border-box sizing models.

## JavaScript Variable Scope Analyzer

```python
# PROBLEM: Analyze JavaScript code for variable scope issues
# TIME: O(n) where n = lines of code SPACE: O(v) where v = variables
def analyze_js_scope(code: str) -> dict:
    """
    Analyze JavaScript code for scope-related issues.
    Detects var, let, const declarations and potential issues.
    """
    import re
    from collections import defaultdict
    
    scopes = defaultdict(list)  # scope_level -> [(var_name, type, line)]
    issues = []
    scope_level = 0
    
    lines = code.split('\n')
    
    # Patterns
    var_pattern = r'\b(var|let|const)\s+(\w+)'
    block_start = r'{'
    block_end = r'}'
    
    for line_num, line in enumerate(lines, 1):
        # Track scope level
        scope_level += line.count('{') - line.count('}')
        
        # Find variable declarations
        for match in re.finditer(var_pattern, line):
            var_type = match.group(1)
            var_name = match.group(2)
            scopes[scope_level].append((var_name, var_type, line_num))
            
            # Check for issues
            if var_type == "var":
                issues.append({
                    "line": line_num,
                    "type": "var_deprecated",
                    "message": f"Consider using 'let' or 'const' instead of 'var' for '{var_name}'"
                })
            
            # Check for shadowing
            for prev_scope in range(scope_level):
                for prev_var, prev_type, prev_line in scopes[prev_scope]:
                    if prev_var == var_name:
                        issues.append({
                            "line": line_num,
                            "type": "variable_shadowing",
                            "message": f"'{var_name}' shadows variable declared at line {prev_line}"
                        })
    
    # Find undeclared variables (simplified)
    usage_pattern = r'\b([a-zA-Z_]\w*)\s*='
    declared = {var for vars in scopes.values() for var, _, _ in vars}
    
    for line_num, line in enumerate(lines, 1):
        for match in re.finditer(usage_pattern, line):
            var_name = match.group(1)
            if var_name not in declared and not var_name.isupper():
                # Skip known globals
                if var_name not in ['window', 'document', 'console', 'Math', 'JSON']:
                    issues.append({
                        "line": line_num,
                        "type": "possible_undeclared",
                        "message": f"'{var_name}' may be used before declaration"
                    })
    
    return {
        "scopes": dict(scopes),
        "issues": issues,
        "total_variables": sum(len(v) for v in scopes.values())
    }
```
**Why**: Helps identify scope-related bugs in JavaScript code; detects var usage, variable shadowing, and potential undeclared variables.

## DOM Element Selector Optimizer

```python
# PROBLEM: Optimize CSS selectors for better performance
# TIME: O(n) where n = selector complexity SPACE: O(1)
def optimize_css_selector(selector: str) -> dict:
    """
    Analyze and optimize CSS selector for performance.
    Returns optimized selector and performance score.
    """
    import re
    
    issues = []
    optimizations = []
    score = 100  # Start with perfect score
    
    # ID selectors are fastest
    if selector.startswith('#'):
        pass  # Already optimal
    
    # Universal selector is slowest
    if selector.startswith('*'):
        issues.append("Universal selector * is very slow")
        optimizations.append("Replace with specific element selector")
        score -= 30
    
    # Check for expensive patterns
    expensive_patterns = [
        (r'\[.*\*=.*\]', "Attribute contains selector is slow", 15),
        (r'\[.*~=.*\]', "Attribute word selector is slow", 10),
        (r':nth-child', "Nth-child requires tree traversal", 10),
        (r':not\(', "Negation pseudo-class adds overhead", 5),
        (r'\s\+\s', "Adjacent sibling combinator is expensive", 10),
        (r'\s~\s', "General sibling combinator is expensive", 10),
        (r'\s{2,}', "Descendant combinator chain is slow", 15),
    ]
    
    for pattern, message, penalty in expensive_patterns:
        if re.search(pattern, selector):
            issues.append(message)
            score -= penalty
    
    # Check selector length
    parts = re.split(r'[\s>+~]', selector)
    if len(parts) > 3:
        issues.append(f"Selector has {len(parts)} parts (recommended: ≤3)")
        score -= (len(parts) - 3) * 5
    
    # Generate optimization suggestions
    if len(parts) > 2:
        optimizations.append("Consider adding a class to target element directly")
    
    # Check for qualified ID selectors (slower than bare ID)
    if re.search(r'\w+#', selector):
        issues.append("Qualified ID selector is unnecessary")
        optimizations.append("Remove element name from ID selector")
        score -= 5
    
    # Check for redundant selectors
    if re.search(r'div\.', selector) or re.search(r'span\.', selector):
        issues.append("Redundant element qualifier before class")
        optimizations.append("Remove element name (e.g., .class instead of div.class)")
        score -= 5
    
    return {
        "original": selector,
        "score": max(0, score),
        "issues": issues,
        "optimizations": optimizations,
        "rating": "Excellent" if score >= 80 else "Good" if score >= 60 else "Fair" if score >= 40 else "Poor"
    }
```
**Why**: CSS selector performance impacts rendering speed; this analyzer identifies expensive patterns and suggests optimizations for faster page loads.
