## 3. html5_rp_element

```html
<!-- RP (Ruby Parenthesis) Fallback -->
<!DOCTYPE html>
<html lang="ja">
<head>
    <meta charset="UTF-8">
    <title>RP Element - Fallback Support</title>
    <style>
        ruby { font-size: 2em; }
    </style>
</head>
<body>
    <ruby>
        日本語
        <rp>(</rp>
        <rt>にほんご</rt>
        <rp>)</rp>
    </ruby>
    
    <!-- JavaScript check for ruby support -->
    <script>
        function checkRubySupport() {
            // Check if browser supports ruby element
            const ruby = document.createElement('ruby');
            const supported = 'webkitAppearance' in ruby.style;
            
            if (!supported) {
                document.body.innerHTML += '<p style="color: red;">Ruby not supported. Fallback parentheses visible.</p>';
            }
            return supported;
        }
        
        checkRubySupport();
    </script>
</body>
</html>
```

**Why**: The `<rp>` element provides fallback parentheses for browsers that don't support ruby annotations, ensuring graceful degradation.

**Time**: O(1) | **Space**: O(1)

## 4. html5_bdi_element

```html
<!-- BDI (Bi-Directional Isolation) Element -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>BDI Element - Bi-Directional Text</title>
    <style>
        ul { list-style: none; padding: 0; }
        li {
            padding: 10px;
            margin: 5px 0;
            border: 1px solid #ddd;
            display: flex;
            justify-content: space-between;
        }
        bdi { font-weight: bold; }
    </style>
</head>
<body>
    <ul>
        <li>User: <bdi>إيان</bdi> - Score: 90</li>
        <li>User: <bdi>Julia</bdi> - Score: 88</li>
        <li>User: <bdi>الفضل</bdi> - Score: 95</li>
        <li>User: <bdi>Sarah</bdi> - Score: 82</li>
    </ul>
    
    <script>
        // Dynamically add users with mixed direction names
        function addUser(name, score) {
            const ul = document.querySelector('ul');
            const li = document.createElement('li');
            const bdi = document.createElement('bdi');
            bdi.textContent = name;
            li.appendChild(document.createTextNode('User: '));
            li.appendChild(bdi);
            li.appendChild(document.createTextNode(` - Score: ${score}`));
            ul.appendChild(li);
        }
        
        // Add more users with RTL names
        addUser('יוסף', 87);
        addUser('Michael', 92);
    </script>
</body>
</html>
```

**Why**: The `<bdi>` element isolates text with unknown directionality, preventing layout issues when mixing LTR and RTL languages.

**Time**: O(1) | **Space**: O(1)

## 5. html5_wbr_element

```html
<!-- WBR (Word Break Opportunity) Element -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>WBR Element - Line Break Control</title>
    <style>
        .container {
            width: 300px;
            border: 2px solid #333;
            padding: 20px;
            margin: 20px 0;
            background: #f5f5f5;
        }
        .url {
            font-family: monospace;
            background: #e0e0e0;
            padding: 5px;
        }
    </style>
</head>
<body>
    <h3>Without WBR (may overflow)</h3>
    <div class="container">
        <p class="url">https://www.example.com/very/long/path/to/some/resource/that/might/overflow</p>
    </div>
    
    <h3>With WBR (breaks at logical points)</h3>
    <div class="container">
        <p class="url">https://www<wbr>.example<wbr>.com<wbr>/very<wbr>/long<wbr>/path<wbr>/to<wbr>/some<wbr>/resource<wbr>/that<wbr>/might<wbr>/overflow</p>
    </div>
    
    <h3>Long compound word</h3>
    <div class="container">
        <p>Supercalifragilistic<wbr>expialidocious<wbr>antidisestablishment<wbr>arianism</p>
    </div>
    
    <script>
        // Add WBR elements programmatically
        function addWordBreakOpportunities(text, segments) {
            return segments.join('<wbr>');
        }
        
        // Example: Add WBR to a long email
        const longEmail = 'john.doe.smith.junior@someverylongcompanyname.com';
        const segments = longEmail.split(/([@.])/);
        document.body.innerHTML += `
            <div class="container">
                <p><strong>Email with WBR:</strong><br>
                <span class="url">${segments.join('<wbr>')}</span></p>
            </div>
        `;
    </script>
</body>
</html>
```

**Why**: The `<wbr>` element suggests line break opportunities within long words or URLs, improving text flow without forcing breaks.

**Time**: O(n) where n = segments | **Space**: O(n)

## 6. html5_data_element

```html
<!-- DATA Element - Machine-Readable Data -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>DATA Element - Machine Readable Values</title>
    <style>
        table {
            border-collapse: collapse;
            width: 100%;
            max-width: 600px;
        }
        th, td {
            border: 1px solid #ddd;
            padding: 12px;
            text-align: left;
        }
        th {
            background-color: #007bff;
            color: white;
        }
        tr:hover {
            background-color: #f5f5f5;
        }
    </style>
</head>
<body>
    <h2>Product Prices</h2>
    <table>
        <thead>
            <tr>
                <th>Product</th>
                <th>Price</th>
                <th>Rating</th>
            </tr>
        </thead>
        <tbody>
            <tr>
                <td>Smartphone</td>
                <td><data value="599.99">$599.99</data></td>
                <td><data value="4.5">★★★★☆</data></td>
            </tr>
            <tr>
                <td>Laptop</td>
                <td><data value="1299.00">$1,299.00</data></td>
                <td><data value="4.8">★★★★★</data></td>
            </tr>
            <tr>
                <td>Headphones</td>
                <td><data value="149.50">$149.50</data></td>
                <td><data value="4.2">★★★★☆</data></td>
            </tr>
        </tbody>
    </table>
    
    <script>
        // Extract and process machine-readable data
        function calculateTotal() {
            const prices = Array.from(document.querySelectorAll('data[value]'))
                .filter(data => !isNaN(parseFloat(data.value)));
            
            const total = prices.reduce((sum, data) => {
                return sum + parseFloat(data.value);
            }, 0);
            
            console.log(`Total price: $${total.toFixed(2)}`);
            return total;
        }
        
        // Sort table by price (machine-readable value)
        function sortByPrice() {
            const tbody = document.querySelector('tbody');
            const rows = Array.from(tbody.querySelectorAll('tr'));
            
            rows.sort((a, b) => {
                const priceA = parseFloat(a.querySelector('data').value);
                const priceB = parseFloat(b.querySelector('data').value);
                return priceA - priceB;
            });
            
            rows.forEach(row => tbody.appendChild(row));
        }
        
        // Add a button to sort
        document.body.innerHTML += `
            <br><button onclick="sortByPrice()">Sort by Price</button>
            <button onclick="alert('Total: $' + calculateTotal().toFixed(2))">Calculate Total</button>
        `;
        
        // Log machine-readable values
        document.querySelectorAll('data').forEach(data => {
            console.log(`Display: ${data.textContent}, Value: ${data.value}`);
        });
    </script>
</body>
</html>
```

**Why**: The `<data>` element links human-readable content with machine-readable values, enabling automated processing while maintaining readable display.

**Time**: O(n log n) for sorting | **Space**: O(n)

```html

<!DOCTYPE html>

        <div class="form-group">
            <label>Slider 1: <output name="s1val">50</output></label>
            <input type="range" name="slider1" value="50" min="0" max="100" 
                   oninput="s1val.value = slider1.value">
        </div>
        <div class="form-group">
            <label>Slider 2: <output name="s2val">30</output></label>
            <input type="range" name="slider2" value="30" min="0" max="100"
                   oninput="s2val.value = slider2.value">
        </div>
        <div class="form-group">
            <label>Slider 3: <output name="s3val">20</output></label>
            <input type="range" name="slider3" value="20" min="0" max="100"
                   oninput="s3val.value = slider3.value">
        </div>
        <div class="form-group">
            <label>Total:</label>
            <output name="total">100</output>
        </div>
    </form>
    
    <script>
        // Access output elements programmatically
        const outputs = document.querySelectorAll('output');
        
        outputs.forEach(output => {
            // Add validation
            output.addEventListener('invalid', function(e) {
                e.target.style.background = '#dc3545';
            });
            
            // Log value changes
            const observer = new MutationObserver((mutations) => {
                mutations.forEach((mutation) => {
                    if (mutation.type === 'attributes' && mutation.attributeName === 'value') {
                        console.log(`${mutation.target.name}: ${mutation.target.value}`);
                    }
                });
            });
            
            observer.observe(output, { attributes: true });
        });
        
        // Get all output values as JSON
        function getFormOutputs() {
            const formData = {};
            document.querySelectorAll('output[name]').forEach(output => {
                formData[output.name] = output.value;
            });
            console.log(JSON.stringify(formData, null, 2));
            return formData;
        }
    </script>
</body>
</html>
```

**Why**: The `<output>` element represents the result of a calculation or user action, providing semantic meaning to form results with accessibility support.

**Time**: O(1) for display | **Space**: O(1)

```html

        }
    </style>
</head>
<body>
    <h2>⚠️ Keygen Element (Deprecated)</h2>
    
    <div class="warning">
        <strong>Warning:</strong> The &lt;keygen&gt; element is deprecated and removed from 
        modern browsers. This example is for educational purposes only. Use the Web Crypto API instead.
    </div>
    
    <!-- Legacy example (won't work in modern browsers) -->
    <form id="legacyForm" action="/submit" method="post">
        <div class="form-group">
            <label>Username:</label>
            <input type="text" name="username" required>
        </div>
        <div class="form-group">
            <label>Key Strength:</label>
            <!-- <keygen name="security" challenge="challenge-string" keytype="RSA"> -->
            <select name="keysize">
                <option value="2048">2048 bits</option>
                <option value="4096">4096 bits</option>
            </select>
        </div>
        <button type="submit">Register</button>
    </form>
    
    <h2>✅ Modern Alternative: Web Crypto API</h2>
    <div id="modernApproach"></div>
    
    <script>
        // Modern replacement using Web Crypto API
        async function generateKeyPair() {
            try {
                // Generate RSA key pair
                const keyPair = await window.crypto.subtle.generateKey(
                    {
                        name: "RSASSA-PKCS1-v1_5",
                        modulusLength: 2048,
                        publicExponent: new Uint8Array([1, 0, 1]),
                        hash: "SHA-256",
                    },
                    true,
                    ["sign", "verify"]
                );
                
                // Export public key
                const exportedPublicKey = await window.crypto.subtle.exportKey(
                    "spki",
                    keyPair.publicKey
                );
                
                // Convert to base64
                const publicKeyBase64 = btoa(
                    String.fromCharCode(...new Uint8Array(exportedPublicKey))
                );
                
                // Export private key (for demo - in production, keep secure!)
                const exportedPrivateKey = await window.crypto.subtle.exportKey(
                    "pkcs8",
                    keyPair.privateKey
                );
                
                const privateKeyBase64 = btoa(
                    String.fromCharCode(...new Uint8Array(exportedPrivateKey))
                );
                
                const output = document.getElementById('modernApproach');
                output.innerHTML = `
                    <div class="form-group">
                        <label>Public Key:</label>
                        <textarea style="width: 100%; height: 100px; font-family: monospace; font-size: 10px;">-----BEGIN PUBLIC KEY-----\n${publicKeyBase64}\n-----END PUBLIC KEY-----</textarea>
                    </div>
                    <div class="form-group">
                        <label>Private Key (Keep Secret!):</label>
                        <textarea style="width: 100%; height: 100px; font-family: monospace; font-size: 10px;">-----BEGIN PRIVATE KEY-----\n${privateKeyBase64}\n-----END PRIVATE KEY-----</textarea>
                    </div>
                    <p><strong>Key pair generated successfully!</strong></p>
                `;
                
                return { publicKey: publicKeyBase64, privateKey: privateKeyBase64 };
            } catch (error) {
                console.error('Key generation failed:', error);
                throw error;
            }
        }
        
        // Auto-generate on page load
        generateKeyPair();
    </script>
</body>
</html>
```

**Why**: The `<keygen>` element (now deprecated) was used for key pair generation. Modern applications should use the Web Crypto API for secure key generation.

**Time**: O(1) | **Space**: O(n) where n = key size

## 9. html5_template_element

```html
<!-- TEMPLATE Element - Reusable HTML Templates -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>TEMPLATE Element - Clonable Content</title>
    <style>
        .card {
            border: 1px solid #ddd;

        }

            margin: 5px 0;
            color: #666;
        }

            font-size: 1.2em;
            font-weight: bold;
            color: #007bff;
        }
        #products {
            display: flex;
            flex-wrap: wrap;
        }
    </style>
</head>
<body>
    <!-- Template - Not rendered initially -->
    <template id="productTemplate">
        <div class="card">
            <img src="" alt="Product Image">
            <h3 class="name"></h3>
            <p class="description"></p>
            <div class="price"></div>
            <button onclick="addToCart(this)">Add to Cart</button>

    <div id="products"></div>
    
    <script>
        // Product data
        const products = [
            {
                id: 1,
                name: "Wireless Headphones",
                description: "Premium noise-canceling headphones",
                price: 299.99,
                image: "https://via.placeholder.com/300x150/007bff/ffffff?text=Headphones"
            },
            {
                id: 2,
                name: "Smart Watch",
                description: "Fitness tracking smartwatch",
                price: 199.99,
                image: "https://via.placeholder.com/300x150/28a745/ffffff?text=Smart+Watch"
            },
            {
                id: 3,
                name: "Bluetooth Speaker",
                description: "Portable waterproof speaker",
                price: 79.99,
                image: "https://via.placeholder.com/300x150/dc3545/ffffff?text=Speaker"
            },
            {
                id: 4,
                name: "USB-C Hub",
                description: "7-in-1 multiport adapter",
                price: 49.99,
                image: "https://via.placeholder.com/300x150/ffc107/000000?text=USB-C+Hub"
            }
        ];
        
        // Function to render products using template
        function renderProducts() {
            const template = document.getElementById('productTemplate');
            const container = document.getElementById('products');
            
            // Clear container
            container.innerHTML = '';
            
            products.forEach(product => {
                // Clone template
                const clone = template.content.cloneNode(true);
                
                // Populate template
                clone.querySelector('img').src = product.image;
                clone.querySelector('img').alt = product.name;
                clone.querySelector('.name').textContent = product.name;
                clone.querySelector('.description').textContent = product.description;
                clone.querySelector('.price').textContent = `$${product.price.toFixed(2)}`;
                
                // Add data attribute
                const card = clone.querySelector('.card');
                card.dataset.productId = product.id;
                
                container.appendChild(clone);
            });
        }
        
        // Add to cart function
        function addToCart(button) {
            const card = button.closest('.card');
            const productId = card.dataset.productId;
            const product = products.find(p => p.id == productId);
            
            alert(`Added "${product.name}" to cart!\nPrice: $${product.price}`);
        }
        
        // Render products on page load
        renderProducts();
        
        // Dynamic template creation example
        function createDynamicTemplate(data) {
            const template = document.createElement('template');
            template.innerHTML = `
                <div class="card" style="border-color: #28a745;">
                    <h3>${data.title}</h3>
                    <p>${data.content}</p>
                </div>
            `;
            return template;
        }
        
        // Usage example
        const dynamicTemplate = createDynamicTemplate({
            title: "Dynamic Card",
            content: "Created programmatically"
        });
        
        document.getElementById('products').appendChild(dynamicTemplate.content.cloneNode(true));
        
        console.log('Templates rendered successfully!');
    </script>
</body>
</html>
```

**Why**: The `<template>` element holds HTML code that can be cloned and inserted multiple times without initial rendering, perfect for dynamic content generation.

**Time**: O(n) for cloning | **Space**: O(n)

## 10. html5_slot_element

```html
<!-- SLOT Element - Web Component Content Projection -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>SLOT Element - Shadow DOM Slots</title>

        <h2 slot="title">🚀 Amazing Feature</h2>
        <p slot="content">
            This content is projected into the card component's slot. 
            You can customize the header, content, and footer independently.
        </p>
        <button slot="footer" onclick="alert('Button clicked!')">Learn More</button>
    </card-component>
    
    <h3>Card 2: Partial Slots</h3>
    <card-component>
        <h2 slot="title">📊 Statistics</h2>
        <div slot="content">
            <p>Users: 1,234</p>
            <p>Revenue: $56,789</p>
        </div>
        <!-- Footer uses default -->
    </card-component>
    
    <h3>Card 3: Default Content</h3>
    <card-component>
        <!-- Uses all default content -->
    </card-component>
    
    <h3>Card 4: Named vs Unnamed Slots</h3>
    <script>
        // Component with both named and default slots
        class ArticleComponent extends HTMLElement {
            constructor() {
                super();
                const shadow = this.attachShadow({ mode: 'open' });
                
                shadow.innerHTML = `
                    <style>
                        :host {
                            display: block;
                            border: 2px solid #007bff;
                            padding: 15px;
                            margin: 10px 0;
                            border-radius: 5px;
                        }
                        .article {
                            background: #f8f9fa;
                        }
                    </style>
                    <div class="article">
                        <slot name="headline"></slot>
                        <slot>This is the default (unnamed) slot content.</slot>
                        <slot name="author"></slot>
                    </div>
                `;
            }
        }
        
        customElements.define('article-component', ArticleComponent);
    </script>
    
    <article-component>
        <h3 slot="headline">📰 Breaking News</h3>
        <p>This goes to the unnamed (default) slot.</p>
        <small slot="author">By: John Doe</small>
    </article-component>
    
    <script>
        // Programmatic slot manipulation
        function addDynamicSlotContent() {
            const card = document.createElement('card-component');
            
            const title = document.createElement('h2');
            title.setAttribute('slot', 'title');
            title.textContent = '⚡ Dynamic Card';
            
            const content = document.createElement('p');
            content.setAttribute('slot', 'content');
            content.textContent = 'This card was created dynamically using JavaScript!';
            
            card.appendChild(title);
            card.appendChild(content);
            
            document.body.appendChild(card);
        }
        
        // Add after 2 seconds
        setTimeout(() => {
            document.body.innerHTML += '<h3>Card 5: Dynamically Created</h3>';
            addDynamicSlotContent();
        }, 2000);
        
        console.log('Slot elements demo loaded!');
    </script>
</body>
</html>
```

**Why**: The `<slot>` element enables content projection in web components, allowing flexible composition and reuse of component templates with customizable content.

**Time**: O(1) | **Space**: O(n) where n = projected content
