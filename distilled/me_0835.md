## Scalable Bloom Filter

```python
# PROBLEM: Dynamically resize bloom filter when capacity is reached
# TIME: O(k) for add/contains where k is number of hash functions
# SPACE: O(m) where m grows exponentially with each filter
import mmh3
from bitarray import bitarray

class ScalableBloomFilter:
    """
    A scalable bloom filter that grows automatically.
    Uses multiple filters with exponentially decreasing error rates.
    """
    def __init__(self, initial_capacity=1000, error_rate=0.001, growth_factor=2):
        self.error_rate = error_rate
        self.growth_factor = growth_factor
        self.filters = []
        self.capacity = 0
        self._add_filter(initial_capacity, error_rate)
    
    def _add_filter(self, capacity, error_rate):
        # Calculate optimal size and number of hash functions
        # m = -n*ln(p) / (ln(2))^2
        import math
        m = int(-capacity * math.log(error_rate) / (math.log(2) ** 2))
        k = int(m / capacity * math.log(2))
        
        self.filters.append({
            'bits': bitarray(m),
            'k': k,
            'count': 0,
            'capacity': capacity,
            'error_rate': error_rate
        })
        self.filters[-1]['bits'].setall(False)
    
    def _hashes(self, item, m, k):
        # Generate k hash values using double hashing
        h1 = mmh3.hash(str(item), 0)
        h2 = mmh3.hash(str(item), h1)
        for i in range(k):
            yield (h1 + i * h2) % m
    
    def add(self, item):
        # Check if we need to scale
        last_filter = self.filters[-1]
        if last_filter['count'] >= last_filter['capacity']:
            # Add new filter with tighter error rate
            new_error = self.error_rate / (len(self.filters) + 1)
            new_capacity = last_filter['capacity'] * self.growth_factor
            self._add_filter(new_capacity, new_error)
            last_filter = self.filters[-1]
        
        # Add to the last filter
        for idx in self._hashes(item, len(last_filter['bits']), last_filter['k']):
            last_filter['bits'][idx] = True
        last_filter['count'] += 1
        self.capacity += 1
    
    def contains(self, item):
        # Check all filters
        for f in self.filters:
            found = True
            for idx in self._hashes(item, len(f['bits']), f['k']):
                if not f['bits'][idx]:
                    found = False
                    break
            if found:
                return True
        return False
```

**Why**: Scales automatically to handle unbounded datasets while maintaining error rate guarantees.

## Quotient Filter

```python
# PROBLEM: Space-efficient probabilistic data structure supporting deletions
# TIME: O(1) amortized for insert/lookup/delete
# SPACE: O(n * log(n/q)) where q is number of slots
class QuotientFilter:
    """
    A quotient filter stores fingerprints using quotienting.
    Supports deletions unlike standard bloom filters.
    """
    def __init__(self, q, r):
        self.q = q  # Quotient bits
        self.r = r  # Remainder bits
        self.size = 1 << q  # 2^q slots
        self.table = [None] * self.size
        self._metadata = [[False, False, False] for _ in range(self.size)]
        # is_occupied, is_continuation, is_shifted
    
    def _hash(self, item):
        h = hash(item) & 0xFFFFFFFF
        quotient = h >> self.r
        remainder = h & ((1 << self.r) - 1)
        return quotient, remainder
    
    def _find_run_start(self, quotient):
        # Navigate to start of run for this quotient
        if not self._metadata[quotient][0]:  # Not occupied
            return quotient
        
        # Count occupied slots before this quotient
        count = 0
        for i in range(quotient):
            if self._metadata[i][0]:
                count += 1
        
        # Find the run by scanning
        pos = 0
        while count > 0:
            if not self._metadata[pos][1]:  # End of a run
                count -= 1
            pos += 1
        
        # Now at start of target run
        while pos < self.size and self._metadata[pos][1]:
            pos += 1
        return pos - 1 if pos > 0 else 0
    
    def insert(self, item):
        q, r = self._hash(item)
        if not self._metadata[q][0]:  # Slot not occupied
            self.table[q] = r
            self._metadata[q] = [True, False, False]
        else:
            # Insert into existing run
            pos = self._find_run_start(q)
            # Shift elements and insert
            self.table[pos] = r
            self._metadata[pos][1] = True  # Mark as continuation
    
    def lookup(self, item):
        q, r = self._hash(item)
        if not self._metadata[q][0]:
            return False
        
        # Search through the run
        pos = self._find_run_start(q)
        while pos < self.size and self.table[pos] != r:
            if not self._metadata[pos][1]:  # End of run
                return False
            pos += 1
        return pos < self.size and self.table[pos] == r
```

**Why**: More space-efficient than bloom filters and supports deletions with good cache locality.

## Distributed Cache (LRU with TTL)

```python
# PROBLEM: Implement distributed cache with eviction and expiration
# TIME: O(1) for get/set with LRU eviction
# SPACE: O(n) for cache entries
import time
from collections import OrderedDict
from threading import Lock

class DistributedCache:
    """
    LRU cache with TTL support for distributed systems.
    Thread-safe with expiration handling.
    """
    def __init__(self, max_size=1000, default_ttl=300):
        self.max_size = max_size
        self.default_ttl = default_ttl
        self.cache = OrderedDict()
        self.lock = Lock()
        self.hits = 0
        self.misses = 0
    
    def get(self, key):
        with self.lock:
            if key not in self.cache:
                self.misses += 1
                return None
            
            value, expiry = self.cache[key]
            if time.time() > expiry:
                # Expired, remove it
                del self.cache[key]
                self.misses += 1
                return None
            
            # Move to end (most recently used)
            self.cache.move_to_end(key)
            self.hits += 1
            return value
    
    def set(self, key, value, ttl=None):
        with self.lock:
            ttl = ttl or self.default_ttl
            expiry = time.time() + ttl
            
            # Remove if exists (to update position)
            if key in self.cache:
                del self.cache[key]
            
            # Evict if at capacity
            while len(self.cache) >= self.max_size:
                self.cache.popitem(last=False)  # Remove oldest
            
            self.cache[key] = (value, expiry)
    
    def delete(self, key):
        with self.lock:
            if key in self.cache:
                del self.cache[key]
                return True
            return False
    
    def clear_expired(self):
        """Remove all expired entries"""
        with self.lock:
            current_time = time.time()
            expired = [k for k, (v, e) in self.cache.items() if current_time > e]
            for key in expired:
                del self.cache[key]
            return len(expired)
    
    def stats(self):
        total = self.hits + self.misses
        hit_rate = self.hits / total if total > 0 else 0
        return {
            'size': len(self.cache),
            'max_size': self.max_size,
            'hits': self.hits,
            'misses': self.misses,
            'hit_rate': hit_rate
        }
```

**Why**: Provides thread-safe caching with automatic expiration and LRU eviction for distributed systems.

```python
# PROBLEM: Create consistent test objects with minimal boilerplate
# TIME: O(1) for object creation
# SPACE: O(1) per factory
from dataclasses import dataclass, field
from typing import Optional
from datetime import datetime

@dataclass
class User:
    id: int
    name: str
    email: str
    age: int
    created_at: datetime = field(default_factory=datetime.now)
    is_active: bool = True

class UserMother:
    """
    Object Mother pattern for User entities.
    Provides pre-configured test fixtures.
    """
    _counter = 0
    
    @classmethod
    def _next_id(cls):
        cls._counter += 1
        return cls._counter
    
    @classmethod
    def create(cls, **kwargs):
        """Create a user with defaults, override with kwargs"""
        defaults = {
            'id': cls._next_id(),
            'name': 'John Doe',

            'age': 30,
            'is_active': True
        }
        defaults.update(kwargs)
        return User(**defaults)
    
    @classmethod

        """Create an active user"""
        return cls.create(is_active=True, **kwargs)
    
    @classmethod
    def inactive_user(cls, **kwargs):
        """Create an inactive user"""
        return cls.create(is_active=False, **kwargs)
    
    @classmethod
    def admin(cls, **kwargs):
        """Create an admin user"""
        return cls.create(
            email='admin@example.com',
            age=35,
            **kwargs
        )
    
    @classmethod

        """Create a user under 18"""
        return cls.create(age=16, **kwargs)
```

```python
from dataclasses import dataclass, field

        self._id = 1
        self._customer_id = 1
        self._items = []
        self._total = 0.0

            'product': product,

        self._status = status
        return self
```

## Mock Object Pattern

```python
# PROBLEM: Create test doubles that verify interactions
# TIME: O(1) for method calls and verifications
# SPACE: O(n) for call history
from typing import Any, List, Dict
from dataclasses import dataclass

@dataclass
class MethodCall:
    name: str
    args: tuple
    kwargs: dict
    returned: Any = None
    raised: Exception = None

class MockObject:
    """
    Generic mock object for testing.
    Records all method calls and allows verification.
    """
    def __init__(self, name: str = "Mock"):
        self._name = name
        self._calls: List[MethodCall] = []
        self._return_values: Dict[str, Any] = {}
        self._side_effects: Dict[str, Exception] = {}
        self._call_counts: Dict[str, int] = {}
    
    def __getattr__(self, name: str):
        def method(*args, **kwargs):
            call = MethodCall(name=name, args=args, kwargs=kwargs)
            self._calls.append(call)
            self._call_counts[name] = self._call_counts.get(name, 0) + 1
            
            # Check for side effect
            if name in self._side_effects:
                call.raised = self._side_effects[name]
                raise self._side_effects[name]
            
            # Return configured value
            if name in self._return_values:
                value = self._return_values[name]
                if callable(value):
                    value = value(*args, **kwargs)
                call.returned = value
                return value
            
            return None
        
        return method
    
    def setup_return(self, method_name: str, value: Any):
        """Configure return value for a method"""
        self._return_values[method_name] = value
        return self
    
    def setup_side_effect(self, method_name: str, exception: Exception):
        """Configure method to raise exception"""
        self._side_effects[method_name] = exception
        return self
    
    def verify_called(self, method_name: str, times: int = None) -> bool:
        """Verify method was called"""
        count = self._call_counts.get(method_name, 0)
        if times is not None:
            return count == times
        return count > 0
    
    def verify_called_with(self, method_name: str, *args, **kwargs) -> bool:
        """Verify method was called with specific arguments"""
        for call in self._calls:
            if call.name == method_name:
                if args and call.args != args:
                    continue
                if kwargs and call.kwargs != kwargs:
                    continue
                return True
        return False
    
    def get_call_count(self, method_name: str) -> int:
        """Get number of times method was called"""
        return self._call_counts.get(method_name, 0)
    
    def reset(self):
        """Clear all recorded calls"""
        self._calls.clear()
        self._call_counts.clear()

# Usage
mock_db = MockObject("Database")
mock_db.setup_return("query", [{"id": 1, "name": "Alice"}])
mock_db.setup_side_effect("connect", ConnectionError("Network down"))

result = mock_db.query("SELECT * FROM users")
assert mock_db.verify_called("query")
assert mock_db.verify_called_with("query", "SELECT * FROM users")
```

**Why**: Enables behavior verification without external dependencies in unit tests.

## Redis Patterns (Distributed Lock)

```python
# PROBLEM: Implement distributed lock using Redis
# TIME: O(1) for lock acquire/release
# SPACE: O(1) per lock
import uuid
import time
import redis

class RedisDistributedLock:
    """
    Distributed lock implementation using Redis SET NX.
    Ensures mutual exclusion across multiple processes/servers.
    """
    def __init__(self, redis_client, lock_name, timeout=10, retry_delay=0.1):
        self.redis = redis_client
        self.lock_name = f"lock:{lock_name}"
        self.timeout = timeout
        self.retry_delay = retry_delay
        self.identifier = str(uuid.uuid4())
        self._acquired = False
    
    def acquire(self, blocking=True, timeout=None):
        """
        Acquire the lock.
        Returns True if lock acquired, False otherwise.
        """
        deadline = time.time() + timeout if timeout else None
        
        while True:
            # SET key value NX PX milliseconds
            acquired = self.redis.set(
                self.lock_name,
                self.identifier,
                nx=True,  # Only set if not exists
                px=int(self.timeout * 1000)  # Expire in milliseconds
            )
            
            if acquired:
                self._acquired = True
                return True
            
            if not blocking:
                return False
            
            if deadline and time.time() >= deadline:
                return False
            
            time.sleep(self.retry_delay)
    
    def release(self):
        """
        Release the lock using Lua script for atomicity.
        Only releases if we own the lock.
        """
        if not self._acquired:
            return False
        
        # Lua script ensures atomic check-and-delete
        lua_script = """
        if redis.call("get", KEYS[1]) == ARGV[1] then
            return redis.call("del", KEYS[1])
        else
            return 0
        end
        """
        
        result = self.redis.eval(lua_script, 1, self.lock_name, self.identifier)
        self._acquired = False
        return result == 1
    
    def extend(self, additional_time):
        """
        Extend the lock timeout.
        Only extends if we still own the lock.
        """
        lua_script = """
        if redis.call("get", KEYS[1]) == ARGV[1] then
            return redis.call("pexpire", KEYS[1], ARGV[2])
        else
            return 0
        end
        """
        
        result = self.redis.eval(
            lua_script, 
            1, 
            self.lock_name, 
            self.identifier,
            int(additional_time * 1000)
        )
        return result == 1
    
    def __enter__(self):
        self.acquire()
        return self
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        self.release()
        return False

# Usage
client = redis.Redis(host='localhost', port=6379)
with RedisDistributedLock(client, "resource_1", timeout=5) as lock:
    if lock._acquired:
        # Critical section
        print("Lock acquired, doing work...")
```

**Why**: Provides mutual exclusion across distributed systems with automatic expiration to prevent deadlocks.

## BASE Properties

```python
# PROBLEM: Implement BASE-compliant event store for eventual consistency
# TIME: O(1) for event append, O(n) for projection rebuild
# SPACE: O(n) for event log
from dataclasses import dataclass
from datetime import datetime
from typing import List, Dict, Any
import json

@dataclass
class Event:
    id: int
    aggregate_id: str
    event_type: str
    data: Dict[str, Any]
    timestamp: datetime
    version: int

class EventStore:
    """
    Event store implementing BASE properties:
    - Basically Available: Always accept writes
    - Soft state: State may change without input
    - Eventually consistent: Converges to consistent state
    """
    def __init__(self):
        self.events: List[Event] = []
        self._sequence = 0
        self._projections: Dict[str, Dict] = {}
    
    def append(self, aggregate_id: str, event_type: str, data: dict) -> Event:
        """
        Append event to stream.
        Always succeeds (Basically Available).
        """
        self._sequence += 1
        event = Event(
            id=self._sequence,
            aggregate_id=aggregate_id,
            event_type=event_type,
            data=data,
            timestamp=datetime.now(),
            version=len([e for e in self.events if e.aggregate_id == aggregate_id]) + 1
        )
        self.events.append(event)
        # Trigger async projection update
        self._schedule_projection_update(aggregate_id)
        return event
    
    def get_events(self, aggregate_id: str, from_version: int = 0) -> List[Event]:
        """Get all events for an aggregate"""
        return [
            e for e in self.events
            if e.aggregate_id == aggregate_id and e.version > from_version
        ]
    
    def _schedule_projection_update(self, aggregate_id: str):
        """
        Schedule projection update (eventual consistency).
        In production, this would be a message queue.
        """
        # Simulated async update
        self._update_projection(aggregate_id)
    
    def _update_projection(self, aggregate_id: str):
        """Rebuild projection from event stream"""
        events = self.get_events(aggregate_id)
        if not events:
            return
        
        projection = {'id': aggregate_id, 'version': 0}
        
        # Apply events in order (event sourcing)
        for event in events:
            if event.event_type == 'UserCreated':
                projection.update({
                    'name': event.data['name'],
                    'email': event.data['email'],
                    'status': 'active'
                })
            elif event.event_type == 'UserEmailChanged':
                projection['email'] = event.data['new_email']
            elif event.event_type == 'UserDeactivated':
                projection['status'] = 'inactive'
            
            projection['version'] = event.version
        
        self._projections[aggregate_id] = projection
    
    def get_projection(self, aggregate_id: str) -> Dict:
        """
        Get current projection (may be stale).
        This is the 'soft state' - eventually consistent.
        """
        return self._projections.get(aggregate_id)
    
    def get_all_events(self, event_type: str = None) -> List[Event]:
        """Get all events, optionally filtered by type"""
        if event_type:
            return [e for e in self.events if e.event_type == event_type]
        return self.events.copy()

# Usage
store = EventStore()
store.append("user-1", "UserCreated", {"name": "Alice", "email": "alice@example.com"})
store.append("user-1", "UserEmailChanged", {"new_email": "alice.new@example.com"})
projection = store.get_projection("user-1")
```

**Why**: Enables high availability and partition tolerance while guaranteeing eventual consistency.

## Eventual Consistency Monitor

```python
# PROBLEM: Track and verify eventual consistency across distributed nodes
# TIME: O(1) for updates, O(n) for consistency check
# SPACE: O(n * m) where n = entities, m = nodes
from collections import defaultdict
from typing import Dict, Set, Tuple
from dataclasses import dataclass
import time

@dataclass
class VectorClock:
    """Vector clock for tracking causality"""
    clocks: Dict[str, int]
    
    def increment(self, node_id: str):
        self.clocks[node_id] = self.clocks.get(node_id, 0) + 1
    
    def merge(self, other: 'VectorClock'):
        for node, clock in other.clocks.items():
            self.clocks[node] = max(self.clocks.get(node, 0), clock)
    
    def happens_before(self, other: 'VectorClock') -> bool:
        """Check if this clock happens-before other"""
        all_less_or_equal = all(
            self.clocks.get(n, 0) <= other.clocks.get(n, 0)
            for n in set(self.clocks.keys()) | set(other.clocks.keys())
        )
        any_less = any(
            self.clocks.get(n, 0) < other.clocks.get(n, 0)
            for n in self.clocks.keys()
        )
        return all_less_or_equal and any_less

class EventualConsistencyMonitor:
    """
    Monitors eventual consistency across distributed nodes.
    Tracks versions and detects when consistency is achieved.
    """
    def __init__(self, nodes: Set[str], consistency_window: float = 5.0):
        self.nodes = nodes
        self.consistency_window = consistency_window  # seconds
        self._entity_versions: Dict[str, Dict[str, Tuple[VectorClock, float]]] = defaultdict(dict)
        self._inconsistencies: List[dict] = []
    
    def update(self, entity_id: str, node_id: str, version: VectorClock):
        """
        Record a version update from a node.
        """
        timestamp = time.time()
        self._entity_versions[entity_id][node_id] = (version, timestamp)
        self._check_consistency(entity_id)
    
    def _check_consistency(self, entity_id: str):
        """Check if entity is consistent across all nodes"""
        node_versions = self._entity_versions[entity_id]
        
        if len(node_versions) < len(self.nodes):
            # Not all nodes have reported
            return
        
        # Get all versions
        versions = [(node, vc, ts) for node, (vc, ts) in node_versions.items()]
        
        # Check for conflicts
        for i, (node1, vc1, ts1) in enumerate(versions):
            for node2, vc2, ts2 in versions[i+1:]:
                if not vc1.happens_before(vc2) and not vc2.happens_before(vc1):
                    # Concurrent versions - inconsistency
                    age = min(time.time() - ts1, time.time() - ts2)
                    
                    if age > self.consistency_window:
                        self._inconsistencies.append({
                            'entity_id': entity_id,
                            'nodes': [node1, node2],
                            'age': age,
                            'timestamp': time.time()
                        })
    
    def is_consistent(self, entity_id: str) -> bool:
        """Check if entity has achieved consistency"""
        node_versions = self._entity_versions[entity_id]
        
        if len(node_versions) != len(self.nodes):
            return False
        
        # All nodes must have the same version
        versions = [vc for vc, _ in node_versions.values()]
        first = versions[0]
        
        return all(
            not first.happens_before(v) and not v.happens_before(first)
            for v in versions[1:]
        )
    
    def get_inconsistencies(self, entity_id: str = None) -> List[dict]:
        """Get detected inconsistencies"""
        if entity_id:
            return [i for i in self._inconsistencies if i['entity_id'] == entity_id]
        return self._inconsistencies.copy()
    
    def consistency_lag(self, entity_id: str) -> float:
        """Get time since last consistent state"""
        node_versions = self._entity_versions[entity_id]
        if not node_versions:
            return 0.0
        
        timestamps = [ts for _, ts in node_versions.values()]
        return time.time() - max(timestamps)

# Usage
nodes = {'node1', 'node2', 'node3'}
```

## Pydantic Patterns (Data Validation)

```python
# PROBLEM: Validate and serialize complex data structures with type safety
# TIME: O(n) for validation where n = fields
# SPACE: O(n) for validated model
from pydantic import BaseModel, validator, Field, constr, confloat
from typing import List, Optional
from datetime import datetime
from enum import Enum

class OrderStatus(str, Enum):
    PENDING = "pending"
    CONFIRMED = "confirmed"
    SHIPPED = "shipped"
    DELIVERED = "delivered"

class Address(BaseModel):
    """Nested model for address"""
    street: constr(min_length=1, max_length=200)
    city: constr(min_length=1, max_length=100)
    country: constr(min_length=2, max_length=100)
    postal_code: constr(regex=r'^\d{5,10}$')
    
    class Config:
        # Immutable after creation
        frozen = True

class OrderItem(BaseModel):
    """Order line item with validation"""
    product_id: int = Field(..., gt=0)
    name: constr(min_length=1)
    quantity: int = Field(..., gt=0, le=100)
    price: confloat(gt=0, le=100000)
    
    @validator('price')
    def validate_price(cls, v, values):
        # Business rule: quantity discount
        if 'quantity' in values and values['quantity'] > 10:
            return v * 0.9  # 10% discount for bulk
        return v

class Order(BaseModel):
    """
    Order model with comprehensive validation.
    Demonstrates Pydantic patterns for production use.
    """
    id: int = Field(..., description="Unique order ID")
    customer_email: constr(regex=r'^[\w\.-]+@[\w\.-]+\.\w+$')
    items: List[OrderItem] = Field(..., min_items=1)
    shipping_address: Address
    status: OrderStatus = OrderStatus.PENDING
    created_at: datetime = Field(default_factory=datetime.now)
    notes: Optional[constr(max_length=500)] = None
    
    class Config:
        # Validate on assignment
        validate_assignment = True
        # Use enum values
        use_enum_values = True
    
    @validator('items')
    def validate_items(cls, v):
        # Business rule: max 20 items per order
        if len(v) > 20:
            raise ValueError('Order cannot have more than 20 items')
        return v
    
    @property
    def total(self) -> float:
        """Calculate order total"""
        return sum(item.quantity * item.price for item in self.items)
    
    def update_status(self, new_status: OrderStatus):
        """Transition status with validation"""
        valid_transitions = {
            OrderStatus.PENDING: [OrderStatus.CONFIRMED],
            OrderStatus.CONFIRMED: [OrderStatus.SHIPPED],
            OrderStatus.SHIPPED: [OrderStatus.DELIVERED],
        }
        
        if new_status not in valid_transitions.get(self.status, []):
            raise ValueError(f"Invalid status transition: {self.status} -> {new_status}")
        
        self.status = new_status
        return self

# Usage
order_data = {
    "id": 1001,
    "customer_email": "alice@example.com",
    "items": [
        {"product_id": 1, "name": "Widget", "quantity": 5, "price": 10.00},
        {"product_id": 2, "name": "Gadget", "quantity": 15, "price": 20.00}
    ],
    "shipping_address": {
        "street": "123 Main St",
        "city": "Springfield",
        "country": "USA",
        "postal_code": "12345"
    }
}

order = Order(**order_data)  # Validates and coerces
print(f"Order total: ${order.total:.2f}")
order.update_status(OrderStatus.CONFIRMED)
```

**Why**: Provides runtime type safety, automatic validation, and serialization with clear error messages.
