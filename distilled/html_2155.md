/**
 * Parse metadata VTT cues and extract chapter information
 * @param {HTMLTrackElement} track - The track element containing metadata
 * @returns {Array} Array of chapter objects with start, end, and text
 */
function parseMetadataTrack(track) {
  const chapters = [];
  
  // Wait for track to load if not ready
  if (track.readyState === 0) {
    return new Promise((resolve) => {
      track.addEventListener('load', () => {
        resolve(extractCues(track));
      });
    });
  }
  
  return extractCues(track);
}

/**
 * Extract cue data from loaded track
 * @param {HTMLTrackElement} track - Loaded track element
 * @returns {Array} Array of chapter objects
 */
function extractCues(track) {
  const cues = track.track.cues;
  const chapters = [];
  
  for (let i = 0; i < cues.length; i++) {
    const cue = cues[i];
    chapters.push({
      startTime: cue.startTime,
      endTime: cue.endTime,
      text: cue.text,
      id: cue.id || `chapter-${i}`
    });
  }
  
  return chapters;
}

/**
 * Detect optimal source based on media query
 * @param {HTMLPictureElement} picture - The picture element
 * @returns {string} URL of the optimal source
 */
function getOptimalSource(picture) {
  const sources = picture.querySelectorAll('source');
  
  for (const source of sources) {
    const media = source.getAttribute('media');
    
    // Check if media query matches current viewport
    if (media && window.matchMedia(media).matches) {
      return source.srcset || source.src;
    }
  }
  
  // Return fallback image source
  const img = picture.querySelector('img');
  return img ? img.src : null;
}

/**
 * Create responsive video element with media query sources

 * @returns {HTMLVideoElement} Configured video element
 */
function createResponsiveVideo(config) {
  const video = document.createElement('video');
  video.controls = true;
  
  // Define breakpoints and sources
  const breakpoints = [

/**
 * Update source on viewport resize (debounced)
 * @param {HTMLVideoElement} video - The video element
 * @param {Array} sources - Array of {media, src} objects
 */
function setupResponsiveSourceSwitching(video, sources) {
  let debounceTimer;
  
  const updateSource = () => {
    for (const { media, src } of sources) {
      if (window.matchMedia(media).matches && video.src !== src) {
        const currentTime = video.currentTime;
        const wasPaused = video.paused;
        
        video.src = src;
        video.currentTime = currentTime;
        
        if (!wasPaused) {
          video.play();
        }
        break;
      }
    }
  };
  
  window.addEventListener('resize', () => {
    clearTimeout(debounceTimer);
    debounceTimer = setTimeout(updateSource, 250);
  });
}

/**
 * Art direction helper: generate picture element with multiple crops
 * @param {Object} config - Art direction configuration
 * @returns {HTMLPictureElement} Configured picture element
 */
function createArtDirectedImage(config) {
  const picture = document.createElement('picture');
  
  // Process each art direction variant
  config.variants.forEach(variant => {
    const source = document.createElement('source');
    
    // Set media query
    source.media = variant.media;
    
    // Handle srcset with multiple resolutions
    if (variant.srcset && Array.isArray(variant.srcset)) {
      source.srcset = variant.srcset
        .map(s => `${s.url} ${s.descriptor}`)
        .join(', ');

    // Optional: set type for format switching
    if (variant.type) {
      source.type = variant.type;

    picture.appendChild(source);
  });
  
  // Create fallback img element
  const img = document.createElement('img');
  img.src = config.fallback.src;
  img.alt = config.fallback.alt;
  img.loading = config.fallback.lazy ? 'lazy' : 'eager';
  img.decoding = 'async';
  
  picture.appendChild(img);
  
  return picture;
}

/**
 * Accessible Modal Manager
 * Handles focus trapping, keyboard navigation, and ARIA states
 */
class AccessibleModal {
  /**
   * Create a new accessible modal
   * @param {HTMLElement} modal - The modal element
   */
  constructor(modal) {
    this.modal = modal;
    this.focusableSelector = [
      'button:not([disabled])',
      '[href]',
      'input:not([disabled])',
      'select:not([disabled])',
      'textarea:not([disabled])',
      '[tabindex]:not([tabindex="-1"])'
    ].join(', ');
    
    this.previousActiveElement = null;
    this.focusableElements = [];
    
    this.init();
  }
  
  /**
   * Initialize event listeners
   */
  init() {
    // Close on backdrop click
    this.modal.querySelectorAll('[data-close-modal]').forEach(el => {
      el.addEventListener('click', () => this.close());
    });
    
    // Handle keyboard events
    this.modal.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') {
        this.close();
      }
      if (e.key === 'Tab') {
        this.trapFocus(e);
      }
    });
  }
  
  /**
   * Open the modal
   */
  open() {
    // Store current focus to restore later
    this.previousActiveElement = document.activeElement;
    
    // Show modal
    this.modal.hidden = false;
    this.modal.setAttribute('aria-hidden', 'false');
    
    // Add body class to prevent scrolling
    document.body.classList.add('modal-open');
    
    // Get all focusable elements
    this.focusableElements = [
      ...this.modal.querySelectorAll(this.focusableSelector)
    ];
    
    // Focus first element
    if (this.focusableElements.length > 0) {
      this.focusableElements[0].focus();
    }
    
    // Dispatch open event
    this.modal.dispatchEvent(new CustomEvent('modal:open'));
  }
  
  /**
   * Close the modal
   */
  close() {
    this.modal.hidden = true;
    this.modal.setAttribute('aria-hidden', 'true');
    document.body.classList.remove('modal-open');
    
    // Restore focus to previous element
    if (this.previousActiveElement) {
      this.previousActiveElement.focus();
    }
    
    // Dispatch close event
    this.modal.dispatchEvent(new CustomEvent('modal:close'));
  }
  
  /**
   * Trap focus within modal
   * @param {KeyboardEvent} e - The keyboard event
   */
  trapFocus(e) {
    const firstFocusable = this.focusableElements[0];
    const lastFocusable = this.focusableElements[
      this.focusableElements.length - 1
    ];
    
    if (e.shiftKey) {
      // Shift + Tab: going backwards
      if (document.activeElement === firstFocusable) {
        lastFocusable.focus();
        e.preventDefault();
      }
    } else {
      // Tab: going forwards
      if (document.activeElement === lastFocusable) {
        firstFocusable.focus();
        e.preventDefault();
      }
    }
  }
  
  /**
   * Check if modal is open
   * @returns {boolean} True if modal is visible
   */
  isOpen() {
    return !this.modal.hidden;
  }
}

/**
 * Live Region Announcer
 * Manages ARIA live region announcements for screen readers
 */
class LiveAnnouncer {
  /**
   * Create a new announcer
   * @param {HTMLElement} politeRegion - Polite live region element
   * @param {HTMLElement} assertiveRegion - Assertive live region element
   */
  constructor(politeRegion, assertiveRegion) {
    this.politeRegion = politeRegion;
    this.assertiveRegion = assertiveRegion;
    this.announcementQueue = [];
    this.isAnnouncing = false;
  }
  
  /**
   * Announce a message politely (non-interrupting)
   * @param {string} message - Message to announce
   * @param {number} delay - Delay before announcement (ms)
   */
  announcePolite(message, delay = 100) {
    this.queueAnnouncement(message, 'polite', delay);
  }
  
  /**
   * Announce a message assertively (interrupting)
   * @param {string} message - Message to announce
   * @param {number} delay - Delay before announcement (ms)
   */
  announceAssertive(message, delay = 50) {
    this.queueAnnouncement(message, 'assertive', delay);
  }
  
  /**
   * Queue an announcement
   * @param {string} message - Message to announce
   * @param {string} priority - 'polite' or 'assertive'
   * @param {number} delay - Delay before announcement
   */
  queueAnnouncement(message, priority, delay) {
    this.announcementQueue.push({ message, priority, delay });
    
    if (!this.isAnnouncing) {
      this.processQueue();
    }
  }
  
  /**
   * Process the announcement queue
   */
  async processQueue() {
    this.isAnnouncing = true;
    
    while (this.announcementQueue.length > 0) {
      const { message, priority, delay } = this.announcementQueue.shift();
      
      await this.delay(delay);
      
      const region = priority === 'assertive' 
        ? this.assertiveRegion 
        : this.politeRegion;
      
      // Clear and set message (triggers announcement)
      region.textContent = '';
      await this.delay(50);
      region.textContent = message;
      
      // Wait for announcement to complete
      await this.delay(500);
    }
    
    this.isAnnouncing = false;
  }
  
  /**
   * Promise-based delay
   * @param {number} ms - Milliseconds to delay
   * @returns {Promise}
   */
  delay(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

      metaTags.push(this.createMetaTag('twitter:player:height', config.player.height));

  truncate(str, maxLength) {

    return str.substring(0, maxLength - 3) + '...';
  },

   * Escape HTML entities
   * @param {string} str - String to escape
   * @returns {string} Escaped string

  escapeHtml(str) {

    template.innerHTML = html;
    
    template.content.childNodes.forEach(node => {
      document.head.appendChild(node.cloneNode(true));
    });
  }

  preload(href, options = {}) {

    link.rel = 'preload';
    link.href = href;

    }

    }

        const link = this.preload(resource.href, resource);

const ConicGradient = {
  /**
   * Create a conic gradient string
   * @param {Object} config - Gradient configuration
   * @returns {string} CSS gradient string
   */
  create(config) {
    const {
      from = 0,
      position = 'center',
      stops = []
    } = config;
    
    // Build gradient parts
    const parts = [];
    
    // Add from angle
    if (from !== 0) {
      parts.push(`from ${from}deg`);
    }
    
    // Add position
    if (position !== 'center') {
      parts.push(`at ${position}`);
    }
    
    // Add color stops
    const stopsStr = stops.map(stop => {
      if (typeof stop === 'string') return stop;
      return `${stop.color} ${stop.position}`;
    }).join(', ');
    
    return `conic-gradient(${parts.join(' ')}, ${stopsStr})`;
  },
  
  /**
   * Create a pie chart gradient
   * @param {Array} segments - Array of {value, color} objects
   * @returns {string} CSS gradient string
   */
  pieChart(segments) {
    const total = segments.reduce((sum, s) => sum + s.value, 0);
    const stops = [];
    let currentAngle = 0;
    
    segments.forEach(segment => {
      const angle = (segment.value / total) * 360;
      stops.push(`${segment.color} ${currentAngle}deg ${currentAngle + angle}deg`);
      currentAngle += angle;
    });
    
    return `conic-gradient(${stops.join(', ')})`;
  },
