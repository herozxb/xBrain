## css_transition_shorthand

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>transition Shorthand</title>
  <style>
    /* PROBLEM: Create smooth animated transitions between states
       TIME: O(1) SPACE: O(1) */
    
    /* transition: property duration timing-function delay */
    
    .box {
      width: 100px;
      height: 100px;
      background-color: #007bff;
      margin: 20px;
      /* Shorthand: all properties, 0.3s duration, ease timing */
      transition: all 0.3s ease;
    }
    
    .box:hover {
      width: 200px;
      background-color: #28a745;
      border-radius: 50%;
    }
    
    /* Multiple transitions */
    .multi {
      width: 100px;
      height: 100px;
      background-color: #dc3545;
      margin: 20px;
      /* Different timing for different properties */
      transition: width 0.5s ease, 
                  background-color 0.3s ease-in-out, 
                  transform 0.2s linear;
    }
    
    .multi:hover {
      width: 150px;
      background-color: #ffc107;
      transform: rotate(45deg);
    }
    
    /* With delay */
    .delayed {
      width: 100px;
      height: 100px;
      background-color: #17a2b8;
      margin: 20px;
      /* 0.5s delay before transition starts */
      transition: transform 0.3s ease 0.5s;
    }
    
    .delayed:hover {
      transform: scale(1.5);
    }
  </style>
</head>
<body>
  <h2>transition Shorthand Demo</h2>

  <div class="multi">Multiple</div>

</html>
```
**Why**: The `transition` shorthand combines multiple transition properties into one declaration, making animations concise and readable while creating smooth visual feedback.

## js_destructuring_objects

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Object Destructuring</title>
</head>
<body>
  <h2>Object Destructuring Demo</h2>
  <pre id="output"></pre>
  
  <script>
    // PROBLEM: Extract values from objects into variables
    // TIME: O(1) SPACE: O(1)
    
    const log = (text) => {
      document.getElementById('output').textContent += text + '\n';
    };
    
    // Basic destructuring
    const user = {
      name: 'Alice',
      age: 30,
      email: 'alice@example.com'
    };
    
    const { name, age } = user;
    log(`Name: ${name}, Age: ${age}`);
    
    // Renaming variables
    const { name: userName, email: userEmail } = user;
    log(`User: ${userName}, Email: ${userEmail}`);
    
    // Default values
    const { city = 'Unknown' } = user;
    log(`City: ${city}`);
    
    // Nested destructuring
    const person = {
      info: {
        firstName: 'Bob',
        lastName: 'Smith'
      },
      address: {
        city: 'NYC',
        country: 'USA'
      }
    };
    
    const { info: { firstName }, address: { city: personCity } } = person;
    log(`First Name: ${firstName}, City: ${personCity}`);
    
    // Rest properties
    const { name: personName, ...rest } = user;
    log(`Name: ${personName}`);
    log(`Rest: ${JSON.stringify(rest)}`);
    
    // Function parameter destructuring
    function greet({ name, age }) {
      return `Hello ${name}, you are ${age} years old`;
    }
    
    log(greet(user));
    
    // In function defaults
    function setup({ theme = 'light', size = 'medium' } = {}) {
      log(`Theme: ${theme}, Size: ${size}`);
    }
    
    setup({ theme: 'dark' });
    setup();
  </script>
</body>
</html>
```
**Why**: Object destructuring extracts properties into variables with a concise syntax, supports renaming, defaults, and nesting, making code cleaner when working with objects.

## js_promise_catch

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Promise.catch() Method</title>
</head>
<body>
  <h2>Promise.catch() Demo</h2>
  <pre id="output"></pre>
  
  <script>
    // PROBLEM: Handle errors in Promise chains
    // TIME: O(1) SPACE: O(1)
    
    const log = (text) => {
      document.getElementById('output').textContent += text + '\n';
    };
    
    // Basic catch
    const failingPromise = new Promise((resolve, reject) => {
      reject('Something went wrong!');
    });
    
    failingPromise
      .then(result => log(`Success: ${result}`))
      .catch(error => log(`Caught: ${error}`));
    
    // Catch in chain
    function fetchUser(id) {
      return new Promise((resolve, reject) => {
        setTimeout(() => {
          if (id > 0) {
            resolve({ id, name: `User ${id}` });
          } else {
            reject(new Error('Invalid user ID'));
          }
        }, 500);
      });
    }
    
    fetchUser(-1)
      .then(user => log(`User: ${JSON.stringify(user)}`))
      .catch(err => log(`Error: ${err.message}`));
    
    // Catch can recover
    fetchUser(-1)
      .catch(err => {
        log(`Retrying after error: ${err.message}`);
        return { id: 0, name: 'Default User' };
      })
      .then(user => log(`Recovered with: ${JSON.stringify(user)}`));
    
    // Multiple catches in chain
    function step1() {
      return Promise.resolve('Step 1 complete');
    }
    
    function step2() {
      return Promise.reject('Step 2 failed');
    }
    
    function step3() {
      return Promise.resolve('Step 3 complete');
    }
    
    step1()
      .then(result => {
        log(result);
        return step2();
      })
      .catch(err => {
        log(`Caught at step 2: ${err}`);
        throw err; // Re-throw to continue chain
      })
      .then(() => step3())
      .catch(err => log(`Final catch: ${err}`));
    
    // Catch with finally
    fetchUser(1)
      .then(user => log(`Found: ${JSON.stringify(user)}`))
      .catch(err => log(`Error: ${err.message}`))
      .finally(() => log('Operation complete (finally)'));
    
    // Unhandled rejection warning prevention
    window.addEventListener('unhandledrejection', event => {
      log(`Unhandled rejection: ${event.reason}`);
      event.preventDefault();
    });
  </script>
</body>
</html>
```
**Why**: `.catch()` handles Promise rejections in a chain, allows error recovery, and provides centralized error handling for asynchronous operations.

## js_setTimeout

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>setTimeout Function</title>
  <style>
    .box {
      width: 100px;
      height: 100px;
      background-color: #007bff;
      margin: 20px;
      transition: all 0.3s ease;
    }
    .box.active {
      background-color: #28a745;
      transform: scale(1.2);
    }
  </style>
</head>
<body>
  <h2>setTimeout Demo</h2>
  <div class="box" id="box"></div>
  <button onclick="startTimer()">Start Timer (2s)</button>
  <button onclick="cancelTimer()">Cancel Timer</button>
  <p id="status">Click to start</p>
  <pre id="output"></pre>
  
  <script>
    // PROBLEM: Execute code after a delay
    // TIME: O(1) SPACE: O(1)
    
    const log = (text) => {
      document.getElementById('output').textContent += text + '\n';
    };
    
    let timerId = null;
    
    // Basic setTimeout
    function startTimer() {
      document.getElementById('status').textContent = 'Waiting...';
      log('Timer started');
      
      timerId = setTimeout(() => {
        document.getElementById('box').classList.add('active');
        document.getElementById('status').textContent = 'Timer completed!';
        log('Timer executed');
      }, 2000);
    }
    
    function cancelTimer() {
      if (timerId) {
        clearTimeout(timerId);
        document.getElementById('status').textContent = 'Timer cancelled';
        log('Timer cancelled');
        timerId = null;
      }
    }
    
    // setTimeout with parameters
    function greet(name, greeting = 'Hello') {
      log(`${greeting}, ${name}!`);
    }
    
    setTimeout(greet, 1000, 'Alice', 'Hi');
    
    // Recursive setTimeout (interval-like behavior)
    let count = 0;
    function countUp() {
      count++;
      log(`Count: ${count}`);
      if (count < 5) {
        setTimeout(countUp, 500);
      }
    }
    
    log('Starting count:');
    setTimeout(countUp, 500);
    
    // Zero delay - executes after current call stack
    log('Before timeout');
    setTimeout(() => log('Zero delay timeout'), 0);
    log('After timeout');
    
    // Using this context
    const counter = {
      count: 0,
      start() {
        // Without bind, 'this' would be window
        setTimeout(() => {
          this.count++;
          log(`Counter: ${this.count}`);
        }, 1500);
      }
    };
    
    counter.start();
  </script>
</body>
</html>
```
**Why**: `setTimeout` delays code execution, useful for animations, debouncing, polling, and scheduling tasks. Zero-delay timeouts defer execution to the next event loop tick.

## js_string_split

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>String.split() Method</title>
</head>
<body>
  <h2>String.split() Demo</h2>
  <pre id="output"></pre>
  
  <script>
    // PROBLEM: Split string into array of substrings
    // TIME: O(n) SPACE: O(n)
    
    const log = (text) => {
      document.getElementById('output').textContent += text + '\n';
    };
    
    // Basic split
    const str = 'Hello World JavaScript';
    const words = str.split(' ');
    log(`Split by space: ${JSON.stringify(words)}`);
    
    // Split by comma
    const csv = 'apple,banana,cherry,date';
    const fruits = csv.split(',');
    log(`CSV split: ${JSON.stringify(fruits)}`);
    
    // Limit parameter
    const limited = str.split(' ', 2);
    log(`Limited to 2: ${JSON.stringify(limited)}`);
    
    // Split into characters
    const chars = 'Hello'.split('');
    log(`Characters: ${JSON.stringify(chars)}`);
    
    // Split by regex
    const text = 'apple123banana456cherry';
    const byNumber = text.split(/\d+/);
    log(`Split by numbers: ${JSON.stringify(byNumber)}`);
    
    // Split by multiple delimiters
    const mixed = 'apple,banana;cherry|date';
    const byMultiple = mixed.split(/[,;|]/);
    log(`Multiple delimiters: ${JSON.stringify(byMultiple)}`);
    
    // URL parsing
    const url = 'https://example.com/path/to/page?query=value';
    const parts = url.split('/');
    log(`URL parts: ${JSON.stringify(parts)}`);
    
    // Query string parsing
    const queryString = 'name=Alice&age=30&city=NYC';
    const params = queryString.split('&').map(pair => {
      const [key, value] = pair.split('=');
      return { key, value };
    });
    log(`Query params: ${JSON.stringify(params)}`);
    
    // Reverse string using split
    const reversed = 'Hello'.split('').reverse().join('');
    log(`Reversed: ${reversed}`);
    
    // Count words
    const sentence = 'The quick brown fox jumps over the lazy dog';
    const wordCount = sentence.split(' ').length;
    log(`Word count: ${wordCount}`);
    
    // Split with capturing groups (included in result)
    const date = '2026-02-25';
    const dateParts = date.split(/(-)/);
    log(`With delimiters: ${JSON.stringify(dateParts)}`);
    
    // Empty string split
    const empty = ''.split(',');
    log(`Empty split: ${JSON.stringify(empty)}`);
    
    // Practical example: parse CSV line
    function parseCSVLine(line) {
      return line.split(',').map(field => field.trim());
    }
    
    const csvLine = 'Alice, 30, alice@example.com, NYC';
    log(`Parsed CSV: ${JSON.stringify(parseCSVLine(csvLine))}`);
  </script>
</body>
</html>
```
**Why**: `String.split()` divides strings into arrays using a separator, essential for parsing CSV data, tokenizing text, and manipulating string content.
