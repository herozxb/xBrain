## Line Intersection

```python
# PROBLEM: Determine if two line segments intersect and find intersection point
# TIME: O(1) SPACE: O(1)
def line_intersection(p1, p2, p3, p4):
    """
    Find intersection point of two line segments.
    Each point is a tuple (x, y).
    Returns intersection point or None if segments don't intersect.
    """
    x1, y1 = p1
    x2, y2 = p2
    x3, y3 = p3
    x4, y4 = p4
    
    # Calculate denominator for parametric equations
    denom = (x1 - x2) * (y3 - y4) - (y1 - y2) * (x3 - x4)
    
    # Lines are parallel if denominator is zero
    if abs(denom) < 1e-10:
        return None
    
    # Calculate parameters t and u
    t = ((x1 - x3) * (y3 - y4) - (y1 - y3) * (x3 - x4)) / denom
    u = -((x1 - x2) * (y1 - y3) - (y1 - y2) * (x1 - x3)) / denom
    
    # Check if intersection point lies on both segments
    if 0 <= t <= 1 and 0 <= u <= 1:
        ix = x1 + t * (x2 - x1)
        iy = y1 + t * (y2 - y1)
        return (ix, iy)
    
    return None
```
**Why**: Fundamental computational geometry problem used in graphics, CAD, collision detection, and map applications.

## Edit Distance Variants (Damerau-Levenshtein)

```python
# PROBLEM: Calculate edit distance allowing adjacent transpositions
# TIME: O(m*n) SPACE: O(m*n)
def damerau_levenshtein(s1, s2):
    """
    Compute Damerau-Levenshtein distance.
    Allows insertions, deletions, substitutions, and adjacent swaps.
    """
    m, n = len(s1), len(s2)
    
    # Create DP table
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    
    # Initialize base cases
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j
    
    # Fill DP table
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            cost = 0 if s1[i-1] == s2[j-1] else 1
            
            dp[i][j] = min(
                dp[i-1][j] + 1,      # deletion
                dp[i][j-1] + 1,      # insertion
                dp[i-1][j-1] + cost  # substitution
            )
            
            # Check for adjacent transposition
            if (i > 1 and j > 1 and 
                s1[i-1] == s2[j-2] and s1[i-2] == s2[j-1]):
                dp[i][j] = min(dp[i][j], dp[i-2][j-2] + 1)
    
    return dp[m][n]
```
**Why**: More realistic edit distance for typos where adjacent characters are swapped; used in spell checkers and fuzzy string matching.

```python
# PROBLEM: Find all palindromic substrings in a string
# TIME: O(n^2) SPACE: O(1)
def count_palindrome_substrings(s):
    """
    Count all palindromic substrings using center expansion.
    """
    n = len(s)
    count = 0
    
    def expand_around_center(left, right):
        """Expand outward from center while palindrome holds."""
        local_count = 0
        while left >= 0 and right < n and s[left] == s[right]:
            local_count += 1
            left -= 1
            right += 1
        return local_count
    
    # Check all possible centers
    for i in range(n):
        # Odd length palindromes (single character center)
        count += expand_around_center(i, i)
        
        # Even length palindromes (two character center)
        count += expand_around_center(i, i + 1)
    
    return count

def longest_palindromic_substring(s):
    """
    Find the longest palindromic substring.
    """
    if not s:
        return ""
    
    start, max_len = 0, 1
    
    def expand_around_center(left, right):
        nonlocal start, max_len
        while left >= 0 and right < len(s) and s[left] == s[right]:
            current_len = right - left + 1
            if current_len > max_len:
                start = left
                max_len = current_len
            left -= 1
            right += 1
    
    for i in range(len(s)):
        expand_around_center(i, i)     # odd length
        expand_around_center(i, i + 1) # even length
    
    return s[start:start + max_len]
```
**Why**: Core string algorithm for palindrome detection; applications in DNA sequence analysis and text processing.

## Partition Equal Subset Sum

```python
# PROBLEM: Determine if array can be partitioned into two equal-sum subsets
# TIME: O(n * sum) SPACE: O(sum)
def can_partition_equal_subset(nums):
    """
    Check if array can be split into two subsets with equal sum.
    Uses 1D DP optimization for space efficiency.
    """
    total = sum(nums)
    
    # If total is odd, equal partition impossible
    if total % 2 != 0:
        return False
    
    target = total // 2
    
    # dp[j] = True if sum j is achievable
    dp = [False] * (target + 1)
    dp[0] = True
    
    for num in nums:
        # Traverse backwards to avoid using same element twice
        for j in range(target, num - 1, -1):
            dp[j] = dp[j] or dp[j - num]
    
    return dp[target]
```
**Why**: Classic NP-complete problem; demonstrates subset sum DP optimization; used in resource allocation and scheduling.

```python
# PROBLEM: Determine if string can be segmented into dictionary words
# TIME: O(n^2) SPACE: O(n)
def word_break(s, word_dict):
    """
    Check if string can be segmented into space-separated
    sequence of dictionary words.
    """
    word_set = set(word_dict)
    n = len(s)
    
    # dp[i] = True if s[0:i] can be segmented
    dp = [False] * (n + 1)
    dp[0] = True  # empty string is always valid
    
    for i in range(1, n + 1):
        for j in range(i):
            # If s[0:j] is valid and s[j:i] is in dictionary
            if dp[j] and s[j:i] in word_set:
                dp[i] = True
                break
    
    return dp[n]

def word_break_with_solutions(s, word_dict):
    """
    Return all possible word break combinations.
    """
    word_set = set(word_dict)
    memo = {}
    
    def backtrack(start):
        if start in memo:
            return memo[start]
        
        if start == len(s):
            return [""]  # Found valid segmentation
        
        results = []
        for end in range(start + 1, len(s) + 1):
            word = s[start:end]
            if word in word_set:
                # Recursively get rest of sentence
                for subsentence in backtrack(end):
                    if subsentence:
                        results.append(word + " " + subsentence)
                    else:
                        results.append(word)
        
        memo[start] = results
        return results
    
    return backtrack(0)
```
**Why**: Essential NLP algorithm used in tokenization, spell checking, and text segmentation in multiple languages.

## Double Ended Queue (Deque)

```python
# PROBLEM: Implement a double-ended queue with O(1) operations
# TIME: O(1) for all operations SPACE: O(n)
class Deque:
    """
    Double-ended queue implementation using doubly linked list.
    Supports O(1) insertions and deletions at both ends.
    """
    class Node:
        __slots__ = ['val', 'prev', 'next']
        def __init__(self, val):
            self.val = val
            self.prev = None
            self.next = None
    
    def __init__(self):
        self.head = None
        self.tail = None
        self.size = 0
    
    def append_left(self, val):
        """Add element to front."""
        new_node = self.Node(val)
        if self.size == 0:
            self.head = self.tail = new_node
        else:
            new_node.next = self.head
            self.head.prev = new_node
            self.head = new_node
        self.size += 1
    
    def append_right(self, val):
        """Add element to back."""
        new_node = self.Node(val)
        if self.size == 0:
            self.head = self.tail = new_node
        else:
            new_node.prev = self.tail
            self.tail.next = new_node
            self.tail = new_node
        self.size += 1
    
    def pop_left(self):
        """Remove and return front element."""
        if self.size == 0:
            raise IndexError("pop from empty deque")
        
        val = self.head.val
        self.head = self.head.next
        if self.head:
            self.head.prev = None
        else:
            self.tail = None
        self.size -= 1
        return val
    
    def pop_right(self):
        """Remove and return back element."""
        if self.size == 0:
            raise IndexError("pop from empty deque")
        
        val = self.tail.val
        self.tail = self.tail.prev
        if self.tail:
            self.tail.next = None
        else:
            self.head = None
        self.size -= 1
        return val
    
    def peek_left(self):
        """View front element without removing."""
        if self.size == 0:
            raise IndexError("peek from empty deque")
        return self.head.val
    
    def peek_right(self):
        """View back element without removing."""
        if self.size == 0:
            raise IndexError("peek from empty deque")
        return self.tail.val
    
    def __len__(self):
        return self.size
    
    def is_empty(self):
        return self.size == 0
```
**Why**: Versatile data structure for sliding window algorithms, BFS traversal, and palindrome checking; Python's collections.deque is optimized version.

## Binomial Heap

```python
# PROBLEM: Implement mergeable heap supporting efficient union
# TIME: insert=O(log n), extract_min=O(log n), union=O(log n)
class BinomialNode:
    """Node in a binomial heap."""
    __slots__ = ['key', 'degree', 'parent', 'child', 'sibling']
    
    def __init__(self, key):
        self.key = key
        self.degree = 0
        self.parent = None
        self.child = None
        self.sibling = None

class BinomialHeap:
    """
    Binomial heap supporting efficient union operations.
    Collection of binomial trees satisfying min-heap property.
    """
    def __init__(self):
        self.head = None
        self.size = 0
    
    def is_empty(self):
        return self.head is None
    
    def merge(self, other):
        """Merge two heaps by linking root lists."""
        if other.head is None:
            return self
        
        if self.head is None:
            self.head = other.head
            return self
        
        # Merge sorted root lists
        new_head = None
        current = None
        
        h1, h2 = self.head, other.head
        
        while h1 and h2:
            if h1.degree <= h2.degree:
                if current:
                    current.sibling = h1
                else:
                    new_head = h1
                current = h1
                h1 = h1.sibling
            else:
                if current:
                    current.sibling = h2
                else:
                    new_head = h2
                current = h2
                h2 = h2.sibling
        
        # Attach remaining nodes
        remaining = h1 if h1 else h2
        if current:
            current.sibling = remaining
        
        self.head = new_head
        self.size += other.size
        self._consolidate()
        return self
    
    def _consolidate(self):
        """Ensure only one tree per degree."""
        if self.head is None:
            return
        
        # Map degree to tree
        degree_map = {}
        current = self.head
        
        while current:
            next_node = current.sibling
            current.sibling = None
            
            degree = current.degree
            while degree in degree_map:
                other = degree_map[degree]
                del degree_map[degree]
                current = self._link(current, other)
                degree += 1
            
            degree_map[degree] = current
            current = next_node
        
        # Rebuild root list
        sorted_trees = sorted(degree_map.values(), key=lambda x: x.degree)
        for i in range(len(sorted_trees) - 1):
            sorted_trees[i].sibling = sorted_trees[i + 1]
        
        self.head = sorted_trees[0] if sorted_trees else None
    
    def _link(self, tree1, tree2):
        """Link two binomial trees of same degree."""
        if tree1.key > tree2.key:
            tree1, tree2 = tree2, tree1
        
        tree2.parent = tree1
        tree2.sibling = tree1.child
        tree1.child = tree2
        tree1.degree += 1
        
        return tree1
    
    def insert(self, key):
        """Insert new key into heap."""
        new_heap = BinomialHeap()
        new_heap.head = BinomialNode(key)
        new_heap.size = 1
        self.merge(new_heap)
    
    def find_min(self):
        """Find minimum key in heap."""
        if self.head is None:
            return None
        
        min_key = self.head.key
        current = self.head.sibling
        
        while current:
            if current.key < min_key:
                min_key = current.key
            current = current.sibling
        
        return min_key
    
    def extract_min(self):
        """Remove and return minimum key."""
        if self.head is None:
            return None
        
        # Find min node and its predecessor
        min_node = self.head
        min_prev = None
        current = self.head
        prev = None
        
        while current:
            if current.key < min_node.key:
                min_node = current
                min_prev = prev
            prev = current
            current = current.sibling
        
        # Remove min node from root list
        if min_prev:
            min_prev.sibling = min_node.sibling
        else:
            self.head = min_node.sibling
        
        # Reverse min node's children to create new heap
        child_heap = BinomialHeap()
        child = min_node.child
        prev_child = None
        
        while child:
            next_child = child.sibling
            child.sibling = prev_child
            child.parent = None
            prev_child = child
            child = next_child
        
        child_heap.head = prev_child
        
        self.merge(child_heap)
        self.size -= 1
        
        return min_node.key
```
**Why**: Supports efficient heap union in O(log n); used in priority queue applications requiring frequent merges like Dijkstra's with decrease-key.

## MinHash (Locality Sensitive Hashing)

```python
# PROBLEM: Estimate Jaccard similarity between sets efficiently
# TIME: O(k) per comparison SPACE: O(k) for k hash functions
import random

class MinHash:
    """
    MinHash for estimating set similarity.
    Uses multiple hash functions to create compact signatures.
    """
    def __init__(self, num_hashes=128, seed=42):
        """
        Initialize MinHash with k hash functions.
        More hashes = better accuracy, larger signatures.
        """
        self.num_hashes = num_hashes
        random.seed(seed)
        
        # Generate random coefficients for hash functions
        # h(x) = (a*x + b) mod LARGE_PRIME
        self.max_hash = (1 << 32) - 1  # Large prime
        self.a_coeffs = [random.randint(1, self.max_hash) for _ in range(num_hashes)]
        self.b_coeffs = [random.randint(0, self.max_hash) for _ in range(num_hashes)]
    
    def _hash(self, x, i):
        """Compute ith hash function on value x."""
        return (self.a_coeffs[i] * x + self.b_coeffs[i]) % self.max_hash
    
    def create_signature(self, items):
        """
        Create MinHash signature for a set of items.
        Items can be strings, will be hashed to integers.
        """
        signature = [self.max_hash] * self.num_hashes
        
        for item in items:
            # Convert string to integer if needed
            if isinstance(item, str):
                item_hash = hash(item) & self.max_hash
            else:
                item_hash = item
            
            # Update minimum for each hash function
            for i in range(self.num_hashes):
                h = self._hash(item_hash, i)
                if h < signature[i]:
                    signature[i] = h
        
        return signature
    
    def jaccard_similarity(self, sig1, sig2):
        """
        Estimate Jaccard similarity from two signatures.
        Jaccard(A,B) = |A ∩ B| / |A ∪ B|
        """
        if len(sig1) != len(sig2):
            raise ValueError("Signatures must have same length")
        
        matches = sum(1 for a, b in zip(sig1, sig2) if a == b)
        return matches / len(sig1)
    
    def true_jaccard(self, set1, set2):
        """Compute actual Jaccard similarity (for validation)."""
        intersection = len(set1 & set2)
        union = len(set1 | set2)
        return intersection / union if union > 0 else 0.0

# Example usage for document similarity
def document_similarity(doc1_words, doc2_words, num_hashes=128):
    """
    Estimate similarity between two documents.
    More efficient than comparing full word sets.
    """
    mh = MinHash(num_hashes=num_hashes)
    
    sig1 = mh.create_signature(doc1_words)
    sig2 = mh.create_signature(doc2_words)
    
    return mh.jaccard_similarity(sig1, sig2)
```
**Why**: Enables fast similarity estimation in massive datasets; crucial for duplicate detection, recommendation systems, and near-duplicate document search.

## Interpreter Pattern

```python
# PROBLEM: Evaluate expressions in a custom language
# TIME: O(n) for parsing, O(depth) for evaluation SPACE: O(n)
from abc import ABC, abstractmethod
from typing import Dict, List

class Expression(ABC):
    """Abstract expression in interpreter pattern."""
    
    @abstractmethod
    def interpret(self, context: Dict[str, float]) -> float:
        """Evaluate expression with given variable context."""
        pass
    
    @abstractmethod
    def __str__(self) -> str:
        pass

class Number(Expression):
    """Terminal expression for numeric values."""
    
    def __init__(self, value: float):
        self.value = value
    
    def interpret(self, context: Dict[str, float]) -> float:
        return self.value
    
    def __str__(self) -> str:
        return str(self.value)

class Variable(Expression):
    """Terminal expression for variables."""
    
    def __init__(self, name: str):
        self.name = name
    
    def interpret(self, context: Dict[str, float]) -> float:
        if self.name not in context:
            raise ValueError(f"Variable '{self.name}' not defined")
        return context[self.name]
    
    def __str__(self) -> str:
        return self.name

class BinaryOperation(Expression):
    """Non-terminal expression for binary operations."""
    
    def __init__(self, left: Expression, right: Expression, operator: str):
        self.left = left
        self.right = right
        self.operator = operator
    
    def interpret(self, context: Dict[str, float]) -> float:
        left_val = self.left.interpret(context)
        right_val = self.right.interpret(context)
        
        if self.operator == '+':
            return left_val + right_val
        elif self.operator == '-':
            return left_val - right_val
        elif self.operator == '*':
            return left_val * right_val
        elif self.operator == '/':
            if right_val == 0:
                raise ValueError("Division by zero")
            return left_val / right_val
        else:
            raise ValueError(f"Unknown operator: {self.operator}")
    
    def __str__(self) -> str:
        return f"({self.left} {self.operator} {self.right})"

class Parser:
    """
    Simple arithmetic expression parser.
    Supports +, -, *, / and parentheses with proper precedence.
    """
    
    def __init__(self, text: str):
        self.tokens = self._tokenize(text)
        self.pos = 0
    
    def _tokenize(self, text: str) -> List[str]:
        """Convert input string to tokens."""
        import re
        pattern = r'\d+\.?\d*|[a-zA-Z_]\w*|[+\-*/()]'
        return re.findall(pattern, text.replace(' ', ''))
    
    def parse(self) -> Expression:
        """Parse and return expression tree."""
        result = self._parse_expression()
        if self.pos < len(self.tokens):
            raise ValueError(f"Unexpected token: {self.tokens[self.pos]}")
        return result
    
    def _parse_expression(self) -> Expression:
        """Parse addition/subtraction (lowest precedence)."""
        left = self._parse_term()
        
        while self.pos < len(self.tokens) and self.tokens[self.pos] in '+-':
            op = self.tokens[self.pos]
            self.pos += 1
            right = self._parse_term()
            left = BinaryOperation(left, right, op)
        
        return left
    
    def _parse_term(self) -> Expression:
        """Parse multiplication/division (higher precedence)."""
        left = self._parse_factor()
        
        while self.pos < len(self.tokens) and self.tokens[self.pos] in '*/':
            op = self.tokens[self.pos]
            self.pos += 1
            right = self._parse_factor()
            left = BinaryOperation(left, right, op)
        
        return left
    
    def _parse_factor(self) -> Expression:
        """Parse numbers, variables, and parentheses."""
        if self.pos >= len(self.tokens):
            raise ValueError("Unexpected end of expression")
        
        token = self.tokens[self.pos]
        
        if token == '(':
            self.pos += 1
            result = self._parse_expression()
            if self.pos >= len(self.tokens) or self.tokens[self.pos] != ')':
                raise ValueError("Missing closing parenthesis")
            self.pos += 1
            return result
        elif token[0].isdigit() or (token[0] == '-' and len(token) > 1):
            self.pos += 1
            return Number(float(token))
        elif token[0].isalpha() or token[0] == '_':
            self.pos += 1
            return Variable(token)
        else:
            raise ValueError(f"Unexpected token: {token}")

# Interpreter with context
class ArithmeticInterpreter:
    """Complete interpreter for arithmetic expressions."""
    
    def __init__(self):
        self.context: Dict[str, float] = {}
    
    def set_variable(self, name: str, value: float):
        """Define a variable."""
        self.context[name] = value
    
    def evaluate(self, expression: str) -> float:
        """Parse and evaluate an expression string."""
        parser = Parser(expression)
        ast = parser.parse()
        return ast.interpret(self.context)
```
**Why**: Enables domain-specific languages, rule engines, and expression evaluation; foundation for DSLs in business rules and configuration.

## Sidecar Pattern

```python
# PROBLEM: Add cross-cutting concerns to services without modifying them
# TIME: O(1) for most operations SPACE: O(n)
import threading
import time
import logging
from typing import Callable, Any, Optional
from dataclasses import dataclass
from functools import wraps

@dataclass
class Request:
    """Represents an incoming request."""
    id: str
    data: Any
    headers: dict
    timestamp: float = time.time()

@dataclass
class Response:
    """Represents an outgoing response."""
    request_id: str
    status: int
    data: Any
    duration_ms: float

class Sidecar:
    """
    Sidecar pattern implementation.
    Wraps a service to add cross-cutting concerns:
    logging, metrics, retry, circuit breaker, etc.
    """
    
    def __init__(self, service_name: str):
        self.service_name = service_name
        self.logger = logging.getLogger(f"sidecar.{service_name}")
        self.metrics = {
            'requests_total': 0,
            'requests_success': 0,
            'requests_failed': 0,
            'total_duration_ms': 0
        }
        self._lock = threading.Lock()
        
        # Circuit breaker state
        self.circuit_open = False
        self.failure_count = 0
        self.failure_threshold = 5
        self.recovery_timeout = 30  # seconds
        self.last_failure_time = 0
    
    def _record_metric(self, metric: str, value: float = 1):
        """Thread-safe metric recording."""
        with self._lock:
            self.metrics[metric] += value
    
    def _check_circuit(self) -> bool:
        """Check if circuit breaker should allow request."""
        if not self.circuit_open:
            return True
        
        # Check if recovery timeout has passed
        if time.time() - self.last_failure_time > self.recovery_timeout:
            self.logger.info("Circuit breaker entering half-open state")
            return True
        
        return False
    
    def _record_success(self):
        """Record successful request."""
        with self._lock:
            self.failure_count = 0
            self.circuit_open = False
            self.metrics['requests_success'] += 1
    
    def _record_failure(self):
        """Record failed request."""
        with self._lock:
            self.failure_count += 1
            self.metrics['requests_failed'] += 1
            self.last_failure_time = time.time()
            
            if self.failure_count >= self.failure_threshold:
                self.circuit_open = True
                self.logger.warning("Circuit breaker opened due to failures")
    
    def intercept(self, func: Callable) -> Callable:
        """Decorator to add sidecar concerns to service method."""
        @wraps(func)
        def wrapper(request: Request) -> Response:
            request_id = request.id
            
            # Pre-processing: logging and metrics
            self.logger.info(f"[{request_id}] Processing request")
            self._record_metric('requests_total')
            
            # Circuit breaker check
            if not self._check_circuit():
                self.logger.warning(f"[{request_id}] Request rejected - circuit open")
                return Response(
                    request_id=request_id,
                    status=503,
                    data={"error": "Service unavailable - circuit breaker open"},
                    duration_ms=0
                )
            
            # Execute with timing
            start_time = time.time()
            try:
                result = func(request)
                duration_ms = (time.time() - start_time) * 1000
                
                # Post-processing: success metrics
                self._record_success()
                self._record_metric('total_duration_ms', duration_ms)
                
                self.logger.info(
                    f"[{request_id}] Request completed in {duration_ms:.2f}ms"
                )
                
                return Response(
                    request_id=request_id,
                    status=200,
                    data=result,
                    duration_ms=duration_ms
                )
                
            except Exception as e:
                duration_ms = (time.time() - start_time) * 1000
                
                # Post-processing: failure handling
                self._record_failure()
                self.logger.error(f"[{request_id}] Request failed: {e}")
                
                return Response(
                    request_id=request_id,
                    status=500,
                    data={"error": str(e)},
                    duration_ms=duration_ms
                )
        
        return wrapper
    
    def get_metrics(self) -> dict:
        """Return current metrics snapshot."""
        with self._lock:
            avg_duration = (
                self.metrics['total_duration_ms'] / self.metrics['requests_total']
                if self.metrics['requests_total'] > 0 else 0
            )
            return {
                **self.metrics,
                'avg_duration_ms': avg_duration,
                'circuit_open': self.circuit_open,
                'failure_count': self.failure_count
            }

# Example service using sidecar
class UserService:
    """Example service with sidecar pattern."""
    
    def __init__(self):
        self.sidecar = Sidecar("user-service")
        self.users = {
            "1": {"id": "1", "name": "Alice", "email": "alice@example.com"},
            "2": {"id": "2", "name": "Bob", "email": "bob@example.com"}
        }
    
    @Sidecar.intercept
    def get_user(self, request: Request) -> dict:
        """Get user by ID - sidecar adds logging/metrics."""
        user_id = request.data.get('user_id')
        if user_id not in self.users:
            raise ValueError(f"User {user_id} not found")
        return self.users[user_id]
    
    def get_sidecar_metrics(self) -> dict:
        """Access sidecar metrics."""
        return self.sidecar.get_metrics()
```
**Why**: Implements separation of concerns in microservices; adds observability, resilience, and security without modifying application code.
