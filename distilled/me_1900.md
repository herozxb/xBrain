## 1. Suffix Tree Construction
```python
# PROBLEM: Build a suffix tree for efficient pattern matching in strings
# TIME: O(n²) SPACE: O(n²) for naive implementation
class SuffixTreeNode:
    """Node in the suffix tree structure."""
    def __init__(self):
        self.children = {}  # Map from character to child node
        self.start = None   # Start index of the edge label
        self.end = None     # End index of the edge label
        self.suffix_link = None  # Link to another node for optimization
        self.suffix_index = -1   # Starting index of suffix if leaf

class SuffixTree:
    """Suffix tree for pattern matching and text analysis."""
    
    def __init__(self, text):
        self.text = text
        self.root = SuffixTreeNode()
        self._build_suffix_tree()
    
    def _build_suffix_tree(self):
        """Construct suffix tree by adding each suffix."""
        n = len(self.text)
        
        # Add each suffix to the tree
        for i in range(n):
            self._add_suffix(i)
    
    def _add_suffix(self, start_idx):
        """Add a suffix starting at start_idx to the tree."""
        current = self.root
        i = start_idx
        
        # Traverse the tree and add new nodes as needed
        while i < len(self.text):
            char = self.text[i]
            
            if char not in current.children:
                # Create new leaf node
                node = SuffixTreeNode()
                node.start = i
                node.end = len(self.text) - 1
                node.suffix_index = start_idx
                current.children[char] = node
                return
            
            # Follow existing path
            child = current.children[char]
            j = child.start
            
            # Find matching prefix
            while j <= child.end and i < len(self.text) and self.text[i] == self.text[j]:
                i += 1
                j += 1
            
            if j > child.end:
                # Continue with child node
                current = child
            else:
                # Split the node
                split_node = SuffixTreeNode()
                split_node.start = child.start
                split_node.end = j - 1
                
                current.children[char] = split_node
                split_node.children[self.text[j]] = child
                child.start = j
                
                # Add new leaf
                leaf = SuffixTreeNode()
                leaf.start = i
                leaf.end = len(self.text) - 1
                leaf.suffix_index = start_idx
                split_node.children[self.text[i]] = leaf
                return
    
    def search(self, pattern):
        """Search for pattern in the text. Returns True if found."""
        current = self.root
        
        for i, char in enumerate(pattern):
            if char not in current.children:
                return False
            
            child = current.children[char]
            j = child.start
            
            # Match along the edge
            while j <= child.end and i < len(pattern):
                if self.text[j] != pattern[i]:
                    return False
                j += 1
                i += 1
            
            if i < len(pattern):
                current = child
        
        return True
```
**Why**: Suffix trees enable O(m) pattern matching after O(n²) preprocessing, essential for bioinformatics and text search applications.

## 2. Link-Cut Tree
```python
# PROBLEM: Maintain a dynamic forest with path queries and link/cut operations
# TIME: O(log n) amortized per operation SPACE: O(n)
class LinkCutNode:
    """Node in the link-cut tree structure."""
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None
        self.parent = None
        self.path_parent = None  # For link-cut tree
        self.size = 1
        self.reversed = False

class LinkCutTree:
    """Link-cut tree for dynamic tree operations."""
    
    def __init__(self, n):
        """Initialize n nodes with values 0 to n-1."""
        self.nodes = [LinkCutNode(i) for i in range(n)]
    
    def _push(self, node):
        """Push lazy propagation for reversal."""
        if node and node.reversed:
            node.left, node.right = node.right, node.left
            if node.left:
                node.left.reversed ^= True
            if node.right:
                node.right.reversed ^= True
            node.reversed = False
    
    def _update(self, node):
        """Update node size."""
        if node:
            node.size = 1
            if node.left:
                node.size += node.left.size
            if node.right:
                node.size += node.right.size
    
    def _rotate(self, node):
        """Rotate node up in the splay tree."""
        parent = node.parent
        grandparent = parent.parent
        
        if parent.left == node:
            parent.left = node.right
            if node.right:
                node.right.parent = parent
            node.right = parent
        else:
            parent.right = node.left
            if node.left:
                node.left.parent = parent
            node.left = parent
        
        node.parent = grandparent
        parent.parent = node
        
        if grandparent:
            if grandparent.left == parent:
                grandparent.left = node
            else:
                grandparent.right = node
        else:
            # Update path parent
            node.path_parent = parent.path_parent
            parent.path_parent = None
        
        self._update(parent)
        self._update(node)
    
    def _splay(self, node):
        """Splay node to root of its auxiliary tree."""
        self._push(node)
        
        while node.parent:
            parent = node.parent
            grandparent = parent.parent
            
            if grandparent:
                self._push(grandparent)
            self._push(parent)
            self._push(node)
            
            if grandparent:
                # Zig-zag or zig-zig
                if (grandparent.left == parent) == (parent.left == node):
                    self._rotate(parent)
                else:
                    self._rotate(node)
                self._rotate(node)
            else:
                # Zig
                self._rotate(node)
    
    def _access(self, node):
        """Make node the root of its preferred path."""
        last = None
        current = node
        
        while current:
            self._splay(current)
            
            # Change preferred path
            if current.right:
                current.right.path_parent = current
                current.right.parent = None
            
            current.right = last
            if last:
                last.parent = current
                last.path_parent = None
            
            self._update(current)
            last = current
            current = current.path_parent
        
        self._splay(node)
        return last
    
    def find_root(self, node):
        """Find the root of the tree containing node."""
        self._access(node)
        current = node
        
        while current.left:
            self._push(current)
            current = current.left
        
        self._splay(current)
        return current
    
    def link(self, u, v):
        """Make v a child of u."""
        self._access(v)
        v.path_parent = self.nodes[u]
    
    def cut(self, node):
        """Cut node from its parent."""
        self._access(node)
        if node.left:
            node.left.parent = None
            node.left.path_parent = node.path_parent
            node.left = None
            self._update(node)
    
    def path_query(self, u, v):
        """Query the path between u and v."""
        # Make u the root
        self._access(self.nodes[u])
        # Access v to get the path
        self._access(self.nodes[v])
        return self.nodes[v].size
```
**Why**: Link-cut trees support dynamic tree operations in O(log n) amortized time, critical for network flow algorithms and dynamic graph problems.

## 3. Null Object Pattern
```python
# PROBLEM: Eliminate null checks by providing default behavior
# TIME: O(1) SPACE: O(1)
from abc import ABC, abstractmethod

class Logger(ABC):
    """Abstract logger interface."""
    
    @abstractmethod
    def log(self, message):
        """Log a message."""
        pass

class ConsoleLogger(Logger):
    """Concrete logger that writes to console."""
    
    def log(self, message):
        print(f"[LOG] {message}")

class FileLogger(Logger):
    """Concrete logger that writes to file."""
    
    def __init__(self, filename):
        self.filename = filename
    
    def log(self, message):
        with open(self.filename, 'a') as f:
            f.write(f"[LOG] {message}\n")

class NullLogger(Logger):
    """Null object that provides no-op behavior."""
    
    def log(self, message):
        # Do nothing - this is the null object pattern
        pass

class LoggerFactory:
    """Factory to create appropriate logger based on configuration."""
    
    @staticmethod
    def create_logger(config):
        """Create logger based on config, returns NullLogger if disabled."""
        if not config.get('logging_enabled', False):
            return NullLogger()
        
        log_type = config.get('log_type', 'console')
        
        if log_type == 'console':
            return ConsoleLogger()
        elif log_type == 'file':
            return FileLogger(config.get('log_file', 'app.log'))
        else:
            return NullLogger()

class UserService:
    """Example service using null object pattern."""
    
    def __init__(self, logger=None):
        # Use null logger if none provided
        self.logger = logger if logger else NullLogger()
    
    def create_user(self, username):
        """Create a new user."""
        # No null check needed!
        self.logger.log(f"Creating user: {username}")
        
        # Business logic here
        print(f"User {username} created")
        
        self.logger.log(f"User {username} created successfully")

# Usage without null checks
def demonstrate_null_object():
    """Demonstrate null object pattern usage."""
    # With logging enabled
    config_enabled = {'logging_enabled': True, 'log_type': 'console'}
    logger = LoggerFactory.create_logger(config_enabled)
    service1 = UserService(logger)
    service1.create_user("Alice")
    
    # With logging disabled - no null checks needed!
    config_disabled = {'logging_enabled': False}
    logger = LoggerFactory.create_logger(config_disabled)
    service2 = UserService(logger)
    service2.create_user("Bob")  # Works without any null checks
```
**Why**: Eliminates null pointer exceptions and conditional logic, making code cleaner and more maintainable by providing default no-op behavior.

## 4. Special Case Pattern
```python
# PROBLEM: Handle exceptional cases with specialized objects instead of conditionals
# TIME: O(1) SPACE: O(1)
from abc import ABC, abstractmethod
from datetime import datetime

class Customer(ABC):
    """Abstract customer interface."""
    
    @abstractmethod
    def get_discount(self):
        """Return discount rate."""
        pass
    
    @abstractmethod
    def get_delivery_date(self, order_date):
        """Calculate delivery date."""
        pass
    
    @abstractmethod
    def is_eligible_for_promotion(self):
        """Check if customer is eligible for promotions."""
        pass

class RegularCustomer(Customer):
    """Regular customer with standard terms."""
    
    def get_discount(self):
        return 0.0  # No discount
    
    def get_delivery_date(self, order_date):
        # 5 business days delivery
        return datetime.fromisoformat(order_date) + timedelta(days=7)
    
    def is_eligible_for_promotion(self):
        return True

class VIPCustomer(Customer):
    """VIP customer with premium benefits."""
    
    def get_discount(self):
        return 0.15  # 15% discount
    
    def get_delivery_date(self, order_date):
        # Next day delivery
        return datetime.fromisoformat(order_date) + timedelta(days=1)
    
    def is_eligible_for_promotion(self):
        return True

class GuestCustomer(Customer):
    """Guest customer - special case with no account."""
    
    def get_discount(self):
        return 0.0  # No discount
    
    def get_delivery_date(self, order_date):
        # 7 business days delivery
        return datetime.fromisoformat(order_date) + timedelta(days=10)
    
    def is_eligible_for_promotion(self):
        return False  # Not eligible

class BlockedCustomer(Customer):
    """Blocked customer - special case for suspended accounts."""
    
    def get_discount(self):
        return 0.0
    
    def get_delivery_date(self, order_date):
        raise ValueError("Cannot process orders for blocked customers")
    
    def is_eligible_for_promotion(self):
        return False

class OrderProcessor:
    """Process orders without checking customer types."""
    
    def process_order(self, customer, amount, order_date):
        """Process an order without conditionals for customer types."""
        # Calculate discount - polymorphism handles all cases
        discount = customer.get_discount()
        final_amount = amount * (1 - discount)
        
        # Get delivery date - special cases handle themselves
        try:
            delivery_date = customer.get_delivery_date(order_date)
            print(f"Order processed: ${final_amount:.2f}, delivery by {delivery_date.date()}")
            return True
        except ValueError as e:
            print(f"Order failed: {e}")
            return False

# Usage without conditionals
def demonstrate_special_case():
    """Demonstrate special case pattern."""
    processor = OrderProcessor()
    
    # Different customer types handled polymorphically
    customers = [
        VIPCustomer(),
        RegularCustomer(),
        GuestCustomer(),
        BlockedCustomer()
    ]
    
    for customer in customers:
        processor.process_order(customer, 100.0, "2026-02-20")
```
**Why**: Replaces complex conditional logic with polymorphism, making the code more maintainable and extensible for new customer types.

## 5. Flyweight Factory Combination
```python
# PROBLEM: Share common parts of objects to reduce memory usage
# TIME: O(1) for retrieval SPACE: O(n) for shared objects
class TreeType:
    """Flyweight object containing shared tree attributes."""
    
    def __init__(self, name, color, texture):
        self.name = name
        self.color = color
        self.texture = texture
    
    def draw(self, canvas, x, y):
        """Draw tree using shared attributes."""
        print(f"Drawing {self.name} tree at ({x}, {y}) "
              f"with {self.color} color and {self.texture} texture")

class Tree:
    """Context object containing extrinsic state."""
    
    def __init__(self, x, y, tree_type):
        self.x = x
        self.y = y
        self.tree_type = tree_type  # Reference to flyweight
    
    def draw(self, canvas):
        """Delegate drawing to flyweight with position."""
        self.tree_type.draw(canvas, self.x, self.y)

class TreeFactory:
    """Factory that manages flyweight objects."""
    
    _tree_types = {}  # Cache of flyweight objects
    
    @classmethod
    def get_tree_type(cls, name, color, texture):
        """Get or create a flyweight tree type."""
        key = f"{name}_{color}_{texture}"
        
        if key not in cls._tree_types:
            print(f"Creating new flyweight: {key}")
            cls._tree_types[key] = TreeType(name, color, texture)
        
        return cls._tree_types[key]
    
    @classmethod
    def get_flyweight_count(cls):
        """Return number of flyweight objects."""
        return len(cls._tree_types)

class Forest:
    """Forest containing many trees sharing flyweights."""
    
    def __init__(self):
        self.trees = []
    
    def plant_tree(self, x, y, name, color, texture):
        """Plant a tree using flyweight pattern."""
        tree_type = TreeFactory.get_tree_type(name, color, texture)
        tree = Tree(x, y, tree_type)
        self.trees.append(tree)
    
    def draw(self, canvas):
        """Draw all trees."""
        for tree in self.trees:
            tree.draw(canvas)

def demonstrate_flyweight_factory():
    """Demonstrate flyweight pattern with factory."""
    forest = Forest()
    
    # Plant 1000 trees but only 3 flyweights created
    for i in range(1000):
        # Only 3 different tree types
        if i % 3 == 0:
            forest.plant_tree(i, i*2, "Oak", "Green", "Rough")
        elif i % 3 == 1:
            forest.plant_tree(i, i*2, "Pine", "Dark Green", "Smooth")
        else:
            forest.plant_tree(i, i*2, "Maple", "Red", "Medium")
    
    print(f"Trees planted: {len(forest.trees)}")
    print(f"Flyweights created: {TreeFactory.get_flyweight_count()}")
    # Memory saved: 1000 trees but only 3 flyweight objects!
```
**Why**: Dramatically reduces memory usage when many objects share common state, essential for games and graphics applications with numerous similar objects.

## 6. Rate Limiter - Sliding Window
```python
# PROBLEM: Limit request rate using sliding window algorithm
# TIME: O(log n) for cleanup SPACE: O(n) for timestamp storage
import time
from collections import deque

class SlidingWindowRateLimiter:
    """Rate limiter using sliding window algorithm."""
    
    def __init__(self, max_requests, window_seconds):
        """
        Initialize rate limiter.
        
        Args:
            max_requests: Maximum requests allowed in window
            window_seconds: Time window in seconds
        """
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self.requests = deque()  # Stores timestamps
    
    def _cleanup(self):
        """Remove old requests outside the window."""
        cutoff = time.time() - self.window_seconds
        
        while self.requests and self.requests[0] < cutoff:
            self.requests.popleft()
    
    def allow_request(self):
        """
        Check if request is allowed.
        
        Returns:
            bool: True if request is allowed, False otherwise
        """
        self._cleanup()
        
        if len(self.requests) < self.max_requests:
            self.requests.append(time.time())
            return True
        
        return False
    
    def get_current_count(self):
        """Get current number of requests in window."""
        self._cleanup()
        return len(self.requests)
    
    def get_remaining(self):
        """Get number of remaining requests allowed."""
        self._cleanup()
        return max(0, self.max_requests - len(self.requests))

class MultiUserRateLimiter:
    """Rate limiter for multiple users with sliding window."""
    
    def __init__(self, max_requests, window_seconds):
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self.user_limiters = {}  # user_id -> SlidingWindowRateLimiter
    
    def allow_request(self, user_id):
        """Check if user can make a request."""
        if user_id not in self.user_limiters:
            self.user_limiters[user_id] = SlidingWindowRateLimiter(
                self.max_requests, self.window_seconds
            )
        
        return self.user_limiters[user_id].allow_request()
    
    def get_user_stats(self, user_id):
        """Get rate limit stats for a user."""
        if user_id not in self.user_limiters:
            return {'current': 0, 'remaining': self.max_requests}
        
        limiter = self.user_limiters[user_id]
        return {
            'current': limiter.get_current_count(),
            'remaining': limiter.get_remaining()
        }

def demonstrate_sliding_window():
    """Demonstrate sliding window rate limiter."""
    limiter = SlidingWindowRateLimiter(max_requests=5, window_seconds=10)
    
    # Make requests
    for i in range(7):
        if limiter.allow_request():
            print(f"Request {i+1}: ALLOWED (count: {limiter.get_current_count()})")
        else:
            print(f"Request {i+1}: REJECTED (limit reached)")
    
    # Wait and try again
    time.sleep(11)
    print(f"After 11 seconds: {limiter.get_current_count()} requests in window")
```
**Why**: Provides smooth rate limiting without burstiness of fixed window, essential for API rate limiting and preventing abuse.

## 7. Circuit Breaker State Machine
```python
# PROBLEM: Prevent cascading failures with circuit breaker pattern
# TIME: O(1) for state transitions SPACE: O(1)
import time
from enum import Enum

class CircuitState(Enum):
    """Circuit breaker states."""
    CLOSED = "closed"      # Normal operation
    OPEN = "open"          # Failing, reject requests
    HALF_OPEN = "half_open"  # Testing if service recovered

class CircuitBreaker:
    """Circuit breaker with state machine pattern."""
    
    def __init__(self, failure_threshold=5, timeout=60, success_threshold=3):
        """
        Initialize circuit breaker.
        
        Args:
            failure_threshold: Failures before opening circuit
            timeout: Seconds to wait before trying half-open
            success_threshold: Successes needed to close circuit
        """
        self.failure_threshold = failure_threshold
        self.timeout = timeout
        self.success_threshold = success_threshold
        
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.success_count = 0
        self.last_failure_time = None
    
    def call(self, func, *args, **kwargs):
        """
        Execute function through circuit breaker.
        
        Args:
            func: Function to execute
            *args, **kwargs: Arguments for function
        
        Returns:
            Result of function or raises exception
        """
        if self.state == CircuitState.OPEN:
            if self._should_attempt_reset():
                self._transition_to_half_open()
            else:
                raise Exception("Circuit breaker is OPEN")
        
        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise e
    
    def _should_attempt_reset(self):
        """Check if enough time has passed to try reset."""
        if self.last_failure_time is None:
            return False
        return time.time() - self.last_failure_time >= self.timeout
    
    def _on_success(self):
        """Handle successful call."""
        self.failure_count = 0
        
        if self.state == CircuitState.HALF_OPEN:
            self.success_count += 1
            if self.success_count >= self.success_threshold:
                self._transition_to_closed()
    
    def _on_failure(self):
        """Handle failed call."""
        self.failure_count += 1
        self.success_count = 0
        self.last_failure_time = time.time()
        
        if self.state == CircuitState.HALF_OPEN:
            self._transition_to_open()
        elif self.failure_count >= self.failure_threshold:
            self._transition_to_open()
    
    def _transition_to_open(self):
        """Transition to OPEN state."""
        self.state = CircuitState.OPEN
        print("Circuit breaker: CLOSED -> OPEN")
    
    def _transition_to_half_open(self):
        """Transition to HALF_OPEN state."""
        self.state = CircuitState.HALF_OPEN
        self.success_count = 0
        print("Circuit breaker: OPEN -> HALF_OPEN")
    
    def _transition_to_closed(self):
        """Transition to CLOSED state."""
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.success_count = 0
        print("Circuit breaker: HALF_OPEN -> CLOSED")
    
    def get_state(self):
        """Get current state."""
        return self.state.value

def unreliable_service():
    """Simulated unreliable service."""
    import random
    if random.random() < 0.7:  # 70% failure rate
        raise Exception("Service unavailable")
    return "Success"

def demonstrate_circuit_breaker():
    """Demonstrate circuit breaker pattern."""
    cb = CircuitBreaker(failure_threshold=3, timeout=5, success_threshold=2)
    
    for i in range(20):
        try:
            result = cb.call(unreliable_service)
            print(f"Call {i+1}: {result} [State: {cb.get_state()}]")
        except Exception as e:
            print(f"Call {i+1}: {e} [State: {cb.get_state()}]")
        
        time.sleep(0.5)
```
**Why**: Prevents cascading failures in distributed systems by failing fast when a service is unhealthy, allowing it time to recover.

## 8. Bulkhead Isolation
```python
# PROBLEM: Isolate failures to prevent total system failure
# TIME: O(1) for isolation SPACE: O(n) for thread pools
import time
from concurrent.futures import ThreadPoolExecutor, TimeoutError as FuturesTimeoutError
from threading import Semaphore

class Bulkhead:
    """Bulkhead pattern to isolate components."""
    
    def __init__(self, max_concurrent_calls, name="default"):
        """
        Initialize bulkhead.
        
        Args:
            max_concurrent_calls: Maximum concurrent calls allowed
            name: Name of the bulkhead for logging
        """
        self.max_concurrent_calls = max_concurrent_calls
        self.name = name
        self.semaphore = Semaphore(max_concurrent_calls)
        self.current_calls = 0
    
    def execute(self, func, *args, **kwargs):
        """
        Execute function with bulkhead protection.
        
        Args:
            func: Function to execute
            *args, **kwargs: Function arguments
        
        Returns:
            Result of function
        
        Raises:
            Exception: If bulkhead is full
        """
        if not self.semaphore.acquire(blocking=False):
            raise Exception(
                f"Bulkhead '{self.name}' is full "
                f"({self.max_concurrent_calls} concurrent calls)"
            )
        
        try:
            self.current_calls += 1
            result = func(*args, **kwargs)
            return result
        finally:
            self.current_calls -= 1
            self.semaphore.release()
    
    def get_stats(self):
        """Get bulkhead statistics."""
        return {
            'name': self.name,
            'max_concurrent': self.max_concurrent_calls,
            'current_calls': self.current_calls,
            'available': self.max_concurrent_calls - self.current_calls
        }

class BulkheadWithTimeout:
    """Bulkhead with timeout support."""
    
    def __init__(self, max_concurrent_calls, timeout_seconds, name="default"):
        self.max_concurrent_calls = max_concurrent_calls
        self.timeout_seconds = timeout_seconds
        self.name = name
        self.executor = ThreadPoolExecutor(max_workers=max_concurrent_calls)
        self.semaphore = Semaphore(max_concurrent_calls)
    
    def execute(self, func, *args, **kwargs):
        """Execute with bulkhead and timeout."""
        if not self.semaphore.acquire(blocking=False):
            raise Exception(f"Bulkhead '{self.name}' is full")
        
        try:
            future = self.executor.submit(func, *args, **kwargs)
            return future.result(timeout=self.timeout_seconds)
        except FuturesTimeoutError:
            raise Exception(f"Operation timed out after {self.timeout_seconds}s")
        finally:
            self.semaphore.release()
    
    def shutdown(self):
        """Shutdown the executor."""
        self.executor.shutdown(wait=False)

class IsolatedSystem:
    """System with multiple isolated bulkheads."""
    
    def __init__(self):
        # Different bulkheads for different services
        self.database_bulkhead = Bulkhead(max_concurrent_calls=10, name="database")
        self.api_bulkhead = Bulkhead(max_concurrent_calls=20, name="api")
        self.cache_bulkhead = Bulkhead(max_concurrent_calls=50, name="cache")
    
    def call_database(self, query):
        """Execute database query with bulkhead."""
        return self.database_bulkhead.execute(
            self._execute_query, query
        )
    
    def call_api(self, endpoint):
        """Call external API with bulkhead."""
        return self.api_bulkhead.execute(
            self._call_endpoint, endpoint
        )
    
    def get_cache(self, key):
        """Get from cache with bulkhead."""
        return self.cache_bulkhead.execute(
            self._cache_get, key
        )
    
    def _execute_query(self, query):
        """Simulate database query."""
        time.sleep(0.1)
        return f"Result for: {query}"
    
    def _call_endpoint(self, endpoint):
        """Simulate API call."""
        time.sleep(0.2)
        return f"Response from: {endpoint}"
    
    def _cache_get(self, key):
        """Simulate cache get."""
        time.sleep(0.01)
        return f"Cache value: {key}"
    
    def get_all_stats(self):
        """Get statistics for all bulkheads."""
        return {
            'database': self.database_bulkhead.get_stats(),
            'api': self.api_bulkhead.get_stats(),
            'cache': self.cache_bulkhead.get_stats()
        }

def demonstrate_bulkhead():
```

## 9. Query Optimization with Indexing
```python
# PROBLEM: Optimize database queries using proper indexing strategies
# TIME: O(log n) for indexed search SPACE: O(n) for indexes
from collections import defaultdict, OrderedDict
import bisect

class InMemoryDatabase:
    """In-memory database with indexing optimization."""
    
    def __init__(self):
        self.tables = {}  # table_name -> list of rows
        self.indexes = {}  # table_name -> {column_name -> sorted_values}
        self.primary_keys = {}  # table_name -> primary_key_column
    
    def create_table(self, table_name, primary_key=None):
        """Create a new table."""
        self.tables[table_name] = []
        self.indexes[table_name] = {}
        if primary_key:
            self.primary_keys[table_name] = primary_key
    
    def create_index(self, table_name, column_name):
        """Create an index on a column."""
        if table_name not in self.indexes:
            raise ValueError(f"Table {table_name} does not exist")
        
        # Build sorted index
        index = []
        for i, row in enumerate(self.tables[table_name]):
            value = row.get(column_name)
            if value is not None:
                bisect.insort(index, (value, i))
        
        self.indexes[table_name][column_name] = index
    
    def insert(self, table_name, row):
        """Insert a row into a table."""
        if table_name not in self.tables:
            raise ValueError(f"Table {table_name} does not exist")
        
        # Check primary key constraint
        if table_name in self.primary_keys:
            pk_column = self.primary_keys[table_name]
            pk_value = row.get(pk_column)
            
            for existing_row in self.tables[table_name]:
                if existing_row.get(pk_column) == pk_value:
                    raise ValueError(f"Duplicate primary key: {pk_value}")
        
        row_index = len(self.tables[table_name])
        self.tables[table_name].append(row)
        
        # Update indexes
        for column_name in self.indexes[table_name]:
            value = row.get(column_name)
            if value is not None:
                bisect.insort(
                    self.indexes[table_name][column_name],
                    (value, row_index)
                )
    
    def query_with_index(self, table_name, column_name, value):
        """Query using index (O(log n))."""
        if table_name not in self.indexes:
            raise ValueError(f"Table {table_name} does not exist")
        
        if column_name not in self.indexes[table_name]:
            # Fall back to full table scan
            return self.query_full_scan(table_name, column_name, value)
        
        # Binary search on index
        index = self.indexes[table_name][column_name]
        left = bisect.bisect_left(index, (value, 0))
        right = bisect.bisect_right(index, (value, float('inf')))
        
        results = []
        for i in range(left, right):
            if i < len(index) and index[i][0] == value:
                row_index = index[i][1]
                results.append(self.tables[table_name][row_index])
        
        return results
    
    def query_full_scan(self, table_name, column_name, value):
        """Full table scan (O(n))."""
        if table_name not in self.tables:
            raise ValueError(f"Table {table_name} does not exist")
        
        results = []
        for row in self.tables[table_name]:
            if row.get(column_name) == value:
                results.append(row)
        
        return results
    
    def range_query(self, table_name, column_name, min_value, max_value):
        """Range query using index."""
        if column_name not in self.indexes.get(table_name, {}):
            # Fall back to full scan
            results = []
            for row in self.tables[table_name]:
                val = row.get(column_name)
                if val is not None and min_value <= val <= max_value:
                    results.append(row)
            return results
        
        # Use index for range query
        index = self.indexes[table_name][column_name]
        left = bisect.bisect_left(index, (min_value, 0))
        right = bisect.bisect_right(index, (max_value, float('inf')))
        
        results = []
        for i in range(left, right):
            row_index = index[i][1]
            results.append(self.tables[table_name][row_index])
        
        return results
    
    def get_query_plan(self, table_name, column_name):
        """Get execution plan for a query."""
        uses_index = column_name in self.indexes.get(table_name, {})
        
        return {
            'table': table_name,
            'column': column_name,
            'uses_index': uses_index,
            'index_type': 'btree' if uses_index else None,
            'estimated_cost': 'O(log n)' if uses_index else 'O(n)',
            'rows_scanned': len(self.indexes[table_name].get(column_name, []))
                if uses_index else len(self.tables[table_name])
        }

def demonstrate_indexing():
    """Demonstrate query optimization with indexing."""
    db = InMemoryDatabase()
    
    # Create table
    db.create_table('users', primary_key='id')
    
    # Insert data
    for i in range(10000):
        db.insert('users', {
            'id': i,
            'name': f'User{i}',
            'age': 20 + (i % 50),
            'email': f'user{i}@example.com'
        })
    
    # Create index on age
    db.create_index('users', 'age')
    
    # Query with index
    import time
    start = time.time()
    results = db.query_with_index('users', 'age', 25)
    indexed_time = time.time() - start
    
    # Query without index
    start = time.time()
    results = db.query_full_scan('users', 'name', 'User25')
    scan_time = time.time() - start
    
    print(f"Indexed query time: {indexed_time:.4f}s")
    print(f"Full scan time: {scan_time:.4f}s")
    print(f"Speedup: {scan_time/indexed_time:.2f}x")
    
    # Show query plan
    plan = db.get_query_plan('users', 'age')
    print(f"\nQuery plan: {plan}")
```
**Why**: Proper indexing reduces query time from O(n) to O(log n), critical for database performance and scalability.

## 10. Transaction with Optimistic Concurrency Control
```python
# PROBLEM: Handle concurrent transactions without locking
# TIME: O(n) for validation SPACE: O(n) for version storage
import time
from threading import Lock
from copy import deepcopy

class VersionedValue:
    """Value with version for optimistic locking."""
    
    def __init__(self, value, version=0):
        self.value = value
        self.version = version

class OCCDatabase:
    """Database with optimistic concurrency control."""
    
    def __init__(self):
        self.data = {}  # key -> VersionedValue
        self.lock = Lock()
        self.transactions = {}  # transaction_id -> transaction_data
    
    def begin_transaction(self, transaction_id):
        """Begin a new transaction."""
        with self.lock:
            self.transactions[transaction_id] = {
                'read_set': {},   # key -> (value, version) at read time
                'write_set': {},  # key -> new_value
                'start_time': time.time()
            }
    
    def read(self, transaction_id, key):
        """Read a value in transaction context."""
        with self.lock:
            if transaction_id not in self.transactions:
                raise ValueError(f"Transaction {transaction_id} not found")
            
            # Check write set first (read-your-own-writes)
            if key in self.transactions[transaction_id]['write_set']:
                return self.transactions[transaction_id]['write_set'][key]
            
            # Read from database
            if key not in self.data:
                return None
            
            versioned = self.data[key]
            
            # Record in read set
            self.transactions[transaction_id]['read_set'][key] = (
                versioned.value, versioned.version
            )
            
            return versioned.value
    
    def write(self, transaction_id, key, value):
        """Write a value in transaction context."""
        with self.lock:
            if transaction_id not in self.transactions:
                raise ValueError(f"Transaction {transaction_id} not found")
            
            # Add to write set
            self.transactions[transaction_id]['write_set'][key] = value
    
    def commit(self, transaction_id):
        """Commit transaction with optimistic validation."""
        with self.lock:
            if transaction_id not in self.transactions:
                raise ValueError(f"Transaction {transaction_id} not found")
            
            transaction = self.transactions[transaction_id]
            
            # Validation phase: check if read set is still valid
            for key, (value, version) in transaction['read_set'].items():
                if key in self.data:
                    current_version = self.data[key].version
                    if current_version != version:
                        # Validation failed - abort transaction
                        del self.transactions[transaction_id]
                        raise Exception(
                            f"Transaction {transaction_id} aborted: "
                            f"concurrent modification detected for key '{key}'"
                        )
            
            # Write phase: apply all writes
            for key, value in transaction['write_set'].items():
                if key not in self.data:
                    self.data[key] = VersionedValue(value, version=1)
                else:
                    self.data[key] = VersionedValue(
                        value,
                        self.data[key].version + 1
                    )
            
            # Clean up transaction
            del self.transactions[transaction_id]
            return True
    
    def rollback(self, transaction_id):
        """Rollback a transaction."""
        with self.lock:
            if transaction_id in self.transactions:
                del self.transactions[transaction_id]
    
    def get_version(self, key):
        """Get current version of a key."""
        with self.lock:
            if key in self.data:
                return self.data[key].version
            return None

class BankAccountService:
    """Bank service using OCC for transfers."""
    
    def __init__(self):
        self.db = OCCDatabase()
        self.transaction_counter = 0
    
    def create_account(self, account_id, initial_balance):
        """Create a new account."""
        tx_id = self._get_next_transaction_id()
        self.db.begin_transaction(tx_id)
        
        try:
            self.db.write(tx_id, f'account:{account_id}', initial_balance)
            self.db.commit(tx_id)
            return True
        except:
            self.db.rollback(tx_id)
            raise
    
    def transfer(self, from_account, to_account, amount):
        """Transfer money with optimistic concurrency control."""
        tx_id = self._get_next_transaction_id()
        self.db.begin_transaction(tx_id)
        
        try:
            # Read both accounts
            from_balance = self.db.read(tx_id, f'account:{from_account}')
            to_balance = self.db.read(tx_id, f'account:{to_account}')
            
            if from_balance is None or to_balance is None:
                raise ValueError("Account not found")
            
            if from_balance < amount:
                raise ValueError("Insufficient funds")
            
            # Write new balances
            self.db.write(
                tx_id,
                f'account:{from_account}',
                from_balance - amount
            )
            self.db.write(
                tx_id,
                f'account:{to_account}',
                to_balance + amount
            )
            
            # Commit (will validate versions)
            self.db.commit(tx_id)
            return True
        except Exception as e:
            self.db.rollback(tx_id)
            raise e
    
    def _get_next_transaction_id(self):
        """Get next transaction ID."""
        self.transaction_counter += 1
        return f'tx_{self.transaction_counter}'
    
    def get_balance(self, account_id):
        """Get account balance."""
        tx_id = self._get_next_transaction_id()
        self.db.begin_transaction(tx_id)
        
        try:
            balance = self.db.read(tx_id, f'account:{account_id}')
            self.db.commit(tx_id)
            return balance
        except:
            self.db.rollback(tx_id)
            raise
```
