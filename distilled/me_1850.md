## Boyer-Moore String Search
```python
# PROBLEM: Find pattern in text using bad character and good suffix heuristics
# TIME: O(n/m) average, O(nm) worst SPACE: O(k) for alphabet size
def boyer_moore_search(text, pattern):
    """
    Boyer-Moore string search algorithm.
    Scans pattern right-to-left, skips ahead using bad character rule.
    """
    if not pattern or not text:
        return []
    
    # Build bad character table: rightmost position of each char in pattern
    def build_bad_char_table(pat):
        table = {}
        for i, char in enumerate(pat):
            table[char] = i
        return table
    
    bad_char = build_bad_char_table(pattern)
    n, m = len(text), len(pattern)
    matches = []
    shift = 0
    
    while shift <= n - m:
        # Compare pattern right-to-left
        j = m - 1
        while j >= 0 and pattern[j] == text[shift + j]:
            j -= 1
        
        if j < 0:
            # Found match
            matches.append(shift)
            # Move to next potential match
            shift += m - bad_char.get(text[shift + m], -1) if shift + m < n else 1
        else:
            # Bad character mismatch: align pattern with text's bad char
            bc_shift = j - bad_char.get(text[shift + j], -1)
            shift += max(1, bc_shift)
    
    return matches

# Example usage
if __name__ == "__main__":
    text = "ABAAABCDBABCDABCDBABCD"
    pattern = "ABC"
    print(boyer_moore_search(text, pattern))  # [4, 8, 12]
```
**Why**: Boyer-Moore achieves sublinear time on average by skipping characters. The bad character heuristic allows large jumps, making it practical for search in large texts.

## KMP String Matching
```python
# PROBLEM: Find all pattern occurrences in text using failure function
# TIME: O(n + m) SPACE: O(m)
def kmp_search(text, pattern):
    """
    Knuth-Morris-Pratt string matching algorithm.
    Uses failure function (LPS array) to avoid re-checking characters.
    """
    if not pattern or not text:
        return []
    
    # Build Longest Prefix Suffix (LPS) array
    def build_lps(pat):
        m = len(pat)
        lps = [0] * m
        length = 0  # Length of previous longest prefix suffix
        i = 1
        
        while i < m:
            if pat[i] == pat[length]:
                length += 1
                lps[i] = length
                i += 1
            else:
                if length != 0:
                    # Fall back to previous longest prefix
                    length = lps[length - 1]
                else:
                    lps[i] = 0
                    i += 1
        return lps
    
    lps = build_lps(pattern)
    n, m = len(text), len(pattern)
    matches = []
    i = j = 0  # i for text, j for pattern
    
    while i < n:
        if text[i] == pattern[j]:
            i += 1
            j += 1
            
            if j == m:
                # Found complete match
                matches.append(i - j)
                j = lps[j - 1]  # Continue searching
        else:
            if j != 0:
                # Use failure function to skip
                j = lps[j - 1]
            else:
                i += 1
    
    return matches

# Example usage
if __name__ == "__main__":
    text = "ABABDABACDABABCABAB"
    pattern = "ABABCABAB"
    print(kmp_search(text, pattern))  # [10]
```
**Why**: KMP guarantees linear time by never backtracking on the text. The failure function precomputes where to resume matching after a mismatch, critical for streaming or unidirectional input.

## Rabin-Karp Rolling Hash
```python
# PROBLEM: Find pattern in text using hash comparison with rolling hash
# TIME: O(n + m) average, O(nm) worst SPACE: O(1)
def rabin_karp_search(text, pattern, base=256, prime=101):
    """
    Rabin-Karp string search using rolling hash.
    Compares hash values, verifies only when hashes match.
    """
    if not pattern or not text or len(pattern) > len(text):
        return []
    
    n, m = len(text), len(pattern)
    pattern_hash = 0  # Hash value for pattern
    text_hash = 0     # Hash value for current text window
    h = 1             # Base^(m-1) % prime for rolling hash
    
    matches = []
    
    # Calculate h = base^(m-1) % prime
    for _ in range(m - 1):
        h = (h * base) % prime
    
    # Calculate initial hash values
    for i in range(m):
        pattern_hash = (base * pattern_hash + ord(pattern[i])) % prime
        text_hash = (base * text_hash + ord(text[i])) % prime
    
    # Slide pattern over text
    for i in range(n - m + 1):
        # Check if hash values match
        if pattern_hash == text_hash:
            # Verify character by character
            if text[i:i + m] == pattern:
                matches.append(i)
        
        # Calculate hash for next window (rolling)
        if i < n - m:
            text_hash = (base * (text_hash - ord(text[i]) * h) + ord(text[i + m])) % prime
            # Handle negative hash
            if text_hash < 0:
                text_hash += prime
    
    return matches

# Example usage
if __name__ == "__main__":
    text = "GEEKS FOR GEEKS"
    pattern = "GEEK"
    print(rabin_karp_search(text, pattern))  # [0, 10]
```
**Why**: Rabin-Karp uses hash comparison for O(1) window shifts. Ideal for multiple pattern search (same hash for all patterns) and plagiarism detection where hash-based filtering is efficient.

## Z-Algorithm
```python
# PROBLEM: Compute Z-array for pattern matching and string analysis
# TIME: O(n) SPACE: O(n)
def z_algorithm(s):
    """
    Z-algorithm: computes Z-array where Z[i] is the length of the longest
    substring starting from i that matches prefix of s.
    """
    n = len(s)
    Z = [0] * n
    Z[0] = n  # First element is the string itself
    
    left = right = 0
    
    for i in range(1, n):
        # Case 1: i is outside current Z-box
        if i > right:
            left = right = i
            while right < n and s[right - left] == s[right]:
                right += 1
            Z[i] = right - left
            right -= 1
        # Case 2: i is inside current Z-box
        else:
            k = i - left
            # If Z[k] doesn't reach box boundary, copy it
            if Z[k] < right - i + 1:
                Z[i] = Z[k]
            else:
                # Otherwise, extend from right boundary
                left = i
                while right < n and s[right - left] == s[right]:
                    right += 1
                Z[i] = right - left
                right -= 1
    
    return Z

def z_search(text, pattern):
    """Find all occurrences of pattern in text using Z-algorithm."""
    if not pattern or not text:
        return []
    
    # Concatenate: pattern + '$' + text
    combined = pattern + '$' + text
    Z = z_algorithm(combined)
    
    m = len(pattern)
    matches = []
    for i in range(m + 1, len(Z)):
        if Z[i] == m:
            matches.append(i - m - 1)
    
    return matches

# Example usage
if __name__ == "__main__":
    text = "ABABDABACDABABCABAB"
    pattern = "ABAB"
    print(z_search(text, pattern))  # [0, 10, 15]
    print(z_algorithm("aabcaabxaaz"))  # [11, 1, 0, 0, 3, 1, 0, 0, 2, 1, 0]
```
**Why**: Z-algorithm processes string in linear time with guaranteed O(n) complexity. Useful for pattern matching, finding longest palindromic prefix, and string periodicity problems.

## Two Pointer Technique
```python
# PROBLEM: Solve array problems using two pointers moving in same or opposite directions
# TIME: O(n) SPACE: O(1)
def two_sum_sorted(arr, target):
    """
    Find two numbers in sorted array that sum to target.
    Uses two pointers from opposite ends.
    """
    left, right = 0, len(arr) - 1
    
    while left < right:
        current_sum = arr[left] + arr[right]
        if current_sum == target:
            return (left, right)
        elif current_sum < target:
            left += 1  # Need larger sum
        else:
            right -= 1  # Need smaller sum
    
    return None

def remove_duplicates_sorted(arr):
    """
    Remove duplicates from sorted array in-place.
    Returns new length.
    """
    if not arr:
        return 0
    
    write_idx = 1  # Position to write next unique element
    for read_idx in range(1, len(arr)):
        if arr[read_idx] != arr[read_idx - 1]:
            arr[write_idx] = arr[read_idx]
            write_idx += 1
    
    return write_idx

def three_sum(arr, target):
    """
    Find all triplets summing to target in sorted array.
    """
    arr.sort()
    n = len(arr)
    triplets = []
    
    for i in range(n - 2):
        if i > 0 and arr[i] == arr[i - 1]:
            continue  # Skip duplicates
        
        left, right = i + 1, n - 1
        while left < right:
            current = arr[i] + arr[left] + arr[right]
            if current == target:
                triplets.append([arr[i], arr[left], arr[right]])
                # Skip duplicates
                while left < right and arr[left] == arr[left + 1]:
                    left += 1
                while left < right and arr[right] == arr[right - 1]:
                    right -= 1
                left += 1
                right -= 1
            elif current < target:
                left += 1
            else:
                right -= 1
    
    return triplets

# Example usage
if __name__ == "__main__":
    arr = [1, 2, 3, 4, 5, 6, 7]
    print(two_sum_sorted(arr, 9))  # (2, 5) -> arr[2]=3, arr[5]=6
    
    arr2 = [1, 1, 2, 2, 3, 4, 4]
    new_len = remove_duplicates_sorted(arr2)
    print(arr2[:new_len])  # [1, 2, 3, 4]
    
    arr3 = [-1, 0, 1, 2, -1, -4]
    print(three_sum(arr3, 0))  # [[-1, -1, 2], [-1, 0, 1]]
```
**Why**: Two pointers reduce O(n²) to O(n) for sorted arrays. Eliminates nested loops by smart pointer movement. Essential for array/string problems with sorted input or palindromic checks.

## Sliding Window Maximum
```python
# PROBLEM: Find maximum in each sliding window of size k
# TIME: O(n) SPACE: O(k)
from collections import deque

def sliding_window_maximum(arr, k):
    """
    Find maximum in each sliding window using deque.
    Deque stores indices of useful elements in decreasing order.
    """
    if not arr or k <= 0:
        return []
    
    n = len(arr)
    if k >= n:
        return [max(arr)]
    
    result = []
    dq = deque()  # Stores indices
    
    for i in range(n):
        # Remove indices outside current window
        while dq and dq[0] < i - k + 1:
            dq.popleft()
        
        # Remove indices of smaller elements (they won't be max)
        while dq and arr[dq[-1]] < arr[i]:
            dq.pop()
        
        dq.append(i)
        
        # Add maximum to result once window is fully formed
        if i >= k - 1:
            result.append(arr[dq[0]])
    
    return result

def sliding_window_minimum(arr, k):
    """Find minimum in each sliding window."""
    if not arr or k <= 0:
        return []
    
    n = len(arr)
    result = []
    dq = deque()  # Stores indices in increasing order
    
    for i in range(n):
        # Remove indices outside window
        while dq and dq[0] < i - k + 1:
            dq.popleft()
        
        # Remove indices of larger elements
        while dq and arr[dq[-1]] > arr[i]:
            dq.pop()
        
        dq.append(i)
        
        if i >= k - 1:
            result.append(arr[dq[0]])
    
    return result

# Example usage
if __name__ == "__main__":
    arr = [1, 3, -1, -3, 5, 3, 6, 7]
    k = 3
    print(sliding_window_maximum(arr, k))  # [3, 3, 5, 5, 6, 7]
    print(sliding_window_minimum(arr, k))  # [-1, -3, -3, -3, 3, 3]
```
**Why**: Deque maintains candidates in sorted order, achieving O(n) instead of O(nk). Front always contains current window's max/min. Used in streaming analytics and real-time monitoring.

## Minimum Window Substring
```python
# PROBLEM: Find smallest window in string s containing all characters of string t
# TIME: O(n + m) SPACE: O(k) for character set
from collections import Counter

def min_window_substring(s, t):
    """
    Find minimum window substring containing all characters of t.
    Uses sliding window with character frequency tracking.
    """
    if not s or not t or len(s) < len(t):
        return ""
    
    # Count characters in t
    target_counts = Counter(t)
    required = len(target_counts)  # Unique characters needed
    
    # Window state
    window_counts = {}
    formed = 0  # How many unique chars satisfy requirement
    
    left = right = 0
    min_len = float('inf')
    min_window = (0, 0)
    
    while right < len(s):
        # Expand window by adding character at right
        char = s[right]
        window_counts[char] = window_counts.get(char, 0) + 1
        
        # Check if current char's count matches target
        if char in target_counts and window_counts[char] == target_counts[char]:
            formed += 1
        
        # Try to contract window from left
        while left <= right and formed == required:
            char = s[left]
            
            # Update minimum window
            if right - left + 1 < min_len:
                min_len = right - left + 1
                min_window = (left, right + 1)
            
            # Remove leftmost character
            window_counts[char] -= 1
            if char in target_counts and window_counts[char] < target_counts[char]:
                formed -= 1
            
            left += 1
        
        right += 1
    
    return s[min_window[0]:min_window[1]] if min_len != float('inf') else ""

# Example usage
if __name__ == "__main__":
    s = "ADOBECODEBANC"
    t = "ABC"
    print(min_window_substring(s, t))  # "BANC"
    
    s2 = "a"
    t2 = "a"
    print(min_window_substring(s2, t2))  # "a"
    
    s3 = "a"
    t3 = "aa"
    print(min_window_substring(s3, t3))  # ""
```
**Why**: Sliding window optimally balances expansion and contraction. Tracks when all required characters are satisfied. Essential for substring pattern matching and DNA sequence analysis.

## Interval Tree
```python
# PROBLEM: Store intervals and query for overlaps efficiently
# TIME: O(n log n) construction, O(log n + k) query SPACE: O(n)
class IntervalNode:
    def __init__(self, interval):
        self.interval = interval  # (start, end)
        self.max_end = interval[1]  # Maximum end in subtree
        self.left = None
        self.right = None

class IntervalTree:
    """
    Augmented BST for interval storage and overlap queries.
    Each node stores maximum end point in its subtree.
    """
    
    def __init__(self):
        self.root = None
    
    def insert(self, interval):
        """Insert interval into tree."""
        if not self.root:
            self.root = IntervalNode(interval)
        else:
            self._insert(self.root, interval)
    
    def _insert(self, node, interval):
        # Compare by start time
        if interval[0] < node.interval[0]:
            if node.left:
                self._insert(node.left, interval)
            else:
                node.left = IntervalNode(interval)
        else:
            if node.right:
                self._insert(node.right, interval)
            else:
                node.right = IntervalNode(interval)
        
        # Update max_end
        node.max_end = max(node.max_end, interval[1])
    
    def query_overlaps(self, interval):
        """Find all intervals overlapping with given interval."""
        results = []
        self._query(self.root, interval, results)
        return results
    
    def _query(self, node, interval, results):
        if not node:
            return
        
        # Check if intervals overlap
        node_interval = node.interval
        if self._overlap(node_interval, interval):
            results.append(node_interval)
        
        # If left child exists and its max_end >= interval start, search left
        if node.left and node.left.max_end >= interval[0]:
            self._query(node.left, interval, results)
        
        # Search right if node's start <= interval's end
        if node.right and node_interval[0] <= interval[1]:
            self._query(node.right, interval, results)
    
    def _overlap(self, int1, int2):
        """Check if two intervals overlap."""
        return int1[0] < int2[1] and int2[0] < int1[1]

# Example usage
if __name__ == "__main__":
    tree = IntervalTree()
    intervals = [(15, 20), (10, 30), (17, 19), (5, 20), (12, 15), (30, 40)]
    
    for interval in intervals:
        tree.insert(interval)
    
    # Query for overlaps with (14, 16)
    overlaps = tree.query_overlaps((14, 16))
    print(overlaps)  # [(15, 20), (10, 30), (5, 20), (12, 15)]
```
**Why**: Interval trees enable efficient overlap queries in O(log n + k) time. Augmentation with max_end prunes search space. Critical for scheduling systems, computational geometry, and genomic analysis.

## Segment Tree with Lazy Propagation
```python
# PROBLEM: Support range updates and range queries efficiently
# TIME: O(n) construction, O(log n) query/update SPACE: O(n)
class LazySegmentTree:
    """
    Segment tree with lazy propagation for range updates.
    Supports range sum queries and range increment updates.
    """
    
    def __init__(self, arr):
        self.n = len(arr)
        self.tree = [0] * (4 * self.n)  # Segment tree
        self.lazy = [0] * (4 * self.n)  # Lazy propagation
        self._build(arr, 0, 0, self.n - 1)
    
    def _build(self, arr, node, start, end):
        """Build segment tree from array."""
        if start == end:
            self.tree[node] = arr[start]
        else:
            mid = (start + end) // 2
            self._build(arr, 2 * node + 1, start, mid)
            self._build(arr, 2 * node + 2, mid + 1, end)
            self.tree[node] = self.tree[2 * node + 1] + self.tree[2 * node + 2]
    
    def _propagate(self, node, start, end):
        """Propagate lazy value to children."""
        if self.lazy[node] != 0:
            self.tree[node] += (end - start + 1) * self.lazy[node]
            if start != end:
                self.lazy[2 * node + 1] += self.lazy[node]
                self.lazy[2 * node + 2] += self.lazy[node]
            self.lazy[node] = 0
    
    def update_range(self, l, r, val):
        """Add val to all elements in range [l, r]."""
        self._update_range(0, 0, self.n - 1, l, r, val)
    
    def _update_range(self, node, start, end, l, r, val):
        self._propagate(node, start, end)
        
        if start > r or end < l:
            return  # Out of range
        
        if l <= start and end <= r:
            self.lazy[node] += val
            self._propagate(node, start, end)
            return
        
        mid = (start + end) // 2
        self._update_range(2 * node + 1, start, mid, l, r, val)
        self._update_range(2 * node + 2, mid + 1, end, l, r, val)
        self.tree[node] = self.tree[2 * node + 1] + self.tree[2 * node + 2]
    
    def query_range(self, l, r):
        """Query sum of range [l, r]."""
        return self._query_range(0, 0, self.n - 1, l, r)
    
    def _query_range(self, node, start, end, l, r):
        self._propagate(node, start, end)
        
        if start > r or end < l:
            return 0  # Out of range
        
        if l <= start and end <= r:
            return self.tree[node]
        
        mid = (start + end) // 2
        left_sum = self._query_range(2 * node + 1, start, mid, l, r)
        right_sum = self._query_range(2 * node + 2, mid + 1, end, l, r)
        return left_sum + right_sum

# Example usage
if __name__ == "__main__":
    arr = [1, 3, 5, 7, 9, 11]
    st = LazySegmentTree(arr)
    
    print(st.query_range(1, 3))  # 15 (3+5+7)
    
    st.update_range(1, 3, 10)  # Add 10 to elements [1, 3]
    print(st.query_range(1, 3))  # 45 (13+15+17)
    
    print(st.query_range(0, 5))  # 66 (1+13+15+17+9+11)
```
**Why**: Lazy propagation defers updates until necessary, achieving O(log n) for range operations. Essential for problems with frequent range updates (game engines, interval statistics, image processing).

## Rate Limiter - Token Bucket
```python
# PROBLEM: Implement rate limiting with token bucket algorithm
# TIME: O(1) for allow request SPACE: O(1)
import time

class TokenBucket:
    """
    Token bucket rate limiter.
    Tokens are added at fixed rate, each request consumes one token.
    """
    
    def __init__(self, capacity, refill_rate):
        """
        capacity: Maximum tokens in bucket
        refill_rate: Tokens added per second
        """
        self.capacity = capacity
        self.refill_rate = refill_rate
        self.tokens = capacity
        self.last_refill = time.time()
    
    def allow_request(self, tokens=1):
        """
        Check if request with given token cost is allowed.
        Returns True if allowed, False otherwise.
        """
        self._refill()
        
        if self.tokens >= tokens:
            self.tokens -= tokens
            return True
        return False
    
    def _refill(self):
        """Refill tokens based on time elapsed."""
        now = time.time()
        time_passed = now - self.last_refill
        tokens_to_add = time_passed * self.refill_rate
        
        self.tokens = min(self.capacity, self.tokens + tokens_to_add)
        self.last_refill = now
    
    def get_state(self):
        """Return current bucket state."""
        self._refill()
        return {
            'tokens': self.tokens,
            'capacity': self.capacity,
            'refill_rate': self.refill_rate
        }

class DistributedTokenBucket:
    """
    Token bucket with Redis backing for distributed rate limiting.
    """
    
    def __init__(self, redis_client, key, capacity, refill_rate):
        self.redis = redis_client
        self.key = key
        self.capacity = capacity
        self.refill_rate = refill_rate
    
    def allow_request(self, tokens=1):
        """Atomic check-and-consume using Redis Lua script."""
        lua_script = """
        local key = KEYS[1]
        local capacity = tonumber(ARGV[1])
        local refill_rate = tonumber(ARGV[2])
        local tokens_needed = tonumber(ARGV[3])
        local now = tonumber(ARGV[4])
        
        local bucket = redis.call('HMGET', key, 'tokens', 'last_refill')
        local current_tokens = tonumber(bucket[1]) or capacity
        local last_refill = tonumber(bucket[2]) or now
        
        -- Refill tokens
        local time_passed = now - last_refill
        local tokens_to_add = time_passed * refill_rate
        current_tokens = math.min(capacity, current_tokens + tokens_to_add)
        
        -- Check and consume
        if current_tokens >= tokens_needed then
            current_tokens = current_tokens - tokens_needed
            redis.call('HMSET', key, 'tokens', current_tokens, 'last_refill', now)
            return 1
        else
            return 0
        end
        """
        
        now = time.time()
        result = self.redis.eval(
            lua_script, 1, self.key,
            self.capacity, self.refill_rate, tokens, now
        )
        return result == 1

# Example usage
if __name__ == "__main__":
    # Simple rate limiter: 10 tokens, 2 tokens/second refill
    limiter = TokenBucket(capacity=10, refill_rate=2)
    
    # Burst of requests
    for i in range(12):
        allowed = limiter.allow_request()
        print(f"Request {i + 1}: {'✓' if allowed else '✗'}")
        time.sleep(0.1)
    
    # Wait and try again
    time.sleep(3)
    print(f"After 3s: {'✓' if limiter.allow_request() else '✗'}")
```
**Why**: Token bucket allows burst traffic while maintaining average rate. Simple O(1) operations with memory efficiency. Industry standard for API rate limiting (AWS, Stripe use variants). Supports distributed deployment with Redis.
