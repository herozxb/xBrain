## html_body_element
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Body Element Example</title>
</head>
<!-- PROBLEM: Create a functional body element with event handlers
     TIME: O(1) SPACE: O(1) -->
<body onload="initPage()" onunload="cleanupPage()">
    <h1>Welcome to My Page</h1>
    <p>This is the body content.</p>
    
    <script>
        // Initialize page when body loads
        function initPage() {
            console.log('Page loaded successfully');
            document.body.style.backgroundColor = '#f0f0f0';
        }
        
        // Cleanup when page unloads
        function cleanupPage() {
            console.log('Page unloaded');
        }
        
        // Dynamically modify body attributes
        function setBodyAttribute(name, value) {
            document.body.setAttribute(name, value);
            return document.body.getAttribute(name);
        }
    </script>
</body>
</html>
```
**Why**: The body element is the container for all visible page content and supports lifecycle events for initialization and cleanup.

## html_link_tag
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Link Tag Example</title>
    <!-- PROBLEM: Dynamically load external stylesheets and resources
         TIME: O(1) SPACE: O(1) -->
    <link rel="stylesheet" href="styles.css">
    <link rel="icon" href="favicon.ico" type="image/x-icon">
</head>
<body>
    <h1>Dynamic Link Loader</h1>
    <button onclick="loadStylesheet('dark-theme.css')">Load Dark Theme</button>
    
    <script>
        // Dynamically add a stylesheet link
        function loadStylesheet(url) {
            const link = document.createElement('link');
            link.rel = 'stylesheet';
            link.href = url;
            link.onload = () => console.log('Stylesheet loaded:', url);
            link.onerror = () => console.error('Failed to load:', url);
            
            document.head.appendChild(link);
            return link;
        }
        
        // Remove a stylesheet by URL
        function removeStylesheet(url) {
            const links = document.querySelectorAll('link[rel="stylesheet"]');
            links.forEach(link => {
                if (link.href.includes(url)) {
                    link.remove();
                    console.log('Stylesheet removed:', url);
                }
            });
        }
        
        // Check if stylesheet is loaded
        function isStylesheetLoaded(url) {
            const links = document.querySelectorAll('link[rel="stylesheet"]');
            return Array.from(links).some(link => link.href.includes(url));
        }
    </script>
</body>
</html>
```
**Why**: Link tags are essential for loading external resources like CSS stylesheets, icons, and other page resources.

## html_style_tag
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Style Tag Example</title>
    <!-- PROBLEM: Define and manage inline CSS styles
         TIME: O(1) SPACE: O(n) where n is CSS rules -->
    <style>
        body {
            font-family: Arial, sans-serif;
            margin: 0;
            padding: 20px;
        }
        .highlight {
            background-color: yellow;
        }
    </style>
</head>
<body>
    <h1>Dynamic Style Manager</h1>
    <p class="highlight">This paragraph is highlighted.</p>
    
    <script>
        // Add CSS rule to the first style tag
        function addCSSRule(selector, styles) {
            const styleTag = document.querySelector('style');
            const cssText = `${selector} { ${styles} }`;
            styleTag.textContent += '\n' + cssText;
            console.log('CSS rule added:', selector);
        }
        
        // Create a new style tag
        function createStyleTag(id) {
            const style = document.createElement('style');
            style.id = id;
            document.head.appendChild(style);
            return style;
        }
        
        // Apply styles dynamically
        function applyDynamicStyles() {
            addCSSRule('.dynamic', 'color: blue; font-weight: bold;');
            addCSSRule('#special', 'border: 2px solid red; padding: 10px;');
        }
        
        // Call to demonstrate
        applyDynamicStyles();
    </script>
    
    <p class="dynamic">This has dynamic styles applied.</p>
    <div id="special">This element has special styles.</div>
</body>
</html>
```
**Why**: Style tags allow embedding CSS directly in HTML, enabling scoped and dynamic styling without external files.

## html_script_tag
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Script Tag Example</title>
    <!-- PROBLEM: Load and execute JavaScript code dynamically
         TIME: O(1) SPACE: O(n) where n is script size -->
</head>
<body>
    <h1>Script Loader Demo</h1>
    <button onclick="loadExternalScript()">Load External Script</button>
    <button onclick="executeInlineScript()">Execute Inline Code</button>
    
    <script>
        // Load external JavaScript file dynamically
        function loadExternalScript() {
            const script = document.createElement('script');
            script.src = 'https://example.com/script.js';
            script.async = true;
            
            script.onload = () => console.log('External script loaded');
            script.onerror = () => console.error('Failed to load script');
            
            document.head.appendChild(script);
        }
        
        // Execute inline JavaScript code
        function executeInlineScript() {
            const script = document.createElement('script');
            script.textContent = `
                alert('Inline script executed!');
                console.log('This code runs immediately');
            `;
            document.body.appendChild(script);
            script.remove(); // Clean up
        }
        
        // Load script with callback
        function loadScriptWithCallback(url, callback) {
            const script = document.createElement('script');
            script.src = url;
            
            script.onload = () => {
                console.log('Script loaded:', url);
                if (callback) callback();
            };
            
            document.head.appendChild(script);
        }
    </script>
</body>
</html>
```
**Why**: Script tags are used to include or embed JavaScript code, enabling dynamic behavior and interactivity.

## html_comments
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>HTML Comments Example</title>
    <!-- PROBLEM: Add and manage HTML comments for documentation
         TIME: O(1) SPACE: O(1) -->
    <!-- This is a head comment -->
</head>
<body>
    <!-- Main content starts here -->
    <h1>Comment Manager</h1>
    
    <!-- 
        Multi-line comment example:
        - This explains the code structure
        - Helps with documentation
        - Not visible in browser
    -->
    
    <div id="content">
        <!-- Content will be loaded here -->
    </div>
    
    <script>
        // Add comment to an element
        function addComment(element, commentText) {
            const comment = document.createComment(commentText);
            element.parentNode.insertBefore(comment, element);
            return comment;
        }
        
        // Remove all comments from an element's children
        function removeComments(element) {
            const walker = document.createTreeWalker(
                element,
                NodeFilter.SHOW_COMMENT,
                null,
                false
            );
            
            const comments = [];
            while (walker.nextNode()) {
                comments.push(walker.currentNode);
            }
            
            comments.forEach(comment => comment.remove());
            return comments.length;
        }
        
        // Count comments in an element
        function countComments(element) {
            const comments = element.querySelectorAll('*');
            let count = 0;
            
            comments.forEach(node => {
                node.childNodes.forEach(child => {
                    if (child.nodeType === Node.COMMENT_NODE) {
                        count++;
                    }
                });
            });
            
            return count;
        }
        
        // Demonstrate comment usage
        addComment(document.getElementById('content'), 'Dynamic comment added');
    </script>
    
    <!-- Main content ends here -->
</body>
</html>
```
**Why**: Comments are essential for documenting code, explaining logic, and temporarily disabling code blocks during development.

## html_heading_tags
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Heading Tags Example</title>
    <!-- PROBLEM: Create and manage hierarchical heading structure
         TIME: O(1) SPACE: O(1) -->
    <style>
        h1 { color: navy; }
        h2 { color: darkblue; }
        h3 { color: steelblue; }
        h4 { color: lightblue; }
        h5 { color: skyblue; }
        h6 { color: lightgray; }
    </style>
</head>
<body>
    <h1>Main Heading (H1)</h1>
    <h2>Subheading (H2)</h2>
    <h3>Section (H3)</h3>
    <h4>Subsection (H4)</h4>
    <h5>Minor Heading (H5)</h5>
    <h6>Smallest Heading (H6)</h6>
    
    <script>
        // Create heading element
        function createHeading(level, text) {
            if (level < 1 || level > 6) {
                console.error('Heading level must be 1-6');
                return null;
            }
            
            const heading = document.createElement(`h${level}`);
            heading.textContent = text;
            return heading;
        }
        
        // Extract heading hierarchy
        function getHeadingHierarchy() {
            const headings = document.querySelectorAll('h1, h2, h3, h4, h5, h6');
            return Array.from(headings).map(h => ({
                level: parseInt(h.tagName.substring(1)),
                text: h.textContent,
                element: h
            }));
        }
        
        // Generate table of contents
        function generateTOC() {
            const toc = document.createElement('div');
            toc.innerHTML = '<h2>Table of Contents</h2>';
            
            const headings = getHeadingHierarchy();
            const list = document.createElement('ul');
            
            headings.forEach(h => {
                const item = document.createElement('li');
                item.style.marginLeft = `${(h.level - 1) * 20}px`;
                item.textContent = h.text;
                list.appendChild(item);
            });
            
            toc.appendChild(list);
            return toc;
        }
        
        // Add TOC to page
        document.body.insertBefore(generateTOC(), document.body.firstChild);
    </script>
</body>
</html>
```
**Why**: Heading tags (H1-H6) create semantic document structure, improve accessibility, and are crucial for SEO.

## html_paragraph_tag
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Paragraph Tag Example</title>
    <!-- PROBLEM: Create and manipulate paragraph elements dynamically
         TIME: O(1) SPACE: O(1) -->
    <style>
        p { 
            margin: 15px 0; 
            line-height: 1.6; 
        }
        .highlight {
            background-color: yellow;
            padding: 5px;
        }
    </style>
</head>
<body>
    <h1>Paragraph Manager</h1>
    <div id="paragraphContainer">
        <p>This is a sample paragraph.</p>
        <p class="highlight">This paragraph has a highlight class.</p>
    </div>
    
    <script>
        // Create a new paragraph
        function createParagraph(text, className = '') {
            const p = document.createElement('p');
            p.textContent = text;
            if (className) p.className = className;
            return p;
        }
        
        // Add paragraph to container
        function addParagraph(text, className = '') {
            const container = document.getElementById('paragraphContainer');
            const p = createParagraph(text, className);
            container.appendChild(p);
            return p;
        }
        
        // Get all paragraphs
        function getAllParagraphs() {
            return Array.from(document.querySelectorAll('p'));
        }
        
        // Count words in all paragraphs
        function countTotalWords() {
            const paragraphs = getAllParagraphs();
            return paragraphs.reduce((total, p) => {
                return total + p.textContent.split(/\s+/).filter(word => word.length > 0).length;
            }, 0);
        }
        
        // Find paragraph containing specific text
        function findParagraphContaining(text) {
            const paragraphs = getAllParagraphs();
            return paragraphs.find(p => p.textContent.includes(text));
        }
        
        // Highlight text in paragraphs
        function highlightTextInParagraphs(searchText) {
            const paragraphs = getAllParagraphs();
            paragraphs.forEach(p => {
                if (p.textContent.includes(searchText)) {
                    p.style.backgroundColor = 'yellow';
                }
            });
        }
        
        // Add some paragraphs dynamically
        addParagraph('This is a dynamically added paragraph.');
        addParagraph('Another paragraph with custom class.', 'highlight');
        
        console.log('Total words:', countTotalWords());
    </script>
</body>
</html>
```
**Why**: Paragraph tags define blocks of text, providing semantic structure and improving readability and accessibility.

## html_line_break
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Line Break Example</title>
    <!-- PROBLEM: Insert and manage line breaks in text content
         TIME: O(1) SPACE: O(1) -->
    <style>
        .poem {
            font-family: Georgia, serif;
            font-style: italic;
            margin: 20px;
        }
    </style>
</head>
<body>
    <h1>Line Break Manager</h1>
    <div id="content">
        <p>
            This is line one.<br>
            This is line two.<br>
            This is line three.
        </p>
        
        <div class="poem" id="poem">
            Roses are red<br>
            Violets are blue<br>
            Sugar is sweet<br>
            And so are you
        </div>
    </div>
    
    <script>
        // Add line break after an element
        function addLineBreak(element) {
            const br = document.createElement('br');
            element.parentNode.insertBefore(br, element.nextSibling);
            return br;
        }
        
        // Add multiple line breaks
        function addMultipleLineBreaks(element, count) {
            const breaks = [];
            for (let i = 0; i < count; i++) {
                breaks.push(addLineBreak(element));
            }
            return breaks;
        }
        
        // Convert newlines to <br> tags
        function convertNewlinesToBreaks(text) {
            return text.replace(/\n/g, '<br>');
        }
        
        // Remove all line breaks from element
        function removeLineBreaks(element) {
            const breaks = element.querySelectorAll('br');
            breaks.forEach(br => br.remove());
            return breaks.length;
        }
        
        // Count line breaks in element
        function countLineBreaks(element) {
            return element.querySelectorAll('br').length;
        }
        
        // Format text with line breaks
        function formatTextWithBreaks(text, maxLineLength) {
            const words = text.split(' ');
            let currentLine = '';
            let result = '';
            
            words.forEach(word => {
                if (currentLine.length + word.length > maxLineLength) {
                    result += currentLine + '<br>';
                    currentLine = word + ' ';
                } else {
                    currentLine += word + ' ';
                }
            });
            
            result += currentLine;
            return result;
        }
        
        // Demonstrate usage
        const poemElement = document.getElementById('poem');
        console.log('Line breaks in poem:', countLineBreaks(poemElement));
        
        // Format a long text
        const longText = 'This is a very long line of text that should be broken into multiple lines for better readability.';
        document.getElementById('content').innerHTML += '<p>' + formatTextWithBreaks(longText, 30) + '</p>';
    </script>
</body>
</html>
```
**Why**: Line breaks create visual line separation without starting new paragraphs, useful for poems, addresses, and formatted text.

## html_horizontal_rule
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Horizontal Rule Example</title>
    <!-- PROBLEM: Create thematic breaks between content sections
         TIME: O(1) SPACE: O(1) -->
    <style>
        hr {
            border: none;
            height: 2px;
            background: linear-gradient(to right, transparent, navy, transparent);
            margin: 30px 0;
        }
        
        hr.dashed {
            border-top: 2px dashed #999;
            background: none;
        }
        
        hr.dotted {
            border-top: 3px dotted #666;
            background: none;
        }
    </style>
</head>
<body>
    <h1>Content Section 1</h1>
    <p>This is the first section of content.</p>
    
    <hr>
    
    <h2>Content Section 2</h2>
    <p>This is the second section of content.</p>
    
    <hr class="dashed">
    
    <h3>Content Section 3</h3>
    <p>This is the third section of content.</p>
    
    <hr class="dotted">
    
    <script>
        // Create horizontal rule
        function createHorizontalRule(className = '') {
            const hr = document.createElement('hr');
            if (className) hr.className = className;
            return hr;
        }
        
        // Add HR after element
        function addSeparatorAfter(element, className = '') {
            const hr = createHorizontalRule(className);
            element.parentNode.insertBefore(hr, element.nextSibling);
            return hr;
        }
        
        // Add HR before element
        function addSeparatorBefore(element, className = '') {
            const hr = createHorizontalRule(className);
            element.parentNode.insertBefore(hr, element);
            return hr;
        }
        
        // Remove all HR elements
        function removeAllSeparators() {
            const hrs = document.querySelectorAll('hr');
            hrs.forEach(hr => hr.remove());
            return hrs.length;
        }
        
        // Style HR dynamically
        function styleHorizontalRule(hr, options) {
            if (options.color) hr.style.borderColor = options.color;
            if (options.height) hr.style.height = options.height;
            if (options.width) hr.style.width = options.width;
            if (options.margin) hr.style.margin = options.margin;
            return hr;
        }
        
        // Add decorative separators
        function addDecorativeSeparator() {
            const hr = createHorizontalRule();
            hr.style.background = 'linear-gradient(to right, red, orange, yellow, green, blue, violet)';
            hr.style.height = '3px';
            hr.style.border = 'none';
            document.body.appendChild(hr);
            return hr;
        }
        
        // Demonstrate
        addDecorativeSeparator();
        
        const heading = document.querySelector('h2');
        addSeparatorAfter(heading);
        
        console.log('Total HR elements:', document.querySelectorAll('hr').length);
    </script>
</body>
</html>
```
**Why**: Horizontal rules create visual separators between content sections, improving readability and document structure.
