```html
  -->

      const currentSrc = video.currentSrc;

      // Detect format from URL or MIME type
      const format = currentSrc.split('.').pop();
      console.log(`Format in use: ${format}`);
    });
    
    // Programmatically check supported formats

      const testVideo = document.createElement('video');
      return testVideo.canPlayType(mimeType) !== '';
```

**Why**: The `type` attribute helps browsers efficiently select the best media format without downloading headers, improving page load performance and ensuring cross-browser compatibility.

## html_aria_role_menuitem

```html
<!DOCTYPE html>
<html lang="en">
<head>

  <title>ARIA Menuitem Role - Accessible Menu System</title>

    .menu {

    PROBLEM: Create keyboard-accessible menus for screen readers

        <a role="menuitem" 
           href="#new" 
           tabindex="-1"

          New File

      </li>

           tabindex="-1">
          Save As...
        </a>
      </li>

        <hr>
      </li>
      <li role="none">
        <a role="menuitem" 
           href="#print" 
           tabindex="-1"
           aria-keyshortcuts="Ctrl+P">
          Print
        </a>
      </li>
      <li role="none">
        <span role="menuitem" 
              tabindex="-1"
              aria-disabled="true">
          Export (Coming Soon)
        </span>
      </li>
    </ul>
  </nav>

    class AccessibleMenu {
      constructor(menuElement) {
        this.menu = menuElement;
        this.items = Array.from(
          menuElement.querySelectorAll('[role="menuitem"]')
        );
        this.currentIndex = -1;
        
        this.init();
      }
      
      init() {
        // Handle keyboard navigation
        this.menu.addEventListener('keydown', (e) => {
          this.handleKeydown(e);
        });
        
        // Handle click focus
        this.items.forEach((item, index) => {
          item.addEventListener('click', () => {
            this.focusItem(index);
          });
        });
      }
      
      handleKeydown(e) {
        const enabledItems = this.items.filter(
          item => item.getAttribute('aria-disabled') !== 'true'
        );
        
        switch(e.key) {
          case 'ArrowDown':
          case 'ArrowRight':
            e.preventDefault();
            this.focusNext(enabledItems);
            break;
            
          case 'ArrowUp':
          case 'ArrowLeft':
            e.preventDefault();
            this.focusPrevious(enabledItems);
            break;
            
          case 'Home':
            e.preventDefault();
            this.focusFirst(enabledItems);
            break;
            
          case 'End':
            e.preventDefault();
            this.focusLast(enabledItems);
            break;
            
          case 'Enter':
          case ' ':
            e.preventDefault();
            this.activateCurrent();
            break;
            
          case 'Escape':
            this.closeMenu();
            break;
        }
      }
      
      focusItem(items, index) {
        items.forEach(item => item.setAttribute('tabindex', '-1'));
        items[index].setAttribute('tabindex', '0');
        items[index].focus();
        this.currentIndex = items.indexOf(items[index]);
      }
      
      focusNext(items) {
        const currentInEnabled = items.indexOf(
          items.find(item => item === document.activeElement)
        );
        const nextIndex = (currentInEnabled + 1) % items.length;
        this.focusItem(items, nextIndex);
      }
      
      focusPrevious(items) {
        const currentInEnabled = items.indexOf(
          items.find(item => item === document.activeElement)
        );
        const prevIndex = (currentInEnabled - 1 + items.length) % items.length;
        this.focusItem(items, prevIndex);
      }
      
      focusFirst(items) {
        this.focusItem(items, 0);
      }
      
      focusLast(items) {
        this.focusItem(items, items.length - 1);
      }
      
      activateCurrent() {
        const activeItem = document.activeElement;
        if (activeItem.getAttribute('aria-disabled') !== 'true') {
          console.log(`Activated: ${activeItem.textContent.trim()}`);
          activeItem.click();
        }
      }
      
      closeMenu() {
        // Move focus outside menu
        this.menu.blur();
        console.log('Menu closed');
      }
    }
    
    // Initialize menu
    const menu = new AccessibleMenu(document.querySelector('[role="menu"]'));
  </script>
</body>
</html>
```

**Why**: `role="menuitem"` provides semantic meaning for screen readers, enabling proper keyboard navigation (arrow keys, Enter, Escape) and ensuring menus are accessible to users with disabilities.

```html

<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Open Graph URL - Social Media Sharing</title>
  
  <!--
    PROBLEM: Define canonical URL for social media sharing previews
    TIME: O(1) SPACE: O(1)
    
    og:url specifies the canonical URL for the page.

  -->
  
  <!-- Basic Open Graph Tags -->
  <meta property="og:url" 
        content="https://example.com/articles/2025/html-meta-guide">

  <meta property="article:tag" content="Meta Tags">
  <meta property="article:tag" content="SEO">

<body>

  </article>
  
  <script>
    // Dynamically validate Open Graph tags
    function validateOpenGraphTags() {
      const requiredTags = ['og:url', 'og:type', 'og:title', 'og:image'];
      const issues = [];
      
      requiredTags.forEach(tag => {
        const meta = document.querySelector(`meta[property="${tag}"]`);
        if (!meta || !meta.content) {
          issues.push(`Missing: ${tag}`);
        }
      });
      
      // Validate og:url is absolute and canonical
      const ogUrl = document.querySelector('meta[property="og:url"]');
      if (ogUrl) {
        try {
          const url = new URL(ogUrl.content);
          
          // Check for common issues
          if (url.protocol !== 'https:') {
            issues.push('og:url should use HTTPS');
          }
          
          // Check if it matches canonical link
          const canonical = document.querySelector('link[rel="canonical"]');
          if (canonical && canonical.href !== ogUrl.content) {
            issues.push('og:url differs from canonical link');
          }
          
          // Remove trailing slash inconsistency
          const currentPath = window.location.href;
          if (currentPath !== ogUrl.content) {
            console.log('Note: og:url differs from current page URL');
            console.log(`og:url: ${ogUrl.content}`);
            console.log(`Current: ${currentPath}`);
          }
          
        } catch (e) {
          issues.push('og:url is not a valid URL');
        }
      }
      
      return issues;
    }
    
    // Get Open Graph data as object
    function getOpenGraphData() {
      const ogData = {};
      const metaTags = document.querySelectorAll('meta[property^="og:"]');
      
      metaTags.forEach(meta => {
        const property = meta.getAttribute('property').replace('og:', '');
        
        // Handle multiple values (like article:tag)
        if (ogData[property]) {
          if (Array.isArray(ogData[property])) {
            ogData[property].push(meta.content);
          } else {
            ogData[property] = [ogData[property], meta.content];
          }
        } else {
          ogData[property] = meta.content;
        }
      });
      
      return ogData;
    }
    
    // Run validation
    const issues = validateOpenGraphTags();
    if (issues.length > 0) {
      console.warn('Open Graph Issues:', issues);
    } else {
      console.log('Open Graph tags validated successfully');
    }
    
    // Display current OG data
    console.log('Open Graph Data:', getOpenGraphData());
    
    // Generate Facebook Debugger URL
    const ogUrl = document.querySelector('meta[property="og:url"]');
    if (ogUrl) {
      const debuggerUrl = `https://developers.facebook.com/tools/debug/?q=${encodeURIComponent(ogUrl.content)}`;
      console.log('Facebook Debugger:', debuggerUrl);
    }
  </script>
</body>
</html>
```

**Why**: `og:url` ensures social platforms recognize the canonical version of your content, preventing duplicate content issues when the same page is accessible via multiple URLs.

## html_link_rel_dns_prefetch

```html
<html lang="en">

  <!--
    PROBLEM: Reduce latency by resolving DNS before resources are requested
    TIME: O(1) SPACE: O(1)
    
    dns-prefetch hints the browser to resolve domain names early.

  -->
  
  <!-- Prefetch DNS for external resources -->
  <link rel="dns-prefetch" href="//fonts.googleapis.com">
  <link rel="dns-prefetch" href="//fonts.gstatic.com">

  <link rel="dns-prefetch" href="//analytics.google.com">

  <link rel="preconnect" href="https://fonts.googleapis.com" crossorigin>
  <link rel="preconnect" href="https://fonts.gstatic.com" crossorigin>
  
  <!-- Then load the resource -->
  <link href="https://fonts.googleapis.com/css2?family=Roboto&display=swap" 
        rel="stylesheet">
</head>
<body>
  <h1>DNS Prefetch Demo</h1>
  <p style="font-family: 'Roboto', sans-serif;">
    This text uses a Google Font. DNS resolution happened early!
  </p>
  
  <script>
    // Measure DNS resolution time
    async function measureDNSResolution(domain) {
      const start = performance.now();
      
      try {
        await fetch(`https://${domain}/favicon.ico`, {
          mode: 'no-cors',
          cache: 'no-store'
        });
      } catch (e) {
        // Expected for no-cors, we just want timing
      }
      
      return performance.now() - start;
    }
    
    // Analyze resource hints
    function analyzeResourceHints() {
      const dnsPrefetches = document.querySelectorAll('link[rel="dns-prefetch"]');
      const preconnects = document.querySelectorAll('link[rel="preconnect"]');
      
      console.log('DNS Prefetch hints:', dnsPrefetches.length);
      console.log('Preconnect hints:', preconnects.length);
      
      const hints = {
        dnsPrefetch: Array.from(dnsPrefetches).map(l => l.href),
        preconnect: Array.from(preconnects).map(l => l.href)
      };
      
      return hints;
    }
    
    // Check if DNS was cached
    function checkDNSCache(domain) {
      // Use Performance API to check timing
      const entries = performance.getEntriesByType('resource');
      const domainEntries = entries.filter(e => e.name.includes(domain));
      
      domainEntries.forEach(entry => {
        console.log(`${domain}:`);
        console.log(`  DNS Lookup: ${entry.domainLookupEnd - entry.domainLookupStart}ms`);
        console.log(`  TCP Connection: ${entry.connectEnd - entry.connectStart}ms`);
      });
    }
    
    // Dynamically add DNS prefetch hints
    function addDNSPrefetch(domain) {
      // Check if already exists
      const existing = document.querySelector(
        `link[rel="dns-prefetch"][href="//${domain}"]`
      );
      
      if (!existing) {
        const link = document.createElement('link');
        link.rel = 'dns-prefetch';
        link.href = `//${domain}`;

    console.log('Current hints:', analyzeResourceHints());

      setTimeout(() => {
        checkDNSCache('fonts.googleapis.com');
      }, 100);
    });
  </script>

```

**Why**: `dns-prefetch` reduces latency by ~50-300ms per domain by resolving DNS names in parallel with page parsing, before resources are actually needed.

## css_flexbox_order

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Flexbox Order - Visual Reordering</title>
  <style>
    /*
      PROBLEM: Reorder flex items visually without changing DOM
      TIME: O(1) SPACE: O(1)
      
      The order property changes visual display order of flex items.
      Default is 0. Lower values appear first.
    */
    
    .container {
      display: flex;
      gap: 10px;
      padding: 20px;
      background: #f0f0f0;
    }
    
    .item {
      padding: 20px;
      background: #007bff;
      color: white;
      border-radius: 4px;
      font-weight: bold;
    }
    
    /* Visual order differs from DOM order */
    .item-1 { order: 3; background: #dc3545; }
    .item-2 { order: 1; background: #28a745; }
    .item-3 { order: 4; background: #ffc107; color: #333; }
    .item-4 { order: 2; background: #17a2b8; }
    .item-5 { order: 0; background: #6f42c1; }
    
    /* Negative order values */
    .first { order: -1; }
    
    /* Responsive reordering */
    @media (max-width: 600px) {
      .mobile-first { order: -1; }
      .mobile-last { order: 999; }
    }
  </style>
</head>
<body>
  <h2>Flexbox Order Demo</h2>
  
  <!-- Items appear in order: 5, 2, 4, 1, 3 -->
  <div class="container">
    <div class="item item-1">Item 1 (order: 3)</div>
    <div class="item item-2">Item 2 (order: 1)</div>
    <div class="item item-3">Item 3 (order: 4)</div>
    <div class="item item-4">Item 4 (order: 2)</div>
    <div class="item item-5">Item 5 (order: 0)</div>
  </div>
  
  <script>
    // Dynamically reorder items
    class FlexOrderManager {
      constructor(container) {
        this.container = container;
        this.items = Array.from(container.children);
      }
      
      // Set order for specific item
      setOrder(element, order) {
        element.style.order = order;
      }
      
      // Shuffle items randomly
      shuffle() {
        this.items.forEach(item => {
          item.style.order = Math.floor(Math.random() * 10);
        });
      }
      
      // Sort by order value
      sortByOrder() {
        this.items.sort((a, b) => {
          const orderA = parseInt(getComputedStyle(a).order) || 0;
          const orderB = parseInt(getComputedStyle(b).order) || 0;
          return orderA - orderB;
        });
        
        // Log visual order (DOM unchanged)
        console.log('Visual order:', 
          this.items.map(i => i.textContent.trim())
        );
      }
      
      // Reset to DOM order
      reset() {
        this.items.forEach((item, index) => {
          item.style.order = '';
        });
      }
      
      // Move item to front (negative order)
      toFront(element) {
        // Find minimum current order
        const orders = this.items.map(item => 
          parseInt(getComputedStyle(item).order) || 0
        );
        const minOrder = Math.min(...orders);
        element.style.order = minOrder - 1;
      }
      
      // Move item to back
      toBack(element) {
        const orders = this.items.map(item => 
          parseInt(getComputedStyle(item).order) || 0
        );
        const maxOrder = Math.max(...orders);
        element.style.order = maxOrder + 1;
      }
      
      // Swap two items
      swap(item1, item2) {
        const order1 = getComputedStyle(item1).order || 0;
        const order2 = getComputedStyle(item2).order || 0;
        item1.style.order = order2;
        item2.style.order = order1;
      }
      
      // Group items with same order
      group(items, order) {
        items.forEach(item => {
          item.style.order = order;
        });
      }
    }
    
    // Initialize
    const container = document.querySelector('.container');
    const manager = new FlexOrderManager(container);
    
    // Get current visual order
    manager.sortByOrder();
    
    // Example: Shuffle on button click
    document.body.insertAdjacentHTML('beforeend', `
      <div style="margin: 20px; display: flex; gap: 10px;">
        <button id="shuffleBtn">Shuffle</button>
        <button id="resetBtn">Reset</button>
      </div>
    `);
    
    document.getElementById('shuffleBtn').addEventListener('click', () => {
      manager.shuffle();
    });
    
    document.getElementById('resetBtn').addEventListener('click', () => {
      manager.reset();
    });
  </script>
</body>
</html>
```

**Why**: `order` enables responsive layouts where visual priority differs from DOM order (e.g., mobile layouts), but remember it doesn't affect tab order or screen reader navigation—accessibility requires DOM order to match visual expectations.
