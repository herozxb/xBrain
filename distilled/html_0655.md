```html
      padding: 20px;
      background-color: #e0e0e0;

      .resolution-indicator {
        background-color: #fff;
```

```html
<!DOCTYPE html>
<html lang="en">
<head>

    // PROBLEM: Load modules on-demand instead of at page load
    // TIME: O(1) for import call, async loading varies SPACE: O(n) for module cache
    
    /**
     * Dynamically imports a module when needed
     * Why: Reduces initial bundle size, improves page load performance
     * @returns {Promise<void>}
     */
    async function loadChartOnDemand() {
      // Show loading state
      const container = document.getElementById('chartContainer');
      container.textContent = 'Loading chart module...';
      
      try {
        // Dynamic import returns a Promise that resolves to the module
        // Syntax: import(modulePath) - must be a string or template literal
        const chartModule = await import('./chart-utils.js');
        
        // Use the dynamically loaded module
        const chart = chartModule.createBarChart([10, 25, 45, 30]);
        container.innerHTML = '';
        container.appendChild(chart);
        
        console.log('Chart module loaded successfully!');
        
      } catch (error) {
        // Handle import failures (network issues, module not found)
        container.textContent = `Failed to load: ${error.message}`;
        console.error('Dynamic import failed:', error);
      }
    }
    
    // Attach click handler - module only loads when user interacts
    document.getElementById('loadChartBtn').addEventListener('click', loadChartOnDemand);
    
    // Pattern: Conditional loading based on feature detection
    async function loadPolyfillIfNeeded() {
      if (!window.customElements) {
        // Only load Web Components polyfill if browser doesn't support it
        await import('./web-components-polyfill.js');
        console.log('Polyfill loaded');
      }
    }
    
    // Pattern: Load different modules based on environment
    async function loadEnvironmentModule() {
      const isDevelopment = process.env.NODE_ENV === 'development';
      
      // Conditional dynamic import
      const config = await import(
        isDevelopment ? './config.dev.js' : './config.prod.js'
      );
      
      return config.default;
    }
  </script>
</body>
</html>
```
**Why:** Dynamic `import()` enables on-demand code loading, reducing initial bundle size. Returns a Promise, works great with `async/await`, ideal for route-based code splitting.

## js_object_create

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Object.create() - Prototype Inheritance</title>
</head>
<body>
  <script>
    // PROBLEM: Create new object with specified prototype
    // TIME: O(1) SPACE: O(1) for object + property descriptors
    
    /**
     * Demonstrates Object.create() for prototypal inheritance
     * Why: Clean way to set up prototype chains without constructor functions
     */
    
    // Create a prototype object with shared methods
    const animalPrototype = {
      // Method inherited by all animals
      speak() {
        return `${this.name} makes a sound`;
      },
      
      // Method to get species info
      describe() {
        return `${this.name} is a ${this.species}`;
      },
      
      // Initialize instance properties
      init(name, species) {
        this.name = name;
        this.species = species;
        return this; // Enable method chaining
      }
    };
    
    // Create a new animal using Object.create()
    const dog = Object.create(animalPrototype);
    dog.init('Buddy', 'Dog');
    
    console.log(dog.speak());    // "Buddy makes a sound"
    console.log(dog.describe()); // "Buddy is a Dog"
    
    // Second parameter: property descriptors object
    const cat = Object.create(animalPrototype, {
      name: {
        value: 'Whiskers',
        writable: false,    // Cannot be changed
        enumerable: true,
        configurable: false
      },
      species: {
        value: 'Cat',
        writable: true,
        enumerable: true,
        configurable: true
      }
    });
    
    console.log(cat.name); // "Whiskers"
    // cat.name = 'Max'; // Fails silently in non-strict mode
    
    // Create object with null prototype (no inherited properties)
    const pureObject = Object.create(null);
    pureObject.customProp = 'No toString, no hasOwnProperty';
    // pureObject.toString(); // TypeError!
    
    // Practical use: Object.create for inheritance chain
    const mammalPrototype = Object.create(animalPrototype);
    mammalPrototype.nurse = function() {
      return `${this.name} is nursing offspring`;
    };
    
    const whale = Object.create(mammalPrototype);
    whale.init('Blue Whale', 'Whale');
    
    console.log(whale.speak());  // Inherited from animalPrototype
    console.log(whale.nurse());  // Own method from mammalPrototype
  </script>
</body>
</html>
```
**Why:** `Object.create(proto, descriptors)` creates objects with explicit prototype chain. Second argument provides fine-grained property control via descriptors. Use `Object.create(null)` for dictionary objects without inherited methods.

```html
    // PROBLEM: Add a node as the last child of a parent element

     * Demonstrates appendChild() for DOM manipulation
     * Why: Fundamental method for building dynamic DOM structures
     */
    
    const list = document.getElementById('todoList');

      const li = document.createElement('li');
      li.textContent = text;
      
      // appendChild adds node as last child, returns the node

    }
    
    addBtn.addEventListener('click', () => {

    });
    
    // Important: appendChild moves existing nodes
    function demonstrateNodeMovement() {
      const container = document.createElement('div');
      container.innerHTML = '<span>First</span><span>Second</span>';
      document.body.appendChild(container);
      
      const firstSpan = container.querySelector('span');
      // Appending existing node MOVES it to end
      container.appendChild(firstSpan);
      // Result: Second, First (First was moved to end)
    }
    
    // DocumentFragment for batch appending (better performance)
    function appendMultipleItems(items) {
      // Create fragment to hold multiple nodes
      const fragment = document.createDocumentFragment();
      
      items.forEach(item => {
        const li = document.createElement('li');
        li.textContent = item;
        // Append to fragment (no reflow yet)
        fragment.appendChild(li);
      });
      
      // Single reflow for all items
      list.appendChild(fragment);
    }
    
    // Example: Building nested structure
    function createArticleCard(title, content) {
      const article = document.createElement('article');
      const h2 = document.createElement('h2');
      const p = document.createElement('p');
      
      h2.textContent = title;
      p.textContent = content;
      
      // Build hierarchy with appendChild
      article.appendChild(h2);
      article.appendChild(p);
      
      return article;
    }
    
    // Append card to body
    const card = createArticleCard(
      'appendChild() Guide',
      'Learn how to manipulate DOM nodes dynamically.'

```
**Why:** `parentNode.appendChild(node)` inserts node as last child. Returns the appended node. Important: appending existing node MOVES it. Use `DocumentFragment` for batch operations to minimize reflows.

## js_event_stopPropagation

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>stopPropagation() - Event Flow Control</title>
  <style>
    .outer {
      padding: 40px;
      background: #e0e0e0;
      border: 2px solid #666;
    }
    .middle {
      padding: 30px;
      background: #a0c8a0;
      border: 2px solid #4a4;
    }
    .inner {
      padding: 20px;
      background: #90b8f0;
      border: 2px solid #44a;
      cursor: pointer;
    }
    #log { margin-top: 20px; font-family: monospace; }
  </style>
</head>
<body>
  <div class="outer" id="outer">
    Outer
    <div class="middle" id="middle">
      Middle
      <div class="inner" id="inner">
        Inner (click me!)
      </div>
    </div>
  </div>
  
  <div id="log"></div>

  <script>
    // PROBLEM: Stop event from bubbling up or capturing down
    // TIME: O(1) SPACE: O(1)
    
    /**
     * Demonstrates stopPropagation() for event flow control
     * Why: Prevent parent handlers from receiving the same event
     */
    
    const log = (msg) => {
      const logDiv = document.getElementById('log');
      logDiv.innerHTML += msg + '<br>';
    };
    
    // Get all three divs
    const outer = document.getElementById('outer');
    const middle = document.getElementById('middle');
    const inner = document.getElementById('inner');
    
    // Without stopPropagation - all handlers fire (bubbling phase)
    outer.addEventListener('click', () => log('OUTER clicked'));
    middle.addEventListener('click', () => log('MIDDLE clicked'));
    inner.addEventListener('click', () => log('INNER clicked'));
    
    // With stopPropagation - event stops at this element
    const stoppingMiddle = document.createElement('div');
    stoppingMiddle.className = 'middle';
    stoppingMiddle.textContent = 'Middle (stops propagation)';
    
    const stoppingInner = document.createElement('div');
    stoppingInner.className = 'inner';
    stoppingInner.textContent = 'Inner (in stopping section)';
    stoppingMiddle.appendChild(stoppingInner);
    outer.appendChild(stoppingMiddle);
    
    stoppingMiddle.addEventListener('click', (e) => {
      log('STOPPING MIDDLE - event stops here!');
      e.stopPropagation(); // Event won't reach outer
    });
    
    stoppingInner.addEventListener('click', (e) => {
      log('STOPPING INNER clicked');
      // Event bubbles to stoppingMiddle, which stops it
    });
    
    // Practical use: Modal with backdrop click-to-close
    function createModal(content) {
      const backdrop = document.createElement('div');
      backdrop.className = 'modal-backdrop';
      
      const modal = document.createElement('div');
      modal.className = 'modal-content';
      modal.innerHTML = content;
      
      backdrop.appendChild(modal);
      
      // Close on backdrop click
      backdrop.addEventListener('click', () => {
        backdrop.remove();
      });
      
      // Prevent modal content clicks from closing
      modal.addEventListener('click', (e) => {
        e.stopPropagation(); // Stop here, don't reach backdrop
      });
      
      document.body.appendChild(backdrop);
    }
    
    // Practical use: Nested dropdowns
    document.querySelector('.dropdown-menu')?.addEventListener('click', (e) => {
      // Prevent document click handler from closing dropdown
      e.stopPropagation();
    });
    
    // Note: stopPropagation vs stopImmediatePropagation
    // - stopPropagation: stops bubbling to parents, but other handlers on SAME element still fire
    // - stopImmediatePropagation: stops everything including other handlers on same element
    
    // Warning: Overuse can make debugging difficult
    // Prefer checking event.target in parent handlers when possible
  </script>
</body>
</html>
```
**Why:** `event.stopPropagation()` stops event bubbling (or capturing). Essential for nested interactive elements (modals, dropdowns). Note: doesn't prevent default action - use `preventDefault()` for that.

## js_indexeddb_open

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>IndexedDB - Opening Database</title>
</head>
<body>
  <div id="status">Status: Ready</div>
  <button id="openBtn">Open Database</button>

  <script>
    // PROBLEM: Open/create an IndexedDB database for client-side storage
    // TIME: O(1) to initiate, async for completion SPACE: O(n) database size
    
    /**
     * Opens or creates an IndexedDB database
     * Why: Store large amounts of structured data client-side
     */
    
    let db = null;
    
    function openDatabase() {
      const statusDiv = document.getElementById('status');
      
      // Open database (or create if doesn't exist)
      // Parameters: name, version (version must be integer)
      const request = indexedDB.open('MyAppDatabase', 1);
      
      // Handle upgrade needed - runs when DB is new or version increased
      request.onupgradeneeded = (event) => {
        statusDiv.textContent = 'Status: Upgrading database...';
        
        const database = event.target.result;
        
        // Create object store (like a table) if it doesn't exist
        if (!database.objectStoreNames.contains('users')) {
          // keyPath specifies primary key property
          const usersStore = database.createObjectStore('users', {
            keyPath: 'id',
            autoIncrement: true
          });
          
          // Create indexes for searching
          usersStore.createIndex('email', 'email', { unique: true });
          usersStore.createIndex('name', 'name', { unique: false });
        }
        
        if (!database.objectStoreNames.contains('settings')) {
          database.createObjectStore('settings', { keyPath: 'key' });
        }
        
        statusDiv.textContent = 'Status: Database schema created/updated';
      };
      
      // Handle successful open
      request.onsuccess = (event) => {
        db = event.target.result;
        statusDiv.textContent = `Status: Database "${db.name}" opened (v${db.version})`;
        
        // Listen for unexpected closes (e.g., disk full, browser issues)
        db.onerror = (event) => {
          console.error('Database error:', event.target.error);
        };
        
        db.onclose = () => {
          console.log('Database connection closed');
          db = null;
        };
        
        db.onversionchange = () => {
          // Another tab wants to upgrade - close this connection
          db.close();
          alert('Database outdated, please reload the page');
        };
      };
      
      // Handle errors
      request.onerror = (event) => {
        statusDiv.textContent = `Status: Error - ${event.target.error}`;
        console.error('IndexedDB error:', event.target.error);
      };
      
      // Handle blocked (another tab has older version open)
      request.onblocked = () => {
        statusDiv.textContent = 'Status: Blocked - close other tabs using this DB';
      };
    }
    
    // Helper: Add user to database
    function addUser(user) {
      return new Promise((resolve, reject) => {
        if (!db) {
          reject(new Error('Database not open'));
          return;
        }
        
        // Create transaction with readwrite access
        const transaction = db.transaction(['users'], 'readwrite');
        const store = transaction.objectStore('users');
        
        const request = store.add(user);
        
        request.onsuccess = () => resolve(request.result);
        request.onerror = () => reject(request.error);
      });
    }
    
    document.getElementById('openBtn').addEventListener('click', openDatabase);
    
    // Check for IndexedDB support
    if (!window.indexedDB) {
      document.getElementById('status').textContent = 
        'Status: IndexedDB not supported in this browser';
    }
  </script>
</body>
</html>
```
**Why:** `indexedDB.open(name, version)` creates/opens a NoSQL database in browser. `onupgradeneeded` handles schema changes. Supports much larger data than localStorage. Use transactions for all operations.

## js_fetch_headers

```html
<!DOCTYPE html>
<html lang="en">
<head>

  <title>Fetch API - Working with Headers</title>

<body>
  <script>
    // PROBLEM: Manage HTTP request and response headers with Fetch API
    // TIME: O(1) for header operations SPACE: O(n) for header storage
    
    /**
     * Demonstrates Headers API for HTTP request/response customization
     * Why: Control content types, authentication, CORS, and caching
     */
    
    // Create Headers object for request
    const requestHeaders = new Headers();
    
    // Method 1: Append headers (allows multiple values for same name)
    requestHeaders.append('Content-Type', 'application/json');
    requestHeaders.append('Authorization', 'Bearer token123');
    requestHeaders.append('Accept', 'application/json');
    requestHeaders.append('X-Custom-Header', 'custom-value');
    
    // Method 2: Set header (overwrites existing values)
    requestHeaders.set('Accept', 'text/html');
    
    // Check if header exists
    console.log(requestHeaders.has('Authorization')); // true
    
    // Get header value
    console.log(requestHeaders.get('Content-Type')); // 'application/json'
    
    // Method 3: Pass object to constructor
    const altHeaders = new Headers({
      'Content-Type': 'application/json',
      'Authorization': 'Bearer token123'
    });
    
    // Fetch with custom headers
    async function fetchWithAuth(url) {
      const headers = new Headers({
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${localStorage.getItem('token')}`
      });
      
      const response = await fetch(url, {
        method: 'GET',
        headers: headers
      });
      
      // Access response headers
      const contentType = response.headers.get('Content-Type');
      const server = response.headers.get('Server');
      const rateLimit = response.headers.get('X-RateLimit-Remaining');
      
      console.log('Response Content-Type:', contentType);
      console.log('Server:', server);
      console.log('Rate limit remaining:', rateLimit);
      
      return response.json();
    }
    
    // POST with JSON body and headers
    async function postData(url, data) {
      const response = await fetch(url, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Accept': 'application/json'
        },
        body: JSON.stringify(data)
      });
      
      return response;
    }
    
    // Iterate over headers
    const headers = new Headers({
      'A': '1',
      'B': '2',
      'C': '3'
    });
    
    // entries(), keys(), values() methods
    for (const [name, value] of headers.entries()) {
      console.log(`${name}: ${value}`);
    }
    
    // Delete header
    requestHeaders.delete('X-Custom-Header');
    
    // Common headers reference:
    const commonHeaders = {
      // Content negotiation
      'Accept': 'application/json, text/plain, */*',
      'Accept-Language': 'en-US,en;q=0.9',
      'Accept-Encoding': 'gzip, deflate, br',
      
      // Content type
      'Content-Type': 'application/json; charset=utf-8',
      
      // Authentication
      'Authorization': 'Bearer <token>',
      
      // CORS
      'Origin': 'https://example.com',
      
      // Caching
      'Cache-Control': 'no-cache',
      'If-None-Match': '<etag>',
      
      // User agent
      'User-Agent': 'MyApp/1.0'
    };
  </script>
</body>
</html>
```
**Why:** `Headers` object manages HTTP headers with methods like `get()`, `set()`, `append()`, `delete()`. Pass to `fetch()` in options. Access response headers via `response.headers`. Handles multi-value headers correctly.

```html
    // PROBLEM: Programmatically check if form meets all validation constraints
    // TIME: O(n) where n is number of form elements SPACE: O(1)
    
    /**
     * Demonstrates checkValidity() for form validation
     * Why: Validate forms without submitting, enable custom error display
     */
    
    const form = document.getElementById('userForm');
    const resultDiv = document.getElementById('result');
    
    // Check entire form validity
    function validateForm() {
      // checkValidity() returns boolean
      const isValid = form.checkValidity();
      
      if (isValid) {
        resultDiv.textContent = '✅ Form is valid!';
        resultDiv.style.color = 'green';
      } else {
        resultDiv.textContent = '❌ Form has errors';
        resultDiv.style.color = 'red';
      }
      
      return isValid;
    }

    function validateField(input) {
      const errorSpan = document.getElementById(`${input.name}Error`);
      
      // checkValidity() works on individual elements too
      if (!input.checkValidity()) {
        // Show specific validation error
        errorSpan.textContent = input.validationMessage;
        return false;
      } else {
        errorSpan.textContent = '';
        return true;
      }
    }
    
    // Validate on input change (real-time feedback)
    form.querySelectorAll('input').forEach(input => {
      input.addEventListener('blur', () => validateField(input));
    });

    form.addEventListener('submit', (event) => {
      // Prevent default form submission
      event.preventDefault();
      
      // Check validity before processing
      if (!form.checkValidity()) {
        // form.reportValidity() shows native validation UI
        // form.reportValidity();
        
        // Or use custom handling:
        const invalidInputs = form.querySelectorAll(':invalid');
        invalidInputs.forEach(input => validateField(input));
        
        // Focus first invalid field
        invalidInputs[0]?.focus();
        return;

      // Form is valid - process data
      const formData = new FormData(form);
      const data = Object.fromEntries(formData);
      
      console.log('Submitting:', data);
      resultDiv.textContent = `Submitted: ${JSON.stringify(data)}`;

      const isValid = form.checkValidity();

    function getValidationDetails(input) {
      const validity = input.validity;
      
      // validity object has boolean flags:
      console.log({
        valueMissing: validity.valueMissing,   // required field empty
        typeMismatch: validity.typeMismatch,   // email/url pattern wrong
        patternMismatch: validity.patternMismatch, // pattern attribute fails
        tooLong: validity.tooLong,             // exceeds maxlength
        tooShort: validity.tooShort,           // below minlength
        rangeUnderflow: validity.rangeUnderflow, // below min
        rangeOverflow: validity.rangeOverflow,   // above max
        stepMismatch: validity.stepMismatch,     // doesn't match step
        badInput: validity.badInput,             // browser can't convert
        customError: validity.customError,       // setCustomValidity() used
        valid: validity.valid                    // true if all pass
      });
    }
```

## js_date_toISOString

```html
<!DOCTYPE html>

  <script>
    // PROBLEM: Convert Date object to ISO 8601 string format
    // TIME: O(1) SPACE: O(1)
    
    /**
     * Demonstrates toISOString() for standardized date formatting
     * Why: Universal format for APIs, databases, and data exchange
     */
    
    // Create a date
    const now = new Date();
    
    // toISOString returns: YYYY-MM-DDTHH:mm:ss.sssZ
    // Always in UTC (Z = UTC+00:00)
    console.log(now.toISOString());
    // Example: "2026-02-26T22:55:00.000Z"
    
    // Parsing ISO string back to Date
    const isoString = '2026-12-25T10:30:00.000Z';
    const parsedDate = new Date(isoString);
    console.log(parsedDate.toISOString()); // Same string
    
    // Create ISO string for current moment
    function getCurrentISOTimestamp() {
      return new Date().toISOString();
    }
    
    // For API requests
    async function saveRecord(data) {
      const record = {
        ...data,
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString()
      };
      
      // ISO format is universally accepted
      const response = await fetch('/api/records', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(record)
      });
      
      return response.json();
    }
    
    // Get ISO date only (without time)
    function getISODateString(date = new Date()) {
      // Returns: "2026-02-26"
      return date.toISOString().split('T')[0];
    }
    
    // Get ISO time only (without date)
    function getISOTimeString(date = new Date()) {
      // Returns: "22:55:00.000Z"
      return date.toISOString().split('T')[1];
    }
    
    // Create ISO string for specific date/time
    const specificDate = new Date(2026, 11, 25, 18, 30, 0); // Dec 25, 2026 18:30
    console.log(specificDate.toISOString());
    // Note: Converts local time to UTC
    
    // Compare dates using ISO strings
    function isAfter(date1, date2) {
      return date1.toISOString() > date2.toISOString();
    }
    
    // Sort dates
    const dates = [
      new Date('2026-03-15'),
      new Date('2026-01-10'),
      new Date('2026-06-20')
    ];
    
    // ISO strings sort correctly alphabetically
    const sortedISOStrings = dates
      .map(d => d.toISOString())
      .sort();
    
    console.log('Sorted:', sortedISOStrings);
    
    // Handle timezone conversion display
    function displayLocalTime(isoString) {
      const date = new Date(isoString);
      // Use toLocaleString for user-friendly display
      return date.toLocaleString('en-US', {
        weekday: 'short',
        year: 'numeric',
        month: 'short',
        day: 'numeric',
        hour: '2-digit',
        minute: '2-digit',
        timeZoneName: 'short'
      });
    }
    
    console.log(displayLocalTime('2026-02-26T22:55:00.000Z'));
    // Example: "Thu, Feb 27, 2026, 06:55 AM GMT+8"
    
    //toISOString throws RangeError for invalid dates
    function safeToISOString(date) {
      try {
        const iso = date.toISOString();
        return iso;
      } catch (e) {
        return null; // Invalid date
      }
    }
```

**Why:** `date.toISOString()` returns standardized ISO 8601 format (`YYYY-MM-DDTHH:mm:ss.sssZ`). Always UTC. Perfect for APIs, databases, and data exchange. Use `new Date(isoString)` to parse back.
