## 1. html_form_min_max

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Form Min/Max Validation</title>
  <style>
    .form-group { margin-bottom: 16px; }
    input:invalid { border-color: #e74c3c; }
    input:valid { border-color: #27ae60; }
    .error { color: #e74c3c; font-size: 12px; }
  </style>
</head>
<body>
  <!-- 
    PROBLEM: Validate numeric input within min/max range
    TIME: O(1) SPACE: O(1)
    
    WHY: Ensures user input stays within acceptable bounds
         without server round-trip
  -->
  <form id="rangeForm">
    <div class="form-group">
      <label for="age">Age (18-100):</label>
      <input type="number" id="age" name="age" min="18" max="100" required>
      <span class="error" id="ageError"></span>
    </div>
    <div class="form-group">
      <label for="quantity">Quantity (1-50):</label>
      <input type="number" id="quantity" name="quantity" min="1" max="50" required>
      <span class="error" id="quantityError"></span>
    </div>
    <button type="submit">Submit</button>
  </form>

  <script>
    // Validate min/max constraints on form submission
    function validateMinMax(input) {
      const min = parseFloat(input.min);
      const max = parseFloat(input.max);
      const value = parseFloat(input.value);
      
      if (isNaN(value)) return { valid: false, message: 'Please enter a number' };
      if (value < min) return { valid: false, message: `Minimum value is ${min}` };
      if (value > max) return { valid: false, message: `Maximum value is ${max}` };
      
      return { valid: true, message: '' };
    }

    // Attach validation listeners
    document.querySelectorAll('input[min][max]').forEach(input => {
      input.addEventListener('input', function() {
        const result = validateMinMax(this);
        const errorSpan = document.getElementById(this.id + 'Error');
        errorSpan.textContent = result.valid ? '' : result.message;
      });
    });
  </script>
</body>
</html>
```

```html
    }
```

```html
  <script>
    // Sample video playlist data
    const videos = [
      { id: 1, title: 'Introduction', src: 'intro.mp4', duration: '5:30' },
      { id: 2, title: 'Getting Started', src: 'start.mp4', duration: '8:15' },
      { id: 3, title: 'Advanced Topics', src: 'advanced.mp4', duration: '12:00' },
      { id: 4, title: 'Best Practices', src: 'practices.mp4', duration: '6:45' },
      { id: 5, title: 'Conclusion', src: 'conclusion.mp4', duration: '3:20' }
    ];

    let currentIndex = 0;

    // Render playlist items
    function renderPlaylist() {
      const playlist = document.getElementById('playlist');
      playlist.innerHTML = videos.map((video, index) => `
        <div class="playlist-item ${index === currentIndex ? 'active' : ''}" 
             data-index="${index}">
          <div class="title">${video.title}</div>
          <div class="duration">${video.duration}</div>
        </div>
      `).join('');

      // Attach click handlers
      playlist.querySelectorAll('.playlist-item').forEach(item => {
        item.addEventListener('click', () => {
          playVideo(parseInt(item.dataset.index));
        });
      });
    }

    // Play video at specific index
    function playVideo(index) {
      if (index < 0 || index >= videos.length) return;
      
      currentIndex = index;
      const video = videos[index];
      const player = document.getElementById('mainVideo');
      
      player.src = video.src;
      player.play();
      
      document.getElementById('currentTitle').textContent = video.title;
      renderPlaylist(); // Update active state
    }

    // Auto-advance to next video
    document.getElementById('mainVideo').addEventListener('ended', () => {
      if (currentIndex < videos.length - 1) {
        playVideo(currentIndex + 1);
      }
    });

    // Initialize
    renderPlaylist();
  </script>
```

```html
<body>
  <!-- 
    PROBLEM: Handle variable number of function arguments
    TIME: O(n) SPACE: O(n)
    
    WHY: Rest parameters replace arguments object with
         real arrays, enabling array methods
  -->
  <h2>JavaScript Rest Parameters</h2>
  
  <div class="demo">
    <button onclick="demoSum()">Sum Numbers</button>
    <button onclick="demoMerge()">Merge Arrays</button>
    <button onclick="demoValidate()">Validate Args</button>

  <script>
    const output = document.getElementById('output');

    // Basic rest parameter - collects all arguments into array
    function sum(...numbers) {
      return numbers.reduce((total, n) => total + n, 0);
    }

    // Rest with regular parameters
    function greet(greeting, ...names) {
      return names.map(name => `${greeting}, ${name}!`).join('\n');
    }

    // Merge multiple arrays
    function mergeArrays(...arrays) {
      return arrays.flat();
    }

    // Validation with required and optional params
    function validateUser(required, ...optional) {
      return {
        required: required,
        optional: optional.length > 0 ? optional : 'None',
        total: 1 + optional.length
      };
    }

    // Rest in destructuring
    function destructureDemo() {
      const [first, second, ...rest] = [1, 2, 3, 4, 5];
      const { a, ...others } = { a: 1, b: 2, c: 3 };
      return { first, second, rest, a, others };
    }

    // Demo handlers
    function demoSum() {
      const result = sum(1, 2, 3, 4, 5);
      output.textContent = `sum(1, 2, 3, 4, 5) = ${result}\n\n${greet('Hello', 'Alice', 'Bob', 'Charlie')}`;
    }

    function demoMerge() {
      const result = mergeArrays([1, 2], [3, 4], [5, 6]);
      output.textContent = `mergeArrays([1,2], [3,4], [5,6])\n=> [${result.join(', ')}]`;
    }

    function demoValidate() {
      const result = validateUser('id-required', 'email', 'phone', 'address');
      output.textContent = `validateUser('id-required', 'email', 'phone', 'address')\n${JSON.stringify(result, null, 2)}`;
    }

    function demoDestructure() {
      const result = destructureDemo();
      output.textContent = `Destructuring with rest:\n${JSON.stringify(result, null, 2)}`;
    }
  </script>
```
