```python
# PROBLEM: Spatial partitioning for 3D points
# TIME: O(log n) insertion/query, O(n) space
class OctreeNode:
    """8-ary tree for 3D spatial indexing."""
    def __init__(self, boundary, capacity=4, depth=0):
        self.boundary = boundary  # (x, y, z, width, height, depth)
        self.capacity = capacity
        self.depth = depth
        self.points = []
        self.children = [None] * 8  # 8 octants
        self.divided = False
    
    def _get_octant(self, point):
        """Determine which octant contains the point."""
        x, y, z = point
        cx, cy, cz, w, h, d = self.boundary
        mid_x, mid_y, mid_z = cx + w/2, cy + h/2, cz + d/2
        
        index = 0
        if x >= mid_x: index |= 1
        if y >= mid_y: index |= 2
        if z >= mid_z: index |= 4
        return index
    
    def _subdivide(self):
        """Create 8 child octants."""
        x, y, z, w, h, d = self.boundary
        hw, hh, hd = w/2, h/2, d/2
        
        for i in range(8):
            nx = x + hw if i & 1 else x
            ny = y + hh if i & 2 else y
            nz = z + hd if i & 4 else z
            self.children[i] = OctreeNode(
                (nx, ny, nz, hw, hh, hd),
                self.capacity,
                self.depth + 1
            )
        self.divided = True
    
    def insert(self, point):
        """Insert a 3D point into the octree."""
        if len(self.points) < self.capacity:
            self.points.append(point)
            return True
        
        if not self.divided:
            self._subdivide()
        
        octant = self._get_octant(point)
        return self.children[octant].insert(point)
    
    def query_range(self, range_box):
        """Find all points within a 3D range box."""
        found = []
        # Check if ranges overlap (simplified)
        # Implementation would check AABB intersection
        for point in self.points:
            if self._point_in_box(point, range_box):
                found.append(point)
        
        if self.divided:
            for child in self.children:
                if child:
                    found.extend(child.query_range(range_box))
        return found
```

**Why**: Efficient spatial queries for 3D graphics, collision detection, and nearest neighbor searches.

```python
        """Recalculate MBR from all entries."""
        if not self.entries:
            self.mbr = None
            return
        
        min_x = min(e[0][0] for e in self.entries)
        min_y = min(e[0][1] for e in self.entries)
        max_x = max(e[0][2] for e in self.entries)
        max_y = max(e[0][3] for e in self.entries)
```

```python
# PROBLEM: Memory-efficient string storage with common prefixes
# TIME: O(k) search/insert where k = key length, O(n) space
class CompressedTrieNode:
    """Space-efficient trie with path compression."""
    def __init__(self):
        self.children = {}  # char -> (label, node)
        self.is_end = False
        self.value = None
    
    def insert(self, word, value=None):
        """Insert word with path compression."""
        if not word:
            self.is_end = True
            self.value = value
            return
        
        # Find child with matching prefix
        for char, (label, child) in list(self.children.items()):
            # Find common prefix
            i = 0
            while i < min(len(word), len(label)) and word[i] == label[i]:
                i += 1
            
            if i > 0:  # Found common prefix
                if i == len(label):
                    # Word extends beyond label
                    child.insert(word[i:], value)
                else:
                    # Split the node
                    new_node = CompressedTrieNode()
                    new_node.children[label[i]] = (label[i:], child)
                    self.children[char] = (label[:i], new_node)
                    
                    if i == len(word):
                        new_node.is_end = True
                        new_node.value = value
                    else:
                        new_node.insert(word[i:], value)
                return
        
        # No matching prefix, create new edge
        self.children[word[0]] = (word, CompressedTrieNode())
        self.children[word[0]][1].is_end = True
        self.children[word[0]][1].value = value
    
    def search(self, word):
        """Search for word in compressed trie."""
        if not word:
            return self.is_end, self.value
        
        for char, (label, child) in self.children.items():
            if word.startswith(label):
                return child.search(word[len(label):])
        
        return False, None
```
**Why**: Autocomplete systems, IP routing tables, and dictionary implementations.

### Leftist Heap

```python
# PROBLEM: Efficient mergeable priority queue
# TIME: O(log n) merge/extract, O(1) insert amortized, O(n) space
class LeftistHeap:
    """Self-adjusting heap with efficient merge operation."""
    def __init__(self, value=None):
        self.value = value
        self.left = None
        self.right = None
        self.rank = 0  # Null path length
    
    def _update_rank(self):
        """Maintain leftist property."""
        if self.left is None or self.right is None:
            self.rank = 0
        else:
            self.rank = min(self.left.rank, self.right.rank) + 1
    
    @staticmethod
    def merge(h1, h2):
        """Merge two leftist heaps."""
        if h1 is None: return h2
        if h2 is None: return h1
        
        # Ensure h1 has smaller root
        if h1.value > h2.value:
            h1, h2 = h2, h1
        
        # Merge right subtree with h2
        h1.right = LeftistHeap.merge(h1.right, h2)
        
        # Maintain leftist property
        if h1.left is None or (h1.right and h1.left.rank < h1.right.rank):
            h1.left, h1.right = h1.right, h1.left
        
        h1._update_rank()
        return h1
    
    def insert(self, value):
        """Insert value by merging with singleton heap."""
        new_heap = LeftistHeap(value)
        return LeftistHeap.merge(self, new_heap)
    
    def extract_min(self):
        """Remove and return minimum value."""
        if self.value is None:
            return None, None
        
        min_val = self.value
        merged = LeftistHeap.merge(self.left, self.right)
        return min_val, merged
```
**Why**: Task scheduling with frequent merges, graph algorithms needing priority queue merges.

### Rollback Union-Find

```python
# PROBLEM: Union-Find with undo capability
# TIME: O(log n) find/union/rollback, O(n) space
class RollbackUnionFind:
    """Disjoint set with checkpoint/rollback support."""
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.history = []  # Stack for rollback
        self.checkpoint = 0
    
    def find(self, x):
        """Find root without path compression (preserves rollback)."""
        while self.parent[x] != x:
            x = self.parent[x]
        return x
    
    def union(self, x, y):
        """Unite sets, recording changes for rollback."""
        px, py = self.find(x), self.find(y)
        if px == py:
            self.history.append((False,))  # No change marker
            return False
        
        # Union by rank
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        
        # Record state before change
        self.history.append((True, py, self.parent[py], px, self.rank[px]))
        
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        return True
    
    def checkpoint(self):
        """Mark current state for rollback."""
        self.checkpoint = len(self.history)
    
    def rollback(self):
        """Undo operations until last checkpoint."""
        while len(self.history) > self.checkpoint:
            record = self.history.pop()
            if record[0]:  # Was a real change
                _, py, old_parent, px, old_rank = record
                self.parent[py] = old_parent
                self.rank[px] = old_rank
    
    def connected(self, x, y):
        """Check if x and y are in same set."""
        return self.find(x) == self.find(y)
```
**Why**: Dynamic connectivity with backtracking, offline algorithms, competitive programming.

### Robin Hood Hashing

```python
# PROBLEM: Hash table with reduced variance in probe lengths
# TIME: O(1) avg insert/lookup, O(n) space
class RobinHoodHashTable:
    """Open addressing with robin hood eviction."""
    def __init__(self, capacity=16, load_factor=0.7):
        self.capacity = capacity
        self.load_factor = load_factor
        self.size = 0
        self.keys = [None] * capacity
        self.values = [None] * capacity
        self.hashes = [None] * capacity  # Store hash for fast lookup
    
    def _probe_distance(self, pos, desired_pos):
        """Calculate how far element is from ideal position."""
        return (pos - desired_pos) % self.capacity
    
    def _find_slot(self, key):
        """Find appropriate slot using robin hood insertion."""
        h = hash(key)
        pos = h % self.capacity
        ideal_pos = pos
        current_dist = 0
        
        while self.keys[pos] is not None:
            if self.keys[pos] == key:
                return pos, True  # Found existing key
            
            # Robin hood: swap if we've traveled further
            existing_dist = self._probe_distance(pos, self.hashes[pos] % self.capacity)
            if current_dist > existing_dist:
                return pos, False  # Take this slot, evict occupant
            
            pos = (pos + 1) % self.capacity
            current_dist += 1
        
        return pos, False
    
    def insert(self, key, value):
        """Insert with robin hood eviction."""
        if self.size >= self.capacity * self.load_factor:
            self._resize()
        
        pos, found = self._find_slot(key)
        
        if found:
            self.values[pos] = value  # Update existing
            return
        
        # Robin hood insertion
        h = hash(key)
        current_key, current_val, current_hash = key, value, h
        
        while self.keys[pos] is not None:
            # Evict occupant if we've traveled further
            existing_dist = self._probe_distance(pos, self.hashes[pos] % self.capacity)
            our_dist = self._probe_distance(pos, current_hash % self.capacity)
            
            if our_dist > existing_dist:
                # Swap with occupant
                self.keys[pos], current_key = current_key, self.keys[pos]
                self.values[pos], current_val = current_val, self.values[pos]
                self.hashes[pos], current_hash = current_hash, self.hashes[pos]
            
            pos = (pos + 1) % self.capacity
        
        self.keys[pos] = current_key
        self.values[pos] = current_val
        self.hashes[pos] = current_hash
        self.size += 1
    
    def get(self, key):
        """Retrieve value by key."""
        pos, found = self._find_slot(key)
        if found:
            return self.values[pos]
        raise KeyError(key)
```
**Why**: Cache-friendly hash table with bounded probe lengths, game development, high-performance systems.

### Outbox Pattern

```python
# PROBLEM: Reliable message delivery with database transactions
# TIME: O(1) per message, O(n) storage
import threading
import queue
from dataclasses import dataclass
from datetime import datetime

@dataclass
class OutboxMessage:
    """Message stored for reliable delivery."""
    id: int
    aggregate_type: str
    aggregate_id: str
    event_type: str
    payload: dict
    created_at: datetime
    sent_at: datetime = None

class Outbox:
    """Ensures events are published exactly once."""
    def __init__(self, db_connection):
        self.db = db_connection
        self.pending = queue.Queue()
        self._start_publisher()
    
    def save_event(self, aggregate_type, aggregate_id, event_type, payload):
        """Save event within same transaction as state change."""
        # This would be called within the same DB transaction
        cursor = self.db.cursor()
        cursor.execute("""
            INSERT INTO outbox (aggregate_type, aggregate_id, event_type, payload, created_at)
            VALUES (?, ?, ?, ?, ?)
        """, (aggregate_type, aggregate_id, event_type, payload, datetime.now()))
        
        msg_id = cursor.lastrowid
        self.pending.put(msg_id)
        return msg_id
    
    def _start_publisher(self):
        """Background thread publishes pending messages."""
        def publisher():
            while True:
                msg_id = self.pending.get()
                self._publish_message(msg_id)
        
        thread = threading.Thread(target=publisher, daemon=True)
        thread.start()
    
    def _publish_message(self, msg_id):
        """Publish message and mark as sent."""
        cursor = self.db.cursor()
        cursor.execute("SELECT * FROM outbox WHERE id = ?", (msg_id,))
        row = cursor.fetchone()
        
        if row and row['sent_at'] is None:
            # Publish to message broker (pseudo-code)
            # broker.publish(row['event_type'], row['payload'])
            
            cursor.execute("""
                UPDATE outbox SET sent_at = ? WHERE id = ?
            """, (datetime.now(), msg_id))
            self.db.commit()
```
**Why**: Microservices event sourcing, ensuring domain events are published reliably.

### Saga Pattern (Choreography)

```python
# PROBLEM: Distributed transactions without 2PC
# TIME: O(n) for n services, O(n) storage
from dataclasses import dataclass
from enum import Enum
from typing import Callable, List

class SagaState(Enum):
    PENDING = "pending"
    COMPLETED = "completed"
    COMPENSATING = "compensating"
    FAILED = "failed"

@dataclass
class SagaStep:
    """Single step in saga with compensation."""
    name: str
    action: Callable
    compensate: Callable

class Saga:
    """Orchestrates distributed transaction with compensation."""
    def __init__(self, steps: List[SagaStep]):
        self.steps = steps
        self.completed_steps = []
        self.state = SagaState.PENDING
        self.results = {}
    
    def execute(self):
        """Execute steps forward, compensate on failure."""
        for step in self.steps:
            try:
                result = step.action()
                self.completed_steps.append(step)
                self.results[step.name] = result
                print(f"✓ Step '{step.name}' completed")
            except Exception as e:
                print(f"✗ Step '{step.name}' failed: {e}")
                self.state = SagaState.COMPENSATING
                self._compensate()
                self.state = SagaState.FAILED
                return False
        
        self.state = SagaState.COMPLETED
        return True
    
    def _compensate(self):
        """Rollback completed steps in reverse order."""
        print("Starting compensation...")
        for step in reversed(self.completed_steps):
            try:
                step.compensate(self.results.get(step.name))
                print(f"↩ Compensated '{step.name}'")
            except Exception as e:
                print(f"⚠ Compensation failed for '{step.name}': {e}")

# Example usage
def book_hotel(): return {"booking_id": "H123"}
def cancel_hotel(booking): print(f"Cancelled hotel {booking}")
def book_flight(): raise Exception("No seats available")
def cancel_flight(booking): pass

saga = Saga([
    SagaStep("hotel", book_hotel, cancel_hotel),
    SagaStep("flight", book_flight, cancel_flight),
])
saga.execute()  # Will book hotel, fail on flight, compensate hotel
```
**Why**: Maintaining consistency across microservices without distributed transactions.

```python
from functools import wraps

class StranglerRouter:
    """Route traffic between legacy and new systems."""

        self.legacy = legacy_system
        self.new = new_system
        self.routes = {}  # path -> (new_handler, enabled)

        """Decorator to register migrated routes."""
        def decorator(handler):
            self.routes[path] = (handler, enabled)
            return handler
        return decorator

        """Route request to appropriate system."""
        if path in self.routes:
            handler, enabled = self.routes[path]
            if enabled:
                result = handler(*args, **kwargs)
                if self.shadow_mode:
                    # Compare with legacy for validation
                    legacy_result = self.legacy.handle(path, *args, **kwargs)
                    self._compare_results(result, legacy_result)
                return result
        
        return self.legacy.handle(path, *args, **kwargs)
    
    def _compare_results(self, new_result, legacy_result):
        """Log differences in shadow mode."""
        if new_result != legacy_result:
            print(f"⚠ Result mismatch: new={new_result}, legacy={legacy_result}")
    
    def enable_route(self, path):
        """Enable migrated route."""
        if path in self.routes:
            self.routes[path] = (self.routes[path][0], True)
    
    def migrate_percentage(self, path, percentage):
        """Route percentage of traffic to new system."""
        import random
        if path in self.routes and random.random() < percentage / 100:
            return self.routes[path][0]
```

### Bulkhead Pattern

```python
# PROBLEM: Isolate failures to prevent cascade
# TIME: O(1) per isolation check, O(n) thread pools
import threading
from concurrent.futures import ThreadPoolExecutor
from dataclasses import dataclass

@dataclass
class BulkheadConfig:
    """Configuration for isolated resource pool."""
    name: str
    max_concurrent: int
    max_wait_ms: int = 1000

class Bulkhead:
    """Isolate resources to prevent failure propagation."""
    def __init__(self, config: BulkheadConfig):
        self.config = config
        self.semaphore = threading.Semaphore(config.max_concurrent)
        self.executor = ThreadPoolExecutor(max_workers=config.max_concurrent)
        self.active_calls = 0
        self.rejected_calls = 0
    
    def execute(self, func, *args, **kwargs):
        """Execute function within bulkhead isolation."""
        acquired = self.semaphore.acquire(
            blocking=True,
            timeout=self.config.max_wait_ms / 1000
        )
        
        if not acquired:
            self.rejected_calls += 1
            raise Exception(f"Bulkhead '{self.config.name}' full")
        
        try:
            self.active_calls += 1
            return func(*args, **kwargs)
        finally:
            self.active_calls -= 1
            self.semaphore.release()
    
    def metrics(self):
        """Return bulkhead statistics."""
        return {
            "name": self.config.name,
            "active": self.active_calls,
            "available": self.config.max_concurrent - self.active_calls,
            "rejected_total": self.rejected_calls
        }

# Usage: Separate bulkheads for different services
db_bulkhead = Bulkhead(BulkheadConfig("database", max_concurrent=10))
api_bulkhead = Bulkhead(BulkheadConfig("external_api", max_concurrent=5))
```
**Why**: Microservices resilience, preventing one slow service from consuming all resources.

### Consistent Hashing

```python
# PROBLEM: Distribute keys across dynamic servers
# TIME: O(log n) lookup with n servers, O(n) space
import hashlib
import bisect

class ConsistentHash:
    """Distribute keys across servers with minimal rebalancing."""
    def __init__(self, virtual_nodes=150):
        self.virtual_nodes = virtual_nodes
        self.ring = []  # Sorted hash positions
        self.nodes = {}  # hash -> node
        self.node_keys = {}  # node -> set of keys
    
    def _hash(self, key):
        """Generate consistent hash value."""
        return int(hashlib.md5(str(key).encode()).hexdigest(), 16)
    
    def add_node(self, node):
        """Add server with virtual nodes."""
        for i in range(self.virtual_nodes):
            virtual_key = f"{node}#{i}"
            h = self._hash(virtual_key)
            bisect.insort(self.ring, h)
            self.nodes[h] = node
        
        if node not in self.node_keys:
            self.node_keys[node] = set()
    
    def remove_node(self, node):
        """Remove server and its virtual nodes."""
        for i in range(self.virtual_nodes):
            virtual_key = f"{node}#{i}"
            h = self._hash(virtual_key)
            self.ring.remove(h)
            del self.nodes[h]
        
        del self.node_keys[node]
    
    def get_node(self, key):
        """Find node responsible for key."""
        if not self.ring:
            return None
        
        h = self._hash(key)
        idx = bisect.bisect_left(self.ring, h)
        
        # Wrap around to first node if needed
        if idx >= len(self.ring):
            idx = 0
        
        return self.nodes[self.ring[idx]]
    
    def get_distribution(self):
        """Show key distribution across nodes."""
        return {node: len(keys) for node, keys in self.node_keys.items()}
```
**Why**: Distributed caching, load balancing with minimal rebalancing on node changes.

```python
    def _reset_election_timer(self):
        """Reset election timeout (150-300ms)."""
        self.election_deadline = random.randint(150, 300)
    
    def request_vote(self, term, candidate_id, last_log_index, last_log_term):
        """Handle vote request from candidate."""
        if term < self.current_term:
            return False, self.current_term
        
        if term > self.current_term:
            self.current_term = term
            self.state = State.FOLLOWER
            self.voted_for = None
        
        can_vote = (
            self.voted_for is None or self.voted_for == candidate_id
        ) and self._is_log_up_to_date(last_log_index, last_log_term)
        
        if can_vote:
            self.voted_for = candidate_id
            self._reset_election_timer()
        
        return can_vote, self.current_term
    
    def _is_log_up_to_date(self, last_log_index, last_log_term):
        """Check if candidate's log is at least as current."""
        if not self.log:
            return True
        
        my_last_term = self.log[-1].term if self.log else 0
        my_last_index = len(self.log) - 1
        
        if last_log_term != my_last_term:
            return last_log_term > my_last_term
        return last_log_index >= my_last_index
    
    def append_entries(self, term, leader_id, prev_log_index, entries, leader_commit):
        """Handle append entries from leader."""
        if term < self.current_term:
            return False
        
        if term > self.current_term:
            self.current_term = term
            self.state = State.FOLLOWER
        
        self._reset_election_timer()
        
        # Append new entries (simplified)
        if entries:
            self.log.extend(entries)
        
        # Commit entries
        if leader_commit > self.commit_index:
            self.commit_index = min(leader_commit, len(self.log) - 1)
        
        return True
```
**Why**: Distributed databases, configuration management, leader election in clusters.

### Adaptive Rate Limiting

```python
# PROBLEM: Dynamically adjust rate limits based on system load
# TIME: O(1) per request check, O(1) space
import time
from collections import deque

class AdaptiveRateLimiter:
    """Rate limiter that adjusts based on system metrics."""
    def __init__(self, initial_rps=100, min_rps=10, max_rps=1000):
        self.current_limit = initial_rps
        self.min_limit = min_rps
        self.max_limit = max_rps
        
        self.requests = deque()  # Timestamps of recent requests
        self.errors = deque()  # Recent error timestamps
        self.latencies = deque(maxlen=100)  # Recent latencies
        
        self.window_ms = 1000
        self.target_latency_ms = 200
        self.adaptation_factor = 0.1
    
    def _clean_old(self, deque_obj):
        """Remove entries older than window."""
        cutoff = time.time() - self.window_ms / 1000
        while deque_obj and deque_obj[0] < cutoff:
            deque_obj.popleft()
    
    def allow_request(self):
        """Check if request should be allowed."""
        now = time.time()
        self._clean_old(self.requests)
        
        current_rps = len(self.requests)
        
        if current_rps >= self.current_limit:
            return False
        
        self.requests.append(now)
        return True
    
    def record_response(self, latency_ms: float, is_error: bool):
        """Record response for adaptive adjustment."""
        now = time.time()
        
        self.latencies.append(latency_ms)
        
        if is_error:
            self.errors.append(now)
        
        # Adapt limit based on metrics
        self._adapt_limit()
    
    def _adapt_limit(self):
        """Adjust rate limit based on system health."""
        if not self.latencies:
            return
        
        avg_latency = sum(self.latencies) / len(self.latencies)
        error_rate = len(self.errors) / max(len(self.requests), 1)
        
        # Decrease limit if high latency or errors
        if avg_latency > self.target_latency_ms * 2 or error_rate > 0.05:
            self.current_limit = max(
                self.min_limit,
                int(self.current_limit * (1 - self.adaptation_factor))
            )
        # Increase limit if system healthy
        elif avg_latency < self.target_latency_ms and error_rate < 0.01:
            self.current_limit = min(
                self.max_limit,
                int(self.current_limit * (1 + self.adaptation_factor))
            )
```
**Why**: API protection, preventing cascading failures, auto-scaling admission control.

### Optimistic vs Pessimistic Locking

```python
# PROBLEM: Handle concurrent data access
# TIME: O(1) per lock check, O(n) contention resolution
import time
import threading

class OptimisticLock:
    """Version-based concurrency control."""
    def __init__(self):
        self.data = {}
        self.versions = {}
        self.lock = threading.Lock()
    
    def read(self, key):
        """Read with version number."""
        with self.lock:
            return self.data.get(key), self.versions.get(key, 0)
    
    def write(self, key, value, expected_version):
        """Write only if version matches."""
        with self.lock:
            current_version = self.versions.get(key, 0)
            
            if current_version != expected_version:
                raise Exception(
                    f"Conflict: expected version {expected_version}, "
                    f"but current is {current_version}"
                )
            
            self.data[key] = value
            self.versions[key] = current_version + 1
            return True

class PessimisticLock:
    """Explicit locking for conflict prevention."""
    def __init__(self):
        self.data = {}
        self.locks = {}  # key -> Lock
        self.global_lock = threading.Lock()
    
    def _get_lock(self, key):
        """Get or create lock for key."""
        with self.global_lock:
            if key not in self.locks:
                self.locks[key] = threading.Lock()
            return self.locks[key]
    
    def read_with_lock(self, key):
        """Read with explicit lock."""
        lock = self._get_lock(key)
        with lock:
            return self.data.get(key)
    
    def write_with_lock(self, key, value):
        """Write with explicit lock."""
        lock = self._get_lock(key)
        with lock:
            self.data[key] = value
            return True
```
**Why**: Optimistic for low-contention reads, pessimistic for high-contention writes.

### Event Sourcing Basics

```python
# PROBLEM: Store all state changes as events
# TIME: O(1) append, O(n) replay, O(n) storage
from dataclasses import dataclass
from datetime import datetime
from typing import List, Callable
import json

@dataclass
class Event:
    """Immutable state change record."""
    event_type: str
    aggregate_id: str
    data: dict
    timestamp: datetime
    version: int

class EventStore:
    """Append-only store of all events."""
    def __init__(self):
        self.events: List[Event] = []
        self.subscribers: List[Callable] = []
        self.version = 0
    
    def append(self, event_type: str, aggregate_id: str, data: dict):
        """Add new event to stream."""
        self.version += 1
        event = Event(
            event_type=event_type,
            aggregate_id=aggregate_id,
            data=data,
            timestamp=datetime.now(),
            version=self.version
        )
        self.events.append(event)
        self._notify_subscribers(event)
        return event
    
    def get_events(self, aggregate_id: str = None, from_version: int = 0):
        """Retrieve events, optionally filtered."""
        for event in self.events:
            if event.version > from_version:
                if aggregate_id is None or event.aggregate_id == aggregate_id:
                    yield event
    
    def subscribe(self, callback: Callable):
        """Register for event notifications."""
        self.subscribers.append(callback)
    
    def _notify_subscribers(self, event: Event):
        """Notify all subscribers of new event."""
        for callback in self.subscribers:
            callback(event)

class BankAccount:
    """Example aggregate using event sourcing."""
    def __init__(self, account_id: str, event_store: EventStore):
        self.account_id = account_id
        self.event_store = event_store
        self.balance = 0
        self._replay_events()
    
    def _replay_events(self):
        """Rebuild state from event history."""
        for event in self.event_store.get_events(self.account_id):
            self._apply(event)
    
    def _apply(self, event: Event):
        """Apply event to current state."""
        if event.event_type == "AccountOpened":
            self.balance = 0
        elif event.event_type == "MoneyDeposited":
            self.balance += event.data["amount"]
        elif event.event_type == "MoneyWithdrawn":
            self.balance -= event.data["amount"]
    
    def deposit(self, amount: int):
        """Record deposit as event."""
        event = self.event_store.append(
            "MoneyDeposited",
            self.account_id,
            {"amount": amount}
        )
        self._apply(event)
```
**Why**: Complete audit trail, temporal queries, event replay for debugging/recovery.

```python
T = TypeVar('T')

    """Generic function with constraint."""
    if not items:
        raise ValueError("Empty list")
    
    max_item = items[0]
    for item in items[1:]:
        if item > max_item:
            max_item = item
    return max_item
```
