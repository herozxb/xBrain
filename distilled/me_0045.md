## 1. Flyweight Pattern Variants

```python
# PROBLEM: Share common parts of state among multiple objects to minimize memory usage
# TIME: O(1) for access SPACE: O(n) for shared state storage
class FlyweightFactory:
    """Factory for managing shared flyweight objects."""
    
    def __init__(self):
        self._flyweights = {}
    
    def get_flyweight(self, shared_state):
        """Get existing flyweight or create new one."""
        key = self._get_key(shared_state)
        if key not in self._flyweights:
            self._flyweights[key] = Flyweight(shared_state)
        return self._flyweights[key]
    
    def _get_key(self, state):
        """Generate unique key from shared state."""
        return tuple(sorted(state.items()))


class Flyweight:
    """Shared object containing intrinsic state."""
    
    def __init__(self, shared_state):
        self._shared_state = shared_state
    
    def operation(self, unique_state):
        """Execute operation combining shared and unique state."""
        return {**self._shared_state, **unique_state}


class Context:
    """Contains extrinsic state and reference to flyweight."""
    
    def __init__(self, flyweight, unique_state):
        self._flyweight = flyweight
        self._unique_state = unique_state
    
    def operation(self):
        """Delegate to flyweight with unique state."""
        return self._flyweight.operation(self._unique_state)
```

**Why**: Reduces memory footprint when many objects share identical state. Commonly used in text editors (character formatting), game engines (tree/terrain data), and GUI systems.

## 2. State Pattern Variants

```python
# PROBLEM: Allow object to alter behavior when internal state changes
# TIME: O(1) for state transition SPACE: O(n) for number of states
from abc import ABC, abstractmethod


class State(ABC):
    """Abstract base class for all states."""
    
    @abstractmethod
    def handle(self, context):
        """Handle request in current state."""
        pass


class ConcreteStateA(State):
    """First concrete state implementation."""
    
    def handle(self, context):
        print("State A handling request")
        # Transition to State B
        context.state = ConcreteStateB()


class ConcreteStateB(State):
    """Second concrete state implementation."""
    
    def handle(self, context):
        print("State B handling request")
        # Transition to State A
        context.state = ConcreteStateA()


class Context:
    """Maintains current state and delegates requests."""
    
    def __init__(self, initial_state):
        self._state = initial_state
    
    @property
    def state(self):
        return self._state
    
    @state.setter
    def state(self, new_state):
        self._state = new_state
    
    def request(self):
        """Delegate handling to current state."""
        self._state.handle(self)
```

**Why**: Eliminates large conditional statements. Makes state transitions explicit. Useful for order processing, TCP connections, document workflows, and game character states.

## 3. Read Model Pattern (CQRS)

```python
# PROBLEM: Optimize read operations separate from write operations
# TIME: O(1) for reads SPACE: O(n) for projection storage
class ReadModel:
    """Optimized read-only projection of domain data."""
    
    def __init__(self):
        self._projections = {}
        self._indexes = {}
    
    def update_projection(self, entity_id, data):
        """Update read projection with latest data."""
        self._projections[entity_id] = {
            'data': data,
            'version': data.get('version', 0),
            'timestamp': data.get('timestamp')
        }
        self._update_indexes(entity_id, data)
    
    def get_by_id(self, entity_id):
        """Fast lookup by entity ID."""
        return self._projections.get(entity_id)
    
    def get_by_index(self, index_name, value):
        """Query by secondary index."""
        index_key = f"{index_name}:{value}"
        entity_ids = self._indexes.get(index_key, [])
        return [self._projections[eid] for eid in entity_ids]
    
    def _update_indexes(self, entity_id, data):
        """Maintain secondary indexes."""
        for index_name, value in data.items():
            if index_name in ['status', 'category', 'user_id']:
                index_key = f"{index_name}:{value}"
                if index_key not in self._indexes:
                    self._indexes[index_key] = []
                if entity_id not in self._indexes[index_key]:
                    self._indexes[index_key].append(entity_id)


class ReadModelFacade:
    """Facade for querying read models."""
    
    def __init__(self, read_model):
        self._read_model = read_model
    
    def find_by_status(self, status):
        """Find all entities with given status."""
        return self._read_model.get_by_index('status', status)
    
    def find_by_user(self, user_id):
        """Find all entities for a user."""
        return self._read_model.get_by_index('user_id', user_id)
```

**Why**: Separates read and write concerns for scalability. Enables denormalized views optimized for queries. Critical for high-performance CQRS systems and event-sourced architectures.

```python
        self.version += 1
        # Override in subclasses
    
    def mark_events_committed(self):
        """Clear uncommitted events after persistence."""
        self.uncommitted_events = []


class WriteModelRepository:
    """Repository for write model aggregates."""
    
    def __init__(self, event_store):
        self._event_store = event_store
        self._aggregates = {}
    
    def save(self, aggregate):
        """Persist aggregate events."""
        if aggregate.uncommitted_events:
            self._event_store.append_events(
                aggregate.aggregate_id,
                aggregate.uncommitted_events,
                aggregate.version
            )
            aggregate.mark_events_committed()
    
    def load(self, aggregate_id, aggregate_class):
        """Rebuild aggregate from events."""
        events = self._event_store.get_events(aggregate_id)
        aggregate = aggregate_class(aggregate_id)
        for event in events:
            aggregate._apply_event(event)
        return aggregate
```

**Why**: Ensures consistency boundaries. Validates all invariants before persisting changes. Foundation of CQRS architecture with clear command validation and event sourcing.

## 5. Query Bus Pattern

```python
# PROBLEM: Decouple query execution from query definition
# TIME: O(1) for dispatch SPACE: O(n) for handler registry
from abc import ABC, abstractmethod
from typing import Type, Dict, Any


class Query(ABC):
    """Base query class."""
    pass


class QueryHandler(ABC):
    """Abstract handler for queries."""
    
    @abstractmethod
    def handle(self, query: Query) -> Any:
        """Execute query and return result."""
        pass


class QueryBus:
    """Dispatches queries to registered handlers."""
    
    def __init__(self):
        self._handlers: Dict[Type[Query], QueryHandler] = {}
    
    def register(self, query_class: Type[Query], handler: QueryHandler):
        """Register handler for query type."""
        self._handlers[query_class] = handler
    
    def dispatch(self, query: Query) -> Any:
        """Execute query and return result."""
        query_class = type(query)
        handler = self._handlers.get(query_class)
        if not handler:
            raise ValueError(f"No handler registered for {query_class.__name__}")
        return handler.handle(query)


# Example usage
class GetUserQuery(Query):
    """Query to fetch user by ID."""
    
    def __init__(self, user_id):
        self.user_id = user_id


class GetUserHandler(QueryHandler):
    """Handler for GetUserQuery."""
    
    def __init__(self, user_repository):
        self.user_repository = user_repository
    
    def handle(self, query: GetUserQuery):
        return self.user_repository.find_by_id(query.user_id)
```

**Why**: Separates query definition from execution. Enables middleware (caching, logging). Testable and extensible. Standard pattern in CQRS architectures.

## 6. Handler Pattern

```python
# PROBLEM: Encapsulate request handling logic in dedicated classes
# TIME: O(1) for dispatch SPACE: O(n) for handlers
from abc import ABC, abstractmethod
from typing import Any, Callable


class Handler(ABC):
    """Base handler interface."""
    
    @abstractmethod
    def handle(self, request: Any) -> Any:
        """Process request and return response."""
        pass
    
    @abstractmethod
    def can_handle(self, request: Any) -> bool:
        """Check if this handler can process the request."""
        pass


class HandlerChain:
    """Chain of responsibility for handlers."""
    
    def __init__(self):
        self._handlers = []
    
    def add_handler(self, handler: Handler):
        """Add handler to chain."""
        self._handlers.append(handler)
        return self
    
    def handle(self, request: Any) -> Any:
        """Find matching handler and execute."""
        for handler in self._handlers:
            if handler.can_handle(request):
                return handler.handle(request)
        raise ValueError("No handler found for request")


class RequestHandler(Handler):
    """Concrete handler for specific request type."""
    
    def __init__(self, request_type, processor: Callable):
        self._request_type = request_type
        self._processor = processor
    
    def can_handle(self, request: Any) -> bool:
        return isinstance(request, self._request_type)
    
    def handle(self, request: Any) -> Any:
        return self._processor(request)


class LoggingHandler(Handler):
    """Decorator handler that logs all requests."""
    
    def __init__(self, wrapped_handler: Handler, logger):
        self._handler = wrapped_handler
        self._logger = logger
    
    def can_handle(self, request: Any) -> bool:
        return self._handler.can_handle(request)
    
    def handle(self, request: Any) -> Any:
        self._logger.info(f"Handling request: {type(request).__name__}")
        result = self._handler.handle(request)
        self._logger.info(f"Request completed")
        return result
```

**Why**: Single responsibility principle. Each handler focuses on one type of request. Easy to add new handlers without modifying existing code. Common in command/query systems.

## 7. Pipeline Pattern

```python
# PROBLEM: Process data through a series of transformation stages
# TIME: O(n*m) where n=data size, m=pipeline stages SPACE: O(n) for intermediate data
from abc import ABC, abstractmethod
from typing import Any, List


class PipelineStage(ABC):
    """Abstract pipeline stage."""
    
    @abstractmethod
    def process(self, data: Any) -> Any:
        """Transform input data."""
        pass


class Pipeline:
    """Chain of processing stages."""
    
    def __init__(self):
        self._stages: List[PipelineStage] = []
    
    def add_stage(self, stage: PipelineStage):
        """Add stage to pipeline."""
        self._stages.append(stage)
        return self
    
    def execute(self, data: Any) -> Any:
        """Pass data through all stages."""
        result = data
        for stage in self._stages:
            result = stage.process(result)
        return result


class FilterStage(PipelineStage):
    """Filter out items not matching predicate."""
    
    def __init__(self, predicate):
        self._predicate = predicate
    
    def process(self, data: List[Any]) -> List[Any]:
        return [item for item in data if self._predicate(item)]


class TransformStage(PipelineStage):
    """Transform each item using function."""
    
    def __init__(self, transformer):
        self._transformer = transformer
    
    def process(self, data: List[Any]) -> List[Any]:
        return [self._transformer(item) for item in data]


class AggregateStage(PipelineStage):
    """Aggregate all items into single value."""
    
    def __init__(self, aggregator, initial=None):
        self._aggregator = aggregator
        self._initial = initial
    
    def process(self, data: List[Any]) -> Any:
        if not data:
            return self._initial
        result = self._initial
        for item in data:
            result = self._aggregator(result, item)
        return result

```

**Why**: Composable data processing. Each stage is independent and testable. Easy to add/remove stages. Used in data engineering, ETL, image processing, and middleware systems.

## 8. GIN Index Implementation

```python
# PROBLEM: Efficiently search composite values (arrays, full-text) using inverted index
# TIME: O(k) for search where k=number of matching items SPACE: O(n*m) where n=items, m=avg elements
from collections import defaultdict
from typing import List, Set, Any


class GINIndex:
    """Generalized Inverted Index for composite values."""
    
    def __init__(self):
        self._inverted_index = defaultdict(set)  # element -> set of item_ids
        self._items = {}  # item_id -> composite value
    
    def insert(self, item_id: Any, composite_value: List[Any]):
        """Insert item with composite value (e.g., array, tokens)."""
        # Remove old entries if updating
        if item_id in self._items:
            self._delete_from_index(item_id)
        
        # Store item
        self._items[item_id] = composite_value
        
        # Build inverted index entries
        for element in set(composite_value):  # Use set to avoid duplicates
            self._inverted_index[element].add(item_id)
    
    def search(self, query_elements: List[Any]) -> Set[Any]:
        """Find items containing all query elements."""
        if not query_elements:
            return set()
        
        # Get items matching first element
        result = self._inverted_index.get(query_elements[0], set()).copy()
        
        # Intersect with items matching remaining elements
        for element in query_elements[1:]:
            matching_items = self._inverted_index.get(element, set())
            result &= matching_items
            if not result:
                break
        
        return result
    
    def search_any(self, query_elements: List[Any]) -> Set[Any]:
        """Find items containing any query element."""
        result = set()
        for element in query_elements:
            result |= self._inverted_index.get(element, set())
        return result
    
    def delete(self, item_id: Any):
        """Remove item from index."""
        if item_id in self._items:
            self._delete_from_index(item_id)
            del self._items[item_id]
    
    def _delete_from_index(self, item_id: Any):
        """Remove item from inverted index."""
        composite_value = self._items[item_id]
        for element in composite_value:
            self._inverted_index[element].discard(item_id)
            if not self._inverted_index[element]:
                del self._inverted_index[element]
```

**Why**: Efficient for array containment queries, full-text search, and JSON indexing. PostgreSQL uses GIN for tsvector, array, and jsonb types. Supports fast intersection queries on composite values.

## 9. Compression Algorithms

```python
# PROBLEM: Reduce data size through encoding techniques
# TIME: O(n) for encoding/decoding SPACE: O(n) for compressed output
from collections import Counter
import heapq
from typing import Dict, Tuple


class RunLengthEncoding:
    """Compress consecutive repeated characters."""
    
    @staticmethod
    def encode(data: str) -> str:
        """Compress using run-length encoding."""
        if not data:
            return ""
        
        result = []
        count = 1
        
        for i in range(1, len(data)):
            if data[i] == data[i - 1]:
                count += 1
            else:
                result.append(f"{data[i - 1]}{count}")
                count = 1
        
        result.append(f"{data[-1]}{count}")
        return "".join(result)
    
    @staticmethod
    def decode(compressed: str) -> str:
        """Decompress run-length encoded data."""
        result = []
        i = 0
        
        while i < len(compressed):
            char = compressed[i]
            i += 1
            
            # Extract count
            count_str = ""
            while i < len(compressed) and compressed[i].isdigit():
                count_str += compressed[i]
                i += 1
            
            count = int(count_str) if count_str else 1
            result.append(char * count)
        
        return "".join(result)


class HuffmanCoding:
    """Variable-length prefix coding based on frequency."""
    
    def __init__(self):
        self._codes: Dict[str, str] = {}
        self._tree = None
    
    def build_tree(self, data: str):
        """Build Huffman tree from data."""
        # Count frequencies
        freq = Counter(data)
        
        # Create leaf nodes
        heap = [[weight, [char, ""]] for char, weight in freq.items()]
        heapq.heapify(heap)
        
        # Build tree
        while len(heap) > 1:
            lo = heapq.heappop(heap)
            hi = heapq.heappop(heap)
            
            for pair in lo[1:]:
                pair[1] = '0' + pair[1]
            for pair in hi[1:]:
                pair[1] = '1' + pair[1]
            
            heapq.heappush(heap, [lo[0] + hi[0]] + lo[1:] + hi[1:])
        
        # Extract codes
        self._codes = {char: code for char, code in heap[0][1:]}
    
    def encode(self, data: str) -> str:
        """Compress data using Huffman codes."""
        if not self._codes:
            self.build_tree(data)
        return "".join(self._codes[char] for char in data)
    
    def decode(self, encoded: str) -> str:
        """Decompress Huffman encoded data."""
        # Reverse mapping
        reverse_codes = {code: char for char, code in self._codes.items()}
        
        result = []
        current_code = ""
        
        for bit in encoded:
            current_code += bit
            if current_code in reverse_codes:
                result.append(reverse_codes[current_code])
                current_code = ""
        
        return "".join(result)


class LZWCompression:
    """Dictionary-based compression algorithm."""
    
    @staticmethod
    def encode(data: str) -> List[int]:
        """Compress using LZW algorithm."""
        # Initialize dictionary with single characters
        dictionary = {chr(i): i for i in range(256)}
        next_code = 256
        
        result = []
        current = ""
        
        for char in data:
            combined = current + char
            if combined in dictionary:
                current = combined
            else:
                result.append(dictionary[current])
                dictionary[combined] = next_code
                next_code += 1
                current = char
        
        if current:
            result.append(dictionary[current])
        
        return result
    
    @staticmethod
    def decode(compressed: List[int]) -> str:
        """Decompress LZW encoded data."""
        # Initialize dictionary
        dictionary = {i: chr(i) for i in range(256)}
        next_code = 256
        
        result = []
        current = dictionary[compressed[0]]
        result.append(current)
        
        for code in compressed[1:]:
            if code in dictionary:
                entry = dictionary[code]
            elif code == next_code:
                entry = current + current[0]
            else:
                raise ValueError("Invalid compressed data")
            
            result.append(entry)
            dictionary[next_code] = current + entry[0]
            next_code += 1
            current = entry
        
        return "".join(result)
```

**Why**: Fundamental for data storage and transmission. RLE for simple repeated patterns. Huffman for optimal prefix coding. LZW for dictionary-based compression (used in GIF, ZIP). Critical for databases and file systems.

## 10. Await Implementation

```python
# PROBLEM: Suspend execution until awaitable completes
# TIME: O(1) for suspension SPACE: O(n) for pending coroutines
import asyncio
from typing import Any, Coroutine


class Awaitable:
    """Custom awaitable object."""
    
    def __init__(self, value):
        self._value = value
    
    def __await__(self):
        """Make object awaitable."""
        # Yield control to event loop
        yield
        return self._value


class AsyncContext:
    """Async context manager example."""
    
    async def __aenter__(self):
        """Enter async context."""
        print("Entering async context")
        await asyncio.sleep(0.1)
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        """Exit async context."""
        print("Exiting async context")
        await asyncio.sleep(0.1)
        return False


class AsyncIterator:
    """Async iterator implementation."""
    
    def __init__(self, items):
        self._items = items
        self._index = 0
    
    def __aiter__(self):
        """Return async iterator."""
        return self
    
    async def __anext__(self):
        """Get next item asynchronously."""
        await asyncio.sleep(0.01)  # Simulate async operation
        
        if self._index >= len(self._items):
            raise StopAsyncIteration
        
        item = self._items[self._index]
        self._index += 1
        return item


async def gather_with_concurrency_limit(n: int, *coros: Coroutine) -> list:
    """Execute coroutines with limited concurrency."""
    semaphore = asyncio.Semaphore(n)
    
    async def bounded_coro(coro):
        async with semaphore:
            return await coro
    
    return await asyncio.gather(*[bounded_coro(c) for c in coros])


async def wait_with_timeout(coro: Coroutine, timeout: float) -> Any:
    """Await coroutine with timeout."""
    try:
        return await asyncio.wait_for(coro, timeout=timeout)
    except asyncio.TimeoutError:
        raise TimeoutError(f"Coroutine exceeded {timeout}s timeout")


async def retry_with_backoff(coro_factory, max_retries: int = 3, base_delay: float = 1.0):
    """Retry failed coroutine with exponential backoff."""
    for attempt in range(max_retries):
        try:
            return await coro_factory()
        except Exception as e:
            if attempt == max_retries - 1:
                raise
            delay = base_delay * (2 ** attempt)
            await asyncio.sleep(delay)


# Example usage
async def fetch_data(url):
    """Simulate async data fetching."""
    await asyncio.sleep(0.1)
    return f"Data from {url}"


async def main():
    """Demonstrate await patterns."""
    # Basic await
    data = await fetch_data("https://example.com")
    
    # Async context manager
    async with AsyncContext() as ctx:
        print("Inside context")
    
    # Async iteration
    async for item in AsyncIterator([1, 2, 3]):
        print(item)
    
    # Concurrent execution with limit
    results = await gather_with_concurrency_limit(
        2,
        fetch_data("url1"),
        fetch_data("url2"),
        fetch_data("url3")
    )
    
    # Timeout
    try:
        result = await wait_with_timeout(fetch_data("url"), timeout=0.05)
    except TimeoutError:
        print("Request timed out")
    
    # Retry with backoff
    result = await retry_with_backoff(lambda: fetch_data("url"))
```

**Why**: Foundation of Python async/await. Enables non-blocking I/O operations. Critical for high-performance web servers, API clients, and concurrent data processing. Provides structured concurrency and resource management.
