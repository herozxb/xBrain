# Memento Design Pattern

## [Memento Pattern - Undo/Redo System]

```python
# PROBLEM: Implement an undo/redo mechanism without violating encapsulation
# TIME: O(1) for save/undo/redo operations SPACE: O(n) where n is number of states

class Memento:
    """
    Memento class stores the internal state of the Originator.
    The state is immutable and can only be accessed by the Originator.
    """
    
    def __init__(self, state: str):
        """Initialize memento with a state"""
        self._state = state  # Store the state to be saved
    
    def get_state(self) -> str:
        """Retrieve the saved state - only Originator should call this"""
        return self._state


class Originator:
    """
    Originator creates a memento containing a snapshot of its internal state.
    It can also use the memento to restore its previous state.
    """
    
    def __init__(self):
        """Initialize originator with an empty state"""
        self._state = ""  # Current internal state
    
    def set_state(self, state: str) -> None:
        """Set a new state"""
        self._state = state  # Update the current state
        print(f"Originator: State changed to: {state}")
    
    def get_state(self) -> str:
        """Get current state"""
        return self._state  # Return current state
    
    def create_memento(self) -> Memento:
        """Create a memento containing a snapshot of current state"""
        return Memento(self._state)  # Save current state to memento
    
    def restore_from_memento(self, memento: Memento) -> None:
        """Restore state from a memento"""
        self._state = memento.get_state()  # Restore state from memento
        print(f"Originator: State restored to: {self._state}")


class Caretaker:
    """
    Caretaker is responsible for keeping track of mementos.
    It never modifies or inspects the contents of mementos.
    """
    
    def __init__(self):
        """Initialize caretaker with empty history"""
        self._history = []  # Stack to store mementos (for undo)
        self._future = []   # Stack to store undone mementos (for redo)
    
    def save_state(self, memento: Memento) -> None:
        """Save a memento to history"""
        self._history.append(memento)  # Add memento to history
        self._future.clear()  # Clear future states when new state is saved
        print("Caretaker: State saved to history")
    
    def undo(self, originator: Originator) -> bool:
        """Undo the last state change"""
        if len(self._history) <= 1:
            print("Caretaker: Cannot undo - no history available")
            return False
        
        # Move current state to future stack
        current_memento = self._history.pop()
        self._future.append(current_memento)
        
        # Restore previous state
        previous_memento = self._history[-1]
        originator.restore_from_memento(previous_memento)
        print("Caretaker: Undo successful")
        return True
    
    def redo(self, originator: Originator) -> bool:
        """Redo the last undone state change"""
        if not self._future:
            print("Caretaker: Cannot redo - no future state available")
            return False
        
        # Move state from future to history
        memento = self._future.pop()
        self._history.append(memento)
        
        # Restore the state
        originator.restore_from_memento(memento)
        print("Caretaker: Redo successful")
        return True
    
    def get_history_size(self) -> int:
        """Get number of states in history"""
        return len(self._history)  # Return history count


# Demo usage
def main():
    """Demonstrate the Memento pattern with undo/redo functionality"""
    
    # Create originator and caretaker
    originator = Originator()
    caretaker = Caretaker()
    
    # Set initial state and save
    originator.set_state("State 1: Initial")
    caretaker.save_state(originator.create_memento())
    
    # Make changes and save each state
    originator.set_state("State 2: Added feature A")
    caretaker.save_state(originator.create_memento())
    
    originator.set_state("State 3: Added feature B")
    caretaker.save_state(originator.create_memento())
    
    originator.set_state("State 4: Added feature C")
    caretaker.save_state(originator.create_memento())
    
    print(f"\nHistory size: {caretaker.get_history_size()}")
    
    # Test undo functionality
    print("\n=== Testing Undo ===")
    caretaker.undo(originator)  # Back to State 3
    caretaker.undo(originator)  # Back to State 2
    
    # Test redo functionality
    print("\n=== Testing Redo ===")
    caretaker.redo(originator)  # Forward to State 3
    
    # Make a new change (clears future states)
    print("\n=== Making New Change ===")
    originator.set_state("State 5: New feature D")
    caretaker.save_state(originator.create_memento())
    
    # Try to redo (should fail - future cleared)
    print("\n=== Trying Redo After New Change ===")
    caretaker.redo(originator)  # No future state available
    
    print(f"\nFinal state: {originator.get_state()}")
    print(f"History size: {caretaker.get_history_size()}")


if __name__ == "__main__":
    main()
```

## Why Use the Memento Pattern?

**Purpose**: The Memento pattern captures and externalizes an object's internal state without violating encapsulation, allowing the object to be restored to this state later.

**Key Benefits**:
1. **Encapsulation Preserved**: The internal state of the Originator is not exposed to external objects
2. **Undo/Redo Support**: Enables easy implementation of undo/redo functionality
3. **State History**: Maintains a history of states without compromising the Originator's design
4. **Separation of Concerns**: Caretaker manages history while Originator manages state

**Real-World Applications**:
- Text editors (undo/redo operations)
- Database transactions (rollback mechanisms)
- Game state saves (save/load game progress)
- Configuration management (version control for settings)

**Complexity Analysis**:
- **Save Operation**: O(1) - Adding memento to history stack
- **Undo Operation**: O(1) - Popping from history, pushing to future stack
- **Redo Operation**: O(1) - Popping from future, pushing to history stack
- **Space Complexity**: O(n) where n is the number of saved states

**Design Pattern Category**: Behavioral Pattern
