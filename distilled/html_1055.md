```html
  <!-- PROBLEM: Create disabled button that remains focusable for accessibility -->
  <!-- TIME: O(1) SPACE: O(1) -->

  <button id="action-btn" aria-disabled="false">Submit Form</button>
  <button id="toggle-disable">Toggle Disable</button>

  <script>
    /**
     * Set aria-disabled state on an element
     * Unlike HTML disabled, aria-disabled keeps element focusable
     * @param {HTMLElement} element - Target element
     * @param {boolean} disabled - Whether to disable
     */
    function setAriaDisabled(element, disabled) {
      element.setAttribute('aria-disabled', String(disabled));
      
      if (disabled) {
        // Prevent click action but keep focusable
        element.dataset.wasDisabled = 'true';
      } else {
        delete element.dataset.wasDisabled;
      }
    }

    const actionBtn = document.getElementById('action-btn');
    const toggleBtn = document.getElementById('toggle-disable');

    // Intercept clicks on aria-disabled elements
    actionBtn.addEventListener('click', function() {
      if (this.getAttribute('aria-disabled') === 'true') {
        console.log('Button is disabled');
        return;
      }
      console.log('Form submitted!');
    });

    // Toggle disabled state
    toggleBtn.addEventListener('click', function() {
      const isDisabled = actionBtn.getAttribute('aria-disabled') === 'true';
      setAriaDisabled(actionBtn, !isDisabled);
      this.textContent = !isDisabled ? 'Enable Button' : 'Toggle Disable';
    });
  </script>
</body>
</html>
```

**Why**: `aria-disabled` indicates an element is perceivable but not operable. Unlike HTML `disabled`, it keeps elements focusable so screen readers can announce the disabled state.

```html
<!DOCTYPE html>
<html lang="en">
<head>
```

```html
<body>
  <h2>Array.pop() - Remove Last Element</h2>
  <pre id="output"></pre>

  <script>
    /* PROBLEM: Remove and return last element from array */
    /* TIME: O(1) SPACE: O(1) */

    /**
     * Remove and return the last element from an array
     * Mutates the original array
     * @param {Array} arr - Input array
     * @returns {*} The removed element, or undefined if empty
     */
    function arrayPop(arr) {
      // Array.prototype.pop() removes last element
      // Returns undefined for empty arrays
      return arr.pop();
    }

    /**
     * Pop multiple elements from end of array
     * @param {Array} arr - Input array
     * @param {number} count - Number of elements to pop
     * @returns {Array} Array of popped elements (reversed order)
     */
    function popMultiple(arr, count) {
      const result = [];
      for (let i = 0; i < count && arr.length > 0; i++) {
        result.unshift(arr.pop()); // Maintain order
      }
      return result;
    }

    /**
     * Pop until condition is met (excludes matching element)
     * @param {Array} arr - Input array
     * @param {Function} predicate - Condition function
     * @returns {Array} Popped elements
     */
    function popUntil(arr, predicate) {
      const result = [];
      while (arr.length > 0 && !predicate(arr[arr.length - 1])) {
        result.unshift(arr.pop());
      }
      return result;
    }

    /**
     * Safe pop that doesn't mutate original array
     * @param {Array} arr - Input array
     * @returns {Object} { element, array: new array without last element }
     */
    function popImmutable(arr) {
      if (arr.length === 0) {
        return { element: undefined, array: [] };
      }
      return {
        element: arr[arr.length - 1],
        array: arr.slice(0, -1)
      };
    }

    // === Demo ===
    const log = (msg) => document.getElementById('output').textContent += msg + '\n';

    // Basic pop
    let fruits = ['apple', 'banana', 'cherry', 'date'];
    log('Original: ' + JSON.stringify(fruits));

    const removed = arrayPop(fruits);
    log('Popped: ' + removed);
    log('After pop: ' + JSON.stringify(fruits));

    // Pop multiple
    let numbers = [1, 2, 3, 4, 5, 6, 7, 8];
    log('\nOriginal numbers: ' + JSON.stringify(numbers));
    const popped = popMultiple(numbers, 3);
    log('Popped 3: ' + JSON.stringify(popped));
    log('Remaining: ' + JSON.stringify(numbers));

    // Pop until
    let values = [10, 20, 30, 40, 50, 0];
    log('\nOriginal values: ' + JSON.stringify(values));
    const beforeZero = popUntil(values, v => v === 0);
    log('Before zero: ' + JSON.stringify(beforeZero));
    log('Remaining: ' + JSON.stringify(values));

    // Immutable pop
    let original = ['a', 'b', 'c'];
    log('\nOriginal: ' + JSON.stringify(original));
    const { element, array } = popImmutable(original);
    log('Immutable pop - element: ' + element);
    log('New array: ' + JSON.stringify(array));
    log('Original unchanged: ' + JSON.stringify(original));
  </script>
</body>
</html>
```

**Why**: `Array.pop()` removes the last element in O(1) time. Mutates the array and returns the removed element. Returns `undefined` for empty arrays. Use for stack-like operations.

```html
    /* PROBLEM: Add key-value pairs to Map with any key type */
    /* TIME: O(1) average SPACE: O(1) per entry */

    /**
     * Create and populate a Map
     * Map keys can be any type (objects, functions, primitives)
     * @param {Array} entries - Array of [key, value] pairs
     * @returns {Map} New Map instance
     */
    function createMap(entries = []) {

    }

    /**
     * Set a value in a Map (returns the Map for chaining)
     * @param {Map} map - Target Map
     * @param {*} key - Any value as key
     * @param {*} value - Value to store
     * @returns {Map} The same Map (for chaining)
     */
    function mapSet(map, key, value) {
      return map.set(key, value);
    }

    /**
     * Set multiple entries at once
     * @param {Map} map - Target Map
     * @param {Array} entries - Array of [key, value] pairs
     * @returns {Map} The same Map
     */
    function mapSetMultiple(map, entries) {
      entries.forEach(([key, value]) => map.set(key, value));
      return map;
    }

    /**
     * Set with expiration (time-based cache)
     * @param {Map} map - Target Map
     * @param {*} key - Cache key
     * @param {*} value - Value to cache
     * @param {number} ttlMs - Time to live in milliseconds
     */
    function setWithExpiry(map, key, value, ttlMs) {
      const expiry = Date.now() + ttlMs;
      map.set(key, { value, expiry });
    }

    /**
     * Get with expiry check
     * @param {Map} map - Source Map
     * @param {*} key - Cache key
     * @returns {*} Value or undefined if expired/missing
     */
    function getWithExpiry(map, key) {
      const entry = map.get(key);
      if (!entry) return undefined;
      
      if (Date.now() > entry.expiry) {
        map.delete(key);
        return undefined;
      }
      return entry.value;
    }

    /**
     * Convert object to Map
     * @param {Object} obj - Plain object
     * @returns {Map} Map with string keys
     */
    function objectToMap(obj) {
      const map = new Map();
      Object.entries(obj).forEach(([key, value]) => map.set(key, value));
      return map;
    }
```

```html
    /**
     * Basic GET request with fetch
     * @param {string} url - Request URL
     * @returns {Promise} Resolves to JSON data
     */
    async function fetchGet(url) {
      const response = await fetch(url);
      
      if (!response.ok) {
        throw new Error(`HTTP ${response.status}: ${response.statusText}`);
      }
      
      return response.json();
    }

    /**
     * GET request with custom headers
     * @param {string} url - Request URL
     * @param {Object} headers - Custom headers
     * @returns {Promise} Resolves to JSON data
     */
    async function fetchGetWithHeaders(url, headers = {}) {
      const response = await fetch(url, {
        method: 'GET',
        headers: {
          'Content-Type': 'application/json',
          ...headers
        }
      });
      
      if (!response.ok) {
        throw new Error(`HTTP ${response.status}`);
      }
      
      return response.json();
    }

    /**
     * GET request with query parameters
     * @param {string} baseUrl - Base URL
     * @param {Object} params - Query parameters
     * @returns {Promise} Resolves to JSON data
     */
    async function fetchGetWithParams(baseUrl, params = {}) {
      const url = new URL(baseUrl);
      Object.entries(params).forEach(([key, value]) => {
        url.searchParams.append(key, value);
      });
      
      const response = await fetch(url);
      
      if (!response.ok) {
        throw new Error(`HTTP ${response.status}`);
      }
      
      return response.json();
    }

    /**
     * GET request with timeout
     * @param {string} url - Request URL
     * @param {number} timeoutMs - Timeout in milliseconds
     * @returns {Promise} Resolves to JSON data
     */
    async function fetchGetWithTimeout(url, timeoutMs = 5000) {
      const controller = new AbortController();
      const timeoutId = setTimeout(() => controller.abort(), timeoutMs);
      
      try {
        const response = await fetch(url, {
          signal: controller.signal
        });
        
        if (!response.ok) {
          throw new Error(`HTTP ${response.status}`);
        }
        
        return response.json();
      } finally {
        clearTimeout(timeoutId);
      }
    }

    /**
     * GET request with retry logic
     * @param {string} url - Request URL
     * @param {number} retries - Number of retry attempts
     * @param {number} delayMs - Delay between retries
     * @returns {Promise} Resolves to JSON data
     */
    async function fetchGetWithRetry(url, retries = 3, delayMs = 1000) {
      for (let i = 0; i < retries; i++) {
        try {
          const response = await fetch(url);
          
          if (response.ok) {
            return response.json();
          }
          
          // Don't retry on client errors (4xx)
          if (response.status >= 400 && response.status < 500) {
            throw new Error(`Client error: ${response.status}`);
          }
        } catch (error) {
          if (i === retries - 1) throw error;
          await new Promise(resolve => setTimeout(resolve, delayMs));
        }
      }
    }

    // === Demo ===

      }
```
