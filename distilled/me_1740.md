## Boruvka's Algorithm

```python
# PROBLEM: Find Minimum Spanning Tree (MST) of a weighted undirected graph
# TIME: O(E log V) SPACE: O(V + E)
class UnionFind:
    """Disjoint Set Union for connectivity checks."""
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank = [0] * n
    
    def find(self, x: int) -> int:
        """Find root with path compression."""
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x: int, y: int) -> bool:
        """Union by rank. Returns True if merged."""
        px, py = self.find(x), self.find(y)
        if px == py:
            return False
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        return True

def boruvka_mst(n: int, edges: list[tuple[int, int, int]]) -> list[tuple[int, int, int]]:
    """
    Boruvka's algorithm for MST.
    Each iteration, find minimum edge for each component.
    
    Args:
        n: Number of vertices
        edges: List of (u, v, weight) tuples
    
    Returns:
        List of edges in the MST
    """
    uf = UnionFind(n)
    mst = []
    mst_weight = 0
    
    # Continue until we have n-1 edges or no more progress
    while len(mst) < n - 1:
        # Track minimum edge for each component
        min_edge = {}  # component_root -> (weight, u, v, edge_idx)
        
        for idx, (u, v, w) in enumerate(edges):
            root_u = uf.find(u)
            root_v = uf.find(v)
            
            # Skip if same component
            if root_u == root_v:
                continue
            
            # Update minimum edge for each component
            for root in [root_u, root_v]:
                if root not in min_edge or w < min_edge[root][0]:
                    min_edge[root] = (w, u, v, idx)
        
        # No progress means graph is disconnected
        if not min_edge:
            break
        
        # Add selected edges to MST
        added_edges = set()
        for root, (w, u, v, idx) in min_edge.items():
            if idx not in added_edges and uf.union(u, v):
                mst.append((u, v, w))
                mst_weight += w
                added_edges.add(idx)
    
    return mst if len(mst) == n - 1 else []
```
**Why:** Parallelizable MST algorithm. Each component independently finds its minimum outgoing edge, making it suitable for distributed computing and GPU implementations.

## Euler Path

```python
# PROBLEM: Find a path that visits every edge exactly once (Eulerian path/circuit)
# TIME: O(V + E) SPACE: O(V + E)
from collections import defaultdict, deque

def find_euler_path(n: int, edges: list[tuple[int, int]]) -> list[int] | None:
    """
    Find Eulerian path or circuit using Hierholzer's algorithm.
    
    Eulerian circuit exists if all vertices have even degree.
    Eulerian path exists if exactly 0 or 2 vertices have odd degree.
    
    Args:
        n: Number of vertices
        edges: List of (u, v) undirected edges
    
    Returns:
        List of vertices in Eulerian path, or None if impossible
    """
    # Build adjacency list
    graph = defaultdict(list)
    degree = [0] * n
    
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
        degree[u] += 1
        degree[v] += 1
    
    # Count vertices with odd degree
    odd_vertices = [i for i in range(n) if degree[i] % 2 == 1]
    
    # Check Eulerian path/circuit conditions
    if len(odd_vertices) not in [0, 2]:
        return None
    
    # Start from odd degree vertex (for path) or any vertex (for circuit)
    start = odd_vertices[0] if odd_vertices else 0
    
    # Hierholzer's algorithm
    stack = [start]
    path = []
    
    while stack:
        current = stack[-1]
        if graph[current]:
            # Remove edge and move to neighbor
            neighbor = graph[current].pop()
            graph[neighbor].remove(current)
            stack.append(neighbor)
        else:
            # No more edges, add to path
            path.append(stack.pop())
    
    # Verify we used all edges
    if any(graph.values()):
        return None  # Graph is disconnected
    
    return path[::-1]
```
**Why:** Fundamental graph traversal with applications in DNA sequencing, circuit design, and the classic "Seven Bridges of Königsberg" problem.

## Max Flow Min Cut (Ford-Fulkerson)

```python
# PROBLEM: Find maximum flow from source to sink in a flow network
# TIME: O(E * max_flow) for Ford-Fulkerson, O(V * E^2) for Edmonds-Karp
# SPACE: O(V + E)
from collections import deque

def max_flow_edmonds_karp(
    n: int, 
    capacity: list[list[int]], 
    source: int, 
    sink: int
) -> tuple[int, list[list[int]]]:
    """
    Edmonds-Karp algorithm for maximum flow.
    Uses BFS to find shortest augmenting paths.
    
    Args:
        n: Number of vertices
        capacity: Adjacency matrix of capacities
        source: Source vertex
        sink: Sink vertex
    
    Returns:
        (max_flow, residual_graph) tuple
    """
    # Build residual graph
    residual = [row[:] for row in capacity]
    flow = [[0] * n for _ in range(n)]
    max_flow = 0
    
    def bfs_find_path() -> list[int] | None:
        """BFS to find augmenting path. Returns parent array."""
        parent = [-1] * n
        parent[source] = source
        queue = deque([source])
        
        while queue:
            u = queue.popleft()
            for v in range(n):
                if parent[v] == -1 and residual[u][v] > 0:
                    parent[v] = u
                    if v == sink:
                        return parent
                    queue.append(v)
        return None
    
    # Keep augmenting while path exists
    while True:
        parent = bfs_find_path()
        if not parent:
            break
        
        # Find minimum residual capacity on path
        path_flow = float('inf')
        v = sink
        while v != source:
            u = parent[v]
            path_flow = min(path_flow, residual[u][v])
            v = u
        
        # Update residual capacities
        v = sink
        while v != source:
            u = parent[v]
            residual[u][v] -= path_flow
            residual[v][u] += path_flow
            flow[u][v] += path_flow
            flow[v][u] -= path_flow
            v = u
        
        max_flow += path_flow
    
    return max_flow, residual

def min_cut(n: int, residual: list[list[int]], source: int) -> set[int]:
    """Find min cut from residual graph after max flow."""
    reachable = {source}
    queue = deque([source])
    
    while queue:
        u = queue.popleft()
        for v in range(n):
            if v not in reachable and residual[u][v] > 0:
                reachable.add(v)
                queue.append(v)
    
    return reachable
```
**Why:** Core optimization algorithm with applications in network routing, bipartite matching, image segmentation, and resource allocation.

## Z-Algorithm

```python
# PROBLEM: Find all occurrences of pattern in text in O(n+m) time
# TIME: O(n + m) SPACE: O(n + m)
def z_algorithm(s: str) -> list[int]:
    """
    Compute Z-array where Z[i] is the length of longest substring
    starting from i that matches prefix of s.
    
    Example: "aabxaabcaa" -> [0, 1, 0, 0, 4, 1, 0, 0, 1, 1]
    """
    n = len(s)
    z = [0] * n
    z[0] = n  # By definition, whole string matches itself
    
    left = right = 0
    
    for i in range(1, n):
        # Case 1: i is outside current Z-box
        if i > right:
            left = right = i
            while right < n and s[right - left] == s[right]:
                right += 1
            z[i] = right - left
            right -= 1
        # Case 2: i is inside current Z-box
        else:
            k = i - left
            # If Z[k] doesn't reach box boundary, copy it
            if z[k] < right - i + 1:
                z[i] = z[k]
            else:
                # Otherwise, extend from boundary
                left = i
                while right < n and s[right - left] == s[right]:
                    right += 1
                z[i] = right - left
                right -= 1
    
    return z

def z_search(text: str, pattern: str) -> list[int]:
    """
    Find all occurrences of pattern in text using Z-algorithm.
    
    Returns:
        List of starting indices where pattern occurs
    """
    # Concatenate with special character not in text/pattern
    combined = pattern + "$" + text
    z = z_algorithm(combined)
    
    # Find positions where Z value equals pattern length
    m = len(pattern)
    return [i - m - 1 for i in range(m + 1, len(combined)) if z[i] == m]
```
**Why:** Linear-time string matching alternative to KMP. Simpler to implement and often faster in practice due to better cache locality.

## Chinese Remainder Theorem

```python
# PROBLEM: Solve system of modular congruences: x ≡ a[i] (mod m[i])
# TIME: O(n log M) where M is product of moduli
# SPACE: O(1)
from math import gcd

def extended_gcd(a: int, b: int) -> tuple[int, int, int]:
    """Extended Euclidean algorithm. Returns (gcd, x, y) where ax + by = gcd."""
    if b == 0:
        return a, 1, 0
    g, x, y = extended_gcd(b, a % b)
    return g, y, x - (a // b) * y

def mod_inverse(a: int, m: int) -> int | None:
    """Find modular inverse of a mod m if gcd(a, m) = 1."""
    g, x, _ = extended_gcd(a % m, m)
    if g != 1:
        return None
    return x % m

def chinese_remainder_theorem(
    remainders: list[int], 
    moduli: list[int]
) -> int | None:
    """
    Solve x ≡ remainders[i] (mod moduli[i]) for all i.
    Moduli must be pairwise coprime.
    
    Args:
        remainders: List of residues [a_1, a_2, ...]
        moduli: List of moduli [m_1, m_2, ...]
    
    Returns:
        Unique solution x modulo M (product of all moduli), or None
    """
    if len(remainders) != len(moduli):
        return None
    
    # Verify pairwise coprimality
    for i in range(len(moduli)):
        for j in range(i + 1, len(moduli)):
            if gcd(moduli[i], moduli[j]) != 1:
                return None
    
    # Compute product of all moduli
    M = 1
    for m in moduli:
        M *= m
    
    result = 0
    for a, m in zip(remainders, moduli):
        Mi = M // m  # Product of all other moduli
        yi = mod_inverse(Mi, m)  # Modular inverse
        if yi is None:
            return None
        result += a * Mi * yi
    
    return result % M

# General CRT for non-coprime moduli
def crt_general(remainders: list[int], moduli: list[int]) -> tuple[int, int] | None:
    """
    General CRT that works even when moduli aren't coprime.
    
    Returns:
        (x, lcm) where x is solution mod lcm of all moduli, or None
    """
    if not remainders:
        return None
    
    x, lcm = remainders[0], moduli[0]
    
    for a, m in zip(remainders[1:], moduli[1:]):
        g, p, q = extended_gcd(lcm, m)
        
        # Check if solution exists
        if (a - x) % g != 0:
            return None
        
        # Update solution
        x += (a - x) // g * p % (m // g) * lcm
        lcm = lcm * m // g  # LCM of previous and current modulus
        x %= lcm
    
    return x, lcm
```
**Why:** Essential for modular arithmetic problems in cryptography, competitive programming, and computing with large numbers.

## B+ Tree

```python
# PROBLEM: Implement a disk-efficient ordered dictionary with range queries
# TIME: O(log n) for insert/delete/search, O(log n + k) for range
# SPACE: O(n)
class BPlusTreeNode:
    """Node in B+ Tree. Internal nodes have only keys, leaves store data."""
    
    def __init__(self, is_leaf: bool = False):
        self.keys: list = []
        self.children: list['BPlusTreeNode'] = []  # Internal nodes
        self.values: list = []  # Leaf nodes only
        self.is_leaf = is_leaf
        self.next: 'BPlusTreeNode' | None = None  # Leaf linkage

class BPlusTree:
    """
    B+ Tree implementation for efficient disk storage.
    All data lives in leaf nodes; internal nodes are routing only.
    """
    
    def __init__(self, order: int = 4):
        """
        Initialize B+ tree with given order (max children per node).
        Min children = ceil(order/2), except root.
        """
        self.order = order
        self.root = BPlusTreeNode(is_leaf=True)
        self.min_keys = (order + 1) // 2 - 1
    
    def search(self, key) -> tuple | None:
        """Find value for key, or None if not found."""
        node = self._find_leaf(key)
        for i, k in enumerate(node.keys):
            if k == key:
                return (k, node.values[i])
        return None
    
    def _find_leaf(self, key) -> BPlusTreeNode:
        """Navigate to leaf node where key would exist."""
        node = self.root
        while not node.is_leaf:
            idx = 0
            while idx < len(node.keys) and key >= node.keys[idx]:
                idx += 1
            node = node.children[idx]
        return node
    
    def insert(self, key, value) -> None:
        """Insert key-value pair, splitting nodes as needed."""
        leaf = self._find_leaf(key)
        
        # Find insertion point
        idx = 0
        while idx < len(leaf.keys) and leaf.keys[idx] < key:
            idx += 1
        
        # Handle duplicate keys
        if idx < len(leaf.keys) and leaf.keys[idx] == key:
            leaf.values[idx] = value
            return
        
        leaf.keys.insert(idx, key)
        leaf.values.insert(idx, value)
        
        # Split if overflow
        if len(leaf.keys) >= self.order:
            self._split_leaf(leaf)
    
    def _split_leaf(self, leaf: BPlusTreeNode) -> None:
        """Split leaf node and propagate up."""
        mid = len(leaf.keys) // 2
        
        # Create new leaf with right half
        new_leaf = BPlusTreeNode(is_leaf=True)
        new_leaf.keys = leaf.keys[mid:]
        new_leaf.values = leaf.values[mid:]
        new_leaf.next = leaf.next
        
        # Update original leaf
        leaf.keys = leaf.keys[:mid]
        leaf.values = leaf.values[:mid]
        leaf.next = new_leaf
        
        # Propagate split
        self._insert_parent(leaf, new_leaf.keys[0], new_leaf)
    
    def _insert_parent(
        self, 
        left: BPlusTreeNode, 
        key, 
        right: BPlusTreeNode
    ) -> None:
        """Insert key and right child into parent of left."""
        if left == self.root:
            # Create new root
            new_root = BPlusTreeNode()
            new_root.keys = [key]
            new_root.children = [left, right]
            self.root = new_root
            return
        
        # Find parent (simplified - in practice, store parent pointer)
        parent = self._find_parent(self.root, left)
        
        # Insert key and right child
        idx = parent.children.index(left) + 1
        parent.keys.insert(idx - 1, key)
        parent.children.insert(idx, right)
        
        # Split internal node if overflow
        if len(parent.keys) >= self.order:
            self._split_internal(parent)
    
    def _find_parent(
        self, 
        node: BPlusTreeNode, 
        child: BPlusTreeNode
    ) -> BPlusTreeNode | None:
        """Find parent of child node starting from node."""
        if node.is_leaf:
            return None
        
        for c in node.children:
            if c == child:
                return node
            result = self._find_parent(c, child)
            if result:
                return result
        return None
    
    def range_query(self, start, end) -> list:
        """Get all key-value pairs in [start, end]."""
        results = []
        leaf = self._find_leaf(start)
        
        while leaf:
            for k, v in zip(leaf.keys, leaf.values):
                if start <= k <= end:
                    results.append((k, v))
                elif k > end:
                    return results
            leaf = leaf.next
        
        return results
```
**Why:** Standard index structure in databases (MySQL, PostgreSQL). Efficient range queries, high fanout, and disk-friendly page alignment.

## LFU Cache

```python
# PROBLEM: Evict least frequently used items when cache is full
# TIME: O(1) for get and put operations
# SPACE: O(n)
from collections import defaultdict

class Node:
    """Doubly linked list node for O(1) operations."""
    __slots__ = ['key', 'value', 'freq', 'prev', 'next']
    
    def __init__(self, key=0, value=0):
        self.key = key
        self.value = value
        self.freq = 1
        self.prev = None
        self.next = None

class DLinkedList:
    """Doubly linked list with sentinel nodes."""
    
    def __init__(self):
        self.head = Node()
        self.tail = Node()
        self.head.next = self.tail
        self.tail.prev = self.head
        self.size = 0
    
    def append(self, node: Node) -> None:
        """Add node to front (most recent)."""
        node.prev = self.head
        node.next = self.head.next
        self.head.next.prev = node
        self.head.next = node
        self.size += 1
    
    def remove(self, node: Node) -> None:
        """Remove node from list."""
        node.prev.next = node.next
        node.next.prev = node.prev
        self.size -= 1
    
    def pop(self) -> Node | None:
        """Remove and return least recent node."""
        if self.size == 0:
            return None
        node = self.tail.prev
        self.remove(node)
        return node

class LFUCache:
    """
    Least Frequently Used cache with O(1) operations.
    Uses frequency buckets with doubly linked lists.
    """
    
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.size = 0
        self.min_freq = 0
        self.key_to_node: dict[int, Node] = {}
        self.freq_to_list: dict[int, DLinkedList] = defaultdict(DLinkedList)
    
    def get(self, key: int) -> int:
        """Get value and update frequency."""
        if key not in self.key_to_node:
            return -1
        
        node = self.key_to_node[key]
        self._update_freq(node)
        return node.value
    
    def put(self, key: int, value: int) -> None:
        """Insert or update key-value pair."""
        if self.capacity == 0:
            return
        
        if key in self.key_to_node:
            node = self.key_to_node[key]
            node.value = value
            self._update_freq(node)
            return
        
        # Evict if at capacity
        if self.size == self.capacity:
            # Remove from least frequent list
            lfu_list = self.freq_to_list[self.min_freq]
            node = lfu_list.pop()
            if node:
                del self.key_to_node[node.key]
            self.size -= 1
        
        # Insert new node
        node = Node(key, value)
        self.key_to_node[key] = node
        self.freq_to_list[1].append(node)
        self.min_freq = 1
        self.size += 1
    
    def _update_freq(self, node: Node) -> None:
        """Move node to next frequency bucket."""
        freq = node.freq
        
        # Remove from current list
        self.freq_to_list[freq].remove(node)
        
        # Update min_freq if needed
        if freq == self.min_freq and self.freq_to_list[freq].size == 0:
            self.min_freq += 1
        
        # Add to next frequency list
        node.freq += 1
        self.freq_to_list[node.freq].append(node)
```
**Why:** Better cache eviction than LRU for workloads with hot/cold data patterns. O(1) operations make it practical for high-performance systems.

## Event Sourcing

```python
# PROBLEM: Store state changes as immutable events for audit and replay
# TIME: O(1) append, O(n) replay SPACE: O(e) where e = events
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any, Callable
from abc import ABC, abstractmethod
import json

@dataclass
class Event:
    """Base event with metadata."""
    event_type: str
    aggregate_id: str
    timestamp: datetime = field(default_factory=datetime.now)
    version: int = 1
    data: dict = field(default_factory=dict)

class EventStore:
    """Append-only storage for events."""
    
    def __init__(self):
        self.events: list[Event] = []
        self.subscribers: list[Callable[[Event], None]] = []
    
    def append(self, event: Event) -> None:
        """Store event and notify subscribers."""
        event.version = len(self.events) + 1
        self.events.append(event)
        for subscriber in self.subscribers:
            subscriber(event)
    
    def get_events(
        self, 
        aggregate_id: str | None = None,
        after_version: int = 0
    ) -> list[Event]:
        """Get events, optionally filtered by aggregate and version."""
        result = []
        for event in self.events:
            if event.version <= after_version:
                continue
            if aggregate_id and event.aggregate_id != aggregate_id:
                continue
            result.append(event)
        return result
    
    def subscribe(self, callback: Callable[[Event], None]) -> None:
        """Register event handler."""
        self.subscribers.append(callback)

class Aggregate(ABC):
    """Base class for event-sourced aggregates."""
    
    def __init__(self, aggregate_id: str):
        self.aggregate_id = aggregate_id
        self.version = 0
        self._pending_events: list[Event] = []
    
    @abstractmethod
    def apply(self, event: Event) -> None:
        """Apply event to update state."""
        pass
    
    def load_from_history(self, events: list[Event]) -> None:
        """Rebuild state from event history."""
        for event in events:
            self.apply(event)
            self.version = event.version
    
    def _add_event(self, event: Event) -> None:
        """Create and apply new event."""
        self.apply(event)
        self._pending_events.append(event)

# Example: Bank Account
class BankAccount(Aggregate):
    """Event-sourced bank account."""
    
    def __init__(self, account_id: str):
        super().__init__(account_id)
        self.balance = 0.0
        self.status = "inactive"
    
    def apply(self, event: Event) -> None:
        """Apply events to update account state."""
        if event.event_type == "AccountOpened":
            self.status = "active"
            self.balance = 0.0
        elif event.event_type == "MoneyDeposited":
            self.balance += event.data["amount"]
        elif event.event_type == "MoneyWithdrawn":
            self.balance -= event.data["amount"]
        elif event.event_type == "AccountClosed":
            self.status = "closed"
        self.version = event.version
    
    def open(self) -> None:
        """Open new account."""
        if self.status != "inactive":
            raise ValueError("Account already opened")
        self._add_event(Event(
            event_type="AccountOpened",
            aggregate_id=self.aggregate_id
        ))
    
    def deposit(self, amount: float) -> None:
        """Deposit money."""
        if self.status != "active":
            raise ValueError("Account not active")
        if amount <= 0:
            raise ValueError("Amount must be positive")
        self._add_event(Event(
            event_type="MoneyDeposited",
            aggregate_id=self.aggregate_id,
            data={"amount": amount}
        ))
    
    def withdraw(self, amount: float) -> None:
        """Withdraw money."""
        if self.status != "active":
            raise ValueError("Account not active")
        if amount <= 0:
            raise ValueError("Amount must be positive")
        if self.balance < amount:
            raise ValueError("Insufficient funds")
        self._add_event(Event(
            event_type="MoneyWithdrawn",
            aggregate_id=self.aggregate_id,
            data={"amount": amount}
        ))
    
    def close(self) -> None:
        """Close account."""
        if self.balance != 0:
            raise ValueError("Cannot close account with balance")
        self._add_event(Event(
            event_type="AccountClosed",
            aggregate_id=self.aggregate_id
        ))
```
**Why:** Complete audit trail, time travel debugging, event replay for bug fixes, and natural fit for distributed systems. No lost data.

```python
@dataclass
class SearchUsers(Query):
    name_pattern: str

# Command Handler (Write Model)
class UserRepository:
    """Write model - normalized, transactional."""
    
    def __init__(self):
        self.users: dict[str, dict] = {}
        self.events: list[dict] = []
    
    def create(self, user_id: str, name: str, email: str) -> None:
        if user_id in self.users:
            raise ValueError("User already exists")
        self.users[user_id] = {
            "id": user_id,
            "name": name,
            "email": email,
            "active": True
        }
        self._emit_event("UserCreated", user_id, {"name": name, "email": email})
    
    def update_email(self, user_id: str, new_email: str) -> None:
        if user_id not in self.users:
            raise ValueError("User not found")
        old_email = self.users[user_id]["email"]
        self.users[user_id]["email"] = new_email
        self._emit_event("EmailUpdated", user_id, {"old": old_email, "new": new_email})
    
    def deactivate(self, user_id: str) -> None:
        if user_id not in self.users:
            raise ValueError("User not found")
        self.users[user_id]["active"] = False
        self._emit_event("UserDeactivated", user_id, {})
    
    def _emit_event(self, event_type: str, user_id: str, data: dict) -> None:
        self.events.append({
            "type": event_type,
            "user_id": user_id,
            "data": data
        })

# Query Handler (Read Model / Projection)
class UserProjection:
    """Read model - denormalized, optimized for queries."""
    
    def __init__(self):
        # Optimized for different query patterns
        self.by_id: dict[str, dict] = {}
        self.active_users: list[dict] = []
        self.name_index: dict[str, list[str]] = defaultdict(list)
    
    def apply_event(self, event: dict) -> None:
        """Update projections based on events."""
        event_type = event["type"]
        user_id = event["user_id"]
        data = event["data"]
        
        if event_type == "UserCreated":
            user = {"id": user_id, **data, "active": True}
            self.by_id[user_id] = user
            self.active_users.append(user)
            self.name_index[data["name"].lower()].append(user_id)
        
        elif event_type == "EmailUpdated":
            if user_id in self.by_id:
                self.by_id[user_id]["email"] = data["new"]
        
        elif event_type == "UserDeactivated":
            if user_id in self.by_id:
                self.by_id[user_id]["active"] = False
                self.active_users = [
                    u for u in self.active_users if u["id"] != user_id
                ]

# CQRS Bus
class CQRSBus:
    """Dispatcher for commands and queries."""
    
    def __init__(self):
        self.command_handlers: dict[type, Callable] = {}
        self.query_handlers: dict[type, Callable] = {}
        self.write_model = UserRepository()
        self.read_model = UserProjection()
        self._sync_projections()
    
    def _sync_projections(self) -> None:
        """Sync read model with write model events."""
        for event in self.write_model.events:
            self.read_model.apply_event(event)
    
    def register_command(
        self, 
        command_type: type, 
        handler: Callable[[Command], None]
    ) -> None:
        self.command_handlers[command_type] = handler
    
    def register_query(
        self, 
        query_type: type, 
        handler: Callable[[Query], Any]
    ) -> None:
        self.query_handlers[query_type] = handler
    
    def execute(self, command: Command) -> None:
        """Execute command (writes)."""
        handler = self.command_handlers.get(type(command))
        if not handler:
            raise ValueError(f"No handler for {type(command)}")
        handler(command)
        self._sync_projections()
    
    def query(self, q: Query) -> Any:
        """Execute query (reads)."""
        handler = self.query_handlers.get(type(q))
        if not handler:
            raise ValueError(f"No handler for {type(q)}")
        return handler(q)

# Setup and usage
bus = CQRSBus()

# Register command handlers
bus.register_command(CreateUser, lambda c: 
    bus.write_model.create(c.user_id, c.name, c.email))
bus.register_command(UpdateEmail, lambda c: 
    bus.write_model.update_email(c.user_id, c.new_email))
bus.register_command(DeactivateUser, lambda c: 
    bus.write_model.deactivate(c.user_id))

# Register query handlers
bus.register_query(GetUserById, lambda q: 
    bus.read_model.by_id.get(q.user_id))
bus.register_query(GetActiveUsers, lambda q: 
    bus.read_model.active_users)
bus.register_query(SearchUsers, lambda q: [
    bus.read_model.by_id[uid] 
    for name, uids in bus.read_model.name_index.items()
    if q.name_pattern.lower() in name
    for uid in uids
])
```
**Why:** Scale reads independently from writes. Use different data models optimized for each. Natural with event sourcing for eventual consistency.

## Rate Limiting Patterns

```python
# PROBLEM: Control request rate to prevent abuse and ensure fair usage
# TIME: O(1) for all algorithms SPACE: O(n) for user state
import time
from collections import deque
from dataclasses import dataclass, field
from threading import Lock
from typing import Callable

@dataclass
class RateLimitResult:
    allowed: bool
    remaining: int
    reset_time: float  # Unix timestamp
    retry_after: float | None = None

# Fixed Window Counter
class FixedWindowRateLimiter:
    """
    Simple but has edge case: allows 2x rate at window boundary.
    If limit is 10/min, can do 10 at 0:59 and 10 at 1:01.
    """
    
    def __init__(self, limit: int, window_seconds: float):
        self.limit = limit
        self.window = window_seconds
        self.counters: dict[str, tuple[int, float]] = {}
        self.lock = Lock()
    
    def check(self, key: str) -> RateLimitResult:
        with self.lock:
            now = time.time()
            count, window_start = self.counters.get(key, (0, now))
            
            # Reset if window expired
            if now - window_start >= self.window:
                count = 0
                window_start = now
            
            reset_time = window_start + self.window
            
            if count >= self.limit:
                return RateLimitResult(
                    allowed=False,
                    remaining=0,
                    reset_time=reset_time,
                    retry_after=reset_time - now
                )
            
            self.counters[key] = (count + 1, window_start)
            return RateLimitResult(
                allowed=True,
                remaining=self.limit - count - 1,
                reset_time=reset_time
            )

# Sliding Window Log
class SlidingWindowRateLimiter:
    """
    Precise but memory-intensive. Stores all request timestamps.
    """
    
    def __init__(self, limit: int, window_seconds: float):
        self.limit = limit
        self.window = window_seconds
        self.requests: dict[str, deque] = {}
        self.lock = Lock()
    
    def check(self, key: str) -> RateLimitResult:
        with self.lock:
            now = time.time()
            cutoff = now - self.window
            
            # Get or create request queue
            if key not in self.requests:
                self.requests[key] = deque()
            requests = self.requests[key]
            
            # Remove expired requests
            while requests and requests[0] < cutoff:
                requests.popleft()
            
            # Check limit
            if len(requests) >= self.limit:
                oldest = requests[0]
                retry_after = oldest + self.window - now
                return RateLimitResult(
                    allowed=False,
                    remaining=0,
                    reset_time=oldest + self.window,
                    retry_after=retry_after
                )
            
            requests.append(now)
            return RateLimitResult(
                allowed=True,
                remaining=self.limit - len(requests),
                reset_time=now + self.window
            )

# Token Bucket
class TokenBucketRateLimiter:
    """
    Allows burst up to bucket size. Good for API rate limiting.
    Refills continuously at specified rate.
    """
    
    def __init__(self, capacity: int, refill_rate: float):
        """
        Args:
            capacity: Maximum tokens (burst size)
            refill_rate: Tokens added per second
        """
        self.capacity = capacity
        self.refill_rate = refill_rate
        self.buckets: dict[str, tuple[float, float]] = {}  # (tokens, last_update)
        self.lock = Lock()
    
    def check(self, key: str, tokens_needed: int = 1) -> RateLimitResult:
        with self.lock:
            now = time.time()
            tokens, last_update = self.buckets.get(key, (self.capacity, now))
            
            # Refill tokens
            elapsed = now - last_update
            tokens = min(self.capacity, tokens + elapsed * self.refill_rate)
            
            if tokens < tokens_needed:
                # Calculate time until enough tokens
                deficit = tokens_needed - tokens
                retry_after = deficit / self.refill_rate
                return RateLimitResult(
                    allowed=False,
                    remaining=int(tokens),
                    reset_time=now + retry_after,
                    retry_after=retry_after
                )
            
            tokens -= tokens_needed
            self.buckets[key] = (tokens, now)
            
            return RateLimitResult(
                allowed=True,
                remaining=int(tokens),
                reset_time=now + (self.capacity - tokens) / self.refill_rate
            )

# Leaky Bucket
class LeakyBucketRateLimiter:
    """
    Processes requests at constant rate. Good for smoothing traffic.
    Queue fills up, drips at constant rate.
    """
    
    def __init__(self, capacity: int, drip_rate: float):
        """
        Args:
            capacity: Maximum queue size
            drip_rate: Requests processed per second
        """
        self.capacity = capacity
        self.drip_rate = drip_rate
        self.buckets: dict[str, tuple[int, float]] = {}  # (water, last_drip)
        self.lock = Lock()
    
    def check(self, key: str) -> RateLimitResult:
        with self.lock:
            now = time.time()
            water, last_drip = self.buckets.get(key, (0, now))
            
            # Drip (process) requests
            elapsed = now - last_drip
            water = max(0, water - elapsed * self.drip_rate)
            
            if water >= self.capacity:
                retry_after = (water - self.capacity + 1) / self.drip_rate
                return RateLimitResult(
                    allowed=False,
                    remaining=0,
                    reset_time=now + water / self.drip_rate,
                    retry_after=retry_after
                )
            
            # Add request to bucket
            water += 1
            self.buckets[key] = (water, now)
            
            return RateLimitResult(
                allowed=True,
                remaining=self.capacity - int(water),
                reset_time=now + water / self.drip_rate
            )

# Decorator for easy use
def rate_limit(
    limiter: TokenBucketRateLimiter, 
    key_func: Callable = None
):
    """Decorator to rate limit a function."""
    def decorator(func):
        def wrapper(*args, **kwargs):
            key = key_func(*args, **kwargs) if key_func else "default"
            result = limiter.check(key)
            if not result.allowed:
                raise Exception(f"Rate limit exceeded. Retry after {result.retry_after:.2f}s")
            return func(*args, **kwargs)
        return wrapper
    return decorator
```
**Why:** Essential for API protection. Token bucket for bursts, leaky bucket for smoothing, sliding window for precision. Choose based on use case.
