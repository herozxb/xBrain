## 1. Maximum Subarray Sum (Sliding Window)
```python
# PROBLEM: Find the maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)
def max_subarray_sum(arr, k):
    """
    Find maximum sum of any subarray of length k using sliding window.
    Instead of recalculating sum for each window, slide by removing
    the element leaving and adding the element entering.
    """
    if len(arr) < k:
        return None
    
    # Calculate sum of first window
    window_sum = sum(arr[:k])
    max_sum = window_sum
    
    # Slide window through array
    for i in range(len(arr) - k):
        # Remove leftmost element, add new rightmost element
        window_sum = window_sum - arr[i] + arr[i + k]
        max_sum = max(max_sum, window_sum)
    
    return max_sum
```
**Why**: Classic sliding window pattern. Avoids O(n*k) brute force by reusing previous computation. Window slides in O(1) per step.

## 2. Two Sum II (Two Pointers)
```python
# PROBLEM: Find two numbers in sorted array that sum to target
# TIME: O(n) SPACE: O(1)
def two_sum_sorted(arr, target):
    """
    Use two pointers from both ends of sorted array.
    If sum too small, move left pointer right (increase).
    If sum too large, move right pointer left (decrease).
    Only works because array is sorted.
    """
    left, right = 0, len(arr) - 1
    
    while left < right:
        current_sum = arr[left] + arr[right]
        
        if current_sum == target:
            return [left, right]
        elif current_sum < target:
            left += 1  # Need larger sum
        else:
            right -= 1  # Need smaller sum
    
    return None  # No solution
```
**Why**: Two pointers exploit sorted property to achieve O(n) without extra space. Fundamental technique for pair problems in sorted arrays.

## 3. N-Queens (Backtracking)
```python
# PROBLEM: Place N queens on NxN board so no two attack each other
# TIME: O(N!) SPACE: O(N)
def solve_n_queens(n):
    """
    Backtracking: place queens row by row, backtrack when conflict.
    Use sets to track attacked columns and diagonals for O(1) checks.
    """
    def backtrack(row, cols, diag1, diag2, board):
        if row == n:
            solutions.append([''.join(r) for r in board])
            return
        
        for col in range(n):
            # Check if position is safe
            d1 = row - col  # Diagonal \
            d2 = row + col  # Diagonal /
            
            if col in cols or d1 in diag1 or d2 in diag2:
                continue  # Skip attacked positions
            
            # Place queen
            board[row][col] = 'Q'
            cols.add(col)
            diag1.add(d1)
            diag2.add(d2)
            
            # Recurse to next row
            backtrack(row + 1, cols, diag1, diag2, board)
            
            # Backtrack: remove queen
            board[row][col] = '.'
            cols.remove(col)
            diag1.remove(d1)
            diag2.remove(d2)
    
    solutions = []
    board = [['.' for _ in range(n)] for _ in range(n)]
    backtrack(0, set(), set(), set(), board)
    return solutions
```
**Why**: Classic backtracking problem. Demonstrates state exploration with pruning. Set-based conflict detection makes validity check O(1).

Topics: backtracking

## 4. Number of Connected Components (Union-Find)
```python
# PROBLEM: Count connected components in undirected graph
# TIME: O(E * α(V)) SPACE: O(V), where α is inverse Ackermann
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.components = n
    
    def find(self, x):
        """Find root with path compression."""
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        """Union by rank. Returns True if merged."""
        px, py = self.find(x), self.find(y)
        if px == py:
            return False  # Already connected
        
        # Union by rank
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        self.components -= 1
        return True

def count_components(n, edges):
    """
    Use Union-Find to merge connected nodes.
    Final component count is answer.
    """
    uf = UnionFind(n)
    for u, v in edges:
        uf.union(u, v)
    return uf.components
```
**Why**: Union-Find is optimal for dynamic connectivity. Path compression and union by rank achieve near-constant amortized time.

Topics: union_find

## 5. Longest Increasing Subsequence (DP)
```python
# PROBLEM: Find length of longest strictly increasing subsequence
# TIME: O(n log n) SPACE: O(n)
def length_of_lis(nums):
    """
    Patience sorting approach: maintain piles where each pile's
    top card is the smallest possible end of an increasing
    subsequence of that length. Binary search for placement.
    """
    if not nums:
        return 0
    
    # tails[i] = smallest tail of LIS of length i+1
    tails = []
    
    for num in nums:
        # Binary search for first tail >= num
        left, right = 0, len(tails)
        while left < right:
            mid = (left + right) // 2
            if tails[mid] < num:
                left = mid + 1
            else:
                right = mid
        
        # Extend or replace
        if left == len(tails):
            tails.append(num)
        else:
            tails[left] = num
    
    return len(tails)
```
**Why**: Clever DP optimization using binary search. Standard DP is O(n²), but patience sorting reduces to O(n log n).

## 6. Dijkstra's Shortest Path (Graphs + Heap)
```python
# PROBLEM: Find shortest path distances from source to all nodes
# TIME: O((V + E) log V) SPACE: O(V)
import heapq

def dijkstra(graph, start):
    """
    Greedy BFS with min-heap. Always process closest unvisited node.
    Graph format: {node: [(neighbor, weight), ...]}
    """
    # Initialize distances to infinity
    dist = {node: float('inf') for node in graph}
    dist[start] = 0
    
    # Priority queue: (distance, node)
    pq = [(0, start)]
    visited = set()
    
    while pq:
        d, node = heapq.heappop(pq)
        
        if node in visited:
            continue
        visited.add(node)
        
        # Relax all edges
        for neighbor, weight in graph[node]:
            if neighbor in visited:
                continue
            new_dist = d + weight
            if new_dist < dist[neighbor]:
                dist[neighbor] = new_dist
                heapq.heappush(pq, (new_dist, neighbor))
    
    return dist
```
**Why**: Fundamental graph algorithm. Heap ensures we always process the closest node next. Only works for non-negative weights.

Topics: graphs, heap

## 7. Serialize and Deserialize Binary Tree (Trees)
```python
# PROBLEM: Convert binary tree to string and back
# TIME: O(n) SPACE: O(n)
from collections import deque

def serialize(root):
    """
    Level-order serialization with 'null' markers.
    Format: "1,2,3,null,null,4,5"
    """
    if not root:
        return "null"
    
    result = []
    queue = deque([root])
    
    while queue:
        node = queue.popleft()
        if node:
            result.append(str(node.val))
            queue.append(node.left)
            queue.append(node.right)
        else:
            result.append("null")
    
    # Remove trailing nulls
    while result and result[-1] == "null":
        result.pop()
    
    return ",".join(result)

def deserialize(data):
    """Reconstruct tree from serialized string."""
    if data == "null":
        return None
    
    values = data.split(",")
    root = TreeNode(int(values[0]))
    queue = deque([root])
    i = 1
    
    while queue and i < len(values):
        node = queue.popleft()
        
        # Left child
        if i < len(values) and values[i] != "null":
            node.left = TreeNode(int(values[i]))
            queue.append(node.left)
        i += 1
        
        # Right child
        if i < len(values) and values[i] != "null":
            node.right = TreeNode(int(values[i]))
            queue.append(node.right)
        i += 1
    
    return root

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right
```
**Why**: Complete tree serialization. BFS preserves structure including null positions. Essential for tree storage/transmission.

Topics: trees

## 8. Kth Largest Element (Heap)
```python
# PROBLEM: Find kth largest element in unsorted array
# TIME: O(n log k) SPACE: O(k)
import heapq

def find_kth_largest(nums, k):
    """
    Maintain min-heap of size k containing k largest elements.
    Root is the kth largest. Smaller elements are naturally evicted.
    """
    # Min-heap of size k
    heap = []
    
    for num in nums:
        heapq.heappush(heap, num)
        if len(heap) > k:
            heapq.heappop(heap)  # Remove smallest
    
    return heap[0]  # Root is kth largest
```
**Why**: Heap pattern for top-k problems. O(n log k) beats sorting O(n log n) when k << n. Alternative: QuickSelect for O(n) average.

Topics: heap

## 9. Implement Trie (Prefix Tree)
```python
# PROBLEM: Implement insert, search, and startsWith for a trie
# TIME: O(m) per operation where m = word length SPACE: O(alphabet * nodes)
class TrieNode:
    def __init__(self):
        self.children = {}  # char -> TrieNode
        self.is_end = False

class Trie:
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word):
        """Insert word into trie."""
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
    
    def search(self, word):
        """Check if exact word exists."""
        node = self.root
        for char in word:
            if char not in node.children:
                return False
            node = node.children[char]
        return node.is_end
    
    def starts_with(self, prefix):
        """Check if any word starts with prefix."""
        node = self.root
        for char in prefix:
            if char not in node.children:
                return False
            node = node.children[char]
        return True
```
**Why**: Efficient prefix matching. Each node represents a character path. O(m) regardless of total words stored. Essential for autocomplete.

Topics: trie

## 10. Word Break (DP + LRU Cache)
```python
# PROBLEM: Check if string can be segmented into dictionary words
# TIME: O(n²) with memoization SPACE: O(n)
from functools import lru_cache

def word_break(s, word_dict):
    """
    Top-down DP with memoization. Try each prefix, recursively check suffix.
    LRU cache prevents redundant subproblem computation.
    """
    word_set = set(word_dict)
    
    @lru_cache(maxsize=None)
    def can_break(start):
        # Base case: reached end successfully
        if start == len(s):
            return True
        
        # Try all possible end positions
        for end in range(start + 1, len(s) + 1):
            prefix = s[start:end]
            if prefix in word_set and can_break(end):
                return True
        
        return False
    
    return can_break(0)
```
**Why**: Memoization turns exponential brute force into polynomial. @lru_cache decorator automatically caches function results. Clean top-down DP.

Topics: dp, lru_cache
