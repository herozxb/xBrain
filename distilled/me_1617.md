## Tim Sort

```python
# PROBLEM: Hybrid stable sorting algorithm combining merge sort and insertion sort
# TIME: O(n log n) worst/average, O(n) best (already sorted) SPACE: O(n)

MIN_RUN = 32

def insertion_sort(arr, left, right):
    """Sort a small subarray using insertion sort.
    
    Args:
        arr: The array to sort
        left: Starting index (inclusive)
        right: Ending index (inclusive)
    """
    for i in range(left + 1, right + 1):
        key = arr[i]
        j = i - 1
        # Shift elements greater than key to the right
        while j >= left and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key


def merge(arr, left, mid, right):
    """Merge two sorted subarrays.
    
    Args:
        arr: Array containing both subarrays
        left: Start of first subarray
        mid: End of first subarray (mid+1 starts second)
        right: End of second subarray
    """
    left_arr = arr[left:mid + 1]
    right_arr = arr[mid + 1:right + 1]
    
    i = j = 0
    k = left
    
    # Merge in sorted order
    while i < len(left_arr) and j < len(right_arr):
        if left_arr[i] <= right_arr[j]:
            arr[k] = left_arr[i]
            i += 1
        else:
            arr[k] = right_arr[j]
            j += 1
        k += 1
    
    # Copy remaining elements
    while i < len(left_arr):
        arr[k] = left_arr[i]
        i += 1
        k += 1
    while j < len(right_arr):
        arr[k] = right_arr[j]
        j += 1
        k += 1


def calc_min_run(n):
    """Calculate minimum run size for Tim sort.
    
    Returns a value in range [32, 64] that divides n evenly if possible.
    """
    r = 0
    while n >= MIN_RUN:
        r |= n & 1
        n >>= 1
    return n + r


def tim_sort(arr):
    """Sort array using Tim sort algorithm.
    
    Tim sort is Python's built-in sorting algorithm. It uses:
    1. Insertion sort for small runs (<= 32 elements)
    2. Merge sort to combine runs
    
    Args:
        arr: List to sort in-place
        
    Returns:
        The sorted list
    """
    n = len(arr)
    if n < 2:
        return arr
    
    min_run = calc_min_run(n)
    
    # Sort individual runs using insertion sort
    for start in range(0, n, min_run):
        end = min(start + min_run - 1, n - 1)
        insertion_sort(arr, start, end)
    
    # Merge runs of increasing size
    size = min_run
    while size < n:
        for left in range(0, n, 2 * size):
            mid = min(left + size - 1, n - 1)
            right = min(left + 2 * size - 1, n - 1)
            if mid < right:
                merge(arr, left, mid, right)
        size *= 2
    
    return arr
```

**Why**: Tim sort is the default sorting algorithm in Python and Java. It's a practical hybrid that achieves O(n) on sorted data while maintaining O(n log n) worst-case. Understanding it helps write better performance-critical code.

## Edmonds-Karp Algorithm

```python
# PROBLEM: Find maximum flow in a flow network using BFS
# TIME: O(V * E^2) SPACE: O(V^2)

from collections import deque

def edmonds_karp(capacity, source, sink):
    """Find maximum flow from source to sink using Edmonds-Karp.
    
    Edmonds-Karp is the BFS implementation of Ford-Fulkerson.
    It always finds the shortest augmenting path in terms of edges.
    
    Args:
        capacity: 2D list where capacity[u][v] is edge capacity
        source: Source vertex index
        sink: Sink vertex index
        
    Returns:
        Maximum flow value
    """
    n = len(capacity)
    # Residual capacity matrix
    residual = [row[:] for row in capacity]
    max_flow = 0
    
    def bfs():
        """Find augmenting path using BFS.
        
        Returns:
            Tuple of (parent dict, flow value) or (None, 0) if no path
        """
        parent = [-1] * n
        parent[source] = source
        queue = deque([(source, float('inf'))])
        
        while queue:
            u, flow = queue.popleft()
            
            for v in range(n):
                # Check if edge has residual capacity and v not visited
                if parent[v] == -1 and residual[u][v] > 0:
                    parent[v] = u
                    # Minimum flow along the path
                    new_flow = min(flow, residual[u][v])
                    
                    if v == sink:
                        return parent, new_flow
                    
                    queue.append((v, new_flow))
        
        return None, 0
    
    # Repeatedly find augmenting paths
    while True:
        parent, path_flow = bfs()
        
        if path_flow == 0:
            break  # No more augmenting paths
        
        max_flow += path_flow
        
        # Update residual capacities along the path
        v = sink
        while v != source:
            u = parent[v]
            residual[u][v] -= path_flow  # Forward edge
            residual[v][u] += path_flow  # Backward edge
            v = u
    
    return max_flow


# Example: Bipartite matching as flow problem
def max_bipartite_matching(graph, left_size, right_size):
    """Find maximum bipartite matching using max flow.
    
    Args:
        graph: Adjacency list of bipartite graph
        left_size: Number of vertices in left partition
        right_size: Number of vertices in right partition
        
    Returns:
        Maximum matching size
    """
    n = left_size + right_size + 2  # +2 for source and sink
    source = left_size + right_size
    sink = source + 1
    
    # Build capacity matrix
    capacity = [[0] * n for _ in range(n)]
    
    # Source to left partition
    for i in range(left_size):
        capacity[source][i] = 1
    
    # Left to right edges
    for u in range(left_size):
        for v in graph[u]:
            capacity[u][v + left_size] = 1
    
    # Right partition to sink
    for i in range(left_size, left_size + right_size):
        capacity[i][sink] = 1
    
    return edmonds_karp(capacity, source, sink)
```

**Why**: Maximum flow solves many real problems: bipartite matching, network routing, image segmentation. Edmonds-Karp guarantees polynomial time unlike naive Ford-Fulkerson.

## Boyer-Moore String Matching

```python
# PROBLEM: Find pattern occurrences in text using bad character heuristic
# TIME: O(n/m) best, O(n+m) worst SPACE: O(k) where k is alphabet size

def boyer_moore(text, pattern):
    """Find all occurrences of pattern in text using Boyer-Moore.
    
    Boyer-Moore skips sections of text by matching from right to left
    and using the bad character heuristic for larger skips.
    
    Args:
        text: String to search in
        pattern: String to search for
        
    Returns:
        List of starting indices where pattern occurs
    """
    n, m = len(text), len(pattern)
    if m == 0 or m > n:
        return []
    
    # Preprocess: build bad character table
    # bad_char[c] = rightmost position of c in pattern (or -1 if not present)
    bad_char = {}
    for i in range(m):
        bad_char[pattern[i]] = i
    
    matches = []
    shift = 0  # Current alignment of pattern with text
    
    while shift <= n - m:
        j = m - 1  # Start matching from rightmost character
        
        # Match from right to left
        while j >= 0 and pattern[j] == text[shift + j]:
            j -= 1
        
        if j < 0:
            # Pattern found!
            matches.append(shift)
            
            # Shift pattern to find next occurrence
            # Use character after current alignment
            if shift + m < n:
                next_char = text[shift + m]
                skip = m - bad_char.get(next_char, -1)
            else:
                skip = 1
            shift += max(1, skip)
        else:
            # Mismatch at position j
            # Calculate shift using bad character heuristic
            mismatched = text[shift + j]
            last_pos = bad_char.get(mismatched, -1)
            
            # Shift so mismatched character in text aligns with
            # its rightmost occurrence in pattern (or skip past it)
            bad_char_shift = j - last_pos
            
            # Always shift at least 1
            shift += max(1, bad_char_shift)
    
    return matches


def boyer_moore_full(text, pattern):
    """Boyer-Moore with both bad character and good suffix heuristics.
    
    The good suffix heuristic handles cases where a matched suffix
    appears elsewhere in the pattern.
    
    Args:
        text: String to search in
        pattern: String to search for
        
    Returns:
        List of starting indices
    """
    n, m = len(text), len(pattern)
    if m == 0 or m > n:
        return []
    
    # Bad character table
    bad_char = {}
    for i in range(m):
        bad_char[pattern[i]] = i
    
    # Good suffix table
    # suffix[i] = rightmost position where pattern[i+1:m] appears
    # as a prefix of the pattern
    suffix = [0] * m
    last_prefix = m
    for i in range(m - 1, -1, -1):
        # Check if pattern[i+1:] is a prefix of pattern
        if pattern[i+1:] == pattern[:m-i-1]:
            last_prefix = i + 1
        suffix[i] = last_prefix
    
    matches = []
    shift = 0
    
    while shift <= n - m:
        j = m - 1
        
        while j >= 0 and pattern[j] == text[shift + j]:
            j -= 1
        
        if j < 0:
            matches.append(shift)
            shift += suffix[0]
        else:
            # Bad character shift
            bc_shift = max(1, j - bad_char.get(text[shift + j], -1))
            # Good suffix shift
            gs_shift = suffix[j]
            shift += max(bc_shift, gs_shift)
    
    return matches
```

**Why**: Boyer-Moore is the standard algorithm for grep and text editors. It can skip ahead by m characters per comparison, achieving sublinear time in practice.

## Composite Pattern

```python
# PROBLEM: Represent part-whole hierarchies as tree structures
# TIME: O(n) for traversal SPACE: O(d) recursion depth where d is tree depth

from abc import ABC, abstractmethod
from typing import List

# Component interface
class FileSystemNode(ABC):
    """Abstract base class for all file system nodes."""
    
    @abstractmethod
    def get_size(self) -> int:
        """Return size in bytes."""
        pass
    
    @abstractmethod
    def print_structure(self, indent: int = 0) -> None:
        """Print the structure with indentation."""
        pass
    
    @abstractmethod
    def find_by_name(self, name: str) -> 'FileSystemNode':
        """Find node by name. Returns None if not found."""
        pass


# Leaf class
class File(FileSystemNode):
    """Represents a file (leaf node)."""
    
    def __init__(self, name: str, size: int):
        self.name = name
        self._size = size
    
    def get_size(self) -> int:
        return self._size
    
    def print_structure(self, indent: int = 0) -> None:
        print(f"{'  ' * indent}📄 {self.name} ({self._size} bytes)")
    
    def find_by_name(self, name: str) -> FileSystemNode:
        return self if self.name == name else None


# Composite class
class Directory(FileSystemNode):
    """Represents a directory (composite node that can contain other nodes)."""
    
    def __init__(self, name: str):
        self.name = name
        self._children: List[FileSystemNode] = []
    
    def add(self, node: FileSystemNode) -> None:
        """Add a child node."""
        self._children.append(node)
    
    def remove(self, node: FileSystemNode) -> None:
        """Remove a child node."""
        self._children.remove(node)
    
    def get_size(self) -> int:
        """Sum of all children sizes."""
        return sum(child.get_size() for child in self._children)
    
    def print_structure(self, indent: int = 0) -> None:
        print(f"{'  ' * indent}📁 {self.name}/ ({self.get_size()} bytes total)")
        for child in self._children:
            child.print_structure(indent + 1)
    
    def find_by_name(self, name: str) -> FileSystemNode:
        if self.name == name:
            return self
        
        for child in self._children:
            result = child.find_by_name(name)
            if result:
                return result
        return None
    
    def get_files_by_extension(self, ext: str) -> List[File]:
        """Find all files with given extension."""
        files = []
        for child in self._children:
            if isinstance(child, File):
                if child.name.endswith(ext):
                    files.append(child)
            elif isinstance(child, Directory):
                files.extend(child.get_files_by_extension(ext))
        return files


# Usage example
def build_filesystem():
    """Create a sample file system structure."""
    root = Directory("root")
    
    src = Directory("src")
    src.add(File("main.py", 2048))
    src.add(File("utils.py", 1024))
    
    tests = Directory("tests")
    tests.add(File("test_main.py", 512))
    tests.add(File("test_utils.py", 384))
    
    docs = Directory("docs")
    docs.add(File("readme.md", 2560))
    
    root.add(src)
    root.add(tests)
    root.add(docs)
    root.add(File("config.json", 128))
    
    return root


if __name__ == "__main__":
    fs = build_filesystem()
    fs.print_structure()
    # Output:
    # 📁 root/ (6656 bytes total)
    #   📁 src/ (3072 bytes total)
    #     📄 main.py (2048 bytes)
    #     📄 utils.py (1024 bytes)
    #   📁 tests/ (896 bytes total)
    #     📄 test_main.py (512 bytes)
    #     📄 test_utils.py (384 bytes)
    #   📁 docs/ (2560 bytes total)
    #     📄 readme.md (2560 bytes)
    #   📄 config.json (128 bytes)
```

**Why**: Composite pattern lets clients treat individual objects and compositions uniformly. Essential for UI frameworks (widgets in containers), file systems, and organizational hierarchies.

## Bloom Filter

```python
# PROBLEM: Space-efficient probabilistic set membership test
# TIME: O(k) for add/contains where k is number of hash functions
# SPACE: O(m) bits where m depends on desired false positive rate

import hashlib
from bitarray import bitarray  # pip install bitarray

class BloomFilter:
    """Probabilistic data structure for set membership.
    
    Properties:
    - No false negatives: if contains() returns False, element is definitely not present
    - Possible false positives: if contains() returns True, element might not be present
    - Very space efficient: uses fixed space regardless of number of elements
    """
    
    def __init__(self, expected_items: int, false_positive_rate: float = 0.01):
        """Initialize bloom filter with optimal parameters.
        
        Args:
            expected_items: Expected number of items to store
            false_positive_rate: Desired false positive probability (0 < p < 1)
        """
        import math
        
        self.n = expected_items
        self.p = false_positive_rate
        
        # Calculate optimal bit array size: m = -n*ln(p) / (ln(2)^2)
        self.m = int(-self.n * math.log(self.p) / (math.log(2) ** 2))
        
        # Calculate optimal number of hash functions: k = (m/n) * ln(2)
        self.k = int((self.m / self.n) * math.log(2))
        
        # Initialize bit array
        self.bits = bitarray(self.m)
        self.bits.setall(False)
        
        self._count = 0  # Approximate count
    
    def _hashes(self, item) -> list:
        """Generate k hash values for an item using double hashing.
        
        Uses formula: h(i) = (hash1 + i * hash2) % m
        This technique only needs 2 hash functions to generate k values.
        """
        item_bytes = str(item).encode('utf-8')
        
        # Use MD5 and SHA1 as the two base hashes
        hash1 = int(hashlib.md5(item_bytes).hexdigest(), 16)
        hash2 = int(hashlib.sha1(item_bytes).hexdigest(), 16)
        
        return [(hash1 + i * hash2) % self.m for i in range(self.k)]
    
    def add(self, item) -> None:
        """Add an item to the bloom filter."""
        for pos in self._hashes(item):
            self.bits[pos] = True
        self._count += 1
    
    def contains(self, item) -> bool:
        """Check if item might be in the set.
        
        Returns:
            True if item might be in set (may be false positive)
            False if item is definitely not in set
        """
        return all(self.bits[pos] for pos in self._hashes(item))
    
    def __contains__(self, item) -> bool:
        """Allow 'in' operator syntax."""
        return self.contains(item)
    
    def current_false_positive_rate(self) -> float:
        """Calculate current false positive rate based on fill level."""
        import math
        # p = (1 - e^(-kn/m))^k
        bits_set = self.bits.count(True)
        fill_ratio = bits_set / self.m
        return (1 - (1 - fill_ratio) ** self.k) ** self.k
    
    @property
    def fill_ratio(self) -> float:
        """Return the fraction of bits set to 1."""
        return self.bits.count(True) / self.m


class CountingBloomFilter(BloomFilter):
    """Bloom filter variant that supports removal.
    
    Uses counters instead of bits, allowing decrement on removal.
    Uses 4x more space than standard bloom filter.
    """
    
    def __init__(self, expected_items: int, false_positive_rate: float = 0.01):
        super().__init__(expected_items, false_positive_rate)
        # Replace bitarray with counter array
        self.counters = [0] * self.m
        self.bits = None  # Not using bitarray
    
    def add(self, item) -> None:
        for pos in self._hashes(item):
            self.counters[pos] += 1
        self._count += 1
    
    def remove(self, item) -> bool:
        """Remove an item. Returns False if item wasn't present."""
        if not self.contains(item):
            return False
        
        for pos in self._hashes(item):
            self.counters[pos] = max(0, self.counters[pos] - 1)
        self._count -= 1
        return True
    
    def contains(self, item) -> bool:
        return all(self.counters[pos] > 0 for pos in self._hashes(item))


# Usage example
def demo_bloom_filter():
    # Create filter for 1000 items with 1% false positive rate
    bf = BloomFilter(expected_items=1000, false_positive_rate=0.01)
    
    # Add items
    usernames = ["alice", "bob", "charlie", "diana"]
    for name in usernames:
        bf.add(name)
    
    # Check membership
    print(f"alice in filter: {'alice' in bf}")  # True
    print(f"eve in filter: {'eve' in bf}")      # Probably False
    print(f"Fill ratio: {bf.fill_ratio:.2%}")
    print(f"Current FP rate: {bf.current_false_positive_rate():.4f}")


if __name__ == "__main__":
    demo_bloom_filter()
```

**Why**: Bloom filters are crucial for databases (avoid disk lookups), web caches, spell checkers, and distributed systems. They trade accuracy for massive space savings—determining "definitely not present" vs "might be present".

## Dependency Injection Pattern

```python
# PROBLEM: Decouple component creation from business logic
# TIME: O(1) for registration and resolution SPACE: O(n) for registered services

from abc import ABC, abstractmethod
from typing import Dict, Type, Callable, Any, Optional
from dataclasses import dataclass
from enum import Enum, auto

class ServiceLifetime(Enum):
    """Lifecycle of a service in the container."""
    TRANSIENT = auto()  # New instance every time
    SINGLETON = auto()  # One instance for all requests
    SCOPED = auto()     # One instance per scope


class DIContainer:
    """Simple dependency injection container.
    
    Supports:
    - Service registration with interfaces
    - Constructor injection
    - Different lifetimes (transient, singleton, scoped)
    - Factory functions for complex creation
    """
    
    def __init__(self):
        self._registrations: Dict[Type, Dict] = {}
        self._singletons: Dict[Type, Any] = {}
        self._scoped_instances: Dict[Type, Any] = {}
    
    def register(
        self,
        interface: Type,
        implementation: Type,
        lifetime: ServiceLifetime = ServiceLifetime.TRANSIENT
    ) -> None:
        """Register a service implementation for an interface."""
        self._registrations[interface] = {
            'implementation': implementation,
            'lifetime': lifetime,
            'factory': None
        }
    
    def register_factory(
        self,
        interface: Type,
        factory: Callable[['DIContainer'], Any],
        lifetime: ServiceLifetime = ServiceLifetime.TRANSIENT
    ) -> None:
        """Register a factory function for complex service creation."""
        self._registrations[interface] = {
            'implementation': None,
            'lifetime': lifetime,
            'factory': factory
        }
    
    def resolve(self, interface: Type) -> Any:
        """Resolve a service by its interface."""
        if interface not in self._registrations:
            raise ValueError(f"Service {interface.__name__} not registered")
        
        registration = self._registrations[interface]
        lifetime = registration['lifetime']
        
        # Handle singleton lifetime
        if lifetime == ServiceLifetime.SINGLETON:
            if interface not in self._singletons:
                self._singletons[interface] = self._create_instance(registration)
            return self._singletons[interface]
        
        # Handle scoped lifetime
        if lifetime == ServiceLifetime.SCOPED:
            if interface not in self._scoped_instances:
                self._scoped_instances[interface] = self._create_instance(registration)
            return self._scoped_instances[interface]
        
        # Transient: always create new
        return self._create_instance(registration)
    
    def _create_instance(self, registration: Dict) -> Any:
        """Create a new instance using factory or constructor."""
        if registration['factory']:
            return registration['factory'](self)
        
        impl = registration['implementation']
        # Auto-resolve constructor dependencies
        return self._auto_wire(impl)
    
    def _auto_wire(self, cls: Type) -> Any:
        """Automatically resolve constructor parameters."""
        import inspect
        
        signature = inspect.signature(cls.__init__)
        kwargs = {}
        
        for param_name, param in signature.parameters.items():
            if param_name == 'self':
                continue
            
            annotation = param.annotation
            if annotation != inspect.Parameter.empty:
                # Try to resolve the annotated type
                if annotation in self._registrations:
                    kwargs[param_name] = self.resolve(annotation)
        
        return cls(**kwargs)
    
    def create_scope(self) -> 'DIScope':
        """Create a new scope for scoped services."""
        return DIScope(self)


class DIScope:
    """A scope for managing scoped service lifetimes."""
    
    def __init__(self, container: DIContainer):
        self._container = container
        self._scoped_instances: Dict[Type, Any] = {}
    
    def resolve(self, interface: Type) -> Any:
        """Resolve within this scope."""
        registration = self._container._registrations.get(interface)
        
        if registration and registration['lifetime'] == ServiceLifetime.SCOPED:
            if interface not in self._scoped_instances:
                self._scoped_instances[interface] = self._container._create_instance(registration)
            return self._scoped_instances[interface]
        
        return self._container.resolve(interface)


# Example: Building a web application architecture

# Domain interfaces
class ILogger(ABC):
    @abstractmethod
    def log(self, message: str) -> None: pass

class IUserRepository(ABC):
    @abstractmethod
    def get_user(self, user_id: int) -> dict: pass
    
    @abstractmethod
    def save_user(self, user: dict) -> None: pass

class ICache(ABC):
    @abstractmethod
    def get(self, key: str) -> Optional[Any]: pass
    
    @abstractmethod
    def set(self, key: str, value: Any, ttl: int = 3600) -> None: pass


# Implementations
class ConsoleLogger(ILogger):
    def log(self, message: str) -> None:
        print(f"[LOG] {message}")

class InMemoryCache(ICache):
    def __init__(self):
        self._cache: Dict[str, Any] = {}
    
    def get(self, key: str) -> Optional[Any]:
        return self._cache.get(key)
    
    def set(self, key: str, value: Any, ttl: int = 3600) -> None:
        self._cache[key] = value

class UserRepository(IUserRepository):
    def __init__(self, logger: ILogger, cache: ICache):
        self._logger = logger
        self._cache = cache
        self._db = {1: {"id": 1, "name": "Alice", "email": "alice@example.com"}}
    
    def get_user(self, user_id: int) -> dict:
        # Check cache first
        cache_key = f"user:{user_id}"
        cached = self._cache.get(cache_key)
        if cached:
            self._logger.log(f"Cache hit for user {user_id}")
            return cached
        
        # Fetch from database
        user = self._db.get(user_id)
        if user:
            self._cache.set(cache_key, user)
            self._logger.log(f"Fetched user {user_id} from database")
        
        return user
    
    def save_user(self, user: dict) -> None:
        self._db[user["id"]] = user
        self._cache.set(f"user:{user['id']}", user)
        self._logger.log(f"Saved user {user['id']}")


# Application service
class UserService:
    def __init__(self, user_repo: IUserRepository, logger: ILogger):
        self._repo = user_repo
        self._logger = logger
    
    def get_user_name(self, user_id: int) -> str:
        user = self._repo.get_user(user_id)
        if user:
            return user["name"]
        self._logger.log(f"User {user_id} not found")
        return "Unknown"


# Setup DI container
def configure_container() -> DIContainer:
    container = DIContainer()
    
    # Register services
    container.register(ILogger, ConsoleLogger, ServiceLifetime.SINGLETON)
    container.register(ICache, InMemoryCache, ServiceLifetime.SINGLETON)
    container.register(IUserRepository, UserRepository, ServiceLifetime.SCOPED)
    
    # Register factory for UserService
```

## Distributed Locks (Redlock)

```python
# PROBLEM: Acquire distributed lock across multiple Redis instances
# TIME: O(n) where n is number of Redis instances SPACE: O(1)

import time
import uuid
import redis
from typing import List, Optional
from dataclasses import dataclass

@dataclass
class LockResult:
    """Result of a lock acquisition attempt."""
    acquired: bool
    lock_id: Optional[str] = None
    validity_time: float = 0.0  # Milliseconds


class RedLock:
    """Distributed lock implementation using Redis (Redlock algorithm).
    
    Properties:
    - Mutual exclusion: Only one client can hold the lock at a time
    - Deadlock-free: Locks expire automatically
    - Fault-tolerant: Works even if some Redis instances fail
    
    The algorithm requires majority quorum (N/2 + 1 instances).
    """
    
    def __init__(
        self,
        redis_hosts: List[str],
        retry_delay: float = 0.1,
        clock_drift_factor: float = 0.01
    ):
        """Initialize Redlock with multiple Redis instances.
        
        Args:
            redis_hosts: List of Redis connection strings
            retry_delay: Delay between retry attempts (seconds)
            clock_drift_factor: Factor for clock drift compensation
        """
        if len(redis_hosts) < 3:
            raise ValueError("Redlock requires at least 3 Redis instances")
        
        self.clients = [
            redis.from_str(host) for host in redis_hosts
        ]
        self.quorum = len(self.clients) // 2 + 1
        self.retry_delay = retry_delay
        self.clock_drift_factor = clock_drift_factor
    
    def acquire(
        self,
        resource: str,
        ttl_ms: int,
        max_retries: int = 3
    ) -> LockResult:
        """Attempt to acquire a distributed lock.
        
        Args:
            resource: Resource name to lock
            ttl_ms: Lock time-to-live in milliseconds
            max_retries: Maximum number of acquisition attempts
            
        Returns:
            LockResult with acquisition status
        """
        lock_id = str(uuid.uuid4())
        
        for attempt in range(max_retries + 1):
            start_time = time.time()
            
            # Try to acquire lock on all instances
            acquired_count = 0
            for client in self.clients:
                try:
                    # SET with NX (only if not exists) and PX (milliseconds TTL)
                    acquired = client.set(
                        f"lock:{resource}",
                        lock_id,
                        nx=True,
                        px=ttl_ms
                    )
                    if acquired:
                        acquired_count += 1
                except redis.RedisError:
                    continue  # Instance failed, try others
            
            # Calculate time elapsed
            elapsed_ms = (time.time() - start_time) * 1000
            
            # Check if we got quorum and lock is still valid
            if acquired_count >= self.quorum:
                # Subtract elapsed time and clock drift
                validity = ttl_ms - elapsed_ms - (ttl_ms * self.clock_drift_factor)
                
                if validity > 0:
                    return LockResult(
                        acquired=True,
                        lock_id=lock_id,
                        validity_time=validity
                    )
            
            # Failed to acquire, release any partial locks
            self._release_instance(resource, lock_id)
            
            # Wait before retry
            if attempt < max_retries:
                time.sleep(self.retry_delay)
        
        return LockResult(acquired=False)
    
    def release(self, resource: str, lock_id: str) -> bool:
        """Release a distributed lock.
        
        Only releases if the lock still has our lock_id (prevents
        releasing another client's lock).
        
        Args:
            resource: Resource name
            lock_id: Lock identifier from acquisition
            
        Returns:
            True if released on majority of instances
        """
        return self._release_instance(resource, lock_id) >= self.quorum
    
    def _release_instance(self, resource: str, lock_id: str) -> int:
        """Release lock on all instances using Lua script.
        
        Lua script ensures atomicity:
        - Only delete if value matches our lock_id
        """
        lua_script = """
        if redis.call("get", KEYS[1]) == ARGV[1] then
            return redis.call("del", KEYS[1])
        else
            return 0
        end
        """
        
        released_count = 0
        for client in self.clients:
            try:
                result = client.eval(
                    lua_script,
                    1,
                    f"lock:{resource}",
                    lock_id
                )
                if result:
                    released_count += 1
            except redis.RedisError:
                continue
        
        return released_count
    
    def extend(
        self,
        resource: str,
        lock_id: str,
        ttl_ms: int
    ) -> bool:
        """Extend an existing lock's TTL.
        
        Args:
            resource: Resource name
            lock_id: Existing lock identifier
            ttl_ms: New TTL from now
            
        Returns:
            True if extended on majority of instances
        """
        lua_script = """
        if redis.call("get", KEYS[1]) == ARGV[1] then
            return redis.call("pexpire", KEYS[1], ARGV[2])
        else
            return 0
        end
        """
        
        extended_count = 0
        for client in self.clients:
            try:
                result = client.eval(
                    lua_script,
                    1,
                    f"lock:{resource}",
                    lock_id,
                    ttl_ms
                )
                if result:
                    extended_count += 1
            except redis.RedisError:
                continue
        
        return extended_count >= self.quorum


# Context manager for convenient usage
class distributed_lock:
    """Context manager for distributed locking."""
    
    def __init__(
        self,
        redlock: RedLock,
        resource: str,
        ttl_ms: int = 10000,
        max_retries: int = 3
    ):
        self.redlock = redlock
        self.resource = resource
        self.ttl_ms = ttl_ms
        self.max_retries = max_retries
        self.lock_result: Optional[LockResult] = None
    
    def __enter__(self) -> bool:
        self.lock_result = self.redlock.acquire(
            self.resource,
            self.ttl_ms,
            self.max_retries
        )
        return self.lock_result.acquired
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.lock_result and self.lock_result.acquired:
            self.redlock.release(self.resource, self.lock_result.lock_id)


# Usage example
def example_usage():
    # Configure with 5 Redis instances for fault tolerance
    redis_hosts = [
        "redis://localhost:6379/0",
        "redis://localhost:6380/0",
        "redis://localhost:6381/0",
        "redis://localhost:6382/0",
        "redis://localhost:6383/0",
    ]
    
    lock_manager = RedLock(redis_hosts)
    
    # Method 1: Manual lock
    result = lock_manager.acquire("user:123:balance", ttl_ms=5000)
    if result.acquired:
        try:
            # Critical section - update balance
            print(f"Lock acquired, valid for {result.validity_time:.0f}ms")
            time.sleep(0.1)  # Simulate work
        finally:
            lock_manager.release("user:123:balance", result.lock_id)
    
    # Method 2: Context manager
    with distributed_lock(lock_manager, "order:456:process", ttl_ms=10000) as acquired:
        if acquired:
            print("Processing order...")
            # Do work
        else:
            print("Could not acquire lock, order is being processed elsewhere")
```
