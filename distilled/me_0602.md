## Double Hashing

```python
# PROBLEM: Implement hash table collision resolution using double hashing
# TIME: O(1) average, O(n) worst case SPACE: O(n)
class DoubleHashTable:
    """Hash table using double hashing for collision resolution."""
    
    def __init__(self, size: int = 16):
        self.size = size
        self.table = [None] * size
        self.DELETED = object()  # Marker for deleted slots
    
    def _hash1(self, key: int) -> int:
        """Primary hash function."""
        return key % self.size
    
    def _hash2(self, key: int) -> int:
        """Secondary hash function - must not return 0."""
        # Use a prime smaller than table size
        return 1 + (key % (self.size - 1))
    
    def _probe(self, key: int, i: int) -> int:
        """Calculate probe position: (h1 + i * h2) % size."""
        return (self._hash1(key) + i * self._hash2(key)) % self.size
    
    def insert(self, key: int, value: any) -> bool:
        """Insert key-value pair, return False if table is full."""
        for i in range(self.size):
            pos = self._probe(key, i)
            if self.table[pos] is None or self.table[pos] is self.DELETED:
                self.table[pos] = (key, value)
                return True
            # Update existing key
            if self.table[pos][0] == key:
                self.table[pos] = (key, value)
                return True
        return False  # Table full
    
    def search(self, key: int) -> any:
        """Search for key, return value or None."""
        for i in range(self.size):
            pos = self._probe(key, i)
            if self.table[pos] is None:
                return None  # Key not found
            if self.table[pos] is not self.DELETED and self.table[pos][0] == key:
                return self.table[pos][1]
        return None
    
    def delete(self, key: int) -> bool:
        """Delete key using lazy deletion."""
        for i in range(self.size):
            pos = self._probe(key, i)
            if self.table[pos] is None:
                return False
            if self.table[pos] is not self.DELETED and self.table[pos][0] == key:
                self.table[pos] = self.DELETED
                return True
        return False
```
**Why**: Double hashing provides better distribution than linear/quadratic probing, reducing clustering while maintaining O(1) average operations.

## Perfect Hashing

```python
# PROBLEM: Implement perfect hashing for static key sets with O(1) worst-case lookup
# TIME: O(1) worst case for lookup SPACE: O(n)
import random
from typing import List, Any, Optional

class PerfectHash:
    """Two-level perfect hashing for static sets."""
    
    def __init__(self, keys: List[int], values: List[Any]):
        self.n = len(keys)
        self._build(keys, values)
    
    def _hash(self, key: int, a: int, b: int, p: int, m: int) -> int:
        """Universal hash function: ((a*key + b) % p) % m."""
        return ((a * key + b) % p) % m
    
    def _build(self, keys: List[int], values: List[Any]):
        """Build two-level perfect hash table."""
        # Large prime for universal hashing
        p = 2**31 - 1
        
        # First level: try until no bucket has > sqrt(n) collisions
        while True:
            self.a1 = random.randint(1, p - 1)
            self.b1 = random.randint(0, p - 1)
            buckets = [[] for _ in range(self.n)]
            
            for i, key in enumerate(keys):
                h = self._hash(key, self.a1, self.b1, p, self.n)
                buckets[h].append((key, values[i]))
            
            # Check if distribution is acceptable
            total_secondary = sum(len(b) ** 2 for b in buckets)
            if total_secondary <= 4 * self.n:
                break
        
        # Second level: perfect hash each bucket
        self.secondary = []
        for bucket in buckets:
            if not bucket:
                self.secondary.append(None)
                continue
            
            m = len(bucket) ** 2  # Square the bucket size
            # Find hash parameters with no collisions
            while True:
                a2 = random.randint(1, p - 1)
                b2 = random.randint(0, p - 1)
                table = [None] * m
                collision = False
                
                for key, val in bucket:
                    h = self._hash(key, a2, b2, p, m)
                    if table[h] is not None:
                        collision = True
                        break
                    table[h] = (key, val)
                
                if not collision:
                    self.secondary.append({'a': a2, 'b': b2, 'm': m, 'table': table})
                    break
    
    def lookup(self, key: int) -> Optional[Any]:
        """O(1) worst-case lookup."""
        p = 2**31 - 1
        h1 = self._hash(key, self.a1, self.b1, p, self.n)
        
        if self.secondary[h1] is None:
            return None
        
        sec = self.secondary[h1]
        h2 = self._hash(key, sec['a'], sec['b'], p, sec['m'])
        
        if sec['table'][h2] is not None and sec['table'][h2][0] == key:
            return sec['table'][h2][1]
        return None
```
**Why**: Perfect hashing guarantees O(1) worst-case lookup for static key sets, ideal for read-heavy scenarios with known keys (e.g., keyword lookup, static dictionaries).

```python
# TIME: O(k) where k is number of hash functions SPACE: O(m) bits
import hashlib
from typing import List, Callable

class CountingBloomFilter:
    """Bloom filter with deletion support using counters."""
    
    def __init__(self, size: int = 1000, hash_count: int = 7):
        self.size = size
        self.hash_count = hash_count
        self.counters = [0] * size
    
    def _hashes(self, item: any) -> List[int]:
        """Generate k hash values using double hashing technique."""
        data = str(item).encode()
        h1 = int(hashlib.md5(data).hexdigest(), 16)
        h2 = int(hashlib.sha1(data).hexdigest(), 16)
        
        return [(h1 + i * h2) % self.size for i in range(self.hash_count)]
    
    def add(self, item: any):
        """Add item to filter, increment counters."""
        for h in self._hashes(item):
            self.counters[h] += 1
    
    def contains(self, item: any) -> bool:
        """Check if item might be in set (may have false positives)."""
        return all(self.counters[h] > 0 for h in self._hashes(item))
    
    def remove(self, item: any) -> bool:
        """Remove item if it exists (no false negatives guaranteed)."""
        hashes = self._hashes(item)
        # Verify all counters > 0 before removing
        if not all(self.counters[h] > 0 for h in hashes):
            return False
        for h in hashes:
            self.counters[h] -= 1
        return True
```

```python
# PROBLEM: Implement cache with random eviction policy
# TIME: O(1) for all operations SPACE: O(capacity)
import random
from typing import Any, Optional

class RandomCache:
    """Cache with random replacement eviction policy."""
    
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache: dict = {}
        self.keys: list = []  # Track keys for random selection
    
    def get(self, key: any) -> Optional[Any]:
        """Get value if exists, return None otherwise."""
        return self.cache.get(key)
    
    def put(self, key: any, value: Any):
        """Add/update key-value pair, evict randomly if at capacity."""
        if key in self.cache:
            self.cache[key] = value
            return
        
        # Evict if at capacity
        if len(self.cache) >= self.capacity:
            # Random selection using reservoir sampling
            victim = random.choice(self.keys)
            self.cache.pop(victim)
            self.keys.remove(victim)
        
        self.cache[key] = value
        self.keys.append(key)
    
    def delete(self, key: any) -> bool:
        """Remove key from cache."""
        if key in self.cache:
            self.cache.pop(key)
            self.keys.remove(key)
            return True
        return False
    
    def clear(self):
        """Clear all entries."""
        self.cache.clear()
        self.keys.clear()


class RandomSampleCache:
    """Random replacement with sampling - evicts from random sample."""
    
    def __init__(self, capacity: int, sample_size: int = 5):
        self.capacity = capacity
        self.sample_size = sample_size
        self.cache: dict = {}
    
    def get(self, key: any) -> Optional[Any]:
        """Get value, update access metadata."""
        if key in self.cache:
            self.cache[key]['accesses'] += 1
            return self.cache[key]['value']
        return None
    
    def put(self, key: any, value: Any):
        """Insert with random sample eviction."""
        if key in self.cache:
            self.cache[key]['value'] = value
            self.cache[key]['accesses'] += 1
            return
        
        if len(self.cache) >= self.capacity:
            # Sample random keys and evict least accessed
            sample = random.sample(list(self.cache.keys()), 
                                   min(self.sample_size, len(self.cache)))
            victim = min(sample, key=lambda k: self.cache[k]['accesses'])
            del self.cache[victim]
        
        self.cache[key] = {'value': value, 'accesses': 1}
```
**Why**: Random replacement is simple and lock-free friendly. Sampling variant improves eviction quality with minimal overhead, useful for concurrent caches.

## Write-Through Cache

```python
# PROBLEM: Implement write-through cache pattern for data consistency
# TIME: O(1) cache + O(storage) for writes SPACE: O(cache_size)
from typing import Any, Optional, Callable
from dataclasses import dataclass
import threading

@dataclass
class CacheEntry:
    """Cache entry with metadata."""
    value: Any
    timestamp: float
    ttl: float = 0  # 0 means no expiry

class WriteThroughCache:
    """Cache that writes to backing store synchronously."""
    
    def __init__(self, 
                 capacity: int,
                 write_fn: Callable[[any, Any], bool],
                 read_fn: Callable[[any], Any],
                 default_ttl: float = 3600):
        self.capacity = capacity
        self.write_fn = write_fn  # Function to write to backing store
        self.read_fn = read_fn    # Function to read from backing store
        self.default_ttl = default_ttl
        self.cache: dict = {}
        self.access_order: list = []  # For LRU eviction
        self.lock = threading.RLock()
    
    def get(self, key: any) -> Optional[Any]:
        """Read from cache, fall back to backing store (cache-aside on miss)."""
        with self.lock:
            import time
            now = time.time()
            
            if key in self.cache:
                entry = self.cache[key]
                # Check TTL
                if entry.ttl == 0 or now - entry.timestamp < entry.ttl:
                    # Update access order for LRU
                    self._update_access(key)
                    return entry.value
                else:
                    # Expired, remove from cache
                    del self.cache[key]
                    self.access_order.remove(key)
            
            # Cache miss: read from backing store
            value = self.read_fn(key)
            if value is not None:
                self._evict_if_needed()
                self.cache[key] = CacheEntry(value, now, self.default_ttl)
                self.access_order.append(key)
            return value
    
    def put(self, key: any, value: Any) -> bool:
        """Write to both cache and backing store atomically."""
        # Write to backing store first (write-through)
        if not self.write_fn(key, value):
            return False
        
        with self.lock:
            import time
            self._evict_if_needed()
            
            if key in self.cache:
                self.access_order.remove(key)
            
            self.cache[key] = CacheEntry(value, time.time(), self.default_ttl)
            self.access_order.append(key)
            return True
    
    def delete(self, key: any) -> bool:
        """Remove from cache and backing store."""
        # Write-through delete
        if not self.write_fn(key, None):  # Assume None means delete
            return False
        
        with self.lock:
            if key in self.cache:
                del self.cache[key]
                self.access_order.remove(key)
            return True
    
    def _evict_if_needed(self):
        """Evict LRU entry if at capacity (must be called with lock)."""
        if len(self.cache) >= self.capacity and self.access_order:
            lru_key = self.access_order.pop(0)
            del self.cache[lru_key]
    
    def _update_access(self, key: any):
        """Move key to end of access order (most recent)."""
        if key in self.access_order:
            self.access_order.remove(key)
            self.access_order.append(key)
```
**Why**: Write-through ensures cache and backing store are always synchronized. Reads are fast after initial load, writes are slower but consistent. Best for read-heavy workloads where stale data is unacceptable.

## Read-Through Cache

```python
# PROBLEM: Implement read-through cache that lazy-loads from backing store
# TIME: O(1) cache hit, O(storage) on miss SPACE: O(cache_size)
from typing import Any, Optional, Callable
from dataclasses import dataclass
import threading
import time

class ReadThroughCache:
    """Cache that automatically loads missing data from source."""
    
    def __init__(self,
                 capacity: int,
                 load_fn: Callable[[any], Any],
                 ttl: float = 3600):
        self.capacity = capacity
        self.load_fn = load_fn  # Function to load data on cache miss
        self.ttl = ttl
        self.cache: dict = {}
        self.lock = threading.RLock()
    
    def get(self, key: any) -> Optional[Any]:
        """Get from cache or load from source."""
        with self.lock:
            # Check cache
            if key in self.cache:
                entry = self.cache[key]
                if time.time() - entry['timestamp'] < self.ttl:
                    return entry['value']
                else:
                    # Expired, remove and reload
                    del self.cache[key]
            
            # Cache miss: load from source
            value = self._load_from_source(key)
            if value is not None:
                self._add_to_cache(key, value)
            return value
    
    def _load_from_source(self, key: any) -> Optional[Any]:
        """Load data from backing store (can be overridden)."""
        return self.load_fn(key)
    
    def _add_to_cache(self, key: any, value: Any):
        """Add to cache with LRU eviction."""
        if len(self.cache) >= self.capacity:
            # Evict oldest entry
            oldest_key = min(self.cache.keys(), 
                           key=lambda k: self.cache[k]['timestamp'])
            del self.cache[oldest_key]
        
        self.cache[key] = {
            'value': value,
            'timestamp': time.time()
        }
    
    def invalidate(self, key: any):
        """Remove specific key from cache."""
        with self.lock:
            self.cache.pop(key, None)
    
    def refresh(self, key: any) -> Optional[Any]:
        """Force reload from source."""
        with self.lock:
            self.cache.pop(key, None)
            value = self._load_from_source(key)
            if value is not None:
                self._add_to_cache(key, value)
            return value
    
    def preload(self, keys: list):
        """Pre-load multiple keys into cache."""
        for key in keys:
            self.get(key)  # Will load if not cached


class AsyncReadThroughCache:
    """Read-through cache with async refresh for stale entries."""
    
    def __init__(self, 
                 capacity: int,
                 load_fn: Callable,
                 ttl: float = 3600,
                 stale_while_revalidate: float = 600):
        self.capacity = capacity
        self.load_fn = load_fn
        self.ttl = ttl
        self.stale_ttl = stale_while_revalidate
        self.cache: dict = {}
        self.lock = threading.RLock()
        self.refreshing = set()
    
    def get(self, key: any) -> Optional[Any]:
        """Return stale data while refreshing in background."""
        with self.lock:
            if key not in self.cache:
                # Miss: load synchronously
                value = self.load_fn(key)
                if value is not None:
                    self._add(key, value)
                return value
            
            entry = self.cache[key]
            age = time.time() - entry['timestamp']
            
            if age < self.ttl:
                # Fresh: return directly
                return entry['value']
            
            if age < self.ttl + self.stale_ttl:
                # Stale but acceptable: trigger background refresh
                if key not in self.refreshing:
                    self.refreshing.add(key)
                    threading.Thread(target=self._background_refresh, 
                                   args=(key,), daemon=True).start()
                return entry['value']
            
            # Too stale: load synchronously
            value = self.load_fn(key)
            if value is not None:
                self._add(key, value)
            return value
    
    def _background_refresh(self, key: any):
        """Refresh entry in background."""
        try:
            value = self.load_fn(key)
            if value is not None:
                with self.lock:
                    self._add(key, value)
        finally:
            with self.lock:
                self.refreshing.discard(key)
    
    def _add(self, key: any, value: Any):
        """Add entry with eviction."""
        if len(self.cache) >= self.capacity:
            oldest = min(self.cache.keys(),
                        key=lambda k: self.cache[k]['timestamp'])
            del self.cache[oldest]
        self.cache[key] = {'value': value, 'timestamp': time.time()}
```
**Why**: Read-through abstracts the data source from the application. Stale-while-revalidate variant provides low latency by serving slightly stale data while refreshing in background.

## Double-Checked Locking

```python
# PROBLEM: Implement thread-safe singleton with efficient lazy initialization
# TIME: O(1) after initialization SPACE: O(1)
import threading
from typing import Optional, Callable, TypeVar, Generic

T = TypeVar('T')

class Singleton:
    """Thread-safe singleton using double-checked locking."""
    
    _instance: Optional['Singleton'] = None
    _lock = threading.Lock()
    
    def __new__(cls):
        if cls._instance is None:  # First check (no lock)
            with cls._lock:
                if cls._instance is None:  # Second check (with lock)
                    cls._instance = super().__new__(cls)
        return cls._instance


class LazyInitializer(Generic[T]):
    """Generic lazy initialization with double-checked locking."""
    
    def __init__(self, factory: Callable[[], T]):
        self._factory = factory
        self._instance: Optional[T] = None
        self._lock = threading.Lock()
        self._initialized = False
    
    def get(self) -> T:
        """Get instance, initializing if needed."""
        if not self._initialized:  # First check
            with self._lock:
                if not self._initialized:  # Second check
                    self._instance = self._factory()
                    self._initialized = True
        return self._instance
    
    def reset(self):
        """Reset to uninitialized state."""
        with self._lock:
            self._instance = None
            self._initialized = False


class ConnectionPool:
    """Database connection pool using double-checked locking."""
    
    _instance: Optional['ConnectionPool'] = None
    _lock = threading.Lock()
    
    def __init__(self):
        if self._instance is not None:
            raise RuntimeError("Use get_instance()")
        self._pool: list = []
        self._in_use: set = set()
        self._pool_lock = threading.Lock()
    
    @classmethod
    def get_instance(cls) -> 'ConnectionPool':
        """Get singleton instance with thread-safe lazy init."""
        if cls._instance is None:  # First check
            with cls._lock:
                if cls._instance is None:  # Second check
                    cls._instance = cls()
        return cls._instance
    
    def acquire(self):
        """Get a connection from the pool."""
        with self._pool_lock:
            if self._pool:
                conn = self._pool.pop()
            else:
                conn = self._create_connection()
            self._in_use.add(conn)
            return conn
    
    def release(self, conn):
        """Return connection to pool."""
        with self._pool_lock:
            self._in_use.discard(conn)
            self._pool.append(conn)
    
    def _create_connection(self):
        """Create new connection (placeholder)."""
        return f"Connection-{id(self)}"


# Python 3.12+ alternative using parametric singleton
def singleton(cls):
    """Decorator for singleton pattern with double-checked locking."""
    instances = {}
    lock = threading.Lock()
    
    def get_instance(*args, **kwargs):
        if cls not in instances:
            with lock:
                if cls not in instances:
                    instances[cls] = cls(*args, **kwargs)
        return instances[cls]
    
    return get_instance


@singleton
class ConfigManager:
    """Application config as singleton."""
    def __init__(self):
        self.config = {}
```
**Why**: Double-checked locking avoids synchronization overhead after initialization while remaining thread-safe. Essential for singleton patterns in multi-threaded applications. Note: In Python, the GIL provides some protection, but explicit locking is still needed for complex initialization.

## Message Broker Patterns

```python
# PROBLEM: Implement core message broker patterns: pub/sub, queues, routing
# TIME: O(1) publish/subscribe SPACE: O(subscribers + messages)
import threading
from typing import Callable, Dict, List, Set, Any, Optional
from queue import Queue
from dataclasses import dataclass
from enum import Enum

class MessageBroker:
    """In-memory message broker supporting multiple patterns."""
    
    def __init__(self):
        self.lock = threading.RLock()
        # Pub/Sub
        self.topics: Dict[str, Set[Callable]] = {}
        # Point-to-Point queues
        self.queues: Dict[str, Queue] = {}
        # Routing (topic exchange)
        self.bindings: Dict[str, Set[str]] = {}  # pattern -> queue names
    
    # ============ Pub/Sub Pattern ============
    def subscribe(self, topic: str, callback: Callable[[Any], None]):
        """Subscribe to topic, receive all messages."""
        with self.lock:
            if topic not in self.topics:
                self.topics[topic] = set()
            self.topics[topic].add(callback)
    
    def unsubscribe(self, topic: str, callback: Callable):
        """Remove subscription."""
        with self.lock:
            if topic in self.topics:
                self.topics[topic].discard(callback)
    
    def publish(self, topic: str, message: Any):
        """Broadcast to all subscribers."""
        with self.lock:
            subscribers = self.topics.get(topic, set()).copy()
        
        for callback in subscribers:
            try:
                callback(message)
            except Exception as e:
                print(f"Callback error: {e}")
    
    # ============ Point-to-Point Queue ============
    def create_queue(self, name: str, maxsize: int = 0):
        """Create a named queue."""
        with self.lock:
            if name not in self.queues:
                self.queues[name] = Queue(maxsize)
    
    def send_to_queue(self, name: str, message: Any, block: bool = True):
        """Send message to specific queue."""
        if name not in self.queues:
            self.create_queue(name)
        self.queues[name].put(message, block=block)
    
    def receive_from_queue(self, name: str, block: bool = True, 
                          timeout: Optional[float] = None) -> Any:
        """Receive from specific queue."""
        if name not in self.queues:
            raise ValueError(f"Queue {name} does not exist")
        return self.queues[name].get(block=block, timeout=timeout)
    
    # ============ Topic Exchange Pattern ============
    def bind_queue(self, queue_name: str, routing_pattern: str):
        """Bind queue to routing pattern (supports * and # wildcards)."""
        with self.lock:
            if routing_pattern not in self.bindings:
                self.bindings[routing_pattern] = set()
            self.bindings[routing_pattern].add(queue_name)
            if queue_name not in self.queues:
                self.queues[queue_name] = Queue()
    
    def publish_routed(self, routing_key: str, message: Any):
        """Publish to all queues matching routing pattern."""
        with self.lock:
            matched_queues = []
            for pattern, queues in self.bindings.items():
                if self._match_routing(routing_key, pattern):
                    matched_queues.extend(queues)
        
        for qname in set(matched_queues):
            self.queues[qname].put(message)
    
    def _match_routing(self, key: str, pattern: str) -> bool:
        """Match routing key against pattern (* = one word, # = multiple)."""
        key_parts = key.split('.')
        pattern_parts = pattern.split('.')
        
        ki, pi = 0, 0
        while ki < len(key_parts) and pi < len(pattern_parts):
            if pattern_parts[pi] == '#':
                # Match zero or more words
                if pi == len(pattern_parts) - 1:
                    return True
                pi += 1
                while ki < len(key_parts):
                    if self._match_routing('.'.join(key_parts[ki:]), 
                                          '.'.join(pattern_parts[pi:])):
                        return True
                    ki += 1
                return False
            elif pattern_parts[pi] == '*' or pattern_parts[pi] == key_parts[ki]:
                ki += 1
                pi += 1
            else:
                return False
        
        return ki == len(key_parts) and pi == len(pattern_parts)


# Request-Reply Pattern
class RequestReplyBroker:
    """Request-reply messaging with correlation IDs."""
    
    def __init__(self):
        self.pending: Dict[str, threading.Event] = {}
        self.responses: Dict[str, Any] = {}
        self.lock = threading.Lock()
        self.queue = Queue()
    
    def send_request(self, request: Any, timeout: float = 5.0) -> Any:
        """Send request and wait for reply."""
        import uuid
        correlation_id = str(uuid.uuid4())
        
        event = threading.Event()
        with self.lock:
            self.pending[correlation_id] = event
        
        self.queue.put({'id': correlation_id, 'request': request})
        
        # Wait for response
        if event.wait(timeout):
            with self.lock:
                response = self.responses.pop(correlation_id, None)
                self.pending.pop(correlation_id, None)
            return response
        else:
            with self.lock:
                self.pending.pop(correlation_id, None)
            raise TimeoutError("Request timed out")
    
    def send_reply(self, correlation_id: str, response: Any):
        """Send reply for a request."""
        with self.lock:
            if correlation_id in self.pending:
                self.responses[correlation_id] = response
                self.pending[correlation_id].set()
```
**Why**: Message brokers enable decoupled, scalable architectures. Pub/Sub for broadcasting, queues for load distribution, topic exchanges for flexible routing, request-reply for synchronous patterns over async messaging.

```python
# PROBLEM: Implement Kafka-style patterns: partitions, consumer groups, offsets
# TIME: O(log n) partition selection SPACE: O(partitions * messages)
import threading
from typing import Dict, List, Any, Callable, Optional
from queue import Queue
from dataclasses import dataclass
import hashlib

@dataclass
class Message:
    """Kafka-style message with metadata."""
    key: Any
    value: Any
    partition: int
    offset: int
    timestamp: float

class TopicPartition:
    """Single partition of a topic."""
    
    def __init__(self, partition_id: int):
        self.partition_id = partition_id
        self.messages: List[Message] = []
        self.lock = threading.Lock()
        self.next_offset = 0
    
    def append(self, key: Any, value: Any) -> int:
        """Add message, return offset."""
        import time
        with self.lock:
            msg = Message(key, value, self.partition_id, 
                         self.next_offset, time.time())
            self.messages.append(msg)
            offset = self.next_offset
            self.next_offset += 1
            return offset
    
    def read(self, from_offset: int, max_messages: int = 100) -> List[Message]:
        """Read messages from offset."""
        with self.lock:
            return self.messages[from_offset:from_offset + max_messages]


class KafkaLikeTopic:
    """Topic with multiple partitions."""
    
    def __init__(self, name: str, num_partitions: int = 3):
        self.name = name
        self.partitions = [TopicPartition(i) for i in range(num_partitions)]
    
    def _select_partition(self, key: Any) -> int:
        """Select partition based on key hash."""
        if key is None:
            import random
            return random.randint(0, len(self.partitions) - 1)
        
        key_hash = int(hashlib.md5(str(key).encode()).hexdigest(), 16)
        return key_hash % len(self.partitions)
    
    def produce(self, key: Any, value: Any) -> tuple:
        """Produce message, return (partition, offset)."""
        partition_id = self._select_partition(key)
        offset = self.partitions[partition_id].append(key, value)
        return partition_id, offset
    
    def consume(self, partition: int, from_offset: int) -> List[Message]:
        """Consume from specific partition and offset."""
        return self.partitions[partition].read(from_offset)


class ConsumerGroup:
    """Consumer group with partition assignment."""
    
    def __init__(self, group_id: str, topic: KafkaLikeTopic):
        self.group_id = group_id
        self.topic = topic
        self.consumers: Dict[str, List[int]] = {}  # consumer_id -> partitions
        self.offsets: Dict[int, int] = {}  # partition -> committed offset
        self.lock = threading.Lock()
    
    def join(self, consumer_id: str):
        """Add consumer and rebalance partitions."""
        with self.lock:
            self.consumers[consumer_id] = []
            self._rebalance()
    
    def leave(self, consumer_id: str):
        """Remove consumer and rebalance."""
        with self.lock:
            self.consumers.pop(consumer_id, None)
            self._rebalance()
    
    def _rebalance(self):
        """Redistribute partitions among consumers."""
        if not self.consumers:
            return
        
        num_partitions = len(self.topic.partitions)
        consumer_ids = sorted(self.consumers.keys())
        num_consumers = len(consumer_ids)
        
        # Round-robin assignment
        for i, cid in enumerate(consumer_ids):
            # Each consumer gets roughly equal partitions
            self.consumers[cid] = [
                p for p in range(num_partitions) 
                if p % num_consumers == i
            ]
    
    def get_assignment(self, consumer_id: str) -> List[int]:
        """Get partitions assigned to consumer."""
        with self.lock:
            return self.consumers.get(consumer_id, []).copy()
    
    def poll(self, consumer_id: str, max_per_partition: int = 10) -> List[Message]:
        """Poll all assigned partitions."""
        messages = []
        for partition_id in self.get_assignment(consumer_id):
            offset = self.offsets.get(partition_id, 0)
            msgs = self.topic.consume(partition_id, offset)
            if msgs:
                messages.extend(msgs)
                self.offsets[partition_id] = msgs[-1].offset + 1
        return messages
    
    def commit(self, partition: int, offset: int):
        """Manually commit offset."""
        with self.lock:
            self.offsets[partition] = offset


class KafkaLikeBroker:
    """Simple Kafka-style broker."""
    
    def __init__(self):
        self.topics: Dict[str, KafkaLikeTopic] = {}
        self.groups: Dict[str, ConsumerGroup] = {}
        self.lock = threading.Lock()
    
    def create_topic(self, name: str, partitions: int = 3):
        """Create topic with partitions."""
        with self.lock:
            if name not in self.topics:
                self.topics[name] = KafkaLikeTopic(name, partitions)
    
    def produce(self, topic: str, key: Any, value: Any) -> tuple:
        """Produce to topic."""
        if topic not in self.topics:
            self.create_topic(topic)
        return self.topics[topic].produce(key, value)
    
    def create_group(self, group_id: str, topic: str) -> ConsumerGroup:
        """Create consumer group for topic."""
        with self.lock:
            if topic not in self.topics:
                self.create_topic(topic)
            if group_id not in self.groups:
                self.groups[group_id] = ConsumerGroup(group_id, self.topics[topic])
            return self.groups[group_id]
```
**Why**: Kafka patterns enable high-throughput, fault-tolerant streaming. Partitions allow parallel consumption, consumer groups provide scalability and fault tolerance, offset tracking enables exactly-once or at-least-once semantics.

```python
# PROBLEM: Implement common NoSQL patterns: document store, key-value, column family
# TIME: O(1) key-based access SPACE: O(data)
from typing import Dict, List, Any, Optional
from dataclasses import dataclass
import json
import threading

# ============ Document Store Pattern ============
class DocumentStore:
    """MongoDB-style document store."""
    
    def __init__(self):
        self.collections: Dict[str, Dict[str, dict]] = {}
        self.indexes: Dict[str, Dict[str, Dict]] = {}  # collection -> field -> {value: [ids]}
        self.lock = threading.RLock()
    
    def insert(self, collection: str, doc: dict) -> str:
        """Insert document, auto-generate ID if missing."""
        import uuid
        doc_id = doc.get('_id', str(uuid.uuid4()))
        doc['_id'] = doc_id
        
        with self.lock:
            if collection not in self.collections:
                self.collections[collection] = {}
                self.indexes[collection] = {}
            
            self.collections[collection][doc_id] = doc.copy()
            self._update_indexes(collection, doc_id, doc)
        
        return doc_id
    
    def find(self, collection: str, query: dict) -> List[dict]:
        """Find documents matching query."""
        with self.lock:
            if collection not in self.collections:
                return []
            
            # Use index if available
            indexed_candidates = None
            for field, value in query.items():
                if field in self.indexes.get(collection, {}):
                    candidates = set(self.indexes[collection][field].get(value, []))
                    if indexed_candidates is None:
                        indexed_candidates = candidates
                    else:
                        indexed_candidates &= candidates
            
            # If no index or index incomplete, scan all
            if indexed_candidates is None:
                docs = list(self.collections[collection].values())
            else:
                docs = [self.collections[collection][doc_id] 
                       for doc_id in indexed_candidates]
            
            # Apply remaining filters
            return [d for d in docs if self._matches(d, query)]
    
    def update(self, collection: str, query: dict, updates: dict) -> int:
        """Update matching documents."""
        docs = self.find(collection, query)
        with self.lock:
            for doc in docs:
                doc_id = doc['_id']
                old_doc = self.collections[collection][doc_id].copy()
                doc.update(updates)
                self._update_indexes(collection, doc_id, doc, old_doc)
        return len(docs)
    
    def create_index(self, collection: str, field: str):
        """Create index on field."""
        with self.lock:
            if collection not in self.indexes:
                self.indexes[collection] = {}
            self.indexes[collection][field] = {}
            
            # Build index
            for doc_id, doc in self.collections.get(collection, {}).items():
                value = doc.get(field)
                if value not in self.indexes[collection][field]:
                    self.indexes[collection][field][value] = []
                self.indexes[collection][field][value].append(doc_id)
    
    def _matches(self, doc: dict, query: dict) -> bool:
        """Check if document matches query."""
        for key, value in query.items():
            if '.' in key:
                # Nested field query
                parts = key.split('.')
                current = doc
                for part in parts:
                    if not isinstance(current, dict) or part not in current:
                        return False
                    current = current[part]
                if current != value:
                    return False
            elif doc.get(key) != value:
                return False
        return True
    
    def _update_indexes(self, collection: str, doc_id: str, 
                       doc: dict, old_doc: dict = None):
        """Update indexes after document change."""
        if old_doc:
            # Remove from old indexes
            for field in self.indexes.get(collection, {}):
                old_value = old_doc.get(field)
                if old_value is not None:
                    self.indexes[collection][field].get(old_value, []).remove(doc_id)
        
        # Add to new indexes
        for field in self.indexes.get(collection, {}):
            value = doc.get(field)
            if value is not None:
                if value not in self.indexes[collection][field]:
                    self.indexes[collection][field][value] = []
                self.indexes[collection][field][value].append(doc_id)

class ColumnFamilyStore:
    """Cassandra-style column family store."""
    
    def __init__(self):
        self.column_families: Dict[str, Dict[str, Dict[str, Any]]] = {}
        self.lock = threading.RLock()
    
    def create_family(self, name: str):
        """Create column family."""
        with self.lock:
            if name not in self.column_families:
                self.column_families[name] = {}
    
    def put(self, family: str, row_key: str, column: str, value: Any, 
            ttl: Optional[int] = None):
        """Put column value for row key."""
        import time
        with self.lock:
            if family not in self.column_families:
                self.column_families[family] = {}
            if row_key not in self.column_families[family]:
                self.column_families[family][row_key] = {}
            
            entry = {'value': value, 'timestamp': time.time()}
            if ttl:
                entry['expires'] = time.time() + ttl
            
            self.column_families[family][row_key][column] = entry
    
    def get(self, family: str, row_key: str, column: str) -> Optional[Any]:
        """Get column value."""
        with self.lock:
            import time
            family_data = self.column_families.get(family, {})
            row = family_data.get(row_key, {})
            entry = row.get(column)
            
            if entry is None:
                return None
            
            # Check TTL
            if 'expires' in entry and time.time() > entry['expires']:
                del row[column]
                return None
            
            return entry['value']
    
    def get_row(self, family: str, row_key: str) -> Dict[str, Any]:
        """Get all columns for row."""
        with self.lock:
            import time
            row = self.column_families.get(family, {}).get(row_key, {}).copy()
            
            # Filter expired and return values only
            result = {}
            for col, entry in row.items():
                if 'expires' in entry and time.time() > entry['expires']:
                    continue
                result[col] = entry['value']
            return result
    
    def delete(self, family: str, row_key: str, column: Optional[str] = None):
        """Delete column or entire row."""
        with self.lock:
            if column is None:
                self.column_families.get(family, {}).pop(row_key, None)
            else:
                self.column_families.get(family, {}).get(row_key, {}).pop(column, None)

class KeyValueStore:
    """Redis-style key-value store with TTL."""
    
    def __init__(self):
        self.store: Dict[str, Any] = {}
        self.ttls: Dict[str, float] = {}
        self.lock = threading.RLock()
    
    def set(self, key: str, value: Any, ttl: Optional[int] = None):
        """Set key with optional TTL."""
        import time
        with self.lock:
            self.store[key] = value
            if ttl:
                self.ttls[key] = time.time() + ttl
            else:
                self.ttls.pop(key, None)
    
    def get(self, key: str) -> Optional[Any]:
        """Get value, return None if expired or missing."""
        import time
        with self.lock:
            if key in self.ttls and time.time() > self.ttls[key]:
                del self.store[key]
                del self.ttls[key]
                return None
            return self.store.get(key)
    
    def delete(self, key: str):
        """Delete key."""
        with self.lock:
            self.store.pop(key, None)
            self.ttls.pop(key, None)
    
    def increment(self, key: str, delta: int = 1) -> int:
        """Atomic increment."""
        with self.lock:
            value = self.store.get(key, 0)
            if not isinstance(value, (int, float)):
                raise TypeError("Value is not numeric")
            value += delta
            self.store[key] = value
            return value
    
    def set_if_not_exists(self, key: str, value: Any) -> bool:
        """Set only if key doesn't exist (atomic)."""
        with self.lock:
            if key in self.store:
                return False
            self.store[key] = value
            return True
```

**Why**: NoSQL patterns address different data access patterns than relational databases. Document stores for nested data, column families for wide rows and time-series, key-value for simple caching with TTL support.
