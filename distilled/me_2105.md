## Kruskal's Minimum Spanning Tree

```python
# PROBLEM: Find minimum spanning tree in weighted undirected graph
# TIME: O(E log E) SPACE: O(V + E)
class UnionFind:
    """Union-Find data structure with path compression and union by rank."""
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank = [0] * n
    
    def find(self, x: int) -> int:
        # Path compression: make every node point directly to root
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x: int, y: int) -> bool:
        # Union by rank: attach smaller tree under root of larger tree
        px, py = self.find(x), self.find(y)
        if px == py:
            return False  # Already in same set
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        return True

def kruskal_mst(n: int, edges: list[tuple[int, int, int]]) -> list[tuple[int, int, int]]:
    """
    Find MST using Kruskal's algorithm.
    
    Args:
        n: Number of vertices
        edges: List of (weight, u, v) tuples
    
    Returns:
        List of edges in MST
    """
    # Sort edges by weight
    edges.sort(key=lambda e: e[0])
    
    uf = UnionFind(n)
    mst = []
    
    for weight, u, v in edges:
        # Add edge if it doesn't create a cycle
        if uf.union(u, v):
            mst.append((weight, u, v))
            if len(mst) == n - 1:
                break  # MST complete
    
    return mst
```
**Why**: Greedy approach - always pick minimum weight edge that doesn't form cycle. Uses Union-Find for efficient cycle detection.

## Prim's Minimum Spanning Tree

```python
# PROBLEM: Find minimum spanning tree starting from any vertex
# TIME: O(E log V) SPACE: O(V + E)
import heapq

def prim_mst(n: int, adj: list[list[tuple[int, int]]]) -> tuple[int, list[tuple[int, int, int]]]:
    """
    Find MST using Prim's algorithm with priority queue.
    
    Args:
        n: Number of vertices
        adj: Adjacency list where adj[u] = [(v, weight), ...]
    
    Returns:
        (total_weight, list of edges in MST)
    """
    visited = [False] * n
    mst_edges = []
    total_weight = 0
    
    # Start from vertex 0
    # Priority queue: (weight, parent, vertex)
    pq = [(0, -1, 0)]
    
    while pq and len(mst_edges) < n:
        weight, parent, u = heapq.heappop(pq)
        
        if visited[u]:
            continue
        
        visited[u] = True
        if parent != -1:
            mst_edges.append((weight, parent, u))
            total_weight += weight
        
        # Add all edges from current vertex
        for v, w in adj[u]:
            if not visited[v]:
                heapq.heappush(pq, (w, u, v))
    
    return total_weight, mst_edges
```
**Why**: Grows MST one vertex at a time. Always adds minimum weight edge connecting tree to non-tree vertex. More efficient for dense graphs.

## Union-Find with Path Compression

```python
# PROBLEM: Efficiently manage disjoint sets with union and find operations
# TIME: O(α(n)) amortized per operation SPACE: O(n)
class UnionFindOptimized:
    """
    Optimized Union-Find with path compression and union by rank.
    α(n) is inverse Ackermann function, effectively constant.
    """
    
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.size = [1] * n  # Track component sizes
        self.components = n  # Number of connected components
    
    def find(self, x: int) -> int:
        """Find root with full path compression."""
        if self.parent[x] != x:
            # Recursive path compression
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def find_iterative(self, x: int) -> int:
        """Find root with two-pass path compression."""
        root = x
        # Find root
        while self.parent[root] != root:
            root = self.parent[root]
        # Compress path
        while self.parent[x] != root:
            next_node = self.parent[x]
            self.parent[x] = root
            x = next_node
        return root
    
    def union(self, x: int, y: int) -> bool:
        """Union by rank. Returns True if merged, False if already connected."""
        px, py = self.find(x), self.find(y)
        
        if px == py:
            return False
        
        # Union by rank
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        
        self.parent[py] = px
        self.size[px] += self.size[py]
        
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        self.components -= 1
        return True
    
    def connected(self, x: int, y: int) -> bool:
        """Check if x and y are in same component."""
        return self.find(x) == self.find(y)
    
    def get_size(self, x: int) -> int:
        """Get size of component containing x."""
        return self.size[self.find(x)]
```
**Why**: Near-constant time operations make it ideal for Kruskal's MST, dynamic connectivity, and cycle detection in graphs.

## Persistent Segment Tree

```python
# PROBLEM: Query historical versions of segment tree without full copies
# TIME: O(log n) per query/update SPACE: O(n + q log n)
class PersistentSegmentTree:
    """
    Immutable segment tree supporting queries on any historical version.
    Each update creates new path from root to leaf (O(log n) nodes).
    """
    
    class Node:
        def __init__(self, val=0, left=None, right=None):
            self.val = val
            self.left = left
            self.right = right
    
    def __init__(self, arr: list[int]):
        self.n = len(arr)
        self.roots = []  # Store all versions
        self.roots.append(self._build(arr, 0, self.n - 1))
    
    def _build(self, arr: list[int], l: int, r: int) -> Node:
        """Build initial tree."""
        if l == r:
            return self.Node(arr[l])
        
        mid = (l + r) // 2
        left = self._build(arr, l, mid)
        right = self._build(arr, mid + 1, r)
        
        return self.Node(left.val + right.val, left, right)
    
    def update(self, version: int, idx: int, val: int) -> int:
        """
        Create new version with arr[idx] = val.
        Returns index of new version.
        """
        new_root = self._update(self.roots[version], 0, self.n - 1, idx, val)
        self.roots.append(new_root)
        return len(self.roots) - 1
    
    def _update(self, node: Node, l: int, r: int, idx: int, val: int) -> Node:
        """Update with persistence - creates new nodes on path."""
        if l == r:
            return self.Node(val)
        
        mid = (l + r) // 2
        
        if idx <= mid:
            new_left = self._update(node.left, l, mid, idx, val)
            return self.Node(new_left.val + node.right.val, new_left, node.right)
        else:
            new_right = self._update(node.right, mid + 1, r, idx, val)
            return self.Node(node.left.val + new_right.val, node.left, new_right)
    
    def query(self, version: int, ql: int, qr: int) -> int:
        """Query range sum on specific version."""
        return self._query(self.roots[version], 0, self.n - 1, ql, qr)
    
    def _query(self, node: Node, l: int, r: int, ql: int, qr: int) -> int:
        """Standard segment tree query."""
        if qr < l or r < ql:
            return 0
        if ql <= l and r <= qr:
            return node.val
        
        mid = (l + r) // 2
        return (self._query(node.left, l, mid, ql, qr) +
                self._query(node.right, mid + 1, r, ql, qr))
```
**Why**: Enables time-travel queries without storing full copies. Perfect for version control systems, rollback operations, and range queries on historical data.

## Treap (Tree + Heap)

```python
# PROBLEM: Maintain balanced BST with simple randomized balancing
# TIME: O(log n) expected per operation SPACE: O(n)
import random

class TreapNode:
    def __init__(self, key: int, priority: int = None):
        self.key = key
        self.priority = priority if priority else random.randint(1, 10**9)
        self.left = None
        self.right = None

class Treap:
    """
    Randomized BST where keys follow BST property and priorities follow heap property.
    Expected O(log n) height due to randomization.
    """
    
    def rotate_right(self, y: TreapNode) -> TreapNode:
        """Right rotation."""
        x = y.left
        y.left = x.right
        x.right = y
        return x
    
    def rotate_left(self, x: TreapNode) -> TreapNode:
        """Left rotation."""
        y = x.right
        x.right = y.left
        y.left = x
        return y
    
    def insert(self, root: TreapNode, key: int) -> TreapNode:
        """Insert key maintaining BST and heap properties."""
        if not root:
            return TreapNode(key)
        
        if key <= root.key:
            root.left = self.insert(root.left, key)
            # Maintain heap property
            if root.left.priority > root.priority:
                root = self.rotate_right(root)
        else:
            root.right = self.insert(root.right, key)
            # Maintain heap property
            if root.right.priority > root.priority:
                root = self.rotate_left(root)
        
        return root
    
    def delete(self, root: TreapNode, key: int) -> TreapNode:
        """Delete key from treap."""
        if not root:
            return None
        
        if key < root.key:
            root.left = self.delete(root.left, key)
        elif key > root.key:
            root.right = self.delete(root.right, key)
        else:
            # Found key, rotate down to leaf
            if not root.left:
                return root.right
            if not root.right:
                return root.left
            
            # Rotate with child having higher priority
            if root.left.priority > root.right.priority:
                root = self.rotate_right(root)
                root.right = self.delete(root.right, key)
            else:
                root = self.rotate_left(root)
                root.left = self.delete(root.left, key)
        
        return root
    
    def search(self, root: TreapNode, key: int) -> bool:
        """Standard BST search."""
        if not root:
            return False
        if key == root.key:
            return True
        if key < root.key:
            return self.search(root.left, key)
        return self.search(root.right, key)
```
**Why**: Simpler than AVL/Red-Black trees. Randomization provides expected balanced performance without complex rebalancing logic.

## Event Sourcing Pattern

```python
# PROBLEM: Maintain complete audit trail by storing events instead of current state
# TIME: O(1) append, O(n) replay SPACE: O(n) for n events
from dataclasses import dataclass
from datetime import datetime
from typing import Any, Callable
from collections import defaultdict
import json

@dataclass
class Event:
    event_type: str
    aggregate_id: str
    data: dict
    timestamp: datetime
    version: int

class EventStore:
    """
    Append-only store for domain events.
    Current state is derived by replaying events.
    """
    
    def __init__(self):
        self.events: list[Event] = []
        self.subscribers: dict[str, list[Callable]] = defaultdict(list)
        self.snapshots: dict[str, tuple[int, Any]] = {}  # aggregate_id -> (version, state)
    
    def append(self, event: Event) -> None:
        """Append event and notify subscribers."""
        self.events.append(event)
        
        # Notify subscribers
        for handler in self.subscribers[event.event_type]:
            handler(event)
        
        # Also notify wildcard subscribers
        for handler in self.subscribers['*']:
            handler(event)
    
    def get_events(self, aggregate_id: str, from_version: int = 0) -> list[Event]:
        """Get all events for aggregate starting from version."""
        return [
            e for e in self.events
            if e.aggregate_id == aggregate_id and e.version > from_version
        ]
    
    def subscribe(self, event_type: str, handler: Callable) -> None:
        """Subscribe to events of specific type (or '*' for all)."""
        self.subscribers[event_type].append(handler)
    
    def save_snapshot(self, aggregate_id: str, version: int, state: Any) -> None:
        """Save aggregate state at specific version."""
        self.snapshots[aggregate_id] = (version, state)
    
    def get_snapshot(self, aggregate_id: str) -> tuple[int, Any] | None:
        """Get latest snapshot for aggregate."""
        return self.snapshots.get(aggregate_id)


class BankAccount:
    """Example aggregate using event sourcing."""
    
    def __init__(self, account_id: str):
        self.account_id = account_id
        self.balance = 0
        self.version = 0
    
    def apply(self, event: Event) -> None:
        """Apply event to update state."""
        if event.event_type == 'AccountOpened':
            self.balance = 0
        elif event.event_type == 'MoneyDeposited':
            self.balance += event.data['amount']
        elif event.event_type == 'MoneyWithdrawn':
            self.balance -= event.data['amount']
        
        self.version = event.version
    
    @staticmethod
    def rebuild(event_store: EventStore, account_id: str) -> 'BankAccount':
        """Rebuild account state from events."""
        account = BankAccount(account_id)
        
        # Try to load from snapshot
        snapshot = event_store.get_snapshot(account_id)
        if snapshot:
            account.version, account.balance = snapshot
        
        # Replay events since snapshot
        for event in event_store.get_events(account_id, account.version):
            account.apply(event)
        
        return account
```
**Why**: Complete audit trail, temporal queries, easy debugging, and event replay for system recovery. Enables CQRS and powerful analytics.

## LSM Tree Internals

```python
# PROBLEM: Optimize write-heavy workloads with merge-based storage
# TIME: O(log n) write, O(log n) read with bloom filter SPACE: O(n)
from bisect import bisect_left, insort
from collections import defaultdict
import hashlib

class MemTable:
    """In-memory sorted structure (usually skip list or red-black tree)."""
    
    def __init__(self, max_size: int = 1000):
        self.data: dict[str, str | None] = {}  # key -> value (None = tombstone)
        self.size = 0
        self.max_size = max_size
    
    def put(self, key: str, value: str | None) -> bool:
        """Insert/update/delete. Returns True if memtable is full."""
        self.data[key] = value
        self.size += 1
        return self.size >= self.max_size
    
    def get(self, key: str) -> str | None | bool:
        """Get value or None if deleted. Returns (value, found)."""
        if key in self.data:
            return self.data[key], True
        return None, False
    
    def to_sorted_list(self) -> list[tuple[str, str | None]]:
        """Export as sorted list for SSTable."""
        return sorted(self.data.items())


class SSTable:
    """Sorted String Table - immutable sorted file on disk."""
    
    def __init__(self, data: list[tuple[str, str | None]]):
        self.keys = [k for k, v in data]
        self.values = [v for k, v in data]
        self.bloom_filter = BloomFilter()
        
        # Build bloom filter for fast negative lookups
        for key in self.keys:
            self.bloom_filter.add(key)
    
    def get(self, key: str) -> tuple[str | None, bool]:
        """Get value with bloom filter optimization."""
        # Fast negative check
        if key not in self.bloom_filter:
            return None, False
        
        # Binary search
        idx = bisect_left(self.keys, key)
        if idx < len(self.keys) and self.keys[idx] == key:
            return self.values[idx], True
        return None, False


class BloomFilter:
    """Probabilistic filter for membership testing."""
    
    def __init__(self, size: int = 10000, hash_count: int = 7):
        self.size = size
        self.hash_count = hash_count
        self.bit_array = [False] * size
    
    def _hashes(self, item: str) -> list[int]:
        """Generate multiple hash values."""
        hashes = []
        for i in range(self.hash_count):
            h = hashlib.md5(f"{item}:{i}".encode()).hexdigest()
            hashes.append(int(h, 16) % self.size)
        return hashes
    
    def add(self, item: str) -> None:
        """Add item to filter."""
        for h in self._hashes(item):
            self.bit_array[h] = True
    
    def __contains__(self, item: str) -> bool:
        """Check if item might be in set (may have false positives)."""
        return all(self.bit_array[h] for h in self._hashes(item))


class LSMTree:
    """
    Log-Structured Merge Tree.
    Optimized for write-heavy workloads.
    """
    
    def __init__(self, memtable_size: int = 1000):
        self.memtable = MemTable(memtable_size)
        self.immutable_memtables: list[MemTable] = []
        self.sstables: list[SSTable] = []  # Level 0 (newest) to N (oldest)
        self.compaction_threshold = 4
    
    def put(self, key: str, value: str) -> None:
        """Write to memtable, flush if full."""
        if self.memtable.put(key, value):
            # Flush memtable to SSTable
            self._flush_memtable()
    
    def delete(self, key: str) -> None:
        """Tombstone deletion."""
        self.put(key, None)
    
    def get(self, key: str) -> str | None:
        """Read from memtable -> immutable -> SSTables (newest to oldest)."""
        # Check active memtable
        value, found = self.memtable.get(key)
        if found:
            return value  # None means deleted
        
        # Check immutable memtables
        for mt in reversed(self.immutable_memtables):
            value, found = mt.get(key)
            if found:
                return value
        
        # Check SSTables (newest first)
        for sst in self.sstables:
            value, found = sst.get(key)
            if found:
                return value
        
        return None  # Key not found
    
    def _flush_memtable(self) -> None:
        """Convert memtable to SSTable."""
        self.immutable_memtables.append(self.memtable)
        self.memtable = MemTable(self.memtable.max_size)
        
        # Background compaction
        if len(self.immutable_memtables) >= self.compaction_threshold:
            self._compact()
    
    def _compact(self) -> None:
        """Merge multiple SSTables (simplified)."""
        # Merge all immutable memtables into one SSTable
        all_data = []
        for mt in self.immutable_memtables:
            all_data.extend(mt.to_sorted_list())
        
        # Sort and dedupe (keep newest)
        merged = {}
        for k, v in sorted(all_data):
            merged[k] = v
        
        # Create new SSTable
        sst = SSTable(list(merged.items()))
        self.sstables.insert(0, sst)  # Add to front (newest)
        self.immutable_memtables.clear()
```
**Why**: Enables fast writes via sequential I/O. Used in RocksDB, Cassandra, LevelDB. Bloom filters minimize disk reads for non-existent keys.

## Lock-Free Stack

```python
# PROBLEM: Thread-safe stack without locks for high concurrency
# TIME: O(1) per operation SPACE: O(n)
import threading
from typing import Generic, TypeVar, Optional
from dataclasses import dataclass

T = TypeVar('T')

@dataclass
class Node(Generic[T]):
    value: T
    next: Optional['Node[T]'] = None

class LockFreeStack(Generic[T]):
    """
    Lock-free stack using CAS (Compare-And-Swap) operations.
    Implements Treiber's algorithm.
    """
    
    def __init__(self):
        self.head: Optional[Node[T]] = None
        # In Python, we use a lock for CAS simulation
        # Real implementations use atomic CAS instructions
        self._cas_lock = threading.Lock()
    
    def _cas(self, expected: Optional[Node[T]], new_value: Optional[Node[T]]) -> bool:
        """Compare-and-swap operation (simulated in Python)."""
        with self._cas_lock:
            if self.head == expected:
                self.head = new_value
                return True
            return False
    
    def push(self, value: T) -> None:
        """Push value onto stack (lock-free)."""
        new_node = Node(value)
        
        while True:
            old_head = self.head
            new_node.next = old_head
            
            if self._cas(old_head, new_node):
                return
            # CAS failed, retry with updated head
    
    def pop(self) -> Optional[T]:
        """Pop value from stack (lock-free)."""
        while True:
            old_head = self.head
            
            if old_head is None:
                return None  # Stack empty
            
            new_head = old_head.next
            
            if self._cas(old_head, new_head):
                return old_head.value
            # CAS failed, retry
    
    def peek(self) -> Optional[T]:
        """Peek at top value."""
        head = self.head
        return head.value if head else None


class LockFreeQueue(Generic[T]):
    """
    Lock-free queue using Michael-Scott algorithm.
    """
    
    def __init__(self):
        dummy = Node(None)  # Sentinel node
        self.head: Node[T] = dummy
        self.tail: Node[T] = dummy
        self._lock = threading.Lock()
    
    def enqueue(self, value: T) -> None:
        """Add to back of queue."""
        new_node = Node(value)
        
        with self._lock:  # Simplified for Python
            self.tail.next = new_node
            self.tail = new_node
    
    def dequeue(self) -> Optional[T]:
        """Remove from front of queue."""
        with self._lock:
            if self.head.next is None:
                return None  # Queue empty
            
            value = self.head.next.value
            self.head = self.head.next
            return value
```

## Unit of Work Pattern

```python
# PROBLEM: Coordinate multiple database operations as single atomic unit
# TIME: O(1) per operation, O(n) commit SPACE: O(n) for tracked changes
from dataclasses import dataclass, field
from typing import Any, Callable, Protocol
from enum import Enum
from collections import defaultdict

class OperationType(Enum):
    INSERT = "insert"
    UPDATE = "update"
    DELETE = "delete"

@dataclass
class Change:
    entity_type: str
    entity_id: Any
    operation: OperationType
    old_value: dict | None = None
    new_value: dict | None = None

class Repository(Protocol):
    """Repository interface for entities."""
    def add(self, entity: Any) -> None: ...
    def update(self, entity: Any) -> None: ...
    def delete(self, entity: Any) -> None: ...
    def get_by_id(self, entity_id: Any) -> Any | None: ...

class UnitOfWork:
    """
    Tracks changes and coordinates commit across repositories.
    Implements transaction script pattern.
    """
    
    def __init__(self):
        self.changes: list[Change] = []
        self.identity_map: dict[tuple[str, Any], Any] = {}  # Cache
        self.new_entities: list[Any] = []
        self.dirty_entities: list[Any] = []
        self.deleted_entities: list[Any] = []
        self.repositories: dict[str, Repository] = {}
        self._committed = False
        self._rolled_back = False
    
    def register_repository(self, name: str, repo: Repository) -> None:
        """Register repository for entity type."""
        self.repositories[name] = repo
    
    def register_new(self, entity: Any, entity_type: str) -> None:
        """Register entity for insertion."""
        self.new_entities.append((entity_type, entity))
        self.changes.append(Change(
            entity_type=entity_type,
            entity_id=getattr(entity, 'id', None),
            operation=OperationType.INSERT,
            new_value=self._entity_to_dict(entity)
        ))
    
    def register_dirty(self, entity: Any, entity_type: str, 
                       old_value: dict | None = None) -> None:
        """Register entity for update."""
        self.dirty_entities.append((entity_type, entity))
        self.changes.append(Change(
            entity_type=entity_type,
            entity_id=getattr(entity, 'id', None),
            operation=OperationType.UPDATE,
            old_value=old_value,
            new_value=self._entity_to_dict(entity)
        ))
    
    def register_deleted(self, entity: Any, entity_type: str) -> None:
        """Register entity for deletion."""
        self.deleted_entities.append((entity_type, entity))
        self.changes.append(Change(
            entity_type=entity_type,
            entity_id=getattr(entity, 'id', None),
            operation=OperationType.DELETE,
            old_value=self._entity_to_dict(entity)
        ))
    
    def get_by_id(self, entity_type: str, entity_id: Any) -> Any | None:
        """Get entity from identity map or repository."""
        key = (entity_type, entity_id)
        
        if key in self.identity_map:
            return self.identity_map[key]
        
        repo = self.repositories.get(entity_type)
        if repo:
            entity = repo.get_by_id(entity_id)
            if entity:
                self.identity_map[key] = entity
            return entity
        
        return None
    
    def commit(self) -> None:
        """Commit all tracked changes."""
        if self._committed or self._rolled_back:
            raise RuntimeError("Unit of work already finalized")
        
        try:
            # Insert new entities
            for entity_type, entity in self.new_entities:
                repo = self.repositories[entity_type]
                repo.add(entity)
            
            # Update dirty entities
            for entity_type, entity in self.dirty_entities:
                repo = self.repositories[entity_type]
                repo.update(entity)
            
            # Delete removed entities
            for entity_type, entity in self.deleted_entities:
                repo = self.repositories[entity_type]
                repo.delete(entity)
            
            self._committed = True
            
        except Exception as e:
            self.rollback()
            raise
    
    def rollback(self) -> None:
        """Discard all tracked changes."""
        self._rolled_back = True
        self.changes.clear()
        self.new_entities.clear()
        self.dirty_entities.clear()
        self.deleted_entities.clear()
    
    def _entity_to_dict(self, entity: Any) -> dict:
        """Convert entity to dictionary."""
        return {
            k: v for k, v in entity.__dict__.items()
            if not k.startswith('_')
        }


# Usage with context manager
class UnitOfWorkManager:
    """Manages unit of work lifecycle."""
    
    def __enter__(self) -> UnitOfWork:
        self.uow = UnitOfWork()
        return self.uow
    
    def __exit__(self, exc_type, exc_val, exc_tb) -> None:
        if exc_type:
            self.uow.rollback()
        elif not self.uow._committed:
            self.uow.commit()

# Example usage:
"""
with UnitOfWorkManager() as uow:
    uow.register_repository('users', user_repo)
    uow.register_repository('orders', order_repo)
    
    user = User(id=1, name="Alice")
    uow.register_new(user, 'users')
    
    order = Order(id=1, user_id=1, amount=100)
    uow.register_new(order, 'orders')

"""
```
