## 1. CSS Place Content
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>CSS Place Content</title>
  <style>
    /* 
      PROBLEM: Center content both horizontally and vertically
      TIME: O(1) SPACE: O(1)
      
      place-content is a shorthand for align-content and justify-content
      It works in Grid and Flexbox containers
    */
    
    .container {
      display: grid;
      place-content: center;
      height: 300px;
      background: #f0f0f0;
      border: 2px solid #333;
    }
    
    .item {
      padding: 20px;
      background: #007bff;
      color: white;
      border-radius: 8px;
    }
  </style>
</head>
<body>
  <div class="container">
    <div class="item">Centered Content</div>
  </div>
</body>
</html>
```
**Why**: Simplifies centering alignment in Grid/Flexbox layouts with a single property.

## 4. JavaScript JSON Methods
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>JavaScript JSON Methods</title>
</head>
<body>
  <script>
    /* 
      PROBLEM: Serialize and deserialize JavaScript objects
      TIME: O(n) SPACE: O(n)
      
      JSON.stringify() - Convert object to JSON string
      JSON.parse() - Convert JSON string to object
    */
    
    // Original object
    const user = {
      name: 'Alice',
      age: 28,
      skills: ['JavaScript', 'Python', 'React'],
      active: true
    };
    
    // Serialize to JSON string
    const jsonString = JSON.stringify(user, null, 2);
    console.log('JSON String:', jsonString);
    
    // Deserialize back to object
    const parsedUser = JSON.parse(jsonString);
    console.log('Parsed Object:', parsedUser);
    
    // Filter properties during serialization
    const filtered = JSON.stringify(user, ['name', 'age'], 2);
    console.log('Filtered JSON:', filtered);
    
    // Custom toJSON method
    const product = {
      name: 'Laptop',
      price: 999,
      toJSON() {
        return {
          productName: this.name,
          productPrice: `$${this.price}`
        };
      }
    };
    
    console.log('Custom JSON:', JSON.stringify(product));
  </script>
</body>
</html>
```
**Why**: Essential for data exchange, API communication, and localStorage persistence.

## 5. DOM Template Content
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>DOM Template Content</title>
</head>
<body>
  <!-- 
    PROBLEM: Create reusable HTML templates
    TIME: O(1) SPACE: O(1)
    
    <template> elements are inert until cloned
    Perfect for creating reusable component templates
  -->
  
  <template id="card-template">
    <div class="card">
      <h2 class="card-title"></h2>
      <p class="card-content"></p>
    </div>
  </template>
  
  <div id="card-container"></div>
  
  <script>
    const template = document.getElementById('card-template');
    const container = document.getElementById('card-container');
    
    const cards = [
      { title: 'Card 1', content: 'First card content' },
      { title: 'Card 2', content: 'Second card content' },
      { title: 'Card 3', content: 'Third card content' }
    ];
    
    cards.forEach(card => {
      // Clone the template
      const clone = template.content.cloneNode(true);
      
      // Populate the cloned template
      clone.querySelector('.card-title').textContent = card.title;
      clone.querySelector('.card-content').textContent = card.content;
      
      // Append to container
      container.appendChild(clone);
    });
  </script>
</body>
</html>
```
**Why**: Efficient way to create reusable HTML components without string manipulation.

## 6. DOM Resize Observer
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>DOM Resize Observer</title>
  <style>
    /* 
      PROBLEM: Detect element size changes
      TIME: O(1) SPACE: O(1)
    */
    
    .resizable {
      resize: both;
      overflow: hidden;
      width: 200px;
      height: 200px;
      background: #6c757d;
      color: white;
      padding: 20px;
      border: 2px solid #495057;
    }
  </style>
</head>
<body>
  <div class="resizable" id="observed-element">
    Resize me! Current size will be displayed below.
  </div>
  <p id="size-display">Size: 200px × 200px</p>
  
  <script>
    const element = document.getElementById('observed-element');
    const sizeDisplay = document.getElementById('size-display');
    
    // Create ResizeObserver
    const resizeObserver = new ResizeObserver(entries => {
      for (let entry of entries) {
        const { width, height } = entry.contentRect;
        sizeDisplay.textContent = `Size: ${Math.round(width)}px × ${Math.round(height)}px`;
      }
    });
    
    // Start observing
    resizeObserver.observe(element);
    
    // Cleanup when needed
    // resizeObserver.unobserve(element);
    // resizeObserver.disconnect();
  </script>
</body>
</html>
```
**Why**: Provides efficient way to react to element size changes without polling or resize events.

## 7. Event Stop Propagation
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Event Stop Propagation</title>
  <style>
    /* 
      PROBLEM: Prevent event bubbling or capturing
      TIME: O(1) SPACE: O(1)
    */
    
    .outer {
      padding: 40px;
      background: #17a2b8;
      cursor: pointer;
    }
    
    .middle {
      padding: 30px;
      background: #20c997;
      cursor: pointer;
    }
    
    .inner {
      padding: 20px;
      background: #fd7e14;
      cursor: pointer;
      color: white;
    }
  </style>
</head>
<body>
  <div class="outer" id="outer">
    Outer (Click me!)
    <div class="middle" id="middle">
      Middle (Click me!)
      <div class="inner" id="inner">
        Inner (Click me!)
      </div>
    </div>
  </div>
  
  <script>
    const outer = document.getElementById('outer');
    const middle = document.getElementById('middle');
    const inner = document.getElementById('inner');
    
    // Without stopPropagation - all three fire
    outer.addEventListener('click', () => {
      console.log('Outer clicked');
    });
    
    middle.addEventListener('click', () => {
      console.log('Middle clicked');
    });
    
    // With stopPropagation - only inner fires
    inner.addEventListener('click', (event) => {
      event.stopPropagation();
      console.log('Inner clicked (propagation stopped)');
    });
  </script>
</body>
</html>
```
**Why**: Essential for controlling event flow in nested interactive elements.

## 8. Event Animation
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Event Animation</title>
  <style>
    /* 
      PROBLEM: Detect CSS animation lifecycle
      TIME: O(1) SPACE: O(1)
    */
    
    .animated-box {
      width: 100px;
      height: 100px;
      background: #e83e8c;
      animation: pulse 2s ease-in-out;
    }
    
    @keyframes pulse {
      0%, 100% { transform: scale(1); opacity: 1; }
      50% { transform: scale(1.5); opacity: 0.5; }
    }
  </style>
</head>
<body>
  <button id="start-btn">Start Animation</button>
  <div id="box" class="animated-box"></div>
  <p id="status">Status: Waiting...</p>
  
  <script>
    const box = document.getElementById('box');
    const status = document.getElementById('status');
    const startBtn = document.getElementById('start-btn');
    
    // Listen for animation start
    box.addEventListener('animationstart', () => {
      status.textContent = 'Status: Animation started';
      console.log('Animation started');
    });
    
    // Listen for animation iteration
    box.addEventListener('animationiteration', () => {
      status.textContent = 'Status: Animation iteration';
      console.log('Animation iteration');
    });
    
    // Listen for animation end
    box.addEventListener('animationend', () => {
      status.textContent = 'Status: Animation completed';
      console.log('Animation completed');
    });
    
    // Restart animation on button click
    startBtn.addEventListener('click', () => {
      box.style.animation = 'none';
      // Trigger reflow
      box.offsetHeight;
      box.style.animation = 'pulse 2s ease-in-out';
    });
  </script>
</body>
</html>
```
**Why**: Enables JavaScript to respond to CSS animation lifecycle events.

## 9. Event Transition
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Event Transition</title>
  <style>
    /* 
      PROBLEM: Detect CSS transition completion
      TIME: O(1) SPACE: O(1)
    */
    
    .box {
      width: 100px;
      height: 100px;
      background: #6f42c1;
      transition: all 0.5s ease;
      cursor: pointer;
    }
    
    .box.active {
      width: 200px;
      height: 200px;
      background: #d63384;
      border-radius: 50%;
    }
  </style>
</head>
<body>
  <div id="transition-box" class="box"></div>
  <p id="transition-status">Click the box to toggle</p>
  
  <script>
    const box = document.getElementById('transition-box');
    const status = document.getElementById('transition-status');
    
    let isToggled = false;
    
    box.addEventListener('click', () => {
      isToggled = !isToggled;
      box.classList.toggle('active', isToggled);
      status.textContent = 'Transition running...';
    });
    
    // Listen for transition start
    box.addEventListener('transitionstart', () => {
      console.log('Transition started');
    });
    
    // Listen for transition run (repeated during transition)
    box.addEventListener('transitionrun', () => {
      console.log('Transition running');
    });
    
    // Listen for transition end
    box.addEventListener('transitionend', (e) => {
      console.log(`Transition ended for ${e.propertyName}`);
      status.textContent = 'Transition completed!';
    });
    
    // Listen for transition cancel
    box.addEventListener('transitioncancel', () => {
      console.log('Transition cancelled');
    });
  </script>
</body>
</html>
```
**Why**: Allows JavaScript to synchronize with CSS transition completion.

## 10. Event Clipboard
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Event Clipboard</title>
  <style>
    /* 
      PROBLEM: Handle copy, cut, and paste events
      TIME: O(1) SPACE: O(1)
    */
    
    textarea {
      width: 100%;
      height: 100px;
      margin: 10px 0;
      padding: 10px;
    }
    
    button {
      padding: 10px 20px;
      margin: 5px;
      background: #007bff;
      color: white;
      border: none;
      border-radius: 4px;
      cursor: pointer;
    }
    
    button:hover {
      background: #0056b3;
    }
  </style>
</head>
<body>
  <h2>Clipboard Events Demo</h2>
  
  <textarea id="text-area" placeholder="Type something and try copy/cut/paste..."></textarea>
  
  <button id="copy-btn">Copy Text</button>
  <button id="cut-btn">Cut Text</button>
  <button id="paste-btn">Paste Text</button>
  
  <div id="log"></div>
  
  <script>
    const textarea = document.getElementById('text-area');
    const log = document.getElementById('log');
    const copyBtn = document.getElementById('copy-btn');
    const cutBtn = document.getElementById('cut-btn');
    const pasteBtn = document.getElementById('paste-btn');
    
    function addLog(message) {
      const p = document.createElement('p');
      p.textContent = `${new Date().toLocaleTimeString()}: ${message}`;
      log.appendChild(p);
    }
    
    // Listen for copy event
    textarea.addEventListener('copy', (e) => {
      addLog('Copy event triggered');
      // You can modify the copied data
      // e.clipboardData.setData('text/plain', 'Modified: ' + document.getSelection());
    });
    
    // Listen for cut event
    textarea.addEventListener('cut', (e) => {
      addLog('Cut event triggered');
    });
    
    // Listen for paste event
    textarea.addEventListener('paste', (e) => {
      addLog('Paste event triggered');
      // You can read or modify pasted data
      // const pastedData = e.clipboardData.getData('text');
      // console.log('Pasted:', pastedData);
    });
    
    // Programmatic clipboard operations
    copyBtn.addEventListener('click', async () => {
      try {
        await navigator.clipboard.writeText(textarea.value);
        addLog('Text copied to clipboard');
      } catch (err) {
        addLog('Failed to copy: ' + err);
      }
    });
    
    cutBtn.addEventListener('click', async () => {
      try {
        await navigator.clipboard.writeText(textarea.value);
        textarea.value = '';
        addLog('Text cut to clipboard');
      } catch (err) {
        addLog('Failed to cut: ' + err);
      }
    });
    
    pasteBtn.addEventListener('click', async () => {
      try {
        const text = await navigator.clipboard.readText();
        textarea.value += text;
        addLog('Text pasted from clipboard');
      } catch (err) {
        addLog('Failed to paste: ' + err);
      }
    });
  </script>
</body>
</html>
```
**Why**: Enables custom handling of clipboard operations and clipboard API integration.
