   * Sets up all event listeners for audio controls.
   * @returns {void}

    this.audio.addEventListener('loadedmetadata', () => this.updateDuration());
    this.audio.addEventListener('ended', () => this.handleAudioEnd());
    this.volumeSlider.addEventListener('input', (e) => this.setVolume(e.target.value));

      this.playPauseBtn.setAttribute('aria-label', 'Pause');
    } else {
      this.audio.pause();
      this.playPauseBtn.textContent = '▶️';
      this.playPauseBtn.setAttribute('aria-label', 'Play');
    }
  }

   * Updates the progress bar based on current playback position.
   * @returns {void}

    this.progressBar.setAttribute('aria-valuenow', Math.round(progress));
    this.currentTimeDisplay.textContent = this.formatTime(this.audio.currentTime);
  }
  
  /**
   * Updates the duration display when audio metadata loads.
   * @returns {void}
   */
  updateDuration() {
    this.durationDisplay.textContent = this.formatTime(this.audio.duration);
  }
  
  /**
   * Seeks to a position based on click location.
   * @param {MouseEvent} e - Click event
   * @returns {void}
   */
  seek(e) {
    const container = e.currentTarget;
    const clickX = e.offsetX;
    const width = container.offsetWidth;
    const seekTime = (clickX / width) * this.audio.duration;
    this.audio.currentTime = seekTime;
  }
  
  /**
   * Sets the audio volume.

   * @param {number} seconds - Time in seconds

    const secs = Math.floor(seconds % 60);

/**
 * Manages subtitle language selection for video tracks.
 * Provides programmatic control over text track display.
 */
class SubtitleManager {
  constructor(videoId) {
    this.video = document.getElementById(videoId);
    this.selector = document.getElementById('subtitle-lang');
    
    if (!this.video || !this.selector) {
      console.error('Video or selector not found');
      return;
    }
    
    this.init();
  }
  
  /**
   * Initializes event listeners and default state.
   * @returns {void}
   */
  init() {
    // Listen for language selection changes
    this.selector.addEventListener('change', (e) => {
      this.setSubtitleLanguage(e.target.value);
    });
    
    // Detect browser language and set default if available
    this.detectPreferredLanguage();
  }
  
  /**
   * Sets the active subtitle track based on language code.
   * @param {string} langCode - ISO language code (e.g., 'en', 'es')
   * @returns {void}
   */
  setSubtitleLanguage(langCode) {
    const tracks = this.video.textTracks;
    
    // Disable all tracks first
    for (let i = 0; i < tracks.length; i++) {
      tracks[i].mode = 'disabled';
    }
    
    // Enable selected track
    if (langCode !== 'none') {
      for (let i = 0; i < tracks.length; i++) {
        if (tracks[i].language === langCode) {
          tracks[i].mode = 'showing';
          console.log(`Subtitles set to: ${tracks[i].label}`);
          break;
        }
      }
    }
  }
  
  /**
   * Detects user's preferred language and applies it if available.
   * @returns {void}
   */
  detectPreferredLanguage() {
    const browserLang = navigator.language.split('-')[0]; // 'en-US' -> 'en'
    const availableLangs = Array.from(this.video.textTracks)
      .map(track => track.language);
    
    if (availableLangs.includes(browserLang)) {
      this.selector.value = browserLang;
      this.setSubtitleLanguage(browserLang);
    }
  }
}

// Initialize subtitle manager
document.addEventListener('DOMContentLoaded', () => {
  new SubtitleManager('multilingual-video');
});
</script>

/**
 * Utility to display current subtitle text outside the video player.
 * Useful for custom subtitle displays or accessibility tools.
 * 
 * @param {HTMLVideoElement} video - Video element with text tracks
 * @param {HTMLElement} display - Element to show subtitle text
 */
function setupExternalSubtitleDisplay(video, display) {
  if (!video || !display) return;
  
  // Listen for cue changes in active text track
  video.addEventListener('cuechange', () => {
    const activeTrack = Array.from(video.textTracks)
      .find(track => track.mode === 'showing');
    
    if (activeTrack && activeTrack.activeCues.length > 0) {
      const currentCue = activeTrack.activeCues[0];
      display.textContent = currentCue.text;
    } else {
      display.textContent = '';
    }
  });
}

/**
 * Validates that table headers properly use scope attributes.
 * Helps ensure accessibility compliance.
 * 
 * @param {HTMLTableElement} table - Table to validate
 * @returns {Object} Validation results with warnings
 */
function validateTableScope(table) {
  const results = {
    valid: true,
    warnings: []
  };
  
  // Check for colgroup headers
  const colgroupHeaders = table.querySelectorAll('th[scope="colgroup"]');
  const colgroups = table.querySelectorAll('colgroup');
  
  if (colgroupHeaders.length > 0 && colgroups.length === 0) {
    results.warnings.push('colgroup scope used but no colgroup elements defined');
    results.valid = false;
  }
  
  // Check for row headers
  const rowHeaders = table.querySelectorAll('tbody th');
  rowHeaders.forEach((th, index) => {
    if (!th.hasAttribute('scope')) {
      results.warnings.push(`Row header ${index + 1} missing scope attribute`);
    }
  });
  
  console.log('Table Scope Validation:', results);
  return results;
}

  const table = document.querySelector('.financial-report');

/**
 * Toggles between border-collapse models for demonstration.
 * Useful for comparing table rendering options.
 * 
 * @param {HTMLTableElement} table - Table element to modify
 * @param {string} model - 'collapse' or 'separate'
 */
function setBorderModel(table, model) {
  if (model === 'collapse') {
    table.style.borderCollapse = 'collapse';
    table.style.borderSpacing = '0';
  } else if (model === 'separate') {
    table.style.borderCollapse = 'separate';
    table.style.borderSpacing = '5px';
  }
  
  console.log(`Border model set to: ${model}`);
}

/**
 * Calculates total border width for a table cell.
 * Important for precise layout calculations.
 * 
 * @param {HTMLTableCellElement} cell - Table cell to measure
 * @returns {number} Total border width in pixels
 */
function calculateCellBorderWidth(cell) {
  const style = window.getComputedStyle(cell);
  const table = cell.closest('table');
  const tableStyle = window.getComputedStyle(table);
  
  if (tableStyle.borderCollapse === 'collapse') {
    // In collapsed model, borders are shared
    return parseFloat(style.borderWidth);
  } else {
    // In separate model, each cell has its own border
    return parseFloat(style.borderWidth) * 2 + parseFloat(tableStyle.borderSpacing);
  }
}

/**
 * Accessible Tablist Controller.
 * Implements WAI-ARIA Authoring Practices for tabs.
 */
class TablistController {
  constructor(container) {
    this.container = container;
    this.tabs = Array.from(container.querySelectorAll('[role="tab"]'));
    this.panels = Array.from(container.querySelectorAll('[role="tabpanel"]'));
    
    this.init();
  }
  
  /**
   * Initializes event listeners and sets up keyboard navigation.
   * @returns {void}
   */
  init() {
    // Click events
    this.tabs.forEach(tab => {
      tab.addEventListener('click', () => this.selectTab(tab));
    });
  
    // Keyboard navigation
    this.container.addEventListener('keydown', (e) => this.handleKeydown(e));
  }
  
  /**
   * Selects a tab and shows its associated panel.
   * @param {HTMLElement} selectedTab - Tab element to activate
   * @returns {void}
   */
  selectTab(selectedTab) {
    // Update all tabs
    this.tabs.forEach(tab => {
      const isSelected = tab === selectedTab;
      tab.setAttribute('aria-selected', isSelected);
      tab.setAttribute('tabindex', isSelected ? '0' : '-1');
    });
  
    // Update all panels
    this.panels.forEach(panel => {
      const isActive = panel.id === selectedTab.getAttribute('aria-controls');
      panel.hidden = !isActive;
    });
  
    // Move focus to selected tab
    selectedTab.focus();
  }
  
  /**
   * Handles keyboard navigation for accessibility.
   * @param {KeyboardEvent} e - Keyboard event
   * @returns {void}
   */
  handleKeydown(e) {
    const currentTab = document.activeElement;
    const currentIndex = this.tabs.indexOf(currentTab);
  
    if (currentIndex === -1) return;
  
    let newIndex;
  
    switch (e.key) {
      case 'ArrowRight':
      case 'ArrowDown':
        e.preventDefault();
        newIndex = (currentIndex + 1) % this.tabs.length;
        this.tabs[newIndex].focus();
        break;
  
      case 'ArrowLeft':
      case 'ArrowUp':
        e.preventDefault();
        newIndex = (currentIndex - 1 + this.tabs.length) % this.tabs.length;
        this.tabs[newIndex].focus();
        break;
  
      case 'Home':
        e.preventDefault();
        this.tabs[0].focus();
        break;
  
      case 'End':
        e.preventDefault();
        this.tabs[this.tabs.length - 1].focus();
        break;
  
      case 'Enter':
      case ' ':
        e.preventDefault();
        this.selectTab(currentTab);
        break;
    }
  }
}

// Initialize all tablists on page
document.addEventListener('DOMContentLoaded', () => {

    const container = tablist.closest('.tabs-container');
    if (container) {
      new TablistController(container);
    }
  });
});
</script>

/**
 * Extracts and validates author metadata from the document.
 * Useful for content management systems and SEO tools.
 * 
 * @returns {Object} Author information object
 */
function extractAuthorMetadata() {
  const authorMeta = document.querySelector('meta[name="author"]');
  const articleAuthor = document.querySelector('meta[property="article:author"]');
  const twitterCreator = document.querySelector('meta[name="twitter:creator"]');
  const authorLink = document.querySelector('a[rel="author"]');
  
  return {
    // Basic author name
    name: authorMeta ? authorMeta.content : null,
    
    // Social profiles
    facebook: articleAuthor ? articleAuthor.content : null,
    twitter: twitterCreator ? twitterCreator.content : null,
    website: authorLink ? authorLink.href : null,
    
    // Validation
    hasAllMetadata: !!(authorMeta && articleAuthor && twitterCreator),
    
    /**
     * Generates a formatted author attribution string.
     * @returns {string} Formatted author credit
     */
    getAttribution() {
      const parts = [];
      if (this.name) parts.push(this.name);
      if (this.twitter) parts.push(`(${this.twitter})`);
      return parts.join(' ');
    }
  };
}

/**
 * Dynamically updates author meta tags.
 * Useful for single-page applications with multiple authors.
 * 
 * @param {Object} authorData - Author information
 * @returns {void}
 */
function updateAuthorMetadata(authorData) {
  // Update or create basic author meta
  let authorMeta = document.querySelector('meta[name="author"]');
  if (!authorMeta) {
    authorMeta = document.createElement('meta');
    authorMeta.name = 'author';
    document.head.appendChild(authorMeta);
  }
  authorMeta.content = authorData.name;
  
  // Update structured data
  const existingScript = document.querySelector('script[type="application/ld+json"]');
  if (existingScript && authorData.schema) {
    existingScript.textContent = JSON.stringify(authorData.schema);
  }
  
  console.log('Author metadata updated:', authorData.name);
}

  console.log('Complete metadata:', authorInfo.hasAllMetadata);
});
