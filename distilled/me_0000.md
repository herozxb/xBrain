## String Compression

```python
# PROBLEM: Compress a string using run-length encoding (e.g., "aaabbc" → "a3b2c1")
# TIME: O(n) SPACE: O(n) for output string
def string_compression(s: str) -> str:
    """
    Compress string using run-length encoding.
    Returns compressed string if shorter than original, else original.
    """
    if not s:
        return s
    
    # Initialize compression variables
    compressed = []
    count = 1
    n = len(s)
    
    # Iterate through string, counting consecutive characters
    for i in range(1, n):
        if s[i] == s[i - 1]:
            count += 1
        else:
            # Append character and its count to result
            compressed.append(s[i - 1] + str(count))
            count = 1
    
    # Don't forget the last group
    compressed.append(s[-1] + str(count))
    
    # Join and return if shorter than original
    result = ''.join(compressed)
    return result if len(result) < len(s) else s
```

**Why**: Fundamental string processing technique used in file compression, image encoding, and data transmission. Demonstrates single-pass iteration and in-place counting.

## Determinant Matrix

```python
# PROBLEM: Calculate the determinant of an n×n matrix
# TIME: O(n!) for naive recursion, O(n³) for LU decomposition
# SPACE: O(n²) for matrix storage
def determinant_matrix(matrix: list[list[float]]) -> float:
    """
    Calculate determinant using recursive cofactor expansion.
    Works for any n×n matrix.
    """
    n = len(matrix)
    
    # Base case: 1×1 matrix
    if n == 1:
        return matrix[0][0]
    
    # Base case: 2×2 matrix (ad - bc)
    if n == 2:
        return matrix[0][0] * matrix[1][1] - matrix[0][1] * matrix[1][0]
    
    det = 0.0
    
    # Expand along first row using cofactor expansion
    for col in range(n):
        # Calculate minor by removing first row and current column
        minor = []
        for i in range(1, n):
            minor_row = []
            for j in range(n):
                if j != col:
                    minor_row.append(matrix[i][j])
            minor.append(minor_row)
        
        # Cofactor sign: (-1)^(row+col)
        sign = 1 if col % 2 == 0 else -1
        
        # Recursive determinant calculation
        det += sign * matrix[0][col] * determinant_matrix(minor)
    
    return det
```

**Why**: Core linear algebra operation essential for solving systems of equations, calculating matrix inverses, and determining if a matrix is invertible. Foundation for graphics transformations and ML algorithms.

## Rat Maze

```python
# PROBLEM: Find a path from top-left to bottom-right in a maze with obstacles
# TIME: O(4^(n²)) worst case, where n is maze dimension
# SPACE: O(n²) for solution matrix and recursion stack
def rat_maze(maze: list[list[int]]) -> list[list[int]] | None:
    """
    Solve maze using backtracking. 
    1 = valid path, 0 = blocked cell.
    Returns solution matrix or None if no path exists.
    """
    n = len(maze)
    
    # Initialize solution matrix with zeros
    solution = [[0] * n for _ in range(n)]
    
    def is_safe(x: int, y: int) -> bool:
        """Check if position (x, y) is valid and not blocked."""
        return 0 <= x < n and 0 <= y < n and maze[x][y] == 1 and solution[x][y] == 0
    
    def solve(x: int, y: int) -> bool:
        """Recursively find path using backtracking."""
        # Base case: reached destination
        if x == n - 1 and y == n - 1:
            solution[x][y] = 1
            return True
        
        # Check if current position is safe
        if is_safe(x, y):
            # Mark current cell as part of solution path
            solution[x][y] = 1
            
            # Try moving right
            if solve(x, y + 1):
                return True
            
            # Try moving down
            if solve(x + 1, y):
                return True
            
            # Try moving up (if needed)
            if solve(x - 1, y):
                return True
            
            # Try moving left (if needed)
            if solve(x, y - 1):
                return True
            
            # Backtrack: unmark current cell
            solution[x][y] = 0
            return False
        
        return False
    
    # Start from top-left corner
    if solve(0, 0):
        return solution
    return None
```

**Why**: Classic backtracking problem demonstrating state-space search, constraint satisfaction, and systematic exploration. Used in robotics path planning and game AI.

## Predict Winner

```python
# PROBLEM: Predict if player 1 can win a game where players pick from array ends
# TIME: O(n²) SPACE: O(n²)
def predict_winner(nums: list[int]) -> bool:
    """
    Two players take turns picking from either end of array.
    Player with higher sum wins. Determine if player 1 can win.
    Uses dynamic programming to calculate maximum score difference.
    """
    n = len(nums)
    
    # dp[i][j] = maximum score difference player 1 can achieve 
    # over player 2 for subarray nums[i:j+1]
    dp = [[0] * n for _ in range(n)]
    
    # Base case: single element - player 1 takes it
    for i in range(n):
        dp[i][i] = nums[i]
    
    # Fill DP table for subarrays of increasing length
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            
            # Player 1 can pick either end
            # If picks nums[i], player 2 plays optimally on remaining
            # Score difference = nums[i] - optimal_diff_of_remaining
            pick_left = nums[i] - dp[i + 1][j]
            pick_right = nums[j] - dp[i][j - 1]
            
            # Player 1 chooses optimally
            dp[i][j] = max(pick_left, pick_right)
    
    # Player 1 wins if final difference >= 0
    return dp[0][n - 1] >= 0
```

**Why**: Demonstrates game theory DP, optimal substructure, and minimax thinking. Essential for understanding competitive game algorithms and decision-making under adversarial conditions.

## Sudoku Solver Backtrack

```python
# PROBLEM: Solve a 9×9 Sudoku puzzle using backtracking
# TIME: O(9^(n²)) worst case, but much better with constraint propagation
# SPACE: O(n²) for board
def sudoku_solver_backtrack(board: list[list[int]]) -> bool:
    """
    Solve Sudoku in-place using backtracking with constraint checking.
    0 represents empty cell. Returns True if solution exists.
    """
    def is_valid(row: int, col: int, num: int) -> bool:
        """Check if placing num at (row, col) is valid."""
        # Check row
        if num in board[row]:
            return False
        
        # Check column
        if num in [board[i][col] for i in range(9)]:
            return False
        
        # Check 3×3 sub-box
        box_row, box_col = 3 * (row // 3), 3 * (col // 3)
        for i in range(box_row, box_row + 3):
            for j in range(box_col, box_col + 3):
                if board[i][j] == num:
                    return False
        
        return True
    
    def find_empty() -> tuple[int, int] | None:
        """Find next empty cell (with value 0)."""
        for i in range(9):
            for j in range(9):
                if board[i][j] == 0:
                    return (i, j)
        return None
    
    def solve() -> bool:
        """Recursive backtracking solver."""
        empty = find_empty()
        
        # No empty cell means puzzle is solved
        if not empty:
            return True
        
        row, col = empty
        
        # Try digits 1-9
        for num in range(1, 10):
            if is_valid(row, col, num):
                # Place digit
                board[row][col] = num
                
                # Recursively solve rest
                if solve():
                    return True
                
                # Backtrack
                board[row][col] = 0
        
        return False
    
    return solve()
```

**Why**: Premier example of constraint satisfaction problem (CSP). Combines backtracking with efficient constraint propagation. Foundations extend to scheduling, planning, and combinatorial optimization.

## Expression Operators

```python
# PROBLEM: Add operators (+, -, *) to form target value from digit string
# TIME: O(4^n) where n is string length SPACE: O(n) for recursion depth
def expression_operators(num: str, target: int) -> list[str]:
    """
    Find all ways to insert +, -, * operators into num string to get target.
    Multiplication has higher precedence.
    """
    result = []
    n = len(num)
    
    def backtrack(index: int, path: str, value: int, prev: int):
        """
        Backtrack through all possible operator placements.
        
        Args:
            index: current position in num string
            path: current expression string
            value: current evaluated value
            prev: previous operand (for multiplication precedence)
        """
        # Base case: reached end of string
        if index == n:
            if value == target:
                result.append(path)
            return
        
        # Try all possible lengths for next operand
        for i in range(index, n):
            # Skip numbers with leading zero (except "0" itself)
            if i > index and num[index] == '0':
                break
            
            # Extract current operand
            curr_str = num[index:i + 1]
            curr = int(curr_str)
            
            if index == 0:
                # First number - no operator needed
                backtrack(i + 1, curr_str, curr, curr)
            else:
                # Try addition: value + curr
                backtrack(i + 1, path + '+' + curr_str, value + curr, curr)
                
                # Try subtraction: value - curr
                backtrack(i + 1, path + '-' + curr_str, value - curr, -curr)
                
                # Try multiplication: (value - prev) + (prev * curr)
                # This handles operator precedence correctly
                new_value = value - prev + prev * curr
                backtrack(i + 1, path + '*' + curr_str, new_value, prev * curr)
    
    backtrack(0, '', 0, 0)
    return result
```

**Why**: Combines DFS, expression evaluation with operator precedence, and careful state management. Tests understanding of arithmetic parsing and backtrack state design.

## Randomized BST

```python
# PROBLEM: Implement a BST with randomized insertion to maintain balance
# TIME: O(log n) expected for insert/search/delete SPACE: O(n)
import random

class RandomizedBST:
    """
    Binary Search Tree that uses randomization for insertions.
    Provides expected O(log n) operations without explicit balancing.
    """
    
    class Node:
        def __init__(self, key: int, value: any = None):
            self.key = key
            self.value = value
            self.left: RandomizedBST.Node | None = None
            self.right: RandomizedBST.Node | None = None
            self.size = 1  # Size of subtree rooted at this node
    
    def __init__(self):
        self.root: RandomizedBST.Node | None = None
    
    def _size(self, node: Node | None) -> int:
        """Return size of subtree, handling None."""
        return node.size if node else 0
    
    def _update_size(self, node: Node) -> None:
        """Recalculate node size from children."""
        node.size = 1 + self._size(node.left) + self._size(node.right)
    
    def _rotate_right(self, node: Node) -> Node:
        """Right rotation for tree restructuring."""
        left_child = node.left
        node.left = left_child.right
        left_child.right = node
        self._update_size(node)
        self._update_size(left_child)
        return left_child
    
    def _rotate_left(self, node: Node) -> Node:
        """Left rotation for tree restructuring."""
        right_child = node.right
        node.right = right_child.left
        right_child.left = node
        self._update_size(node)
        self._update_size(right_child)
        return right_child
    
    def insert(self, key: int, value: any = None) -> None:
        """Insert key-value pair with randomization."""
        def _insert(node: RandomizedBST.Node | None) -> RandomizedBST.Node:
            if not node:
                return RandomizedBST.Node(key, value)
            
            # Random decision: insert at root or recursively
            if random.random() < 1.0 / (self._size(node) + 1):
                return self._insert_at_root(node, key, value)
            
            # Standard BST insertion
            if key < node.key:
                node.left = _insert(node.left)
            elif key > node.key:
                node.right = _insert(node.right)
            else:
                node.value = value  # Update existing
            
            self._update_size(node)
            return node
        
        self.root = _insert(self.root)
    
    def _insert_at_root(self, node: Node, key: int, value: any) -> Node:
        """Insert as new root using rotations."""
        if not node:
            return RandomizedBST.Node(key, value)
        
        if key < node.key:
            node.left = self._insert_at_root(node.left, key, value)
            return self._rotate_right(node)
        elif key > node.key:
            node.right = self._insert_at_root(node.right, key, value)
            return self._rotate_left(node)
        else:
            node.value = value
            return node
    
    def search(self, key: int) -> any:
        """Search for key, return associated value or None."""
        node = self.root
        while node:
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                return node.value
        return None
```

**Why**: Combines BST structure with probabilistic analysis. Demonstrates how randomization can simplify algorithms while providing performance guarantees. Foundation for randomized data structures like treaps.

## AA Tree

```python
# PROBLEM: Implement a self-balancing BST with simplified red-black rules
# TIME: O(log n) for all operations SPACE: O(n)
class AATree:
    """
    AA Tree: Simplified balanced BST using levels instead of colors.
    All left children are one level below parent.
    More restrictive than red-black tree, easier to implement.
    """
    
    class Node:
        def __init__(self, key: int, value: any = None):
            self.key = key
            self.value = value
            self.left: AATree.Node | None = None
            self.right: AATree.Node | None = None
            self.level = 1  # Level (1 = leaf)
    
    def __init__(self):
        self.root: AATree.Node | None = None
        self.NIL = AATree.Node(0)  # Sentinel node
        self.NIL.level = 0
    
    def _skew(self, node: Node) -> Node:
        """
        Right rotation to fix left horizontal link.
        A left child at same level violates AA tree rules.
        """
        if not node or not node.left:
            return node
        
        if node.left.level == node.level:
            # Perform right rotation
            left_child = node.left
            node.left = left_child.right
            left_child.right = node
            return left_child
        
        return node
    
    def _split(self, node: Node) -> Node:
        """
        Left rotation to fix two consecutive right horizontal links.
        Two right children at same level need rebalancing.
        """
        if not node or not node.right or not node.right.right:
            return node
        
        if node.right.right.level == node.level:
            # Perform left rotation and increase level
            right_child = node.right
            node.right = right_child.left
            right_child.left = node
            right_child.level += 1
            return right_child
        
        return node
    
    def insert(self, key: int, value: any = None) -> None:
        """Insert key-value pair maintaining AA tree balance."""
        def _insert(node: AATree.Node | None) -> AATree.Node:
            if not node:
                return AATree.Node(key, value)
            
            if key < node.key:
                node.left = _insert(node.left)
            elif key > node.key:
                node.right = _insert(node.right)
            else:
                node.value = value  # Update existing
            
            # Rebalance using skew and split
            node = self._skew(node)
            node = self._split(node)
            return node
        
        self.root = _insert(self.root)
    
    def delete(self, key: int) -> bool:
        """Delete key from tree, return True if found."""
        def _delete(node: AATree.Node | None, k: int) -> tuple[AATree.Node | None, bool]:
            if not node:
                return None, False
            
            found = False
            
            if k < node.key:
                node.left, found = _delete(node.left, k)
            elif k > node.key:
                node.right, found = _delete(node.right, k)
            else:
                found = True
                # Node to delete found
                if not node.left and not node.right:
                    return None, True
                elif not node.left:
                    # Find successor
                    successor = node.right
                    while successor.left:
                        successor = successor.left
                    node.key, node.value = successor.key, successor.value
                    node.right, _ = _delete(node.right, successor.key)
                else:
                    # Find predecessor
                    predecessor = node.left
                    while predecessor.right:
                        predecessor = predecessor.right
                    node.key, node.value = predecessor.key, predecessor.value
                    node.left, _ = _delete(node.left, predecessor.key)
            
            # Rebalance after deletion
            if node:
                # Decrease level if needed
                expected_level = min(
                    node.left.level if node.left else 0,
                    node.right.level if node.right else 0
                ) + 1
                
                if node.level > expected_level:
                    node.level = expected_level
                    if node.right and node.right.level > node.level:
                        node.right.level = node.level
                
                node = self._skew(node)
                node = self._split(node)
            
            return node, found
        
        self.root, found = _delete(self.root, key)
        return found
    
    def search(self, key: int) -> any:
        """Search for key, return value or None."""
        node = self.root
        while node:
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                return node.value
        return None
```

**Why**: Simplified balanced BST easier to implement than red-black trees. Demonstrates how constraints on structure can guarantee balance. Used in databases and file systems.

## Weight Balanced Tree

```python
# PROBLEM: Implement a BST balanced by subtree weight (size)
# TIME: O(log n) for all operations SPACE: O(n)
class WeightBalancedTree:
    """
    Weight-balanced tree where balance is determined by subtree sizes.
    Node is balanced if |size(left) - size(right)| <= threshold.
    Rebalances when weight ratio exceeds limit.
    """
    
    class Node:
        def __init__(self, key: int, value: any = None):
            self.key = key
            self.value = value
            self.left: WeightBalancedTree.Node | None = None
            self.right: WeightBalancedTree.Node | None = None
            self.weight = 1  # Number of nodes in subtree
    
    def __init__(self, alpha: float = 0.3):
        """
        Initialize tree with balance parameter alpha.
        Each subtree must have at least alpha * parent_weight nodes.
        """
        self.root: WeightBalancedTree.Node | None = None
        self.alpha = alpha  # Balance threshold (0.29 typical)
    
    def _weight(self, node: Node | None) -> int:
        """Return weight of subtree, handling None."""
        return node.weight if node else 0
    
    def _update_weight(self, node: Node) -> None:
        """Recalculate node weight from children."""
        node.weight = 1 + self._weight(node.left) + self._weight(node.right)
    
    def _is_balanced(self, node: Node) -> bool:
        """Check if node satisfies weight balance condition."""
        total = node.weight
        left_weight = self._weight(node.left)
        right_weight = self._weight(node.right)
        
        # Each subtree must be at least alpha * total
        return (left_weight >= self.alpha * total or left_weight == 0) and \
               (right_weight >= self.alpha * total or right_weight == 0)
    
    def _rotate_right(self, node: Node) -> Node:
        """Right rotation with weight update."""
        left_child = node.left
        node.left = left_child.right
        left_child.right = node
        
        self._update_weight(node)
        self._update_weight(left_child)
        return left_child
    
    def _rotate_left(self, node: Node) -> Node:
        """Left rotation with weight update."""
        right_child = node.right
        node.right = right_child.left
        right_child.left = node
        
        self._update_weight(node)
        self._update_weight(right_child)
        return right_child
    
    def _rebalance(self, node: Node) -> Node:
        """Rebalance node if weight condition violated."""
        self._update_weight(node)
        
        if self._is_balanced(node):
            return node
        
        left_weight = self._weight(node.left)
        right_weight = self._weight(node.right)
        
        if left_weight > right_weight:
            # Left heavy - rotate right
            if self._weight(node.left.right) > self._weight(node.left.left):
                node.left = self._rotate_left(node.left)
            return self._rotate_right(node)
        else:
            # Right heavy - rotate left
            if self._weight(node.right.left) > self._weight(node.right.right):
                node.right = self._rotate_right(node.right)
            return self._rotate_left(node)
    
    def insert(self, key: int, value: any = None) -> None:
        """Insert key-value pair with rebalancing."""
        def _insert(node: WeightBalancedTree.Node | None) -> WeightBalancedTree.Node:
            if not node:
                return WeightBalancedTree.Node(key, value)
            
            if key < node.key:
                node.left = _insert(node.left)
            elif key > node.key:
                node.right = _insert(node.right)
            else:
                node.value = value  # Update existing
                return node
            
            return self._rebalance(node)
        
        self.root = _insert(self.root)
    
    def search(self, key: int) -> any:
        """Search for key, return value or None."""
        node = self.root
        while node:
            if key < node.key:
                node = node.left
            elif key > node.key:
                node = node.right
            else:
                return node.value
        return None
```

**Why**: Alternative balancing strategy using explicit size constraints. Provides clear balance guarantee and useful when subtree size metadata is needed. Demonstrates relationship between structure and balance.

# PROBLEM: Build a 2D range tree for orthogonal range queries
# TIME: O(n log n) build, O(log² n + k) query where k = result size
# SPACE: O(n log n)
class RangeTree2D:
    """
    2D Range Tree for efficient orthogonal range queries.
    Query: find all points in rectangle [x1, x2] × [y1, y2].
    Uses fractional cascading for O(log n + k) query in practice.
    """
    
    class Point:
        def __init__(self, x: float, y: float, data: any = None):
            self.x = x
            self.y = y
            self.data = data
        
        def __repr__(self):
            return f"Point({self.x}, {self.y})"
    
    class Node:
        def __init__(self):
            self.point: RangeTree2D.Point | None = None
            self.left: RangeTree2D.Node | None = None
            self.right: RangeTree2D.Node | None = None
            self.y_tree: list[RangeTree2D.Point] = []  # Sorted by y for this subtree
    
    def __init__(self, points: list[tuple[float, float, any]] = None):
        """Build 2D range tree from list of (x, y, data) tuples."""
        self.root = None
        if points:
            pts = [self.Point(x, y, data) for x, y, data in points]
            self.root = self._build_x_tree(pts)
    
    def _build_x_tree(self, points: list[Point]) -> Node | None:
        """Build primary BST keyed by x-coordinate."""
        if not points:
            return None
        
        # Sort by x-coordinate
        points = sorted(points, key=lambda p: p.x)
        
        node = self.Node()
        
        if len(points) == 1:
            # Leaf node
            node.point = points[0]
            node.y_tree = [points[0]]  # Single point sorted by y
        else:
            # Internal node
            mid = len(points) // 2
            node.point = points[mid]  # Split point
            node.left = self._build_x_tree(points[:mid])
            node.right = self._build_x_tree(points[mid:])
            
            # Build y-sorted list for this subtree
            left_y = node.left.y_tree if node.left else []
            right_y = node.right.y_tree if node.right else []
            node.y_tree = self._merge_by_y(left_y, right_y)
        
        return node
    
    def _merge_by_y(self, left: list[Point], right: list[Point]) -> list[Point]:
        """Merge two y-sorted lists into one y-sorted list."""
        result = []
        i, j = 0, 0
        
        while i < len(left) and j < len(right):
            if left[i].y <= right[j].y:
                result.append(left[i])
                i += 1
            else:
                result.append(right[j])
                j += 1
        
        result.extend(left[i:])
        result.extend(right[j:])
        return result
    
    def range_query(self, x1: float, x2: float, y1: float, y2: float) -> list[Point]:
        """
        Find all points with x in [x1, x2] and y in [y1, y2].
        Returns list of points in range.
        """
        result = []
        self._query(self.root, x1, x2, y1, y2, result)
        return result
    
    def _query(self, node: Node, x1: float, x2: float, 
               y1: float, y2: float, result: list[Point]) -> None:
        """Recursive range query implementation."""
        if not node:
            return
        
        # Binary search in y_tree for points in y-range
        def points_in_y_range(points: list[Point]) -> list[Point]:
            # Find first point with y >= y1
            lo, hi = 0, len(points)
            while lo < hi:
                mid = (lo + hi) // 2
                if points[mid].y < y1:
                    lo = mid + 1
                else:
                    hi = mid
            
            # Collect points with y <= y2
            matches = []
            for i in range(lo, len(points)):
                if points[i].y <= y2:
                    matches.append(points[i])
                else:
                    break
