## html_track_kind_chapters
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Video Chapters with Track Element</title>
    <style>
        .video-container { max-width: 800px; margin: 20px auto; }
        .chapter-list { 
            background: #f5f5f5; 
            padding: 15px; 
            border-radius: 8px; 
            margin-top: 10px; 
        }
        .chapter-item { 
            cursor: pointer; 
            padding: 8px 12px; 
            margin: 4px 0; 
            background: white; 
            border-radius: 4px; 
            transition: background 0.2s; 
        }
        .chapter-item:hover { background: #e0e0e0; }
        .chapter-item.active { background: #007bff; color: white; }
    </style>
</head>
<body>
    <!-- 
        PROBLEM: Implement video chapter navigation using HTML5 track element
        TIME: O(1) SPACE: O(n) where n = number of chapters
        
        Why: Track elements with kind="chapters" provide structured navigation
        for video content, enabling users to jump to specific sections.
        VTT format defines chapter titles and timestamps.
    -->
    <div class="video-container">
        <video id="myVideo" controls width="100%">
            <source src="video.mp4" type="video/mp4">
            <!-- Track element with chapter markers -->
            <track 
                kind="chapters" 
                src="chapters.vtt" 
                srclang="en" 
                label="Chapters"
                default
            >
            Your browser does not support video.
        </video>
        
        <div class="chapter-list" id="chapterList"></div>
    </div>

    <script>
        // Chapter data (typically loaded from VTT file)
        const chapters = [
            { start: 0, end: 30, title: "Introduction" },
            { start: 30, end: 120, title: "Getting Started" },
            { start: 120, end: 240, title: "Core Concepts" },
            { start: 240, end: 360, title: "Advanced Topics" },
            { start: 360, end: 480, title: "Conclusion" }
        ];

        const video = document.getElementById('myVideo');
        const chapterList = document.getElementById('chapterList');

        // Render chapter navigation
        function renderChapters() {
            chapterList.innerHTML = chapters.map((ch, i) => `
                <div class="chapter-item" data-index="${i}" data-start="${ch.start}">
                    <strong>${formatTime(ch.start)}</strong> - ${ch.title}
                </div>
            `).join('');

            // Add click handlers for chapter navigation
            chapterList.querySelectorAll('.chapter-item').forEach(item => {
                item.addEventListener('click', () => {
                    const startTime = parseInt(item.dataset.start);
                    video.currentTime = startTime;
                    video.play();
                });
            });
        }

        // Format seconds to MM:SS
        function formatTime(seconds) {
            const mins = Math.floor(seconds / 60);
            const secs = Math.floor(seconds % 60);
            return `${mins}:${secs.toString().padStart(2, '0')}`;
        }

        // Highlight current chapter on timeupdate
        video.addEventListener('timeupdate', () => {
            const currentTime = video.currentTime;
            const items = chapterList.querySelectorAll('.chapter-item');
            
            items.forEach((item, i) => {
                const ch = chapters[i];
                if (currentTime >= ch.start && currentTime < ch.end) {
                    item.classList.add('active');
                } else {
                    item.classList.remove('active');
                }
            });
        });

        renderChapters();
    </script>
</body>
</html>
```

         aria-modal="true" aria-labelledby="dialogTitle" aria-describedby="dialogDesc">
        <div class="dialog-box">
            <h2 class="dialog-title" id="dialogTitle">Confirm Deletion</h2>
            <p class="dialog-message" id="dialogDesc">
                Are you sure you want to delete your account? This action cannot be undone.
            </p>
            <div class="dialog-buttons">
                <button class="btn btn-secondary" id="cancelBtn">Cancel</button>
                <button class="btn btn-danger" id="confirmBtn">Delete</button>
            </div>
        </div>
    </div>

    <script>
        const overlay = document.getElementById('dialogOverlay');
        const openBtn = document.getElementById('openDialog');
        const cancelBtn = document.getElementById('cancelBtn');
        const confirmBtn = document.getElementById('confirmBtn');
        let previousActiveElement = null;

        function openDialog() {
            previousActiveElement = document.activeElement;
            overlay.classList.add('open');
            // Focus first focusable element in dialog
            confirmBtn.focus();
            // Trap focus within dialog
            document.addEventListener('keydown', handleKeydown);
        }

        function closeDialog() {
            overlay.classList.remove('open');
            document.removeEventListener('keydown', handleKeydown);
            // Restore focus to trigger element
            if (previousActiveElement) {
                previousActiveElement.focus();
            }
        }

        function handleKeydown(e) {
            if (e.key === 'Escape') {
                closeDialog();
            }
            // Focus trap implementation
            if (e.key === 'Tab') {
                const focusable = overlay.querySelectorAll('button');
                const first = focusable[0];
                const last = focusable[focusable.length - 1];
                
                if (e.shiftKey && document.activeElement === first) {
                    e.preventDefault();
                    last.focus();
                } else if (!e.shiftKey && document.activeElement === last) {
                    e.preventDefault();
                    first.focus();
                }
            }
        }

        openBtn.addEventListener('click', openDialog);
        cancelBtn.addEventListener('click', closeDialog);
        confirmBtn.addEventListener('click', () => {
            alert('Account deleted!');
            closeDialog();
        });
    </script>
</body>
</html>

    <script>
        // Handle PWA installation prompt
        let deferredPrompt;
        const installPrompt = document.getElementById('installPrompt');
        const installBtn = document.getElementById('installBtn');

        // Capture the beforeinstallprompt event
        window.addEventListener('beforeinstallprompt', (e) => {
            e.preventDefault();
            deferredPrompt = e;
            installPrompt.classList.add('visible');
        });

        // Handle install button click
        installBtn.addEventListener('click', async () => {
            if (!deferredPrompt) return;
            
            // Show the browser's install prompt
            deferredPrompt.prompt();
            
            // Wait for user response
            const { outcome } = await deferredPrompt.userChoice;
            console.log(`User ${outcome} the install prompt`);
            
            // Clear deferred prompt
            deferredPrompt = null;
            installPrompt.classList.remove('visible');
        });

        // Detect if app is running in standalone mode
        if (window.matchMedia('(display-mode: standalone)').matches) {
            console.log('Running as installed PWA');
        }
    </script>
</body>
</html>

        const resultBox = document.getElementById('result');
        
        // Simulate API call
        function fetchData(shouldSucceed = true) {
            return new Promise((resolve, reject) => {
                setTimeout(() => {
                    if (shouldSucceed) {
                        resolve({ data: 'Success!', timestamp: Date.now() });
                    } else {
                        reject(new Error('Network error'));
                    }
                }, 2000);
            });
        }
        
        async function simulateSuccess() {
            resultBox.textContent = '';
            resultBox.className = 'result-box';

                const data = await fetchData(true)
                    .then(response => {
                        resultBox.textContent = `✅ ${JSON.stringify(response)}`;
                        resultBox.className = 'result-box success';
                        return response;
                    })
                    .catch(error => {
                        resultBox.textContent = `❌ ${error.message}`;
                        resultBox.className = 'result-box error';
                    })
                    .finally(() => {
                        // This ALWAYS runs!
                        loader.classList.remove('active');
                        console.log('Request completed (finally)');
                    });
            } catch (e) {
                // Handle any errors
            }
        }

                        resultBox.className = 'result-box error';

                    .finally(() => {
