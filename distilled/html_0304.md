```html
<html lang="en">

        label { display: block; margin-bottom: 5px; font-weight: bold; }
        input, select { width: 100%; padding: 8px; border: 1px solid #ddd; border-radius: 4px; }

    </style>
</head>
<body>
    <div class="form-container">
        <h2>Understanding Form Methods</h2>
        
        <!-- GET Method: Data visible in URL, for non-sensitive data -->
        <form method="GET" action="/search" id="searchForm">
            <h3>GET Method Example (Search)</h3>
            <div class="form-group">
                <label for="searchQuery">Search:</label>
                <input type="text" id="searchQuery" name="q" placeholder="Enter search term">
            </div>
            <button type="submit">Search (GET)</button>
        </form>

        <hr style="margin: 30px 0;">

        <!-- POST Method: Data in request body, for sensitive data -->
        <form method="POST" action="/login" id="loginForm">
            <h3>POST Method Example (Login)</h3>
            <div class="form-group">
                <label for="username">Username:</label>
                <input type="text" id="username" name="username" required>
            </div>
            <div class="form-group">
                <label for="password">Password:</label>
                <input type="password" id="password" name="password" required>
            </div>
            <button type="submit">Login (POST)</button>
        </form>

        <div class="info-box">
            <h4>Key Differences:</h4>
            <ul>
                <li><strong>GET:</strong> Data in URL, bookmarkable, cached, max ~2048 chars</li>
                <li><strong>POST:</strong> Data in body, not bookmarkable, not cached, no size limit</li>
            </ul>
        </div>
    </div>

    <script>
        // Demonstrate form submission behavior
        document.getElementById('searchForm').addEventListener('submit', function(e) {
            e.preventDefault();
            const formData = new FormData(this);
            const params = new URLSearchParams(formData);
            console.log('GET Request URL:', this.action + '?' + params.toString());
            alert(`GET Request would send:\n${this.action}?${params.toString()}`);
        });

        document.getElementById('loginForm').addEventListener('submit', function(e) {
            e.preventDefault();
            const formData = new FormData(this);
            console.log('POST Request Body:', Object.fromEntries(formData));
            alert(`POST Request would send data in body (not visible in URL)`);
        });
    </script>
</body>
</html>
```
**Why:** Demonstrates the critical difference between GET (URL parameters, for retrieval) and POST (request body, for secure submissions).

```html
            <button onclick="checkNetworkState()">Check Network State</button>
        </div>

    </div>

    <script>
        // Monitor video loading behavior
        const videos = {
            auto: document.getElementById('videoAuto'),
            metadata: document.getElementById('videoMeta'),
            none: document.getElementById('videoNone')
        };

        // Track buffered ranges
        function checkBuffered() {
            let status = '<h4>Buffered Ranges:</h4>';
            for (let [name, video] of Object.entries(videos)) {
                const buffered = video.buffered;
                let ranges = [];
                for (let i = 0; i < buffered.length; i++) {
                    ranges.push(`${buffered.start(i).toFixed(2)}s - ${buffered.end(i).toFixed(2)}s`);
                }
                status += `<strong>${name}:</strong> ${ranges.length ? ranges.join(', ') : 'None'}<br>`;
            }
            document.getElementById('status').innerHTML = status;
        }

        // Check network state
        function checkNetworkState() {
            const states = {
                0: 'NETWORK_EMPTY',
                1: 'NETWORK_IDLE',
                2: 'NETWORK_LOADING',
                3: 'NETWORK_NO_SOURCE'
            };
            let status = '<h4>Network States:</h4>';
            for (let [name, video] of Object.entries(videos)) {
                status += `<strong>${name}:</strong> ${states[video.networkState]}<br>`;
            }
            document.getElementById('status').innerHTML = status;
        }

        // Monitor progress events
        Object.entries(videos).forEach(([name, video]) => {
            video.addEventListener('progress', () => {
                console.log(`${name}: Progress event fired`);
            });
            
            video.addEventListener('loadedmetadata', () => {
                console.log(`${name}: Metadata loaded, duration: ${video.duration}s`);
            });
        });

        // Auto-check on load
        window.addEventListener('load', () => {
            setTimeout(checkBuffered, 1000);
        });
    </script>
</body>
</html>
```
**Why:** Shows how to optimize video loading performance by controlling preloading behavior to balance user experience and bandwidth.

```html
        <thead>

                <!-- scope="row" makes this cell a row header -->

                <td>2023-01-15</td>
            </tr>
            <tr>
                <th scope="row">Bob Smith</th>
                <td>Marketing</td>
                <td>$72,000</td>
                <td>2023-03-20</td>
            </tr>
            <tr>
                <th scope="row">Carol Davis</th>
                <td>Sales</td>
                <td>$68,000</td>
                <td>2023-02-10</td>
            </tr>
        </tbody>
    </table>

    <div class="info">
        <strong>Accessibility Benefit:</strong> Screen readers will announce "Alice Johnson, Department, Engineering" 
        instead of just "Department, Engineering" when navigating cells in Alice's row.
    </div>

    <script>
        // Demonstrate programmatic access to scope attribute
        document.addEventListener('DOMContentLoaded', function() {
            const rowHeaders = document.querySelectorAll('tbody th[scope="row"]');
            
            console.log('Row Headers Found:', rowHeaders.length);
            
            rowHeaders.forEach((header, index) => {
                // Get all cells in this row
                const row = header.parentElement;
                const cells = Array.from(row.querySelectorAll('td'));
                
                console.log(`Row ${index + 1}:`);
                console.log(`  Header: ${header.textContent}`);
                cells.forEach((cell, cellIndex) => {
                    console.log(`  Cell ${cellIndex + 1}: ${cell.textContent}`);
                });
            });

            // Add interactive highlighting
            rowHeaders.forEach(header => {
                header.addEventListener('mouseenter', function() {
                    this.parentElement.style.backgroundColor = '#fffacd';
                });
                header.addEventListener('mouseleave', function() {
                    this.parentElement.style.backgroundColor = '';
                });
            });
        });
    </script>
</body>
</html>
```
**Why:** Improves table accessibility by properly marking row headers, enabling screen readers to provide better context to users with visual impairments.

## html_aria_role_checkbox
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>ARIA Checkbox Role</title>

<body>
    <div class="checkbox-container">
        <h2>ARIA Checkbox Role (role="checkbox")</h2>
        
        <div class="info">
            <strong>Purpose:</strong> Create custom checkboxes with proper accessibility support using ARIA attributes.
        </div>

        <h3>Custom Checkboxes with ARIA:</h3>
        
        <!-- Custom checkbox using ARIA -->
        <div 
            class="custom-checkbox" 
            role="checkbox" 
            aria-checked="false"
            aria-label="Accept terms and conditions"
            tabindex="0"
            onclick="toggleCheckbox(this)"
            onkeydown="handleCheckboxKey(event, this)">
            <span class="checkbox-box"></span>
            <span>Accept terms and conditions</span>
        </div>

        <div 
            class="custom-checkbox" 
            role="checkbox" 
            aria-checked="true"
            aria-label="Subscribe to newsletter"
            tabindex="0"
            onclick="toggleCheckbox(this)"
            onkeydown="handleCheckboxKey(event, this)">
            <span class="checkbox-box checked"></span>
            <span>Subscribe to newsletter</span>
        </div>

        <div 
            class="custom-checkbox" 
            role="checkbox" 
            aria-checked="false"
            aria-label="Enable notifications"
            tabindex="0"
            onclick="toggleCheckbox(this)"
            onkeydown="handleCheckboxKey(event, this)">
            <span class="checkbox-box"></span>
            <span>Enable notifications</span>
        </div>

        <div class="info" style="margin-top: 30px;">
            <h4>Required ARIA Attributes:</h4>
            <ul>
                <li><code>role="checkbox"</code> - Identifies element as checkbox</li>
                <li><code>aria-checked="true/false"</code> - Current state</li>
                <li><code>tabindex="0"</code> - Make focusable</li>
                <li><code>aria-label</code> - Accessible name</li>
            </ul>
        </div>
    </div>

    <script>
        // Toggle checkbox state
        function toggleCheckbox(element) {
            const checkboxBox = element.querySelector('.checkbox-box');
            const isChecked = element.getAttribute('aria-checked') === 'true';
            
            // Toggle state
            element.setAttribute('aria-checked', !isChecked);
            checkboxBox.classList.toggle('checked', !isChecked);
            
            // Announce change to screen readers
            announceChange(element, !isChecked);
            
            console.log(`Checkbox "${element.getAttribute('aria-label')}" is now ${!isChecked ? 'checked' : 'unchecked'}`);
        }

        // Handle keyboard navigation
        function handleCheckboxKey(event, element) {
            // Space or Enter toggles checkbox
            if (event.key === ' ' || event.key === 'Enter') {
                event.preventDefault();
                toggleCheckbox(element);
            }
        }

        // Announce state change for screen readers
        function announceChange(element, isChecked) {
            const label = element.getAttribute('aria-label');
            const announcement = document.createElement('div');
            announcement.setAttribute('role', 'status');
            announcement.setAttribute('aria-live', 'polite');
            announcement.className = 'sr-only';
            announcement.style.cssText = 'position: absolute; left: -10000px; width: 1px; height: 1px; overflow: hidden;';
            announcement.textContent = `${label} ${isChecked ? 'checked' : 'unchecked'}`;
            document.body.appendChild(announcement);
            setTimeout(() => announcement.remove(), 1000);
        }

        // Add visual feedback for focus
        document.querySelectorAll('.custom-checkbox').forEach(checkbox => {
            checkbox.addEventListener('focus', function() {
                this.style.boxShadow = '0 0 0 3px rgba(76, 175, 80, 0.3)';
            });
            checkbox.addEventListener('blur', function() {
                this.style.boxShadow = '';
            });
        });
    </script>
</body>
</html>
```
**Why:** Enables creation of custom-styled checkboxes while maintaining full keyboard and screen reader accessibility.

## html_meta_keywords
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    
    <!-- Meta Keywords: Help search engines understand page content -->
    <meta name="keywords" content="HTML, CSS, JavaScript, web development, tutorial, programming">
    
    <!-- Related meta tags for SEO -->
    <meta name="description" content="Learn web development with HTML, CSS, and JavaScript tutorials">
    <meta name="author" content="Web Developer">
    <meta name="robots" content="index, follow">
    
    <title>Meta Keywords Example - SEO Best Practices</title>

        .container { max-width: 800px; margin: 20px auto; padding: 20px; }
        .meta-display { background: #f5f5f5; padding: 15px; margin: 15px 0; border-radius: 4px; }

    </style>
</head>
<body>
    <div class="container">
        <h1>Meta Keywords Tag</h1>
        
        <div class="meta-display">
            <h3>Current Page Meta Keywords:</h3>
            <code id="currentKeywords"></code>
        </div>

        <div class="warning">
            <strong>⚠️ Important:</strong> Most modern search engines (including Google) <strong>ignore</strong> the meta keywords tag 
            for ranking purposes. It's kept primarily for historical compatibility.
        </div>

        <h2>Meta Keywords Syntax</h2>
        <pre><code>&lt;meta name="keywords" content="keyword1, keyword2, keyword3"&gt;</code></pre>

        <div class="best-practice">
            <h3>✅ Best Practices (If Using):</h3>
            <ul>
                <li>Use 5-10 highly relevant keywords</li>
                <li>Separate with commas</li>
                <li>Include variations and misspellings (if relevant)</li>
                <li>Keep total length under 170 characters</li>
                <li>Ensure keywords appear in page content</li>
            </ul>
        </div>

        <h2>Modern SEO Alternatives</h2>
        <div class="meta-display">
            <h3>1. Meta Description (More Important)</h3>
            <code>&lt;meta name="description" content="..."&gt;</code>
        </div>

        <div class="meta-display">
            <h3>2. Semantic HTML</h3>
            <code>&lt;article&gt;, &lt;section&gt;, &lt;header&gt;, etc.</code>
        </div>

        <div class="meta-display">
            <h3>3. Structured Data (Schema.org)</h3>
            <code>&lt;script type="application/ld+json"&gt;...&lt;/script&gt;</code>
        </div>

        <div class="meta-display">
            <h3>4. Open Graph Tags</h3>
            <code>&lt;meta property="og:title" content="..."&gt;</code>
        </div>
    </div>

    <script>
        // Extract and display current meta keywords
        document.addEventListener('DOMContentLoaded', function() {
            const keywordsMeta = document.querySelector('meta[name="keywords"]');
            const display = document.getElementById('currentKeywords');
            
            if (keywordsMeta) {
                const keywords = keywordsMeta.getAttribute('content');
                display.textContent = keywords;
                
                // Analyze keywords
                const keywordList = keywords.split(',').map(k => k.trim());
                console.log('Total Keywords:', keywordList.length);
                console.log('Keywords:', keywordList);
                console.log('Total Characters:', keywords.length);
                
                // Warn if too many keywords
                if (keywordList.length > 10) {
                    console.warn('Warning: More than 10 keywords may be considered keyword stuffing');
                }
            }
        });

        // Function to programmatically set meta keywords
        function setMetaKeywords(keywords) {
            let meta = document.querySelector('meta[name="keywords"]');
            if (!meta) {
                meta = document.createElement('meta');
                meta.name = 'keywords';
                document.head.appendChild(meta);
            }
            meta.setAttribute('content', keywords);
            console.log('Updated keywords to:', keywords);
        }

        // Example: Dynamic keywords based on content
        function generateKeywordsFromContent() {
            const content = document.body.textContent;
            const words = content.toLowerCase().match(/\b[a-z]{4,}\b/g);
            const frequency = {};
            
            words.forEach(word => {
                frequency[word] = (frequency[word] || 0) + 1;
            });
            
            const topKeywords = Object.entries(frequency)
                .sort((a, b) => b[1] - a[1])
                .slice(0, 10)
                .map(([word]) => word);
            
            console.log('Auto-generated keywords:', topKeywords.join(', '));
            return topKeywords.join(', ');
        }
    </script>
</body>
</html>
```
**Why:** Explains the historical use and current irrelevance of meta keywords for SEO, while introducing modern alternatives.

```html
            <h3>Key Differences:</h3>

                <li><strong>align-content:</strong> Aligns flex lines (when wrapped)</li>
                <li><strong>align-items:</strong> Aligns items within each line</li>
            </ul>
        </div>

    <script>
        const flexContainer = document.getElementById('flexContainer');
        const buttons = document.querySelectorAll('.controls button');
        
        const descriptions = {
            'stretch': 'Lines stretch to fill the container\'s cross-axis space.',
            'flex-start': 'Lines packed to the start of the container.',
            'flex-end': 'Lines packed to the end of the container.',
            'center': 'Lines packed to the center of the container.',
            'space-between': 'Lines evenly distributed; first line at start, last at end.',
            'space-around': 'Lines evenly distributed with equal space around each line.',
            'space-evenly': 'Lines evenly distributed with equal space between and around lines.'
        };

        function setAlignContent(value) {
            // Update CSS
            flexContainer.style.alignContent = value;
            
            // Update UI
            document.getElementById('currentValue').textContent = `align-content: ${value}`;
            document.getElementById('description').textContent = descriptions[value];
            
            // Update button states
            buttons.forEach(btn => {
                btn.classList.remove('active');
                if (btn.textContent.includes(value) || btn.textContent === value) {
                    btn.classList.add('active');
                }
            });
            
            console.log(`align-content changed to: ${value}`);
        }

        // Add keyboard navigation
        buttons.forEach((btn, index) => {
```
