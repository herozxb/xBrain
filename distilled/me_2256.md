## Factory Method Variants

```python
# PROBLEM: Create objects without specifying exact class, allowing subclasses to decide
# TIME: O(1) for creation SPACE: O(1) per factory instance
from abc import ABC, abstractmethod
from typing import TypeVar, Generic, Type

T = TypeVar('T')

class Product(ABC):
    """Abstract product interface."""
    @abstractmethod
    def operation(self) -> str:
        pass

class Creator(ABC, Generic[T]):
    """Factory method pattern with generic return type."""
    
    @abstractmethod
    def factory_method(self) -> T:
        """Subclasses override to create specific products."""
        pass
    
    def some_operation(self) -> str:
        """Main business logic that uses the product."""
        product = self.factory_method()
        return f"Creator: {product.operation()}"

class ConcreteProductA(Product):
    def operation(self) -> str:
        return "Result of ConcreteProductA"

class ConcreteProductB(Product):
    def operation(self) -> str:
        return "Result of ConcreteProductB"

class ConcreteCreatorA(Creator[Product]):
    def factory_method(self) -> Product:
        return ConcreteProductA()

class ConcreteCreatorB(Creator[Product]):
    def factory_method(self) -> Product:
        return ConcreteProductB()

# Parameterized factory variant
class ParameterizedFactory:
    """Factory that selects product based on parameter."""
    _registry: dict[str, type[Product]] = {}
    
    @classmethod
    def register(cls, key: str, product_class: type[Product]) -> None:
        cls._registry[key] = product_class
    
    @classmethod
    def create(cls, key: str) -> Product:
        if key not in cls._registry:
            raise ValueError(f"Unknown product: {key}")
        return cls._registry[key]()
```
**Why**: Factory Method decouples object creation from usage, enabling extensibility. The parameterized variant allows runtime registration without modifying factory code.

## Abstract Factory Variants

```python
# PROBLEM: Create families of related objects without specifying concrete classes
# TIME: O(1) SPACE: O(n) for product families
from abc import ABC, abstractmethod
from typing import Protocol, runtime_checkable

@runtime_checkable
class Button(Protocol):
    def render(self) -> str: ...
    def click(self) -> str: ...

@runtime_checkable  
class Checkbox(Protocol):
    def render(self) -> str: ...
    def check(self) -> str: ...

class GUIFactory(ABC):
    """Abstract factory for creating UI component families."""
    
    @abstractmethod
    def create_button(self) -> Button: pass
    
    @abstractmethod
    def create_checkbox(self) -> Checkbox: pass

# Windows family
class WindowsButton:
    def render(self) -> str:
        return "Windows-style button rendered"
    def click(self) -> str:
        return "Windows click handler"

class WindowsCheckbox:
    def render(self) -> str:
        return "Windows-style checkbox rendered"
    def check(self) -> str:
        return "Windows check toggle"

class WindowsFactory(GUIFactory):
    def create_button(self) -> Button:
        return WindowsButton()
    def create_checkbox(self) -> Checkbox:
        return WindowsCheckbox()

# Mac family
class MacButton:
    def render(self) -> str:
        return "Mac-style button rendered"
    def click(self) -> str:
        return "Mac click handler"

class MacCheckbox:
    def render(self) -> str:
        return "Mac-style checkbox rendered"
    def check(self) -> str:
        return "Mac check toggle"

class MacFactory(GUIFactory):
    def create_button(self) -> Button:
        return MacButton()
    def create_checkbox(self) -> Checkbox:
        return MacCheckbox()

# Singleton registry variant
class FactoryRegistry:
    """Registry pattern for abstract factories."""
    _instance = None
    _factories: dict[str, GUIFactory] = {}
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance
    
    @classmethod
    def register(cls, name: str, factory: GUIFactory) -> None:
        cls._factories[name] = factory
    
    @classmethod
    def get(cls, name: str) -> GUIFactory:
        return cls._factories[name]
```
**Why**: Abstract Factory ensures product families stay consistent. The registry variant allows dynamic factory registration and switching at runtime.

## Shared Database Pattern

```python
# PROBLEM: Multiple services need access to same data without duplication
# TIME: O(1) for read, O(log n) for indexed query SPACE: O(n) for data
from dataclasses import dataclass
from typing import Optional, Protocol
from contextlib import contextmanager
import sqlite3

@dataclass
class User:
    id: int
    name: str
    email: str

class DatabaseAccessor(Protocol):
    """Protocol for database access abstraction."""
    def get_user(self, user_id: int) -> Optional[User]: ...
    def save_user(self, user: User) -> None: ...

class SharedDatabase:
    """Shared database accessible by multiple services."""
    
    def __init__(self, db_path: str = ":memory:"):
        self._conn = sqlite3.connect(db_path)
        self._init_schema()
    
    def _init_schema(self) -> None:
        self._conn.execute("""
            CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY,
                name TEXT NOT NULL,
                email TEXT UNIQUE NOT NULL
            )
        """)
        self._conn.commit()
    
    @contextmanager
    def transaction(self):
        """Provide transactional boundary."""
        try:
            yield self._conn
            self._conn.commit()
        except Exception:
            self._conn.rollback()
            raise
    
    def get_user(self, user_id: int) -> Optional[User]:
        cursor = self._conn.execute(
            "SELECT id, name, email FROM users WHERE id = ?", 
            (user_id,)
        )
        row = cursor.fetchone()
        return User(*row) if row else None
    
    def save_user(self, user: User) -> None:
        self._conn.execute(
            "INSERT OR REPLACE INTO users (id, name, email) VALUES (?, ?, ?)",
            (user.id, user.name, user.email)
        )
        self._conn.commit()

class UserService:
    """Service using shared database."""
    def __init__(self, db: SharedDatabase):
        self._db = db
    
    def get_user_name(self, user_id: int) -> Optional[str]:
        user = self._db.get_user(user_id)
        return user.name if user else None

class OrderService:
    """Another service using same database."""
    def __init__(self, db: SharedDatabase):
        self._db = db
    
    def validate_user(self, user_id: int) -> bool:
        return self._db.get_user(user_id) is not None
```
**Why**: Shared Database reduces data duplication and ensures consistency across services. Trade-off: tight coupling and potential bottleneck.

## Compensating Transaction

```python
# PROBLEM: Rollback distributed transactions when individual steps fail
# TIME: O(n) for n compensating steps SPACE: O(n) for action history
from dataclasses import dataclass
from typing import Callable, Any, Optional
from enum import Enum

class TransactionStatus(Enum):
    PENDING = "pending"
    COMMITTED = "committed"
    COMPENSATED = "compensated"
    FAILED = "failed"

@dataclass
class TransactionStep:
    """Represents a single step in a saga."""
    name: str
    action: Callable[[], Any]
    compensate: Callable[[], Any]
    result: Any = None
    executed: bool = False

class CompensatingTransaction:
    """Manages saga-style compensating transactions."""
    
    def __init__(self):
        self._steps: list[TransactionStep] = []
        self._status = TransactionStatus.PENDING
    
    def add_step(self, name: str, action: Callable, compensate: Callable) -> None:
        """Add a step with its compensation logic."""
        self._steps.append(TransactionStep(name, action, compensate))
    
    def execute(self) -> bool:
        """Execute all steps, compensate on failure."""
        executed_steps: list[TransactionStep] = []
        
        for step in self._steps:
            try:
                step.result = step.action()
                step.executed = True
                executed_steps.append(step)
            except Exception as e:
                # Compensate executed steps in reverse order
                self._compensate_all(executed_steps)
                self._status = TransactionStatus.FAILED
                return False
        
        self._status = TransactionStatus.COMMITTED
        return True
    
    def _compensate_all(self, steps: list[TransactionStep]) -> None:
        """Compensate steps in reverse order."""
        for step in reversed(steps):
            try:
                step.compensate()
            except Exception:
                # Log but continue compensating other steps
                pass
        self._status = TransactionStatus.COMPENSATED
    
    @property
    def status(self) -> TransactionStatus:
        return self._status

# Example usage
def book_hotel() -> str:
    return "hotel_booked"

def cancel_hotel() -> None:
    print("Hotel cancelled")

def book_flight() -> str:
    raise RuntimeError("Flight booking failed")

def cancel_flight() -> None:
    print("Flight cancelled")
```
**Why**: Compensating transactions provide rollback semantics in distributed systems without 2PC. Each step has inverse compensation logic for eventual consistency.

## Phantom Read Prevention

```python
# PROBLEM: Prevent phantom reads where new rows appear in repeated queries
# TIME: O(n log n) with range locks SPACE: O(m) for lock table
from dataclasses import dataclass
from typing import Optional
from threading import Lock
from collections import defaultdict
import bisect

@dataclass
class Range:
    """Represents a locked range of values."""
    start: int
    end: int

class RangeLockManager:
    """Manages range locks for phantom read prevention."""
    
    def __init__(self):
        self._lock = Lock()
        self._read_ranges: dict[int, list[Range]] = defaultdict(list)
        self._write_ranges: dict[int, list[Range]] = defaultdict(list)
    
    def acquire_read_range(self, table_id: int, start: int, end: int) -> bool:
        """Acquire shared lock on range. Blocks if write-locked."""
        with self._lock:
            # Check for conflicting write locks
            for r in self._write_ranges[table_id]:
                if self._ranges_overlap(r, Range(start, end)):
                    return False
            
            # Add read lock
            bisect.insort(
                self._read_ranges[table_id],
                Range(start, end),
                key=lambda x: x.start
            )
            return True
    
    def acquire_write_range(self, table_id: int, start: int, end: int) -> bool:
        """Acquire exclusive lock on range. Blocks if any lock exists."""
        with self._lock:
            # Check for conflicting read locks
            for r in self._read_ranges[table_id]:
                if self._ranges_overlap(r, Range(start, end)):
                    return False
            
            # Check for conflicting write locks
            for r in self._write_ranges[table_id]:
                if self._ranges_overlap(r, Range(start, end)):
                    return False
            
            bisect.insort(
                self._write_ranges[table_id],
                Range(start, end),
                key=lambda x: x.start
            )
            return True
    
    def release_read_range(self, table_id: int, start: int, end: int) -> None:
        with self._lock:
            self._read_ranges[table_id] = [
                r for r in self._read_ranges[table_id]
                if not (r.start == start and r.end == end)
            ]
    
    def release_write_range(self, table_id: int, start: int, end: int) -> None:
        with self._lock:
            self._write_ranges[table_id] = [
                r for r in self._write_ranges[table_id]
                if not (r.start == start and r.end == end)
            ]
    
    @staticmethod
    def _ranges_overlap(r1: Range, r2: Range) -> bool:
        return r1.start <= r2.end and r2.start <= r1.end
```
**Why**: Range locks prevent phantom reads by locking value ranges, not just existing rows. This ensures consistent query results across transactions at serializable isolation.

## Dirty Read Prevention

```python
# PROBLEM: Prevent reading uncommitted changes from other transactions
# TIME: O(1) for lock check SPACE: O(n) for lock table
from dataclasses import dataclass, field
from typing import Optional
from threading import Lock
from enum import Enum
import time

class LockType(Enum):
    SHARED = "shared"
    EXCLUSIVE = "exclusive"

@dataclass
class RowLock:
    transaction_id: int
    lock_type: LockType
    timestamp: float = field(default_factory=time.time)

class DirtyReadPreventer:
    """Implements read-committed isolation using row-level locks."""
    
    def __init__(self):
        self._lock = Lock()
        self._row_locks: dict[str, list[RowLock]] = {}
        self._pending_writes: dict[str, dict[int, any]] = {}
    
    def read_row(self, row_id: str, transaction_id: int) -> Optional[any]:
        """Read row - only sees committed data."""
        with self._lock:
            # Check for exclusive locks (uncommitted writes)
            for row_lock in self._row_locks.get(row_id, []):
                if (row_lock.lock_type == LockType.EXCLUSIVE and 
                    row_lock.transaction_id != transaction_id):
                    return None  # Row is locked for write
            
            # Return committed value (not pending writes from others)
            return self._get_committed_value(row_id)
    
    def write_row(self, row_id: str, value: any, transaction_id: int) -> bool:
        """Write row - requires exclusive lock."""
        with self._lock:
            # Check for conflicting locks
            for row_lock in self._row_locks.get(row_id, []):
                if row_lock.transaction_id != transaction_id:
                    return False  # Cannot acquire exclusive lock
            
            # Acquire exclusive lock
            if row_id not in self._row_locks:
                self._row_locks[row_id] = []
            self._row_locks[row_id].append(
                RowLock(transaction_id, LockType.EXCLUSIVE)
            )
            
            # Store as pending write
            if row_id not in self._pending_writes:
                self._pending_writes[row_id] = {}
            self._pending_writes[row_id][transaction_id] = value
            return True
    
    def commit(self, row_id: str, transaction_id: int) -> None:
        """Commit pending write, release locks."""
        with self._lock:
            if row_id in self._pending_writes:
                pending = self._pending_writes[row_id].pop(transaction_id, None)
                if pending is not None:
                    self._apply_committed_value(row_id, pending)
            
            # Release locks
            self._row_locks[row_id] = [
                l for l in self._row_locks.get(row_id, [])
                if l.transaction_id != transaction_id
            ]
    
    def rollback(self, row_id: str, transaction_id: int) -> None:
        """Discard pending write, release locks."""
        with self._lock:
            self._pending_writes.get(row_id, {}).pop(transaction_id, None)
            self._row_locks[row_id] = [
                l for l in self._row_locks.get(row_id, [])
                if l.transaction_id != transaction_id
            ]
    
    def _get_committed_value(self, row_id: str) -> any:
        """Return only committed values."""
        # Implementation would fetch from actual storage
        return None
    
    def _apply_committed_value(self, row_id: str, value: any) -> None:
        """Persist committed value."""
        pass
```
**Why**: Dirty read prevention ensures transactions only see committed data. Exclusive locks block reads until writes commit, maintaining read-committed isolation.

## Mixin Patterns

```python
# PROBLEM: Add reusable functionality to classes without inheritance hierarchy
# TIME: O(1) SPACE: O(1) per mixin
from typing import Protocol, TypeVar, Generic, Any

T = TypeVar('T')

class SerializableMixin:
    """Mixin for JSON serialization."""
    
    def to_dict(self) -> dict[str, Any]:
        return {
            key: self._serialize_value(value)
            for key, value in self.__dict__.items()
            if not key.startswith('_')
        }
    
    def _serialize_value(self, value: Any) -> Any:
        if hasattr(value, 'to_dict'):
            return value.to_dict()
        elif isinstance(value, list):
            return [self._serialize_value(v) for v in value]
        return value

class ComparableMixin:
    """Mixin for comparison operations."""
    
    def __eq__(self, other: Any) -> bool:
        if not isinstance(other, self.__class__):
            return False
        return self.__dict__ == other.__dict__
    
    def __lt__(self, other: Any) -> bool:
        if not isinstance(other, self.__class__):
            return NotImplemented
        return self._compare_key() < other._compare_key()
    
    def _compare_key(self) -> Any:
        raise NotImplementedError

class CacheableMixin:
    """Mixin for caching computed properties."""
    _cache: dict[str, Any]
    
    def __init_subclass__(cls, **kwargs):
        super().__init_subclass__(**kwargs)
        cls._cache = {}
    
    def cached_property(self, name: str, computer: callable) -> Any:
        if name not in self._cache:
            self._cache[name] = computer()
        return self._cache[name]
    
    def invalidate_cache(self, name: str = None) -> None:
        if name:
            self._cache.pop(name, None)
        else:
            self._cache.clear()

class ValidatableMixin:
    """Mixin for validation logic."""
    
    def validate(self) -> bool:
        """Override to implement validation."""
        return True
    
    def validate_or_raise(self) -> None:
        if not self.validate():
            raise ValueError(f"Validation failed for {self.__class__.__name__}")

# Usage example
class Product(
    SerializableMixin, 
    ComparableMixin, 
    CacheableMixin,
    ValidatableMixin
):
    def __init__(self, id: int, name: str, price: float):
        self.id = id
        self.name = name
        self.price = price
    
    def _compare_key(self) -> tuple:
        return (self.price, self.name)
    
    def validate(self) -> bool:
        return self.price >= 0 and len(self.name) > 0
```
**Why**: Mixins provide horizontal composition of behaviors without deep inheritance. Each mixin adds specific capability while staying independent.

## Composition Patterns

```python
# PROBLEM: Build complex objects from simpler parts without inheritance
# TIME: O(1) for delegation SPACE: O(n) for component objects
from typing import Protocol, Any
from dataclasses import dataclass

class Behavior(Protocol):
    """Protocol for composable behaviors."""
    def execute(self) -> str: ...

@dataclass
class LoggingBehavior:
    """Behavior that logs actions."""
    prefix: str
    
    def execute(self) -> str:
        return f"[{self.prefix}] Logging action"

@dataclass
class TimingBehavior:
    """Behavior that measures execution time."""
    wrapped: Behavior
    
    def execute(self) -> str:
        import time
        start = time.time()
        result = self.wrapped.execute()
        elapsed = time.time() - start
        return f"{result} (took {elapsed:.4f}s)"

@dataclass
class RetryBehavior:
    """Behavior that retries on failure."""
    wrapped: Behavior
    max_retries: int = 3
    
    def execute(self) -> str:
        for attempt in range(self.max_retries):
            try:
                return self.wrapped.execute()
            except Exception as e:
                if attempt == self.max_retries - 1:
                    return f"Failed after {self.max_retries} attempts"
        return ""

class ComposedObject:
    """Object built from composed behaviors."""
    
    def __init__(self):
        self._behaviors: list[Behavior] = []
    
    def add_behavior(self, behavior: Behavior) -> 'ComposedObject':
        self._behaviors.append(behavior)
        return self
    
    def execute_all(self) -> list[str]:
        return [b.execute() for b in self._behaviors]

# Delegation-based composition
class Engine:
    def start(self) -> str:
        return "Engine started"

class Wheels:
    def rotate(self) -> str:
        return "Wheels rotating"

class Car:
    """Car composed of parts, not inherited."""
    
    def __init__(self):
        self._engine = Engine()
        self._wheels = Wheels()
    
    def drive(self) -> str:
        return f"{self._engine.start()}, {self._wheels.rotate()}"
```
**Why**: Composition over inheritance enables flexible object assembly. Components can be swapped, decorated, or replaced independently.

## Adapter Pattern Variants

```python
# PROBLEM: Make incompatible interfaces work together
# TIME: O(1) for delegation SPACE: O(1)
from typing import Protocol

class Target(Protocol):
    """Expected interface by client."""
    def request(self) -> str: ...

class Adaptee:
    """Existing interface to adapt."""
    def specific_request(self) -> str:
        return "Special behavior from adaptee"

# Class adapter (inheritance-based)
class ClassAdapter(Adaptee, Target):
    """Adapter using multiple inheritance."""
    
    def request(self) -> str:
        # Adapt the interface
        return f"Adapted: {self.specific_request()}"

# Object adapter (composition-based) - preferred
class ObjectAdapter:
    """Adapter using composition."""
    
    def __init__(self, adaptee: Adaptee):
        self._adaptee = adaptee
    
    def request(self) -> str:
        return f"Adapted: {self._adaptee.specific_request()}"

# Two-way adapter
class TwoWayAdapter(Target):
    """Adapter that works both ways."""
    
    def __init__(self, adaptee: Adaptee = None):
        self._adaptee = adaptee or Adaptee()
    
    def request(self) -> str:
        return self._adaptee.specific_request()
    
    def specific_request(self) -> str:
        # Also expose adaptee interface
        return self._adaptee.specific_request()

# Pluggable adapter
class PluggableAdapter:
    """Adapter with configurable adaptation strategy."""
    
    def __init__(self, adaptee: Any, 
                 request_method: str = "specific_request"):
        self._adaptee = adaptee
        self._request_method = request_method
    
    def request(self) -> str:
        method = getattr(self._adaptee, self._request_method)
        return f"Adapted: {method()}"

# Bidirectional adapter
class BidirectionalAdapter:
    """Adapter that converts both ways."""
    
    def __init__(self):
        self._formats = {
            'xml': self._to_xml,
            'json': self._to_json,
            'yaml': self._to_yaml
        }
    
    def convert(self, data: dict, from_format: str, to_format: str) -> str:
        # Normalize to dict then convert
        normalized = self._from_format(data, from_format)
        return self._formats[to_format](normalized)
    
    def _from_format(self, data: Any, format: str) -> dict:
        # Implementation would parse from format
        return data if isinstance(data, dict) else {}
    
    def _to_xml(self, data: dict) -> str:
        return f"<root>{data}</root>"
    
    def _to_json(self, data: dict) -> str:
        import json
        return json.dumps(data)
    
    def _to_yaml(self, data: dict) -> str:
        return f"data: {data}"
```
**Why**: Adapters bridge incompatible interfaces without modifying existing code. Object adapter is preferred for flexibility; pluggable adapter enables runtime configuration.

## Domain-Driven Design

```python
# PROBLEM: Model complex business domains with rich behavior
# TIME: O(1) to O(n) depending on domain logic SPACE: O(n) for aggregates
from dataclasses import dataclass, field
from typing import Optional, NewType, List
from datetime import datetime
from enum import Enum

# Value Objects - identity by attributes
@dataclass(frozen=True)
class Money:
    amount: int  # Store as cents to avoid floating point
    currency: str
    
    def __add__(self, other: 'Money') -> 'Money':
        if other.currency != self.currency:
            raise ValueError("Cannot add different currencies")
        return Money(self.amount + other.amount, self.currency)
    
    def __lt__(self, other: 'Money') -> bool:
        if other.currency != self.currency:
            raise ValueError("Cannot compare different currencies")
        return self.amount < other.amount

@dataclass(frozen=True)
class Email:
    value: str
    
    def __post_init__(self):
        if '@' not in self.value:
            raise ValueError("Invalid email format")

# Entities - identity by ID
@dataclass
class OrderId:
    value: str

class OrderStatus(Enum):
    PENDING = "pending"
    CONFIRMED = "confirmed"
    SHIPPED = "shipped"

# Aggregate Root - controls consistency boundary
@dataclass
class Order:
    """Aggregate root for Order bounded context."""
    id: OrderId
    customer_email: Email
    _items: List['OrderItem'] = field(default_factory=list)
    _status: OrderStatus = OrderStatus.PENDING
    _events: List['DomainEvent'] = field(default_factory=list)
    
    def add_item(self, product_name: str, price: Money) -> None:
        """Business rule: cannot modify confirmed orders."""
        if self._status != OrderStatus.PENDING:
            raise ValueError("Cannot modify confirmed order")
        
        item = OrderItem(product_name, price)
        self._items.append(item)
        
        # Emit domain event
        self._events.append(ItemAddedEvent(self.id, product_name))
    
    def confirm(self) -> None:
        """Business rule: order must have items to confirm."""
        if not self._items:
            raise ValueError("Cannot confirm empty order")
        
        self._status = OrderStatus.CONFIRMED
        self._events.append(OrderConfirmedEvent(self.id))
    
    @property
    def total(self) -> Money:
        return sum(
            (item.price for item in self._items),
            Money(0, "USD")
        )
    
    def pull_events(self) -> List['DomainEvent']:
        """Clear and return events for publishing."""
        events = self._events.copy()
        self._events.clear()
        return events

# Entity within aggregate
@dataclass
class OrderItem:
    product_name: str
    price: Money

# Domain Events
@dataclass
class DomainEvent:
    occurred_at: datetime = field(default_factory=datetime.now)

@dataclass
class ItemAddedEvent(DomainEvent):
    order_id: OrderId
    product_name: str

@dataclass
class OrderConfirmedEvent(DomainEvent):
    order_id: OrderId

# Repository interface
class OrderRepository(Protocol):
    def find_by_id(self, id: OrderId) -> Optional[Order]: ...
    def save(self, order: Order) -> None: ...

# Domain Service
class PricingService:
    """Service for complex pricing logic not fitting entity."""
    
    def calculate_discount(self, order: Order, customer_tier: str) -> Money:
        if customer_tier == "gold":
            return Money(
                int(order.total.amount * 0.1),
                order.total.currency
            )
        return Money(0, order.total.currency)
```
**Why**: DDD captures business rules in code. Value objects are immutable; entities have identity; aggregates ensure consistency; domain events enable loose coupling between bounded contexts.
