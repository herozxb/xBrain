```python
# PROBLEM: Implement partner pattern for service collaboration with health monitoring
# TIME: O(n) SPACE: O(n)
class ServicePartner:
    """
    Implements the Partner Pattern for microservices collaboration.
    Partners are services that work together with health checks and failover.
    """
    
    def __init__(self, service_name: str):
        self.service_name = service_name
        self.partners: dict[str, dict] = {}
        self.health_status: dict[str, bool] = {}
    
    def register_partner(self, partner_name: str, endpoint: str, priority: int = 1) -> None:
        """Register a partner service with endpoint and priority."""
        self.partners[partner_name] = {
            'endpoint': endpoint,
            'priority': priority,
            'last_check': None
        }
        self.health_status[partner_name] = True
    
    def check_partner_health(self, partner_name: str) -> bool:
        """Check if partner service is healthy."""
        if partner_name not in self.partners:
            return False
        # Simulate health check (in real impl, make HTTP request)
        import time
        self.partners[partner_name]['last_check'] = time.time()
        return self.health_status.get(partner_name, False)
    
    def get_healthy_partners(self) -> list[str]:
        """Get list of healthy partners sorted by priority."""
        healthy = [
            name for name in self.partners 
            if self.health_status.get(name, False)
        ]
        return sorted(healthy, key=lambda x: self.partners[x]['priority'])
    
    def failover(self, failed_partner: str) -> str | None:
        """Get next available partner for failover."""
        if failed_partner in self.health_status:
            self.health_status[failed_partner] = False
        healthy = self.get_healthy_partners()
        return healthy[0] if healthy else None

# Usage: Partner pattern enables resilient service collaboration
partner = ServicePartner("order-service")
partner.register_partner("inventory-service", "http://inventory:8080", 1)
partner.register_partner("payment-service", "http://payment:8080", 2)
```

**Why**: Partner pattern creates peer-to-peer service relationships with built-in health monitoring and automatic failover.

## Upstream Downstream Pattern

```python
# PROBLEM: Implement upstream/downstream service dependency management
# TIME: O(n log n) SPACE: O(n)
class ServiceDependencyGraph:
    """
    Manages upstream (providers) and downstream (consumers) service dependencies.
    Enables cascading updates and dependency-aware deployments.
    """
    
    def __init__(self):
        self.upstream: dict[str, set[str]] = {}    # service -> services it depends on
        self.downstream: dict[str, set[str]] = {}  # service -> services depending on it
    
    def add_dependency(self, service: str, depends_on: str) -> None:
        """Add upstream dependency: service depends on depends_on."""
        if service not in self.upstream:
            self.upstream[service] = set()
        if depends_on not in self.downstream:
            self.downstream[depends_on] = set()
        
        self.upstream[service].add(depends_on)
        self.downstream[depends_on].add(service)
    
    def get_upstream_services(self, service: str) -> set[str]:
        """Get all services this service depends on (recursively)."""
        visited = set()
        stack = list(self.upstream.get(service, set()))
        
        while stack:
            current = stack.pop()
            if current not in visited:
                visited.add(current)
                stack.extend(self.upstream.get(current, set()))
        
        return visited
    
    def get_downstream_services(self, service: str) -> set[str]:
        """Get all services depending on this service (recursively)."""
        visited = set()
        stack = list(self.downstream.get(service, set()))
        
        while stack:
            current = stack.pop()
            if current not in visited:
                visited.add(current)
                stack.extend(self.downstream.get(current, set()))
        
        return visited
    
    def deployment_order(self, service: str) -> list[str]:
        """Get topological order for deploying a service and its dependencies."""
        # Deploy upstream first, then service, then downstream
        upstream = list(self.get_upstream_services(service))
        downstream = list(self.get_downstream_services(service))
        
        return upstream + [service] + downstream

# Usage: Track service dependencies for coordinated deployments
graph = ServiceDependencyGraph()
graph.add_dependency("frontend", "backend-api")
graph.add_dependency("backend-api", "database")
```

**Why**: Upstream/downstream pattern enables dependency-aware deployments and change propagation.

```python
# PROBLEM: Implement Domain Service for cross-aggregate operations
# TIME: O(n) SPACE: O(1)
from dataclasses import dataclass
from decimal import Decimal

@dataclass
class Account:
    """Account aggregate."""
    account_id: str
    balance: Decimal
    status: str
    
    def can_debit(self, amount: Decimal) -> bool:
        """Check if account can be debited."""
        return self.status == "ACTIVE" and self.balance >= amount
    
    def debit(self, amount: Decimal) -> None:
        """Debit amount from account."""
        if not self.can_debit(amount):
            raise ValueError("Insufficient funds or inactive account")
        self.balance -= amount
    
    def credit(self, amount: Decimal) -> None:
        """Credit amount to account."""
        if self.status != "ACTIVE":
            raise ValueError("Account is not active")
        self.balance += amount

class TransferService:
    """
    Domain Service for money transfer between accounts.
    Used when operation spans multiple aggregates.
    """
    
    def __init__(self, account_repository: dict[str, Account]):
        self.accounts = account_repository
    
    def transfer(
        self, 
        from_account_id: str, 
        to_account_id: str, 
        amount: Decimal
    ) -> dict:
        """
        Execute transfer between two accounts.
        Returns transfer result with transaction details.
        """
        from_account = self.accounts.get(from_account_id)
        to_account = self.accounts.get(to_account_id)
        
        if not from_account or not to_account:
            raise ValueError("One or both accounts not found")
        
        if not from_account.can_debit(amount):
            raise ValueError("Source account cannot be debited")
        
        # Execute the transfer
        from_account.debit(amount)
        to_account.credit(amount)
        
        return {
            'status': 'SUCCESS',
            'from_account': from_account_id,
            'to_account': to_account_id,
            'amount': str(amount),
            'from_balance': str(from_account.balance),
            'to_balance': str(to_account.balance)
        }
    
    def calculate_total_balance(self, account_ids: list[str]) -> Decimal:
        """Calculate total balance across multiple accounts."""
        total = Decimal('0')
        for acc_id in account_ids:
            if acc_id in self.accounts:
                total += self.accounts[acc_id].balance
        return total

# Usage: Domain services handle operations that don't belong to a single aggregate
service = TransferService({
    "ACC-1": Account("ACC-1", Decimal('1000.00'), "ACTIVE"),
    "ACC-2": Account("ACC-2", Decimal('500.00'), "ACTIVE")
})
```

**Why**: DDD Services encapsulate domain operations that span multiple aggregates or don't naturally belong to entities.

```python
# PROBLEM: Implement Domain Events for decoupled communication
# TIME: O(n) SPACE: O(n)
from dataclasses import dataclass, field
from datetime import datetime
from typing import Callable, Any
from collections import defaultdict

@dataclass(frozen=True)
class DomainEvent:
    """Base class for domain events."""
    event_id: str
    occurred_at: datetime
    aggregate_id: str
    event_type: str = field(default="domain_event")

@dataclass(frozen=True)
class OrderCreatedEvent(DomainEvent):
    """Event emitted when order is created."""
    customer_id: str = ""
    total_amount: float = 0.0
    event_type: str = "order_created"

@dataclass(frozen=True)
class OrderShippedEvent(DomainEvent):
    """Event emitted when order is shipped."""
    tracking_number: str = ""
    event_type: str = "order_shipped"

class DomainEventPublisher:
    """
    Publish-subscribe system for domain events.
    Enables decoupled communication between bounded contexts.
    """
    
    def __init__(self):
        self._handlers: dict[str, list[Callable]] = defaultdict(list)
        self._event_log: list[DomainEvent] = []
    
    def subscribe(self, event_type: str, handler: Callable) -> None:
        """Subscribe handler to specific event type."""
        self._handlers[event_type].append(handler)
    
    def unsubscribe(self, event_type: str, handler: Callable) -> None:
        """Unsubscribe handler from event type."""
        if event_type in self._handlers:
            self._handlers[event_type].remove(handler)
    
    def publish(self, event: DomainEvent) -> None:
        """Publish event to all subscribed handlers."""
        self._event_log.append(event)
        
        handlers = self._handlers.get(event.event_type, [])
        for handler in handlers:
            handler(event)
        
        # Also notify wildcard handlers
        for handler in self._handlers.get('*', []):
            handler(event)
    
    def get_events(self, aggregate_id: str | None = None) -> list[DomainEvent]:
        """Get events, optionally filtered by aggregate."""
        if aggregate_id is None:
            return list(self._event_log)
        return [e for e in self._event_log if e.aggregate_id == aggregate_id]

# Usage: Domain events enable loose coupling between aggregates
publisher = DomainEventPublisher()

def send_order_confirmation(event: OrderCreatedEvent) -> None:
    print(f"Sending confirmation for order {event.aggregate_id}")

publisher.subscribe("order_created", send_order_confirmation)
publisher.publish(OrderCreatedEvent(
    event_id="EVT-001",
    occurred_at=datetime.now(),
    aggregate_id="ORD-123",
    customer_id="CUST-456",
    total_amount=199.99
))
```

**Why**: Domain Events enable asynchronous, decoupled communication between bounded contexts and aggregates.

```python
# PROBLEM: Implement functional index for computed/expression-based lookups
# TIME: O(n log n) for lookup SPACE: O(n)
from dataclasses import dataclass
from typing import Callable, Any
import bisect

@dataclass
class IndexEntry:
    """Entry in functional index."""
    computed_key: Any
    record_id: str
    value: Any

class FunctionalIndex:
    """
    Index based on computed function over records.
    Similar to database function-based indexes.
    """
    
    def __init__(self, key_function: Callable[[dict], Any]):
        self.key_function = key_function
        self.entries: list[IndexEntry] = []
        self.record_to_entry: dict[str, int] = {}
    
    def insert(self, record_id: str, record: dict) -> None:
        """Insert record into index with computed key."""
        computed_key = self.key_function(record)
        entry = IndexEntry(
            computed_key=computed_key,
            record_id=record_id,
            value=record
        )
        
        # Find insertion point to maintain sorted order
        keys = [e.computed_key for e in self.entries]
        idx = bisect.bisect_left(keys, computed_key)
        
        self.entries.insert(idx, entry)
        self._rebuild_record_map()
    
    def _rebuild_record_map(self) -> None:
        """Rebuild record_id to index mapping."""
        self.record_to_entry = {
            e.record_id: i for i, e in enumerate(self.entries)
        }
    
    def find_by_key(self, key: Any) -> list[IndexEntry]:
        """Find all entries matching computed key."""
        keys = [e.computed_key for e in self.entries]
        left = bisect.bisect_left(keys, key)
        right = bisect.bisect_right(keys, key)
        return self.entries[left:right]
    
    def find_range(self, low: Any, high: Any) -> list[IndexEntry]:
        """Find entries where computed key is in range."""
        keys = [e.computed_key for e in self.entries]
        left = bisect.bisect_left(keys, low)
        right = bisect.bisect_right(keys, high)
        return self.entries[left:right]
    
    def delete(self, record_id: str) -> bool:
        """Remove record from index."""
        if record_id not in self.record_to_entry:
            return False
        
        idx = self.record_to_entry[record_id]
        del self.entries[idx]
        self._rebuild_record_map()
        return True

# Usage: Functional index enables queries on computed values
users = [
    {'id': 'U1', 'first': 'John', 'last': 'Doe', 'age': 30},
    {'id': 'U2', 'first': 'Jane', 'last': 'Smith', 'age': 25},
]

# Index on full name (computed)
name_index = FunctionalIndex(lambda r: f"{r['last']}, {r['first']}")
for user in users:
    name_index.insert(user['id'], user)

# Index on age for range queries
age_index = FunctionalIndex(lambda r: r['age'])
for user in users:
    age_index.insert(user['id'], user)

# Find users aged 25-30
result = age_index.find_range(25, 30)
```

**Why**: Functional indexes enable efficient queries on computed/derived values without storing redundant data.

```python
# PROBLEM: Implement forward index for document-to-terms mapping
# TIME: O(1) for lookup SPACE: O(n * m) where m is avg terms per doc
from dataclasses import dataclass
from typing import set
from collections import defaultdict

@dataclass
class Document:
    """Document with content and metadata."""
    doc_id: str
    content: str
    metadata: dict

class ForwardIndex:
    """
    Forward index maps document IDs to their terms/content.
    Used in search engines alongside inverted index.
    """
    
    def __init__(self, tokenizer: Callable[[str], list[str]] = None):
        self.tokenizer = tokenizer or self._default_tokenizer
        self.documents: dict[str, Document] = {}
        self.doc_terms: dict[str, set[str]] = {}
        self.term_positions: dict[str, dict[str, list[int]]] = defaultdict(dict)
    
    @staticmethod
    def _default_tokenizer(text: str) -> list[str]:
        """Simple whitespace tokenizer with lowercasing."""
        return text.lower().split()
    
    def add_document(self, document: Document) -> None:
        """Add document to forward index with term extraction."""
        self.documents[document.doc_id] = document
        
        tokens = self.tokenizer(document.content)
        self.doc_terms[document.doc_id] = set(tokens)
        
        # Store term positions for phrase queries
        for position, term in enumerate(tokens):
            if document.doc_id not in self.term_positions[term]:
                self.term_positions[term][document.doc_id] = []
            self.term_positions[term][document.doc_id].append(position)
    
    def get_document(self, doc_id: str) -> Document | None:
        """Retrieve document by ID."""
        return self.documents.get(doc_id)
    
    def get_terms(self, doc_id: str) -> set[str]:
        """Get all terms in a document."""
        return self.doc_terms.get(doc_id, set())
    
    def get_term_positions(self, doc_id: str, term: str) -> list[int]:
        """Get positions of term in document."""
        return self.term_positions.get(term, {}).get(doc_id, [])
    
    def contains_term(self, doc_id: str, term: str) -> bool:
        """Check if document contains term."""
        return term in self.doc_terms.get(doc_id, set())
    
    def get_document_length(self, doc_id: str) -> int:
        """Get number of unique terms in document."""
        return len(self.doc_terms.get(doc_id, set()))
    
    def remove_document(self, doc_id: str) -> bool:
        """Remove document from index."""
        if doc_id not in self.documents:
            return False
        
        # Remove term positions
        for term in self.doc_terms[doc_id]:
            if doc_id in self.term_positions[term]:
                del self.term_positions[term][doc_id]
        
        del self.documents[doc_id]
        del self.doc_terms[doc_id]
        return True

# Usage: Forward index stores document content for retrieval
index = ForwardIndex()
index.add_document(Document(
    doc_id="DOC-1",
    content="The quick brown fox jumps over the lazy dog",
    metadata={'author': 'Unknown'}
))
```

**Why**: Forward index enables fast document retrieval and position-aware queries for search engines.

## Generic Patterns

```python
# PROBLEM: Implement type-safe generic container patterns
# TIME: O(1) or O(n) depending on operation SPACE: O(n)
from typing import TypeVar, Generic, Callable, Optional
from dataclasses import dataclass

T = TypeVar('T')
K = TypeVar('K')
V = TypeVar('V')

@dataclass
class Result(Generic[T]):
    """Generic Result monad for error handling."""
    value: Optional[T] = None
    error: Optional[str] = None
    
    def is_ok(self) -> bool:
        return self.error is None
    
    def is_error(self) -> bool:
        return self.error is not None
    
    def unwrap(self) -> T:
        if self.error:
            raise ValueError(f"Cannot unwrap error: {self.error}")
        return self.value
    
    def unwrap_or(self, default: T) -> T:
        return self.value if self.is_ok() else default
    
    def map(self, func: Callable[[T], T]) -> 'Result[T]':
        if self.is_error():
            return Result(error=self.error)
        try:
            return Result(value=func(self.value))
        except Exception as e:
            return Result(error=str(e))

class Cache(Generic[K, V]):
    """Generic cache with configurable eviction."""
    
    def __init__(self, max_size: int = 100):
        self.max_size = max_size
        self._data: dict[K, V] = {}
        self._access_order: list[K] = []
    
    def get(self, key: K) -> Result[V]:
        if key in self._data:
            # Move to end (LRU)
            self._access_order.remove(key)
            self._access_order.append(key)
            return Result(value=self._data[key])
        return Result(error=f"Key not found: {key}")
    
    def put(self, key: K, value: V) -> None:
        if key in self._data:
            self._access_order.remove(key)
        elif len(self._data) >= self.max_size:
            # Evict least recently used
            lru_key = self._access_order.pop(0)
            del self._data[lru_key]
        
        self._data[key] = value
        self._access_order.append(key)
    
    def compute_if_absent(self, key: K, supplier: Callable[[], V]) -> V:
        result = self.get(key)
        if result.is_ok():
            return result.unwrap()
        
        value = supplier()
        self.put(key, value)
        return value

class Repository(Generic[T]):
    """Generic repository pattern implementation."""
    
    def __init__(self):
        self._items: dict[str, T] = {}
    
    def save(self, id: str, item: T) -> None:
        self._items[id] = item
    
    def find_by_id(self, id: str) -> Result[T]:
        if id in self._items:
            return Result(value=self._items[id])
        return Result(error=f"Item not found: {id}")
    
    def find_all(self) -> list[T]:
        return list(self._items.values())
    
    def delete(self, id: str) -> bool:
        if id in self._items:
            del self._items[id]
            return True
        return False

# Usage: Generic patterns provide type-safe, reusable components
cache: Cache[str, int] = Cache(max_size=10)
cache.put("age", 30)
result = cache.get("age").map(lambda x: x + 1)
```

**Why**: Generic patterns enable type-safe, reusable components that work with any data type.

## Command Patterns

```python
# PROBLEM: Implement Command pattern for undo/redo operations
# TIME: O(1) per command SPACE: O(n) for history
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Any

class Command(ABC):
    """Abstract command interface."""
    
    @abstractmethod
    def execute(self) -> Any:
        """Execute the command."""
        pass
    
    @abstractmethod
    def undo(self) -> Any:
        """Undo the command."""
        pass
    
    @abstractmethod
    def redo(self) -> Any:
        """Redo the command."""
        pass

@dataclass
class TextEditor:
    """Simple text editor state."""
    content: str = ""
    cursor_pos: int = 0

class InsertTextCommand(Command):
    """Command to insert text at cursor position."""
    
    def __init__(self, editor: TextEditor, text: str):
        self.editor = editor
        self.text = text
        self.position = editor.cursor_pos
    
    def execute(self) -> None:
        self.editor.content = (
            self.editor.content[:self.position] + 
            self.text + 
            self.editor.content[self.position:]
        )
        self.editor.cursor_pos = self.position + len(self.text)
    
    def undo(self) -> None:
        end_pos = self.position + len(self.text)
        self.editor.content = (
            self.editor.content[:self.position] + 
            self.editor.content[end_pos:]
        )
        self.editor.cursor_pos = self.position
    
    def redo(self) -> None:
        self.execute()

class DeleteTextCommand(Command):
    """Command to delete text from cursor position."""
    
    def __init__(self, editor: TextEditor, length: int):
        self.editor = editor
        self.length = length
        self.position = editor.cursor_pos
        self.deleted_text = ""
    
    def execute(self) -> None:
        end_pos = self.position + self.length
        self.deleted_text = self.editor.content[self.position:end_pos]
        self.editor.content = (
            self.editor.content[:self.position] + 
            self.editor.content[end_pos:]
        )
    
    def undo(self) -> None:
        self.editor.content = (
            self.editor.content[:self.position] + 
            self.deleted_text + 
            self.editor.content[self.position:]
        )
        self.editor.cursor_pos = self.position
    
    def redo(self) -> None:
        self.execute()

class CommandHistory:
    """Manages command history for undo/redo."""
    
    def __init__(self, max_history: int = 100):
        self.history: list[Command] = []
        self.current_index: int = -1
        self.max_history = max_history
    
    def execute(self, command: Command) -> Any:
        """Execute command and add to history."""
        result = command.execute()
        
        # Remove any commands after current position
        self.history = self.history[:self.current_index + 1]
        
        self.history.append(command)
        if len(self.history) > self.max_history:
            self.history.pop(0)
        else:
            self.current_index += 1
        
        return result
    
    def undo(self) -> bool:
        """Undo last command."""
        if self.current_index >= 0:
            self.history[self.current_index].undo()
            self.current_index -= 1
            return True
        return False
    
    def redo(self) -> bool:
        """Redo next command."""
        if self.current_index < len(self.history) - 1:
            self.current_index += 1
            self.history[self.current_index].redo()
            return True
        return False

# Usage: Command pattern enables undo/redo functionality
editor = TextEditor()
history = CommandHistory()
history.execute(InsertTextCommand(editor, "Hello "))
history.execute(InsertTextCommand(editor, "World"))
history.undo()  # "Hello "
```

**Why**: Command pattern encapsulates operations as objects, enabling undo/redo, queuing, and logging.

```python
# PROBLEM: Implement State pattern for finite state machine behavior
# TIME: O(1) for transitions SPACE: O(n) for states
from abc import ABC, abstractmethod
from typing import Callable, Optional
from enum import Enum

class OrderState(ABC):
    """Abstract state interface."""
    
    @abstractmethod
    def process(self, order: 'Order') -> 'OrderState':
        """Process order in current state."""
        pass
    
    @abstractmethod
    def cancel(self, order: 'Order') -> 'OrderState':
        """Cancel order in current state."""
        pass
    
    @abstractmethod
    def ship(self, order: 'Order') -> 'OrderState':
        """Ship order in current state."""
        pass
    
    @property
    @abstractmethod
    def name(self) -> str:
        """State name."""
        pass

class NewState(OrderState):
    name = "NEW"
    
    def process(self, order: 'Order') -> OrderState:
        print(f"Processing order {order.order_id}")
        return ProcessingState()
    
    def cancel(self, order: 'Order') -> OrderState:
        print(f"Cancelling new order {order.order_id}")
        return CancelledState()
    
    def ship(self, order: 'Order') -> OrderState:
        raise ValueError("Cannot ship new order - must process first")

class ProcessingState(OrderState):
    name = "PROCESSING"
    
    def process(self, order: 'Order') -> OrderState:
        raise ValueError("Order already processing")
    
    def cancel(self, order: 'Order') -> OrderState:
        print(f"Cancelling processing order {order.order_id}")
        return CancelledState()
    
    def ship(self, order: 'Order') -> OrderState:
        print(f"Shipping order {order.order_id}")
        return ShippedState()

class ShippedState(OrderState):
    name = "SHIPPED"
    
    def process(self, order: 'Order') -> OrderState:
        raise ValueError("Cannot process shipped order")
    
    def cancel(self, order: 'Order') -> OrderState:
        print(f"Processing return for shipped order {order.order_id}")
        return ReturnedState()
    
    def ship(self, order: 'Order') -> OrderState:
        raise ValueError("Order already shipped")

class CancelledState(OrderState):
    name = "CANCELLED"
    
    def process(self, order: 'Order') -> OrderState:
        raise ValueError("Cannot process cancelled order")
    
    def cancel(self, order: 'Order') -> OrderState:
        raise ValueError("Order already cancelled")
    
    def ship(self, order: 'Order') -> OrderState:
        raise ValueError("Cannot ship cancelled order")

class ReturnedState(OrderState):
    name = "RETURNED"
    
    def process(self, order: 'Order') -> OrderState:
        raise ValueError("Cannot process returned order")
    
    def cancel(self, order: 'Order') -> OrderState:
        raise ValueError("Cannot cancel returned order")
    
    def ship(self, order: 'Order') -> OrderState:
        raise ValueError("Cannot ship returned order")

class Order:
    """Order with state machine behavior."""
    
    def __init__(self, order_id: str):
        self.order_id = order_id
        self._state: OrderState = NewState()
    
    @property
    def state(self) -> OrderState:
        return self._state
    
    def set_state(self, state: OrderState) -> None:
        self._state = state
    
    def process(self) -> None:
        new_state = self._state.process(self)
        self._state = new_state
        print(f"Order {self.order_id} now in state: {self._state.name}")
    
    def cancel(self) -> None:
        new_state = self._state.cancel(self)
        self._state = new_state
        print(f"Order {self.order_id} now in state: {self._state.name}")
    
    def ship(self) -> None:
        new_state = self._state.ship(self)
        self._state = new_state
        print(f"Order {self.order_id} now in state: {self._state.name}")

# Usage: State pattern encapsulates state-specific behavior
order = Order("ORD-001")
order.process()  # NEW -> PROCESSING
order.ship()     # PROCESSING -> SHIPPED
order.cancel()   # SHIPPED -> RETURNED
```

**Why**: State pattern encapsulates state-specific behavior and transitions, making state machines explicit and maintainable.
