**Why**: The `<option>` element defines items in a `<select>` dropdown. Each option has a value attribute (sent to server) and display text (shown to user). Essential for form inputs requiring predefined choices.

        // Add sortable functionality to table headers
        document.querySelectorAll('th[scope="col"]').forEach((header, index) => {
            header.style.cursor = 'pointer';
            header.addEventListener('click', () => sortTable(index));
        });
        
        // Sort table by column index
        function sortTable(columnIndex) {
            const table = document.getElementById('employeeTable');
            const tbody = table.querySelector('tbody');
            const rows = Array.from(tbody.querySelectorAll('tr'));
            
            // Sort rows based on column content
            rows.sort((a, b) => {
                const aText = a.cells[columnIndex].textContent;
                const bText = b.cells[columnIndex].textContent;
                return aText.localeCompare(bText);
            });
            
            // Re-append sorted rows
            rows.forEach(row => tbody.appendChild(row));
        }
    </script>

            video.preload = value;
            updateStatus(`Preload set to: ${value}`);
            
            // Load video with new preload setting
            video.load();
        }
        
        // Monitor buffering progress
        video.addEventListener('progress', () => {
            if (video.buffered.length > 0) {
                const buffered = video.buffered.end(0);
                const duration = video.duration || 0;
                const percent = Math.round((buffered / duration) * 100);
                updateStatus(`Buffered: ${percent}%`);
            }
        });
        
        // Display metadata when loaded
        video.addEventListener('loadedmetadata', () => {
            updateStatus(`
                Duration: ${Math.round(video.duration)}s<br>
                Dimensions: ${video.videoWidth}x${video.videoHeight}
            `);
        });
        
        // Update status display
        function updateStatus(message) {
            status.innerHTML = message;
        }
        
        // Initial status
        updateStatus('Current preload: ' + video.preload);
    </script>
</body>
</html>
```

**Why**: The `preload` attribute controls how much video data loads before playback: `none` (saves bandwidth), `metadata` (loads duration/dimensions only), `auto` (preloads entire video). Choose based on user behavior and bandwidth constraints to optimize performance.

        // Animate opacity with JavaScript
        function fadeIn(element, duration = 500) {
            element.style.opacity = 0;
            element.style.display = 'block';
            
            let start = null;
            
            function animate(timestamp) {
                if (!start) start = timestamp;
                const progress = timestamp - start;
                const opacity = Math.min(progress / duration, 1);
                
                element.style.opacity = opacity;
                
                if (progress < duration) {
                    requestAnimationFrame(animate);
                }
            }
            
            requestAnimationFrame(animate);
        }
        
        // Fade out element
        function fadeOut(element, duration = 500) {
            let start = null;
            
            function animate(timestamp) {
                if (!start) start = timestamp;
                const progress = timestamp - start;
                const opacity = Math.max(1 - progress / duration, 0);
                
                element.style.opacity = opacity;
                
                if (progress < duration) {
                    requestAnimationFrame(animate);
                } else {
                    element.style.display = 'none';
                }
            }
            
            requestAnimationFrame(animate);
        }
        
        // Example usage
        const box = document.querySelector('.box');
        box.addEventListener('click', () => fadeOut(box, 300));
    </script>

**Why**: `opacity` controls element transparency from 0 (invisible) to 1 (fully visible). Unlike `visibility: hidden`, opacity affects the entire element including children, and creates stacking contexts. Use for overlays, transitions, and visual hierarchy. Alternative: `rgba()` for background-only transparency.

    <script>
        // Simulate API call with delay
        function simulateApiCall(data, delay = 1000) {
            return new Promise((resolve) => {
                setTimeout(() => resolve(data), delay);
            });
        }
        
        // Async function to fetch user data
        async function fetchUserData() {
            const resultDiv = document.getElementById('result');
            resultDiv.className = 'result loading';
            resultDiv.textContent = 'Loading...';
            
            try {
                // Sequential async calls with await
                const user = await simulateApiCall(
                    { id: 1, name: 'John Doe' }, 
                    800
                );
                
                const posts = await simulateApiCall(
                    [{ id: 1, title: 'First Post' }], 
                    600
                );
                
                // Display results
                resultDiv.className = 'result';
                resultDiv.innerHTML = `
                    <strong>User:</strong> ${user.name}<br>
                    <strong>Posts:</strong> ${posts.length}
                `;
                
                return { user, posts };
                
            } catch (error) {
                resultDiv.className = 'result error';
                resultDiv.textContent = `Error: ${error.message}`;
            }
        }
        
        // Parallel execution with Promise.all
        async function fetchParallel() {
            const [users, products, orders] = await Promise.all([
                simulateApiCall(['user1', 'user2'], 500),
                simulateApiCall(['product1'], 700),
                simulateApiCall(['order1'], 600)
            ]);
            
            console.log('All data fetched in parallel:', { users, products, orders });
            return { users, products, orders };
        }
        
        // Error handling example
        async function fetchWithErrorHandling() {
            try {
                const response = await fetch('/api/data');
                
                if (!response.ok) {
                    throw new Error(`HTTP ${response.status}`);
                }
                
                const data = await response.json();
                return data;
                
            } catch (error) {
                console.error('Fetch failed:', error);
                throw error; // Re-throw for caller to handle
            }
        }
        
        // Top-level await (ES2022) - in modules only
        // const moduleData = await fetchModuleConfig();

**Why**: `async/await` provides syntactic sugar over Promises, making asynchronous code read like synchronous code. `await` pauses execution until Promise resolves. Use `try/catch` for error handling. For parallel operations, use `Promise.all()` with multiple awaits. Cleaner than `.then()` chains.

            contentDiv.textContent = userInput; // Escapes HTML
            // or

        }

    </div>
    
    <div class="log" id="log"></div>
    
    <script>
        const log = document.getElementById('log');
        
        // Helper to log events
        function logEvent(event) {
            const time = new Date().toLocaleTimeString();
            log.innerHTML += `[${time}] ${event.type} on ${event.target.id}<br>`;
            log.scrollTop = log.scrollHeight;
        }
        
        // mouseenter: fires once when entering element
        // Does NOT bubble to parent!
        document.getElementById('parent').addEventListener('mouseenter', (e) => {
            logEvent(e);
            e.target.style.background = '#27ae60';
        });
        
        document.getElementById('parent').addEventListener('mouseleave', (e) => {
            logEvent(e);
            e.target.style.background = '#e74c3c';
        });
        
        // mouseover: fires when entering element OR its children
        // BUBBLES to parent!
        document.getElementById('parent').addEventListener('mouseover', (e) => {
            console.log('mouseover (bubbles):', e.target.id);
        });
        
        // Practical example: Card hover effect
        function createHoverCard(element) {
            element.addEventListener('mouseenter', () => {
                element.classList.add('hovered');
                // Load lazy content, show tooltip, etc.
            });
            
            element.addEventListener('mouseleave', () => {
                element.classList.remove('hovered');
            });
        }
        
        // Tooltip on hover
        function showTooltip(element, text) {
            let tooltip = null;
            
            element.addEventListener('mouseenter', (e) => {
                tooltip = document.createElement('div');
                tooltip.textContent = text;
                tooltip.style.cssText = `
                    position: absolute;
                    background: black;
                    color: white;
                    padding: 5px 10px;
                    border-radius: 4px;
                    font-size: 12px;
                `;
                document.body.appendChild(tooltip);
                
                // Position tooltip
                const rect = element.getBoundingClientRect();
                tooltip.style.left = rect.left + 'px';
                tooltip.style.top = (rect.top - 30) + 'px';
            });
            
            element.addEventListener('mouseleave', () => {
                if (tooltip) tooltip.remove();
            });
        }
    </script>
</body>
</html>
```

**Why**: `mouseenter` fires when mouse enters element bounds, but doesn't bubble to parents (unlike `mouseover`). Use for hover effects, tooltips, lazy loading on hover. Paired with `mouseleave`. More efficient than `mouseover` when you don't need bubbling behavior.
