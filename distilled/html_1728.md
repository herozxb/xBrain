```html
    // Animation events: animationstart, animationiteration, animationend
    const box = document.getElementById('box');
    const log = document.getElementById('log');
    
    // Log helper function for clean output
    function logEvent(eventName, event) {
      const entry = document.createElement('div');
      entry.textContent = `${new Date().toLocaleTimeString()}: ${eventName} - elapsedTime: ${event.elapsedTime.toFixed(2)}s`;

    box.addEventListener('animationiteration', (e) => {
      logEvent('animationiteration', e);
    });
    
    // Fires when animation completes
    box.addEventListener('animationend', (e) => {
      logEvent('animationend', e);
      console.log('Animation completed:', e.animationName);
    });

      box.offsetHeight; // Trigger reflow

    };
    document.body.insertBefore(replayBtn, log);
```

```html
  </div>

  <script>
    // Transition events: transitionrun, transitionstart, transitionend, transitioncancel
    const card = document.getElementById('card');
    const status = document.getElementById('status');
    
    // Fires when transition is created (before delay)
    card.addEventListener('transitionrun', (e) => {
      console.log('Transition running:', e.propertyName);
    });
    
    // Fires when transition actually starts (after delay)
    card.addEventListener('transitionstart', (e) => {
      status.textContent = `Transition started: ${e.propertyName}`;
      status.style.background = '#fff3e0';
    });
    
    // Fires when transition completes
    card.addEventListener('transitionend', (e) => {
      status.textContent = `Transition ended: ${e.propertyName} (${e.elapsedTime.toFixed(2)}s)`;
      status.style.background = '#e8f5e9';
    });
    
    // Fires if transition is interrupted
    card.addEventListener('transitioncancel', (e) => {
      status.textContent = `Transition cancelled: ${e.propertyName}`;
      status.style.background = '#ffebee';
    });
    
    // Chain transitions programmatically
    function chainTransitions(element, properties) {
      return new Promise((resolve) => {
        let completed = 0;
        const total = properties.length;
        
        element.addEventListener('transitionend', function handler(e) {
          completed++;
          if (completed >= total) {
            element.removeEventListener('transitionend', handler);
            resolve();
          }
        });
      });
    }
  </script>
</body>
</html>
```

**Why**: Critical for choreographing multi-step UI animations and ensuring JavaScript runs only after visual transitions complete.

```html
  <div class="container" id="container">

      <li data-id="3">Item 3</li>
    </ul>
  </div>
  
  <div class="bubble-path" id="path"></div>

  <script>
    // Event bubbling: event propagates from target up through ancestors
    const container = document.getElementById('container');
    const list = document.getElementById('list');
    const pathEl = document.getElementById('path');
    
    function showBubblePath(event) {
      pathEl.innerHTML = '';
      
      // Build the propagation path
      const path = event.composedPath();
      
      path.forEach((el, index) => {
        if (el.nodeType === Node.ELEMENT_NODE) {
          const step = document.createElement('span');
          step.className = 'bubble-step';
          step.textContent = el.tagName.toLowerCase() + 
            (el.id ? `#${el.id}` : '') + 
            (el.className ? `.${el.className.split(' ')[0]}` : '');
          
          // Highlight the original target
          if (index === 0) {
            step.style.background = '#c8e6c9';
            step.style.fontWeight = 'bold';
          }
          
          pathEl.appendChild(step);
          
          if (index < path.length - 1 && path[index + 1].nodeType === Node.ELEMENT_NODE) {
            pathEl.appendChild(document.createTextNode(' → '));
          }
        }
      });
    }
    
    // Event delegation pattern using bubbling
    // Single listener handles all list items
    list.addEventListener('click', (e) => {
      // Check if a list item was clicked
      if (e.target.tagName === 'LI') {
        const itemId = e.target.dataset.id;
        console.log(`Item ${itemId} clicked via delegation`);
        showBubblePath(e);
      }
    });
    
    // Container also receives the bubbled event
    container.addEventListener('click', (e) => {
      console.log('Container received bubbled click');
    });
    
    // Document receives it last
    document.addEventListener('click', (e) => {
      console.log('Document received bubbled click');
    }, { capture: false }); // explicit bubbling phase
  </script>
</body>
</html>
```

**Why**: Foundation of event delegation—allows one parent listener to handle events from many children efficiently.

```html
    const scrollContainer = document.getElementById('scrollContainer');
    const touchArea = document.getElementById('touchArea');
    const scrollEventsEl = document.getElementById('scrollEvents');
    const touchEventsEl = document.getElementById('touchEvents');
    
    let scrollCount = 0;
    let touchCount = 0;
    
    // PASSIVE listener - browser can scroll immediately without waiting for JS
    // { passive: true } promises we won't call preventDefault()

      scrollCount++;
      scrollEventsEl.textContent = scrollCount;

    let supportsPassive = false;
    try {
      const opts = Object.defineProperty({}, 'passive', {
        get: function() { supportsPassive = true; }
      });
      window.addEventListener('test', null, opts);
    } catch (e) {}
    
    // Safe passive option
    const passiveOption = supportsPassive ? { passive: true } : false;
    
    // Why passive matters:
    // Without passive:true, browser waits for listener to finish
    // to check if preventDefault() was called before scrolling
    // This can cause scroll jank on touch devices
```

```html
    function createModal() {

      const okBtn = overlay.querySelector('#modalOk');

        if (e.target === overlay) {

          document.body.removeChild(overlay);

      }, { capture: true }); // Capture is key!

      // With capture, we can check if click was ON the modal first

      });
      
      cancelBtn.onclick = () => {
        document.body.removeChild(overlay);

      };

        document.body.removeChild(overlay);

      };

          document.body.removeChild(overlay);
          document.removeEventListener('keydown', escHandler);
          logMsg('Modal closed via Escape');
        }
      };

      logMsg('Modal opened with capture-phase close handler');
    }

      logMsg('click2 handler ran');
    });
```

```html
        this.shortcuts = new Map();
        this.init();

        document.addEventListener('keydown', (e) => {
          const handler = this.findHandler(e);
          if (handler) {
            e.preventDefault();
            handler.callback(e);
            this.showNotification(handler.description);
          }
        });
      }
      
      // Generate a unique key from event
      getKey(e) {
        const parts = [];
        if (e.ctrlKey || e.metaKey) parts.push('ctrl');
        if (e.altKey) parts.push('alt');
        if (e.shiftKey) parts.push('shift');
        
        // Use e.key for the actual key
        const key = e.key.toLowerCase();
        if (!['control', 'alt', 'shift', 'meta'].includes(key)) {
          parts.push(key);
        }
        
        return parts.join('+');

        // Parse combo string like "ctrl+shift+s"
        const normalized = combo.toLowerCase().replace(/\s+/g, '').split('+').sort().join('+');
        this.shortcuts.set(normalized, { callback, description });
      }
      
      // Find matching handler for event
      findHandler(e) {
        const key = this.getKey(e);
        // Sort parts for consistent matching
        const normalized = key.split('+').sort().join('+');
        return this.shortcuts.get(normalized);
      }

        const notif = document.getElementById('notification');
        notif.textContent = message;
        notif.classList.add('show');
```

```html
    const gestureArea = document.getElementById('gestureArea');
    const box = document.getElementById('box');
    const gestureTypeEl = document.getElementById('gestureType');
    const scaleEl = document.getElementById('scale');
    const rotationEl = document.getElementById('rotation');
    const positionEl = document.getElementById('position');
    const logEl = document.getElementById('log');
    
    // Gesture state
    let state = {
      // Single touch / drag
      startX: 0,
      startY: 0,
      lastX: 0,
      lastY: 0,
      translateX: 0,
      translateY: 0,

      startDistance: 0,

      rotation: 0,

      entry.textContent = `${new Date().toLocaleTimeString()}: ${msg}`;
      logEl.insertBefore(entry, logEl.firstChild);
      if (logEl.children.length > 20) {

    function updateDisplay() {
      box.style.transform = `translate(${state.translateX}px, ${state.translateY}px) scale(${state.scale}) rotate(${state.rotation}deg)`;
      scaleEl.textContent = state.scale.toFixed(2);

      const dy = touch2.clientY - touch1.clientY;

      const dy = endY - state.swipeStartY;
      const dt = Date.now() - state.swipeStartTime;
      
      const minDistance = 50;
      const maxTime = 300;
      
      if (dt > maxTime) return null;
      
      const absDx = Math.abs(dx);
      const absDy = Math.abs(dy);
      
      if (absDx > minDistance || absDy > minDistance) {
        if (absDx > absDy) {
          return dx > 0 ? 'right' : 'left';

        const angleDiff = currentAngle - state.startAngle;
        state.rotation += angleDiff;
        state.startAngle = currentAngle;
```
