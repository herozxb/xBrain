## Aho-Corasick Algorithm

```python
# PROBLEM: Multi-pattern string matching - find all occurrences of multiple patterns in text
# TIME: O(n + m + z) where n = text length, m = total pattern length, z = matches
# SPACE: O(m * alphabet_size) for the trie

from collections import deque, defaultdict

class AhoCorasick:
    """Multi-pattern string matching using Aho-Corasick algorithm."""
    
    def __init__(self, patterns):
        """Initialize Aho-Corasick automaton with given patterns."""
        self.trie = defaultdict(dict)
        self.fail = {}
        self.output = defaultdict(list)
        self.state_count = 0
        self._build_trie(patterns)
        self._build_failure_links()
    
    def _build_trie(self, patterns):
        """Build initial trie from patterns."""
        for pattern_idx, pattern in enumerate(patterns):
            state = 0
            for char in pattern:
                if char not in self.trie[state]:
                    self.state_count += 1
                    self.trie[state][char] = self.state_count
                state = self.trie[state][char]
            self.output[state].append((pattern, pattern_idx))
    
    def _build_failure_links(self):
        """Build failure links using BFS for pattern matching."""
        queue = deque()
        
        # Initialize failure links for depth-1 nodes
        for char, state in self.trie[0].items():
            self.fail[state] = 0
            queue.append(state)
        
        # BFS to build remaining failure links
        while queue:
            current = queue.popleft()
            
            for char, next_state in self.trie[current].items():
                queue.append(next_state)
                
                # Find failure link
                fail_state = self.fail[current]
                while fail_state != 0 and char not in self.trie[fail_state]:
                    fail_state = self.fail[fail_state]
                
                self.fail[next_state] = self.trie[fail_state].get(char, 0)
                
                # Merge output from failure link
                self.output[next_state].extend(self.output[self.fail[next_state]])
    
    def search(self, text):
        """Find all pattern occurrences in text."""
        results = []
        state = 0
        
        for i, char in enumerate(text):
            # Follow failure links until we find a match or reach root
            while state != 0 and char not in self.trie[state]:
                state = self.fail[state]
            
            state = self.trie[state].get(char, 0)
            
            # Report all matches at current position
            for pattern, pattern_idx in self.output[state]:
                start_pos = i - len(pattern) + 1
                results.append((pattern, start_pos, i))
        
        return results

# Example usage
patterns = ["he", "she", "his", "hers"]
ac = AhoCorasick(patterns)
text = "ushers"
matches = ac.search(text)
# Output: [('she', 1, 3), ('he', 2, 3), ('hers', 2, 5)]
```

**Why**: Aho-Corasick efficiently searches for multiple patterns simultaneously in a single text pass, making it ideal for intrusion detection, spam filtering, and bioinformatics applications.

## Euler Tour Technique

```python
# PROBLEM: Tree traversal and subtree queries using linearization
# TIME: O(n) preprocessing, O(1) for subtree range queries
# SPACE: O(n)

class EulerTour:
    """Euler Tour Technique for tree linearization and subtree queries."""
    
    def __init__(self, n):
        """Initialize with number of nodes (0-indexed)."""
        self.n = n
        self.adj = [[] for _ in range(n)]
        self.euler = []          # Euler tour sequence
        self.first = [-1] * n    # First occurrence of each node
        self.last = [-1] * n     # Last occurrence of each node
        self.depth = [0] * n     # Depth of each node
        self.time = 0
    
    def add_edge(self, u, v):
        """Add undirected edge between u and v."""
        self.adj[u].append(v)
        self.adj[v].append(u)
    
    def build(self, root=0):
        """Build Euler tour starting from root."""
        self._dfs(root, -1, 0)
    
    def _dfs(self, node, parent, d):
        """DFS to generate Euler tour."""
        self.first[node] = self.time
        self.euler.append(node)
        self.depth[node] = d
        self.time += 1
        
        for child in self.adj[node]:
            if child != parent:
                self._dfs(child, node, d + 1)
                # Return to parent in Euler tour
                self.euler.append(node)
                self.time += 1
        
        self.last[node] = self.time - 1
    
    def get_subtree_range(self, node):
        """Get range [l, r] in Euler tour representing subtree of node."""
        return (self.first[node], self.last[node])
    
    def is_ancestor(self, u, v):
        """Check if u is ancestor of v using Euler tour property."""
        return self.first[u] <= self.first[v] and self.last[u] >= self.last[v]
    
    def get_lca_range(self, u, v):
        """Get LCA using Euler tour - returns node with minimum depth in range."""
        l = min(self.first[u], self.first[v])
        r = max(self.first[u], self.first[v])
        
        # In practice, use sparse table or segment tree for O(1) LCA
        min_depth_node = u if self.depth[u] < self.depth[v] else v
        for i in range(l, r + 1):
            if self.depth[self.euler[i]] < self.depth[min_depth_node]:
                min_depth_node = self.euler[i]
        return min_depth_node
```

```python
class CentroidDecomposition:
    """Centroid Decomposition for efficient tree divide-and-conquer."""
    
    def __init__(self, n):
        """Initialize with number of nodes."""
        self.n = n
        self.adj = [[] for _ in range(n)]
        self.parent = [-1] * n      # Parent in centroid tree
        self.subtree_size = [0] * n
        self.removed = [False] * n
    
    def add_edge(self, u, v):
        """Add undirected edge."""
        self.adj[u].append(v)
        self.adj[v].append(u)
    
    def _calc_subtree_size(self, node, par):
        """Calculate subtree sizes for centroid finding."""
        self.subtree_size[node] = 1
        for child in self.adj[node]:
            if child != par and not self.removed[child]:
                self._calc_subtree_size(child, node)
                self.subtree_size[node] += self.subtree_size[child]
    
    def _find_centroid(self, node, par, tree_size):
        """Find centroid of tree rooted at node."""
        for child in self.adj[node]:
            if child != par and not self.removed[child]:
                if self.subtree_size[child] > tree_size // 2:
                    return self._find_centroid(child, node, tree_size)
        return node
    
    def decompose(self, node=0, centroid_parent=-1):
        """Decompose tree starting from node with given centroid parent."""
        # Calculate subtree sizes
        self._calc_subtree_size(node, -1)
        
        # Find centroid
        centroid = self._find_centroid(node, -1, self.subtree_size[node])
        self.parent[centroid] = centroid_parent
        self.removed[centroid] = True
        
        # Recursively decompose remaining subtrees
        for child in self.adj[centroid]:
            if not self.removed[child]:
                self.decompose(child, centroid)
        
        return centroid
    
    def get_centroid_tree_depth(self):
        """Get depth of centroid tree (O(log n) for balanced tree)."""
        depth = [0] * self.n
        for i in range(self.n):
            node = i
            while self.parent[node] != -1:
                depth[i] += 1
                node = self.parent[node]
        return max(depth)
```

## Convex Hull Trick

```python
# PROBLEM: Optimize DP transitions of form: dp[i] = min(dp[j] + m[j]*x[i] + c[j])
# TIME: O(n) or O(n log n) depending on query order
# SPACE: O(n)

from collections import deque

class ConvexHullTrick:
    """
    Convex Hull Trick for optimizing linear DP transitions.
    Maintains lower hull for minimum queries (or upper for maximum).
    """
    
    def __init__(self, get_min=True):
        """
        Initialize CHT.
        get_min: True for minimum queries, False for maximum
        """
        self.lines = deque()  # Stores (m, c) tuples
        self.get_min = get_min
    
    def _intersect_x(self, l1, l2):
        """Get x-coordinate of intersection of two lines."""
        m1, c1 = l1
        m2, c2 = l2
        # Division by slope difference - use fraction for precision
        return (c2 - c1) / (m1 - m2)
    
    def _is_bad(self, l1, l2, l3):
        """Check if l2 is unnecessary (l3 becomes better before l2)."""
        return self._intersect_x(l1, l2) >= self._intersect_x(l2, l3)
    
    def add_line(self, m, c):
        """Add line y = mx + c to the hull."""
        # Adjust for max query
        if not self.get_min:
            m, c = -m, -c
        
        new_line = (m, c)
        
        # Remove lines that are no longer part of convex hull
        while len(self.lines) >= 2 and self._is_bad(
            self.lines[-2], self.lines[-1], new_line
        ):
            self.lines.pop()
        
        self.lines.append(new_line)
    
    def query(self, x):
        """Query minimum (or maximum) y value at given x."""
        # Binary search for x in monotonic case
        # For non-monotonic x, use full binary search
        if len(self.lines) == 0:
            return float('inf') if self.get_min else float('-inf')
        
        # Linear scan for deque (works when x is monotonic)
        while len(self.lines) >= 2:
            m1, c1 = self.lines[0]
            m2, c2 = self.lines[1]
            if m1 * x + c1 > m2 * x + c2:
                self.lines.popleft()
            else:
                break
        
        m, c = self.lines[0]
        result = m * x + c
        
        if not self.get_min:
            result = -result
        
        return result

class ConvexHullTrickBinary:
    """CHT with binary search for non-monotonic x queries."""
    
    def __init__(self, get_min=True):
        self.lines = []
        self.get_min = get_min
    
    def _intersect_x(self, l1, l2):
        m1, c1 = l1
        m2, c2 = l2
        return (c2 - c1) / (m1 - m2)
    
    def add_line(self, m, c):
        if not self.get_min:
            m, c = -m, -c
        
        new_line = (m, c)
        
        while len(self.lines) >= 2:
            if self._intersect_x(self.lines[-2], self.lines[-1]) >= \
               self._intersect_x(self.lines[-1], new_line):
                self.lines.pop()
            else:
                break
        
        self.lines.append(new_line)
    
    def query(self, x):
        """Binary search for optimal line."""
        if not self.lines:
            return float('inf') if self.get_min else float('-inf')
        
        lo, hi = 0, len(self.lines) - 1
        while lo < hi:
            mid = (lo + hi) // 2
            m1, c1 = self.lines[mid]
            m2, c2 = self.lines[mid + 1]
            
            if m1 * x + c1 > m2 * x + c2:
                lo = mid + 1
            else:
                hi = mid
        
        m, c = self.lines[lo]
        result = m * x + c
        
        if not self.get_min:
            result = -result
        
        return result

# Example: DP optimization
# dp[i] = min(dp[j] + a[i] * b[j]) for j < i
def solve_dp_with_cht(a, b):
    n = len(a)
    cht = ConvexHullTrickBinary(get_min=True)
    dp = [0] * n
    
    # dp[0] = 0 (base case)
    cht.add_line(b[0], dp[0])
    
    for i in range(1, n):
        dp[i] = cht.query(a[i])
        cht.add_line(b[i], dp[i])
    
    return dp
```

**Why**: Convex hull trick reduces O(n²) DP to O(n log n) or O(n), crucial for problems like "Batch Scheduling", "Commando", and various optimization problems with linear transitions.

## Li Chao Tree

```python
# PROBLEM: Dynamic line segment tree for maximum/minimum line queries at any point
# TIME: O(log C) per insert/query where C is coordinate range
# SPACE: O(Q log C) where Q is number of lines

class LiChaoTree:
    """
    Li Chao Segment Tree for dynamic line insertions and point queries.
    Supports inserting lines and querying max/min at any point.
    """
    
    def __init__(self, x_left, x_right, get_max=True):
        """
        Initialize Li Chao tree over range [x_left, x_right].
        get_max: True for maximum queries, False for minimum
        """
        self.x_left = x_left
        self.x_right = x_right
        self.get_max = get_max
        self.tree = {}  # node_id -> (m, c) line
    
    def _f(self, line, x):
        """Evaluate line at point x."""
        m, c = line
        return m * x + c
    
    def _better(self, line1, line2, x):
        """Check if line1 is better than line2 at point x."""
        val1 = self._f(line1, x)
        val2 = self._f(line2, x)
        if self.get_max:
            return val1 > val2
        return val1 < val2
    
    def insert(self, m, c, node=1, l=None, r=None):
        """Insert line y = mx + c into the tree."""
        if l is None:
            l = self.x_left
        if r is None:
            r = self.x_right
        
        new_line = (m, c)
        
        if node not in self.tree:
            self.tree[node] = new_line
            return
        
        mid = (l + r) // 2
        curr_line = self.tree[node]
        
        # Determine which line is better at mid
        if self._better(new_line, curr_line, mid):
            self.tree[node] = new_line
            new_line, curr_line = curr_line, new_line
        
        if l == r:
            return
        
        # Propagate the worse line to child where it might be better
        if self._better(new_line, curr_line, l) != self._better(new_line, curr_line, r):
            if self._better(new_line, curr_line, l):
                self.insert(new_line[0], new_line[1], 2*node, l, mid)
            else:
                self.insert(new_line[0], new_line[1], 2*node+1, mid+1, r)
    
    def query(self, x, node=1, l=None, r=None):
        """Query maximum/minimum value at point x."""
        if l is None:
            l = self.x_left
        if r is None:
            r = self.x_right
        
        result = float('-inf') if self.get_max else float('inf')
        
        if node in self.tree:
            result = self._f(self.tree[node], x)
        
        if l == r:
            return result
        
        mid = (l + r) // 2
        
        if x <= mid:
            child_result = self.query(x, 2*node, l, mid)
        else:
            child_result = self.query(x, 2*node+1, mid+1, r)
        
        if self.get_max:
            return max(result, child_result)
        return min(result, child_result)

        swap = False
        if self.get_max:
            if new_at_l >= curr_at_l and new_at_r >= curr_at_r:

        else:
            if new_at_l <= curr_at_l and new_at_r <= curr_at_r:

            self.tree[node] = new_line
            new_line = curr

            if new_at_l < self._f(self.tree[node], l):
```

```python
            self.size *= 2

        self.maxv = [float('-inf')] * (2 * self.size)
        self.smax = [float('-inf')] * (2 * self.size)  # Second max
        self.maxc = [0] * (2 * self.size)              # Count of max

        self.lazy = [None] * (2 * self.size)

        """Build segment tree from array."""
        # Initialize leaves
        for i in range(self.n):
            self.maxv[self.size + i] = arr[i]
            self.smax[self.size + i] = float('-inf')
            self.maxc[self.size + i] = 1
            self.sum[self.size + i] = arr[i]

        for i in range(self.size - 1, 0, -1):
            self._pull(i)
```

**Why**: Segment tree beats handles range min/max updates efficiently with amortized O(n log n) complexity, solving problems impossible with standard segment trees like range chmin with sum queries.

## Wavelet Tree

```python
# PROBLEM: String/array indexing for rank, select, and k-th smallest queries
# TIME: O(log σ) per query where σ is alphabet size
# SPACE: O(n log σ)

class WaveletTree:
    """
    Wavelet Tree for rank/select queries and k-th smallest.
    Supports counting occurrences and finding k-th element in ranges.
    """
    
    def __init__(self, arr, alphabet_size=None):
        """Build wavelet tree from array."""
        if alphabet_size is None:
            alphabet_size = max(arr) + 1 if arr else 1
        
        self.alphabet_size = alphabet_size
        self.arr = arr[:]
        self.n = len(arr)
        
        # Build tree
        self.bitvectors = {}
        self.prefix_sums = {}
        self._build(0, alphabet_size - 1, arr, 1)
    
    def _build(self, lo, hi, arr, node):
        """Recursively build wavelet tree node."""
        if lo == hi or not arr:
            return
        
        mid = (lo + hi) // 2
        
        # Create bitvector: 1 if element goes right, 0 if left
        bitvector = []
        left_arr = []
        right_arr = []
        
        for x in arr:
            if x <= mid:
                bitvector.append(0)
                left_arr.append(x)
            else:
                bitvector.append(1)
                right_arr.append(x)
        
        self.bitvectors[node] = bitvector
        
        # Build prefix sums for rank queries
        prefix = [0]
        for b in bitvector:
            prefix.append(prefix[-1] + b)
        self.prefix_sums[node] = prefix
        
        # Recurse on children
        self._build(lo, mid, left_arr, 2*node)
        self._build(mid+1, hi, right_arr, 2*node+1)
    
    def _rank(self, bitvector, prefix, pos, bit):
        """Count occurrences of bit in bitvector[0:pos]."""
        if bit == 0:
            return pos - prefix[pos]
        return prefix[pos]
    
    def rank(self, c, pos):
        """Count occurrences of character c in arr[0:pos]."""
        return self._rank_helper(c, pos, 0, self.alphabet_size-1, 1)
    
    def _rank_helper(self, c, pos, lo, hi, node):
        """Recursive rank helper."""
        if lo == hi:
            return pos
        
        if node not in self.bitvectors:
            return 0
        
        mid = (lo + hi) // 2
        bitvector = self.bitvectors[node]
        prefix = self.prefix_sums[node]
        
        if c <= mid:
            # Go left
            new_pos = self._rank(bitvector, prefix, pos, 0)
            return self._rank_helper(c, new_pos, lo, mid, 2*node)
        else:
            # Go right
            new_pos = self._rank(bitvector, prefix, pos, 1)
            return self._rank_helper(c, new_pos, mid+1, hi, 2*node+1)
    
    def kth_smallest(self, l, r, k):
        """Find k-th smallest element in arr[l:r]."""
        return self._kth_helper(l, r, k, 0, self.alphabet_size-1, 1)
    
    def _kth_helper(self, l, r, k, lo, hi, node):
        """Recursive k-th smallest helper."""
        if lo == hi:
            return lo
        
        if node not in self.bitvectors:
            return lo
        
        mid = (lo + hi) // 2
        bitvector = self.bitvectors[node]
        prefix = self.prefix_sums[node]
        
        # Count elements going left (smaller elements)
        left_count = r - l - (self._rank(bitvector, prefix, r, 1) - 
                              self._rank(bitvector, prefix, l, 1))
        
        if k <= left_count:
            # k-th is in left subtree
            new_l = self._rank(bitvector, prefix, l, 0)
            new_r = self._rank(bitvector, prefix, r, 0)
            return self._kth_helper(new_l, new_r, k, lo, mid, 2*node)
        else:
            # k-th is in right subtree
            new_l = self._rank(bitvector, prefix, l, 1)
            new_r = self._rank(bitvector, prefix, r, 1)
            return self._kth_helper(new_l, new_r, k - left_count, mid+1, hi, 2*node+1)
    
    def range_count(self, l, r, a, b):
        """Count elements in arr[l:r] with value in [a, b]."""
        return self._range_count_helper(l, r, a, b, 0, self.alphabet_size-1, 1)
    
    def _range_count_helper(self, l, r, a, b, lo, hi, node):
        """Recursive range count helper."""
        if b < lo or hi < a or l >= r:
            return 0
        
        if a <= lo and hi <= b:
            return r - l
        
        if node not in self.bitvectors:
            return 0
        
        mid = (lo + hi) // 2
        bitvector = self.bitvectors[node]
        prefix = self.prefix_sums[node]
        
        # Map positions to children
        left_l = self._rank(bitvector, prefix, l, 0)
        left_r = self._rank(bitvector, prefix, r, 0)
        right_l = self._rank(bitvector, prefix, l, 1)
        right_r = self._rank(bitvector, prefix, r, 1)
        
        return (self._range_count_helper(left_l, left_r, a, b, lo, mid, 2*node) +
                self._range_count_helper(right_l, right_r, a, b, mid+1, hi, 2*node+1))

# Example usage
arr = [3, 1, 4, 1, 5, 9, 2, 6, 5, 3, 5]
wt = WaveletTree(arr)

# Count occurrences of 5 in arr[0:11]
count_5 = wt.rank(5, 11)  # 3

# Find 4th smallest in arr[2:8]
kth = wt.kth_smallest(2, 8, 4)  # 5

# Count elements in [3, 5] in arr[1:9]
range_cnt = wt.range_count(1, 9, 3, 5)  # 4
```

**Why**: Wavelet trees enable efficient rank/select queries and range quantile queries, useful in compressed text indexes, computational biology, and range query problems.

```python
# PROBLEM: Find all intersections of horizontal and vertical line segments
# TIME: O((n + k) log n) where n = segments, k = intersections
# SPACE: O(n)

from collections import defaultdict
from bisect import bisect_left, bisect_right, insort

class LineSweepIntersections:
    """
    Line sweep algorithm to find intersections of axis-aligned segments.
    Sweeps from left to right, maintaining active horizontal segments.
    """
    
    def __init__(self):
        self.horizontal = []  # (y, x1, x2)
        self.vertical = []    # (x, y1, y2)
    
    def add_horizontal(self, y, x1, x2):
        """Add horizontal segment at y from x1 to x2."""
        if x1 > x2:
            x1, x2 = x2, x1
        self.horizontal.append((y, x1, x2))
    
    def add_vertical(self, x, y1, y2):
        """Add vertical segment at x from y1 to y2."""
        if y1 > y2:
            y1, y2 = y2, y1
        self.vertical.append((x, y1, y2))
    
    def find_intersections(self):
        """Find all intersection points between segments."""
        # Create events: (x, type, data)
        # Type: 0 = vertical, 1 = horizontal start, 2 = horizontal end
        events = []
        
        for x, y1, y2 in self.vertical:
            events.append((x, 0, (y1, y2)))
        
        for y, x1, x2 in self.horizontal:
            events.append((x1, 1, y))
            events.append((x2, 2, y))
        
        # Sort events by x-coordinate
        events.sort()
        
        intersections = []
        active_y = []  # Sorted list of active horizontal segment y-coordinates
        
        for x, event_type, data in events:
            if event_type == 0:  # Vertical segment
                y1, y2 = data
                # Find active horizontal segments in range [y1, y2]
                left = bisect_left(active_y, y1)
                right = bisect_right(active_y, y2)
                
                for i in range(left, right):
                    intersections.append((x, active_y[i]))
            
            elif event_type == 1:  # Horizontal start
                y = data
                insort(active_y, y)
            
            else:  # Horizontal end
                y = data
                idx = bisect_left(active_y, y)
                if idx < len(active_y) and active_y[idx] == y:
                    active_y.pop(idx)
        
        return intersections

class RectangleArea:
    """Calculate total area covered by rectangles using line sweep."""
    
    def __init__(self):
        self.rectangles = []
    
    def add_rectangle(self, x1, y1, x2, y2):
        """Add rectangle with corners (x1,y1) and (x2,y2)."""
        self.rectangles.append((x1, y1, x2, y2))
    
    def total_area(self):
        """Calculate total area covered by all rectangles."""
        if not self.rectangles:
            return 0
        
        # Get all unique y-coordinates
        y_coords = set()
        for x1, y1, x2, y2 in self.rectangles:
            y_coords.add(y1)
            y_coords.add(y2)
        
        y_list = sorted(y_coords)
        y_to_idx = {y: i for i, y in enumerate(y_list)}
        
        # Create events: (x, delta, y1, y2)
        events = []
        for x1, y1, x2, y2 in self.rectangles:
            events.append((x1, 1, y1, y2))
            events.append((x2, -1, y1, y2))
        
        events.sort()
        
        # Count of active rectangles for each y-interval
        count = [0] * (len(y_list) - 1)
        
        total = 0
        prev_x = events[0][0] if events else 0
        
        for x, delta, y1, y2 in events:
            # Calculate covered height at previous x
            width = x - prev_x
            if width > 0:
                covered_height = 0
                for i in range(len(y_list) - 1):
                    if count[i] > 0:
                        covered_height += y_list[i+1] - y_list[i]
                total += width * covered_height
            
            # Update counts
            idx1 = y_to_idx[y1]
            idx2 = y_to_idx[y2]
            for i in range(idx1, idx2):
                count[i] += delta
            
            prev_x = x
        
        return total

class ClosestPairSweep:
    """Find closest pair of points using line sweep (alternative to divide-conquer)."""
    
    def __init__(self, points):
        """Initialize with list of (x, y) points."""
        self.points = points
    
    def find_closest_pair(self):
        """Find closest pair of points."""
        if len(self.points) < 2:
            return None, float('inf')
        
        # Sort by x-coordinate
        sorted_points = sorted(self.points)
        
        # Active points within strip
        active = []
        min_dist = float('inf')
        closest_pair = None
        
        for point in sorted_points:
            x, y = point
            
            # Remove points too far in x
            while active and active[0][0] < x - min_dist:
                active.pop(0)
            
            # Check points within y-range
            for ax, ay in active:
                if abs(ay - y) < min_dist:
                    dist = ((x - ax)**2 + (y - ay)**2) ** 0.5
                    if dist < min_dist:
                        min_dist = dist
                        closest_pair = ((ax, ay), (x, y))
            
            insort(active, point)
        
        return closest_pair, min_dist

```

## Rotating Calipers

```python
# PROBLEM: Find diameter, width, and distance measures of convex polygon
# TIME: O(n) after convex hull computation
# SPACE: O(1)

import math

class RotatingCalipers:
    """
    Rotating Calipers algorithm for convex polygon measurements.
    Computes diameter, width, and other geometric properties.
    """
    
    def __init__(self, convex_hull):
        """Initialize with convex hull vertices in CCW order."""
        self.hull = convex_hull
        self.n = len(convex_hull)
    
    def _cross(self, o, a, b):
        """Cross product of OA and OB vectors."""
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
    
    def _distance(self, p1, p2):
        """Euclidean distance between two points."""
        return math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)
    
    def _point_line_distance(self, point, line_p1, line_p2):
        """Distance from point to line defined by two points."""
        x0, y0 = point
        x1, y1 = line_p1
        x2, y2 = line_p2
        
        # Distance = |Ax0 + By0 + C| / sqrt(A^2 + B^2)
        # where line is Ax + By + C = 0
        num = abs((y2-y1)*x0 - (x2-x1)*y0 + x2*y1 - y2*x1)
        den = math.sqrt((y2-y1)**2 + (x2-x1)**2)
        
        return num / den if den > 0 else 0
    
    def diameter(self):
        """Find the diameter (maximum distance) of the convex hull."""
        if self.n < 2:
            return 0
        if self.n == 2:
            return self._distance(self.hull[0], self.hull[1])
        
        max_dist = 0
        j = 1
        
        for i in range(self.n):
            # Find antipodal point for edge i -> i+1
            while True:
                next_j = (j + 1) % self.n
                curr_dist = self._distance(self.hull[i], self.hull[j])
                next_dist = self._distance(self.hull[i], self.hull[next_j])
                
                if next_dist > curr_dist:
                    j = next_j
                else:
                    break
            
            max_dist = max(max_dist, self._distance(self.hull[i], self.hull[j]))
        
        return max_dist
    
    def width(self):
        """Find the width (minimum distance between parallel supporting lines)."""
        if self.n < 2:
            return 0
        
        min_width = float('inf')
        j = 1
        
        for i in range(self.n):
            next_i = (i + 1) % self.n
            
            # Move j to find farthest point from edge (i, i+1)
            while True:
                next_j = (j + 1) % self.n
                
                # Calculate distance from j and next_j to edge (i, i+1)
                curr_dist = self._point_line_distance(
                    self.hull[j], self.hull[i], self.hull[next_i]
                )
                next_dist = self._point_line_distance(
                    self.hull[next_j], self.hull[i], self.hull[next_i]
                )
                
                if next_dist > curr_dist:
                    j = next_j
                else:
                    break
            
            width = self._point_line_distance(
                self.hull[j], self.hull[i], self.hull[next_i]
            )
            min_width = min(min_width, width)
        
        return min_width
    
    def minimum_enclosing_rectangle(self):
        """Find minimum area enclosing rectangle."""
        if self.n < 3:
            return 0, None
        
        min_area = float('inf')
        best_rect = None
        
        j = 1  # Antipodal point
        k = 1
        l = 1
        
        for i in range(self.n):
            next_i = (i + 1) % self.n
            
            # Direction of edge
            dx = self.hull[next_i][0] - self.hull[i][0]
            dy = self.hull[next_i][1] - self.hull[i][1]
            
            # Find farthest point from edge (antipodal)
            while True:
                next_j = (j + 1) % self.n
                if abs(self._cross(self.hull[i], self.hull[next_i], self.hull[next_j])) > \
                   abs(self._cross(self.hull[i], self.hull[next_i], self.hull[j])):
                    j = next_j
                else:
                    break
            
            # Find rightmost point
            while True:
                next_k = (k + 1) % self.n
                if (self.hull[next_k][0] - self.hull[i][0]) * dx + \
                   (self.hull[next_k][1] - self.hull[i][1]) * dy > \
                   (self.hull[k][0] - self.hull[i][0]) * dx + \
                   (self.hull[k][1] - self.hull[i][1]) * dy:
                    k = next_k
                else:
                    break
            
            # Find leftmost point
            while True:
                next_l = (l + 1) % self.n
                if (self.hull[next_l][0] - self.hull[i][0]) * dx + \
                   (self.hull[next_l][1] - self.hull[i][1]) * dy < \
                   (self.hull[l][0] - self.hull[i][0]) * dx + \
                   (self.hull[l][1] - self.hull[i][1]) * dy:
                    l = next_l
                else:
                    break
            
            # Calculate rectangle dimensions
            height = self._point_line_distance(
                self.hull[j], self.hull[i], self.hull[next_i]
            )
            
            # Width calculation (project k and l onto edge direction)
            edge_len = math.sqrt(dx*dx + dy*dy)
            width = ((self.hull[k][0] - self.hull[l][0]) * dx + \
                     (self.hull[k][1] - self.hull[l][1]) * dy) / edge_len
            
            area = height * width
            if area < min_area:
                min_area = area
                best_rect = (self.hull[i], self.hull[next_i], self.hull[j], self.hull[k])
        
        return min_area, best_rect

def convex_hull_graham(points):
    """Compute convex hull using Graham scan."""
    if len(points) < 3:
        return points
    
    # Sort by x, then y
    points = sorted(points)
    
    # Build lower hull
    lower = []
    for p in points:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    
    # Build upper hull
    upper = []
    for p in reversed(points):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    
    return lower[:-1] + upper[:-1]

def cross(o, a, b):
    """Cross product helper."""
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

# Example usage
points = [(0, 0), (1, 4), (2, 2), (3, 5), (4, 1), (5, 3)]
hull = convex_hull_graham(points)
rc = RotatingCalipers(hull)

diameter = rc.diameter()        # Maximum distance
width = rc.width()              # Minimum width
area, rect = rc.minimum_enclosing_rectangle()
```

**Why**: Rotating calipers solves multiple convex polygon problems in O(n) time after hull computation, including diameter, width, bounding box, and tangent calculations.

## Closest Pair of Points

```python
# PROBLEM: Find the closest pair of points in 2D plane
# TIME: O(n log n)
# SPACE: O(n)

import math

class ClosestPair:
    """
    Divide and conquer algorithm to find closest pair of points.
    Recursively divides points and merges results with strip processing.
    """
    
    def __init__(self, points):
        """Initialize with list of (x, y) points."""
        self.points = points
        self.n = len(points)
    
    def _distance(self, p1, p2):
        """Euclidean distance between two points."""
        return math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)
    
    def _brute_force(self, points):
        """Brute force for small point sets."""
        min_dist = float('inf')
        closest = None
        
        for i in range(len(points)):
            for j in range(i + 1, len(points)):
                d = self._distance(points[i], points[j])
                if d < min_dist:
                    min_dist = d
                    closest = (points[i], points[j])
        
        return min_dist, closest
    
    def _strip_closest(self, strip, d):
        """Find closest pair in strip of width d."""
        min_dist = d
        closest = None
        
        # Sort strip by y-coordinate
        strip.sort(key=lambda p: p[1])
        
        # Check each point against next 7 points
        for i in range(len(strip)):
            for j in range(i + 1, len(strip)):
                # Break if y-distance exceeds current minimum
                if strip[j][1] - strip[i][1] >= min_dist:
                    break
                
                dist = self._distance(strip[i], strip[j])
                if dist < min_dist:
                    min_dist = dist
                    closest = (strip[i], strip[j])
        
        return min_dist, closest
    
    def find_closest_pair(self):
        """Main function to find closest pair of points."""
        if self.n < 2:
            return None, float('inf')
        
        # Sort by x-coordinate
        points = sorted(self.points, key=lambda p: p[0])
        
        return self._closest_recursive(points)
    
    def _closest_recursive(self, points):
        """Recursive divide and conquer."""
        n = len(points)
        
        if n <= 3:
            return self._brute_force(points)
        
        # Divide
        mid = n // 2
        mid_point = points[mid]
        
        left = points[:mid]
        right = points[mid:]
        
        # Conquer
        dl, pair_l = self._closest_recursive(left)
        dr, pair_r = self._closest_recursive(right)
        
        # Find minimum of two halves
        if dl < dr:
            d = dl
            closest = pair_l
        else:
            d = dr
            closest = pair_r
        
        # Create strip of points within d of vertical line
        strip = []
        for p in points:
            if abs(p[0] - mid_point[0]) < d:
                strip.append(p)
        
        # Find closest in strip
        ds, pair_s = self._strip_closest(strip, d)
        
        if ds < d:
            return ds, pair_s
        
        return d, closest

class ClosestPairOptimized:
    """Optimized implementation with pre-sorting by y-coordinate."""
    
    def __init__(self, points):
        self.points = points
    
    def _distance(self, p1, p2):
        return math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)
    
    def find_closest_pair(self):
        """Find closest pair using optimized divide and conquer."""
        if len(self.points) < 2:
            return None, float('inf')
        
        # Sort by both x and y
        px = sorted(self.points, key=lambda p: p[0])
        py = sorted(self.points, key=lambda p: p[1])
        
        return self._closest_util(px, py)
    
    def _closest_util(self, px, py):
        """Utility function with both sorted arrays."""
        n = len(px)
        
        if n <= 3:
            return self._brute_force(px)
        
        # Divide
        mid = n // 2
        mid_x = px[mid][0]
        
        # Split py into two halves maintaining y-order
        py_left = []
        py_right = []
        for p in py:
            if p[0] <= mid_x:
                py_left.append(p)
            else:
                py_right.append(p)
        
        # Recurse
        dl, pair_l = self._closest_util(px[:mid], py_left)
        dr, pair_r = self._closest_util(px[mid:], py_right)
        
        if dl < dr:
            d = dl
            closest = pair_l
        else:
            d = dr
            closest = pair_r
        
        # Build strip from pre-sorted py
        strip = [p for p in py if abs(p[0] - mid_x) < d]
        
        # Check strip
        min_dist = d
        for i in range(len(strip)):
            for j in range(i + 1, min(i + 8, len(strip))):
                if strip[j][1] - strip[i][1] >= min_dist:
                    break
                dist = self._distance(strip[i], strip[j])
                if dist < min_dist:
                    min_dist = dist
                    closest = (strip[i], strip[j])
        
        return min_dist, closest
    
    def _brute_force(self, points):
        min_dist = float('inf')
        closest = None
        
        for i in range(len(points)):
            for j in range(i + 1, len(points)):
                d = self._distance(points[i], points[j])
                if d < min_dist:
                    min_dist = d
                    closest = (points[i], points[j])
        
        return min_dist, closest
```
