```html
<!DOCTYPE html>

  <title>ES6 Module Import</title>

<body>

  <!-- Inline module simulation for demonstration -->
  <script type="module">
    // ===== MODULE: mathUtils.js (simulated) =====
    // In real code, this would be in a separate file
    const mathUtilsModule = `
      // Named exports
      export const PI = 3.14159;
      export const E = 2.71828;
      
      // Export functions
      export function add(a, b) {
        return a + b;
      }
      
      export function multiply(a, b) {
        return a * b;
      }
      
      export function square(n) {
        return n * n;
      }
      
      // Default export (one per module)
      export default class Calculator {
        constructor() {
          this.result = 0;
        }
        add(n) { this.result += n; return this; }
        subtract(n) { this.result -= n; return this; }
        getResult() { return this.result; }
      }
    `;
    
    // ===== MAIN SCRIPT =====
    // Named imports - specific exports from module
    import { add, multiply, PI } from './mathUtils.js';
    
    // Import with alias (rename)
    import { square as sq } from './mathUtils.js';
    
    // Import all as namespace object
    import * as MathUtils from './mathUtils.js';
    
    // Default import (no braces needed)
    import Calculator from './mathUtils.js';
    
    // Combined: default + named
    import Calculator, { add, multiply } from './mathUtils.js';
    
    // Dynamic import (returns Promise)
    async function loadModule() {
      const module = await import('./mathUtils.js');
      console.log(module.add(5, 3)); // 8
    }
    
    // Usage examples
    console.log(add(2, 3));           // 5
    console.log(multiply(4, 5));      // 20
    console.log(sq(7));               // 49
    console.log(MathUtils.PI);        // 3.14159
    
    const calc = new Calculator();
    calc.add(10).subtract(3);
    console.log(calc.getResult());    // 7
    
    // Output to page
    document.getElementById('output').innerHTML = `
      <p>add(2,3) = ${add(2,3)}</p>
      <p>multiply(4,5) = ${multiply(4,5)}</p>
      <p>PI = ${PI}</p>
    `;
  </script>
</body>
</html>
```
**Why**: ES6 modules provide native code organization with explicit imports/exports. Named imports allow tree-shaking (dead code elimination), while default exports provide convenient main exports.

## js_microtask_queue
```html
<!-- PROBLEM: Understand microtask queue execution order vs macrotask queue
     TIME: O(n) SPACE: O(n) -->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Microtask Queue Demo</title>
</head>
<body>
  <div id="output"></div>
  
  <script>
    // Microtask queue: Promise.then, queueMicrotask, MutationObserver
    // Macrotask queue: setTimeout, setInterval, setImmediate, I/O
    
    const output = [];
    const log = (msg) => {
      output.push(msg);
      console.log(msg);
    };
    
    log('1. Script start');
    
    // Macrotask - scheduled in macrotask queue
    setTimeout(() => {
      log('5. setTimeout callback');
    }, 0);
    
    // Microtask - scheduled in microtask queue
    Promise.resolve()
      .then(() => {
        log('3. Promise.then 1');
      })
      .then(() => {
        log('4. Promise.then 2');
      });
    
    // queueMicrotask - explicit microtask scheduling
    queueMicrotask(() => {
      log('3.5. queueMicrotask');
    });
    
    log('2. Script end');
    
    // Event loop processes:
    // 1. Execute synchronous code
    // 2. Process ALL microtasks before next macrotask
    // 3. Process one macrotask
    // 4. Repeat from step 2
    
    // Expected output order:
    // 1. Script start
    // 2. Script end
    // 3. Promise.then 1
    // 3.5. queueMicrotask
    // 4. Promise.then 2 (chained, runs after 3 resolves)
    // 5. setTimeout callback
    
    // Display results after event loop completes
    setTimeout(() => {
      document.getElementById('output').innerHTML = 
        '<pre>' + output.join('\n') + '</pre>';
    }, 100);
    
    // Practical example: ensuring code runs after DOM updates
    function updateAndProcess(element, newValue) {
      element.textContent = newValue;
      
      // Microtask runs after current code but before paint
      queueMicrotask(() => {
        console.log('DOM updated, before paint');
      });
      
      // Macrotask runs after paint
      setTimeout(() => {
        console.log('After paint and macrotask');
      }, 0);
    }
  </script>
</body>
</html>
```
**Why**: Understanding microtask vs macrotask queues is critical for async programming. Microtasks always run before the next macrotask, affecting timing of Promises, UI updates, and event handling.

## js_object_defineProperty
```html
<!-- PROBLEM: Define or modify object properties with fine-grained control
     TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Object.defineProperty</title>
</head>
<body>
  <div id="output"></div>
  
  <script>
    const user = {};
    
    // Define a property with full descriptor control
    Object.defineProperty(user, 'name', {
      value: 'Alice',        // Property value
      writable: false,       // Cannot be changed
      enumerable: true,      // Shows in for...in loop
      configurable: false    // Cannot be deleted or reconfigured
    });
    
    // Define getter/setter property
    let _age = 0;
    Object.defineProperty(user, 'age', {
      get() {
        return _age;
      },
      set(value) {
        if (value < 0) {
          throw new Error('Age cannot be negative');
        }
        _age = value;
      },
      enumerable: true,
      configurable: true
    });
    
    user.age = 25;  // Valid
    // user.age = -5;  // Throws Error
    
    // Create read-only computed property
    Object.defineProperty(user, 'birthYear', {
      get() {
        return new Date().getFullYear() - this.age;
      },
      enumerable: true
    });
    
    // Property descriptor object structure:
    // {
    //   value: any,           // Data descriptor
    //   writable: boolean,    // Can value be changed?
    //   get: function,        // Accessor descriptor
    //   set: function,        // Accessor descriptor
    //   enumerable: boolean,  // Visible in for...in/Object.keys?
    //   configurable: boolean // Can descriptor be changed/deleted?
    // }
    
    // Get property descriptor
    const descriptor = Object.getOwnPropertyDescriptor(user, 'name');
    console.log(descriptor);
    // { value: 'Alice', writable: false, enumerable: true, configurable: false }
    
    // Common use case: create constant
    function createConstant(obj, key, value) {
      Object.defineProperty(obj, key, {
        value,
        writable: false,
        enumerable: true,
        configurable: false
      });
    }
    
    createConstant(Math, 'DOUBLE_PI', Math.PI * 2);
    
    // Output demo
    document.getElementById('output').innerHTML = `
      <p>user.name: ${user.name}</p>
      <p>user.age: ${user.age}</p>
      <p>user.birthYear: ${user.birthYear}</p>
      <p>user.name writable: ${descriptor.writable}</p>
    `;
  </script>
</body>
</html>
```
**Why**: `Object.defineProperty` enables precise control over property behavior (read-only, computed, hidden). It's the foundation for reactive frameworks, data binding, and creating robust APIs.

## js_string_replaceAll
```html
<!-- PROBLEM: Replace all occurrences of a substring in a string
     TIME: O(n*m) SPACE: O(n) where n=string length, m=pattern length -->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>String.replaceAll</title>
</head>
<body>
  <div id="output"></div>
  
  <script>
    // Basic replaceAll usage
    const text = 'The quick brown fox jumps over the lazy dog';
    const noSpaces = text.replaceAll(' ', '-');
    console.log(noSpaces); 
    // 'The-quick-brown-fox-jumps-over-the-lazy-dog'
    
    // Case-sensitive replacement
    const mixed = 'Hello hello HELLO';
    console.log(mixed.replaceAll('hello', 'hi'));
    // 'Hello hi HELLO' (only lowercase replaced)
    
    // Replace all with special regex characters (auto-escaped)
    const path = 'C:\\Users\\Documents\\Files';
    const fixedPath = path.replaceAll('\\', '/');
    console.log(fixedPath); 
    // 'C:/Users/Documents/Files'
    
    // Replace all numbers with placeholder
    const version = 'v1.2.3.4';
    const simplified = version.replaceAll(/\d+/g, 'x');
    console.log(simplified); // 'vxxx'
    
    // Using replacer function with replaceAll
    const prices = 'Items: $5, $10, $25, $100';
    const withTax = prices.replaceAll(/\$(\d+)/g, (match, amount) => {
      const taxed = parseInt(amount) * 1.1;
      return `$${taxed.toFixed(2)}`;
    });
    console.log(withTax); 
    // 'Items: $5.50, $11.00, $27.50, $110.00'
    
    // Templating example
    const template = 'Hello {{name}}, welcome to {{place}}!';
    const data = { name: 'Alice', place: 'Wonderland' };
    
    let result = template;
    for (const [key, value] of Object.entries(data)) {
      result = result.replaceAll(`{{${key}}}`, value);
    }
    console.log(result); 
    // 'Hello Alice, welcome to Wonderland!'
    
    // Compare with replace() + global regex
    const str = 'aaa';
    str.replace('a', 'b');        // 'baa' (only first)
    str.replaceAll('a', 'b');     // 'bbb' (all)
    str.replace(/a/g, 'b');       // 'bbb' (same effect)
    
    // Output demo
    document.getElementById('output').innerHTML = `
      <p><strong>Original:</strong> ${text}</p>
      <p><strong>No spaces:</strong> ${noSpaces}</p>
      <p><strong>Template result:</strong> ${result}</p>
    `;
  </script>
</body>
</html>
```
**Why**: `replaceAll` is cleaner than using regex with global flag for simple substring replacement. It automatically escapes special characters, making it safer for user-provided search strings.

## js_addeventlistener_bubbling
```html
<!-- PROBLEM: Handle event bubbling in nested elements
     TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Event Bubbling</title>
  <style>
    .outer {
      padding: 40px;
      background: #3498db;
    }
    .middle {
      padding: 30px;
      background: #2ecc71;
    }
    .inner {
      padding: 20px;
      background: #e74c3c;
      color: white;
      cursor: pointer;
    }
    .log {
      margin-top: 20px;
      padding: 10px;
      background: #f5f5f5;
      max-height: 200px;
      overflow-y: auto;
      font-family: monospace;
      font-size: 12px;
    }
  </style>
</head>
<body>
  <div class="outer" id="outer">
    Outer (Grandparent)
    <div class="middle" id="middle">
      Middle (Parent)
      <div class="inner" id="inner">
        Inner (Target) - Click me!
      </div>
    </div>
  </div>
  
  <div class="log" id="log"></div>
  
  <script>
    const log = document.getElementById('log');
    
    function addLog(msg) {
      log.innerHTML += `<div>${msg}</div>`;
      log.scrollTop = log.scrollHeight;
    }
    
    // Event bubbling: events propagate from target UP to ancestors
    // Click order: inner -> middle -> outer -> body -> document
    
    // Third parameter false = bubbling phase (default)
    document.getElementById('outer').addEventListener('click', (e) => {
      addLog('OUTER - Bubbling phase');
      addLog(`  target: ${e.target.id}`);
      addLog(`  currentTarget: ${e.currentTarget.id}`);
    }, false);
    
    document.getElementById('middle').addEventListener('click', (e) => {
      addLog('MIDDLE - Bubbling phase');
      addLog(`  target: ${e.target.id}`);
      addLog(`  currentTarget: ${e.currentTarget.id}`);
    }, false);
    
    document.getElementById('inner').addEventListener('click', (e) => {
      addLog('INNER - Bubbling phase');
      addLog(`  target: ${e.target.id}`);
      addLog(`  currentTarget: ${e.currentTarget.id}`);
    }, false);
    
    // Stop bubbling with stopPropagation
    document.getElementById('inner').addEventListener('click', (e) => {
      // Uncomment to stop bubbling:
      // e.stopPropagation();
      addLog('(Event continues bubbling... uncomment stopPropagation to stop)');
    });
    
    // Capture phase (third param = true) - runs BEFORE bubbling
    // Order: document -> outer -> middle -> inner -> (bubbling back up)
    document.getElementById('outer').addEventListener('click', (e) => {
      addLog('OUTER - Capture phase (first!)');
    }, true);
    
    // Event delegation: handle events on parent for all children
    document.getElementById('outer').addEventListener('click', (e) => {
      if (e.target.classList.contains('inner')) {
        addLog('Delegation: Inner was clicked!');
      }
    });
    
    // Key properties:
    // e.target - the element that triggered the event
    // e.currentTarget - the element handling the event (where listener is)
    // e.eventPhase - 1=CAPTURING, 2=AT_TARGET, 3=BUBBLING
    
    addLog('Click the red box to see event bubbling!');
  </script>
</body>
</html>
```
**Why**: Event bubbling is fundamental to DOM event handling. Understanding it enables event delegation (efficient handlers on parents), proper event flow control, and debugging complex UI interactions.

## js_localStorage_clear
```html
<!-- PROBLEM: Clear all data from localStorage
     TIME: O(n) SPACE: O(1) where n=number of keys -->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>localStorage.clear()</title>
</head>
<body>
  <div id="output"></div>
  <button id="addBtn">Add Test Data</button>
  <button id="clearBtn">Clear localStorage</button>
  <button id="showBtn">Show Current Data</button>
  
  <script>
    const output = document.getElementById('output');
    
    function updateDisplay() {
      const keys = Object.keys(localStorage);
      if (keys.length === 0) {
        output.innerHTML = '<p>localStorage is empty</p>';
      } else {
        let html = '<p><strong>Current localStorage:</strong></p><ul>';
        for (let i = 0; i < localStorage.length; i++) {
          const key = localStorage.key(i);
          const value = localStorage.getItem(key);
          html += `<li><strong>${key}:</strong> ${value}</li>`;
        }
        html += '</ul>';
        output.innerHTML = html;
      }
    }
    
    // Add test data
    document.getElementById('addBtn').addEventListener('click', () => {
      localStorage.setItem('user', 'Alice');
      localStorage.setItem('theme', 'dark');
      localStorage.setItem('language', 'en');
      localStorage.setItem('timestamp', Date.now().toString());
      localStorage.setItem('preferences', JSON.stringify({
        notifications: true,
        autoSave: false
      }));
      updateDisplay();
    });
    
    // Clear all localStorage data
    document.getElementById('clearBtn').addEventListener('click', () => {
      // Get count before clearing
      const count = localStorage.length;
      
      // Clear ALL data in localStorage
      localStorage.clear();
      
      console.log(`Cleared ${count} items from localStorage`);
      updateDisplay();
    });
    
    // Show current data
    document.getElementById('showBtn').addEventListener('click', updateDisplay);
    
    // Alternative: selective removal
    function clearExcept(keepKeys) {
      const keysToRemove = [];
      for (let i = 0; i < localStorage.length; i++) {
        const key = localStorage.key(i);
        if (!keepKeys.includes(key)) {
          keysToRemove.push(key);
        }
      }
      keysToRemove.forEach(key => localStorage.removeItem(key));
    }
    
    // Usage: clearExcept(['theme', 'language']);
    
    // Check storage quota
    function getStorageSize() {
      let total = 0;
      for (let key in localStorage) {
        if (localStorage.hasOwnProperty(key)) {
          total += localStorage[key].length + key.length;
        }
      }
      return (total * 2 / 1024).toFixed(2) + ' KB'; // UTF-16 = 2 bytes/char
    }
    
    // Storage event (fires in other tabs/windows)
```
