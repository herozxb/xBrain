## 1. Longest Substring Without Repeating Characters
```python
# PROBLEM: Find longest substring with all unique characters
# TIME: O(n) SPACE: O(min(m, n)) where m is charset size
def length_of_longest_substring(s: str) -> int:
    char_index = {}  # sliding window with hash map
    left = max_len = 0
    
    for right, char in enumerate(s):
        if char in char_index and char_index[char] >= left:
            left = char_index[char] + 1  # shrink window
        char_index[char] = right
        max_len = max(max_len, right - left + 1)
    
    return max_len
```
**Why**: Classic sliding window with hash map tracking last seen position. O(n) single pass.

## 2. Container With Most Water
```python
# PROBLEM: Find two lines that form container with max area
# TIME: O(n) SPACE: O(1)
def max_area(height: list[int]) -> int:
    left, right = 0, len(height) - 1
    max_water = 0
    
    while left < right:
        area = min(height[left], height[right]) * (right - left)
        max_water = max(max_water, area)
        
        # Move pointer with smaller height
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1
    
    return max_water
```
**Why**: Two pointers from ends. Moving smaller height pointer is optimal - larger might find better match.

## 3. Combination Sum
```python
# PROBLEM: Find all unique combinations that sum to target
# TIME: O(N^(T/M+1)) SPACE: O(T/M) where T=target, M=min candidate
def combination_sum(candidates: list[int], target: int) -> list[list[int]]:
    result = []
    
    def backtrack(start: int, remaining: int, path: list[int]):
        if remaining == 0:
            result.append(path[:])
            return
        if remaining < 0:
            return
        
        for i in range(start, len(candidates)):
            path.append(candidates[i])
            backtrack(i, remaining - candidates[i], path)  # reuse same element
            path.pop()  # backtrack
    
    backtrack(0, target, [])
    return result
```
**Why**: Backtracking with pruning. Sorted candidates enable early termination.

## 4. Number of Islands
```python
# PROBLEM: Count connected '1' regions in grid
# TIME: O(m*n) SPACE: O(m*n) worst case recursion depth
class UnionFind:
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.count = 0
    
    def find(self, x: int) -> int:
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])  # path compression
        return self.parent[x]
    
    def union(self, x: int, y: int):
        px, py = self.find(x), self.find(y)
        if px == py:
            return
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        self.count -= 1

def num_islands(grid: list[list[str]]) -> int:
    if not grid:
        return 0
    
    rows, cols = len(grid), len(grid[0])
    uf = UnionFind(rows * cols)
    water = 0
    
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == '0':
                water += 1
                continue
            for dr, dc in [(1, 0), (0, 1)]:
                nr, nc = r + dr, c + dc
                if nr < rows and nc < cols and grid[nr][nc] == '1':
                    uf.union(r * cols + c, nr * cols + nc)
    
    return uf.count - water
```
**Why**: Union-Find with path compression and union by rank. Near O(α(n)) per operation.

## 5. Bloom Filter Implementation
```python
# PROBLEM: Space-efficient probabilistic set membership
# TIME: O(k) where k=hash functions SPACE: O(m) bits
import mmh3
from bitarray import bitarray

class BloomFilter:
    def __init__(self, expected_items: int, false_positive_rate: float):
        # Calculate optimal m (bits) and k (hash functions)
        self.m = int(-expected_items * np.log(false_positive_rate) / (np.log(2) ** 2))
        self.k = int(self.m / expected_items * np.log(2))
        self.bit_array = bitarray(self.m)
        self.bit_array.setall(0)
    
    def add(self, item: str):
        for seed in range(self.k):
            index = mmh3.hash(item, seed) % self.m
            self.bit_array[index] = 1
    
    def might_contain(self, item: str) -> bool:
        for seed in range(self.k):
            index = mmh3.hash(item, seed) % self.m
            if not self.bit_array[index]:
                return False
        return True  # may be false positive
```
**Why**: Trade certainty for space. Multiple hash functions reduce false positives. Never false negatives.

## 6. LRU Cache
```python
# PROBLEM: Cache with O(1) get/put, evicts least recently used
# TIME: O(1) SPACE: O(capacity)
from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = OrderedDict()
    
    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)  # mark as recently used
        return self.cache[key]
    
    def put(self, key: int, value: int):
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)  # evict LRU
```
**Why**: OrderedDict maintains insertion order. move_to_end/popitem are O(1). Double-linked list under the hood.

## 7. Token Bucket Rate Limiter
```python
# PROBLEM: Limit requests to N per second with burst allowance
# TIME: O(1) SPACE: O(1)
import time

class TokenBucket:
    def __init__(self, rate: float, capacity: int):
        self.rate = rate  # tokens per second
        self.capacity = capacity
        self.tokens = capacity
        self.last_time = time.time()
    
    def allow_request(self) -> bool:
        now = time.time()
        elapsed = now - self.last_time
        
        # Refill tokens based on elapsed time
        self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)
        self.last_time = now
        
        if self.tokens >= 1:
            self.tokens -= 1
            return True
        return False
```
**Why**: Token bucket allows bursts up to capacity while maintaining average rate. Simple and memory-efficient.

## 8. Consistent Hashing
```python
# PROBLEM: Distribute keys across servers with minimal redistribution on add/remove
# TIME: O(log n) lookup SPACE: O(n * replicas)
import hashlib
from bisect import bisect_right

class ConsistentHash:
    def __init__(self, replicas: int = 150):
        self.replicas = replicas
        self.ring = {}  # hash -> server
        self.sorted_keys = []
    
    def _hash(self, key: str) -> int:
        return int(hashlib.md5(key.encode()).hexdigest(), 16)
    
    def add_server(self, server: str):
        for i in range(self.replicas):
            hash_val = self._hash(f"{server}:{i}")
            self.ring[hash_val] = server
            self.sorted_keys.append(hash_val)
        self.sorted_keys.sort()
    
    def remove_server(self, server: str):
        for i in range(self.replicas):
            hash_val = self._hash(f"{server}:{i}")
            del self.ring[hash_val]
            self.sorted_keys.remove(hash_val)
    
    def get_server(self, key: str) -> str:
        if not self.ring:
            return None
        hash_val = self._hash(key)
        idx = bisect_right(self.sorted_keys, hash_val) % len(self.sorted_keys)
        return self.ring[self.sorted_keys[idx]]
```
**Why**: Virtual nodes (replicas) ensure even distribution. Only K/n keys remap on server change (K=keys, n=servers).

## 9. Minimum Window Substring
```python
# PROBLEM: Find minimum window in s containing all chars of t
# TIME: O(|s| + |t|) SPACE: O(1) charset limited
from collections import Counter

def min_window(s: str, t: str) -> str:
    if not s or not t:
        return ""
    
    need = Counter(t)
    missing = len(t)
    left = 0
    min_len, min_start = float('inf'), 0
    
    for right, char in enumerate(s):
        if need[char] > 0:
            missing -= 1
        need[char] -= 1
        
        # Found valid window, try to shrink
        while missing == 0:
            if right - left + 1 < min_len:
                min_len = right - left + 1
                min_start = left
            
            need[s[left]] += 1
            if need[s[left]] > 0:
                missing += 1
            left += 1
    
    return s[min_start:min_start + min_len] if min_len != float('inf') else ""
```
**Why**: Sliding window with counter. missing==0 means valid window. Shrink while valid for minimum.

## 10. Sliding Window Maximum
```python
# PROBLEM: Find max in each sliding window of size k
# TIME: O(n) SPACE: O(k)
from collections import deque

def max_sliding_window(nums: list[int], k: int) -> list[int]:
    result = []
    dq = deque()  # store indices, decreasing values
    
    for i, num in enumerate(nums):
        # Remove indices outside window
        while dq and dq[0] < i - k + 1:
            dq.popleft()
        
        # Remove smaller elements (they can't be max)
        while dq and nums[dq[-1]] < num:
            dq.pop()
        
        dq.append(i)
        
        if i >= k - 1:
            result.append(nums[dq[0]])
    
    return result
```
**Why**: Monotonic deque maintains decreasing order. Front always has maximum. O(1) amortized per element.
