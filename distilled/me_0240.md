## 1. Application Controller Pattern

```python
# PROBLEM: Centralize request dispatching logic and decouple view selection from business logic
# TIME: O(1) SPACE: O(n) where n is number of registered commands
class ApplicationController:
    """
    Central dispatcher that maps commands to handlers and views.
    Implements front controller with command-based routing.
    """
    
    def __init__(self):
        self._command_handlers = {}
        self._view_mappings = {}
    
    def register_command(self, command_name, handler_class, view_name):
        """Register a command with its handler and associated view."""
        self._command_handlers[command_name] = handler_class
        self._view_mappings[command_name] = view_name
    
    def process_request(self, command_name, request_data):
        """Dispatch request to appropriate handler and return view."""
        if command_name not in self._command_handlers:
            raise ValueError(f"Unknown command: {command_name}")
        
        # Get handler instance and execute
        handler = self._command_handlers[command_name]()
        result = handler.execute(request_data)
        
        # Return view name with processed data
        return {
            'view': self._view_mappings[command_name],
            'data': result
        }

# Example usage
class LoginHandler:
    def execute(self, data):
        # Business logic for login
        return {'user': data.get('username'), 'status': 'authenticated'}

# Setup
app_controller = ApplicationController()
app_controller.register_command('login', LoginHandler, 'dashboard_view')
response = app_controller.process_request('login', {'username': 'john'})
```

**Why**: Eliminates duplicate dispatch logic across controllers, provides single point for cross-cutting concerns (auth, logging), and decouples request handling from view rendering.

## 2. Service to Worker Pattern

```python
# PROBLEM: Coordinate workflow between service layer and worker tasks with clean separation
# TIME: O(m) where m is number of workers SPACE: O(w) for worker pool
from queue import Queue
from threading import Thread
import time

class ServiceToWorker:
    """
    Implements service-to-worker pattern where dispatcher coordinates
    between client requests and worker pool processing.
    """
    
    def __init__(self, num_workers=3):
        self.task_queue = Queue()
        self.workers = []
        self.result_store = {}
        
        # Initialize worker pool
        for i in range(num_workers):
            worker = Thread(target=self._worker_loop, args=(i,))
            worker.daemon = True
            worker.start()
            self.workers.append(worker)
    
    def _worker_loop(self, worker_id):
        """Worker thread that processes tasks from queue."""
        while True:
            task_id, task_func, args = self.task_queue.get()
            try:
                # Execute task and store result
                result = task_func(*args)
                self.result_store[task_id] = {
                    'status': 'completed',
                    'result': result,
                    'worker_id': worker_id
                }
            except Exception as e:
                self.result_store[task_id] = {
                    'status': 'failed',
                    'error': str(e),
                    'worker_id': worker_id
                }
            finally:
                self.task_queue.task_done()
    
    def submit_task(self, task_id, task_func, *args):
        """Submit task to worker pool."""
        self.task_queue.put((task_id, task_func, args))
    
    def get_result(self, task_id, timeout=10):
        """Retrieve result with timeout."""
        start = time.time()
        while time.time() - start < timeout:
            if task_id in self.result_store:
                return self.result_store.pop(task_id)
            time.sleep(0.1)
        return {'status': 'timeout'}

# Example usage
def process_data(data):
    time.sleep(1)  # Simulate work
    return data.upper()

coordinator = ServiceToWorker(num_workers=2)
coordinator.submit_task('task1', process_data, 'hello')
result = coordinator.get_result('task1')
# Output: {'status': 'completed', 'result': 'HELLO', 'worker_id': 0}
```

**Why**: Separates service coordination from task execution, enables concurrent processing with controlled resource usage, and provides clean abstraction for asynchronous workflows.

## 3. Business Delegate Pattern

```python
# PROBLEM: Decouple presentation layer from business services and hide complexity of remote calls
# TIME: O(1) SPACE: O(s) where s is number of services
class BusinessDelegate:
    """
    Acts as client-side facade to business services,
    handling lookup, exception handling, and abstraction.
    """
    
    def __init__(self):
        self._service_factory = BusinessServiceFactory()
        self._service_type = None
    
    def set_service_type(self, service_type):
        """Configure which business service to use."""
        self._service_type = service_type
    
    def do_task(self, *args, **kwargs):
        """Execute business operation with error handling."""
        if not self._service_type:
            raise ValueError("Service type not set")
        
        try:
            service = self._service_factory.get_service(self._service_type)
            return service.perform_operation(*args, **kwargs)
        except ConnectionError:
            # Handle communication failures gracefully
            return self._fallback_operation(*args, **kwargs)
        except Exception as e:
            # Log and transform business exceptions
            raise BusinessException(f"Operation failed: {e}")
    
    def _fallback_operation(self, *args, **kwargs):
        """Provide fallback when service unavailable."""
        return {'status': 'degraded', 'message': 'Service temporarily unavailable'}

class BusinessServiceFactory:
    """Factory for creating business service instances."""
    
    def __init__(self):
        self._services = {}
    
    def register_service(self, name, service_class):
        self._services[name] = service_class
    
    def get_service(self, name):
        if name not in self._services:
            raise ValueError(f"Unknown service: {name}")
        return self._services[name]()

# Example
class OrderService:
    def perform_operation(self, order_id):
        return {'order_id': order_id, 'status': 'processed'}

factory = BusinessServiceFactory()
factory.register_service('order', OrderService)

delegate = BusinessDelegate()
delegate.set_service_type('order')

```

**Why**: Reduces coupling between presentation and business tiers, centralizes service lookup and exception handling, and provides consistent interface despite backend service complexity.

```python
class SessionFacade:
    """
    Coarse-grained facade that encapsulates interactions
    between multiple business objects in a workflow.
    """

        self.inventory_system = InventorySystem()
        self.payment_processor = PaymentProcessor()
        self.notification_service = NotificationService()

        Single method that orchestrates entire order workflow

        """

        for item in items:
            if not self.inventory_system.check_availability(item['id'], item['qty']):
                return {'success': False, 'error': f"Item {item['id']} out of stock"}

        payment_result = self.payment_processor.charge(
            customer_id, 
            payment_info, 
            self._calculate_total(items)
        )
        
        if not payment_result['success']:
            return {'success': False, 'error': 'Payment failed'}

            self.inventory_system.decrease_stock(item['id'], item['qty'])

        self.notification_service.send_confirmation(customer_id, order['order_id'])
        
        return {
            'success': True,
            'order_id': order['order_id'],
            'transaction_id': payment_result['transaction_id']
        }
```

## 5. Composite Entity Pattern

```python
# PROBLEM: Model complex business entities composed of interdependent objects
# TIME: O(c) where c is number of components SPACE: O(c)
class CompositeEntity:
    """
    Represents a coarse-grained entity composed of multiple dependent objects
    with lifecycle managed as a single unit.
    """
    
    def __init__(self, entity_id):
        self.entity_id = entity_id
        self._components = {}
        self._dependent_objects = {}
        self._is_dirty = False
    
    def add_component(self, name, component):
        """Add a component to the composite."""
        self._components[name] = component
        self._is_dirty = True
    
    def add_dependent_object(self, name, dependent_obj):
        """Add dependent object that shares lifecycle."""
        self._dependent_objects[name] = dependent_obj
        self._is_dirty = True
    
    def get_component(self, name):
        """Retrieve component by name."""
        return self._components.get(name)
    
    def get_dependent(self, name):
        """Retrieve dependent object."""
        return self._dependent_objects.get(name)
    
    def save(self):
        """Persist entire composite as single unit."""
        if not self._is_dirty:
            return False
        
        # Save all components
        for component in self._components.values():
            if hasattr(component, 'save'):
                component.save()
        
        # Save dependent objects
        for dep_obj in self._dependent_objects.values():
            if hasattr(dep_obj, 'save'):
                dep_obj.save()
        
        self._is_dirty = False
        return True
    
    def delete(self):
        """Remove entire composite and all dependents."""
        # Delete in reverse dependency order
        for dep_obj in reversed(list(self._dependent_objects.values())):
            if hasattr(dep_obj, 'delete'):
                dep_obj.delete()
        
        for component in reversed(list(self._components.values())):
            if hasattr(component, 'delete'):
                component.delete()

# Example: Order composite with line items and shipping
class OrderComponent:
    def __init__(self, order_id, customer_id):
        self.order_id = order_id
        self.customer_id = customer_id
    
    def save(self):
        print(f"Saving order {self.order_id}")

class LineItemsDependent:
    def __init__(self, items):
        self.items = items
    
    def save(self):
        print(f"Saving {len(self.items)} line items")
    
    def delete(self):
        print("Deleting line items")

# Usage
order_entity = CompositeEntity('ORDER-001')
order_entity.add_component('order_info', OrderComponent('ORDER-001', 'CUST-123'))
order_entity.add_dependent_object('line_items', LineItemsDependent([{'sku': 'A1'}, {'sku': 'B2'}]))

order_entity.save()  # Persists all components atomically
```

**Why**: Manages complex object graphs as single units, ensures referential integrity across dependent objects, simplifies persistence logic, and reduces overhead of fine-grained entity access.

## 6. Event Storming

```python
# PROBLEM: Model complex domain processes through collaborative event discovery
# TIME: O(e * c) where e is events, c is connections SPACE: O(e + a) events + aggregates
from collections import defaultdict
from datetime import datetime

class EventStormingModel:
    """
    Tool for capturing and exploring domain events,
    commands, aggregates, and their relationships.
    """
    
    def __init__(self, domain_name):
        self.domain_name = domain_name
        self.events = {}
        self.commands = {}
        self.aggregates = {}
        self.policies = {}
        self.connections = defaultdict(list)
    
    def add_event(self, event_name, description, aggregate=None):
        """Register a domain event."""
        self.events[event_name] = {
            'name': event_name,
            'description': description,
            'aggregate': aggregate,
            'created': datetime.now()
        }
        
        if aggregate and aggregate in self.aggregates:
            self.connections[aggregate].append(('emits', event_name))
    
    def add_command(self, command_name, triggers_event=None, handled_by=None):
        """Register a command that initiates process."""
        self.commands[command_name] = {
            'name': command_name,
            'triggers': triggers_event,
            'handler': handled_by
        }
        
        if triggers_event:
            self.connections[command_name].append(('triggers', triggers_event))
    
    def add_aggregate(self, aggregate_name, state_fields=None):
        """Register aggregate root."""
        self.aggregates[aggregate_name] = {
            'name': aggregate_name,
            'state': state_fields or []
        }
    
    def add_policy(self, policy_name, reacts_to, executes):
        """Add reactive policy (when X happens, do Y)."""
        self.policies[policy_name] = {
            'reacts_to': reacts_to,
            'executes': executes
        }
        self.connections[reacts_to].append(('policy', policy_name))
    
    def explore_flow(self, start_event):
        """Trace event flow through the model."""
        flow = []
        visited = set()
        
        def trace(event_name, depth=0):
            if event_name in visited or depth > 5:
                return
            visited.add(event_name)
            
            event = self.events.get(event_name)
            if event:
                flow.append(f"{'  ' * depth}📍 {event_name}: {event['description']}")
            
            # Find policies that react to this event
            for conn_type, target in self.connections.get(event_name, []):
                if conn_type == 'policy':
                    policy = self.policies[target]
                    flow.append(f"{'  ' * depth}➡️ Policy: {target}")
                    trace(policy['executes'], depth + 1)
        
        trace(start_event)
        return '\n'.join(flow)
    
    def generate_documentation(self):
        """Export event storming model as text."""
        doc = [f"# Event Storming: {self.domain_name}\n"]
        
        doc.append("## Aggregates")
        for agg in self.aggregates.values():
            doc.append(f"- {agg['name']}")
        
        doc.append("\n## Events")
        for evt in self.events.values():
            doc.append(f"- {evt['name']}: {evt['description']}")
        
        doc.append("\n## Commands")
        for cmd in self.commands.values():
            doc.append(f"- {cmd['name']} → {cmd['triggers']}")
        
        return '\n'.join(doc)

# Example: E-commerce domain modeling
model = EventStormingModel("E-commerce Order")
model.add_aggregate("Order", ["order_id", "status", "total"])
model.add_aggregate("Inventory", ["product_id", "quantity"])

model.add_command("PlaceOrder", triggers_event="OrderPlaced", handled_by="Order")
model.add_event("OrderPlaced", "Customer submitted new order", aggregate="Order")
model.add_event("InventoryReserved", "Stock allocated for order", aggregate="Inventory")
model.add_event("PaymentProcessed", "Payment completed successfully", aggregate="Order")

model.add_policy("ReserveInventory", reacts_to="OrderPlaced", executes="InventoryReserved")

print(model.explore_flow("OrderPlaced"))
```

**Why**: Enables collaborative domain exploration, surfaces hidden business processes, identifies bounded contexts, and creates shared understanding between developers and domain experts.

## 7. Write Model Pattern (CQRS)

```python
# PROBLEM: Separate write operations from read models for optimized command handling
# TIME: O(1) for command dispatch SPACE: O(c) for command handlers
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Any, Dict
import uuid

# Command definitions
@dataclass
class Command:
    """Base command class."""
    command_id: str = None
    
    def __post_init__(self):
        if self.command_id is None:
            self.command_id = str(uuid.uuid4())

@dataclass
class CreateOrderCommand(Command):
    customer_id: str = None
    items: list = None

@dataclass
class CancelOrderCommand(Command):
    order_id: str = None
    reason: str = None

# Command handler interface
class CommandHandler(ABC):
    @abstractmethod
    def handle(self, command: Command) -> Dict[str, Any]:
        pass

# Write model (aggregate root)
class OrderWriteModel:
    """Write-only model for order mutations."""
    
    def __init__(self):
        self._orders = {}  # In production, this would be event-sourced
    
    def create_order(self, customer_id, items):
        order_id = str(uuid.uuid4())
        self._orders[order_id] = {
            'order_id': order_id,
            'customer_id': customer_id,
            'items': items,
            'status': 'created',
            'version': 1
        }
        # Emit event: OrderCreated
        return order_id
    
    def cancel_order(self, order_id, reason):
        if order_id not in self._orders:
            raise ValueError("Order not found")
        
        order = self._orders[order_id]
        if order['status'] == 'shipped':
            raise ValueError("Cannot cancel shipped order")
        
        order['status'] = 'cancelled'
        order['cancellation_reason'] = reason
        order['version'] += 1
        # Emit event: OrderCancelled
        return True

# Concrete command handlers
class CreateOrderHandler(CommandHandler):
    def __init__(self, write_model: OrderWriteModel):
        self.write_model = write_model
    
    def handle(self, command: CreateOrderCommand):
        order_id = self.write_model.create_order(command.customer_id, command.items)
        return {'success': True, 'order_id': order_id}

class CancelOrderHandler(CommandHandler):
    def __init__(self, write_model: OrderWriteModel):
        self.write_model = write_model
    
    def handle(self, command: CancelOrderCommand):
        success = self.write_model.cancel_order(command.order_id, command.reason)
        return {'success': success, 'order_id': command.order_id}

# Command dispatcher
class WriteModelDispatcher:
    """Routes commands to appropriate handlers."""
    
    def __init__(self):
        self._handlers = {}
    
    def register(self, command_type, handler: CommandHandler):
        self._handlers[command_type] = handler
    
    def dispatch(self, command: Command):
        handler = self._handlers.get(type(command))
        if not handler:
            raise ValueError(f"No handler for {type(command).__name__}")
        return handler.handle(command)

# Setup and usage
write_model = OrderWriteModel()
dispatcher = WriteModelDispatcher()

dispatcher.register(CreateOrderCommand, CreateOrderHandler(write_model))
dispatcher.register(CancelOrderCommand, CancelOrderHandler(write_model))

# Execute commands
create_cmd = CreateOrderCommand(customer_id="CUST-001", items=[{"sku": "PROD-A", "qty": 2}])
result = dispatcher.dispatch(create_cmd)
# result: {'success': True, 'order_id': 'uuid-here'}
```

**Why**: Optimizes write performance independent of read patterns, ensures consistency boundaries around aggregates, enables event sourcing, and separates concerns between mutation and query operations.

## 8. Expression Index Implementation

```python
# PROBLEM: Create indexes on computed expressions for faster query performance
# TIME: O(n log n) for index build, O(log n) for lookup SPACE: O(n)
from bisect import bisect_left, insort

class ExpressionIndex:
    """
    Database-style index on computed expressions rather than raw column values.
    Enables fast queries on transformed/derived data.
    """
    
    def __init__(self, expression_func):
        """
        Args:
            expression_func: Function that computes index key from record
        """
        self.expression_func = expression_func
        self.index = []  # Sorted list of (computed_key, record_id)
        self.records = {}  # record_id -> full record
    
    def insert(self, record_id, record):
        """Insert record and compute index key."""
        computed_key = self.expression_func(record)
        self.records[record_id] = record
        insort(self.index, (computed_key, record_id))
    
    def search_by_expression(self, value, operator='='):
        """
        Query index by computed expression value.
        Operators: '=', '<', '>', '<=', '>=', 'range'
        """
        results = []
        
        if operator == '=':
            # Binary search for exact match
            idx = bisect_left(self.index, (value, ''))
            while idx < len(self.index) and self.index[idx][0] == value:
                record_id = self.index[idx][1]
                results.append(self.records[record_id])
                idx += 1
        
        elif operator == '<':
            # All entries with key < value
            idx = bisect_left(self.index, (value, ''))
            for i in range(idx):
                record_id = self.index[i][1]
                results.append(self.records[record_id])
        
        elif operator == '>=':
            # All entries with key >= value
            idx = bisect_left(self.index, (value, ''))
            for i in range(idx, len(self.index)):
                record_id = self.index[i][1]
                results.append(self.records[record_id])
        
        elif operator == 'range':
            # Range query: value is tuple (start, end)
            start, end = value
            idx_start = bisect_left(self.index, (start, ''))
            idx_end = bisect_left(self.index, (end + '\uffff', ''))
            for i in range(idx_start, idx_end):
                record_id = self.index[i][1]
                results.append(self.records[record_id])
        
        return results
    
    def delete(self, record_id):
        """Remove record from index."""
        if record_id not in self.records:
            return False
        
        record = self.records[record_id]
        computed_key = self.expression_func(record)
        self.index.remove((computed_key, record_id))
        del self.records[record_id]
        return True

# Example: Index on UPPER(last_name)
users = [
    {'id': 1, 'name': 'Alice Johnson', 'age': 30, 'email': 'alice@example.com'},
    {'id': 2, 'name': 'Bob Smith', 'age': 25, 'email': 'bob@test.com'},
    {'id': 3, 'name': 'Charlie Brown', 'age': 35, 'email': 'charlie@example.com'},
]

# Create index on email domain (expression: extract domain from email)
def extract_domain(record):
    return record['email'].split('@')[1]

domain_index = ExpressionIndex(extract_domain)

for user in users:
    domain_index.insert(user['id'], user)

# Query: all users from example.com domain
results = domain_index.search_by_expression('example.com', operator='=')
# Returns: users with IDs 1 and 3

# Query: all users with domains starting before 'test.com'
results = domain_index.search_by_expression('test.com', operator='<')
```

**Why**: Avoids expensive expression computation at query time, enables indexed access to derived values, supports functional programming patterns in databases, and improves performance of computed column queries.

## 9. Vectorization Techniques

```python
# PROBLEM: Optimize batch operations using vectorized computation instead of loops
# TIME: O(n) with smaller constant factor SPACE: O(n)
import numpy as np
from typing import List, Callable

class VectorizedProcessor:
    """
    Applies vectorized operations to batch data
    using NumPy for SIMD-optimized computation.
    """
    
    @staticmethod
    def filter_records(records: List[dict], condition: Callable) -> List[dict]:
        """
        Vectorized filtering using boolean masks.
        Faster than list comprehension for large datasets.
        """
        # Extract field values as array
        values = np.array([condition(r) for r in records])
        mask = values.astype(bool)
        
        # Apply mask
        return [records[i] for i in range(len(records)) if mask[i]]
    
    @staticmethod
    def transform_column(records: List[dict], field: str, transform: Callable):
        """Apply transformation to entire column vectorized."""
        values = np.array([r[field] for r in records])
        transformed = transform(values)
        
        for i, record in enumerate(records):
            record[field] = transformed[i]
        
        return records
    
    @staticmethod
    def aggregate_grouped(records: List[dict], 
                         group_key: str, 
                         agg_field: str, 
                         agg_func: str = 'sum'):
        """
        Vectorized group-by aggregation.
        Groups records and applies aggregation function.
        """
        # Build group arrays
        groups = {}
        for record in records:
            key = record[group_key]
            if key not in groups:
                groups[key] = []
            groups[key].append(record[agg_field])
        
        # Vectorized aggregation per group
        results = {}
        for key, values in groups.items():
            arr = np.array(values)
            
            if agg_func == 'sum':
                results[key] = np.sum(arr)
            elif agg_func == 'mean':
                results[key] = np.mean(arr)
            elif agg_func == 'max':
                results[key] = np.max(arr)
            elif agg_func == 'min':
                results[key] = np.min(arr)
            elif agg_func == 'std':
                results[key] = np.std(arr)
        
        return results
    
    @staticmethod
    def normalize_features(records: List[dict], numeric_fields: List[str]):
        """
        Min-max normalization across multiple fields.
        Scales all numeric fields to [0, 1] range.
        """
        for field in numeric_fields:
            values = np.array([r.get(field, 0) for r in records])
            min_val = np.min(values)
            max_val = np.max(values)
            
            if max_val - min_val > 0:
                normalized = (values - min_val) / (max_val - min_val)
            else:
                normalized = np.zeros_like(values)
            
            for i, record in enumerate(records):
                record[f'{field}_normalized'] = float(normalized[i])
        
        return records
```

## 10. Query Parallelization

```python
# PROBLEM: Execute multiple database queries in parallel to reduce total latency
# TIME: O(max(t)) where t is individual query times SPACE: O(q) for query queue
from concurrent.futures import ThreadPoolExecutor, as_completed
from typing import List, Dict, Any, Callable
import time

class QueryParallelizer:
    """
    Executes independent queries in parallel threads
    and combines results with configurable strategies.
    """
    
    def __init__(self, max_workers: int = 4):
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    def execute_parallel(self, queries: List[Dict[str, Any]]) -> Dict[str, Any]:
        """
        Execute multiple queries in parallel and return combined results.
        
        Args:
            queries: List of {'name': str, 'query': str, 'params': dict}
        
        Returns:
            Dict mapping query names to results
        """
        futures = {}
        
        for query_spec in queries:
            future = self.executor.submit(
                self._execute_single,
                query_spec['query'],
                query_spec.get('params', {})
            )
            futures[future] = query_spec['name']
        
        results = {}
        for future in as_completed(futures):
            query_name = futures[future]
            try:
                results[query_name] = future.result()
            except Exception as e:
                results[query_name] = {'error': str(e)}
        
        return results
    
    def _execute_single(self, query: str, params: dict):
        """Execute single query (mock implementation)."""
        # In production, this would connect to actual database
        time.sleep(0.1)  # Simulate query latency
        return {'query': query, 'params': params, 'rows': 42}
    
    def execute_partitioned(self, 
                          base_query: str, 
                          partitions: List[Any],
                          partition_func: Callable) -> List[Any]:
        """
        Partition a large query across multiple parallel executions.
        
        Args:
            base_query: Query template
            partitions: List of partition keys (e.g., date ranges, regions)
            partition_func: Function to modify query for each partition
        """
        futures = []
        
        for partition in partitions:
            partitioned_query = partition_func(base_query, partition)
            future = self.executor.submit(
                self._execute_single,
                partitioned_query,
                {'partition': partition}
            )
            futures.append(future)
        
        # Collect and merge results
        all_results = []
        for future in as_completed(futures):
            try:
                result = future.result()
                all_results.extend(result.get('rows', []))
            except Exception:
                pass  # Handle partial failures
        
        return all_results
    
    def execute_with_timeout(self, queries: List[Dict], timeout_seconds: float):
        """Execute queries with overall timeout constraint."""
        futures = {}
        
        for query_spec in queries:
            future = self.executor.submit(
                self._execute_single,
                query_spec['query'],
                query_spec.get('params', {})
            )
            futures[future] = query_spec['name']
        
        results = {}
        for future in as_completed(futures, timeout=timeout_seconds):
            query_name = futures[future]
            try:
                results[query_name] = future.result()
            except Exception as e:
                results[query_name] = {'error': str(e)}
        
        # Cancel any incomplete futures
        for future in futures:
            if not future.done():
                future.cancel()
        
        return results
    
    def shutdown(self):
        """Clean up thread pool."""
        self.executor.shutdown(wait=True)

# Example: Parallel dashboard data loading
parallelizer = QueryParallelizer(max_workers=3)

queries = [
    {'name': 'recent_orders', 'query': 'SELECT * FROM orders WHERE date > ?', 'params': {'date': '2026-01-01'}},
    {'name': 'top_products', 'query': 'SELECT product_id, COUNT(*) FROM order_items GROUP BY product_id LIMIT 10'},
    {'name': 'customer_stats', 'query': 'SELECT COUNT(*) FROM customers WHERE status = ?', 'params': {'status': 'active'}},
]

# Execute all in parallel (total time ≈ max individual query time)
results = parallelizer.execute_parallel(queries)

# Partitioned query example (parallel range scans)
base_query = "SELECT * FROM events WHERE date BETWEEN ? AND ?"

def partition_by_month(query_template, month):
    start_date = f"2026-{month:02d}-01"
    end_date = f"2026-{month:02d}-28"
    return query_template.replace('?', f"'{start_date}'").replace('?', f"'{end_date}'")

months = [1, 2, 3, 4, 5, 6]
event_data = parallelizer.execute_partitioned(base_query, months, partition_by_month)

parallelizer.shutdown()
```
