## Counting Sort

```python
# PROBLEM: Sort an array of integers in O(n+k) time where k is the range
# TIME: O(n + k) SPACE: O(k)
def counting_sort(arr):
    """
    Non-comparison sorting algorithm that counts occurrences of each value.
    Works best when range of values (k) is not significantly larger than n.
    """
    if not arr:
        return arr
    
    # Find the range of values
    max_val = max(arr)
    min_val = min(arr)
    range_of_elements = max_val - min_val + 1
    
    # Create count array and output array
    count = [0] * range_of_elements
    output = [0] * len(arr)
    
    # Store count of each character
    for num in arr:
        count[num - min_val] += 1
    
    # Change count[i] so it contains actual position in output array
    for i in range(1, len(count)):
        count[i] += count[i - 1]
    
    # Build the output array (traverse backwards for stability)
    for i in range(len(arr) - 1, -1, -1):
        output[count[arr[i] - min_val] - 1] = arr[i]
        count[arr[i] - min_val] -= 1
    
    return output

# Example usage
if __name__ == "__main__":
    arr = [4, 2, 2, 8, 3, 3, 1]
    sorted_arr = counting_sort(arr)
    print(f"Sorted: {sorted_arr}")  # [1, 2, 2, 3, 3, 4, 8]
```

**Why**: Stable sort with linear time complexity when range is small. Perfect for sorting integers or objects with integer keys.

## Radix Sort

```python
# PROBLEM: Sort numbers digit by digit using counting sort as subroutine
# TIME: O(d * (n + k)) SPACE: O(n + k) where d is number of digits
def counting_sort_for_radix(arr, exp):
    """Helper function to sort array based on digit at position exp"""
    n = len(arr)
    output = [0] * n
    count = [0] * 10  # Digits 0-9
    
    # Store count of occurrences
    for i in range(n):
        index = arr[i] // exp
        count[index % 10] += 1
    
    # Change count[i] to contain actual position
    for i in range(1, 10):
        count[i] += count[i - 1]
    
    # Build output array (backwards for stability)
    for i in range(n - 1, -1, -1):
        index = arr[i] // exp
        output[count[index % 10] - 1] = arr[i]
        count[index % 10] -= 1
    
    # Copy output to arr
    for i in range(n):
        arr[i] = output[i]

def radix_sort(arr):
    """
    Sorts by processing individual digits, from least significant to most.
    Uses counting sort as a subroutine for each digit position.
    """
    if not arr:
        return arr
    
    # Find maximum number to know number of digits
    max_val = max(arr)
    
    # Do counting sort for every digit
    exp = 1
    while max_val // exp > 0:
        counting_sort_for_radix(arr, exp)
        exp *= 10
    
    return arr

# Example usage
if __name__ == "__main__":
    arr = [170, 45, 75, 90, 802, 24, 2, 66]
    radix_sort(arr)
    print(f"Sorted: {arr}")  # [2, 24, 45, 66, 75, 90, 170, 802]
```

**Why**: Linear time sort for fixed-width integers. Can handle larger ranges than counting sort by processing digit-by-digit.

## Dijkstra's Shortest Path

```python
# PROBLEM: Find shortest path from source to all vertices in weighted graph
# TIME: O((V + E) log V) SPACE: O(V) with binary heap
import heapq

def dijkstra_shortest_path(graph, start):
    """
    Find shortest paths from start vertex to all other vertices.
    Graph: adjacency dict {vertex: [(neighbor, weight), ...]}
    Returns: distances dict and predecessors dict
    """
    # Initialize distances with infinity
    distances = {vertex: float('infinity') for vertex in graph}
    distances[start] = 0
    
    # Priority queue: (distance, vertex)
    pq = [(0, start)]
    predecessors = {vertex: None for vertex in graph}
    visited = set()
    
    while pq:
        current_dist, current_vertex = heapq.heappop(pq)
        
        # Skip if already visited (better path found)
        if current_vertex in visited:
            continue
        visited.add(current_vertex)
        
        # Explore neighbors
        for neighbor, weight in graph[current_vertex]:
            if neighbor in visited:
                continue
            
            distance = current_dist + weight
            
            # Relaxation step
            if distance < distances[neighbor]:
                distances[neighbor] = distance
                predecessors[neighbor] = current_vertex
                heapq.heappush(pq, (distance, neighbor))
    
    return distances, predecessors

def reconstruct_path(predecessors, start, end):
    """Reconstruct path from start to end using predecessors"""
    path = []
    current = end
    while current is not None:
        path.append(current)
        current = predecessors[current]
    return path[::-1] if path[-1] == start else []

# Example usage
if __name__ == "__main__":
    graph = {
        'A': [('B', 4), ('C', 2)],
        'B': [('A', 4), ('C', 1), ('D', 5)],
        'C': [('A', 2), ('B', 1), ('D', 8), ('E', 10)],
        'D': [('B', 5), ('C', 8), ('E', 2)],
        'E': [('C', 10), ('D', 2)]
    }
    
    distances, predecessors = dijkstra_shortest_path(graph, 'A')
    print(f"Distances: {distances}")
    path = reconstruct_path(predecessors, 'A', 'E')
    print(f"Path to E: {path}")
```

**Why**: Optimal for non-negative edge weights. Foundation for GPS navigation and network routing protocols.

## Bellman-Ford Algorithm

```python
# PROBLEM: Find shortest paths with negative edges and detect negative cycles
# TIME: O(V * E) SPACE: O(V)
def bellman_ford(graph, vertices, start):
    """
    Computes shortest paths from start vertex.
    Can handle negative edge weights and detects negative cycles.
    Graph: list of (u, v, weight) edges
    """
    # Initialize distances
    distances = {v: float('infinity') for v in vertices}
    distances[start] = 0
    predecessors = {v: None for v in vertices}
    
    # Relax all edges V-1 times
    for _ in range(len(vertices) - 1):
        updated = False
        for u, v, weight in graph:
            if distances[u] != float('infinity') and distances[u] + weight < distances[v]:
                distances[v] = distances[u] + weight
                predecessors[v] = u
                updated = True
        if not updated:  # Early termination if no update
            break
    
    # Check for negative cycles
    has_negative_cycle = False
    for u, v, weight in graph:
        if distances[u] != float('infinity') and distances[u] + weight < distances[v]:
            has_negative_cycle = True
            break
    
    return distances, predecessors, has_negative_cycle

# Example usage
if __name__ == "__main__":
    vertices = ['A', 'B', 'C', 'D', 'E']
    edges = [
        ('A', 'B', 4),
        ('A', 'C', 2),
        ('B', 'C', -3),
        ('C', 'D', 5),
        ('D', 'B', 1)
    ]
    
    distances, predecessors, has_cycle = bellman_ford(edges, vertices, 'A')
    print(f"Distances: {distances}")
    print(f"Negative cycle: {has_cycle}")
```

**Why**: Handles negative weights and detects negative cycles. Used in currency arbitrage detection and distributed systems.

## Floyd-Warshall Algorithm

```python
# PROBLEM: Find all-pairs shortest paths in weighted graph
# TIME: O(V^3) SPACE: O(V^2)
def floyd_warshall(graph, vertices):
    """
    Compute shortest distances between all pairs of vertices.
    Graph: adjacency dict {u: {v: weight}}
    Returns: distance matrix and next-hop matrix for path reconstruction
    """
    n = len(vertices)
    vertex_to_idx = {v: i for i, v in enumerate(vertices)}
    
    # Initialize distance matrix
    dist = [[float('infinity')] * n for _ in range(n)]
    next_hop = [[None] * n for _ in range(n)]
    
    # Set diagonal to 0
    for i in range(n):
        dist[i][i] = 0
    
    # Initialize with edge weights
    for u in graph:
        for v, weight in graph[u].items():
            i, j = vertex_to_idx[u], vertex_to_idx[v]
            dist[i][j] = weight
            next_hop[i][j] = v
    
    # Main algorithm
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    next_hop[i][j] = next_hop[i][k]
    
    # Detect negative cycles (diagonal becomes negative)
    has_negative_cycle = any(dist[i][i] < 0 for i in range(n))
    
    return dist, next_hop, vertices, has_negative_cycle

def get_path(next_hop, vertices, start_idx, end_idx):
    """Reconstruct path using next-hop matrix"""
    if next_hop[start_idx][end_idx] is None:
        return []
    path = [vertices[start_idx]]
    current = start_idx
    while current != end_idx:
        current = vertices.index(next_hop[current][end_idx])
        path.append(vertices[current])
    return path

# Example usage
if __name__ == "__main__":
    graph = {
        'A': {'B': 3, 'C': 8},
        'B': {'A': 3, 'C': 2, 'D': 5},
        'C': {'A': 8, 'B': 2, 'D': 1},
        'D': {'B': 5, 'C': 1}
    }
    vertices = ['A', 'B', 'C', 'D']
    
    dist, next_hop, verts, has_cycle = floyd_warshall(graph, vertices)
    print("Distance matrix:")
    for row in dist:
        print([f"{x:5.1f}" for x in row])
```

**Why**: Computes all-pairs shortest paths efficiently for dense graphs. Used in network analysis and transitive closure.

## Mo's Algorithm (Query Square Root Decomposition)

```python
# PROBLEM: Answer offline range queries efficiently using reordering
# TIME: O((n + q) * sqrt(n)) SPACE: O(n)
import math

class MosAlgorithm:
    """
    Efficiently answers range queries by clever ordering.
    Processes queries in sqrt(n) sized blocks.
    """
    def __init__(self, arr):
        self.arr = arr
        self.n = len(arr)
        self.block_size = int(math.sqrt(self.n))
        self.current_l = 0
        self.current_r = -1
        self.current_answer = 0
        self.freq = {}  # Frequency map for current window
    
    def add(self, idx):
        """Add element at idx to current window"""
        val = self.arr[idx]
        self.freq[val] = self.freq.get(val, 0) + 1
        # Update answer based on problem (example: distinct count)
        if self.freq[val] == 1:
            self.current_answer += 1
    
    def remove(self, idx):
        """Remove element at idx from current window"""
        val = self.arr[idx]
        self.freq[val] -= 1
        if self.freq[val] == 0:
            self.current_answer -= 1
    
    def query(self, queries):
        """
        Process multiple range queries [l, r].
        queries: list of (l, r) tuples (0-indexed)
        Returns: dict mapping query index to answer
        """
        # Add query index and sort by Mo's order
        indexed_queries = [(i, l, r) for i, (l, r) in enumerate(queries)]
        
        # Sort by (block of l, r) - odd blocks reverse r order
        indexed_queries.sort(key=lambda x: (
            x[1] // self.block_size,
            x[2] if (x[1] // self.block_size) % 2 == 0 else -x[2]
        ))
        
        results = {}
        for idx, l, r in indexed_queries:
            # Expand/shrink window to [l, r]
            while self.current_l > l:
                self.current_l -= 1
                self.add(self.current_l)
            
            while self.current_r < r:
                self.current_r += 1
                self.add(self.current_r)
            
            while self.current_l < l:
                self.remove(self.current_l)
                self.current_l += 1
            
            while self.current_r > r:
                self.remove(self.current_r)
                self.current_r -= 1
            
            results[idx] = self.current_answer
        
        return results

# Example usage
if __name__ == "__main__":
    arr = [1, 1, 2, 1, 3, 4, 5, 2, 8, 3]
    queries = [(0, 4), (1, 3), (2, 8)]
    
    mo = MosAlgorithm(arr)
    results = mo.query(queries)
    for i, ans in sorted(results.items()):
        print(f"Query {i} {queries[i]}: distinct count = {ans}")
```

**Why**: Dramatically improves performance for multiple range queries. Essential for competitive programming and database query optimization.

## Square Root Decomposition

```python
# PROBLEM: Balance query and update operations in O(sqrt(n))
# TIME: O(sqrt(n)) per query/update SPACE: O(n)
import math

class SqrtDecomposition:
    """
    Divide array into sqrt(n) blocks for efficient range operations.
    Supports point updates and range queries.
    """
    def __init__(self, arr):
        self.arr = arr[:]
        self.n = len(arr)
        self.block_size = int(math.sqrt(self.n)) + 1
        self.num_blocks = (self.n + self.block_size - 1) // self.block_size
        self.blocks = [0] * self.num_blocks
        
        # Precompute block sums
        for i in range(self.n):
            self.blocks[i // self.block_size] += arr[i]
    
    def update(self, idx, new_val):
        """Update element at idx and maintain blocks"""
        if idx < 0 or idx >= self.n:
            return
        
        block_idx = idx // self.block_size
        self.blocks[block_idx] += new_val - self.arr[idx]
        self.arr[idx] = new_val
    
    def range_query(self, l, r):
        """Sum of elements in range [l, r]"""
        if l > r or l < 0 or r >= self.n:
            return 0
        
        result = 0
        block_l = l // self.block_size
        block_r = r // self.block_size
        
        if block_l == block_r:
            # Same block, iterate directly
            for i in range(l, r + 1):
                result += self.arr[i]
        else:
            # Left partial block
            for i in range(l, (block_l + 1) * self.block_size):
                if i < self.n:
                    result += self.arr[i]
            
            # Full blocks
            for i in range(block_l + 1, block_r):
                result += self.blocks[i]
            
            # Right partial block
            for i in range(block_r * self.block_size, r + 1):
                if i < self.n:
                    result += self.arr[i]
        
        return result

# Example usage
if __name__ == "__main__":
    arr = [1, 5, 2, 4, 6, 1, 3, 5, 7, 2]
    sd = SqrtDecomposition(arr)
    
    print(f"Sum [0, 4]: {sd.range_query(0, 4)}")  # 18
    sd.update(2, 10)  # arr[2] = 10
    print(f"Sum [0, 4] after update: {sd.range_query(0, 4)}")  # 26
```

**Why**: Simple alternative to segment trees. Good balance between implementation complexity and performance.

## MVVM Pattern

```python
# PROBLEM: Separate UI logic from business logic with data binding
# TIME: O(1) for notification SPACE: O(n) for observers
from abc import ABC, abstractmethod
from typing import Callable, Any
from weakref import WeakSet

class Observable:
    """Base class for observable data with automatic notification"""
    def __init__(self):
        self._observers = WeakSet()
    
    def bind(self, callback: Callable[[Any], None]):
        """Register observer callback"""
        self._observers.add(callback)
    
    def notify(self, value: Any):
        """Notify all observers of change"""
        for callback in self._observers:
            callback(value)

class ViewModel(ABC):
    """Abstract ViewModel with observable properties"""
    def __init__(self):
        self._properties = {}
    
    def get_property(self, name: str) -> Observable:
        """Get or create observable property"""
        if name not in self._properties:
            self._properties[name] = Observable()
        return self._properties[name]
    
    def set_value(self, name: str, value: Any):
        """Update property and notify observers"""
        if hasattr(self, f'_{name}'):
            setattr(self, f'_{name}', value)
            self.get_property(name).notify(value)

class UserModel:
    """Model: Data layer"""
    def __init__(self):
        self.users = []
    
    def add_user(self, name: str, email: str):
        user = {'name': name, 'email': email}
        self.users.append(user)
        return user

class UserViewModel(ViewModel):
    """ViewModel: Business logic and state management"""
    def __init__(self, model: UserModel):
        super().__init__()
        self.model = model
        self._name = ""
        self._email = ""
        self._message = ""
    
    @property
    def name(self):
        return self._name
    
    @name.setter
    def name(self, value):
        self.set_value('name', value)
    
    @property
    def email(self):
        return self._email
    
    @email.setter
    def email(self, value):
        self.set_value('email', value)
    
    def save_user(self):
        """Business logic for saving user"""
        if not self._name or not self._email:
            self.set_value('message', 'Error: Name and email required')
            return False
        
        self.model.add_user(self._name, self._email)
        self.set_value('message', f'Success: User {self._name} saved')
        self._name = ""
        self._email = ""
        return True

class View:
    """View: UI layer with data binding"""
    def __init__(self, viewmodel: UserViewModel):
        self.vm = viewmodel
        
        # Setup data bindings
        self.vm.get_property('name').bind(self.update_name_display)
        self.vm.get_property('email').bind(self.update_email_display)
        self.vm.get_property('message').bind(self.show_message)
    
    def update_name_display(self, name):
        print(f"[UI] Name field: {name}")
    
    def update_email_display(self, email):
        print(f"[UI] Email field: {email}")
    
    def show_message(self, message):
        print(f"[UI] Message: {message}")
    
    def user_input(self, name, email):
        """Simulate user input"""
        self.vm.name = name
        self.vm.email = email
        self.vm.save_user()

# Example usage
if __name__ == "__main__":
    model = UserModel()
    viewmodel = UserViewModel(model)
    view = View(viewmodel)
    
    view.user_input("Alice", "alice@example.com")

```

**Why**: Clean separation of concerns with automatic UI updates. Essential for modern UI frameworks (WPF, Angular, Vue).

## Repository Pattern

```python
# PROBLEM: Abstract data access layer for clean architecture
# TIME: O(1) for simple operations SPACE: O(n) for caching
from abc import ABC, abstractmethod
from typing import Generic, TypeVar, List, Optional, Dict
from dataclasses import dataclass
from copy import deepcopy

T = TypeVar('T')

@dataclass
class User:
    id: int
    name: str
    email: str
    active: bool = True

class Repository(ABC, Generic[T]):
    """Abstract repository interface"""
    
    @abstractmethod
    def find_by_id(self, id: int) -> Optional[T]:
        pass
    
    @abstractmethod
    def find_all(self) -> List[T]:
        pass
    
    @abstractmethod
    def save(self, entity: T) -> T:
        pass
    
    @abstractmethod
    def delete(self, id: int) -> bool:
        pass

class InMemoryUserRepository(Repository[User]):
    """In-memory implementation for testing/prototyping"""
    
    def __init__(self):
        self._storage: Dict[int, User] = {}
        self._next_id = 1
    
    def find_by_id(self, id: int) -> Optional[User]:
        user = self._storage.get(id)
        return deepcopy(user) if user else None
    
    def find_all(self) -> List[User]:
        return [deepcopy(u) for u in self._storage.values()]
    
    def find_by_email(self, email: str) -> Optional[User]:
        """Domain-specific query"""
        for user in self._storage.values():
            if user.email == email:
                return deepcopy(user)
        return None
    
    def find_active_users(self) -> List[User]:
        """Domain-specific query"""
        return [deepcopy(u) for u in self._storage.values() if u.active]
    
    def save(self, user: User) -> User:
        if user.id is None or user.id not in self._storage:
            # Create new
            user = User(
                id=self._next_id,
                name=user.name,
                email=user.email,
                active=user.active
            )
            self._next_id += 1
        
        self._storage[user.id] = deepcopy(user)
        return user
    
    def delete(self, id: int) -> bool:
        if id in self._storage:
            del self._storage[id]
            return True
        return False

class UserService:
    """Service layer using repository"""
    
    def __init__(self, repository: Repository[User]):
        self.repo = repository
    
    def create_user(self, name: str, email: str) -> User:
        # Business logic
        existing = self.repo.find_by_email(email)
        if existing:
            raise ValueError(f"User with email {email} already exists")
        
        user = User(id=None, name=name, email=email)
        return self.repo.save(user)
    
    def deactivate_user(self, user_id: int) -> bool:
        user = self.repo.find_by_id(user_id)
        if not user:
            return False
        
        user.active = False
        self.repo.save(user)
        return True

# Example usage
if __name__ == "__main__":
    repo = InMemoryUserRepository()
    service = UserService(repo)
    
    # Create users
    alice = service.create_user("Alice", "alice@example.com")
    bob = service.create_user("Bob", "bob@example.com")
    
    print(f"Created: {alice}")
    
    # Query
    active = repo.find_active_users()
    print(f"Active users: {len(active)}")
    
    # Deactivate
    service.deactivate_user(alice.id)
    active = repo.find_active_users()
    print(f"Active after deactivation: {len(active)}")
```

**Why**: Decouples business logic from data access. Enables easy testing with mock repositories and database switching.

## Gossip Protocol

```python
# PROBLEM: Disseminate information in distributed system efficiently
# TIME: O(log n) for propagation SPACE: O(n) for state
import random
import time
from typing import Dict, Set, List, Callable
from dataclasses import dataclass, field
from enum import Enum

class MessageType(Enum):
    GOSSIP = "gossip"
    SYNC = "sync"

@dataclass
class Message:
    id: str
    data: Dict
    timestamp: float
    sender: str
    ttl: int = 3

@dataclass
class NodeState:
    node_id: str
    data: Dict = field(default_factory=dict)
    known_messages: Set[str] = field(default_factory=set)
    peers: Set[str] = field(default_factory=set)
    message_handlers: Dict[str, Callable] = field(default_factory=dict)

class GossipNode:
    """
    Implements gossip protocol for information dissemination.
    Uses push-pull model with anti-entropy for consistency.
    """
    
    def __init__(self, node_id: str, fanout: int = 3):
        self.state = NodeState(node_id=node_id)
        self.fanout = fanout  # Number of peers to gossip to
        self.round_interval = 1.0  # Seconds between rounds
    
    def add_peer(self, peer_id: str):
        """Add a peer to the gossip network"""
        self.state.peers.add(peer_id)
    
    def register_handler(self, message_type: str, handler: Callable):
        """Register callback for message type"""
        self.state.message_handlers[message_type] = handler
    
    def create_message(self, data: Dict) -> Message:
        """Create a new message to broadcast"""
        msg_id = f"{self.state.node_id}:{time.time()}"
        return Message(
            id=msg_id,
            data=data,
            timestamp=time.time(),
            sender=self.state.node_id
        )
    
    def receive_gossip(self, message: Message) -> bool:
        """
        Receive and process gossip message.
        Returns True if message is new, False if already seen.
        """
        if message.id in self.state.known_messages:
            return False
        
        # Mark as seen
        self.state.known_messages.add(message.id)
        
        # Apply data
        self.state.data.update(message.data)
        
        # Call handler if registered
        msg_type = message.data.get('type')
        if msg_type and msg_type in self.state.message_handlers:
            self.state.message_handlers[msg_type](message)
        
        return True
    
    def select_random_peers(self) -> List[str]:
        """Select random subset of peers for gossip"""
        peers = list(self.state.peers)
        if len(peers) <= self.fanout:
            return peers
        return random.sample(peers, self.fanout)
    
    def gossip_round(self) -> Dict[str, Message]:
        """
        Execute one gossip round.
        Returns dict of peer_id -> message to send.
        """
        if not self.state.known_messages:
            return {}
        
        # Select random peers
        selected_peers = self.select_random_peers()
        
        # Prepare messages (in real impl, would include digest of known messages)
        outgoing = {}
        for peer in selected_peers:

class GossipNetwork:
    """Simulator for gossip protocol network"""
    
    def __init__(self, num_nodes: int = 10, fanout: int = 3):
        self.nodes: Dict[str, GossipNode] = {}
        
        # Create nodes
        for i in range(num_nodes):
            node_id = f"node_{i}"
            self.nodes[node_id] = GossipNode(node_id, fanout)
        
        # Connect nodes in a mesh
        node_ids = list(self.nodes.keys())
        for i, node_id in enumerate(node_ids):
            node = self.nodes[node_id]
            # Connect to some peers
            for j in range(fanout):
                peer_id = node_ids[(i + j + 1) % num_nodes]
                node.add_peer(peer_id)
    
    def broadcast(self, origin: str, data: Dict) -> int:
        """Broadcast message from origin node, returns number of nodes reached"""
        origin_node = self.nodes[origin]
        message = origin_node.create_message(data)
        origin_node.receive_gossip(message)
        
        reached = {origin}
        rounds = 0
        max_rounds = 20
        
        while len(reached) < len(self.nodes) and rounds < max_rounds:
            rounds += 1
            all_outgoing = {}
            
            # Collect all outgoing messages
            for node_id, node in self.nodes.items():
                if node_id in reached:
                    outgoing = node.gossip_round()
                    all_outgoing[node_id] = outgoing
            
            # Deliver messages
            for sender_id, outgoing in all_outgoing.items():
                for peer_id, msg in outgoing.items():
                    peer = self.nodes[peer_id]
                    if peer.receive_gossip(msg):
                        reached.add(peer_id)
        
        return len(reached)
```
