        /**
         * Proxy handler for object operation interception
         * PROBLEM: Intercept and control object operations
         * TIME: O(1) SPACE: O(1) for each trap
         * WHY: Enables logging, validation, virtual properties,

        // Logger function
        function log(message, isError = false) {
            const logger = document.getElementById('logger');
            const entry = document.createElement('div');

            entry.textContent = `[${new Date().toLocaleTimeString()}] ${message}`;

        const handler = {

            get(target, property) {

                // Return undefined for non-existent properties
                if (!(property in target)) {
                    log(`⚠️ Property "${property}" does not exist`, true);
                    return undefined;
                }
                
                return target[property];

            set(target, property, value) {

                // Validation example
                if (property === 'age' && (value < 0 || value > 150)) {
                    log(`❌ Validation failed: Invalid age`, true);
                    return false;
                }
                
                target[property] = value;
                log(`✅ Property set successfully`);
                return true;
            },
            
            // Intercept 'in' operator
            has(target, property) {
                log(`🔍 HAS: Checking if "${property}" exists`);
                return property in target;
            }
        };

    <script>
        /**
         * Generator function with asterisk (*) syntax
         * PROBLEM: Generate infinite sequences lazily
         * TIME: O(1) per iteration SPACE: O(1)
         * WHY: Generators enable memory-efficient iteration
         * over large or infinite sequences by computing values
         * on-demand rather than storing entire sequence.
         */
        
        let currentGenerator = null;
        let isFibonacci = true;
        
        /**
         * Generator for Fibonacci sequence
         */
        function* fibonacciGenerator() {
            let prev = 0, curr = 1;
            yield prev;
            yield curr;
            
            while (true) {
                [prev, curr] = [curr, prev + curr];
                yield curr;
            }
        }
        
        /**
         * Generator for prime numbers
         */
        function* primeGenerator() {
            yield 2;
            let num = 3;
            
            while (true) {
                if (isPrime(num)) {
                    yield num;
                }
                num += 2;
            }
        }
        
        // Check if number is prime
        function isPrime(n) {
            if (n < 2) return false;
            if (n === 2) return true;
            if (n % 2 === 0) return false;
            
            for (let i = 3; i <= Math.sqrt(n); i += 2) {
                if (n % i === 0) return false;
            }
            return true;
        }
        
        // Initialize generator
        function initGenerator() {
            currentGenerator = isFibonacci ? 
                fibonacciGenerator() : primeGenerator();
            document.getElementById('mode').textContent = 
                isFibonacci ? 'Fibonacci Sequence' : 'Prime Numbers';
            document.getElementById('display').textContent = '0';
        }
        
        // Get next value
        function nextValue() {
            const result = currentGenerator.next();
            document.getElementById('display').textContent = result.value;
        }
        
        // Reset generator
        function resetGenerator() {
            initGenerator();
        }
        
        // Toggle mode
        function toggleMode() {
            isFibonacci = !isFibonacci;
            initGenerator();
        }
        
        // Initialize
        initGenerator();
    </script>

**Why:** Generator asterisk (*) enables lazy evaluation for memory-efficient processing of infinite or large sequences.
