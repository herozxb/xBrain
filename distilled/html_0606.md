# JavaScript Proxy Apply

```javascript
// PROBLEM: Intercept function calls using Proxy.apply trap
// TIME: O(1) SPACE: O(1)

/**
 * What: The apply trap intercepts function calls on a Proxy.
 * Why: Enables function call interception, logging, validation, and memoization.
 */

// Basic example: Logging function calls
function greet(name) {
  return `Hello, ${name}!`;
}

const greetHandler = {
  apply(target, thisArg, argumentsList) {
    console.log(`Calling ${target.name} with args:`, argumentsList);
    const result = target.apply(thisArg, argumentsList);
    console.log(`Result:`, result);
    return result;
  }
};

const proxiedGreet = new Proxy(greet, greetHandler);
// proxiedGreet('Alice') logs: Calling greet with args: ['Alice'], Result: Hello, Alice!

// Practical example: Memoization
function fibonacci(n) {
  if (n <= 1) return n;
  return fibonacci(n - 1) + fibonacci(n - 2);
}

const memoize = (fn) => {
  const cache = new Map();
  return new Proxy(fn, {
    apply(target, thisArg, args) {
      const key = JSON.stringify(args);
      if (cache.has(key)) {
        console.log(`Cache hit for ${key}`);
        return cache.get(key);
      }
      const result = target.apply(thisArg, args);
      cache.set(key, result);
      return result;
    }
  });
};

const memoizedFib = memoize(fibonacci);
// First call computes, second call uses cache
```

# JavaScript Reflect.set

```javascript
// PROBLEM: Set object properties using Reflect API
// TIME: O(1) SPACE: O(1)

/**
 * What: Reflect.set is the functional equivalent of the assignment operator.
 * Why: Provides a function-based approach to property assignment, useful in Proxy traps.
 */

// Basic usage
const user = { name: 'Bob' };
Reflect.set(user, 'name', 'Alice'); // true
Reflect.set(user, 'age', 25); // true (creates new property)
console.log(user); // { name: 'Alice', age: 25 }

// With receiver (for inheritance)
const parent = { greeting: 'Hello' };
const child = Object.create(parent);

const handler = {
  set(target, property, value, receiver) {
    console.log(`Setting ${property} to ${value}`);
    // Forward the operation to the actual object
    return Reflect.set(target, property, value, receiver);
  }
};

const proxy = new Proxy(child, handler);
proxy.greeting = 'Hi'; // Logs: Setting greeting to Hi

// Validation example
function createValidatedObject(schema) {
  const target = {};
  return new Proxy(target, {
    set(obj, prop, value) {
      if (prop in schema) {
        const validator = schema[prop];
        if (!validator(value)) {
          throw new Error(`Invalid value for ${prop}`);
        }
      }
      return Reflect.set(obj, prop, value);
    }
  });
}

const validated = createValidatedObject({
  age: (val) => typeof val === 'number' && val >= 0
});
validated.age = 25; // OK
// validated.age = -5; // Throws Error
```

# JavaScript for-of Iterator

```javascript
// PROBLEM: Iterate over iterable objects using for-of loop
// TIME: O(n) SPACE: O(1)

/**
 * What: The for-of loop iterates over iterable objects (arrays, strings, maps, sets).
 * Why: Provides clean syntax for iterating values without managing indices or iterators manually.
 */

// Basic array iteration
const fruits = ['apple', 'banana', 'cherry'];
for (const fruit of fruits) {
  console.log(fruit);
}

// String iteration
for (const char of 'Hello') {
  console.log(char); // H, e, l, l, o
}

// Map iteration
const userMap = new Map([
  ['id', 1],
  ['name', 'Alice']
]);
for (const [key, value] of userMap) {
  console.log(`${key}: ${value}`);
}

// Set iteration
const uniqueNumbers = new Set([1, 2, 3, 2, 1]);
for (const num of uniqueNumbers) {
  console.log(num); // 1, 2, 3 (duplicates removed)
}

// Custom iterable object
const range = {
  start: 1,
  end: 5,
  [Symbol.iterator]() {
    let current = this.start;
    const end = this.end;
    return {
      next() {
        if (current <= end) {
          return { value: current++, done: false };
        }
        return { done: true };
      }
    };
  }
};

for (const num of range) {
  console.log(num); // 1, 2, 3, 4, 5
}

// Breaking and continuing
const numbers = [1, 2, 3, 4, 5];
for (const num of numbers) {
  if (num === 3) continue;
  if (num === 5) break;
  console.log(num); // 1, 2, 4
}
```

# JavaScript Prototype Methods

```javascript
// PROBLEM: Add methods to constructor prototypes for shared functionality
// TIME: O(1) SPACE: O(1)

/**
 * What: Prototype methods are functions shared across all instances of a constructor.
 * Why: Memory efficient - methods stored once in prototype, not duplicated per instance.
 */

// Constructor function
function Animal(name, species) {
  this.name = name;
  this.species = species;
}

// Add method to prototype
Animal.prototype.speak = function() {
  return `${this.name} makes a sound`;
};

Animal.prototype.describe = function() {
  return `${this.name} is a ${this.species}`;
};

// Instances share the same method reference
const dog = new Animal('Buddy', 'dog');
const cat = new Animal('Whiskers', 'cat');

console.log(dog.speak()); // Buddy makes a sound
console.log(cat.describe()); // Whiskers is a cat

// Verify shared reference
console.log(dog.speak === cat.speak); // true

// ES6 class syntax (same prototype behavior)
class Vehicle {
  constructor(make, model) {
    this.make = make;
    this.model = model;
  }
  
  // Prototype method
  start() {
    return `${this.make} ${this.model} is starting`;
  }
  
  // Static method (not on prototype)
  static isVehicle(obj) {
    return obj instanceof Vehicle;
  }
}

const car = new Vehicle('Toyota', 'Camry');
console.log(car.start()); // Toyota Camry is starting
console.log(Vehicle.isVehicle(car)); // true

// Extending prototypes (monkey patching - use cautiously)
Array.prototype.first = function() {
  return this[0];
};

const arr = [1, 2, 3];
console.log(arr.first()); // 1
```

# JavaScript Closure Currying

```javascript
// PROBLEM: Transform functions to take arguments one at a time using closures
// TIME: O(1) SPACE: O(1)

/**
 * What: Currying is a technique of translating a function with multiple arguments
 *       into a sequence of functions each with a single argument.
 * Why: Enables partial application, function composition, and more reusable code.
 */

// Manual currying
function add(a) {
  return function(b) {
    return function(c) {
      return a + b + c;
    };
  };
}

console.log(add(1)(2)(3)); // 6

// Partial application
const addFive = add(5);
const addFiveAndTen = addFive(10);
console.log(addFiveAndTen(3)); // 18

// Generic curry function
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    }
    return function(...moreArgs) {
      return curried.apply(this, args.concat(moreArgs));
    };
  };
}

// Usage
function multiply(a, b, c) {
  return a * b * c;
}

const curriedMultiply = curry(multiply);
console.log(curriedMultiply(2)(3)(4)); // 24
console.log(curriedMultiply(2, 3)(4)); // 24
console.log(curriedMultiply(2)(3, 4)); // 24

// Real-world example: Event handler factory
function createEventHandler(eventType, selector) {
  return function(callback) {
    document.querySelector(selector).addEventListener(eventType, callback);
  };
}

const handleClick = createEventHandler('click', '#button');
handleClick(() => console.log('Button clicked!'));

// Practical: API request builder
function apiRequest(baseUrl) {
  return function(endpoint) {
    return function(options) {
      return fetch(`${baseUrl}${endpoint}`, options);
    };
  };
}

const jsonPlaceholder = apiRequest('https://jsonplaceholder.typicode.com');
const usersEndpoint = jsonPlaceholder('/users');
// usersEndpoint({ method: 'GET' }).then(r => r.json());
```

# Web Components Define

```javascript
// PROBLEM: Create custom HTML elements using customElements.define
// TIME: O(1) SPACE: O(1)

/**
 * What: customElements.define() registers a new custom HTML element.
 * Why: Create reusable, encapsulated components with native browser support.
 */

// Basic custom element
class MyGreeting extends HTMLElement {
  constructor() {
    super();
    this.name = this.getAttribute('name') || 'World';
  }
  
  connectedCallback() {
    this.innerHTML = `<h2>Hello, ${this.name}!</h2>`;
  }
}

// Register the custom element
customElements.define('my-greeting', MyGreeting);

// Usage in HTML: <my-greeting name="Alice"></my-greeting>

// Custom element with shadow DOM
class UserCard extends HTMLElement {
  constructor() {
    super();
    this.attachShadow({ mode: 'open' });
  }
  
  connectedCallback() {
    const name = this.getAttribute('name') || 'Anonymous';
    const email = this.getAttribute('email') || '';
    
    this.shadowRoot.innerHTML = `
      <style>
        :host {
          display: block;
          border: 1px solid #ddd;
          border-radius: 8px;
          padding: 16px;
          margin: 8px 0;
          background: white;
        }
        h3 {
          margin: 0 0 8px 0;
          color: #333;
        }
        p {
          margin: 0;
          color: #666;
          font-size: 14px;
        }
      </style>
      <h3>${name}</h3>
      <p>${email}</p>
    `;
  }
}

customElements.define('user-card', UserCard);

// Custom element with observed attributes
class CounterButton extends HTMLElement {
  static get observedAttributes() {
    return ['count'];
  }
  
  constructor() {
    super();
    this.attachShadow({ mode: 'open' });
    this.count = parseInt(this.getAttribute('count')) || 0;
  }
  
  connectedCallback() {
    this.render();
    this.shadowRoot.querySelector('button').addEventListener('click', () => {
      this.count++;
      this.setAttribute('count', this.count);
    });
  }
  
  attributeChangedCallback(name, oldValue, newValue) {
    if (name === 'count' && oldValue !== newValue) {
      this.count = parseInt(newValue);
      this.render();
    }
  }
  
  render() {
    this.shadowRoot.innerHTML = `
      <style>
        button {
          padding: 10px 20px;
          font-size: 16px;
          background: #4CAF50;
          color: white;
          border: none;
          border-radius: 4px;
          cursor: pointer;
        }
        button:hover {
          background: #45a049;
        }
      </style>
      <button>Count: ${this.count}</button>
    `;
  }
}

customElements.define('counter-button', CounterButton);

// HTML usage:
/*
<my-greeting name="Alice"></my-greeting>
<user-card name="John Doe" email="john@example.com"></user-card>
<counter-button count="0"></counter-button>
*/
```
