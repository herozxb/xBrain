## Palindrome Tree (Eertree)

```python
# PROBLEM: Build a data structure that efficiently stores all distinct palindromic substrings of a string
# TIME: O(n) for construction SPACE: O(n * alphabet)
class PalindromeTree:
    """Eertree (Palindromic Tree) for storing all distinct palindromes."""
    
    class Node:
        __slots__ = ['next', 'link', 'len', 'count', 'num']
        
        def __init__(self, length):
            self.next = {}  # edges to longer palindromes
            self.link = 0   # suffix link to longest proper palindromic suffix
            self.len = length  # length of palindrome
            self.count = 0  # occurrence count
            self.num = 0    # number of distinct palindromic suffixes
    
    def __init__(self, s):
        self.s = s
        self.nodes = [self.Node(-1), self.Node(0)]  # root nodes: -1 and 0 length
        self.nodes[0].link = 0  # imaginary node links to itself
        self.nodes[1].link = 0  # empty string links to imaginary
        self.last = 1  # node representing longest suffix palindrome
        self.suff = [1]  # longest palindromic suffix at each position
        self._build()
    
    def _get_link(self, node, pos):
        """Find the longest palindromic suffix that can be extended."""
        while pos - 1 - self.nodes[node].len < 0 or \
              self.s[pos - 1 - self.nodes[node].len] != self.s[pos]:
            node = self.nodes[node].link
        return node
    
    def _build(self):
        """Build the eertree by processing each character."""
        for i, ch in enumerate(self.s):
            # Find longest palindromic suffix that can be extended
            cur = self._get_link(self.last, i)
            
            # Check if this palindrome already exists
            if ch not in self.nodes[cur].next:
                # Create new node
                new_len = self.nodes[cur].len + 2
                new_node = self.Node(new_len)
                self.nodes.append(new_node)
                
                # Set suffix link for new node
                if new_len == 1:
                    new_node.link = 1  # single char links to empty string
                else:
                    link_node = self._get_link(self.nodes[cur].link, i)
                    new_node.link = self.nodes[link_node].next.get(ch, 1)
                
                # Add edge from current node
                self.nodes[cur].next[ch] = len(self.nodes) - 1
            
            # Update last and counts
            self.last = self.nodes[cur].next[ch]
            self.nodes[self.last].count += 1
            self.suff.append(self.last)
    
    def count_palindromes(self):
        """Return total number of distinct palindromes."""
        return len(self.nodes) - 2
    
    def get_all_palindromes(self):
        """Return all distinct palindromes stored in the tree."""
        result = []
        
        def dfs(node_idx, current):
            if node_idx < 2:
                return
            node = self.nodes[node_idx]
            if node.len > 0:
                # Reconstruct palindrome from its length
                # For simplicity, store (length, count)
                result.append((node.len, node.count))
            for ch, next_idx in node.next.items():
                dfs(next_idx, current)
        
        dfs(1, "")  # Start from empty string node
        return result

# Example usage
def main():
    s = "abacaba"
    tree = PalindromeTree(s)
    print(f"Distinct palindromes: {tree.count_palindromes()}")
    # Palindromes: a, b, c, aba, aca, bacab, abacaba

if __name__ == "__main__":
    main()
```
**Why**: Eertree provides O(1) amortized insertion of characters and stores all distinct palindromes efficiently. Used in competitive programming for palindrome-related queries on strings.

## Graph Coloring (Greedy)

```python
# PROBLEM: Color graph vertices such that no adjacent vertices share the same color
# TIME: O(V + E) SPACE: O(V)
def greedy_graph_coloring(adj):
    """
    Greedy graph coloring using Welsh-Powell approach.
    Returns list of colors assigned to each vertex.
    
    Args:
        adj: adjacency list where adj[v] contains neighbors of vertex v
    
    Returns:
        colors: list where colors[v] is the color assigned to vertex v
    """
    n = len(adj)
    colors = [-1] * n  # -1 means uncolored
    
    # Sort vertices by degree (descending) for better results
    vertices = sorted(range(n), key=lambda v: len(adj[v]), reverse=True)
    
    for v in vertices:
        if colors[v] != -1:
            continue
        
        # Find colors used by neighbors
        neighbor_colors = set()
        for neighbor in adj[v]:
            if colors[neighbor] != -1:
                neighbor_colors.add(colors[neighbor])
        
        # Assign smallest available color
        color = 0
        while color in neighbor_colors:
            color += 1
        colors[v] = color
    
    return colors


def dsatur_coloring(adj):
    """
    DSatur algorithm for graph coloring.
    More sophisticated than basic greedy, often produces better results.
    Uses saturation degree (number of different colors in neighbors).
    
    Args:
        adj: adjacency list
    
    Returns:
        colors: list of colors, and chromatic number estimate
    """
    import heapq
    
    n = len(adj)
    colors = [-1] * n
    saturation = [0] * n  # number of different colors in neighborhood
    degree = [len(adj[v]) for v in range(n)]
    colored_count = 0
    
    # Priority queue: (-saturation, -degree, vertex)
    # Max heap based on saturation, then degree
    pq = [(-(degree[v] if colors[v] == -1 else -1), -degree[v], v) 
          for v in range(n)]
    heapq.heapify(pq)
    
    while colored_count < n:
        # Get uncolored vertex with highest saturation
        while pq:
            _, _, v = heapq.heappop(pq)
            if colors[v] == -1:
                break
        else:
            break
        
        # Find colors used by neighbors
        neighbor_colors = set()
        for neighbor in adj[v]:
            if colors[neighbor] != -1:
                neighbor_colors.add(colors[neighbor])
        
        # Assign smallest available color
        color = 0
        while color in neighbor_colors:
            color += 1
        colors[v] = color
        colored_count += 1
        
        # Update saturation for uncolored neighbors
        for neighbor in adj[v]:
            if colors[neighbor] == -1:
                # Recalculate saturation
                neighbor_neighbor_colors = set()
                for nn in adj[neighbor]:
                    if colors[nn] != -1:
                        neighbor_neighbor_colors.add(colors[nn])
                saturation[neighbor] = len(neighbor_neighbor_colors)
                heapq.heappush(pq, (-saturation[neighbor], -degree[neighbor], neighbor))
    
    return colors, max(colors) + 1 if colors else 0


# Example usage
def main():
    # Example graph: triangle + one extra edge
    #    0 --- 1
    #    |   / |
    #    | /   |
    #    2 --- 3
    adj = [
        [1, 2],      # 0's neighbors
        [0, 2, 3],   # 1's neighbors
        [0, 1, 3],   # 2's neighbors
        [1, 2]       # 3's neighbors
    ]
    
    colors = greedy_graph_coloring(adj)
    print(f"Greedy colors: {colors}")
    print(f"Chromatic number (greedy): {max(colors) + 1}")
    
    colors, chromatic = dsatur_coloring(adj)
    print(f"DSatur colors: {colors}")
    print(f"Chromatic number (DSatur): {chromatic}")

if __name__ == "__main__":
    main()
```
**Why**: Graph coloring is NP-hard, but greedy approaches provide fast approximations. DSatur often achieves optimal or near-optimal results in practice. Used in register allocation, scheduling, and map coloring.

## Push-Relabel Maximum Flow

```python
# PROBLEM: Find maximum flow from source to sink in a flow network
# TIME: O(V^2 * E) worst case, O(V^3) simple implementation SPACE: O(V^2)
class PushRelabel:
    """
    Push-Relabel algorithm for maximum flow.
    More efficient than Edmonds-Karp for dense graphs.
    Uses height labels and excess flow concepts.
    """
    
    def __init__(self, n):
        self.n = n
        self.cap = [[0] * n for _ in range(n)]  # capacity matrix
        self.flow = [[0] * n for _ in range(n)]  # flow matrix
        self.excess = [0] * n  # excess flow at each vertex
        self.height = [0] * n  # height label for each vertex
        self.seen = [0] * n  # for gap relabeling optimization
    
    def add_edge(self, u, v, c):
        """Add directed edge from u to v with capacity c."""
        self.cap[u][v] += c
    
    def push(self, u, v):
        """Push flow from u to v."""
        # Calculate flow to push (minimum of excess and residual capacity)
        send = min(self.excess[u], 
                   self.cap[u][v] - self.flow[u][v])
        self.flow[u][v] += send
        self.flow[v][u] -= send
        self.excess[u] -= send
        self.excess[v] += send
        return send > 0
    
    def relabel(self, u):
        """Increase height of vertex u."""
        min_height = float('inf')
        for v in range(self.n):
            if self.cap[u][v] - self.flow[u][v] > 0:
                min_height = min(min_height, self.height[v])
        if min_height < float('inf'):
            self.height[u] = min_height + 1
    
    def discharge(self, u):
        """Discharge excess flow from vertex u."""
        while self.excess[u] > 0:
            if self.seen[u] < self.n:
                v = self.seen[u]
                if self.cap[u][v] - self.flow[u][v] > 0 and \
                   self.height[u] > self.height[v]:
                    self.push(u, v)
                else:
                    self.seen[u] += 1
            else:
                self.relabel(u)
                self.seen[u] = 0
    
    def max_flow(self, s, t):
        """
        Compute maximum flow from source s to sink t.
        Returns the maximum flow value.
        """
        # Initialize
        self.height[s] = self.n
        self.excess[s] = float('inf')
        
        # Push initial flow from source
        for v in range(self.n):
            if v != s:
                self.push(s, v)
        
        # Create list of vertices (excluding source and sink)
        vertices = [v for v in range(self.n) if v != s and v != t]
        
        # Process vertices with excess flow
        i = 0
        while i < len(vertices):
            u = vertices[i]
            old_height = self.height[u]
            self.discharge(u)
            
            # If height changed, move to front (relabel-to-front heuristic)
            if self.height[u] > old_height:
                vertices.insert(0, vertices.pop(i))
                i = 0
            else:
                i += 1
        
        return self.excess[t]


def main():
    # Example: simple flow network
    #     1
    #   / | \
    # 0   |   3
    #   \ | /
    #     2
    pr = PushRelabel(4)
    pr.add_edge(0, 1, 3)
    pr.add_edge(0, 2, 2)
    pr.add_edge(1, 2, 2)
    pr.add_edge(1, 3, 2)
    pr.add_edge(2, 3, 3)
    
    max_flow = pr.max_flow(0, 3)
    print(f"Maximum flow: {max_flow}")  # Should be 5

if __name__ == "__main__":
    main()
```
**Why**: Push-Relabel is more efficient than Ford-Fulkerson for dense graphs and forms the basis for the fastest practical max-flow algorithms. Used in image segmentation, bipartite matching, and network optimization.

## Optimal Binary Search Tree

```python
# PROBLEM: Build BST that minimizes expected search cost given key probabilities
# TIME: O(n^2) SPACE: O(n^2)
def optimal_bst(keys, freq):
    """
    Build optimal BST using dynamic programming.
    
    Args:
        keys: sorted list of keys
        freq: frequency of searching for each key
    
    Returns:
        Minimum expected search cost and the optimal tree structure
    """
    n = len(keys)
    if n == 0:
        return 0, None
    
    # dp[i][j] = minimum cost for subtree containing keys[i..j]
    # sum_freq[i][j] = sum of frequencies from i to j
    dp = [[0] * n for _ in range(n)]
    sum_freq = [[0] * n for _ in range(n)]
    root = [[0] * n for _ in range(n)]  # for tree reconstruction
    
    # Base case: single nodes
    for i in range(n):
        dp[i][i] = freq[i]
        sum_freq[i][i] = freq[i]
        root[i][i] = i
    
    # Fill for chains of length 2 to n
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            dp[i][j] = float('inf')
            sum_freq[i][j] = sum_freq[i][j-1] + freq[j]
            
            # Try each key as root
            for r in range(i, j + 1):
                # Cost = left subtree + right subtree + sum of all frequencies
                left_cost = dp[i][r-1] if r > i else 0
                right_cost = dp[r+1][j] if r < j else 0
                cost = left_cost + right_cost + sum_freq[i][j]
                
                if cost < dp[i][j]:
                    dp[i][j] = cost
                    root[i][j] = r
    
    return dp[0][n-1], root


def reconstruct_tree(keys, root, i, j):
    """Reconstruct the optimal BST from root table."""
    if i > j:
        return None
    
    r = root[i][j]
    return {
        'key': keys[r],
        'left': reconstruct_tree(keys, root, i, r-1),
        'right': reconstruct_tree(keys, root, r+1, j)
    }


def print_tree(node, indent=0):
    """Print tree structure."""
    if node is None:
        return
    print('  ' * indent + str(node['key']))
    if node['left']:
        print_tree(node['left'], indent + 1)
    if node['right']:
        print_tree(node['right'], indent + 1)


# Knuth optimization variant - O(n^2) with better constant
def optimal_bst_knuth(keys, freq):
    """
    Optimal BST with Knuth optimization.
    When root[i][j-1] <= root[i][j] <= root[i+1][j],
    we can reduce the search space for r.
    """
    n = len(keys)
    if n == 0:
        return 0, None
    
    dp = [[0] * n for _ in range(n)]
    sum_freq = [[0] * n for _ in range(n)]
    root = [[0] * n for _ in range(n)]
    
    for i in range(n):
        dp[i][i] = freq[i]
        sum_freq[i][i] = freq[i]
        root[i][i] = i
    
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            dp[i][j] = float('inf')
            sum_freq[i][j] = sum_freq[i][j-1] + freq[j]
            
            # Knuth optimization: only try roots in valid range
            left_root = root[i][j-1] if j > i else i
            right_root = root[i+1][j] if i < n-1 and i+1 <= j else j
            
            for r in range(left_root, right_root + 1):
                left_cost = dp[i][r-1] if r > i else 0
                right_cost = dp[r+1][j] if r < j else 0
                cost = left_cost + right_cost + sum_freq[i][j]
                
                if cost < dp[i][j]:
                    dp[i][j] = cost
                    root[i][j] = r
    
    return dp[0][n-1], root


def main():
    keys = [10, 20, 30, 40, 50]
    freq = [4, 2, 6, 3, 1]  # search frequencies
    
    min_cost, root_table = optimal_bst(keys, freq)
    print(f"Minimum expected search cost: {min_cost}")
    
    print("\nOptimal BST structure:")
    tree = reconstruct_tree(keys, root_table, 0, len(keys)-1)
    print_tree(tree)

if __name__ == "__main__":
    main()
```
**Why**: Optimal BST minimizes expected search time when search frequencies are known. Critical for static dictionaries, symbol tables in compilers, and any scenario where access patterns are predictable.

## Interval Tree (Augmented)

```python
# PROBLEM: Store intervals and efficiently query for overlapping intervals
# TIME: O(log n) query, O(n log n) construction SPACE: O(n)
class IntervalTree:
    """
    Augmented interval tree for efficient interval overlap queries.
    Each node stores an interval and the maximum endpoint in its subtree.
    """
    
    class Node:
        __slots__ = ['interval', 'max_end', 'left', 'right']
        
        def __init__(self, interval):
            self.interval = interval  # (start, end, data)
            self.max_end = interval[1]
            self.left = None
            self.right = None
    
    def __init__(self, intervals=None):
        self.root = None
        if intervals:
            self.root = self._build(sorted(intervals))
    
    def _build(self, intervals):
        """Build balanced tree from sorted intervals."""
        if not intervals:
            return None
        
        mid = len(intervals) // 2
        node = self.Node(intervals[mid])
        
        node.left = self._build(intervals[:mid])
        node.right = self._build(intervals[mid+1:])
        
        # Update max_end
        node.max_end = intervals[mid][1]
        if node.left:
            node.max_end = max(node.max_end, node.left.max_end)
        if node.right:
            node.max_end = max(node.max_end, node.right.max_end)
        
        return node
    
    def insert(self, interval):
        """Insert a single interval into the tree."""
        self.root = self._insert(self.root, interval)
    
    def _insert(self, node, interval):
        if node is None:
            return self.Node(interval)
        
        if interval[0] < node.interval[0]:
            node.left = self._insert(node.left, interval)
        else:
            node.right = self._insert(node.right, interval)
        
        # Update max_end
        node.max_end = max(node.max_end, interval[1])
        return node
    
    def query_point(self, point):
        """Find all intervals containing a point."""
        results = []
        self._query_point(self.root, point, results)
        return results
    
    def _query_point(self, node, point, results):
        if node is None:
            return
        
        start, end, data = node.interval
        
        # Check if point is in current interval
        if start <= point <= end:
            results.append(node.interval)
        
        # Search left if point could overlap
        if node.left and node.left.max_end >= point:
            self._query_point(node.left, point, results)
        
        # Search right if point is >= start of current interval
        if node.right and point >= node.interval[0]:
            self._query_point(node.right, point, results)
    
    def query_interval(self, query_start, query_end):
        """Find all intervals overlapping with [query_start, query_end]."""
        results = []
        self._query_interval(self.root, query_start, query_end, results)
        return results
    
    def _query_interval(self, node, q_start, q_end, results):
        if node is None:
            return
        
        start, end, data = node.interval
        
        # Check for overlap
        if start <= q_end and end >= q_start:
            results.append(node.interval)
        
        # Search left if left subtree could have overlaps
        if node.left and node.left.max_end >= q_start:
            self._query_interval(node.left, q_start, q_end, results)
        
        # Search right if right subtree could have overlaps
        if node.right and node.interval[0] <= q_end:
            self._query_interval(node.right, q_start, q_end, results)


def main():
    # Example intervals: (start, end, label)
    intervals = [
        (15, 20, 'A'),
        (10, 30, 'B'),
        (17, 19, 'C'),
        (5, 20, 'D'),
        (12, 15, 'E'),
        (30, 40, 'F')
    ]
    
    tree = IntervalTree(intervals)
    
    # Query for point
    print("Intervals containing point 14:")
    for interval in tree.query_point(14):
        print(f"  {interval}")
    
    # Query for overlapping interval
    print("\nIntervals overlapping [14, 16]:")
    for interval in tree.query_interval(14, 16):
        print(f"  {interval}")

if __name__ == "__main__":
    main()
```
**Why**: Interval trees enable O(log n + k) queries for overlapping intervals, essential in computational geometry, scheduling systems, and genomic analysis where finding intersecting ranges is frequent.

## Sparse Matrix (CSC Format)

```python
# PROBLEM: Efficiently store and operate on sparse matrices
# TIME: O(nnz) for most operations SPACE: O(nnz)
class SparseMatrixCSC:
    """
    Compressed Sparse Column (CSC) format for sparse matrices.
    Efficient for column-wise operations and matrix-vector multiplication.
    
    Storage:
    - data: non-zero values (column by column)
    - row_indices: row index for each non-zero
    - col_ptr: index in data where each column starts
    """
    
    def __init__(self, shape, data=None, row_indices=None, col_ptr=None):
        self.shape = shape  # (rows, cols)
        self.data = data or []
        self.row_indices = row_indices or []
        self.col_ptr = col_ptr or [0] * (shape[1] + 1)
    
    @classmethod
    def from_dense(cls, matrix):
        """Create CSC matrix from dense 2D array."""
        if not matrix or not matrix[0]:
            return cls((0, 0))
        
        rows, cols = len(matrix), len(matrix[0])
        data = []
        row_indices = []
        col_ptr = [0]
        
        for j in range(cols):
            for i in range(rows):
                if matrix[i][j] != 0:
                    data.append(matrix[i][j])
                    row_indices.append(i)
            col_ptr.append(len(data))
        
        return cls((rows, cols), data, row_indices, col_ptr)
    
    @classmethod
    def from_coo(cls, shape, rows, cols, values):
        """Create CSC from COO (coordinate) format."""
        # Sort by column, then row
        entries = sorted(zip(cols, rows, values))
        
        data = []
        row_indices = []
        col_ptr = [0]
        current_col = 0
        
        for col, row, val in entries:
            while current_col < col:
                col_ptr.append(len(data))
                current_col += 1
            data.append(val)
            row_indices.append(row)
        
        while current_col < shape[1]:
            col_ptr.append(len(data))
            current_col += 1
        
        return cls(shape, data, row_indices, col_ptr)
    
    def get(self, i, j):
        """Get element at position (i, j)."""
        if i < 0 or i >= self.shape[0] or j < 0 or j >= self.shape[1]:
            raise IndexError("Index out of bounds")
        
        # Search in column j
        start = self.col_ptr[j]
        end = self.col_ptr[j + 1]
        
        for idx in range(start, end):
            if self.row_indices[idx] == i:
                return self.data[idx]
        
        return 0
    
    def set(self, i, j, value):
        """Set element at position (i, j). Note: O(nnz) operation."""
        if i < 0 or i >= self.shape[0] or j < 0 or j >= self.shape[1]:
            raise IndexError("Index out of bounds")
        
        start = self.col_ptr[j]
        end = self.col_ptr[j + 1]
        
        # Find position
        insert_pos = start
        for idx in range(start, end):
            if self.row_indices[idx] == i:
                if value == 0:
                    # Remove element
                    del self.data[idx]
                    del self.row_indices[idx]
                    for c in range(j + 1, len(self.col_ptr)):
                        self.col_ptr[c] -= 1
                else:
                    self.data[idx] = value
                return
            if self.row_indices[idx] < i:
                insert_pos = idx + 1
        
        if value != 0:
            self.data.insert(insert_pos, value)
            self.row_indices.insert(insert_pos, i)
            for c in range(j + 1, len(self.col_ptr)):
                self.col_ptr[c] += 1
    
    def matvec(self, vec):
        """Matrix-vector multiplication."""
        if len(vec) != self.shape[1]:
            raise ValueError("Vector dimension mismatch")
        
        result = [0] * self.shape[0]
        for j in range(self.shape[1]):
            for idx in range(self.col_ptr[j], self.col_ptr[j + 1]):
                result[self.row_indices[idx]] += self.data[idx] * vec[j]
        
        return result
    
    def transpose(self):
        """Return transpose of matrix (convert to CSR-like)."""
        # Transpose of CSC is equivalent to CSR
        new_data = []
        new_col_indices = []
        row_ptr = [0]
        
        # Count non-zeros per row
        row_counts = [0] * self.shape[0]
        for row in self.row_indices:
            row_counts[row] += 1
        
        # Build row_ptr
        for count in row_counts:
            row_ptr.append(row_ptr[-1] + count)
        
        # For CSC transpose, we essentially get CSR
        return SparseMatrixCSC(
            (self.shape[1], self.shape[0]),
            self.data[:],
            list(range(self.shape[1])) * len(self.data),  # simplified
            row_ptr
        )
    
    def nnz(self):
        """Return number of non-zero elements."""
        return len(self.data)
    
    def to_dense(self):
        """Convert to dense matrix (for debugging)."""
        result = [[0] * self.shape[1] for _ in range(self.shape[0])]
        for j in range(self.shape[1]):
            for idx in range(self.col_ptr[j], self.col_ptr[j + 1]):
                result[self.row_indices[idx]][j] = self.data[idx]
        return result
    
    def __repr__(self):
        return f"SparseMatrixCSC(shape={self.shape}, nnz={self.nnz()})"
```

## LSM Tree (Log-Structured Merge Tree)

```python
# PROBLEM: Efficiently handle high write throughput with range query support
# TIME: O(1) write amortized, O(log n) read SPACE: O(n)
import bisect
import json
from collections import defaultdict
import os

class SSTable:
    """
    Sorted String Table - immutable sorted key-value file on disk.
    In practice, this would be a real file; here we simulate with memory.
    """
    
    def __init__(self, entries):
        """Create SSTable from sorted entries [(key, value), ...]."""
        self.entries = sorted(entries)
        self.keys = [k for k, v in self.entries]
        self.min_key = self.keys[0] if self.keys else None
        self.max_key = self.keys[-1] if self.keys else None
    
    def get(self, key):
        """Binary search for key."""
        idx = bisect.bisect_left(self.keys, key)
        if idx < len(self.keys) and self.keys[idx] == key:
            return self.entries[idx][1]
        return None
    
    def range_query(self, start, end):
        """Return all entries with start <= key <= end."""
        left = bisect.bisect_left(self.keys, start)
        right = bisect.bisect_right(self.keys, end)
        return self.entries[left:right]


class MemTable:
    """In-memory sorted structure for recent writes."""
    
    def __init__(self, max_size=100):
        self.data = {}
        self.max_size = max_size
        self.size = 0
    
    def put(self, key, value):
        self.data[key] = value
        self.size += 1
        return self.size >= self.max_size
    
    def get(self, key):
        return self.data.get(key)
    
    def is_full(self):
        return self.size >= self.max_size
    
    def to_sstable(self):
        """Convert to SSTable and clear."""
        entries = sorted(self.data.items())
        self.data = {}
        self.size = 0
        return SSTable(entries)


class LSMTree:
    """
    Log-Structured Merge Tree implementation.
    Optimized for write-heavy workloads with range query support.
    """
    
    def __init__(self, memtable_size=100, max_levels=5):
        self.memtable = MemTable(memtable_size)
        self.memtable_size = memtable_size
        self.levels = [[] for _ in range(max_levels)]  # levels[i] = list of SSTables
        self.max_levels = max_levels
    
    def put(self, key, value):
        """Insert key-value pair."""
        if self.memtable.put(key, value):
            self._flush_memtable()
    
    def get(self, key):
        """Retrieve value for key."""
        # Check memtable first
        value = self.memtable.get(key)
        if value is not None:
            return value
        
        # Check levels from newest to oldest
        for level in self.levels:
            for sstable in reversed(level):  # newer tables first
                if sstable.min_key <= key <= sstable.max_key:
                    value = sstable.get(key)
                    if value is not None:
                        return value
        
        return None
    
    def range_query(self, start, end):
        """Get all key-value pairs in range."""
        results = {}
        
        # Check memtable
        for k, v in self.memtable.data.items():
            if start <= k <= end:
                results[k] = v
        
        # Check all levels
        for level in self.levels:
            for sstable in level:
                if sstable.max_key >= start and sstable.min_key <= end:
                    for k, v in sstable.range_query(start, end):
                        if k not in results:  # don't overwrite newer values
                            results[k] = v
        
        return sorted(results.items())
    
    def _flush_memtable(self):
        """Flush memtable to level 0."""
        sstable = self.memtable.to_sstable()
        self.levels[0].append(sstable)
        self._maybe_compact(0)
    
    def _maybe_compact(self, level):
        """Check if level needs compaction."""
        # Simple size-tiered compaction
        threshold = 2 ** level  # level i can have up to 2^i tables
        
        if len(self.levels[level]) > threshold:
            self._compact_level(level)
    
    def _compact_level(self, level):
        """Merge all tables at level into next level."""
        if level >= self.max_levels - 1:
            return
        
        # Merge all tables at this level
        merged = defaultdict(dict)
        
        for sstable in self.levels[level]:
            for key, value in sstable.entries:
                merged[key] = value
        
        # Clear current level
        self.levels[level] = []
        
        # Create new table for next level
        entries = sorted(merged.items())
        if entries:
            new_sstable = SSTable(entries)
            self.levels[level + 1].append(new_sstable)
            self._maybe_compact(level + 1)
    
    def delete(self, key):
        """Delete key using tombstone."""
        self.put(key, None)  # None acts as tombstone


def main():
    lsm = LSMTree(memtable_size=5)
    
    # Insert some data
    print("Inserting data...")
    for i in range(20):
        lsm.put(f"key{i:02d}", f"value{i}")
    
    # Retrieve values
    print(f"\nGet key05: {lsm.get('key05')}")
    print(f"Get key15: {lsm.get('key15')}")
    print(f"Get nonexistent: {lsm.get('key99')}")
    
    # Range query
    print("\nRange query [key03, key07]:")
    for k, v in lsm.range_query("key03", "key07"):
        print(f"  {k}: {v}")
    
    # Delete
    lsm.delete("key05")
    print(f"\nAfter delete, key05: {lsm.get('key05')}")

if __name__ == "__main__":
    main()
```
**Why**: LSM trees power databases like Cassandra, RocksDB, and LevelDB. They excel at write-heavy workloads by converting random writes to sequential I/O, making them ideal for SSDs and log-structured storage systems.

## Bloom Filter Index

```python
# PROBLEM: Probabilistically test set membership with no false negatives
# TIME: O(k) where k is number of hash functions SPACE: O(m) bits
import hashlib

class BloomFilter:
    """
    Bloom filter for space-efficient set membership testing.
    Can have false positives but never false negatives.
    """
    
    def __init__(self, size, hash_count):
        """
        Args:
            size: number of bits in the filter
            hash_count: number of hash functions to use
        """
        self.size = size
        self.hash_count = hash_count
        self.bit_array = [False] * size
        self.count = 0
    
    @classmethod
    def optimal_params(cls, expected_items, false_positive_rate):
        """Calculate optimal size and hash count for desired characteristics."""
        import math
        m = -expected_items * math.log(false_positive_rate) / (math.log(2) ** 2)
        k = m / expected_items * math.log(2)
        return cls(int(m), int(k))
    
    def _hashes(self, item):
        """Generate k hash values for item using double hashing."""
        # Use MD5 and SHA1 as base hashes
        item_bytes = str(item).encode('utf-8')
        h1 = int(hashlib.md5(item_bytes).hexdigest(), 16)
        h2 = int(hashlib.sha1(item_bytes).hexdigest(), 16)
        
        for i in range(self.hash_count):
            # Double hashing: h(i) = h1 + i * h2
            yield (h1 + i * h2) % self.size
    
    def add(self, item):
        """Add item to the filter."""
        for pos in self._hashes(item):
            self.bit_array[pos] = True
        self.count += 1
    
    def contains(self, item):
        """Test if item might be in the filter."""
        return all(self.bit_array[pos] for pos in self._hashes(item))
    
    def __contains__(self, item):
        return self.contains(item)
    
    def false_positive_rate(self):
        """Calculate theoretical false positive rate."""
        import math
        if self.count == 0:
            return 0
        return (1 - math.exp(-self.hash_count * self.count / self.size)) ** self.hash_count


class BloomFilterIndex:
    """
    Bloom filter index for database queries.
    Each block/page has an associated bloom filter for quick elimination.
    """
    
    def __init__(self, block_size=1000, fp_rate=0.01):
        self.blocks = []  # List of (bloom_filter, data_block)
        self.block_size = block_size
        self.fp_rate = fp_rate
    
    def insert_batch(self, items):
        """Insert a batch of items as a new block."""
        bf = BloomFilter.optimal_params(self.block_size, self.fp_rate)
        block = []
        
        for item in items:
            bf.add(item)
            block.append(item)
        
        self.blocks.append((bf, block))
    
    def query(self, item):
        """Find all blocks that might contain item."""
        candidate_blocks = []
        
        for i, (bf, block) in enumerate(self.blocks):
            if item in bf:
                candidate_blocks.append((i, block))
        
        return candidate_blocks
    
    def lookup(self, item):
        """Return True if item exists, checking candidate blocks."""
        for bf, block in self.blocks:
            if item in bf:
                if item in block:  # Confirm with actual lookup
                    return True
        return False


class CountingBloomFilter(BloomFilter):
    """
    Counting Bloom filter that supports deletion.
    Uses counters instead of bits.
    """
    
    def __init__(self, size, hash_count):
        super().__init__(size, hash_count)
        self.counters = [0] * size
    
    def add(self, item):
        for pos in self._hashes(item):
            self.counters[pos] += 1
            self.bit_array[pos] = self.counters[pos] > 0
        self.count += 1
    
    def remove(self, item):
        """Remove item (only if it was added)."""
        if not self.contains(item):
            return False
        
        for pos in self._hashes(item):
            if self.counters[pos] > 0:
                self.counters[pos] -= 1
                self.bit_array[pos] = self.counters[pos] > 0
        
        self.count -= 1
        return True


def main():
    # Basic bloom filter
    bf = BloomFilter.optimal_params(expected_items=1000, false_positive_rate=0.01)
    print(f"Bloom filter: size={bf.size}, hash_count={bf.hash_count}")
    
    # Add some items
    items = [f"user{i}" for i in range(100)]
    for item in items:
        bf.add(item)
    
    # Test membership
    print(f"\n'user50' in filter: {bf.contains('user50')}")
    print(f"'user999' in filter: {bf.contains('user999')}")
    print(f"Theoretical FP rate: {bf.false_positive_rate():.4f}")
    
    # Bloom filter index
    print("\n--- Bloom Filter Index ---")
    index = BloomFilterIndex()
    
    # Insert blocks
    index.insert_batch([f"key{i}" for i in range(100)])
    index.insert_batch([f"key{i}" for i in range(100, 200)])
    index.insert_blocks([f"key{i}" for i in range(200, 300)])
    
    print(f"Lookup 'key50': {index.lookup('key50')}")
    print(f"Lookup 'key999': {index.lookup('key999')}")

if __name__ == "__main__":
    main()
```
**Why**: Bloom filters provide space-efficient probabilistic membership testing with no false negatives. Widely used in databases (Cassandra, PostgreSQL), web caches, spell checkers, and network routers for quick elimination of expensive lookups.

## Steiner Tree Approximation

```python
# PROBLEM: Find minimum weight tree connecting a subset of vertices (terminals)
# TIME: O(k^2 * n^2) for approximation SPACE: O(n^2)
import heapq
from collections import defaultdict

class SteinerTree:
    """
    Steiner Tree approximation using the 2-approximation algorithm.
    The minimum Steiner tree problem is NP-hard; this gives a tree
    with weight at most 2 * OPT.
    """
    
    def __init__(self, n):
        self.n = n
        self.adj = defaultdict(list)  # adj[u] = [(v, weight), ...]
    
    def add_edge(self, u, v, weight):
        """Add undirected edge."""
        self.adj[u].append((v, weight))
        self.adj[v].append((u, weight))
    
    def dijkstra(self, source):
        """Find shortest paths from source to all vertices."""
        dist = [float('inf')] * self.n
        dist[source] = 0
        parent = [-1] * self.n
        visited = [False] * self.n
        
        pq = [(0, source)]
        
        while pq:
            d, u = heapq.heappop(pq)
            if visited[u]:
                continue
            visited[u] = True
            
            for v, w in self.adj[u]:
                if dist[v] > dist[u] + w:
                    dist[v] = dist[u] + w
                    parent[v] = u
                    heapq.heappush(pq, (dist[v], v))
        
        return dist, parent
    
    def approx_steiner_tree(self, terminals):
        """
        2-approximation for Steiner tree.
        1. Compute shortest paths between all terminal pairs
        2. Find MST on the terminal distance graph
        3. Convert MST edges to paths in original graph
        4. Return subtree of union of paths
        """
        if not terminals:
            return [], 0
        
        k = len(terminals)
        if k == 1:
            return [terminals[0]], 0
        
        # Step 1: Compute all-pairs shortest paths between terminals
        terminal_dists = {}
        terminal_parents = {}
        
        for t in terminals:
            dist, parent = self.dijkstra(t)
            terminal_dists[t] = dist
            terminal_parents[t] = parent
        
        # Step 2: Build complete graph on terminals with distances as edge weights
        # and find MST using Prim's
        terminal_to_idx = {t: i for i, t in enumerate(terminals)}
        mst_edges = []
        
        # Prim's MST on terminal graph
        in_mst = [False] * k
        mst_dist = [float('inf')] * k
        mst_parent = [-1] * k
        mst_dist[0] = 0
        
        for _ in range(k):
            # Find minimum distance vertex not in MST
            u = min(range(k), key=lambda i: mst_dist[i] if not in_mst[i] else float('inf'))
            in_mst[u] = True
            
            if mst_parent[u] >= 0:
                mst_edges.append((terminals[mst_parent[u]], terminals[u]))
            
            # Update distances
            for v in range(k):
                if not in_mst[v]:
                    t_u, t_v = terminals[u], terminals[v]
                    d = terminal_dists[t_u][t_v]
                    if d < mst_dist[v]:
                        mst_dist[v] = d
                        mst_parent[v] = u
        
        # Step 3: Convert MST edges to actual paths
        steiner_edges = set()
        
        for t1, t2 in mst_edges:
            # Reconstruct path from t1 to t2 using shortest path tree
            path = [t2]
            current = t2
            while current != t1:
                current = terminal_parents[t1][current]
                path.append(current)
            
            # Add edges from path
            for i in range(len(path) - 1):
                u, v = path[i], path[i + 1]
                if u < v:
                    steiner_edges.add((u, v))
                else:
                    steiner_edges.add((v, u))
        
        # Step 4: Find minimum Steiner tree (subtree of union)
        # For approximation, we return the union
        total_weight = 0
        for u, v in steiner_edges:
            for neighbor, w in self.adj[u]:
                if neighbor == v:
                    total_weight += w
                    break
        
        return list(steiner_edges), total_weight


def main():
    # Example graph:
    #     1 -- 2
    #    /|    |
    #   0 |    |
    #    \|    |
    #     3 -- 4
    # Terminals: 0, 2, 4
    
    st = SteinerTree(5)
    st.add_edge(0, 1, 1)
    st.add_edge(0, 3, 2)
    st.add_edge(1, 2, 1)
    st.add_edge(1, 3, 1)
    st.add_edge(2, 4, 2)
    st.add_edge(3, 4, 1)
    
    terminals = [0, 2, 4]
    edges, weight = st.approx_steiner_tree(terminals)
    
    print(f"Terminals: {terminals}")
    print(f"Steiner tree edges: {edges}")
    print(f"Total weight: {weight}")
    print(f"(Approximation guarantee: weight <= 2 * optimal)")

if __name__ == "__main__":
    main()
```
**Why**: Steiner tree problems arise in network design, VLSI routing, and phylogenetic tree construction. The approximation algorithm provides a practical solution to this NP-hard problem with provable quality guarantees.

## Vertex Cover Approximation

```python
# PROBLEM: Find minimum set of vertices that covers all edges
# TIME: O(E) for 2-approximation SPACE: O(V + E)
class VertexCover:
    """
    Vertex cover approximation algorithms.
    Vertex cover is NP-hard, but 2-approximation is simple and efficient.
    """
    
    def __init__(self, n, edges):
        """
        Args:
            n: number of vertices
            edges: list of (u, v) edges
        """
        self.n = n
        self.edges = edges
    
    def greedy_2approx(self):
        """
        Simple 2-approximation: repeatedly pick uncovered edges
        and add both endpoints to the cover.
        
        Guarantee: |cover| <= 2 * OPT
        """
        cover = set()
        edge_set = set(self.edges)
        
        # Create adjacency for quick edge removal
        adj = [[] for _ in range(self.n)]
        for u, v in self.edges:
            adj[u].append(v)
            adj[v].append(u)
        
        covered = [False] * self.n
        
        for u, v in self.edges:
            if not covered[u] and not covered[v]:
                cover.add(u)
                cover.add(v)
                covered[u] = True
                covered[v] = True
        
        return cover
    
    def greedy_maximal_matching(self):
        """
        Another 2-approximation using maximal matching.
        The endpoints of any maximal matching form a 2-approx vertex cover.
        """
        cover = set()
        matched = [False] * self.n
        
        for u, v in self.edges:
            if not matched[u] and not matched[v]:
                cover.add(u)
                cover.add(v)
                matched[u] = True
                matched[v] = True
        
        return cover
    
    def greedy_degree(self):
        """
        Greedy by degree: repeatedly pick vertex with highest degree.
        No constant approximation guarantee, but often good in practice.
        """
        from collections import defaultdict
        
        cover = set()
        covered_edges = set()
        
        # Build adjacency and degree count
        adj = defaultdict(set)
        degree = defaultdict(int)
        
        for u, v in self.edges:
            adj[u].add(v)
            adj[v].add(u)
            degree[u] += 1
            degree[v] += 1
        
        while len(covered_edges) < len(self.edges):
            # Find vertex with highest effective degree
            best_vertex = None
            best_degree = -1
            
            for v in degree:
                if v in cover:
                    continue
                # Count uncovered edges
                eff_degree = sum(1 for u in adj[v] 
                               if (min(v, u), max(v, u)) not in covered_edges)
                if eff_degree > best_degree:
                    best_degree = eff_degree
                    best_vertex = v
            
            if best_vertex is None or best_degree == 0:
                break
            
            cover.add(best_vertex)
            
            # Mark edges as covered
            for u in adj[best_vertex]:
                edge = (min(best_vertex, u), max(best_vertex, u))
                covered_edges.add(edge)
        
        return cover
    
    def lp_rounding(self):
        """
        LP-based approximation using rounding.
        Solves LP relaxation and rounds vertices with value >= 0.5.
        This is also a 2-approximation.
        
        Note: This is a simplified version; in practice, use an LP solver.
        """
        # LP formulation (simplified, illustrative):
        # min sum(x_v) for all v
        # s.t. x_u + x_v >= 1 for all edges (u, v)
        #      x_v in [0, 1] for all v
        
        # For small graphs, we can solve via enumeration
        # For larger graphs, this should use an LP solver
        
        # Simple rounding heuristic based on edge coverage
        cover = set()
        uncovered = set(self.edges)
        
        # Calculate "fractional" value for each vertex
        fractional = [0.0] * self.n
        edge_count = [0] * self.n
        
        for u, v in self.edges:
            edge_count[u] += 1
            edge_count[v] += 1
        
        # Use degree ratio as proxy for fractional value
        max_edges = max(edge_count) if edge_count else 1
        for v in range(self.n):
            fractional[v] = edge_count[v] / max_edges
        
        # Round vertices with value >= 0.5 first
        while uncovered:
            # Find vertex covering most uncovered edges with fractional >= 0.5
            best = None
            best_cover = 0
            
            for v in range(self.n):
                if v in cover:
                    continue
                cover_count = sum(1 for (u, w) in uncovered if u == v or w == v)
                if cover_count > best_cover:
                    best_cover = cover_count
                    best = v
            
            if best is None:
                break
            
            cover.add(best)
            uncovered = {(u, v) for (u, v) in uncovered 
                        if u != best and v != best}
        
        return cover


def main():
    # Example: triangle graph with an extra vertex
    #   1
    #  / \
    # 0---2---3
    n = 4
    edges = [(0, 1), (1, 2), (0, 2), (2, 3)]
    
    vc = VertexCover(n, edges)
    
    print("Vertex Cover Approximations:")
    print(f"2-approx (edge picking): {vc.greedy_2approx()}")
    print(f"2-approx (maximal matching): {vc.greedy_maximal_matching()}")
    print(f"Greedy by degree: {vc.greedy_degree()}")
    print(f"LP rounding heuristic: {vc.lp_rounding()}")
    
    # Verify cover
    cover = vc.greedy_2approx()
    print(f"\nVerifying cover {cover}:")
    for u, v in edges:
        covered = u in cover or v in cover
        print(f"  Edge ({u}, {v}): {'covered' if covered else 'NOT covered'}")

if __name__ == "__main__":
    main()
```
**Why**: Vertex cover is fundamental in combinatorial optimization with applications in network security, computational biology, and scheduling. The 2-approximation is simple, fast, and optimal unless P=NP (Unique Games Conjecture).
