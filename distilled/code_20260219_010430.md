## Sliding Window - Maximum Subarray
```python
# PROBLEM: Find contiguous subarray with maximum sum
# APPROACH: Kadane's algorithm with sliding window
# TIME: O(n) SPACE: O(1)

def max_subarray(nums):
    max_sum = curr_sum = nums[0]
    for num in nums[1:]:
        curr_sum = max(num, curr_sum + num)
        max_sum = max(max_sum, curr_sum)
    return max_sum
```

## Two Pointers - Container With Most Water
```python
# PROBLEM: Find max area between two lines
# APPROACH: Two pointers from ends
# TIME: O(n) SPACE: O(1)

def max_area(height):
    left, right = 0, len(height) - 1
    max_area = 0
    while left < right:
        area = min(height[left], height[right]) * (right - left)
        max_area = max(max_area, area)
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1
    return max_area
```

## Hash Map - Group Anagrams
```python
# PROBLEM: Group strings that are anagrams
# APPROACH: Sort and group by key
# TIME: O(n * k log k) SPACE: O(n)

def group_anagrams(strs):
    groups = {}
    for s in strs:
        key = ''.join(sorted(s))
        if key not in groups:
            groups[key] = []
        groups[key].append(s)
    return list(groups.values())
```

## BFS - Binary Tree Level Order
```python
# PROBLEM: Traverse tree level by level
# APPROACH: BFS with queue
# TIME: O(n) SPACE: O(n)

from collections import deque

def level_order(root):
    if not root:
        return []
    result = []
    queue = deque([root])
    while queue:
        level = []
        for _ in range(len(queue)):
            node = queue.popleft()
            level.append(node.val)
            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)
        result.append(level)
    return result
```

```python
        return 0

            if grid[i][j] == '1':
                self.dfs(grid, i, j)
                count += 1

def dfs(grid, i, j):
    if i < 0 or j < 0 or i >= len(grid) or j >= len(grid[0]) or grid[i][j] != '1':
        return
    grid[i][j] = '#'
    dfs(grid, i+1, j)
    dfs(grid, i-1, j)
    dfs(grid, i, j+1)
    dfs(grid, i, j-1)
```

## DP - Coin Change
```python
# PROBLEM: Find minimum coins for amount
# APPROACH: Dynamic programming
# TIME: O(amount * n) SPACE: O(amount)

def coin_change(coins, amount):
    dp = [float('inf')] * (amount + 1)
    dp[0] = 0
    for coin in coins:
        for i in range(coin, amount + 1):
            dp[i] = min(dp[i], dp[i - coin] + 1)
    return dp[amount] if dp[amount] != float('inf') else -1
```

## Trie Implementation
```python
# PROBLEM: Implement prefix tree
# APPROACH: Nested dictionaries
# TIME: O(m) per operation SPACE: O(alphabet_size * m)

class Trie:
    def __init__(self):
        self.children = {}
        self.is_end = False
    
    def insert(self, word):
        node = self
        for char in word:
            if char not in node.children:
                node.children[char] = Trie()
            node = node.children[char]
        node.is_end = True
    
    def search(self, word):
        node = self
        for char in word:
            if char not in node.children:
                return False
            node = node.children[char]
        return node.is_end
```

## LRU Cache
```python
# PROBLEM: Implement LRU cache
# APPROACH: OrderedDict for O(1) operations
# TIME: O(1) SPACE: O(capacity)

from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = OrderedDict()
    
    def get(self, key):
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)
        return self.cache[key]
    
    def put(self, key, value):
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)
```

## Binary Search Tree Validation
```python
# PROBLEM: Validate BST
# APPROACH: Check bounds recursively
# TIME: O(n) SPACE: O(h)

def is_valid_bst(root, low=float('-inf'), high=float('inf')):
    if not root:
        return True
    if root.val <= low or root.val >= high:
        return False
    return (is_valid_bst(root.left, low, root.val) and
            is_valid_bst(root.right, root.val, high))
```

## Backtracking - Subsets
```python
# PROBLEM: Generate all subsets
# APPROACH: Backtracking
# TIME: O(2^n) SPACE: O(n)

def subsets(nums):
    result = []
    
    def backtrack(start, path):
        result.append(path[:])
        for i in range(start, len(nums)):
            path.append(nums[i])
            backtrack(i + 1, path)
            path.pop()
    
    backtrack(0, [])
    return result
```
