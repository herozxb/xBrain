## js_node_previousSibling

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>previousSibling Demo</title>
    <style>
        .highlight { background-color: yellow; padding: 5px; }
    </style>
</head>
<body>
    <!-- PROBLEM: Navigate to previous sibling node including text nodes -->
    <!-- TIME: O(1) SPACE: O(1) -->
    
    <ul id="list">
        <li>First Item</li>
        <li id="target">Second Item (Target)</li>
        <li>Third Item</li>
    </ul>
    
    <button onclick="showPreviousSibling()">Show Previous Sibling</button>
    <div id="result"></div>

    <script>
        /**
         * Demonstrates previousSibling property
         * Returns the previous node in the same tree level
         * Includes ALL node types (text, comment, element)
         */
        function showPreviousSibling() {
            const target = document.getElementById('target');
            const previous = target.previousSibling;
            
            // Note: previousSibling includes text nodes (whitespace)
            // Use previousElementSibling for elements only
            
            const result = document.getElementById('result');
            if (previous) {
                result.innerHTML = `
                    <strong>Previous Node:</strong> ${previous.nodeName}<br>
                    <strong>Node Type:</strong> ${previous.nodeType}<br>
                    <strong>Content:</strong> "${previous.textContent?.trim() || 'N/A'}"
                `;
            } else {
                result.textContent = 'No previous sibling exists';
            }
        }
    </script>
</body>
</html>
```

**Why:** `previousSibling` traverses all node types including whitespace text nodes. Use `previousElementSibling` for element-only navigation.

```html
            width: 200px;

            transition: all 0.3s;
        }
    </style>
</head>
<body>
    <!-- PROBLEM: Remove CSS classes from an element dynamically -->
    <!-- TIME: O(n) where n = number of classes to remove SPACE: O(1) -->
    
    <div id="box" class="box active highlight">
        Styled Box
    </div>
    
    <button onclick="removeClass('active')">Remove Active</button>
    <button onclick="removeClass('highlight')">Remove Highlight</button>
    <button onclick="removeMultiple()">Remove Both</button>
    <button onclick="resetClasses()">Reset</button>

    <script>
        /**
         * Removes a single class from the element
         * @param {string} className - The class to remove
         */
        function removeClass(className) {
            const box = document.getElementById('box');
            
            // classList.remove() safely handles non-existent classes
            box.classList.remove(className);
            
            console.log(`Removed: ${className}`);
            console.log('Current classes:', box.classList.toString());
        }
        
        /**
         * Removes multiple classes at once
         * classList.remove() accepts multiple arguments
         */
        function removeMultiple() {
            const box = document.getElementById('box');
            
            // Remove multiple classes in one call
            box.classList.remove('active', 'highlight');
            
            console.log('Removed multiple classes');
        }
        
        /**
         * Reset all classes
         */
        function resetClasses() {
            const box = document.getElementById('box');
            box.className = 'box active highlight';
            console.log('Classes reset');
        }
    </script>
</body>
</html>
```

**Why:** `classList.remove()` is the modern DOM API for removing classes. It's safer than manipulating `className` string and supports removing multiple classes at once.

## js_addeventlistener_capturing

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Event Capturing Demo</title>
    <style>
        #outer { 
            width: 300px; 
            height: 200px; 
            background: lightblue; 
            padding: 20px;
        }
        #middle { 
            width: 200px; 
            height: 120px; 
            background: lightgreen; 
            padding: 20px;
        }
        #inner { 
            width: 100px; 
            height: 60px; 
            background: lightyellow; 
            padding: 10px;
            cursor: pointer;
        }
    </style>
</head>
<body>
    <!-- PROBLEM: Understand event capturing vs bubbling phases -->
    <!-- TIME: O(1) per event handler SPACE: O(1) -->
    
    <h3>Click the yellow box to see event phases:</h3>
    
    <div id="outer">
        Outer (blue)
        <div id="middle">
            Middle (green)
            <div id="inner">Inner (yellow)</div>
        </div>
    </div>
    
    <div id="log" style="margin-top: 20px; font-family: monospace;"></div>
    
    <button onclick="clearLog()">Clear Log</button>

    <script>
        const log = document.getElementById('log');
        
        /**
         * Logs event phase information
         * @param {Event} event - The event object
         * @param {string} element - Element name for logging
         */
        function logEvent(event, element) {
            const phase = event.eventPhase;
            const phaseName = 
                phase === 1 ? 'CAPTURING' :
                phase === 2 ? 'AT_TARGET' :
                phase === 3 ? 'BUBBLING' : 'NONE';
            
            const message = `${element}: ${phaseName} phase<br>`;
            log.innerHTML += message;
        }
        
        /**
         * Third parameter of addEventListener:
         * - false (default): bubbling phase
         * - true: capturing phase
         */
        
        // CAPTURING phase listeners (outer to inner)
        document.getElementById('outer').addEventListener('click', 
            (e) => logEvent(e, 'OUTER'), true);
        document.getElementById('middle').addEventListener('click', 
            (e) => logEvent(e, 'MIDDLE'), true);
        document.getElementById('inner').addEventListener('click', 
            (e) => logEvent(e, 'INNER'), true);
        
        // BUBBLING phase listeners (inner to outer)
        document.getElementById('outer').addEventListener('click', 
            (e) => logEvent(e, 'OUTER'), false);
        document.getElementById('middle').addEventListener('click', 
            (e) => logEvent(e, 'MIDDLE'), false);
        document.getElementById('inner').addEventListener('click', 
            (e) => logEvent(e, 'INNER'), false);
        
        /**
         * Event flow when clicking inner:
         * 1. CAPTURING: document → outer → middle → inner
         * 2. AT_TARGET: inner
         * 3. BUBBLING: inner → middle → outer → document
         */
        
        function clearLog() {
            log.innerHTML = '';
        }
    </script>
</body>
</html>
```

**Why:** Event capturing (third parameter = `true`) triggers handlers from outer to inner before reaching the target. Useful for intercepting events before they reach child elements.

## js_localStorage_key

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>localStorage.key() Demo</title>
</head>
<body>
    <!-- PROBLEM: Iterate through all localStorage keys -->
    <!-- TIME: O(n) where n = number of keys SPACE: O(1) -->
    
    <h2>localStorage Keys</h2>
    
    <button onclick="addSampleData()">Add Sample Data</button>
    <button onclick="listAllKeys()">List All Keys</button>
    <button onclick="clearStorage()">Clear Storage</button>
    
    <div id="output" style="margin-top: 20px;"></div>

    <script>
        /**
         * localStorage.key(index) returns the key name at given index
         * Useful for iterating when you don't know key names
         */
        
        function addSampleData() {
            localStorage.setItem('username', 'john_doe');
            localStorage.setItem('theme', 'dark');
            localStorage.setItem('language', 'en');
            localStorage.setItem('lastVisit', new Date().toISOString());
            
            document.getElementById('output').innerHTML = 
                '<strong>Sample data added!</strong>';
        }
        
        /**
         * Lists all keys using localStorage.key() method
         * Alternative to Object.keys(localStorage)
         */
        function listAllKeys() {
            const output = document.getElementById('output');
            
            if (localStorage.length === 0) {
                output.innerHTML = 'No items in localStorage';
                return;
            }
            
            let html = '<table border="1" cellpadding="5">';
            html += '<tr><th>Index</th><th>Key</th><th>Value</th></tr>';
            
            // Iterate using key() method
            for (let i = 0; i < localStorage.length; i++) {
                const key = localStorage.key(i);  // Get key at index i
                const value = localStorage.getItem(key);
                
                html += `<tr>
                    <td>${i}</td>
                    <td>${key}</td>
                    <td>${value}</td>
                </tr>`;
            }
            
            html += '</table>';
            html += `<p><strong>Total items:</strong> ${localStorage.length}</p>`;
            
            output.innerHTML = html;
        }
        
        /**
         * Clears all localStorage data
         */
        function clearStorage() {
            localStorage.clear();
            document.getElementById('output').innerHTML = 
                '<strong>Storage cleared!</strong>';
        }
        
        // Note: The order of keys is not guaranteed to be consistent
        // across browsers. Don't rely on specific ordering.
    </script>
</body>
</html>
```

**Why:** `localStorage.key(index)` allows iteration when key names are unknown. Combined with `localStorage.length`, it enables dynamic discovery of stored data.

## js_fetch_put

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>fetch PUT Request Demo</title>
</head>
<body>
    <!-- PROBLEM: Update existing resource using HTTP PUT method -->
    <!-- TIME: O(1) for request setup SPACE: O(1) -->
    
    <h2>Update User (PUT Request)</h2>
    
    <form id="updateForm">
        <label>User ID: <input type="number" id="userId" value="1" required></label><br><br>
        <label>Name: <input type="text" id="userName" value="John Doe" required></label><br><br>
        <label>Email: <input type="email" id="userEmail" value="john@example.com" required></label><br><br>
        <button type="submit">Update User</button>
    </form>
    
    <div id="result" style="margin-top: 20px;"></div>

    <script>
        /**
         * PUT request updates/replaces entire resource
         * @param {number} id - User ID to update
         * @param {object} userData - Complete user data
         */
        async function updateUser(id, userData) {
            const resultDiv = document.getElementById('result');
            
            try {
                const response = await fetch(
                    `https://jsonplaceholder.typicode.com/users/${id}`,
                    {
                        method: 'PUT',
                        headers: {
                            'Content-Type': 'application/json',
                        },
                        body: JSON.stringify(userData)
                    }
                );
                
                if (!response.ok) {
                    throw new Error(`HTTP error! status: ${response.status}`);
                }
                
                const updatedUser = await response.json();
                
                resultDiv.innerHTML = `
                    <h3 style="color: green;">✓ User Updated Successfully</h3>
                    <pre>${JSON.stringify(updatedUser, null, 2)}</pre>
                `;
                
                return updatedUser;
                
            } catch (error) {
                resultDiv.innerHTML = `
                    <h3 style="color: red;">✗ Error</h3>
                    <p>${error.message}</p>
                `;
                console.error('Update failed:', error);
            }
        }
        
        // Form submission handler
        document.getElementById('updateForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            
            const id = document.getElementById('userId').value;
            const userData = {
                name: document.getElementById('userName').value,
                email: document.getElementById('userEmail').value,
                updatedAt: new Date().toISOString()
            };
            
            await updateUser(id, userData);
        });
        
        /**
         * PUT vs PATCH:
         * - PUT: Replaces entire resource
         * - PATCH: Partial update
         */
    </script>
</body>
</html>
```

**Why:** `fetch()` with `method: 'PUT'` updates entire resources. Requires setting `Content-Type: application/json` header and `JSON.stringify()` for the body.

## js_form_reportValidity

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>reportValidity() Demo</title>
    <style>
        input:invalid { border: 2px solid red; }
        input:valid { border: 2px solid green; }
    </style>
</head>
<body>
    <!-- PROBLEM: Programmatically trigger form validation UI -->
    <!-- TIME: O(n) where n = number of form elements SPACE: O(1) -->
    
    <h2>Form Validation with reportValidity()</h2>
    
    <form id="myForm">
        <label>Name: 
            <input type="text" name="name" required minlength="2">
        </label><br><br>
        
        <label>Email: 
            <input type="email" name="email" required>
        </label><br><br>
        
        <label>Age: 
            <input type="number" name="age" min="18" max="120" required>
        </label><br><br>
        
        <button type="button" onclick="validateForm()">Validate Form</button>
        <button type="button" onclick="checkWithoutUI()">Check Validity (No UI)</button>
        <button type="submit">Submit</button>
    </form>
    
    <div id="status" style="margin-top: 20px;"></div>

    <script>
        /**
         * reportValidity() checks validity AND shows browser validation UI
         * Returns true if form is valid, false otherwise
         */
        function validateForm() {
            const form = document.getElementById('myForm');
            const status = document.getElementById('status');
            
            // reportValidity() triggers validation UI
            const isValid = form.reportValidity();
            
            if (isValid) {
                status.innerHTML = '<span style="color: green;">✓ Form is valid!</span>';
            } else {
                status.innerHTML = '<span style="color: red;">✗ Form has errors</span>';
            }
            
            return isValid;
        }
        
        /**
         * checkValidity() only checks without showing UI
         * Useful for custom validation feedback
         */
        function checkWithoutUI() {
            const form = document.getElementById('myForm');
            const status = document.getElementById('status');
            
            // checkValidity() returns boolean without UI
            const isValid = form.checkValidity();
            
            status.innerHTML = isValid 
                ? '<span style="color: green;">✓ Valid (silent check)</span>'
                : '<span style="color: orange;">⚠ Invalid (no UI shown)</span>';
            
            return isValid;
        }
        
        /**
         * Form submission with reportValidity
         */
        document.getElementById('myForm').addEventListener('submit', (e) => {
            e.preventDefault();
            
            if (e.target.reportValidity()) {
                // Form is valid, proceed with submission
                const formData = new FormData(e.target);
                const data = Object.fromEntries(formData);
                
                document.getElementById('status').innerHTML = 
                    `<span style="color: green;">✓ Submitted:</span><br>
                    <pre>${JSON.stringify(data, null, 2)}</pre>`;
            }
        });
        
        /**
         * reportValidity() vs checkValidity():
         * - reportValidity(): Shows browser validation messages
         * - checkValidity(): Silent validation check
         */
    </script>
</body>
</html>
```

**Why:** `reportValidity()` triggers native browser validation UI and returns validity status. Unlike `checkValidity()`, it shows validation messages to users automatically.

## html5_geolocation_getCurrentPosition

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Geolocation API Demo</title>
    <style>
        #map { 
            width: 100%; 
            height: 300px; 
            background: #e0e0e0; 
            display: flex;
            align-items: center;
            justify-content: center;
        }
    </style>
</head>
<body>
    <!-- PROBLEM: Get user's current geographic location -->
    <!-- TIME: O(1) for API call, async for GPS retrieval SPACE: O(1) -->
    
    <h2>Geolocation API</h2>
    
    <button onclick="getLocation()">Get My Location</button>
    <button onclick="getLocationHighAccuracy()">High Accuracy</button>
    
    <div id="result" style="margin-top: 20px;"></div>
    <div id="map">
        <p>Location will appear here</p>
    </div>

    <script>
        /**
         * getCurrentPosition() retrieves user's current position
         * Requires HTTPS or localhost
         * User must grant permission
         */
        function getLocation() {
            const resultDiv = document.getElementById('result');
            
            if (!navigator.geolocation) {
                resultDiv.innerHTML = '<span style="color: red;">Geolocation not supported</span>';
                return;
            }
            
            resultDiv.innerHTML = 'Getting location...';
            
            // Basic position request
            navigator.geolocation.getCurrentPosition(
                // Success callback
                (position) => {
                    const { latitude, longitude, accuracy } = position.coords;
                    const timestamp = new Date(position.timestamp);
                    
                    resultDiv.innerHTML = `
                        <h3 style="color: green;">✓ Location Found</h3>
                        <p><strong>Latitude:</strong> ${latitude.toFixed(6)}</p>
                        <p><strong>Longitude:</strong> ${longitude.toFixed(6)}</p>
                        <p><strong>Accuracy:</strong> ${accuracy.toFixed(0)} meters</p>
                        <p><strong>Timestamp:</strong> ${timestamp.toLocaleString()}</p>
                    `;
                    
                    // Show on map placeholder
                    document.getElementById('map').innerHTML = `
                        <a href="https://www.openstreetmap.org/?mlat=${latitude}&mlon=${longitude}&zoom=15" 
                           target="_blank">
                            View on OpenStreetMap
                        </a>
                    `;
                },
                // Error callback
                (error) => {
                    const errorMessages = {
                        1: 'PERMISSION_DENIED - User denied location access',
                        2: 'POSITION_UNAVAILABLE - Location unavailable',
                        3: 'TIMEOUT - Request timed out'
                    };
                    
                    resultDiv.innerHTML = `
                        <span style="color: red;">✗ Error: ${errorMessages[error.code]}</span>
                    `;
                }
            );
        }
        
        /**
         * High accuracy mode with options
         */
        function getLocationHighAccuracy() {
            const resultDiv = document.getElementById('result');
            
            const options = {
                enableHighAccuracy: true,  // Use GPS if available
                timeout: 10000,            // Max wait time (ms)
                maximumAge: 0              // Don't use cached position
            };
            
            resultDiv.innerHTML = 'Getting high-accuracy location (may take longer)...';
            
            navigator.geolocation.getCurrentPosition(
                (position) => {
                    const { latitude, longitude, accuracy, altitude, heading, speed } = position.coords;
                    
                    resultDiv.innerHTML = `
                        <h3 style="color: green;">✓ High-Accuracy Location</h3>
                        <p><strong>Latitude:</strong> ${latitude}</p>
                        <p><strong>Longitude:</strong> ${longitude}</p>
                        <p><strong>Accuracy:</strong> ${accuracy} meters</p>
                        ${altitude ? `<p><strong>Altitude:</strong> ${altitude} meters</p>` : ''}
                        ${heading ? `<p><strong>Heading:</strong> ${heading}°</p>` : ''}
                        ${speed ? `<p><strong>Speed:</strong> ${speed} m/s</p>` : ''}
                    `;
                },
                (error) => {
                    resultDiv.innerHTML = `<span style="color: red;">Error: ${error.message}</span>`;
                },
                options
            );
        }
    </script>
</body>
</html>
```

**Why:** `getCurrentPosition()` retrieves user location via GPS/WiFi/IP. Requires HTTPS, user permission. Use options for high-accuracy GPS mode.

## css_root_selector

**Why:** `:root` pseudo-class targets the document root for defining global CSS variables. Higher specificity than `html` selector. Variables cascade to all child elements.

## css_min_function

```html

        /* Useful for responsive sizing without media queries */

            display: flex;
            align-items: center;
```

```html
        /**
         * throw statement creates custom errors
         * Can throw any value: Error objects, strings, numbers, objects

        // ============================================
        // Custom Error Types
        // ============================================
        
        class ValidationError extends Error {
            constructor(message) {
                super(message);
                this.name = 'ValidationError';
            }
        }
        
        class InsufficientFundsError extends Error {
            constructor(balance, amount) {
                super(`Insufficient funds: $${balance} available, $${amount} requested`);
                this.name = 'InsufficientFundsError';
                this.balance = balance;
                this.amount = amount;
            }
        }
        
        class TemperatureError extends Error {
            constructor(value, reason) {
                super(`Invalid temperature ${value}°C: ${reason}`);
                this.name = 'TemperatureError';
                this.value = value;
            }
        }
        
        // ============================================
        // Temperature Example
        // ============================================
        
        function convertTemperature() {
            const input = document.getElementById('celsius');
            const resultDiv = document.getElementById('result');
            const celsius = parseFloat(input.value);
            
            try {
                // Input validation with throw
                if (isNaN(celsius)) {
                    throw new ValidationError('Please enter a valid number');
                }
                
                // Physical limits validation
                if (celsius < -273.15) {
                    throw new TemperatureError(celsius, 'below absolute zero');
                }
                
                if (celsius > 1000) {
                    throw new TemperatureError(celsius, 'unreasonably high');
                }
                
                const fahrenheit = (celsius * 9/5) + 32;
                resultDiv.innerHTML = `<span style="color: green;">
                    ${celsius}°C = ${fahrenheit.toFixed(2)}°F
                </span>`;
                
            } catch (error) {
                resultDiv.innerHTML = `<span style="color: red;">
                    Error: ${error.message}
                </span>`;
                console.error(error);
            }

        // Bank Account Example

        let balance = 1000;
        
        function withdraw() {
            const amountInput = document.getElementById('amount');
            const transactionDiv = document.getElementById('transaction');
            const amount = parseFloat(amountInput.value);
            
            try {
                if (isNaN(amount) || amount <= 0) {
                    throw new ValidationError('Amount must be a positive number');
                }
                
                if (amount > balance) {
                    throw new InsufficientFundsError(balance, amount);
                }

                    ${error.message}
                </span>`;
```
