function getSiblings(element) {
    const siblings = [];
    let sibling = element.parentElement.firstElementChild;

        if (sibling !== element) {
            siblings.push(sibling);
        }
        sibling = sibling.nextElementSibling;

            if (e.target.tagName === 'LI') {
                clearHighlights();
                selectedElement.classList.remove('selected');
                selectedElement = e.target;
                selectedElement.classList.add('selected');
                updateStatus();
            }
        });

        function selectNext() {
            clearHighlights();
            if (selectedElement.nextElementSibling) {
                selectedElement.classList.remove('selected');
                selectedElement = selectedElement.nextElementSibling;
                selectedElement.classList.add('selected');
                updateStatus();
            }
        }

        function selectPrevious() {
            clearHighlights();
            if (selectedElement.previousElementSibling) {
                selectedElement.classList.remove('selected');
                selectedElement = selectedElement.previousElementSibling;
                selectedElement.classList.add('selected');
                updateStatus();
            }
        }

        function highlightAll() {
            const siblings = getSiblings(selectedElement);
            siblings.forEach(sib => sib.classList.add('highlighted'));
        }

        function clearHighlights() {
            document.querySelectorAll('.highlighted').forEach(el => {
                el.classList.remove('highlighted');

        function getSiblings(element) {
            const siblings = [];
            let sibling = element.parentElement.firstElementChild;
            while (sibling) {
                if (sibling !== element) siblings.push(sibling);
                sibling = sibling.nextElementSibling;
            }
            return siblings;
        }

        function updateStatus() {
            const status = document.getElementById('status');
            const prev = selectedElement.previousElementSibling;
            const next = selectedElement.nextElementSibling;
            const siblings = getSiblings(selectedElement);
            
            status.innerHTML = `
                <strong>Selected:</strong> ${selectedElement.textContent}<br>
                <strong>Previous:</strong> ${prev ? prev.textContent : 'None (first child)'}<br>
                <strong>Next:</strong> ${next ? next.textContent : 'None (last child)'}<br>
                <strong>Total Siblings:</strong> ${siblings.length}
            `;
        }
    </script>

**Why**: Sibling navigation is essential for DOM traversal without using querySelector. Common use cases include: carousel/slider implementations, tabbed interfaces, accordion menus, keyboard navigation (arrow keys), and implementing custom focus management. Always prefer `nextElementSibling`/`previousElementSibling` over `nextSibling`/`previousSibling` to avoid text node issues.
