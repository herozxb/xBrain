## js_event_capture_option

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Event Capture Option Demo</title>
    <style>
        #outer { background: lightblue; padding: 40px; margin: 20px; }
        #middle { background: lightgreen; padding: 30px; }
        #inner { background: lightcoral; padding: 20px; cursor: pointer; }
        .log { margin-top: 20px; padding: 10px; background: #f5f5f5; max-height: 200px; overflow-y: auto; }
    </style>
</head>
<body>
    <h3>Event Capture vs Bubbling Demo</h3>
    <div id="outer">Outer (Capture: ON)
        <div id="middle">Middle (Capture: OFF)
            <div id="inner">Inner - Click Me!</div>
        </div>
    </div>
    <div class="log" id="log"></div>

    <script>
        // PROBLEM: How to use event capture option in addEventListener
        // TIME: O(1) SPACE: O(1)
        
        /**
         * Demonstrates the capture option in addEventListener.
         * When capture is true, events are handled during the capture phase
         * (from window down to target). When false (default), bubbling phase
         * (from target up to window).
         * 
         * @param {string} phase - 'capture' or 'bubble'
         * @param {Event} event - The event object
         */
        function logEvent(phase, event) {
            const logDiv = document.getElementById('log');
            const timestamp = new Date().toLocaleTimeString();
            const targetId = event.currentTarget.id;
            logDiv.innerHTML += `[${timestamp}] ${phase}: ${targetId}<br>`;
            logDiv.scrollTop = logDiv.scrollHeight;
        }

        // Outer: capture = true (handles during capture phase first)
        document.getElementById('outer').addEventListener('click', (e) => {
            logEvent('CAPTURE', e);
        }, true); // <-- Third parameter: capture option

        // Middle: capture = false (default - handles during bubble phase)
        document.getElementById('middle').addEventListener('click', (e) => {
            logEvent('BUBBLE', e);
        }, false);

        // Inner: capture = false
        document.getElementById('inner').addEventListener('click', (e) => {
            logEvent('BUBBLE', e);
            e.stopPropagation(); // Stop further propagation
        });

        // Why: The capture option (3rd parameter) determines WHEN the handler 
        // executes during event propagation. Useful for intercepting events 
        // before they reach child elements (e.g., delegation patterns, security).
    </script>
</body>
</html>
```

**Why**: The `capture` option in `addEventListener(element, handler, capture)` determines event handling phase - true for capture (top-down), false for bubbling (bottom-up). Essential for event delegation and intercepting events before children.

## js_cookies_delete

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Cookie Delete Demo</title>
    <style>
        body { font-family: Arial, sans-serif; padding: 20px; }
        button { margin: 5px; padding: 10px 20px; cursor: pointer; }
        .result { margin-top: 20px; padding: 15px; background: #f0f0f0; border-radius: 5px; }
    </style>
</head>
<body>
    <h3>Cookie Deletion Demo</h3>
    <button onclick="setTestCookie()">Set Test Cookie</button>
    <button onclick="showCookies()">Show Cookies</button>
    <button onclick="deleteCookie('testCookie')">Delete Test Cookie</button>
    <div class="result" id="result"></div>

    <script>
        // PROBLEM: How to delete cookies in JavaScript
        // TIME: O(1) SPACE: O(1)

        /**
         * Sets a test cookie for demonstration purposes.
         */
        function setTestCookie() {
            const expires = new Date(Date.now() + 86400000).toUTCString(); // 1 day
            document.cookie = `testCookie=HelloWorld; expires=${expires}; path=/`;
            displayResult('Cookie set: testCookie=HelloWorld');
        }

        /**
         * Deletes a cookie by setting its expiration date to the past.
         * Cookies are deleted by the browser when expired.
         * 
         * @param {string} name - The name of the cookie to delete
         * @param {string} [path='/'] - Must match the path used when setting
         * @param {string} [domain] - Must match the domain used when setting
         */
        function deleteCookie(name, path = '/', domain = '') {
            // Set expiration to Thu, 01 Jan 1970 00:00:00 GMT
            const expiredDate = 'Thu, 01 Jan 1970 00:00:00 GMT';
            
            // Build cookie string - path and domain must match original
            let cookieStr = `${name}=; expires=${expiredDate}; path=${path}`;
            if (domain) cookieStr += `; domain=${domain}`;
            
            document.cookie = cookieStr;
            displayResult(`Cookie "${name}" deleted`);
        }

        /**
         * Displays all current cookies.
         */
        function showCookies() {
            const cookies = document.cookie || '(no cookies)';
            displayResult(`Current cookies: ${cookies}`);
        }

        /**
         * Helper to display results in the UI.
         */
        function displayResult(message) {
            document.getElementById('result').textContent = message;
        }

        // Why: Cookies can only be deleted by setting their expiration to a past date.
        // The path and domain parameters MUST match those used when setting the cookie,
        // otherwise deletion will fail silently.
    </script>
</body>
</html>
```

**Why**: Cookies cannot be directly deleted via API. Instead, set `expires` to a past date (typically epoch: `Thu, 01 Jan 1970 00:00:00 GMT`). The `path` and `domain` must match the original cookie's attributes.

## js_response_ok

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Response.ok Property Demo</title>
    <style>
        body { font-family: Arial, sans-serif; padding: 20px; }
        button { margin: 5px; padding: 10px 15px; cursor: pointer; }
        .success { color: green; }
        .error { color: red; }
        #result { margin-top: 20px; padding: 15px; background: #f5f5f5; border-radius: 5px; }
    </style>
</head>
<body>
    <h3>Response.ok Property Demo</h3>
    <button onclick="fetchValidData()">Fetch Valid (200)</button>
    <button onclick="fetchNotFound()">Fetch 404</button>
    <button onclick="fetchServerError()">Fetch 500</button>
    <div id="result"></div>

    <script>
        // PROBLEM: How to check if HTTP response was successful using Response.ok
        // TIME: O(1) SPACE: O(1)

        /**
         * Fetches data from a valid endpoint and checks Response.ok.
         * Response.ok is true when status is in the range 200-299.
         */
        async function fetchValidData() {
            try {
                const response = await fetch('https://jsonplaceholder.typicode.com/posts/1');
                
                // response.ok is shorthand for: response.status >= 200 && response.status < 300
                if (response.ok) {
                    const data = await response.json();
                    displayResult(`✅ Success! Status: ${response.status}`, true);
                    console.log('Data:', data);
                } else {
                    displayResult(`❌ HTTP Error: ${response.status}`, false);
                }
            } catch (error) {
                displayResult(`❌ Network Error: ${error.message}`, false);
            }
        }

        /**
         * Fetches a non-existent resource (404) to demonstrate response.ok = false.
         */
        async function fetchNotFound() {
            try {
                const response = await fetch('https://jsonplaceholder.typicode.com/posts/99999');
                
                // 404 means response.ok will be false
                if (response.ok) {
                    displayResult(`✅ Success! Status: ${response.status}`, true);
                } else {
                    displayResult(`❌ Not Found (404). response.ok = ${response.ok}`, false);
                }
            } catch (error) {
                displayResult(`❌ Network Error: ${error.message}`, false);
            }
        }

        /**
         * Simulates a server error response.
         */
        async function fetchServerError() {
            try {
                // Using httpstat.us to simulate 500 error
                const response = await fetch('https://httpstat.us/500');
                
                // 500 means response.ok will be false
                if (response.ok) {
                    displayResult(`✅ Success! Status: ${response.status}`, true);
                } else {
                    displayResult(`❌ Server Error (500). response.ok = ${response.ok}`, false);
                }
            } catch (error) {
                displayResult(`❌ Network Error: ${error.message}`, false);
            }
        }

        /**
         * Helper to display results.
         */
        function displayResult(message, success) {
            const resultDiv = document.getElementById('result');
            resultDiv.className = success ? 'success' : 'error';
            resultDiv.textContent = message;
        }

        // Why: Response.ok is a boolean that simplifies HTTP success checking.
        // It's true only for status codes 200-299. Fetch doesn't reject on HTTP errors,
        // so always check response.ok or response.status before processing data.
    </script>
</body>
</html>
```

**Why**: `Response.ok` is a read-only boolean that returns `true` only when HTTP status is 200-299. Fetch API doesn't throw on HTTP errors (404, 500), so always check `response.ok` before parsing response data.

## js_input_event

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Input Event Demo</title>
    <style>
        body { font-family: Arial, sans-serif; padding: 20px; }
        input, textarea { width: 100%; max-width: 400px; padding: 10px; margin: 10px 0; }
        .preview { padding: 15px; background: #e8f5e9; border-radius: 5px; min-height: 50px; }
        .counter { color: #666; font-size: 0.9em; }
    </style>
</head>
<body>
    <h3>Input Event - Real-time Input Tracking</h3>
    
    <label>Live Character Counter:</label>
    <input type="text" id="textInput" placeholder="Type something...">
    <div class="counter">Characters: <span id="charCount">0</span></div>
    
    <label>Live Preview:</label>
    <textarea id="previewInput" placeholder="Type for live preview..." rows="4"></textarea>
    <div class="preview" id="preview"></div>

    <script>
        // PROBLEM: How to detect real-time input changes using the input event
        // TIME: O(1) SPACE: O(1)

        /**
         * The 'input' event fires on every change to the value of:
         * - <input>, <textarea>, <select> elements
         * - elements with contenteditable="true"
         * 
         * It fires on:
         * - Typing characters
         * - Pasting, cutting, deleting
         * - Drag-and-drop
         * - Speech recognition input
         * 
         * Unlike 'change' event, it fires IMMEDIATELY, not on blur.
         */

        const textInput = document.getElementById('textInput');
        const charCountSpan = document.getElementById('charCount');
        const previewInput = document.getElementById('previewInput');
        const previewDiv = document.getElementById('preview');

        /**
         * Updates character count in real-time.
         */
        textInput.addEventListener('input', function(event) {
            // event.target.value contains the current input value
            const value = event.target.value;
            charCountSpan.textContent = value.length;
            
            // Can also access event.data (the inserted characters)
            // and event.inputType (type of change: insertText, deleteContent, etc.)
            console.log('Input type:', event.inputType);
            console.log('Inserted data:', event.data);
        });

        /**
         * Creates live preview as user types.
         */
        previewInput.addEventListener('input', function(event) {
            const value = event.target.value;
            // Escape HTML to prevent XSS in preview
            const escaped = value
                .replace(/&/g, '&amp;')
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;');
            
            previewDiv.innerHTML = escaped || '<em>Preview will appear here...</em>';
        });

        /**
         * Input event also works with contenteditable elements.
         */
        // Why: The input event is the most reliable way to detect real-time
        // value changes. Unlike keyup/keydown, it catches all input methods
        // including paste, drag-drop, autofill, and speech input.
    </script>
</body>
</html>
```

**Why**: The `input` event fires synchronously on every value change (typing, pasting, cutting, dragging, autofill). Unlike `change` (fires on blur) or `keyup` (misses non-keyboard input), `input` is the most reliable for real-time validation and preview.

## html5_web_workers_postMessage

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Web Worker postMessage Demo</title>
    <style>
        body { font-family: Arial, sans-serif; padding: 20px; }
        button { padding: 10px 20px; margin: 5px; cursor: pointer; }
        #result { margin-top: 20px; padding: 15px; background: #f5f5f5; border-radius: 5px; }
        #counter { font-size: 1.2em; font-weight: bold; color: #2196F3; }
        .note { color: #666; font-style: italic; margin-top: 10px; }
    </style>
</head>
<body>
    <h3>Web Worker postMessage Communication</h3>
    <p>Counter (UI stays responsive): <span id="counter">0</span></p>
    <button onclick="startHeavyComputation()">Start Heavy Computation</button>
    <button onclick="stopWorker()">Stop Worker</button>
    <button onclick="incrementCounter()">Increment Counter (Test UI)</button>
    <div id="result"></div>
    <p class="note">Note: Workers run in a separate thread. In production, use a separate .js file.</p>

    <script>
        // PROBLEM: How to communicate between main thread and Web Worker using postMessage
        // TIME: O(n) for message serialization SPACE: O(message size)

        let worker = null;
        let counter = 0;

        /**
         * Creates an inline Web Worker using a Blob.
         * In production, use a separate .js file.
         */
        function createWorker() {
            // Worker code as a string (normally in separate file)
            const workerCode = `
                // Worker receives messages via onmessage
                self.onmessage = function(event) {
                    const { type, data } = event.data;
                    
                    if (type === 'compute') {
                        // Simulate heavy computation
                        const n = data;
                        let result = 0;
                        for (let i = 0; i < n; i++) {
                            result += Math.sqrt(i);
                        }
                        
                        // Send result back to main thread
                        self.postMessage({
                            type: 'result',
                            data: result
                        });
                    }
                };
            `;

            const blob = new Blob([workerCode], { type: 'application/javascript' });
            const workerUrl = URL.createObjectURL(blob);
            return new Worker(workerUrl);
        }

        /**
         * Starts heavy computation in a background worker.
         * postMessage sends data to the worker thread.
         */
        function startHeavyComputation() {
            if (worker) {
                displayResult('Worker already running...');
                return;
            }

            worker = createWorker();

            // Listen for messages FROM worker
            worker.onmessage = function(event) {
                const { type, data } = event.data;
                
                if (type === 'result') {
                    displayResult(`Computation complete! Result: ${data.toFixed(2)}`);
                    stopWorker();
                }
            };

            // Handle worker errors
            worker.onerror = function(error) {
                displayResult(`Worker error: ${error.message}`);
                stopWorker();
            };

            // Send message TO worker using postMessage
            // Data is STRUCTURED CLONED (not shared by reference)
            worker.postMessage({
                type: 'compute',
                data: 10000000 // 10 million iterations
            });

            displayResult('Computing... (UI remains responsive)');
        }

        /**
         * Terminates the worker and cleans up.
         */
        function stopWorker() {
            if (worker) {
                worker.terminate();
                worker = null;
                displayResult('Worker terminated.');
            }
        }

        /**
         * Demonstrates UI responsiveness while worker runs.
         */
        function incrementCounter() {
            counter++;
            document.getElementById('counter').textContent = counter;
        }

        function displayResult(message) {
            document.getElementById('result').textContent = message;
        }

        // Why: postMessage enables safe communication between threads.
        // Data is serialized (structured clone algorithm), not shared.
        // This prevents race conditions while allowing parallel computation.
        // Workers keep the main thread free for UI interactions.
    </script>
</body>
</html>
```

**Why**: `postMessage()` enables thread-safe communication between the main thread and Web Workers. Messages are structured-cloned (copied), not shared by reference, preventing race conditions. Workers run in parallel, keeping the UI responsive during heavy computations.

## js_typeerror

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">

<body>
    <h3>TypeError - Common Causes and Handling</h3>
    <button onclick="throwNullReference()">Null Reference</button>
    <button onclick="throwNotAFunction()">Not a Function</button>
    <button onclick="throwInvalidAssignment()">Invalid Assignment</button>
    <button onclick="safePropertyAccess()">Safe Property Access</button>
    <div id="output"></div>

    <script>
        // PROBLEM: What causes TypeError and how to handle it
        // TIME: O(1) SPACE: O(1)

        /**
         * TypeError is thrown when:
         * 1. Accessing property on null/undefined
         * 2. Calling something that isn't a function
         * 3. Assigning to non-writable property
         * 4. Using invalid operand types
         */

        /**
         * Demonstrates: Cannot read property of null/undefined
         */
        function throwNullReference() {
            try {
                const user = null;
                // TypeError: Cannot read properties of null
                const name = user.name;
            } catch (error) {
                displayError(error);
            }
        }

        /**
         * Demonstrates: X is not a function
         */
        function throwNotAFunction() {
            try {
                const obj = { method: 'not a function' };
                // TypeError: obj.method is not a function
                obj.method();
            } catch (error) {
                displayError(error);
            }
        }

        /**
         * Demonstrates: Cannot assign to read only property
         */
        function throwInvalidAssignment() {
            try {
                const obj = Object.freeze({ prop: 1 });
                // TypeError: Cannot assign to read only property
                obj.prop = 2;
            } catch (error) {
                displayError(error);
            }
        }

        /**
         * Safe property access using optional chaining (?.)
         * Prevents TypeError from null/undefined access
         */
        function safePropertyAccess() {
            const user = null;
            
            // Old way (causes TypeError):
            // const city = user.address.city;
            
            // Safe way with optional chaining (ES2020):
            const city = user?.address?.city ?? 'Unknown';
            
            // Also works with methods:
            const result = user?.getName?.() ?? 'No name';
            
            displaySuccess(`Safe access result: ${city}`);
        }

        /**
         * Type checking helper to prevent TypeErrors
         */
        function safeCall(obj, methodName, ...args) {
            if (obj && typeof obj[methodName] === 'function') {
                return obj[methodName](...args);
            }
            return undefined;
        }

        function displayError(error) {
            const output = document.getElementById('output');
            output.innerHTML = `<div class="error">
                <strong>${error.name}:</strong> ${error.message}<br>
                <small>Stack: ${error.stack?.split('\n')[1]}</small>
            </div>`;
        }

        function displaySuccess(message) {
            document.getElementById('output').innerHTML = 
                `<div class="success">${message}</div>`;
        }

        // Why: TypeError is one of the most common JS errors. Use optional
        // chaining (?.) and nullish coalescing (??) to prevent null/undefined
        // errors. Always check types before calling functions or accessing
        // properties when types are uncertain.
    </script>
</body>
</html>
```

**Why**: `TypeError` occurs when operations receive unexpected types (accessing property on `null`, calling non-function, assigning to read-only). Prevent with optional chaining (`?.`), type checking (`typeof`), and defensive programming.

        <label><input type="checkbox" id="caseInsensitive"> Case insensitive</label>
        <label><input type="checkbox" id="useRegExp"> Use RegExp</label>
    </div>
    
    <div>
        <label>Replace with:</label>
        <input type="text" id="replaceWith" placeholder="cat" value="cat">
    </div>
    
    <button onclick="performReplace()">Replace All</button>
    
    <div class="result" id="result"></div>

    <script>
        // PROBLEM: How to replace all occurrences using replaceAll() with RegExp
        // TIME: O(n) where n is string length SPACE: O(n) for new string

        /**
         * Demonstrates String.prototype.replaceAll() with RegExp.
         * 
         * Key differences from replace():
         * - replaceAll() requires global flag (g) for RegExp
         * - replaceAll() replaces ALL occurrences by default
         * - replace() with string only replaces first occurrence
         */
        function performReplace() {
            const inputText = document.getElementById('inputText').value;
            const pattern = document.getElementById('searchPattern').value;
            const replacement = document.getElementById('replaceWith').value;
            const caseInsensitive = document.getElementById('caseInsensitive').checked;
            const useRegExp = document.getElementById('useRegExp').checked;

            let result;
            let searchValue;

            if (useRegExp) {
                // Build RegExp with appropriate flags
                // replaceAll REQUIRES global flag (g)
                let flags = 'g';
                if (caseInsensitive) flags += 'i';
                
                try {
                    searchValue = new RegExp(pattern, flags);
                    result = inputText.replaceAll(searchValue, replacement);
                } catch (error) {
                    document.getElementById('result').innerHTML = 
                        `<span style="color:red;">Invalid RegExp: ${error.message}</span>`;
                    return;
                }
            } else {
                // String pattern - replaceAll replaces all occurrences
                // Unlike replace() which only replaces first
                searchValue = pattern;
                result = inputText.replaceAll(searchValue, replacement);
            }

            // Display result with highlighting
            const highlighted = result.replace(
                new RegExp(replacement, 'g'),
                `<mark>${replacement}</mark>`
            );

            document.getElementById('result').innerHTML = `
                <strong>Result:</strong><br>
                ${highlighted}<br><br>
                <small>Replaced ${countOccurrences(inputText, searchValue)} occurrence(s)</small>
            `;
        }

        /**
         * Counts occurrences of pattern in text.
         */
        function countOccurrences(text, pattern) {
            const matches = text.match(
                pattern instanceof RegExp 
                    ? pattern 
                    : new RegExp(escapeRegExp(pattern), 'g')
            );
            return matches ? matches.length : 0;
        }

        /**
         * Escapes special RegExp characters for literal string matching.
         */
        function escapeRegExp(string) {
            return string.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
        }

        // Why: replaceAll() (ES2021) simplifies replacing all occurrences.
        // With RegExp, it REQUIRES the 'g' flag (throws error otherwise).
        // More readable than replace(/pattern/g) and clearer intent.
        
        // Comparison:
        // 'aaa'.replace('a', 'b')      → 'baa' (only first)
        // 'aaa'.replaceAll('a', 'b')   → 'bbb' (all)
        // 'aaa'.replace(/a/g, 'b')     → 'bbb' (all via regex)
        // 'aaa'.replaceAll(/a/g, 'b')  → 'bbb' (requires 'g' flag)
    </script>
</body>
</html>
```

**Why**: `String.prototype.replaceAll()` (ES2021) replaces all occurrences without needing regex `g` flag manually. With RegExp, the `g` flag is **required** (throws otherwise). More explicit than `replace(/pattern/g)` and clearer in intent.

    <script>
        // PROBLEM: How to correctly use Date.getMonth() with zero-based indexing
        // TIME: O(1) SPACE: O(1)

        // Month names for display
        const MONTH_NAMES = [
            'January', 'February', 'March', 'April', 'May', 'June',
            'July', 'August', 'September', 'October', 'November', 'December'
        ];

        const MONTH_NAMES_SHORT = [
            'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
            'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'
        ];

        /**
         * Displays month information for the selected date.
         * Demonstrates correct usage of getMonth().
         */
        function showMonthInfo() {
            const dateInput = document.getElementById('dateInput');
            const date = dateInput.value ? new Date(dateInput.value) : new Date();
            
            // getMonth() returns 0-11
            const monthIndex = date.getMonth(); // 0-11
            const monthName = MONTH_NAMES[monthIndex];
            const monthNumber = monthIndex + 1; // Convert to 1-12 for display
            
            // Other useful date methods
            const year = date.getFullYear();
            const dayOfMonth = date.getDate(); // Note: getDate(), NOT getDay()
            const dayOfWeek = date.getDay(); // 0 (Sunday) to 6 (Saturday)

            // Days in the month
            const daysInMonth = new Date(year, monthIndex + 1, 0).getDate();

            document.getElementById('result').innerHTML = `
                <strong>Date:</strong> ${date.toDateString()}<br><br>
                <strong>getMonth():</strong> ${monthIndex} (zero-indexed)<br>
                <strong>Month Name:</strong> ${monthName}<br>
                <strong>Month Number:</strong> ${monthNumber} (1-12)<br>
                <strong>Year:</strong> ${year}<br>
                <strong>Day of Month:</strong> ${dayOfMonth}<br>
                <strong>Day of Week:</strong> ${['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'][dayOfWeek]}<br>
                <strong>Days in Month:</strong> ${daysInMonth}
            `;
        }

        /**
         * Helper: Get month name from date.
         */
        function getMonthName(date) {
            return MONTH_NAMES[date.getMonth()];
        }

        /**
         * Helper: Get month number (1-12) from date.
         */
        function getMonthNumber(date) {
            return date.getMonth() + 1;
        }

        /**
         * Helper: Create date from month number (1-12).
         * Converts human month to Date month.
         */
        function createDateFromMonth(year, month1to12, day) {
            return new Date(year, month1to12 - 1, day); // Subtract 1!
        }

        /**
         * Helper: Get days in a specific month.
         */
        function getDaysInMonth(year, month1to12) {
            // Day 0 of next month = last day of current month
            return new Date(year, month1to12, 0).getDate();
        }

        // Initialize with today's date
        document.getElementById('dateInput').valueAsDate = new Date();
        showMonthInfo();

        // Why: getMonth() returns 0-11 (zero-indexed), a common source of bugs.
        // Always add 1 for display, subtract 1 when creating dates.
        // This design comes from C's tm_mon struct (1970s).
        // Modern alternative: toLocaleString('en-US', { month: 'long' })
    </script>
</body>
</html>
```

**Why**: `Date.getMonth()` returns 0-11 (zero-indexed), a common source of bugs. Always `+1` for display, `-1` when creating dates. This design originated from C's `tm_mon` (1970s). Modern alternative: `date.toLocaleString('en-US', {month: 'long'})`.
