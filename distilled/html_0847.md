```html
    // Validate step constraint for numeric inputs
    // Uses Constraint Validation API to check if value matches step
    function validateStep(inputElement, validationDiv) {
      const value = inputElement.value;
      const step = parseFloat(inputElement.step);
      const min = parseFloat(inputElement.min) || 0;
      
      // Skip validation if empty
      if (!value) {
        validationDiv.textContent = '';
        return;
      }
      
      // Check validity using native API
      const isValid = inputElement.checkValidity() && 
                      inputElement.validity.valid;
      
      if (isValid) {
        // Additional step validation for numbers
        if (inputElement.type === 'number') {
          const numValue = parseFloat(value);
          const remainder = (numValue - min) % step;
          const epsilon = 0.0001; // Handle floating point precision
          
          if (Math.abs(remainder) > epsilon && 
              Math.abs(remainder - step) > epsilon) {
            validationDiv.textContent = 
              `Value must be in increments of ${step}`;
            validationDiv.className = 'validation-message invalid';
            return;
          }
        }
        
        validationDiv.textContent = '✓ Valid';
        validationDiv.className = 'validation-message valid';
      } else {
        validationDiv.textContent = 
          inputElement.validationMessage || 'Invalid value';
        validationDiv.className = 'validation-message invalid';
      }
    }
    
    // Attach event listeners
    document.getElementById('price').addEventListener('input', (e) => {
      validateStep(e.target, document.getElementById('price-validation'));
    });
    
    document.getElementById('meeting-time').addEventListener('change', (e) => {
      validateStep(e.target, document.getElementById('time-validation'));
    });
    
    document.getElementById('weekly-date').addEventListener('change', (e) => {
      validateStep(e.target, document.getElementById('date-validation'));
```

```html
    // Store directory structure
    let directoryFiles = [];
    let directoryTree = {};
    
    const dirInput = document.getElementById('dir-input');
    const treeDisplay = document.getElementById('tree-display');
    const uploadBtn = document.getElementById('upload-dir-btn');
    const statsDiv = document.getElementById('upload-stats');
    
    // Handle directory selection
    dirInput.addEventListener('change', (e) => {
      const files = Array.from(e.target.files);
      
      if (files.length === 0) return;
      
      directoryFiles = files;
      directoryTree = buildDirectoryTree(files);
      
      // Display tree structure
      renderTree(directoryTree);
      
      // Show stats
      const totalSize = files.reduce((sum, f) => sum + f.size, 0);
      statsDiv.innerHTML = `
        <p><strong>Total Files:</strong> ${files.length}</p>
        <p><strong>Total Size:</strong> ${formatSize(totalSize)}</p>
      `;
      
      uploadBtn.style.display = 'block';
    });
    
    // Build directory tree structure
    function buildDirectoryTree(files) {
      const tree = {};
      
      files.forEach(file => {
        // webkitRelativePath gives full path like "folder/subfolder/file.txt"
        const path = file.webkitRelativePath;
        const parts = path.split('/');
        
        let current = tree;
        
        parts.forEach((part, index) => {
          if (index === parts.length - 1) {
            // File
            current[part] = {
              type: 'file',
              size: file.size,
              file: file
            };
          } else {
            // Folder
            if (!current[part]) {
              current[part] = { type: 'folder', children: {} };
            }
            current = current[part].children || current[part];
          }
        });
      });
      
      return tree;
    }
    
    // Render tree structure
    function renderTree(tree, depth = 0) {
      treeDisplay.innerHTML = '';
      const html = generateTreeHTML(tree, depth);
      treeDisplay.innerHTML = html;
    }
    
    // Generate HTML for tree
    function generateTreeHTML(tree, depth) {
      let html = '';
      
      Object.entries(tree).forEach(([name, node]) => {
        const indent = '  '.repeat(depth);
        
        if (node.type === 'folder') {
          html += `<div class="tree-item"><span class="folder">${indent}📁 ${name}/</span></div>`;
          html += generateTreeHTML(node.children, depth + 1);
        } else if (node.type === 'file') {
          html += `<div class="tree-item"><span class="file">${indent}📄 ${name} (${formatSize(node.size)})</span></div>`;
        }
      });
      
      return html;
    }
    
    // Format file size
    function formatSize(bytes) {
      if (bytes === 0) return '0 B';
      const k = 1024;
      const sizes = ['B', 'KB', 'MB', 'GB'];
      const i = Math.floor(Math.log(bytes) / Math.log(k));
      return (bytes / Math.pow(k, i)).toFixed(2) + ' ' + sizes[i];

      //   method: 'POST',
      //   body: formData
      // });
```

```html
    const captureBtn = document.getElementById('capture-btn');

    const corsStatus = document.getElementById('cors-status');

      corsStatus.textContent = 'anonymous';
      corsStatus.style.color = 'green';
    });

      if (video.error && video.error.code === MediaError.MEDIA_ERR_SRC_NOT_SUPPORTED) {

        corsStatus.style.color = 'red';
      }
    });

    captureBtn.addEventListener('click', () => {
      // Set canvas dimensions to match video
      canvas.width = video.videoWidth;
      canvas.height = video.videoHeight;
      
      try {
        // Draw current video frame to canvas
        ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
        
        // Convert to data URL
        const dataURL = canvas.toDataURL('image/png');
        
        // Display screenshot
        const img = document.createElement('img');
        img.src = dataURL;
        img.style.maxWidth = '640px';
        img.style.border = '2px solid #333';
        
        screenshotDisplay.innerHTML = '<h3>Screenshot:</h3>';
        screenshotDisplay.appendChild(img);

        // CORS error will throw SecurityError
        if (error.name === 'SecurityError') {
          alert('CORS error: Cannot capture screenshot from cross-origin video without proper CORS headers');
        } else {
          throw error;
        }
      }
    });
    
    // Process video frame (grayscale effect)
    processBtn.addEventListener('click', () => {
      canvas.width = video.videoWidth;
      canvas.height = video.videoHeight;
      
      try {
        // Draw frame
        ctx.drawImage(video, 0, 0);
        
        // Get image data
        const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
        const data = imageData.data;
        
        // Apply grayscale filter
        for (let i = 0; i < data.length; i += 4) {
          const avg = (data[i] + data[i + 1] + data[i + 2]) / 3;

          data[i + 1] = avg; // Green
          data[i + 2] = avg; // Blue
          // Alpha (data[i + 3]) stays the same
        }

        // Display processed frame
```

```html
    }
    
    // Expandable menu functionality
    document.querySelectorAll('.menu-button').forEach(button => {
      button.addEventListener('click', () => {
        const isExpanded = button.getAttribute('aria-expanded') === 'true';
        const submenu = document.getElementById(button.getAttribute('aria-controls'));
        
        // Close all other submenus
        document.querySelectorAll('.menu-button').forEach(otherBtn => {
          if (otherBtn !== button) {
            otherBtn.setAttribute('aria-expanded', 'false');
            document.getElementById(otherBtn.getAttribute('aria-controls'))
              .classList.remove('expanded');
          }
        });
        
        // Toggle current submenu
        button.setAttribute('aria-expanded', !isExpanded);
        submenu.classList.toggle('expanded');
      });
    });
    
    // Close menus on Escape key
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') {
        document.querySelectorAll('.menu-button[aria-expanded="true"]').forEach(btn => {
          btn.setAttribute('aria-expanded', 'false');
          document.getElementById(btn.getAttribute('aria-controls'))
            .classList.remove('expanded');
        });
      }
    });
  </script>
</body>
</html>
```

**Why:** Demonstrates proper use of `aria-expanded` attribute for accordion panels and expandable menus, including keyboard navigation and screen reader announcements.
