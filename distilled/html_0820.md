## html_form_action

```html
<!-- PROBLEM: Set where form data is sent on submit -->
<!-- TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Form Action Demo</title>
</head>
<body>
    <!-- 
        The action attribute specifies the URL where form data is sent.
        - Can be absolute URL: action="https://api.example.com/submit"
        - Can be relative URL: action="/api/submit"
        - Omitting action submits to the same page
    -->
    <form action="https://httpbin.org/post" method="POST">
        <label for="username">Username:</label>
        <input type="text" id="username" name="username" required>
        
        <label for="email">Email:</label>
        <input type="email" id="email" name="email" required>
        
        <button type="submit">Submit Form</button>
    </form>

    <script>
        // JavaScript can dynamically change form action
        const form = document.querySelector('form');
        
        // Change action based on user selection
        function updateFormAction(endpoint) {
            switch(endpoint) {
                case 'login':
                    form.action = '/api/login';
                    break;
                case 'register':
                    form.action = '/api/register';
                    break;
                case 'guest':
                    form.action = '/api/guest-checkout';
                    break;
            }
            console.log(`Form action set to: ${form.action}`);
        }

        // Validate before submission
        form.addEventListener('submit', function(e) {
            const email = document.getElementById('email').value;
            if (!email.includes('@')) {
                e.preventDefault(); // Stop form submission
                alert('Please enter a valid email');
            }
        });

        // Example: Conditional action based on form data
        function setConditionalAction(formData) {
            const isPremium = formData.get('account_type') === 'premium';
            form.action = isPremium ? '/api/premium-submit' : '/api/standard-submit';
        }
    </script>
</body>
</html>
```
**Why**: The action attribute is fundamental for form handling, determining where data is sent for server-side processing.

## html_form_target

```html
<!-- PROBLEM: Control where the form response is displayed -->
<!-- TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Form Target Demo</title>
</head>
<body>
    <!--
        The target attribute specifies where to display the response:
        - _self: Same window (default)
        - _blank: New window/tab
        - _parent: Parent frame
        - _top: Full body of the window
        - iframe_name: Named iframe
    -->
    
    <!-- Opens response in new tab -->
    <form action="/api/search" method="GET" target="_blank">
        <input type="search" name="query" placeholder="Search...">
        <button type="submit">Search (New Tab)</button>
    </form>

    <!-- Opens response in specific iframe -->
    <form action="/api/preview" method="POST" target="preview-frame">
        <textarea name="content" placeholder="Enter content..."></textarea>
        <button type="submit">Preview</button>
    </form>

    <!-- Iframe to display response -->
    <iframe name="preview-frame" width="100%" height="200"></iframe>

    <script>
        // Dynamically set target based on context
        function setFormTarget(form, userPreference) {
            const targets = {
                'newTab': '_blank',
                'samePage': '_self',
                'iframe': 'preview-frame'
            };
            form.target = targets[userPreference] || '_self';
        }

        // Handle form with dynamic target
        document.querySelectorAll('form').forEach(form => {
            form.addEventListener('submit', function(e) {
                const openInNewTab = document.getElementById('new-tab-checkbox')?.checked;
                if (openInNewTab) {
                    this.target = '_blank';
                }
            });
        });

        // Security note: _blank can be a security risk
        // Always add rel="noopener noreferrer" when opening new tabs via links
        // Forms don't have rel, but responses should include appropriate headers
    </script>
</body>
</html>
```
**Why**: Target controls response placement, essential for multi-window workflows and iframe-based previews.

```html
        // Toggle entire fieldset based on checkbox
        const digitalOnly = document.getElementById('digital-only');
        const shippingFieldset = document.getElementById('shipping-info');

        digitalOnly.addEventListener('change', function() {
            const inputs = shippingFieldset.querySelectorAll('input');
            inputs.forEach(input => {
                input.disabled = this.checked;
            });
            shippingFieldset.classList.toggle('disabled-section', this.checked);
        });

        // Disable submit during processing
        const submitBtn = document.getElementById('submit-btn');
        const form = document.getElementById('registration-form');

        form.addEventListener('submit', async function(e) {
            e.preventDefault();
            
            // Disable button to prevent double submission
            submitBtn.disabled = true;
            submitBtn.textContent = 'Processing...';

            try {
                const formData = new FormData(form);
                const response = await fetch('/api/register', {
                    method: 'POST',
                    body: formData
                });
                
                if (response.ok) {
                    alert('Registration successful!');
                }
            } catch (error) {
                console.error(error);
                alert('Registration failed');
            } finally {
                // Re-enable button
                submitBtn.disabled = false;
                submitBtn.textContent = 'Register';
            }
        });

        // Batch disable/enable elements
        function setFormDisabled(form, disabled) {
            const elements = form.querySelectorAll('input, button, select, textarea');
            elements.forEach(el => el.disabled = disabled);
        }

        // Check if any element is disabled
        function hasDisabledFields(form) {
            return Array.from(form.elements).some(el => el.disabled);
        }
    </script>
</body>
</html>
```
**Why**: Disabled states prevent unwanted input, show unavailable options, and prevent double submissions.

```html
<!-- PROBLEM: Stream video dynamically using MediaSource API -->
<!-- TIME: O(n) SPACE: O(buffer) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>MediaSource API Demo</title>
</head>
<body>
    <!--
        MediaSource API allows JavaScript to generate media streams

        Useful for:
        - Adaptive streaming (DASH, HLS)
        - Live streaming
        - Custom video processing
    -->
    
    <video id="video" controls width="640" height="360"></video>

    <script>
        // Check browser support
        if (!window.MediaSource) {
            document.getElementById('status').textContent = 
                'MediaSource API not supported';
        }

        const video = document.getElementById('video');
        const status = document.getElementById('status');

        // Create MediaSource object
        const mediaSource = new MediaSource();
        
        // Create object URL for video element
        video.src = URL.createObjectURL(mediaSource);

        // Wait for sourceopen event
        mediaSource.addEventListener('sourceopen', async function() {
            status.textContent = 'Status: MediaSource opened';
            
            // Create SourceBuffer with appropriate MIME type
            const mimeCodec = 'video/mp4; codecs="avc1.42E01E, mp4a.40.2"';
            
            // Check if codec is supported
            if (!MediaSource.isTypeSupported(mimeCodec)) {
                status.textContent = 'Unsupported MIME type or codec';
                return;
            }

            // Create source buffer
            const sourceBuffer = mediaSource.addSourceBuffer(mimeCodec);
            
            // Handle segment append
            sourceBuffer.addEventListener('updateend', function() {
                if (!sourceBuffer.updating && mediaSource.readyState === 'open') {
                    // Close stream when done
                    // mediaSource.endOfStream();
                }
            });

            // Simulate fetching and appending video segments
            async function fetchAndAppendSegment(segmentUrl) {
                try {
                    const response = await fetch(segmentUrl);
                    const arrayBuffer = await response.arrayBuffer();
                    
                    // Wait for buffer to be ready
                    await new Promise(resolve => {
                        if (!sourceBuffer.updating) {
                            resolve();
                        } else {
                            sourceBuffer.addEventListener('updateend', resolve, { once: true });
                        }
                    });
                    
                    sourceBuffer.appendBuffer(arrayBuffer);
                    status.textContent = 'Status: Segment appended';
                } catch (error) {
                    console.error('Error fetching segment:', error);
                }
            }

            // Example: Load multiple segments sequentially
            async function loadVideoSegments(segmentUrls) {
                for (const url of segmentUrls) {
                    await fetchAndAppendSegment(url);
                }
                mediaSource.endOfStream();
                status.textContent = 'Status: All segments loaded';
            }

            // Adaptive bitrate switching example
            function selectQuality(bandwidth) {
                if (bandwidth > 5000000) return 'high';
                if (bandwidth > 2000000) return 'medium';
                return 'low';
            }

            // Buffer management
            function monitorBuffer() {
                if (video.buffered.length > 0) {
                    const bufferedEnd = video.buffered.end(video.buffered.length - 1);
                    const bufferedSeconds = bufferedEnd - video.currentTime;
                    status.textContent = `Buffered: ${bufferedSeconds.toFixed(1)}s`;
                }
            }

            video.addEventListener('timeupdate', monitorBuffer);
        });

        // Cleanup on page unload
        window.addEventListener('unload', function() {
            URL.revokeObjectURL(video.src);
        });
    </script>

</html>
```

```html
        const textTracks = video.textTracks;
        
        // List all tracks
        for (let i = 0; i < textTracks.length; i++) {
            const track = textTracks[i];
            console.log(`Track ${i}: ${track.kind} - ${track.label}`);
        }

        // Get specific track by kind
        function getTrackByKind(kind) {
            for (let i = 0; i < textTracks.length; i++) {
                if (textTracks[i].kind === kind) {
                    return textTracks[i];
                }
            }
            return null;
        }

        // Build chapter navigation
        function buildChapterMenu() {
            const chapterTrack = getTrackByKind('chapters');
            if (!chapterTrack) return;

            chapterTrack.mode = 'hidden'; // Activate without showing
            
            chapterTrack.addEventListener('cuechange', function() {
                const activeCue = this.activeCues[0];
                if (activeCue) {
                    console.log(`Current chapter: ${activeCue.text}`);
                }
            });

            // Wait for cues to load
            chapterTrack.addEventListener('load', function() {
                const cues = chapterTrack.cues;
                chaptersDiv.innerHTML = '<h3>Chapters</h3>';
                
                for (let i = 0; i < cues.length; i++) {
                    const cue = cues[i];
                    const button = document.createElement('button');
                    button.textContent = cue.text;
                    button.onclick = () => {
                        video.currentTime = cue.startTime;
                    };
                    chaptersDiv.appendChild(button);
                }
            });
        }

        // Programmatic caption control
        function toggleCaptions(enabled) {
            const captionTrack = getTrackByKind('captions');
            if (captionTrack) {
                captionTrack.mode = enabled ? 'showing' : 'hidden';
            }
        }

        // Get current caption text
        function getCurrentCaption() {
            const captionTrack = getTrackByKind('captions');
            if (captionTrack && captionTrack.activeCues.length > 0) {
                return captionTrack.activeCues[0].text;
            }
            return null;
        }

        // Add cue dynamically (for live captions)
        function addLiveCaption(text, duration = 3) {
            const track = getTrackByKind('captions');

            const cue = new VTTCue(
                video.currentTime,
                video.currentTime + duration,

Today we'll learn about HTML5
`;

        video.addEventListener('loadedmetadata', buildChapterMenu);
    </script>
</body>
</html>
```

```html
<!-- PROBLEM: Control horizontal spacing between characters -->
<!-- TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Letter Spacing Demo</title>
    <style>
        /* Basic letter-spacing values */
        .normal { letter-spacing: normal; }      /* Default */
        .tight { letter-spacing: -0.05em; }      /* Tighter */
        .loose { letter-spacing: 0.2em; }        /* Looser */
        .custom { letter-spacing: 3px; }         /* Fixed pixels */

        /* Accessibility: respect user preferences */
        @media (prefers-reduced-motion: reduce) {
            .animated { letter-spacing: normal; }
        }

        /* Animated letter spacing */
        .animated {
            letter-spacing: 0;
            transition: letter-spacing 0.3s ease;
        }
        .animated:hover {
            letter-spacing: 0.3em;
        }

        /* Responsive typography */
        .headline {
            letter-spacing: clamp(0.02em, 0.5vw, 0.1em);
        }

        /* Common design patterns */
        .logo-text {
            letter-spacing: 0.5em;
            text-transform: uppercase;
        }

        .button-text {
            letter-spacing: 0.1em;
            text-transform: uppercase;
        }

        .caption {
            letter-spacing: 0.05em;
            font-size: 0.875em;
        }

        /* Negative tracking for display type */
        .display {
            letter-spacing: -0.02em;
            font-weight: 700;
        }
    </style>
</head>
<body>
    <p class="normal">Normal letter spacing</p>
    <p class="tight">Tight letter spacing</p>
    <p class="loose">Loose letter spacing</p>
    <p class="animated">Hover me for animation</p>
    <h1 class="headline">Responsive Headline</h1>

    <script>
        // Dynamic letter spacing based on text length
        function adjustLetterSpacing(element, minSpacing, maxSpacing, maxLength) {
            const textLength = element.textContent.length;
            const ratio = Math.min(textLength / maxLength, 1);
            const spacing = maxSpacing - (ratio * (maxSpacing - minSpacing));
            element.style.letterSpacing = `${spacing}em`;
        }

        // Apply to headlines
        document.querySelectorAll('.headline').forEach(el => {
            adjustLetterSpacing(el, 0.02, 0.15, 50);
        });

        // Letter spacing for readability optimization
        function optimizeReadability(container) {
            const width = container.offsetWidth;
            const fontSize = parseFloat(getComputedStyle(container).fontSize);
            
            // More spacing for smaller text
            if (fontSize < 14) {
                container.style.letterSpacing = '0.02em';
            } else if (fontSize > 24) {
                container.style.letterSpacing = '-0.01em';
            }
        }

        // Animate letter spacing with JavaScript
        function animateLetterSpacing(element, from, to, duration) {
            const start = performance.now();
            
            function update(currentTime) {
                const elapsed = currentTime - start;
                const progress = Math.min(elapsed / duration, 1);
                const eased = 1 - Math.pow(1 - progress, 3); // Ease out cubic
                
                const spacing = from + (to - from) * eased;
                element.style.letterSpacing = `${spacing}em`;
                
                if (progress < 1) {
                    requestAnimationFrame(update);
                }
            }
            
            requestAnimationFrame(update);
        }

        // Example usage
        const animatedEl = document.querySelector('.animated');
        animatedEl.addEventListener('mouseenter', () => {
            animateLetterSpacing(animatedEl, 0, 0.3, 300);
        });
        animatedEl.addEventListener('mouseleave', () => {
            animateLetterSpacing(animatedEl, 0.3, 0, 300);
        });
    </script>
</body>
</html>
```
**Why**: Letter spacing enhances readability, creates visual hierarchy, and enables typographic design patterns like logo styling.

```html
<!-- PROBLEM: Determine the type of a value at runtime -->
<!-- TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Typeof Operator Demo</title>
</head>
<body>
    <script>
        // Basic typeof usage
        console.log(typeof undefined);    // "undefined"
        console.log(typeof null);         // "object" ⚠️ Historical bug
        console.log(typeof true);         // "boolean"
        console.log(typeof 42);           // "number"
        console.log(typeof 'hello');      // "string"
        console.log(typeof Symbol());     // "symbol"
        console.log(typeof {});           // "object"
        console.log(typeof []);           // "object" ⚠️ Arrays are objects
        console.log(typeof function(){}); // "function"
        console.log(typeof 42n);          // "bigint"

        // Comprehensive type checking function
        function getType(value) {
            // Handle null explicitly (typeof bug workaround)
            if (value === null) return 'null';
            
            // Handle arrays
            if (Array.isArray(value)) return 'array';
            
            // Handle NaN
            if (typeof value === 'number' && isNaN(value)) return 'NaN';
            
            // Handle typed arrays
            if (ArrayBuffer.isView(value)) {
                return value.constructor.name.toLowerCase();
            }
            
            // Handle Date, RegExp, Map, Set, etc.
            if (value instanceof Date) return 'date';
            if (value instanceof RegExp) return 'regexp';
            if (value instanceof Map) return 'map';
            if (value instanceof Set) return 'set';
            if (value instanceof Promise) return 'promise';
            
            // Handle custom classes
            if (value?.constructor?.name && value.constructor.name !== 'Object') {
                return value.constructor.name.toLowerCase();
            }
            
            return typeof value;
        }

        // Usage examples
        console.log(getType(null));           // "null"
        console.log(getType([1, 2, 3]));      // "array"
        console.log(getType(new Date()));     // "date"
        console.log(getType(NaN));            // "NaN"
        console.log(getType(new Map()));      // "map"

        // Type guards using typeof
        function processValue(value) {
            if (typeof value === 'string') {
                return value.toUpperCase();
            }
            if (typeof value === 'number') {
                return value * 2;
            }
            if (typeof value === 'boolean') {
                return !value;
            }
            if (typeof value === 'function') {
                return value();
            }
            if (Array.isArray(value)) {
                return value.map(processValue);
            }
            return String(value);
        }

        // Safe property access with typeof
        function safeStringLength(value) {
            if (typeof value !== 'string') {
                return 0;
            }
            return value.length;
        }

        // Validate function parameters
        function validateConfig(config) {
            if (typeof config !== 'object' || config === null) {
                throw new TypeError('Config must be an object');
            }
            if (typeof config.url !== 'string') {
                throw new TypeError('config.url must be a string');
            }
            if (typeof config.timeout !== 'number') {
                config.timeout = 5000; // Default
            }
            return config;
        }

        // Detect primitive vs reference types
        function isPrimitive(value) {
            return value === null || (typeof value !== 'object' && typeof value !== 'function');
        }

        // Check if value is numeric
        function isNumeric(value) {
            return typeof value === 'number' && !isNaN(value) && isFinite(value);
        }

        // Check if value is a valid key (string or symbol)
        function isValidKey(value) {
            return typeof value === 'string' || typeof value === 'symbol';
        }

        // Runtime type checking decorator (concept)
        function typed(paramTypes) {
            return function(target, key, descriptor) {
                const original = descriptor.value;
                descriptor.value = function(...args) {
                    args.forEach((arg, i) => {
                        const expected = paramTypes[i];
                        if (typeof arg !== expected) {
                            throw new TypeError(
                                `Parameter ${i} expected ${expected}, got ${typeof arg}`
                            );
                        }
                    });
                    return original.apply(this, args);
                };
            };
        }
    </script>
```

**Why**: Typeof enables runtime type checking, essential for validation, type guards, and defensive programming in dynamic JavaScript.
