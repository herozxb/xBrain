    /* PROBLEM: Iterate over iterable objects cleanly
       TIME: O(n) SPACE: O(1) */
    
    const log = (msg) => {
      document.getElementById('output').textContent += msg + '\n';
    };
    
    // ============================================
    // Basic array iteration
    // ============================================
    
    const fruits = ['apple', 'banana', 'cherry'];
    
    log('--- Basic Array ---');
    for (const fruit of fruits) {

    }

    // ============================================
    // String iteration (characters)
    // ============================================
    
    log('\n--- String Characters ---');

    }

    // Map iteration

    log('\n--- Map Entries ---');
    const userMap = new Map([

    }
    
    // ============================================
    // Set iteration
    // ============================================
    
    log('\n--- Set Values ---');
    const uniqueNumbers = new Set([1, 2, 3, 2, 1]);

      log(num);  // 1, 2, 3 (no duplicates)

    // NodeList iteration (DOM)
    // ============================================
    
    log('\n--- NodeList (simulated) ---');

    // ============================================
    
    log('\n--- Generator ---');

      yield 1;
      yield 2;
      yield 3;

    for (const num of numberGenerator()) {
      log(`Generated: ${num}`);
    }
    
    // ============================================
    // With entries() for index + value
    // ============================================
    
    log('\n--- With Index (entries) ---');

    log('\n--- Break Example ---');

    for (const n of numbers) {
      if (n > 5) break;  // Stop at 6

    async function* asyncNumbers() {
      yield Promise.resolve(1);
      yield Promise.resolve(2);
      yield Promise.resolve(3);

    (async () => {
      for await (const num of asyncNumbers()) {
        log(`Async: ${num}`);
      }
    })();
    
    // ============================================
    // Custom iterable object
    // ============================================
    
    log('\n--- Custom Iterable ---');
    const range = {
      start: 1,
      end: 5,
      [Symbol.iterator]() {
        let current = this.start;
        const end = this.end;
        return {
          next() {
            if (current <= end) {
              return { value: current++, done: false };
            }
            return { done: true };
          }
        };
      }
    };
    
    for (const num of range) {
      log(`Range: ${num}`);
    }
    // 1, 2, 3, 4, 5
    
    // ============================================
    // Comparison: for-of vs for-in vs forEach
    // ============================================
    log('\n--- Comparison ---');
    log('for-of: Values from iterables (arrays, strings, maps, sets)');
    log('for-in: Enumerable property keys (object properties)');
    log('forEach: Array method, callback-based, cannot break');
    
    // ============================================
    // Summary
    // ============================================
    log('\n--- Key Points ---');
    log('✓ Works with any iterable (has Symbol.iterator)');
    log('✓ Clean syntax, no index management');
    log('✓ Can use break, continue, return');
    log('✓ Works with destructuring');
    log('✓ Supports async with for-await-of');
  </script>
</body>
</html>
```
**Why**: For-of provides clean, readable iteration over any iterable. Superior to for-in (which iterates keys) and forEach (which can't break). Works universally with arrays, strings, Maps, Sets, and custom iterables.

      return acc.concat(arr);
    }, []);
    log(`Flattened: ${flattened}`);  // [5, 6, 3, 4, 1, 2]
    
    // ============================================
    // Build string from right
    // ============================================
    
    log('\n--- Build String ---');
    const chars = ['H', 'e', 'l', 'l', 'o'];
    
    const reversed = chars.reduceRight((acc, char) => acc + char, '');
    log(`Reversed: ${reversed}`);  // "olleH"
    
    // ============================================
    // Compose functions (right to left)
    // ============================================
    
    log('\n--- Function Composition ---');
    const double = x => x * 2;
    const addOne = x => x + 1;
    const square = x => x * x;
    
    const functions = [double, addOne, square];
    
    // Compose: square(addOne(double(x)))
    const composed = functions.reduceRight((acc, fn) => {
      return x => fn(acc(x));
    }, x => x);
    
    log(`Composed(3): ${composed(3)}`);
    // double(3)=6 -> addOne(6)=7 -> square(7)=49
    
    // ============================================
    // Object property chain
    // ============================================
    
    log('\n--- Property Chain ---');
    const user = {
      profile: {
        settings: {
          theme: 'dark'
        }
      }
    };
    
    const path = ['profile', 'settings', 'theme'];
    
    const value = path.reduceRight((acc, key) => {
      return { [key]: acc };
    }, user.profile.settings.theme);
    
    log(`Rebuilt object: ${JSON.stringify(value)}`);
    // {"profile":{"settings":{"theme":"dark"}}}
    
    // ============================================
    // Calculate difference series
    // ============================================
    
    log('\n--- Difference Series ---');
    const values = [100, 20, 10, 5];
    
    // Calculate: 100 - 20 - 10 - 5 from right
    // = 100 - (20 - (10 - 5))
    // = 100 - (20 - 5)
    // = 100 - 15 = 85
    const diffFromRight = values.reduceRight((acc, val) => val - acc);
    log(`Right difference: ${diffFromRight}`);  // 85
    
    // ============================================
    // Practical: Build CSS selector
    // ============================================
    
    log('\n--- CSS Selector Builder ---');
    const elements = ['button', 'div', 'section', 'main'];
    
    const selector = elements.reduceRight((acc, el) => {
      return `${el} > ${acc}`;
    }, '*');
    
    log(`Selector: ${selector}`);
    // "main > section > div > button > *"
    
    // ============================================
    // Practical: Parse right-associative expression
    // ============================================
    
    log('\n--- Right-Associative Math ---');
    // Exponentiation is right-associative: 2^3^2 = 2^(3^2) = 2^9 = 512
    const bases = [2, 3, 2];
    
    const power = bases.reduceRight((acc, base) => {
      return Math.pow(base, acc);
    });
    log(`2^3^2 (right-associative): ${power}`);  // 512
    
    // ============================================
    // Practical: Dependency resolution order
    // ============================================
    
    log('\n--- Dependency Resolution ---');
    const modules = [
      { name: 'app', deps: ['router'] },
      { name: 'router', deps: ['utils'] },
      { name: 'utils', deps: [] }
    ];
    
    const loadOrder = modules.reduceRight((acc, mod) => {
      if (!acc.includes(mod.name)) {
        acc.unshift(mod.name);
      }
      mod.deps.forEach(dep => {
        if (!acc.includes(dep)) {
          acc.unshift(dep);
        }
      });
      return acc;
    }, []);
    
    log(`Load order: ${loadOrder.join(' -> ')}`);
    // utils -> router -> app
    
    // ============================================
    // With initial value
    // ============================================
    
    log('\n--- With Initial Value ---');
    const items = [1, 2, 3];
    
    const withInitial = items.reduceRight((acc, item) => {
      acc.push(item * 10);
      return acc;
    }, [0]);  // Start with [0]
    
    log(`With initial [0]: ${withInitial}`);  // [0, 30, 20, 10]
    
    // ============================================
    // Summary
    // ============================================
    log('\n--- Key Points ---');
    log('✓ Iterates right-to-left (last index to first)');
    log('✓ Same signature as reduce()');
    log('✓ Useful for right-associative operations');
    log('✓ Great for function composition');
    log('✓ Reverses natural left-to-right reduce order');
  </script>

</html>
```
**Why**: ReduceRight processes arrays from end to start, essential for right-associative operations like function composition and exponentiation. Complements reduce() for bidirectional data processing.
