```python
# PROBLEM: Divide a simple polygon into triangles using ear clipping method
# TIME: O(n²) SPACE: O(n)
def polygon_triangulation(vertices):
    """
    Triangulate a simple polygon using the ear clipping algorithm.
    An ear is a triangle formed by three consecutive vertices where
    the triangle is entirely inside the polygon.
    
    Args:
        vertices: List of (x, y) tuples representing polygon vertices in order
    
    Returns:
        List of triangles, each triangle is a tuple of 3 vertex indices
    """
    n = len(vertices)
    if n < 3:
        return []
    if n == 3:
        return [(0, 1, 2)]
    
    # Create a doubly linked list of vertex indices
    indices = list(range(n))
    triangles = []
    
    def cross(o, a, b):
        """Calculate cross product of vectors OA and OB."""
        return (vertices[a][0] - vertices[o][0]) * (vertices[b][1] - vertices[o][1]) - \
               (vertices[a][1] - vertices[o][1]) * (vertices[b][0] - vertices[o][0])
    
    def is_ear(i, indices):
        """Check if vertex at index i forms an ear."""
        n = len(indices)
        if n < 3:
            return False
        
        prev_idx = indices[(i - 1) % n]
        curr_idx = indices[i]
        next_idx = indices[(i + 1) % n]
        
        # Check if the angle is convex (left turn)
        if cross(prev_idx, curr_idx, next_idx) <= 0:
            return False
        
        # Check if any other vertex is inside the triangle
        for j in range(n):
            if j in [(i - 1) % n, i, (i + 1) % n]:
                continue
            
            p = indices[j]
            # Check if point p is inside triangle (prev, curr, next)
            d1 = cross(prev_idx, curr_idx, p)
            d2 = cross(curr_idx, next_idx, p)
            d3 = cross(next_idx, prev_idx, p)
            
            has_neg = (d1 < 0) or (d2 < 0) or (d3 < 0)
            has_pos = (d1 > 0) or (d2 > 0) or (d3 > 0)
            
            if not (has_neg and has_pos):
                return False
        
        return True
    
    while len(indices) > 3:
        ear_found = False
        for i in range(len(indices)):
            if is_ear(i, indices):
                # Add triangle
                prev_idx = indices[(i - 1) % len(indices)]
                curr_idx = indices[i]
                next_idx = indices[(i + 1) % len(indices)]
                triangles.append((prev_idx, curr_idx, next_idx))
                
                # Remove the ear vertex
                indices.pop(i)
                ear_found = True
                break
        
        if not ear_found:
            # Fallback: polygon may be non-simple
            break
    
    # Add the final triangle
    if len(indices) == 3:
        triangles.append((indices[0], indices[1], indices[2]))
    
    return triangles
```
**Why**: Fundamental computational geometry algorithm used in computer graphics, game development, and mesh generation for decomposing complex polygons into renderable triangles.

```python
# PROBLEM: Find the maximum independent set in a graph (NP-hard, greedy approximation)
# TIME: O(V + E) SPACE: O(V)
def maximum_independent_set(graph):
    """
    Find an approximate maximum independent set using a greedy approach.
    An independent set is a set of vertices where no two are adjacent.
    Greedy strategy: repeatedly pick the vertex with minimum degree.
    
    Args:
        graph: Dictionary mapping vertex to set of adjacent vertices
    
    Returns:
        Set of vertices forming an independent set
    """
    import heapq
    
    # Create a copy of the graph
    remaining = {v: set(neighbors) for v, neighbors in graph.items()}
    independent_set = set()
    
    # Priority queue: (degree, vertex)
    # Using degree as priority - pick vertices with fewer constraints first
    heap = [(len(neighbors), v) for v, neighbors in remaining.items()]
    heapq.heapify(heap)
    
    in_heap = set(remaining.keys())
    
    while heap and remaining:
        degree, vertex = heapq.heappop(heap)
        
        # Skip if vertex was already processed or removed
        if vertex not in in_heap or vertex not in remaining:
            continue
        
        # Add vertex to independent set
        independent_set.add(vertex)
        
        # Remove vertex and all its neighbors from consideration
        neighbors_to_remove = list(remaining[vertex])
        
        for neighbor in neighbors_to_remove:
            if neighbor in remaining:
                in_heap.discard(neighbor)
                del remaining[neighbor]
        
        # Remove the selected vertex
        in_heap.discard(vertex)
        del remaining[vertex]
        
        # Update degrees and rebuild heap
        if remaining:
            heap = [(len(neighbors), v) for v, neighbors in remaining.items()]
            heapq.heapify(heap)
    
    return independent_set
```
**Why**: Critical optimization problem in scheduling, resource allocation, wireless networks, and coding theory. The greedy approximation provides practical solutions for this NP-hard problem.

```python
# PROBLEM: Build and evaluate expression trees from postfix notation
# TIME: O(n) for build, O(n) for evaluate SPACE: O(n)
class ExpressionTreeNode:
    """Node in an expression tree."""
    
    def __init__(self, value, left=None, right=None):
        self.value = value
        self.left = left
        self.right = right
    
    def is_operator(self):
        """Check if this node is an operator."""
        return self.value in {'+', '-', '*', '/', '^', '%'}
    
    def evaluate(self):
        """Recursively evaluate the expression tree."""
        if not self.is_operator():
            # It's an operand (number or variable)
            try:
                return float(self.value)
            except ValueError:
                raise ValueError(f"Cannot evaluate variable: {self.value}")
        
        # It's an operator - evaluate children first
        left_val = self.left.evaluate() if self.left else 0
        right_val = self.right.evaluate() if self.right else 0
        
        ops = {
            '+': lambda a, b: a + b,
            '-': lambda a, b: a - b,
            '*': lambda a, b: a * b,
            '/': lambda a, b: a / b if b != 0 else float('inf'),
            '^': lambda a, b: a ** b,
            '%': lambda a, b: a % b,
        }
        
        return ops[self.value](left_val, right_val)
    
    def to_infix(self):
        """Convert expression tree to infix notation string."""
        if not self.is_operator():
            return str(self.value)
        
        left_str = self.left.to_infix() if self.left else ""
        right_str = self.right.to_infix() if self.right else ""
        
        return f"({left_str} {self.value} {right_str})"
    
    def to_prefix(self):
        """Convert expression tree to prefix (Polish) notation."""
        if not self.is_operator():
            return str(self.value)
        
        left_str = self.left.to_prefix() if self.left else ""
        right_str = self.right.to_prefix() if self.right else ""
        
        return f"{self.value} {left_str} {right_str}".strip()


def build_expression_tree(postfix_tokens):
    """
    Build an expression tree from postfix (Reverse Polish) notation.
    
    Args:
        postfix_tokens: List of tokens in postfix order
    
    Returns:
        Root node of the expression tree
    """
    stack = []
    operators = {'+', '-', '*', '/', '^', '%'}
    
    for token in postfix_tokens:
        if token not in operators:
            # Operand - create a leaf node
            stack.append(ExpressionTreeNode(token))
        else:
            # Operator - pop two operands and create internal node
            right = stack.pop() if stack else None
            left = stack.pop() if stack else None
            stack.append(ExpressionTreeNode(token, left, right))
    
    return stack[0] if stack else None


def infix_to_postfix(infix_tokens):
    """
    Convert infix notation to postfix using Shunting-yard algorithm.
    
    Args:
        infix_tokens: List of tokens in infix order (with parentheses)
    
    Returns:
        List of tokens in postfix order
    """
    precedence = {'+': 1, '-': 1, '*': 2, '/': 2, '^': 3, '%': 2}
    associativity = {'+': 'L', '-': 'L', '*': 'L', '/': 'L', '^': 'R', '%': 'L'}
    
    output = []
    operator_stack = []
    operators = set(precedence.keys())
    
    for token in infix_tokens:
        if token == '(':
            operator_stack.append(token)
        elif token == ')':
            # Pop until matching '('
            while operator_stack and operator_stack[-1] != '(':
                output.append(operator_stack.pop())
            operator_stack.pop()  # Remove the '('
        elif token in operators:
            while (operator_stack and 
                   operator_stack[-1] != '(' and
                   (precedence[operator_stack[-1]] > precedence[token] or
                    (precedence[operator_stack[-1]] == precedence[token] and
                     associativity[token] == 'L'))):
                output.append(operator_stack.pop())
            operator_stack.append(token)
        else:
            # Operand
            output.append(token)
    
    while operator_stack:
        output.append(operator_stack.pop())
    
    return output
```
**Why**: Essential for compilers, calculators, and symbolic computation systems. Enables efficient expression evaluation, manipulation, and conversion between notations.

```python
# PROBLEM: Implement circular buffer with overwriting and blocking variants
# TIME: O(1) for all operations SPACE: O(n)
class RingBuffer:
    """Basic ring buffer (circular queue) without overwriting."""
    
    def __init__(self, capacity):
        self.capacity = capacity
        self.buffer = [None] * capacity
        self.head = 0  # Read position
        self.tail = 0  # Write position
        self.size = 0
    
    def is_empty(self):
        """Check if buffer is empty."""
        return self.size == 0
    
    def is_full(self):
        """Check if buffer is full."""
        return self.size == self.capacity
    
    def enqueue(self, item):
        """Add item to buffer. Returns False if full."""
        if self.is_full():
            return False
        self.buffer[self.tail] = item
        self.tail = (self.tail + 1) % self.capacity
        self.size += 1
        return True
    
    def dequeue(self):
        """Remove and return oldest item. Returns None if empty."""
        if self.is_empty():
            return None
        item = self.buffer[self.head]
        self.head = (self.head + 1) % self.capacity
        self.size -= 1
        return item
    
    def peek(self):
        """Return oldest item without removing."""
        if self.is_empty():
            return None
        return self.buffer[self.head]


class OverwritingRingBuffer:
    """Ring buffer that overwrites oldest data when full."""
    
    def __init__(self, capacity):
        self.capacity = capacity
        self.buffer = [None] * capacity
        self.head = 0
        self.tail = 0
        self.size = 0
        self.overwrite_count = 0
    
    def write(self, item):
        """Write item, overwriting oldest if necessary."""
        if self.is_full():
            # Overwrite: advance head
            self.head = (self.head + 1) % self.capacity
            self.overwrite_count += 1
        else:
            self.size += 1
        
        self.buffer[self.tail] = item
        self.tail = (self.tail + 1) % self.capacity
    
    def read(self):
        """Read and remove oldest item."""
        if self.is_empty():
            raise IndexError("Buffer is empty")
        item = self.buffer[self.head]
        self.head = (self.head + 1) % self.capacity
        self.size -= 1
        return item
    
    def is_empty(self):
        return self.size == 0
    
    def is_full(self):
        return self.size == self.capacity
    
    def get_all(self):
        """Get all items in order (oldest to newest)."""
        items = []
        for i in range(self.size):
            idx = (self.head + i) % self.capacity
            items.append(self.buffer[idx])
        return items


class ThreadSafeRingBuffer:
    """Thread-safe ring buffer with blocking operations."""
    
    def __init__(self, capacity):
        import threading
        self.capacity = capacity
        self.buffer = [None] * capacity
        self.head = 0
        self.tail = 0
        self.size = 0
        self.lock = threading.Lock()
        self.not_empty = threading.Condition(self.lock)
        self.not_full = threading.Condition(self.lock)
    
    def put(self, item, block=True, timeout=None):
        """Add item, blocking if full."""
        with self.not_full:
            if not block and self.size == self.capacity:
                raise IndexError("Buffer is full")
            
            # Wait until not full
            while self.size == self.capacity:
                if not self.not_full.wait(timeout):
                    raise TimeoutError("Timeout waiting for space")
            
            self.buffer[self.tail] = item
            self.tail = (self.tail + 1) % self.capacity
            self.size += 1
            self.not_empty.notify()
    
    def get(self, block=True, timeout=None):
        """Remove and return item, blocking if empty."""
        with self.not_empty:
            if not block and self.size == 0:
                raise IndexError("Buffer is empty")
            
            # Wait until not empty
            while self.size == 0:
                if not self.not_empty.wait(timeout):
                    raise TimeoutError("Timeout waiting for item")
            
            item = self.buffer[self.head]
            self.head = (self.head + 1) % self.capacity
            self.size -= 1
            self.not_full.notify()
            return item
```
**Why**: Critical for streaming applications, audio/video processing, producer-consumer patterns, and implementing efficient FIFO queues with fixed memory footprint.

```python
# PROBLEM: Implement DDD Aggregate pattern with consistency boundaries
# TIME: O(1) for most operations SPACE: O(n)
from dataclasses import dataclass, field
from typing import List, Optional, Set
from datetime import datetime
import uuid


@dataclass
class DomainEvent:
    """Base class for domain events."""
    event_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    occurred_at: datetime = field(default_factory=datetime.utcnow)


@dataclass
class Entity:
    """Base class for entities with identity."""
    id: str
    
    def __eq__(self, other):
        if not isinstance(other, Entity):
            return False
        return self.id == other.id
    
    def __hash__(self):
        return hash(self.id)


@dataclass
class OrderItem(Entity):
    """Entity within Order aggregate."""
    product_id: str
    product_name: str
    quantity: int
    unit_price: float
    
    def calculate_total(self):
        """Calculate line item total."""
        return self.quantity * self.unit_price
    
    def update_quantity(self, new_quantity: int):
        """Update quantity with validation."""
        if new_quantity <= 0:
            raise ValueError("Quantity must be positive")
        self.quantity = new_quantity


@dataclass
class OrderCreatedEvent(DomainEvent):
    """Event emitted when order is created."""
    order_id: str
    customer_id: str


@dataclass
class ItemAddedEvent(DomainEvent):
    """Event emitted when item is added to order."""
    order_id: str
    product_id: str
    quantity: int


@dataclass
class OrderSubmittedEvent(DomainEvent):
    """Event emitted when order is submitted."""
    order_id: str
    total_amount: float


class Order:
    """
    Order aggregate root - enforces consistency boundary.
    All modifications to OrderItems must go through this root.
    """
    
    def __init__(self, order_id: str, customer_id: str):
        self.id = order_id
        self.customer_id = customer_id
        self._items: List[OrderItem] = []
        self._status = "draft"
        self._events: List[DomainEvent] = []
        self._version = 0
        
        # Emit creation event
        self._add_event(OrderCreatedEvent(
            order_id=self.id,
            customer_id=customer_id
        ))
    
    def _add_event(self, event: DomainEvent):
        """Add domain event to pending events."""
        self._events.append(event)
    
    def pull_events(self) -> List[DomainEvent]:
        """Get and clear pending events."""
        events = self._events.copy()
        self._events.clear()
        return events
    
    @property
    def items(self) -> List[OrderItem]:
        """Read-only access to items."""
        return self._items.copy()
    
    @property
    def status(self) -> str:
        return self._status
    
    def add_item(self, product_id: str, product_name: str, 
                 quantity: int, unit_price: float) -> OrderItem:
        """Add item to order with invariants enforcement."""
        if self._status != "draft":
            raise InvalidOperationException("Cannot modify submitted order")
        
        # Check if item already exists
        for item in self._items:
            if item.product_id == product_id:
                # Merge with existing item
                item.update_quantity(item.quantity + quantity)
                return item
        
        # Create new item
        item = OrderItem(
            id=str(uuid.uuid4()),
            product_id=product_id,
            product_name=product_name,
            quantity=quantity,
            unit_price=unit_price
        )
        self._items.append(item)
        self._version += 1
        
        self._add_event(ItemAddedEvent(
            order_id=self.id,
            product_id=product_id,
            quantity=quantity
        ))
        
        return item
    
    def remove_item(self, item_id: str):
        """Remove item from order."""
        if self._status != "draft":
            raise InvalidOperationException("Cannot modify submitted order")
        
        self._items = [i for i in self._items if i.id != item_id]
        self._version += 1
    
    def calculate_total(self) -> float:
        """Calculate order total."""
        return sum(item.calculate_total() for item in self._items)
    
    def submit(self):
        """Submit the order - transition to final state."""
        if self._status != "draft":
            raise InvalidOperationException("Order already submitted")
        
        if not self._items:
            raise InvalidOperationException("Cannot submit empty order")
        
        total = self.calculate_total()
        self._status = "submitted"
        self._version += 1
        
        self._add_event(OrderSubmittedEvent(
            order_id=self.id,
            total_amount=total
        ))


class InvalidOperationException(Exception):
    """Exception for invalid operations on aggregate."""
    pass
```
**Why**: Domain-Driven Design pattern that ensures consistency boundaries in complex domains. Aggregates group related entities and enforce business rules through a single entry point (the aggregate root).

```python
# PROBLEM: Implement immutable Value Objects with equality based on attributes

from dataclasses import dataclass, field
from typing import Any, Tuple
import math


@dataclass(frozen=True)
class ValueObject:
    """
    Base class for Value Objects.
    Value objects are immutable and equality is based on all attributes.
    """
    
    def __eq__(self, other):
        if not isinstance(other, self.__class__):
            return False
        return self._get_equality_components() == other._get_equality_components()
    
    def __hash__(self):
        return hash(self._get_equality_components())
    
    def _get_equality_components(self) -> Tuple[Any, ...]:
        """Get tuple of all attributes for equality comparison."""
        return tuple(getattr(self, f.name) for f in self.__dataclass_fields__.values())
    
    def __str__(self):
        components = self._get_equality_components()
        if len(components) == 1:
            return str(components[0])
        return f"{self.__class__.__name__}{components}"


@dataclass(frozen=True)
class Money(ValueObject):
    """
    Money value object with currency.
    Immutable and handles rounding for financial precision.
    """
    amount: float
    currency: str
    
    def __post_init__(self):
        # Round to 2 decimal places for currency
        object.__setattr__(self, 'amount', round(self.amount, 2))
        
        if self.amount < 0:
            raise ValueError("Money amount cannot be negative")
        if not self.currency or len(self.currency) != 3:
            raise ValueError("Currency must be 3-letter code")
    
    def add(self, other: 'Money') -> 'Money':
        """Add two money values of same currency."""
        if self.currency != other.currency:
            raise ValueError(f"Cannot add {other.currency} to {self.currency}")
        return Money(self.amount + other.amount, self.currency)
    
    def subtract(self, other: 'Money') -> 'Money':
        """Subtract two money values of same currency."""
        if self.currency != other.currency:
            raise ValueError(f"Cannot subtract {other.currency} from {self.currency}")
        return Money(self.amount - other.amount, self.currency)
    
    def multiply(self, factor: float) -> 'Money':
        """Multiply money by a factor."""
        return Money(self.amount * factor, self.currency)
    
    def is_greater_than(self, other: 'Money') -> bool:
        """Compare two money values."""
        if self.currency != other.currency:
            raise ValueError(f"Cannot compare {other.currency} with {self.currency}")
        return self.amount > other.amount

class Coordinates(ValueObject):
    """Geographic coordinates value object."""
    latitude: float
    longitude: float
    
    def __post_init__(self):
        if not -90 <= self.latitude <= 90:
            raise ValueError("Latitude must be between -90 and 90")
        if not -180 <= self.longitude <= 180:
            raise ValueError("Longitude must be between -180 and 180")
    
    def distance_to(self, other: 'Coordinates') -> float:
        """
        Calculate distance using Haversine formula.
        Returns distance in kilometers.
        """
        R = 6371  # Earth's radius in km
        
        lat1, lon1 = math.radians(self.latitude), math.radians(self.longitude)
        lat2, lon2 = math.radians(other.latitude), math.radians(other.longitude)
        
        dlat = lat2 - lat1
        dlon = lon2 - lon1
        
        a = math.sin(dlat/2)**2 + math.cos(lat1) * math.cos(lat2) * math.sin(dlon/2)**2
        c = 2 * math.asin(math.sqrt(a))
        
        return R * c
```

```python
class CommandHandler:
    """Dispatches commands to appropriate handlers."""
    
    def __init__(self):

        self._handlers[command_type] = handler

        """Register a handler for a query type."""

        """Dispatch query to registered handler."""
        handler = self._handlers.get(type(query))

            raise ValueError(f"No handler registered for {type(query)}")
        return handler(query)

        self.query_handler = QueryHandler()

            product = ProductAggregate(
                product_id=str(uuid.uuid4()),
                name=cmd.name,
                price=cmd.price,
                stock=cmd.stock
            )
            self.write_repository[product.id] = product
            
            # Update read model
            read_model = ProductReadModel(
                product_id=product.id,
                name=product.name,
                price=product.price,
                available_stock=product.stock,
                last_updated=datetime.utcnow()
            )
            self.read_repository.save(read_model)
            return product.id
        
        def handle_update_price(cmd: UpdatePriceCommand):
            product = self.write_repository[cmd.product_id]
            product.update_price(cmd.new_price)
            
            # Update read model
            read_model = self.read_repository.get(cmd.product_id)
            if read_model:
                read_model.price = cmd.new_price
                read_model.last_updated = datetime.utcnow()
                self.read_repository.save(read_model)
        
        def handle_reserve_stock(cmd: ReserveStockCommand):
            product = self.write_repository[cmd.product_id]
            success = product.reserve_stock(cmd.quantity, cmd.order_id)
            
            if success:
                # Update read model
                read_model = self.read_repository.get(cmd.product_id)
                if read_model:
                    read_model.available_stock -= cmd.quantity
                    self.read_repository.save(read_model)
            
            return success
        
        self.command_handler.register(CreateProductCommand, handle_create)
        self.command_handler.register(UpdatePriceCommand, handle_update_price)
        self.command_handler.register(ReserveStockCommand, handle_reserve_stock)
        
        # Query handlers
        self.query_handler.register(
            GetProductQuery,
            lambda q: self.read_repository.get(q.product_id)
        )
        self.query_handler.register(
            ListProductsQuery,
            lambda q: self.read_repository.list(
                q.min_price, q.max_price, q.in_stock_only
            )
        )
    
    def send_command(self, command: Command) -> Any:
        """Send a command to the write side."""
        return self.command_handler.dispatch(command)
    
    def send_query(self, query: Query) -> Any:
        """Send a query to the read side."""
        return self.query_handler.dispatch(query)
```
**Why**: CQRS separates read and write operations, allowing independent optimization. Read models can be denormalized for query performance while write models maintain domain integrity. Essential for high-scale systems with different read/write patterns.

## 8. Optimistic Locking

```python
# PROBLEM: Implement optimistic locking for concurrent data access
# TIME: O(1) for check SPACE: O(1)
from dataclasses import dataclass, field
from typing import Optional, Any, Callable
from datetime import datetime
import threading


@dataclass
class VersionedEntity:
    """Entity with version number for optimistic locking."""
    id: str
    data: Any
    version: int = 1
    last_modified: datetime = field(default_factory=datetime.utcnow)


class OptimisticLockException(Exception):
    """Exception raised when optimistic lock fails."""
    pass


class OptimisticLockRepository:
    """
    Repository with optimistic locking support.
    Detects concurrent modifications using version numbers.
    """
    
    def __init__(self):
        self._storage: dict[str, VersionedEntity] = {}
        self._lock = threading.Lock()  # For thread-safe operations
    
    def save(self, entity: VersionedEntity) -> VersionedEntity:
        """
        Save entity with optimistic lock check.
        Raises OptimisticLockException if version mismatch.
        """
        with self._lock:
            existing = self._storage.get(entity.id)
            
            if existing is not None:
                # Check version - concurrent modification detection
                if existing.version != entity.version:
                    raise OptimisticLockException(
                        f"Concurrent modification detected. "
                        f"Expected version {existing.version}, got {entity.version}"
                    )
            
            # Increment version and save
            entity.version = (entity.version or 0) + 1
            entity.last_modified = datetime.utcnow()
            self._storage[entity.id] = entity
            return entity
    
    def get(self, entity_id: str) -> Optional[VersionedEntity]:
        """Get entity by ID (returns a copy to prevent external modification)."""
        with self._lock:
            entity = self._storage.get(entity_id)
            if entity:
                # Return a copy so external modifications don't affect stored version
                return VersionedEntity(
                    id=entity.id,
                    data=entity.data.copy() if isinstance(entity.data, dict) else entity.data,
                    version=entity.version,
                    last_modified=entity.last_modified
                )
            return None
    
    def delete(self, entity_id: str, expected_version: int):
        """Delete entity with version check."""
        with self._lock:
            existing = self._storage.get(entity_id)
            if existing is None:
                raise KeyError(f"Entity {entity_id} not found")
            if existing.version != expected_version:
                raise OptimisticLockException(
                    f"Cannot delete: version mismatch. "
                    f"Expected {expected_version}, current is {existing.version}"
                )
            del self._storage[entity_id]


class OptimisticLockTemplate:
    """
    Template for retrying operations on optimistic lock failure.
    Implements the retry pattern for concurrent updates.
    """
    
    def __init__(self, repository: OptimisticLockRepository, max_retries: int = 3):
        self.repository = repository
        self.max_retries = max_retries
    
    def execute_with_retry(self, entity_id: str, 
                          update_func: Callable[[Any], Any]) -> VersionedEntity:
        """
        Execute update with automatic retry on lock failure.
        
        Args:
            entity_id: ID of entity to update
            update_func: Function that takes current data and returns modified data
        
        Returns:
            Updated entity
        
        Raises:
            OptimisticLockException if max retries exceeded
        """
        last_exception = None
        
        for attempt in range(self.max_retries):
            try:
                # Load current version
                entity = self.repository.get(entity_id)
                if entity is None:
                    raise KeyError(f"Entity {entity_id} not found")
                
                # Apply business logic
                entity.data = update_func(entity.data)
                
                # Try to save (will check version)
                return self.repository.save(entity)
                
            except OptimisticLockException as e:
                last_exception = e
                # Wait briefly before retry
                import time
                time.sleep(0.01 * (attempt + 1))
                continue
        
        raise OptimisticLockException(
            f"Failed after {self.max_retries} attempts: {last_exception}"
        )


# Example usage with bank account
@dataclass
class BankAccount:
    """Bank account data for optimistic locking example."""
    owner: str
    balance: float
    transactions: list = field(default_factory=list)


def transfer_money(repo: OptimisticLockRepository,
                   template: OptimisticLockTemplate,
                   from_id: str, to_id: str, amount: float):
    """
    Transfer money between accounts with optimistic locking.
    Will retry if concurrent modifications occur.
    """
    # Deduct from source
    def deduct(data: BankAccount) -> BankAccount:
        if data.balance < amount:
            raise ValueError("Insufficient funds")
        data.balance -= amount
        data.transactions.append({
            'type': 'debit',
            'amount': amount,
            'to': to_id
        })
        return data
    
    # Add to destination
    def credit(data: BankAccount) -> BankAccount:
        data.balance += amount
        data.transactions.append({
            'type': 'credit',
            'amount': amount,
            'from': from_id
        })
        return data
    
    # Execute both updates with retry
    template.execute_with_retry(from_id, deduct)
    template.execute_with_retry(to_id, credit)
```
**Why**: Optimistic locking allows concurrent reads while detecting write conflicts. More scalable than pessimistic locking for read-heavy workloads. Common in distributed systems and microservices where database-level locks are impractical.

## 9. Coroutine Patterns

```python
# PROBLEM: Implement advanced coroutine patterns for async programming
# TIME: O(1) per operation SPACE: O(n) for pending tasks
import asyncio
from typing import List, Any, Callable, TypeVar, Optional
from dataclasses import dataclass, field
from datetime import datetime
import functools

T = TypeVar('T')


# ============ PATTERN 1: ASYNC PIPELINE ============

class AsyncPipeline:
    """
    Chain multiple async operations in a pipeline.
    Each stage processes the output of the previous stage.
    """
    
    def __init__(self):
        self.stages: List[Callable] = []
    
    def add_stage(self, stage: Callable[[Any], Any]) -> 'AsyncPipeline':
        """Add a processing stage to the pipeline."""
        self.stages.append(stage)
        return self
    
    async def execute(self, initial_input: Any) -> Any:
        """Execute the pipeline with the given input."""
        result = initial_input
        for stage in self.stages:
            if asyncio.iscoroutinefunction(stage):
                result = await stage(result)
            else:
                result = stage(result)
        return result
    
    async def execute_batch(self, inputs: List[Any]) -> List[Any]:
        """Execute pipeline on multiple inputs concurrently."""
        tasks = [self.execute(inp) for inp in inputs]
        return await asyncio.gather(*tasks)


# ============ PATTERN 2: ASYNC GENERATOR WITH BACKPRESSURE ============

async def bounded_async_generator(source: List[Any], 
                                   consumer: Callable[[Any], Any],
                                   max_pending: int = 10):
    """
    Async generator with backpressure control.
    Limits the number of pending items to prevent memory overflow.
    """
    semaphore = asyncio.Semaphore(max_pending)
    
    async def process_item(item):
        async with semaphore:
            if asyncio.iscoroutinefunction(consumer):
                return await consumer(item)
            return consumer(item)
    
    tasks = []
    for item in source:
        # Wait if we've hit the limit
        await semaphore.acquire()
        semaphore.release()
        
        task = asyncio.create_task(process_item(item))
        tasks.append(task)
    
    return await asyncio.gather(*tasks)


# ============ PATTERN 3: ASYNC CIRCUIT BREAKER ============

class AsyncCircuitBreaker:
    """
    Circuit breaker pattern for async operations.
    Prevents cascading failures by failing fast when service is unhealthy.
    """
    
    def __init__(self, failure_threshold: int = 5, 
                 recovery_timeout: float = 30.0):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.failures = 0
        self.last_failure_time: Optional[float] = None
        self.state = "closed"  # closed, open, half_open
        self._lock = asyncio.Lock()
    
    async def call(self, func: Callable[..., T], *args, **kwargs) -> T:
        """Execute function with circuit breaker protection."""
        async with self._lock:
            if self.state == "open":
                # Check if we should try half-open
                if self._should_attempt_recovery():
                    self.state = "half_open"
                else:
                    raise CircuitBreakerOpenError("Circuit breaker is open")
        
        try:
            result = await func(*args, **kwargs)
            await self._on_success()
            return result
        except Exception as e:
            await self._on_failure()
            raise
    
    def _should_attempt_recovery(self) -> bool:
        """Check if enough time has passed to attempt recovery."""
        if self.last_failure_time is None:
            return True
        import time
        return (time.time() - self.last_failure_time) >= self.recovery_timeout
    
    async def _on_success(self):
        """Reset on successful call."""
        async with self._lock:
            self.failures = 0
            self.state = "closed"
    
    async def _on_failure(self):
        """Record failure and potentially open circuit."""
        async with self._lock:
            self.failures += 1
            import time
            self.last_failure_time = time.time()
            
            if self.failures >= self.failure_threshold:
                self.state = "open"


class CircuitBreakerOpenError(Exception):
    """Raised when circuit breaker is open."""
    pass


# ============ PATTERN 4: ASYNC RATE LIMITER ============

class AsyncRateLimiter:
    """
    Token bucket rate limiter for async operations.
    Controls the rate of async function calls.
    """
    
    def __init__(self, rate: float, capacity: int):
        """
        Args:
            rate: Tokens added per second
            capacity: Maximum tokens in bucket
        """
        self.rate = rate
        self.capacity = capacity
        self.tokens = capacity
        self.last_update = datetime.utcnow()
        self._lock = asyncio.Lock()
    
    async def acquire(self, tokens: int = 1):
        """Acquire tokens, waiting if necessary."""
        while True:
            async with self._lock:
                self._refill()
                
                if self.tokens >= tokens:
                    self.tokens -= tokens
                    return
                
                # Calculate wait time
                needed = tokens - self.tokens
                wait_seconds = needed / self.rate
            
            # Wait outside lock
            await asyncio.sleep(wait_seconds)
    
    def _refill(self):
        """Refill tokens based on elapsed time."""
        now = datetime.utcnow()
        elapsed = (now - self.last_update).total_seconds()
        self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)
        self.last_update = now
    
    async def wrap(self, func: Callable[..., T]) -> Callable[..., T]:
        """Wrap a function with rate limiting."""
        @functools.wraps(func)
        async def wrapper(*args, **kwargs):
            await self.acquire()
            return await func(*args, **kwargs)
        return wrapper


# ============ PATTERN 5: ASYNC CACHE ============

class AsyncCache:
    """
    Async cache with TTL and automatic refresh.
    Useful for caching async operation results.
    """
    
    def __init__(self, ttl_seconds: float = 60.0):
        self.ttl_seconds = ttl_seconds
        self._cache: dict[str, tuple[Any, float]] = {}
        self._lock = asyncio.Lock()
        self._refreshing: set[str] = set()
    
    async def get_or_fetch(self, key: str, 
                          fetcher: Callable[[], Any]) -> Any:
        """Get from cache or fetch if missing/expired."""
        async with self._lock:
            if key in self._cache:
                value, timestamp = self._cache[key]
                import time
                if time.time() - timestamp < self.ttl_seconds:
                    return value
        
        # Fetch outside lock to allow concurrent reads
        import time
        value = await fetcher()
        
        async with self._lock:
            self._cache[key] = (value, time.time())
        
        return value
    
    async def invalidate(self, key: str):
        """Invalidate a cache entry."""
        async with self._lock:
            self._cache.pop(key, None)
    
    async def invalidate_all(self):
        """Clear entire cache."""
        async with self._lock:
            self._cache.clear()


# ============ PATTERN 6: ASYNC WORKER POOL ============

class AsyncWorkerPool:
    """
    Pool of workers processing tasks from a queue.
    Provides controlled concurrency for task processing.
    """
    
    def __init__(self, num_workers: int, max_queue_size: int = 100):
        self.num_workers = num_workers
        self.queue: asyncio.Queue = asyncio.Queue(maxsize=max_queue_size)
        self.workers: List[asyncio.Task] = []
        self.running = False
    
    async def start(self):
        """Start worker pool."""
        self.running = True
        for i in range(self.num_workers):
            worker = asyncio.create_task(self._worker(i))
            self.workers.append(worker)
    
    async def stop(self):
        """Stop worker pool gracefully."""
        self.running = False
        # Send stop signals
        for _ in range(self.num_workers):
            await self.queue.put(None)
        # Wait for workers to finish
        await asyncio.gather(*self.workers)
    
    async def submit(self, task: Callable[..., Any], *args, **kwargs):
        """Submit a task to the pool."""
        await self.queue.put((task, args, kwargs))
    
    async def _worker(self, worker_id: int):
        """Worker coroutine that processes tasks."""
        while self.running:
            try:
                item = await asyncio.wait_for(self.queue.get(), timeout=1.0)
            except asyncio.TimeoutError:
                continue
            
            if item is None:
                break
            
            task, args, kwargs = item
            try:
                if asyncio.iscoroutinefunction(task):
                    await task(*args, **kwargs)
                else:
                    task(*args, **kwargs)
            except Exception as e:
                print(f"Worker {worker_id} error: {e}")
            
            self.queue.task_done()
```
**Why**: Advanced coroutine patterns enable building robust, scalable async applications. These patterns handle common challenges: flow control (backpressure), fault tolerance (circuit breaker), resource management (rate limiting, caching), and concurrency control (worker pools).

## 10. Edge Coloring

```python
# PROBLEM: Color edges of a graph so no adjacent edges share the same color
# TIME: O(V * E) for greedy, O(E * log(V)) for bipartite SPACE: O(E)
from typing import Dict, List, Set, Tuple
from collections import defaultdict


def edge_coloring_greedy(graph: Dict[int, Set[int]]) -> Dict[Tuple[int, int], int]:
    """
    Greedy edge coloring algorithm.
    Adjacent edges (sharing a vertex) must have different colors.
    Uses at most Δ + 1 colors where Δ is maximum degree.
    
    Args:
        graph: Adjacency list representation {vertex: {neighbors}}
    
    Returns:
        Dictionary mapping edges (v1, v2) to color numbers
    """
    # Build edge list with canonical ordering
    edges = []
    for v in graph:
        for u in graph[v]:
            if v < u:  # Avoid duplicates
                edges.append((v, u))
    
    # Track colors used at each vertex
    vertex_colors: Dict[int, Set[int]] = defaultdict(set)
    edge_colors: Dict[Tuple[int, int], int] = {}
    
    for v, u in edges:
        # Find the smallest color not used by adjacent edges
        used_colors = vertex_colors[v] | vertex_colors[u]
        
        color = 0
        while color in used_colors:
            color += 1
        
        edge_colors[(v, u)] = color
        vertex_colors[v].add(color)
        vertex_colors[u].add(color)
    
    return edge_colors


def edge_coloring_bipartite(graph: Dict[int, Set[int]]) -> Dict[Tuple[int, int], int]:
    """
    Edge coloring for bipartite graphs using König's theorem.
    Uses exactly Δ colors where Δ is maximum degree.
    
    Args:
        graph: Adjacency list of a bipartite graph
    
    Returns:
        Dictionary mapping edges to color numbers
    """
    # First, verify graph is bipartite and get partitions
    partition = _check_bipartite(graph)
    if partition is None:
        raise ValueError("Graph is not bipartite")
    
    left, right = partition
    
    # Calculate maximum degree
    max_degree = max(len(neighbors) for neighbors in graph.values())
    
    # Build edge list
    edges = []
    for v in left:
        for u in graph[v]:
            if u in right:
                edges.append((v, u))
    
    # Track colors used at each vertex
    vertex_colors: Dict[int, Set[int]] = defaultdict(set)
    edge_colors: Dict[Tuple[int, int], int] = {}
    
    # Use a more efficient matching-based approach
    # For each color, find a maximum matching and color those edges
    remaining_edges = set(edges)
    
    for color in range(max_degree):
        # Find edges that can use this color
        for v, u in list(remaining_edges):
            if color not in vertex_colors[v] and color not in vertex_colors[u]:
                edge_colors[(v, u)] = color
                vertex_colors[v].add(color)
                vertex_colors[u].add(color)
                remaining_edges.discard((v, u))
    
    return edge_colors


def _check_bipartite(graph: Dict[int, Set[int]]) -> Tuple[Set[int], Set[int]]:
    """
    Check if graph is bipartite and return partitions.
    Returns None if not bipartite.
    """
    if not graph:
        return set(), set()
    
    color = {}  # 0 = left, 1 = right
    
    def dfs(v, c):
        if v in color:
            return color[v] == c
        color[v] = c
        for u in graph.get(v, set()):
            if not dfs(u, 1 - c):
                return False
        return True
    
    start = next(iter(graph))
    if not dfs(start, 0):
        return None
    
    left = {v for v, c in color.items() if c == 0}
    right = {v for v, c in color.items() if c == 1}
    return left, right


def edge_coloring_misra_gries(graph: Dict[int, Set[int]]) -> Dict[Tuple[int, int], int]:
    """
    Misra & Gries edge coloring algorithm.
    Uses at most Δ + 1 colors with better quality than greedy.
    
    Args:
        graph: Adjacency list representation
    
    Returns:
        Dictionary mapping edges to color numbers
    """
    # Calculate maximum degree
    max_degree = max(len(neighbors) for neighbors in graph.values())
    max_colors = max_degree + 1
    
    # Track colors at each vertex and edge colors
    vertex_colors: Dict[int, Set[int]] = defaultdict(set)
    edge_colors: Dict[Tuple[int, int], int] = {}
    
    # Get all edges
    edges = []
    for v in graph:
        for u in graph[v]:
            if v < u:
                edges.append((v, u))
    
    for v, u in edges:
        # Find free colors at each vertex
        free_v = set(range(max_colors)) - vertex_colors[v]
        free_u = set(range(max_colors)) - vertex_colors[u]
        
        # Try to find a common free color
        common = free_v & free_u
        if common:
            color = min(common)
            edge_colors[(v, u)] = color
            vertex_colors[v].add(color)
            vertex_colors[u].add(color)
            continue
        
        # No common free color - need to recolor
        # Find a free color at v and u
        alpha = min(free_v)  # Free at v
        beta = min(free_u)   # Free at u
        
        # Find maximal alternating path from u using colors alpha/beta
        path = _find_alternating_path(u, alpha, beta, graph, vertex_colors)
        
        # Swap colors along the path
        for i in range(len(path) - 1):
            a, b = path[i], path[i + 1]
            edge_key = (min(a, b), max(a, b))
            
            # Swap colors for this edge
            old_color = edge_colors.get(edge_key)
            if old_color == alpha:
                edge_colors[edge_key] = beta
                vertex_colors[a].discard(alpha)
                vertex_colors[b].discard(alpha)
                vertex_colors[a].add(beta)
                vertex_colors[b].add(beta)
            elif old_color == beta:
                edge_colors[edge_key] = alpha
                vertex_colors[a].discard(beta)
                vertex_colors[b].discard(beta)
                vertex_colors[a].add(alpha)
                vertex_colors[b].add(alpha)
        
        # Now alpha is free at u
        edge_colors[(v, u)] = alpha
        vertex_colors[v].add(alpha)
        vertex_colors[u].add(alpha)
    
    return edge_colors


def _find_alternating_path(start: int, color1: int, color2: int,
                           graph: Dict[int, Set[int]], 
                           vertex_colors: Dict[int, Set[int]]) -> List[int]:
    """Find maximal alternating path starting from vertex."""
    path = [start]
    current = start
    looking_for = color1
    
    visited_edges = set()
    
    while True:
        # Find edge with the color we're looking for
        found = False
        for neighbor in graph.get(current, set()):
            edge_key = (min(current, neighbor), max(current, neighbor))
            if edge_key in visited_edges:
                continue
            if looking_for in vertex_colors.get(neighbor, set()):
                # Check if there's an edge to this neighbor with the looking color
                # This is a simplification; full implementation needs edge-color tracking
                path.append(neighbor)
                current = neighbor
                looking_for = color2 if looking_for == color1 else color1
                visited_edges.add(edge_key)
                found = True
                break
        
        if not found:
            break
    
    return path


def visualize_edge_colors(edge_colors: Dict[Tuple[int, int], int]):
    """
    Print edge coloring result in a readable format.
    Groups edges by color.
    """
    by_color = defaultdict(list)
    for edge, color in edge_colors.items():
        by_color[color].append(edge)
    
    print("Edge Coloring Result:")
    print("=" * 40)
    for color in sorted(by_color.keys()):
        edges = by_color[color]
        edge_str = ", ".join(f"({v}-{u})" for v, u in edges)
        print(f"Color {color}: {edge_str}")
    print(f"\nTotal colors used: {len(by_color)}")
```
