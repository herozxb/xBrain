## 1. Cycle Detection in Linked List

```python
# PROBLEM: Detect if a linked list has a cycle using O(1) memory
# TIME: O(n) SPACE: O(1)
def has_cycle(head):
    """
    Floyd's Tortoise and Hare algorithm.
    Two pointers move at different speeds; if they meet, there's a cycle.
    """
    if not head or not head.next:
        return False
    
    slow = head
    fast = head.next
    
    while slow != fast:
        if not fast or not fast.next:
            return False
        slow = slow.next
        fast = fast.next.next
    
    return True
```

**Why**: Classic interview problem demonstrating two-pointer technique and cycle detection fundamental to many systems (garbage collection, deadlock detection).

## 2. Sliding Window Maximum

```python
# PROBLEM: Find max in each sliding window of size k
# TIME: O(n) SPACE: O(k)
from collections import deque

def sliding_window_maximum(nums, k):
    """
    Use a deque to maintain indices of potential maximums.
    Monotonically decreasing deque ensures front is always the max.
    """
    if not nums or k <= 0:
        return []
    
    result = []
    dq = deque()  # stores indices, values are monotonically decreasing
    
    for i, num in enumerate(nums):
        # Remove indices outside the current window
        while dq and dq[0] <= i - k:
            dq.popleft()
        
        # Remove smaller elements (they can never be max)
        while dq and nums[dq[-1]] < num:
            dq.pop()
        
        dq.append(i)
        
        # Window is complete when i >= k - 1
        if i >= k - 1:
            result.append(nums[dq[0]])
    
    return result
```

**Why**: Essential for streaming data analysis, real-time monitoring systems, and demonstrates deque mastery for monotonic queue patterns.

## 3. LRU Cache Implementation

```python
# PROBLEM: Implement an LRU cache with O(1) get and put
# TIME: O(1) for both operations SPACE: O(capacity)
class Node:
    """Doubly linked list node for O(1) removal."""
    def __init__(self, key=0, value=0):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class LRUCache:
    """
    Combine hash map (O(1) lookup) with doubly linked list (O(1) reordering).
    Head = most recent, Tail = least recent.
    """
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}  # key -> Node
        self.head = Node()  # dummy head
        self.tail = Node()  # dummy tail
        self.head.next = self.tail
        self.tail.prev = self.head
    
    def _remove(self, node):
        """Remove node from list."""
        node.prev.next = node.next
        node.next.prev = node.prev
    
    def _add_to_front(self, node):
        """Add node right after head."""
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node
    
    def get(self, key):
        if key not in self.cache:
            return -1
        
        node = self.cache[key]
        # Move to front (most recently used)
        self._remove(node)
        self._add_to_front(node)
        return node.value
    
    def put(self, key, value):
        if key in self.cache:
            # Update existing and move to front
            node = self.cache[key]
            node.value = value
            self._remove(node)
            self._add_to_front(node)
        else:
            if len(self.cache) >= self.capacity:
                # Evict LRU (tail's prev)
                lru = self.tail.prev
                self._remove(lru)
                del self.cache[lru.key]
            
            new_node = Node(key, value)
            self.cache[key] = new_node
            self._add_to_front(new_node)
```

**Why**: Fundamental caching strategy used in databases, browsers, and OS page replacement. Demonstrates combining data structures for optimal performance.

## 4. Word Search (Backtracking)

```python
# PROBLEM: Find if a word exists in a 2D grid (adjacent cells)
# TIME: O(m * n * 4^L) where L is word length SPACE: O(L)
def exist(board, word):
    """
    DFS with backtracking. Mark visited cells temporarily.
    """
    if not board or not board[0]:
        return False
    
    rows, cols = len(board), len(board[0])
    
    def dfs(r, c, index):
        # Base case: found entire word
        if index == len(word):
            return True
        
        # Boundary and character check
        if (r < 0 or r >= rows or 
            c < 0 or c >= cols or 
            board[r][c] != word[index]):
            return False
        
        # Mark as visited
        temp = board[r][c]
        board[r][c] = '#'
        
        # Explore all 4 directions
        found = (dfs(r + 1, c, index + 1) or
                 dfs(r - 1, c, index + 1) or
                 dfs(r, c + 1, index + 1) or
                 dfs(r, c - 1, index + 1))
        
        # Backtrack: restore original value
        board[r][c] = temp
        return found
    
    # Try starting from each cell
    for r in range(rows):
        for c in range(cols):
            if dfs(r, c, 0):
                return True
    
    return False
```

**Why**: Classic backtracking problem demonstrating state mutation, recursion, and exploration patterns used in puzzle solving and game AI.

```python
# PROBLEM: Merge k sorted linked lists into one sorted list
# TIME: O(n log k) where n = total nodes, k = number of lists
# SPACE: O(k) for heap
import heapq

def merge_k_lists(lists):
    """
    Use min-heap to always get the smallest current node.
    Each pop gives us the next smallest element overall.
    """
    if not lists:
        return None
    
    # Min-heap stores (value, list_index, node)
    # list_index breaks ties when values are equal
    heap = []
    for i, node in enumerate(lists):
        if node:
            heapq.heappush(heap, (node.val, i, node))
    
    dummy = ListNode(0)
    current = dummy
    
    while heap:
        val, list_idx, node = heapq.heappop(heap)
        current.next = node
        current = current.next
        
        # Push next node from same list
        if node.next:
            heapq.heappush(heap, (node.next.val, list_idx, node.next))
    
    return dummy.next
```

**Why**: Efficient merging strategy used in external sorting, map-reduce patterns, and stream processing systems.

## 6. Median of Two Sorted Arrays

```python
# PROBLEM: Find median of two sorted arrays in O(log(m+n))
# TIME: O(log(min(m, n))) SPACE: O(1)
def find_median_sorted_arrays(nums1, nums2):
    """
    Binary search on the smaller array.
    Partition both arrays such that left halves have same count as right.
    """
    # Ensure nums1 is smaller for log(min(m,n)) complexity
    if len(nums1) > len(nums2):
        nums1, nums2 = nums2, nums1
    
    m, n = len(nums1), len(nums2)
    low, high = 0, m
    
    while low <= high:
        partition1 = (low + high) // 2
        partition2 = (m + n + 1) // 2 - partition1
        
        # Handle edge cases with infinities
        max_left1 = float('-inf') if partition1 == 0 else nums1[partition1 - 1]
        min_right1 = float('inf') if partition1 == m else nums1[partition1]
        
        max_left2 = float('-inf') if partition2 == 0 else nums2[partition2 - 1]
        min_right2 = float('inf') if partition2 == n else nums2[partition2]
        
        # Check if partition is correct
        if max_left1 <= min_right2 and max_left2 <= min_right1:
            # Found correct partition
            if (m + n) % 2 == 0:
                return (max(max_left1, max_left2) + min(min_right1, min_right2)) / 2
            else:
                return max(max_left1, max_left2)
        
        # Adjust partition
        elif max_left1 > min_right2:
            high = partition1 - 1
        else:
            low = partition1 + 1
    
    raise ValueError("Input arrays are not sorted")
```

**Why**: Advanced binary search application. Demonstrates partition thinking and edge case handling in divide-and-conquer algorithms.

## 7. Trie (Prefix Tree)

```python
# PROBLEM: Implement a trie with insert, search, and startsWith
# TIME: O(m) for all operations where m = key length
# SPACE: O(alphabet_size * total_characters)
class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_end = False


class Trie:
    """
    Tree where each node represents a character.
    Path from root to node represents a prefix.
    """
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, word):
        """Insert word into trie."""
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
    
    def search(self, word):
        """Check if exact word exists."""
        node = self.root
        for char in word:
            if char not in node.children:
                return False
            node = node.children[char]
        return node.is_end
    
    def starts_with(self, prefix):
        """Check if any word starts with prefix."""
        node = self.root
        for char in prefix:
            if char not in node.children:
                return False
            node = node.children[char]
        return True
    
    def get_words_with_prefix(self, prefix):
        """Return all words starting with prefix."""
        node = self.root
        for char in prefix:
            if char not in node.children:
                return []
            node = node.children[char]
        
        results = []
        self._collect_words(node, prefix, results)
        return results
    
    def _collect_words(self, node, prefix, results):
        """DFS to collect all words from a node."""
        if node.is_end:
            results.append(prefix)
        
        for char, child in node.children.items():
            self._collect_words(child, prefix + char, results)
```

**Why**: Foundation for autocomplete systems, spell checkers, IP routing, and efficient prefix matching in text processing.

## 8. Trapping Rain Water

```python
# PROBLEM: Calculate how much water can be trapped after raining
# TIME: O(n) SPACE: O(1)
def trap(height):
    """
    Two pointers approach: water at each position is determined by
    the minimum of max height to left and right, minus current height.
    """
    if not height:
        return 0
    
    left, right = 0, len(height) - 1
    left_max, right_max = 0, 0
    water = 0
    
    while left < right:
        if height[left] < height[right]:
            # Process left side
            if height[left] >= left_max:
                left_max = height[left]
            else:
                water += left_max - height[left]
            left += 1
        else:
            # Process right side
            if height[right] >= right_max:
                right_max = height[right]
            else:
                water += right_max - height[right]
            right -= 1
    
    return water
```

**Why**: Classic two-pointer problem teaching how to think about constraints from both directions simultaneously. Used in water flow simulation and terrain analysis.

## 9. Serialize and Deserialize Binary Tree

```python
# PROBLEM: Convert binary tree to string and back
# TIME: O(n) for both operations SPACE: O(n)
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


def serialize(root):
    """Convert tree to string using preorder traversal."""
    result = []
    
    def dfs(node):
        if not node:
            result.append('null')
            return
        result.append(str(node.val))
        dfs(node.left)
        dfs(node.right)
    
    dfs(root)
    return ','.join(result)


def deserialize(data):
    """Reconstruct tree from string."""
    values = data.split(',')
    index = [0]  # Use list for mutable reference in nested function
    
    def dfs():
        if values[index[0]] == 'null':
            index[0] += 1
            return None
        
        node = TreeNode(int(values[index[0]]))
        index[0] += 1
        node.left = dfs()
        node.right = dfs()
        return node
    
    return dfs()
```

**Why**: Essential for tree persistence, network transmission of tree structures, and demonstrates recursive serialization patterns used in data interchange formats.

## 10. Lowest Common Ancestor (Binary Tree)

```python
# PROBLEM: Find lowest common ancestor of two nodes in binary tree
# TIME: O(n) SPACE: O(h) where h is height
def lowest_common_ancestor(root, p, q):
    """
    Post-order traversal: if both subtrees return non-null,
    current node is LCA. If one is null, return the non-null one.
    """
    if not root:
        return None
    
    # If current node is p or q, it could be LCA
    if root == p or root == q:
        return root
    
    # Recursively search left and right
    left = lowest_common_ancestor(root.left, p, q)
    right = lowest_common_ancestor(root.right, p, q)
    
    # Both found: current node is LCA
    if left and right:
        return root
    
    # Only one found: return it (could be LCA or just descendant)
    return left if left else right


# Alternative: iterative with parent pointers
def lca_iterative(root, p, q):
    """
    Build parent pointers with BFS, then trace ancestors.
    """
    stack = [root]
    parent = {root: None}
    
    # Build parent map until both p and q are found
    while p not in parent or q not in parent:
        node = stack.pop()
        if node.left:
            parent[node.left] = node
            stack.append(node.left)
        if node.right:
            parent[node.right] = node
            stack.append(node.right)
    
    # Collect all ancestors of p
    ancestors = set()
    while p:
        ancestors.add(p)
        p = parent[p]
    
    # Find first ancestor of q that's also ancestor of p
    while q not in ancestors:
        q = parent[q]
    
    return q
```

**Why**: Critical for tree path queries, version control systems (git merge bases), and hierarchical data navigation. Shows elegant recursive reasoning.
