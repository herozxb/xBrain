## js_classes_super_keyword

```html
<!-- PROBLEM: Use super keyword to call parent class methods and constructor
TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>JavaScript super Keyword</title>
</head>
<body>
    <h2>Class Inheritance with super</h2>
    <div id="output"></div>

    <script>
        // Parent class
        class Animal {
            constructor(name, age) {
                this.name = name;
                this.age = age;
            }

            speak() {
                return `${this.name} makes a sound.`;
            }

            getInfo() {
                return `${this.name} is ${this.age} years old.`;
            }
        }

        // Child class using super in constructor
        class Dog extends Animal {
            constructor(name, age, breed) {
                // Call parent constructor with super
                super(name, age);
                this.breed = breed;
            }

            // Override parent method
            speak() {
                // Call parent method with super, then extend
                return `${super.speak()} Woof!`;
            }

            getInfo() {
                // Combine parent info with child-specific info
                return `${super.getInfo()} Breed: ${this.breed}.`;
            }
        }

        // Another child class
        class Cat extends Animal {
            constructor(name, age, color) {
                super(name, age);
                this.color = color;
            }

            speak() {
                return `${this.name} says: Meow!`;
            }

            getInfo() {
                return `${super.getInfo()} Color: ${this.color}.`;
            }
        }

        // Usage
        const dog = new Dog('Buddy', 3, 'Golden Retriever');
        const cat = new Cat('Whiskers', 5, 'Orange');

        const output = document.getElementById('output');
        output.innerHTML = `
            <h3>Dog:</h3>
            <p>${dog.speak()}</p>
            <p>${dog.getInfo()}</p>
            
            <h3>Cat:</h3>
            <p>${cat.speak()}</p>
            <p>${cat.getInfo()}</p>
        `;

        // Console output
        console.log('Dog speak:', dog.speak());
        console.log('Dog info:', dog.getInfo());
        console.log('Cat speak:', cat.speak());
        console.log('Cat info:', cat.getInfo());
    </script>
</body>
</html>
```
**Why**: The super keyword enables proper inheritance in ES6 classes by allowing child classes to call parent constructors and methods. This promotes code reuse and maintains the prototype chain correctly, essential for object-oriented programming in JavaScript.

## js_classes_private_fields

```html
<!-- PROBLEM: Create truly private class fields using # syntax
TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>JavaScript Private Fields</title>
</head>
<body>
    <h2>Private Class Fields with #</h2>
    <div id="output"></div>

    <script>
        class BankAccount {
            // Private fields (prefixed with #)
            #balance;
            #pin;
            #transactions = [];

            constructor(initialBalance, pin) {
                this.#balance = initialBalance;
                this.#pin = pin;
            }

            // Public method to access private field
            getBalance(enteredPin) {
                if (this.#verifyPin(enteredPin)) {
                    return this.#balance;
                }
                return 'Access denied';
            }

            // Private method
            #verifyPin(enteredPin) {
                return enteredPin === this.#pin;
            }

            // Private method to log transaction
            #addTransaction(type, amount) {
                this.#transactions.push({
                    type,
                    amount,
                    date: new Date().toISOString()
                });
            }

            deposit(amount, enteredPin) {
                if (!this.#verifyPin(enteredPin)) {
                    return 'Invalid PIN';
                }
                if (amount <= 0) {
                    return 'Invalid amount';
                }
                this.#balance += amount;
                this.#addTransaction('deposit', amount);
                return `Deposited $${amount}. New balance: $${this.#balance}`;
            }

            withdraw(amount, enteredPin) {
                if (!this.#verifyPin(enteredPin)) {
                    return 'Invalid PIN';
                }
                if (amount <= 0) {
                    return 'Invalid amount';
                }
                if (amount > this.#balance) {
                    return 'Insufficient funds';
                }
                this.#balance -= amount;
                this.#addTransaction('withdrawal', amount);
                return `Withdrew $${amount}. New balance: $${this.#balance}`;
            }

            getTransactionHistory(enteredPin) {
                if (!this.#verifyPin(enteredPin)) {
                    return 'Access denied';
                }
                return this.#transactions;
            }
        }

        // Usage
        const account = new BankAccount(1000, '1234');

        const output = document.getElementById('output');
        let html = '<h3>Bank Account Operations:</h3>';

        // Try to access private field directly (will cause error)
        // console.log(account.#balance); // SyntaxError

        // Access through public methods
        html += `<p>Initial Balance: ${account.getBalance('1234')}</p>`;
        html += `<p>Wrong PIN: ${account.getBalance('0000')}</p>`;
        html += `<p>${account.deposit(500, '1234')}</p>`;
        html += `<p>${account.withdraw(200, '1234')}</p>`;
        html += `<p>${account.withdraw(2000, '1234')}</p>`;

        // Show transaction history
        const history = account.getTransactionHistory('1234');
        html += '<h3>Transaction History:</h3><ul>';
        history.forEach(trans => {
            html += `<li>${trans.type}: $${trans.amount} on ${trans.date}</li>`;
        });
        html += '</ul>';

        output.innerHTML = html;
    </script>
</body>
</html>
```
**Why**: Private fields (#) provide true encapsulation in JavaScript classes, preventing external access and modification. Unlike closure-based privacy or naming conventions, private fields are enforced by the language runtime, ensuring data integrity and secure object-oriented design.

```html
        }
```

## js_object_getPrototypeOf

```html
<!-- PROBLEM: Get the prototype of an object to inspect inheritance chain
TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Object.getPrototypeOf</title>
</head>
<body>
    <h2>Get Object Prototype</h2>
    <div id="output"></div>

    <script>
        // Create objects using different methods
        const literalObject = { name: 'Literal' };

        function ConstructorFunction(name) {
            this.name = name;
        }
        ConstructorFunction.prototype.greet = function() {
            return `Hello, I'm ${this.name}`;
        };
        const constructedObject = new ConstructorFunction('Constructed');

        const prototypeObject = { 
            type: 'prototype',
            info() {
                return `Type: ${this.type}`;
            }
        };
        const objectWithPrototype = Object.create(prototypeObject);
        objectWithPrototype.name = 'Created with prototype';

        // Class-based object
        class Person {
            constructor(name) {
                this.name = name;
            }
            sayHello() {
                return `Hi, I'm ${this.name}`;
            }
        }
        const classInstance = new Person('Classy');

        // Inspect prototypes
        const output = document.getElementById('output');
        let html = '<h3>Prototype Chain Inspection:</h3>';

        // Helper function to show prototype
        function showPrototype(obj, name) {
            const proto = Object.getPrototypeOf(obj);
            const protoInfo = proto ? proto.constructor.name : 'null';
            const hasOwnProto = proto !== Object.prototype;
            
            return `
                <div style="margin: 20px 0; padding: 10px; border: 1px solid #ccc;">
                    <h4>${name}</h4>
                    <p>Object: ${JSON.stringify(obj)}</p>
                    <p>Prototype: ${protoInfo}</p>
                    <p>Has custom prototype: ${hasOwnProto}</p>
                    ${proto ? `<p>Prototype methods: ${Object.getOwnPropertyNames(proto).filter(p => p !== 'constructor').join(', ') || 'none'}</p>` : ''}
                </div>
            `;
        }

        html += showPrototype(literalObject, 'Object Literal');
        html += showPrototype(constructedObject, 'Constructor Function');
        html += showPrototype(objectWithPrototype, 'Object.create()');
        html += showPrototype(classInstance, 'Class Instance');

        // Walk the prototype chain
        html += '<h3>Full Prototype Chain for Class Instance:</h3>';
        let current = classInstance;
        let chain = [];
        let depth = 0;

        while (current !== null && depth < 5) {
            const proto = Object.getPrototypeOf(current);
            chain.push({
                level: depth,
                type: current.constructor.name,
                isPrototype: current !== classInstance,
                methods: Object.getOwnPropertyNames(current).filter(p => typeof current[p] === 'function')
            });
            current = proto;
            depth++;
        }

        html += '<ul>';
        chain.forEach(item => {
            html += `<li>Level ${item.level}: ${item.type} 
                ${item.isPrototype ? '(prototype)' : '(instance)'}
                ${item.methods.length > 0 ? `- Methods: ${item.methods.join(', ')}` : ''}
            </li>`;
        });
        html += '</ul>';

        output.innerHTML = html;
    </script>
</body>
</html>
```
**Why**: Object.getPrototypeOf is crucial for debugging inheritance issues, implementing mixins, and understanding the prototype chain. It provides a standard way to inspect object relationships, essential for advanced JavaScript patterns and library development.

## js_object_setPrototypeOf

```html
<!-- PROBLEM: Change an object's prototype after creation (use cautiously)
TIME: O(1) SPACE: O(1) -->
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Object.setPrototypeOf</title>
</head>
<body>
    <h2>Set Object Prototype</h2>
    <div id="output"></div>

    <script>
        // Define different prototypes
        const dogPrototype = {
            speak() {
                return `${this.name} says: Woof!`;
            },
            move() {
                return `${this.name} runs on four legs`;
            }
        };

        const birdPrototype = {
            speak() {
                return `${this.name} says: Tweet!`;
            },
            move() {
                return `${this.name} flies through the air`;
            }
        };

        const fishPrototype = {
            speak() {
                return `${this.name} says: Blub!`;
            },
            move() {
                return `${this.name} swims in water`;
            }
        };

        // Create an object and change its prototype
        const animal = {
            name: 'Creature'
        };

        const output = document.getElementById('output');
        let html = '<h3>Dynamic Prototype Switching:</h3>';
        html += '<p><em>Note: setPrototypeOf is slow and should be avoided in performance-critical code</em></p>';

        // Initially no custom prototype
        html += `<p>Original: ${animal.speak ? animal.speak() : 'No speak method'}</p>`;

        // Set to dog prototype
        Object.setPrototypeOf(animal, dogPrototype);
        html += `<p>As Dog: ${animal.speak()}</p>`;
        html += `<p>Movement: ${animal.move()}</p>`;

        // Change to bird prototype
        Object.setPrototypeOf(animal, birdPrototype);
        html += `<p>As Bird: ${animal.speak()}</p>`;
        html += `<p>Movement: ${animal.move()}</p>`;

        // Change to fish prototype
        Object.setPrototypeOf(animal, fishPrototype);
        html += `<p>As Fish: ${animal.speak()}</p>`;
        html += `<p>Movement: ${animal.move()}</p>`;

        // Practical example: Mixin pattern
        html += '<h3>Mixin Pattern Example:</h3>';

        const serializable = {
            serialize() {
                return JSON.stringify(this);
            },
            deserialize(json) {
                const data = JSON.parse(json);
                Object.assign(this, data);
            }
        };

        const comparable = {
            equals(other) {
                return JSON.stringify(this) === JSON.stringify(other);
            }
        };

        const user = {
            name: 'Alice',
            age: 30,
            email: 'alice@example.com'
        };

        // Add serialization capability
        Object.setPrototypeOf(user, { ...Object.getPrototypeOf(user), ...serializable });
        html += `<p>Serialized: ${user.serialize()}</p>`;

        // Create another user and compare
        const user2 = { name: 'Alice', age: 30, email: 'alice@example.com' };
        Object.setPrototypeOf(user2, { ...Object.getPrototypeOf(user2), ...comparable });

        // Can't directly compare since they have different prototypes
        // This demonstrates the limitation and shows better alternatives
        html += `<p>Same data: ${JSON.stringify(user) === JSON.stringify(user2)}</p>`;

        // Better approach: Use Object.create for mixin
        html += '<h3>Better Approach with Object.create:</h3>';
        const betterUser = Object.create({ ...serializable, ...comparable });
        Object.assign(betterUser, { name: 'Bob', age: 25 });

        html += `<p>Serialized: ${betterUser.serialize()}</p>`;

        output.innerHTML = html;
    </script>
</body>
</html>
```
**Why**: Object.setPrototypeOf allows dynamic prototype changes but should be used sparingly due to performance implications. It's useful for polyfills, testing, and advanced metaprogramming, but Object.create is generally preferred for setting up inheritance initially.

```html
            return str1.toLowerCase() === str2.toLowerCase();
        }

        html += `<p><strong>Compare:</strong> "Hello" === "HELLO": ${caseInsensitiveCompare('Hello', 'HELLO')}</p>`;

        // Normalize user input
        function normalizeEmail(email) {
            return email.toLowerCase().trim();
        }

        const emails = ['  USER@EXAMPLE.COM  ', 'Admin@Site.Com', 'test@domain.org'];
        html += '<h3>Email Normalization:</h3><ul>';
        emails.forEach(email => {
            html += `<li>"${email}" → "${normalizeEmail(email)}"</li>`;
        });
        html += '</ul>';

        // Creating URL slugs
        function createSlug(text) {
            return text
                .toLowerCase()
                .replace(/[^a-z0-9]+/g, '-')
                .replace(/(^-|-$)/g, '');
        }

        const titles = ['Hello World!', 'This is a TEST', 'My Blog Post #1'];
        html += '<h3>URL Slug Creation:</h3><ul>';
        titles.forEach(title => {
            html += `<li>"${title}" → "${createSlug(title)}"</li>`;
        });
        html += '</ul>';

        // Case-insensitive sorting
        const fruits = ['banana', 'Apple', 'cherry', 'Date', 'elderberry'];
        const sortedNormal = [...fruits].sort();
        const sortedCaseInsensitive = [...fruits].sort((a, b) => 
            a.toLowerCase().localeCompare(b.toLowerCase())
        );
        html += `<p><strong>Normal sort:</strong> ${sortedNormal.join(', ')}</p>`;
        html += `<p><strong>Case-insensitive sort:</strong> ${sortedCaseInsensitive.join(', ')}</p>`;

        // Performance note
        html += '<h3>Performance Consideration:</h3>';
        html += '<p>For repeated comparisons, cache the lowercase version:</p>';
        html += '<pre><code>const lowerCache = str.toLowerCase();\n// Use lowerCache multiple times</code></pre>';

        output.innerHTML = html;
    </script>
</body>
</html>
```
**Why**: toLowerCase is fundamental for case-insensitive operations, user input normalization, and data processing. Understanding locale-specific behavior (toLocaleLowerCase) is crucial for internationalization, especially for languages with special casing rules like Turkish.

```html
            return arr.filter(item => {

                seen.add(upper);
```
