```html
<head>

    </style>

<body>
    <!--
    PROBLEM: Handle asynchronous operations with callbacks
    TIME: O(1) SPACE: O(1)
    -->
    <h2>JavaScript Callbacks</h2>
    
    <button onclick="demonstrateCallbacks()">Run Examples</button>
    
    <div class="output" id="output"></div>
    
    <script>
        // Basic callback example
        function greet(name, callback) {
            console.log('Hello ' + name);
            callback();
        }
        
        // Callback with parameters
        function processArray(arr, callback) {
            const results = [];
            for (let i = 0; i < arr.length; i++) {
                results.push(callback(arr[i]));
            }
            return results;
        }
        
        // Asynchronous callback
        function fetchData(callback) {
            setTimeout(() => {
                callback({ data: 'Sample data from server' });
            }, 1000);
        }
        
        // Error-first callback pattern
        function divide(a, b, callback) {
            if (b === 0) {
                callback(new Error('Cannot divide by zero'), null);
            } else {
                callback(null, a / b);
            }
        }
        
        function demonstrateCallbacks() {
            const output = document.getElementById('output');
            output.innerHTML = '';
            
            // Example 1: Basic callback
            greet('World', function() {
                output.innerHTML += 'Greeting completed!<br>';
            });
            
            // Example 2: Array processing
            const numbers = [1, 2, 3, 4, 5];
            const doubled = processArray(numbers, num => num * 2);
            output.innerHTML += 'Doubled: ' + doubled.join(', ') + '<br>';
            
            // Example 3: Async callback
            output.innerHTML += 'Fetching data...<br>';
            fetchData(function(result) {
                output.innerHTML += 'Data received: ' + result.data + '<br>';
            });
            
            // Example 4: Error-first callback
            divide(10, 2, function(err, result) {
                if (err) {
                    output.innerHTML += 'Error: ' + err.message + '<br>';
                } else {
                    output.innerHTML += 'Result: ' + result + '<br>';
                }
            });
            
            // Example 5: Event listener callback
            output.innerHTML += '<br>Click the output area to see event callback in action<br>';
            output.addEventListener('click', function(e) {
                alert('Output clicked at: ' + e.clientX + ', ' + e.clientY);
            });
        }
    </script>
</body>
</html>
```

**Why**: Enables asynchronous programming and event-driven architectures in JavaScript.

```html
    <!--
    PROBLEM: Understand JavaScript event loop execution order
    TIME: O(1) SPACE: O(1)

    <div class="output" id="output"></div>
    
    <script>
        function demonstrateEventLoop() {
            const output = document.getElementById('output');
            let log = '';
            
            // Synchronous code - executed immediately
            log += '1. Synchronous start\n';
            
            // Macrotask (setTimeout) - goes to task queue
            setTimeout(() => {
                log += '4. Macrotask (setTimeout)\n';
                output.textContent = log;
            }, 0);
            
            // Microtask (Promise) - goes to microtask queue
            Promise.resolve()
                .then(() => {
                    log += '3. Microtask (Promise)\n';
                    output.textContent = log;
                });
            
            // Another macrotask
            setTimeout(() => {
                log += '5. Another macrotask\n';
                output.textContent = log;
            }, 0);
            
            // Another microtask
            Promise.resolve()
                .then(() => {
                    log += '3.5. Another microtask\n';
                    output.textContent = log;
                });
            
            // Synchronous code continues
            log += '2. Synchronous end\n';
            
            // Update output
            output.textContent = log;
            
            // Explanation
            setTimeout(() => {
                log += '\n--- Execution Order ---\n';
                log += '1. Sync code runs first\n';
                log += '2. Microtasks run after sync\n';
                log += '3. Macrotasks run last\n';
                output.textContent = log;
            }, 100);
        }
        
        // Demonstrate blocking vs non-blocking
        function demonstrateBlocking() {
            const output = document.getElementById('output');
            let log = 'Start\n';
            
            // This would block the event loop
            // for (let i = 0; i < 1e9; i++) {} // BAD
            
            // This doesn't block
            setTimeout(() => {
                log += 'Timer completed\n';
                output.textContent = log;
            }, 1000);
            
            log += 'End (appears immediately)\n';
            output.textContent = log;
        }
    </script>
</body>
</html>
```

**Why**: Explains JavaScript's concurrency model and execution order of synchronous, microtask, and macrotask queues.

```html
<html lang="en">
<head>
    <meta charset="UTF-8">

        }
        .container {

        }
    </style>
</head>
<body>
    <!--
    PROBLEM: Get or replace entire element including itself
    TIME: O(n) SPACE: O(n) where n is element size
    -->
    <h2>DOM outerHTML Property</h2>
    
    <div class="container">
        <div id="target" class="highlight">
            This is the target element
        </div>
        <p>This content stays</p>
    </div>
    
    <button onclick="getOuterHTML()">Get outerHTML</button>
    <button onclick="replaceElement()">Replace Element</button>
    <button onclick="resetElement()">Reset</button>
    
    <div id="output" style="background: #f0f0f0; padding: 10px; margin-top: 20px;"></div>
    
    <script>
        function getOuterHTML() {
            const element = document.getElementById('target');
            const output = document.getElementById('output');
            
            // Get outerHTML (includes the element itself)
            output.innerHTML = '<strong>outerHTML:</strong><br>' + 
                escapeHTML(element.outerHTML) + '<br><br>' +
                '<strong>innerHTML:</strong><br>' + 
                escapeHTML(element.innerHTML);
        }
        
        function replaceElement() {
            const element = document.getElementById('target');
            
            // Replace entire element with new HTML
            element.outerHTML = `
                <div id="target" style="background: lightgreen; padding: 20px; border-radius: 5px;">
                    <h3>New Element</h3>
                    <p>This replaced the old element completely</p>
                </div>
            `;
        }
        
        function resetElement() {
            // Find or recreate the original element
            let element = document.getElementById('target');
            if (!element) {
                const container = document.querySelector('.container');
                const newDiv = document.createElement('div');
                newDiv.id = 'target';
                newDiv.className = 'highlight';
                newDiv.textContent = 'This is the target element';
                container.insertBefore(newDiv, container.firstChild);
            } else {
                element.className = 'highlight';
                element.innerHTML = 'This is the target element';
            }
        }
        
        function escapeHTML(str) {
            const div = document.createElement('div');
            div.textContent = str;
            return div.innerHTML;
        }
        
        // Demonstrate difference between innerHTML and outerHTML
        function demonstrateDifference() {
            const element = document.getElementById('target');
            console.log('innerHTML:', element.innerHTML);
            console.log('outerHTML:', element.outerHTML);
            
            // innerHTML changes content inside
            // outerHTML replaces the entire element
        }
    </script>
</body>
</html>
```

**Why**: Gets or replaces entire element including the element itself, unlike innerHTML which only affects content.

## Event Mouseleave

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Mouseleave Demo</title>
    <style>
        body {
            font-family: Arial, sans-serif;
            max-width: 600px;
            margin: 50px auto;
            padding: 20px;
        }
        .container {
            border: 3px solid #333;
            padding: 40px;

        }
        .target {
            width: 200px;
            height: 200px;
            background: #4CAF50;
            margin: 20px;
            display: inline-block;
            transition: background 0.3s;
            cursor: pointer;
        }
        .target.mouseover {
            background: #45a049;
        }
        .info {
            background: #fff;
            padding: 10px;
            border-left: 4px solid #4CAF50;
            margin: 10px 0;
        }
    </style>
</head>
<body>
    <!--
    PROBLEM: Detect when mouse leaves element (non-bubbling)
    TIME: O(1) SPACE: O(1)
    -->
    <h2>Mouseleave vs Mouseout</h2>
    
    <div class="info">
        <strong>mouseleave:</strong> Does NOT bubble, triggers when mouse leaves element<br>
        <strong>mouseout:</strong> DOES bubble, triggers when mouse leaves element or enters child
    </div>
    
    <div class="container">
        <div id="mouseleaveBox" class="target">
            <div style="padding: 20px; background: rgba(255,255,255,0.3);">
                Child Element
            </div>
        </div>
        
        <div id="mouseoutBox" class="target">
            <div style="padding: 20px; background: rgba(255,255,255,0.3);">
                Child Element
            </div>
        </div>
    </div>
    
    <div id="output" style="font-family: monospace; background: #f0f0f0; padding: 15px; border-radius: 5px;">
        <p><strong>Mouseleave Box:</strong> <span id="mouseleaveCount">0</span> events</p>
        <p><strong>Mouseout Box:</strong> <span id="mouseoutCount">0</span> events</p>
    </div>
    
    <script>
        let mouseleaveCount = 0;
        let mouseoutCount = 0;
        
        // Mouseleave - does not bubble
        const mouseleaveBox = document.getElementById('mouseleaveBox');
        mouseleaveBox.addEventListener('mouseenter', () => {
            mouseleaveBox.classList.add('mouseover');
        });
        mouseleaveBox.addEventListener('mouseleave', () => {
            mouseleaveBox.classList.remove('mouseover');
            mouseleaveCount++;
            document.getElementById('mouseleaveCount').textContent = mouseleaveCount;
            console.log('mouseleave triggered');
        });
        
        // Mouseout - bubbles
        const mouseoutBox = document.getElementById('mouseoutBox');
        mouseoutBox.addEventListener('mouseover', () => {
            mouseoutBox.classList.add('mouseover');
        });
        mouseoutBox.addEventListener('mouseout', () => {
            mouseoutBox.classList.remove('mouseover');
            mouseoutCount++;
            document.getElementById('mouseoutCount').textContent = mouseoutCount;
            console.log('mouseout triggered');
        });
        
        // Hover child elements to see the difference:
        // mouseleave only triggers when leaving the parent
        // mouseout triggers when entering the child too
    </script>
</body>
</html>
```

**Why**: Detects mouse leaving element without bubbling, unlike mouseout which fires when entering child elements.

```html
            if (value.length === 0) {
                usernameOutput.textContent = '';

            } else if (isValid) {
                usernameOutput.innerHTML = '<span style="color: green;">✓ Valid username</span>';

            }

            charCount.textContent = `${count} / ${max}`;

                code: e.code,

            if (keys.length > maxKeys) {

            }
```
