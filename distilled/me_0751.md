## 1. Sparse Matrix CSR (Compressed Sparse Row)

```python
# PROBLEM: Store and access large sparse matrices efficiently
# TIME: O(nnz) for construction, O(k) for row access where k = non-zeros in row
# SPACE: O(nnz + n + 1) where nnz = number of non-zeros, n = rows
class SparseMatrixCSR:
    """Compressed Sparse Row format for efficient sparse matrix storage."""
    
    def __init__(self, rows, cols):
        self.rows = rows
        self.cols = cols
        self.values = []      # Non-zero values
        self.col_indices = [] # Column index for each value
        self.row_ptr = [0]    # Pointer to start of each row in values array
    
    def add_row(self, row_data):
        """Add a row to the matrix (must be called sequentially)."""
        for col, val in enumerate(row_data):
            if val != 0:
                self.values.append(val)
                self.col_indices.append(col)
        self.row_ptr.append(len(self.values))
    
    def get(self, row, col):
        """Get value at (row, col) - returns 0 if not stored."""
        if row >= self.rows or col >= self.cols:
            return 0
        
        # Search within row's range
        start, end = self.row_ptr[row], self.row_ptr[row + 1]
        for i in range(start, end):
            if self.col_indices[i] == col:
                return self.values[i]
        return 0
    
    def get_row(self, row):
        """Get all non-zero values in a row."""
        if row >= self.rows:
            return []
        start, end = self.row_ptr[row], self.row_ptr[row + 1]
        return [(self.col_indices[i], self.values[i]) 
                for i in range(start, end)]
```
**Why**: Reduces memory from O(n²) to O(nnz), ideal for matrices with <10% density.

## 2. XOR Linked List

```python
# PROBLEM: Implement doubly linked list with one pointer per node
# TIME: O(1) for insert/delete, O(n) for traversal
# SPACE: O(n) with half the pointer overhead of standard doubly linked list
class XORNode:
    """Node using XOR of prev/next pointers to save memory."""
    
    def __init__(self, data):
        self.data = data
        self.npx = 0  # XOR of next and prev pointers (as int IDs)
    
    def __repr__(self):
        return f"XORNode({self.data})"


class XORLinkedList:
    """Memory-efficient doubly linked list using XOR pointer compression."""
    
    def __init__(self):
        self.nodes = {}  # Map id -> node (simulate pointer arithmetic)
        self.head_id = None
        self.tail_id = None
    
    def _get_id(self, node):
        return id(node) if node else 0
    
    def insert_front(self, data):
        """Insert node at the front of the list."""
        new_node = XORNode(data)
        new_id = self._get_id(new_node)
        self.nodes[new_id] = new_node
        
        if self.head_id is None:
            self.head_id = self.tail_id = new_id
        else:
            # New node's npx = 0 XOR head_id = head_id
            new_node.npx = self.head_id
            # Old head's npx = (0 XOR old_next) XOR new_id
            old_head = self.nodes[self.head_id]
            old_head.npx = new_id ^ (old_head.npx)
            self.head_id = new_id
    
    def traverse_forward(self):
        """Traverse list from head to tail."""
        result = []
        prev_id = 0
        curr_id = self.head_id
        
        while curr_id:
            curr = self.nodes[curr_id]
            result.append(curr.data)
            # next = prev XOR curr.npx
            next_id = prev_id ^ curr.npx
            prev_id, curr_id = curr_id, next_id
        
        return result
```
**Why**: Saves memory in embedded systems; uses XOR trick to store two pointers in one field.

## 3. Circular Buffer (Ring Buffer)

```python
# PROBLEM: Fixed-size queue with O(1) operations and memory reuse
# TIME: O(1) for enqueue/dequeue/peek
# SPACE: O(capacity) - fixed allocation
import threading


class CircularBuffer:
    """Thread-safe ring buffer for producer-consumer scenarios."""
    
    def __init__(self, capacity):
        self.capacity = capacity
        self.buffer = [None] * capacity
        self.head = 0  # Read position
        self.tail = 0  # Write position
        self.size = 0
        self.lock = threading.Lock()
    
    def enqueue(self, item):
        """Add item to buffer. Returns False if full."""
        with self.lock:
            if self.is_full():
                return False
            
            self.buffer[self.tail] = item
            self.tail = (self.tail + 1) % self.capacity
            self.size += 1
            return True
    
    def dequeue(self):
        """Remove and return oldest item. Returns None if empty."""
        with self.lock:
            if self.is_empty():
                return None
            
            item = self.buffer[self.head]
            self.head = (self.head + 1) % self.capacity
            self.size -= 1
            return item
    
    def peek(self):
        """View oldest item without removing."""
        with self.lock:
            return self.buffer[self.head] if not self.is_empty() else None
    
    def is_empty(self):
        return self.size == 0
    
    def is_full(self):
        return self.size == self.capacity
    
    def __len__(self):
        return self.size
```
**Why**: Eliminates memory allocation/deallocation overhead; ideal for streaming data and bounded queues.

## 4. Difference Array (Range Update Optimization)

```python
# PROBLEM: Apply multiple range updates efficiently
# TIME: O(1) per update, O(n) to reconstruct final array
# SPACE: O(n)
class DifferenceArray:
    """Efficiently handle range increment/decrement operations."""
    
    def __init__(self, arr):
        self.n = len(arr)
        # Build difference array
        self.diff = [0] * self.n
        self.diff[0] = arr[0]
        for i in range(1, self.n):
            self.diff[i] = arr[i] - arr[i - 1]
    
    def range_update(self, left, right, delta):
        """Add delta to all elements in [left, right]."""
        if left < 0 or right >= self.n or left > right:
            return
        
        self.diff[left] += delta
        if right + 1 < self.n:
            self.diff[right + 1] -= delta
    
    def reconstruct(self):
        """Reconstruct the final array after all updates."""
        result = [0] * self.n
        result[0] = self.diff[0]
        for i in range(1, self.n):
            result[i] = result[i - 1] + self.diff[i]
        return result
    
    def batch_updates(self, updates):
        """Apply multiple range updates efficiently."""
        for left, right, delta in updates:
            self.range_update(left, right, delta)
        return self.reconstruct()


# Example: Range increment for multiple queries
def apply_range_increments(arr, queries):
    """
    queries: list of (left, right, increment)
    Returns array after all range increments applied.
    """
    diff_arr = DifferenceArray(arr)
    return diff_arr.batch_updates(queries)
```
**Why**: Converts O(k*n) range updates to O(k + n); essential for problems with many range modifications.

## 5. Sparse Set

```python
# PROBLEM: Fast set operations with O(1) clear and iteration over elements only
# TIME: O(1) for insert/contains/remove, O(k) to iterate k elements
# SPACE: O(universe_size) but only stores sparse members efficiently
class SparseSet:
    """Set optimized for rapid clear and iteration in sparse domains."""
    
    def __init__(self, universe_size):
        self.universe_size = universe_size
        self.dense = []              # Actual elements
        self.sparse = [0] * universe_size  # Index into dense array
    
    def insert(self, x):
        """Add element x to the set."""
        if x < 0 or x >= self.universe_size:
            return
        if not self.contains(x):
            self.sparse[x] = len(self.dense)
            self.dense.append(x)
    
    def contains(self, x):
        """Check if x is in the set."""
        if x < 0 or x >= self.universe_size:
            return False
        idx = self.sparse[x]
        return idx < len(self.dense) and self.dense[idx] == x
    
    def remove(self, x):
        """Remove x from the set."""
        if not self.contains(x):
            return
        
        idx = self.sparse[x]
        last = self.dense[-1]
        
        # Swap with last element
        self.dense[idx] = last
        self.sparse[last] = idx
        
        # Remove last
        self.dense.pop()
    
    def clear(self):
        """Clear all elements - O(1) operation!"""
        self.dense.clear()
    
    def __iter__(self):
        return iter(self.dense)
    
    def __len__(self):
        return len(self.dense)
```
**Why**: O(1) clear operation (just reset dense array); iterates only over actual members, not entire universe.

## 6. Rope Data Structure

```python
# PROBLEM: Efficient string concatenation and insertion
# TIME: O(log n) for concat/insert/split, O(n) to flatten
# SPACE: O(n)
class RopeNode:
    """Node in a rope (binary tree for string operations)."""
    
    def __init__(self, data=None):
        self.data = data      # String if leaf, None if internal
        self.left = None
        self.right = None
        self.weight = 0       # Length of left subtree (or string if leaf)
        
        if data is not None:
            self.weight = len(data)


class Rope:
    """Binary tree-based string for efficient concatenation."""
    
    def __init__(self, s=""):
        self.root = RopeNode(s) if s else None
    
    def _weight(self, node):
        return node.weight if node else 0
    
    def concat(self, other):
        """Concatenate two ropes - O(1)."""
        new_root = RopeNode()
        new_root.left = self.root
        new_root.right = other.root
        new_root.weight = self._total_length(self.root)
        self.root = new_root
        return self
    
    def _total_length(self, node):
        """Get total length of subtree."""
        if not node:
            return 0
        if node.data is not None:
            return len(node.data)
        return self._total_length(node.left) + self._total_length(node.right)
    
    def char_at(self, index):
        """Get character at index - O(log n)."""
        return self._char_at(self.root, index)
    
    def _char_at(self, node, idx):
        if not node:
            return None
        
        if node.data is not None:
            # Leaf node
            if idx < len(node.data):
                return node.data[idx]
            return None
        
        # Internal node
        if idx < node.weight:
            return self._char_at(node.left, idx)
        else:
            return self._char_at(node.right, idx - node.weight)
    
    def to_string(self):
        """Flatten rope to string - O(n)."""
        result = []
        self._inorder(self.root, result)
        return ''.join(result)
    
    def _inorder(self, node, result):
        if not node:
            return
        if node.data is not None:
            result.append(node.data)
        else:
            self._inorder(node.left, result)
            self._inorder(node.right, result)
    
    def __str__(self):
        return self.to_string()
```
**Why**: Text editors use ropes; concatenation is O(1) vs O(n) for strings; supports efficient insert/delete.

## 7. Fenwick Tree 2D (Binary Indexed Tree)

```python
# PROBLEM: Range sum queries and point updates in 2D grid
# TIME: O(log²(n*m)) for both update and query
# SPACE: O(n*m)
class FenwickTree2D:
    """2D Binary Indexed Tree for matrix operations."""
    
    def __init__(self, rows, cols):
        self.rows = rows
        self.cols = cols
        self.tree = [[0] * (cols + 1) for _ in range(rows + 1)]
    
    def _lsb(self, x):
        """Get least significant bit (isolates rightmost 1-bit)."""
        return x & (-x)
    
    def update(self, row, col, delta):
        """Add delta to cell (row, col) - 1-indexed internally."""
        i = row + 1
        while i <= self.rows:
            j = col + 1
            while j <= self.cols:
                self.tree[i][j] += delta
                j += self._lsb(j)
            i += self._lsb(i)
    
    def query(self, row, col):
        """Get sum from (0,0) to (row,col) inclusive."""
        result = 0
        i = row + 1
        while i > 0:
            j = col + 1
            while j > 0:
                result += self.tree[i][j]
                j -= self._lsb(j)
            i -= self._lsb(i)
        return result
    
    def range_query(self, r1, c1, r2, c2):
        """Get sum in rectangle from (r1,c1) to (r2,c2)."""
        # Inclusion-exclusion principle
        return (self.query(r2, c2) 
                - self.query(r1 - 1, c2) 
                - self.query(r2, c1 - 1) 
                + self.query(r1 - 1, c1 - 1))
    
    def build(self, matrix):
        """Build tree from initial matrix."""
        for i in range(self.rows):
            for j in range(self.cols):
                self.update(i, j, matrix[i][j])
```
**Why**: Supports dynamic updates; more space-efficient than 2D segment trees; used in image processing and gaming.

## 8. Unrolled Linked List

```python
# PROBLEM: Cache-friendly linked list with better CPU cache utilization
# TIME: O(n/k) for traversal where k = block size, O(k) for insert/delete
# SPACE: O(n) with better cache performance
class UnrolledBlock:
    """Block containing multiple elements."""
    
    def __init__(self, capacity=8):
        self.capacity = capacity
        self.elements = []
        self.next = None


class UnrolledLinkedList:
    """Linked list with cache-optimized block storage."""
    
    def __init__(self, block_size=8):
        self.head = None
        self.block_size = block_size
        self.size = 0
    
    def append(self, value):
        """Add element to the end."""
        if not self.head:
            self.head = UnrolledBlock(self.block_size)
        
        # Find last block
        curr = self.head
        while curr.next:
            curr = curr.next
        
        # If block is full, create new one
        if len(curr.elements) >= curr.capacity:
            # Split: move half to new block
            new_block = UnrolledBlock(self.block_size)
            mid = len(curr.elements) // 2
            new_block.elements = curr.elements[mid:]
            curr.elements = curr.elements[:mid]
            new_block.next = curr.next
            curr.next = new_block
            
            # Decide which block to add to
            if len(curr.elements) < curr.capacity:
                curr.elements.append(value)
            else:
                new_block.elements.append(value)
        else:
            curr.elements.append(value)
        
        self.size += 1
    
    def get(self, index):
        """Get element at index."""
        if index < 0 or index >= self.size:
            return None
        
        curr = self.head
        count = 0
        
        while curr:
            if count + len(curr.elements) > index:
                return curr.elements[index - count]
            count += len(curr.elements)
            curr = curr.next
        
        return None
    
    def to_list(self):
        """Convert to Python list."""
        result = []
        curr = self.head
        while curr:
            result.extend(curr.elements)
            curr = curr.next
        return result
    
    def __len__(self):
        return self.size
```
**Why**: Reduces cache misses by storing multiple elements per node; used in databases and file systems.

## 9. Zip Tree

```python
# PROBLEM: Simple randomized balanced BST with expected O(log n) height
# TIME: O(log n) expected for insert/delete/search
# SPACE: O(n)
import random


class ZipNode:
    """Node in zip tree with rank for balancing."""
    
    def __init__(self, key, value=None):
        self.key = key
        self.value = value
        self.rank = random.randint(1, 2**31)  # Random priority
        self.left = None
        self.right = None


class ZipTree:
    """Randomized BST using zip operations for balancing."""
    
    def __init__(self):
        self.root = None
    
    def _zip(self, left, right):
        """Merge two trees (all keys in left < all keys in right)."""
        if not left:
            return right
        if not right:
            return left
        
        if left.rank > right.rank:
            left.right = self._zip(left.right, right)
            return left
        else:
            right.left = self._zip(left, right.left)
            return right
    
    def _unzip(self, node, key):
        """Split tree into < key and >= key."""
        if not node:
            return (None, None)
        
        if key < node.key:
            left, right = self._unzip(node.left, key)
            node.left = right
            return (left, node)
        else:
            left, right = self._unzip(node.right, key)
            node.right = left
            return (node, right)
    
    def insert(self, key, value=None):
        """Insert key-value pair."""
        new_node = ZipNode(key, value)
        
        if not self.root:
            self.root = new_node
            return
        
        # Unzip at key
        left, right = self._unzip(self.root, key)
        
        # Insert new node between left and right
        new_node.left = left
        new_node.right = right
        self.root = new_node
    
    def delete(self, key):
        """Remove key from tree."""
        self.root = self._delete(self.root, key)
    
    def _delete(self, node, key):
        if not node:
            return None
        
        if key == node.key:
            return self._zip(node.left, node.right)
        elif key < node.key:
            node.left = self._delete(node.left, key)
        else:
            node.right = self._delete(node.right, key)
        
        return node
    
    def search(self, key):
        """Find value for key."""
        return self._search(self.root, key)
    
    def _search(self, node, key):
        if not node:
            return None
        if key == node.key:
            return node.value
        elif key < node.key:
            return self._search(node.left, key)
        else:
            return self._search(node.right, key)
```
**Why**: Simpler than treaps and skip lists; excellent cache behavior; used in Redis for sorted sets.

## 10. Judy Array (Conceptual Implementation)

```python
# PROBLEM: Fast, memory-efficient sparse array for integer keys
# TIME: O(log n) to O(1) depending on key distribution
# SPACE: Adaptive - uses only what's needed
class JudyArray:
    """Simplified Judy array using hybrid trie/array structure."""
    
    def __init__(self):
        self.root = {}  # Trie-like structure
        self.count = 0
    
    def _encode_key(self, key):
        """Convert integer to byte representation."""
        if key < 0:
            raise ValueError("Judy array only supports non-negative integers")
        # Simple: use hex string as path
        return format(key, 'x')
    
    def insert(self, key, value):
        """Insert key-value pair."""
        path = self._encode_key(key)
        node = self.root
        
        for i, char in enumerate(path):
            if char not in node:
                node[char] = {}
            if i == len(path) - 1:
                if '_value' not in node[char]:
                    self.count += 1
                node[char]['_value'] = value
            else:
                node = node[char]
    
    def get(self, key):
        """Retrieve value for key."""
        path = self._encode_key(key)
        node = self.root
        
        for i, char in enumerate(path):
            if char not in node:
                return None
            if i == len(path) - 1:
                return node[char].get('_value')
            node = node[char]
        
        return None
    
    def delete(self, key):
        """Remove key from array."""
        path = self._encode_key(key)
        nodes = [(self.root, path[0])]
        
        node = self.root
        for char in path:
            if char not in node:
                return False
            node = node[char]
            nodes.append((node, None))
        
        # Check if value exists
        if '_value' not in node:
            return False
        
        del node['_value']
        self.count -= 1
        return True
    
    def __len__(self):
        return self.count
    
    def __contains__(self, key):
        return self.get(key) is not None
    
    def __getitem__(self, key):
        result = self.get(key)
        if result is None:
            raise KeyError(key)
        return result
    
    def __setitem__(self, key, value):
        self.insert(key, value)
```
**Why**: Used in high-performance databases; combines benefits of arrays and trees; extremely memory-efficient for sparse data.
