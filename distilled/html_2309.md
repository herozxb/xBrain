    <script>
        // Keyboard navigation support
        const menuItems = document.querySelectorAll('[role="menuitem"]');
        
        menuItems.forEach((item, index) => {
            item.addEventListener('keydown', (e) => {
                let newIndex;
                
                switch(e.key) {
                    case 'ArrowRight':
                        newIndex = (index + 1) % menuItems.length;
                        menuItems[newIndex].focus();
                        break;
                    case 'ArrowLeft':
                        newIndex = (index - 1 + menuItems.length) % menuItems.length;
                        menuItems[newIndex].focus();
                        break;
                    case 'Home':
                        menuItems[0].focus();
                        break;
                    case 'End':
                        menuItems[menuItems.length - 1].focus();
                        break;
                }
            });
        });
    </script>

**Why**: ARIA navigation roles enhance accessibility by providing semantic meaning to screen readers, ensuring keyboard navigation works properly for users with disabilities.

## [JavaScript Async/Await Parallel]
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Async/Await Parallel Execution</title>
</head>
<body>
    <h2>Parallel API Requests with Async/Await</h2>
    <div id="results"></div>

    <script>
        // PROBLEM: Executing multiple async operations concurrently
        // TIME: O(n) where n = number of promises SPACE: O(n)

        // Simulate API call
        function fetchUser(id) {
            return new Promise(resolve => {
                setTimeout(() => {
                    resolve({ id, name: `User ${id}`, timestamp: Date.now() });
                }, Math.random() * 1000);
            });
        }

        function fetchPosts(userId) {
            return new Promise(resolve => {
                setTimeout(() => {
                    resolve([
                        { id: 1, title: `Post 1 by User ${userId}` },
                        { id: 2, title: `Post 2 by User ${userId}` }
                    ]);
                }, Math.random() * 1000);
            });
        }

        function fetchComments(postId) {
            return new Promise(resolve => {
                setTimeout(() => {
                    resolve([
                        { id: 1, text: `Comment on post ${postId}` },
                        { id: 2, text: `Another comment` }
                    ]);
                }, Math.random() * 1000);
            });
        }

        // Parallel execution with Promise.all()
        async function fetchAllDataParallel() {
            const startTime = Date.now();
            
            try {
                // Execute all requests concurrently
                const [users, posts, comments] = await Promise.all([
                    fetchUser(1),
                    fetchPosts(1),
                    fetchComments(1)
                ]);

                const endTime = Date.now();
                const duration = endTime - startTime;

                return {
                    users,
                    posts,
                    comments,
                    duration
                };
            } catch (error) {
                console.error('Error fetching data:', error);
                throw error;
            }
        }

        // Sequential execution for comparison
        async function fetchAllDataSequential() {
            const startTime = Date.now();
            
            try {
                const users = await fetchUser(1);
                const posts = await fetchPosts(1);
                const comments = await fetchComments(1);

                const endTime = Date.now();
                const duration = endTime - startTime;

                return {
                    users,
                    posts,
                    comments,
                    duration
                };
            } catch (error) {
                console.error('Error fetching data:', error);
                throw error;
            }
        }

        // Execute and display results
        (async () => {
            const resultsDiv = document.getElementById('results');
            
            // Parallel execution
            const parallelResults = await fetchAllDataParallel();
            resultsDiv.innerHTML += `<h3>Parallel Execution: ${parallelResults.duration}ms</h3>`;
            resultsDiv.innerHTML += `<pre>${JSON.stringify(parallelResults, null, 2)}</pre>`;

            // Sequential execution
            const sequentialResults = await fetchAllDataSequential();
            resultsDiv.innerHTML += `<h3>Sequential Execution: ${sequentialResults.duration}ms</h3>`;
            resultsDiv.innerHTML += `<pre>${JSON.stringify(sequentialResults, null, 2)}</pre>`;
        })();
    </script>
</body>
</html>
```
**Why**: Parallel async/await with `Promise.all()` dramatically improves performance by executing independent operations concurrently instead of sequentially, reducing total wait time.

## [JavaScript Object Keys]
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>JavaScript Object.keys()</title>
</head>
<body>
    <h2>Object.keys() Methods and Patterns</h2>
    <div id="output"></div>

    <script>
        // PROBLEM: Extracting and manipulating object keys efficiently
        // TIME: O(n) SPACE: O(n)

        const output = document.getElementById('output');
        
        function log(title, content) {
            output.innerHTML += `<h3>${title}</h3><pre>${JSON.stringify(content, null, 2)}</pre>`;
        }

        // Example object
        const user = {
            id: 1,
            name: 'John Doe',
            email: 'john@example.com',
            age: 30,
            active: true
        };

        // 1. Get all keys
        const keys = Object.keys(user);
        log('All Keys:', keys);

        // 2. Check if key exists
        function hasKey(obj, key) {
            return Object.keys(obj).includes(key);
        }
        log('Has "email" key:', hasKey(user, 'email'));
        log('Has "phone" key:', hasKey(user, 'phone'));

        // 3. Get key count
        function getKeyCount(obj) {
            return Object.keys(obj).length;
        }
        log('Key Count:', getKeyCount(user));

        // 4. Filter keys
        function filterKeys(obj, predicate) {
            return Object.keys(obj).filter(key => predicate(key, obj[key]));
        }
        const stringKeys = filterKeys(user, (key, value) => typeof value === 'string');
        log('String Value Keys:', stringKeys);

        // 5. Map over keys
        function mapKeys(obj, mapper) {
            return Object.keys(obj).map(key => mapper(key, obj[key]));
        }
        const upperCaseKeys = mapKeys(user, (key, value) => key.toUpperCase());
        log('Uppercase Keys:', upperCaseKeys);

        // 6. Get values by keys
        function getValuesByKeys(obj, keys) {
            return keys.map(key => obj[key]);
        }
        const selectedValues = getValuesByKeys(user, ['name', 'email']);
        log('Selected Values:', selectedValues);

        // 7. Transform object keys
        function transformKeys(obj, transformer) {
            return Object.keys(obj).reduce((acc, key) => {
                acc[transformer(key)] = obj[key];
                return acc;
            }, {});
        }
        const prefixedKeys = transformKeys(user, key => `user_${key}`);
        log('Prefixed Keys Object:', prefixedKeys);

        // 8. Get sorted keys
        function getSortedKeys(obj) {
            return Object.keys(obj).sort();
        }
        log('Sorted Keys:', getSortedKeys(user));

        // 9. Keys with specific pattern
        function getKeysByPattern(obj, pattern) {
            return Object.keys(obj).filter(key => pattern.test(key));
        }
        const keysStartingWithA = getKeysByPattern(user, /^a/);
        log('Keys starting with "a":', keysStartingWithA);

        // 10. Convert to array of key-value pairs
        function toKeyValuePairs(obj) {
            return Object.keys(obj).map(key => ({ key, value: obj[key] }));
        }
        log('Key-Value Pairs:', toKeyValuePairs(user));
    </script>
</body>
</html>
```
**Why**: `Object.keys()` is essential for iterating, transforming, and manipulating objects, enabling functional programming patterns and dynamic object inspection.

## [JavaScript Proxy Get]
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>JavaScript Proxy Get Trap</title>
</head>
<body>
    <h2>Proxy Get Trap Examples</h2>
    <div id="output"></div>

    <script>
        // PROBLEM: Intercepting and customizing property access
        // TIME: O(1) SPACE: O(1)

        const output = document.getElementById('output');
        
        function log(title, content) {
            output.innerHTML += `<h3>${title}</h3><pre>${JSON.stringify(content, null, 2)}</pre>`;
        }

        // 1. Basic get trap - logging property access
        const logger = {
            get(target, property, receiver) {
                console.log(`Getting property: ${property}`);
                return Reflect.get(target, property, receiver);
            }
        };

        const user = { name: 'John', age: 30 };
        const loggedUser = new Proxy(user, logger);
        loggedUser.name; // Console: "Getting property: name"
        log('Logged User:', loggedUser.name);

        // 2. Default values for undefined properties
        const withDefaults = {
            get(target, property) {
                return property in target ? target[property] : 'N/A';
            }
        };

        const data = { id: 1, name: 'Product' };
        const safeData = new Proxy(data, withDefaults);
        log('Existing Property:', safeData.name);
        log('Missing Property:', safeData.price); // Returns 'N/A'

        // 3. Case-insensitive property access
        const caseInsensitive = {
            get(target, property) {
                if (typeof property === 'string') {
                    const lowerProp = property.toLowerCase();
                    for (let key in target) {
                        if (key.toLowerCase() === lowerProp) {
                            return target[key];
                        }
                    }
                }
                return target[property];
            }
        };

        const config = { ApiKey: '12345', BaseUrl: 'https://api.example.com' };
        const ciConfig = new Proxy(config, caseInsensitive);
        log('Case-insensitive access:', ciConfig.apikey); // Returns '12345'

        // 4. Computed properties
        const computed = {
            get(target, property) {
                if (property === 'fullName') {
                    return `${target.firstName} ${target.lastName}`;
                }
                if (property === 'ageInDays') {
                    return target.age * 365;
                }
                return target[property];
            }
        };

        const person = { firstName: 'John', lastName: 'Doe', age: 30 };
        const computedPerson = new Proxy(person, computed);
        log('Computed fullName:', computedPerson.fullName);
        log('Computed ageInDays:', computedPerson.ageInDays);

        // 5. Nested property access with dot notation
        const nestedAccess = {
            get(target, property) {
                if (property.includes('.')) {
                    const keys = property.split('.');
                    let value = target;
                    for (let key of keys) {
                        value = value[key];
                        if (value === undefined) break;
                    }
                    return value;
                }
                return target[property];
            }
        };

        const nested = { 
            user: { 
                profile: { 
                    name: 'Alice' 
                } 
            } 
        };
        const nestedProxy = new Proxy(nested, nestedAccess);
        log('Nested access:', nestedProxy['user.profile.name']); // Returns 'Alice'

        // 6. Range validation with array indexing
        const arrayBounds = {
            get(target, property) {
                if (typeof property === 'string' && !isNaN(property)) {
                    const index = Number(property);
                    if (index < 0 || index >= target.length) {
                        return undefined;
                    }
                }
                return target[property];
            }
        };

        const arr = [1, 2, 3, 4, 5];
        const safeArray = new Proxy(arr, arrayBounds);
        log('Valid index:', safeArray[2]); // Returns 3
        log('Out of bounds:', safeArray[10]); // Returns undefined
    </script>
</body>
</html>
```
**Why**: Proxy get traps enable powerful metaprogramming capabilities: logging, validation, computed properties, and custom behavior for property access without modifying the original object.
