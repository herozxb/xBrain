## 1. Maximum Sum Subarray of Size K
```python
# PROBLEM: Find the maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)
def max_sum_subarray_k(arr, k):
    """
    Sliding window technique: maintain a window of size k,
    slide it through the array while tracking the maximum sum.
    """
    if len(arr) < k:
        return None
    
    # Calculate sum of first window
    window_sum = sum(arr[:k])
    max_sum = window_sum
    
    # Slide window: subtract leftmost, add rightmost
    for i in range(k, len(arr)):
        window_sum = window_sum - arr[i - k] + arr[i]
        max_sum = max(max_sum, window_sum)
    
    return max_sum
```
**Why**: Classic sliding window pattern. Efficiently handles fixed-size window problems without nested loops.

## 2. Container With Most Water
```python
# PROBLEM: Find two lines that form container holding the most water
# TIME: O(n) SPACE: O(1)
def max_area(height):
    """
    Two pointers from both ends. Move the shorter line inward
    because only a taller line can increase the area.
    """
    left, right = 0, len(height) - 1
    max_water = 0
    
    while left < right:
        # Area = width * min height
        width = right - left
        h = min(height[left], height[right])
        max_water = max(max_water, width * h)
        
        # Move pointer with shorter line
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1
    
    return max_water
```
**Why**: Two pointers eliminate brute force O(n²). Greedy choice is provably optimal.

## 3. N-Queens
```python
# PROBLEM: Place N queens on NxN board so no two attack each other
# TIME: O(N!) SPACE: O(N²)
def solve_n_queens(n):
    """
    Backtracking: try placing queens row by row, backtrack when conflict.
    Use sets to track occupied columns and diagonals for O(1) lookup.
    """
    def backtrack(row, cols, diag1, diag2, board):
        if row == n:
            solutions.append([''.join(r) for r in board])
            return
        
        for col in range(n):
            # Check if position is safe
            d1 = row - col  # "/" diagonal
            d2 = row + col  # "\" diagonal
            
            if col in cols or d1 in diag1 or d2 in diag2:
                continue
            
            # Place queen
            board[row][col] = 'Q'
            cols.add(col)
            diag1.add(d1)
            diag2.add(d2)
            
            backtrack(row + 1, cols, diag1, diag2, board)
            
            # Backtrack: remove queen
            board[row][col] = '.'
            cols.remove(col)
            diag1.remove(d1)
            diag2.remove(d2)
    
    solutions = []
    board = [['.' for _ in range(n)] for _ in range(n)]
    backtrack(0, set(), set(), set(), board)
    return solutions
```
**Why**: Canonical backtracking problem. Demonstrates state management and pruning invalid branches early.

## 4. Number of Connected Components
```python
# PROBLEM: Count connected components in an undirected graph
# TIME: O(n * α(n)) ≈ O(n) SPACE: O(n)
class UnionFind:
    """Union-Find (Disjoint Set) with path compression and union by rank."""
    
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.components = n
    
    def find(self, x):
        """Find root with path compression."""
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        """Union by rank. Returns True if merged."""
        px, py = self.find(x), self.find(y)
        if px == py:
            return False
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        self.components -= 1
        return True

def count_components(n, edges):
    """
    Use Union-Find to merge connected nodes.
    Remaining components = initial nodes - successful merges.
    """
    uf = UnionFind(n)
    for u, v in edges:
        uf.union(u, v)
    return uf.components
```
**Why**: Union-Find is essential for connectivity problems. Near-linear time with optimizations.

## 5. Longest Common Subsequence
```python
# PROBLEM: Find length of longest common subsequence between two strings
# TIME: O(m*n) SPACE: O(m*n) or O(min(m,n)) with optimization
def longest_common_subsequence(text1, text2):
    """
    Classic DP: dp[i][j] = LCS length for text1[:i] and text2[:j].
    If chars match: dp[i][j] = dp[i-1][j-1] + 1
    Else: dp[i][j] = max(dp[i-1][j], dp[i][j-1])
    """
    m, n = len(text1), len(text2)
    
    # Space optimization: only need previous row
    prev = [0] * (n + 1)
    
    for i in range(1, m + 1):
        curr = [0] * (n + 1)
        for j in range(1, n + 1):
            if text1[i - 1] == text2[j - 1]:
                curr[j] = prev[j - 1] + 1
            else:
                curr[j] = max(prev[j], curr[j - 1])
        prev = curr
    
    return prev[n]
```
**Why**: Foundational DP pattern. Space optimization shows important technique for 2D DP.

## 6. Dijkstra's Shortest Path
```python
# PROBLEM: Find shortest path from source to all nodes in weighted graph
# TIME: O((V + E) log V) SPACE: O(V + E)
import heapq
from collections import defaultdict

def dijkstra(n, edges, source):
    """
    Use min-heap to always process node with smallest known distance.
    Graph stored as adjacency list: {node: [(neighbor, weight), ...]}
    """
    # Build adjacency list
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))  # Remove if directed
    
    # Distance array, initialize to infinity
    dist = [float('inf')] * n
    dist[source] = 0
    
    # Min-heap: (distance, node)
    heap = [(0, source)]
    
    while heap:
        d, node = heapq.heappop(heap)
        
        # Skip if we already found a shorter path
        if d > dist[node]:
            continue
        
        for neighbor, weight in graph[node]:
            new_dist = dist[node] + weight
            if new_dist < dist[neighbor]:
                dist[neighbor] = new_dist
                heapq.heappush(heap, (new_dist, neighbor))
    
    return dist
```
**Why**: Fundamental graph algorithm. Heap-based approach is optimal for non-negative weights.

## 7. Serialize and Deserialize Binary Tree
```python
# PROBLEM: Convert binary tree to string and back
# TIME: O(n) SPACE: O(n)
from collections import deque

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def serialize(root):
    """
    Level-order traversal with null markers.
    Format: "1,2,3,null,null,4,5"
    """
    if not root:
        return "null"
    
    result = []
    queue = deque([root])
    
    while queue:
        node = queue.popleft()
        if node:
            result.append(str(node.val))
            queue.append(node.left)
            queue.append(node.right)
        else:
            result.append("null")
    
    # Trim trailing nulls
    while result and result[-1] == "null":
        result.pop()
    
    return ",".join(result)

def deserialize(data):
    """Reconstruct tree from level-order string."""
    if data == "null":
        return None
    
    values = data.split(",")
    root = TreeNode(int(values[0]))
    queue = deque([root])
    i = 1
    
    while queue and i < len(values):
        node = queue.popleft()
        
        # Left child
        if i < len(values) and values[i] != "null":
            node.left = TreeNode(int(values[i]))
            queue.append(node.left)
        i += 1
        
        # Right child
        if i < len(values) and values[i] != "null":
            node.right = TreeNode(int(values[i]))
            queue.append(node.right)
        i += 1
    
    return root
```
**Why**: Tree serialization is practical for storage/transmission. BFS approach preserves structure.

## 8. Merge K Sorted Lists

```python
# PROBLEM: Merge k sorted linked lists into one sorted list
# TIME: O(n log k) where n = total nodes, k = number of lists
# SPACE: O(k) for heap
import heapq

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next
    
    # For heap comparison
    def __lt__(self, other):
        return self.val < other.val

def merge_k_lists(lists):
    """
    Use min-heap to always get smallest node across all lists.
    Pop smallest, add its next to heap, repeat.
    """
    # Initialize heap with first node of each list
    heap = []
    for node in lists:
        if node:
            heapq.heappush(heap, node)
    
    dummy = ListNode()
    curr = dummy
    
    while heap:
        # Get smallest node
        smallest = heapq.heappop(heap)
        curr.next = smallest
        curr = curr.next
        
        # Add next node from same list to heap
        if smallest.next:
            heapq.heappush(heap, smallest.next)
    
    return dummy.next
```
**Why**: Classic heap application. Demonstrates efficient k-way merge used in external sorting.

## 9. Word Dictionary (Trie with Wildcard)
```python
# PROBLEM: Design data structure supporting addWord and search with '.' wildcard
# TIME: O(L) add, O(26^W * L) search where W = wildcards, L = word length
# SPACE: O(N * L) where N = number of words
class TrieNode:
    def __init__(self):
        self.children = {}  # char -> TrieNode
        self.is_word = False

class WordDictionary:
    """
    Trie structure where each node maps chars to children.
    Wildcard '.' requires DFS to check all possible paths.
    """
    
    def __init__(self):
        self.root = TrieNode()
    
    def addWord(self, word):
        node = self.root
        for ch in word:
            if ch not in node.children:
                node.children[ch] = TrieNode()
            node = node.children[ch]
        node.is_word = True
    
    def search(self, word):
        return self._search_helper(word, 0, self.root)
    
    def _search_helper(self, word, index, node):
        if index == len(word):
            return node.is_word
        
        ch = word[index]
        
        if ch == '.':
            # Try all possible children for wildcard
            for child in node.children.values():
                if self._search_helper(word, index + 1, child):
                    return True
            return False
        else:
            if ch not in node.children:
                return False
            return self._search_helper(word, index + 1, node.children[ch])
```
**Why**: Trie is optimal for prefix/wildcard searches. Recursive search handles '.' elegantly.

## 10. LRU Cache
```python
# PROBLEM: Design cache with O(1) get/put, evicting least recently used when full
# TIME: O(1) get/put SPACE: O(capacity)
class Node:
    """Doubly linked list node for O(1) removal/insertion."""
    def __init__(self, key=0, val=0):
        self.key = key
        self.val = val
        self.prev = None
        self.next = None

class LRUCache:
    """
    HashMap + Doubly Linked List:
    - HashMap gives O(1) access to any node
    - DLL maintains recency order (head = MRU, tail = LRU)
    """
    
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}  # key -> Node
        self.head = Node()  # Dummy head
        self.tail = Node()  # Dummy tail
        self.head.next = self.tail
        self.tail.prev = self.head
    
    def _remove(self, node):
        """Remove node from list."""
        prev, nxt = node.prev, node.next
        prev.next = nxt
        nxt.prev = prev
    
    def _add_to_front(self, node):
        """Add node right after head (most recent)."""
        node.prev = self.head
        node.next = self.head.next
        self.head.next.prev = node
        self.head.next = node
    
    def get(self, key):
        if key not in self.cache:
            return -1
        node = self.cache[key]
        # Move to front (most recently used)
        self._remove(node)
        self._add_to_front(node)
        return node.val
    
    def put(self, key, value):
        if key in self.cache:
            # Update existing, move to front
            node = self.cache[key]
            node.val = value
            self._remove(node)
            self._add_to_front(node)
        else:
            # Add new node
            if len(self.cache) >= self.capacity:
                # Evict LRU (node before tail)
                lru = self.tail.prev
                self._remove(lru)
                del self.cache[lru.key]
            
            new_node = Node(key, value)
            self.cache[key] = new_node
            self._add_to_front(new_node)
```
**Why**: Combines HashMap + DLL for true O(1) operations. Standard interview question pattern.
