## 1. [js_element_dataset] Custom Data Attributes Manager
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Dataset Manager</title>
</head>
<body>
    <div id="user-card" 
         data-user-id="12345" 
         data-user-name="John Doe" 
         data-user-role="admin"
         data-login-count="42">
        User Card
    </div>

    <script>
        // PROBLEM: Access and manipulate custom data attributes
        // TIME: O(1) SPACE: O(1)

        const card = document.getElementById('user-card');

        // Reading data attributes via dataset property
        // All data-* attributes become properties of element.dataset
        // Automatically converts kebab-case to camelCase
        function readUserData(element) {
            return {
                id: element.dataset.userId,        // data-user-id → userId
                name: element.dataset.userName,    // data-user-name → userName
                role: element.dataset.userRole,    // data-user-role → userRole
                logins: element.dataset.loginCount // data-login-count → loginCount
            };
        }

        // Setting data attributes
        // Values are automatically converted to strings
        function updateUserData(element, key, value) {
            element.dataset[key] = value;
            // This creates/updates data-* attribute in HTML
            // e.g., dataset.newProp = 'val' → data-new-prop="val"
        }

        // Deleting data attributes
        function removeUserData(element, key) {
            delete element.dataset[key];
            // Removes the corresponding data-* attribute
        }

        // Example usage
        console.log(readUserData(card));
        // {id: "12345", name: "John Doe", role: "admin", logins: "42"}

        updateUserData(card, 'lastLogin', '2026-02-27');
        console.log(card.dataset.lastLogin); // "2026-02-27"

        removeUserData(card, 'loginCount');
        console.log(card.dataset.loginCount); // undefined
    </script>
</body>
</html>
```
**Why**: The `dataset` property provides a clean, type-safe way to access custom `data-*` attributes without manual string parsing. It automatically handles kebab-case to camelCase conversion and is more readable than `getAttribute('data-...')`.

## 2. [js_element_style_cssText] Bulk Style Updates
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>cssText Bulk Styling</title>
    <style>
        #box { width: 100px; height: 100px; background: #3498db; }
    </style>
</head>
<body>
    <div id="box">Style Me</div>
    <button id="style-btn">Apply Styles</button>

    <script>
        // PROBLEM: Apply multiple CSS styles efficiently
        // TIME: O(1) SPACE: O(1)

        const box = document.getElementById('box');
        const btn = document.getElementById('style-btn');

        // Method 1: Individual style assignments (multiple reflows)
        function applyStylesSlowly(element) {
            element.style.backgroundColor = '#e74c3c';
            element.style.borderRadius = '10px';
            element.style.padding = '20px';
            element.style.margin = '10px';
            element.style.boxShadow = '0 4px 6px rgba(0,0,0,0.3)';
            // Each line can trigger a reflow!
        }

        // Method 2: Using cssText (single reflow, but replaces all inline styles)
        function applyStylesFast(element) {
            element.style.cssText = `
                background-color: #e74c3c;
                border-radius: 10px;
                padding: 20px;
                margin: 10px;
                box-shadow: 0 4px 6px rgba(0,0,0,0.3);
                transform: rotate(5deg);
            `;
            // WARNING: This REPLACES all existing inline styles!
        }

        // Method 3: Append to existing cssText (preserves current styles)
        function appendStyles(element, newStyles) {
            const current = element.style.cssText;
            element.style.cssText = current + ';' + newStyles;
        }

        // Get current cssText
        function getCurrentStyles(element) {
            return element.style.cssText;
        }

        btn.addEventListener('click', () => {
            // Using cssText for batch updates - single reflow
            applyStylesFast(box);
            
            // Check what was applied
            console.log(getCurrentStyles(box));
        });
    </script>
</body>
</html>
```
**Why**: `cssText` allows batch style updates in a single operation, reducing browser reflows. However, it replaces all inline styles, so use carefully or append to existing cssText when preservation is needed.

## 3. [js_cssstyledeclaration] CSS Style Declaration Inspector
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>CSSStyleDeclaration API</title>
    <style>
        #element {
            color: #333;
            font-size: 16px;
            margin: 10px;
        }
    </style>

<body>
    <div id="element" style="background: #f0f0f0; padding: 20px;">
        Styled Element
    </div>

    <script>
        // PROBLEM: Inspect and manipulate CSSStyleDeclaration object
        // TIME: O(n) for iteration SPACE: O(1)

        const element = document.getElementById('element');
        const style = element.style; // CSSStyleDeclaration object

        // CSSStyleDeclaration properties and methods
        function inspectStyles(styleDecl) {
            console.log('=== CSSStyleDeclaration ===');
            
            // length - number of inline style properties
            console.log('Property count:', styleDecl.length);
            
            // Iterate through all set properties
            for (let i = 0; i < styleDecl.length; i++) {
                const prop = styleDecl[i]; // Get property name
                const value = styleDecl.getPropertyValue(prop);
                console.log(`${prop}: ${value}`);
            }
            
            // cssText - get/set all styles as string
            console.log('cssText:', styleDecl.cssText);
        }

        // getPropertyValue - get a specific CSS property
        function getStyle(styleDecl, property) {
            return styleDecl.getPropertyValue(property);
        }

        // setProperty - set a CSS property with optional priority
        function setStyle(styleDecl, property, value, priority = '') {
            styleDecl.setProperty(property, value, priority);
            // priority can be 'important' or ''
        }

        // removeProperty - remove a CSS property
        function removeStyle(styleDecl, property) {
            const oldValue = styleDecl.removeProperty(property);
            return oldValue; // Returns the removed value
        }

        // item(index) - get property name at index
        function getPropertyAt(styleDecl, index) {
            return styleDecl.item(index);
        }

        // Usage examples
        inspectStyles(style);
        
        setStyle(style, 'border', '2px solid #333', 'important');
        console.log('Border:', getStyle(style, 'border'));
        
        removeStyle(style, 'padding');
        console.log('After removing padding:', style.cssText);
    </script>
</body>
</html>
```
**Why**: `CSSStyleDeclaration` is the object returned by `element.style`. It provides methods to get, set, and remove CSS properties programmatically with full control over priorities (like `!important`).

## 4. [js_event_type] Event Type Router
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Event Type Handler</title>
</head>
<body>
    <div id="interactive" style="padding: 50px; background: #ecf0f1; cursor: pointer;">
        Click, Hover, or Press Keys Here
    </div>
    <div id="log" style="margin-top: 20px; font-family: monospace;"></div>

    <script>
        // PROBLEM: Route events based on event type
        // TIME: O(1) SPACE: O(1)

        const interactive = document.getElementById('interactive');
        const logDiv = document.getElementById('log');

        function log(message) {
            logDiv.innerHTML += message + '<br>';
        }

        // Event type to handler mapping
        const eventHandlers = {
            click: (e) => `Clicked at (${e.clientX}, ${e.clientY})`,
            dblclick: (e) => 'Double clicked!',
            mouseenter: (e) => 'Mouse entered',
            mouseleave: (e) => 'Mouse left',
            mousedown: (e) => `Mouse button ${e.button} down`,
            mouseup: (e) => `Mouse button ${e.button} up`,
            keydown: (e) => `Key down: ${e.key} (code: ${e.code})`,
            keyup: (e) => `Key up: ${e.key}`,
            focus: (e) => 'Element focused',
            blur: (e) => 'Element blurred',
            input: (e) => `Input value: ${e.target.value}`,
            change: (e) => `Changed to: ${e.target.value}`
        };

        // Universal event handler using event.type
        function handleEvent(event) {
            const eventType = event.type; // Get the event type string
            
            // Route to appropriate handler
            if (eventHandlers[eventType]) {
                const result = eventHandlers[eventType](event);
                log(`[${eventType.toUpperCase()}] ${result}`);
            } else {
                log(`[UNKNOWN] Event type: ${eventType}`);
            }

            // Log all event properties
            console.log('Event object:', {
                type: event.type,
                target: event.target,
                currentTarget: event.currentTarget,
                timeStamp: event.timeStamp,
                isTrusted: event.isTrusted
            });
        }

        // Register multiple event listeners with single handler
        const eventTypes = ['click', 'dblclick', 'mouseenter', 'mouseleave', 
                           'mousedown', 'mouseup', 'keydown', 'keyup'];
        
        eventTypes.forEach(type => {
            interactive.addEventListener(type, handleEvent);
        });

        // Make div focusable for keyboard events
        interactive.tabIndex = 0;

        // Example: Detect touch vs mouse events
        function detectInputMethod(event) {
            const type = event.type;
            if (type.startsWith('touch')) return 'TOUCH';
            if (type.startsWith('mouse')) return 'MOUSE';
            if (type.startsWith('key')) return 'KEYBOARD';
            return 'OTHER';
        }
    </script>
</body>
</html>
```
**Why**: The `event.type` property identifies what kind of event occurred, enabling a single handler to route multiple event types. Useful for analytics, debugging, and creating unified event management systems.

## 5. [js_event_timeStamp] Event Timing Analyzer
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Event Timing Analysis</title>
</head>
<body>
    <button id="btn">Click Rapidly!</button>
    <div id="stats"></div>

    <script>
        // PROBLEM: Measure time between events using timeStamp
        // TIME: O(1) SPACE: O(n) for history

        const btn = document.getElementById('btn');
        const statsDiv = document.getElementById('stats');

        // Event timing tracker
        class EventTimer {
            constructor() {
                this.events = [];
                this.maxHistory = 100;
            }

            record(event) {
                // timeStamp: milliseconds since page load began
                const timestamp = event.timeStamp;
                
                this.events.push({
                    type: event.type,
                    timestamp: timestamp,
                    date: new Date(timestamp) // Convert to Date
                });

                // Limit history
                if (this.events.length > this.maxHistory) {
                    this.events.shift();
                }

                return this.analyze();
            }

            analyze() {
                if (this.events.length < 2) {
                    return { count: this.events.length };
                }

                const intervals = [];
                for (let i = 1; i < this.events.length; i++) {
                    intervals.push(
                        this.events[i].timestamp - this.events[i-1].timestamp
                    );
                }

                const avgInterval = intervals.reduce((a, b) => a + b, 0) / intervals.length;
                const minInterval = Math.min(...intervals);
                const maxInterval = Math.max(...intervals);

                return {
                    count: this.events.length,
                    avgInterval: avgInterval.toFixed(2) + 'ms',
                    minInterval: minInterval.toFixed(2) + 'ms',
                    maxInterval: maxInterval.toFixed(2) + 'ms',
                    lastTimestamp: this.events[this.events.length - 1].timestamp
                };
            }

            // Detect double-click (two clicks within 300ms)
            isDoubleClick(threshold = 300) {
                if (this.events.length < 2) return false;
                const last = this.events[this.events.length - 1];
                const prev = this.events[this.events.length - 2];
                return (last.timestamp - prev.timestamp) < threshold;
            }

            // Detect rapid clicking (more than 5 clicks in 1 second)
            isRapidClicking(windowMs = 1000, threshold = 5) {
                const now = this.events[this.events.length - 1]?.timestamp;
                if (!now) return false;
                
                const recentEvents = this.events.filter(
                    e => (now - e.timestamp) <= windowMs
                );
                return recentEvents.length >= threshold;
            }
        }

        const timer = new EventTimer();

        btn.addEventListener('click', (event) => {
            const analysis = timer.record(event);
            
            statsDiv.innerHTML = `
                <h3>Click Statistics</h3>
                <p>Total clicks: ${analysis.count}</p>
                <p>Average interval: ${analysis.avgInterval}</p>
                <p>Fastest: ${analysis.minInterval}</p>
                <p>Slowest: ${analysis.maxInterval}</p>
                <p>Double-click detected: ${timer.isDoubleClick()}</p>
                <p>Rapid clicking: ${timer.isRapidClicking()}</p>
            `;
        });

        // Note: event.timeStamp is relative to time origin
        // In browsers, this is when the document was created
        console.log('Time origin:', performance.timeOrigin);
    </script>
</body>
</html>
```
**Why**: `event.timeStamp` provides precise timing for events, enabling click speed analysis, double-click detection, and performance measurement. Critical for UX analytics and implementing time-based interactions.

```html
                    value: event.clientY, 
                    desc: 'Viewport Y (visible area)' 
                },
                'pageX': { 
                    value: event.pageX, 
                    desc: 'Document X (with scroll)' 
                },
                'pageY': { 
                    value: event.pageY, 
                    desc: 'Document Y (with scroll)' 
                },
                'screenX': { 
                    value: event.screenX, 
                    desc: 'Screen X (monitor)' 
                },
                'screenY': { 
                    value: event.screenY, 
                    desc: 'Screen Y (monitor)' 
                },
                'offsetX': { 
                    value: event.offsetX, 
                    desc: 'Element X (relative to target)' 
                },
                'offsetY': { 
                    value: event.offsetY, 
                    desc: 'Element Y (relative to target)' 
                },
                'movementX': { 
                    value: event.movementX, 
                    desc: 'Delta X since last move' 
                },
                'movementY': { 
                    value: event.movementY, 
                    desc: 'Delta Y since last move' 
                }
            };

            // Update cursor position using offset coordinates
            cursor.style.left = event.offsetX + 'px';
            cursor.style.top = event.offsetY + 'px';

            // Update table
            coordsTable.innerHTML = `
                <tr><th>Property</th><th>Value</th><th>Description</th></tr>
                ${Object.entries(coords).map(([key, data]) => `
                    <tr>
                        <td>${key}</td>
                        <td>${data.value}</td>
                        <td>${data.desc}</td>
                    </tr>
                `).join('')}
            `;
        }

        // Track mouse movement
        area.addEventListener('mousemove', updateCoords);

        // Convert between coordinate systems
        function clientToPage(clientX, clientY) {
            return {
                pageX: clientX + window.scrollX,
                pageY: clientY + window.scrollY
            };
        }

        function pageToClient(pageX, pageY) {
            return {
                clientX: pageX - window.scrollX,
                clientY: pageY - window.scrollY
            };
        }

        // Check if mouse is inside an element
        function isMouseInsideElement(event, element) {
            const rect = element.getBoundingClientRect();
            return (
                event.clientX >= rect.left &&
                event.clientX <= rect.right &&
                event.clientY >= rect.top &&
                event.clientY <= rect.bottom
            );
        }
    </script>
</body>
</html>
```
**Why**: Understanding the different mouse coordinate systems (`clientX/Y` vs `pageX/Y` vs `screenX/Y` vs `offsetX/Y`) is essential for drag-and-drop, custom cursors, games, and any mouse-based interaction.

## 7. [js_fetch_cache] Fetch Cache Control
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Fetch Cache Strategies</title>
</head>
<body>
    <button id="reload">Reload Data</button>
    <button id="force">Force Fresh</button>
    <div id="output"></div>

    <script>
        // PROBLEM: Control caching behavior in fetch requests
        // TIME: O(1) SPACE: O(1)

        const output = document.getElementById('output');

        // Cache strategies for different use cases
        const cacheStrategies = {
            // Default: use cache if available, fetch if not
            default: 'default',
            
            // No store: never cache, always fetch fresh
            noStore: 'no-store',
            
            // Reload: always fetch fresh, update cache
            reload: 'reload',
            
            // No cache: use cache but validate with server
            noCache: 'no-cache',
            
            // Force cache: use cache even if stale
            forceCache: 'force-cache',
            
            // Only if cached: only use cache, fail if not cached
            onlyIfCached: 'only-if-cached'
        };

        // Fetch with specific cache strategy
        async function fetchWithCache(url, cacheMode = 'default') {
            try {
                const response = await fetch(url, {
                    cache: cacheMode
                });
                
                // Check cache status from response
                console.log('Cache status:', {
                    type: response.type, // 'basic', 'cors', 'default', 'error', 'opaque'
                    url: response.url
                });
                
                return response;
            } catch (error) {
                console.error('Fetch failed:', error);
            }
        }

        // Implement stale-while-revalidate pattern
        async function staleWhileRevalidate(url) {
            const cache = await caches.open('v1');
            
            // Try cache first
            const cachedResponse = await cache.match(url);
            
            if (cachedResponse) {
                // Return cached data immediately
                output.innerHTML = '<p>From cache (serving stale)</p>';
                
                // Fetch fresh data in background
                fetch(url).then(response => {
                    cache.put(url, response.clone());
                    console.log('Cache updated in background');
                });
                
                return cachedResponse;
            }
            
            // No cache, fetch fresh
            const response = await fetch(url);
            cache.put(url, response.clone());
            output.innerHTML = '<p>Fresh from network</p>';
            return response;
        }

        // Cache with expiration
        class CacheWithExpiry {
            constructor(cacheName) {
                this.cacheName = cacheName;
            }

            async get(key, maxAgeMs) {
                const cache = await caches.open(this.cacheName);
                const response = await cache.match(key);
                
                if (!response) return null;
                
                // Check cached headers for age
                const dateHeader = response.headers.get('date');
                const cachedTime = dateHeader ? new Date(dateHeader).getTime() : 0;
                const age = Date.now() - cachedTime;
                
                if (age > maxAgeMs) {
                    console.log('Cache expired, age:', age);
                    await cache.delete(key);
                    return null;
                }
                
                console.log('Cache hit, age:', age);
                return response;
            }

            async set(key, response) {
                const cache = await caches.open(this.cacheName);
                await cache.put(key, response);
            }
        }

        // Usage
        document.getElementById('reload').addEventListener('click', async () => {
            // Use default caching
            await fetchWithCache('https://api.github.com/zen', 'default');
        });

        document.getElementById('force').addEventListener('click', async () => {
            // Force fresh data
            await fetchWithCache('https://api.github.com/zen', 'no-store');
        });
    </script>
</body>
</html>
```
**Why**: The `cache` option in fetch gives fine-grained control over HTTP caching. Combined with Cache API, you can implement sophisticated caching strategies like stale-while-revalidate for offline-capable web apps.

## 8. [js_response_json] JSON Response Handler
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Response.json() Handler</title>
</head>
<body>
    <button id="fetch-btn">Fetch Users</button>
    <div id="result"></div>

    <script>
        // PROBLEM: Safely parse JSON from fetch responses
        // TIME: O(n) where n is response size SPACE: O(n)

        const fetchBtn = document.getElementById('fetch-btn');
        const resultDiv = document.getElementById('result');

        // Basic JSON response handling
        async function fetchJSON(url) {
            const response = await fetch(url);
            
            // Check if request was successful
            if (!response.ok) {
                throw new Error(`HTTP ${response.status}: ${response.statusText}`);
            }
            
            // Check content-type header
            const contentType = response.headers.get('content-type');
            if (!contentType || !contentType.includes('application/json')) {
                throw new Error('Response is not JSON');
            }
            
            // Parse JSON - response.json() returns a Promise
            return response.json();
        }

        // Enhanced JSON fetcher with validation
        async function fetchValidatedJSON(url, schema = null) {
            try {
                const response = await fetch(url);
                
                if (!response.ok) {
                    // Try to get error message from response body
                    let errorMessage;
                    try {
                        const errorData = await response.json();
                        errorMessage = errorData.message || errorData.error || 'Unknown error';
                    } catch {
                        errorMessage = response.statusText;
                    }
                    throw new Error(`API Error: ${errorMessage}`);
                }
                
                const data = await response.json();
                
                // Optional schema validation
                if (schema && typeof schema === 'function') {
                    if (!schema(data)) {
                        throw new Error('Response validation failed');
                    }
                }
                
                return { data, response };
                
            } catch (error) {
                if (error instanceof SyntaxError) {
                    throw new Error('Invalid JSON in response');
                }
                throw error;
            }
        }

        // Streaming large JSON responses
        async function fetchStreamingJSON(url) {
            const response = await fetch(url);
            const reader = response.body.getReader();
            const decoder = new TextDecoder();
            let buffer = '';
            
            while (true) {
                const { done, value } = await reader.read();
                if (done) break;
                
                buffer += decoder.decode(value, { stream: true });
                
                // Process complete JSON objects if streaming NDJSON
                const lines = buffer.split('\n');
                buffer = lines.pop(); // Keep incomplete line
                
                for (const line of lines) {
                    if (line.trim()) {
                        yield JSON.parse(line);
                    }
                }
            }
        }

        // Usage example
        fetchBtn.addEventListener('click', async () => {
            resultDiv.innerHTML = 'Loading...';
            
            try {
                const { data, response } = await fetchValidatedJSON(
                    'https://jsonplaceholder.typicode.com/users'
                );
                
                // data is already parsed JavaScript object/array
                console.log('Parsed data:', data);
                console.log('Response headers:', [...response.headers.entries()]);
                
                resultDiv.innerHTML = `
                    <h3>Fetched ${data.length} users</h3>
                    <ul>
                        ${data.slice(0, 3).map(user => `
                            <li>${user.name} (${user.email})</li>
                        `).join('')}
                    </ul>
                `;
                
            } catch (error) {
                resultDiv.innerHTML = `<p style="color: red;">Error: ${error.message}</p>`;
            }
        });

        // Important: response.json() can only be called once
        // The response body stream is consumed after parsing
        async function demonstrateSingleConsume() {
            const response = await fetch('https://api.github.com/zen');
            
            const data1 = await response.json(); // Works
            // const data2 = await response.json(); // Error: body stream already read
            
            // To reuse, clone the response first
            const response2 = response.clone();
            const text = await response2.text();
        }
    </script>
</body>
</html>
```
**Why**: `response.json()` asynchronously parses JSON from a fetch response. It's safer than `JSON.parse()` because it handles the stream internally. Remember: response body can only be consumed once unless cloned.

## 9. [html5_fullscreen_api] Fullscreen Mode Manager
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Fullscreen API</title>

        }

        #video-container:fullscreen {
            width: 100vw;
            height: 100vh;
        }
        #video-container:-webkit-full-screen {
            width: 100vw;
            height: 100vh;
        }
        /* Hide controls initially in fullscreen */
        #video-container:fullscreen #controls {
            opacity: 0;
            transition: opacity 0.3s;
        }
        #video-container:fullscreen:hover #controls {
            opacity: 1;
        }
    </style>
</head>
<body>
    <div id="video-container">
        <video id="video" poster="https://via.placeholder.com/640x360">
            <source src="video.mp4" type="video/mp4">
        </video>
        <div id="controls">
            <button id="fullscreen-btn">⛶ Fullscreen</button>
            <button id="exit-fullscreen-btn">Exit Fullscreen</button>
            <span id="fullscreen-status"></span>
        </div>
    </div>

    <script>
        // PROBLEM: Control fullscreen mode programmatically
        // TIME: O(1) SPACE: O(1)

        const container = document.getElementById('video-container');
        const video = document.getElementById('video');
        const fullscreenBtn = document.getElementById('fullscreen-btn');
        const exitBtn = document.getElementById('exit-fullscreen-btn');
        const status = document.getElementById('fullscreen-status');

        // Cross-browser fullscreen methods
        const fullscreenAPI = {
            requestFullscreen: document.fullscreenEnabled ? 'requestFullscreen' :
                               document.webkitFullscreenEnabled ? 'webkitRequestFullscreen' :
                               document.mozFullScreenEnabled ? 'mozRequestFullScreen' :
                               document.msFullscreenEnabled ? 'msRequestFullscreen' : null,
            
            exitFullscreen: document.exitFullscreen ? 'exitFullscreen' :
                           document.webkitExitFullscreen ? 'webkitExitFullscreen' :
                           document.mozCancelFullScreen ? 'mozCancelFullScreen' :
                           document.msExitFullscreen ? 'msExitFullscreen' : null,
            
            fullscreenElement: document.fullscreenElement !== undefined ? 'fullscreenElement' :
                              document.webkitFullscreenElement !== undefined ? 'webkitFullscreenElement' :
                              document.mozFullScreenElement !== undefined ? 'mozFullScreenElement' :
                              document.msFullscreenElement !== undefined ? 'msFullscreenElement' : null,
            
            fullscreenchange: 'onfullscreenchange' in document ? 'fullscreenchange' :
                             'onwebkitfullscreenchange' in document ? 'webkitfullscreenchange' :
                             'onmozfullscreenchange' in document ? 'mozfullscreenchange' :
                             'onmsfullscreenchange' in document ? 'MSFullscreenChange' : null
        };

        // Request fullscreen for element
        function enterFullscreen(element) {
            if (!document.fullscreenEnabled && !document.webkitFullscreenEnabled) {
                alert('Fullscreen not supported');
                return;
            }

            const promise = element.requestFullscreen 
                ? element.requestFullscreen()
                : element.webkitRequestFullscreen 
                ? element.webkitRequestFullscreen()
                : element.mozRequestFullScreen()
                ? element.mozRequestFullScreen()
                : element.msRequestFullscreen();

            return promise; // Returns a Promise
        }

        // Exit fullscreen
        function exitFullscreen() {
            const promise = document.exitFullscreen 
                ? document.exitFullscreen()
                : document.webkitExitFullscreen 
                ? document.webkitExitFullscreen()
                : document.mozCancelFullScreen()
                ? document.mozCancelFullScreen()
                : document.msExitFullscreen();

            return promise;
        }

        // Toggle fullscreen
        function toggleFullscreen(element) {
            if (isFullscreen()) {
                return exitFullscreen();
            } else {
                return enterFullscreen(element);
            }
        }

        // Check if in fullscreen
        function isFullscreen() {
            return !!(document.fullscreenElement || 
                     document.webkitFullscreenElement ||
                     document.mozFullScreenElement ||
                     document.msFullscreenElement);
        }

        // Get current fullscreen element
        function getFullscreenElement() {
            return document.fullscreenElement ||
                   document.webkitFullscreenElement ||
                   document.mozFullScreenElement ||
                   document.msFullscreenElement;
        }

        // Listen for fullscreen changes
        document.addEventListener('fullscreenchange', updateStatus);
        document.addEventListener('webkitfullscreenchange', updateStatus);
        document.addEventListener('mozfullscreenchange', updateStatus);
        document.addEventListener('MSFullscreenChange', updateStatus);

        function updateStatus() {
            if (isFullscreen()) {
                status.textContent = '🟢 Fullscreen';
                console.log('Entered fullscreen:', getFullscreenElement());
            } else {
                status.textContent = '🔴 Windowed';
                console.log('Exited fullscreen');
            }
        }

        // Button handlers
        fullscreenBtn.addEventListener('click', () => {
            enterFullscreen(container);
        });

        exitBtn.addEventListener('click', () => {
            exitFullscreen();
        });

        // Double-click to toggle
        video.addEventListener('dblclick', () => {
            toggleFullscreen(container);
        });

        // ESC key exits fullscreen automatically (browser behavior)
    </script>
</body>
</html>
```
**Why**: The Fullscreen API allows immersive experiences for videos, games, and presentations. It requires vendor prefixes for cross-browser support and user gesture activation (click/tap). Listen for `fullscreenchange` event to update UI.

```html
            logDiv.innerHTML += `[${time}] ${message}<br>`;

                log('❌ Notifications not supported');

            return Notification.permission; // 'granted', 'denied', or 'default'

        async function requestPermission() {
            if (!checkSupport()) return null;
            
            // Permission can only be requested from user gesture
            const permission = await Notification.requestPermission();
            log(`Permission: ${permission}`);
            return permission;
        }

        // Create and show notification
        function showNotification(title, options = {}) {
            if (Notification.permission !== 'granted') {
                log('❌ Permission not granted');
                return null;
            }

            // Default options
            const defaultOptions = {
                body: '',                    // Body text
                icon: '',                    // Icon URL
                badge: '',                   // Badge URL (Android)
                image: '',                   // Large image URL

                requireInteraction: false,   // Stay until user interacts
                silent: false,               // No sound/vibration
                timestamp: Date.now(),       // Event timestamp
                data: null,                  // Custom data
                actions: [],                 // Action buttons
                vibrate: [200, 100, 200]     // Vibration pattern (mobile)
            };

            const notification = new Notification(title, { 
                ...defaultOptions, 
                ...options 
            });

            notification.onclick = (event) => {
                log('🖱️ Notification clicked');
                window.focus(); // Focus the window
                notification.close();
                
                // Handle custom data

                    console.log('Notification data:', options.data);

            };

            notification.onclose = (event) => {
                log('✖️ Notification closed');
            };

            notification.onerror = (event) => {
                log('❌ Notification error');
            };

            notification.onshow = (event) => {
                log('✅ Notification shown');
            };

            return notification;
        }

                } else if (event.action === 'no') {
                    log('User clicked No');
                } else {
                    log('User clicked notification body');
                }
            };
        }

            // and requires a reference to the notification object
        }
```
