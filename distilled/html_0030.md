## Fullscreen API Request

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Fullscreen API - Request Fullscreen</title>
  <style>
    /* PROBLEM: Enter fullscreen mode on an element */
    /* TIME: O(1) SPACE: O(1) */
    
    .fullscreen-container {
      width: 100%;
      height: 300px;
      background: linear-gradient(135deg, #667eea 0%, #764ba2 100%);
      display: flex;
      align-items: center;
      justify-content: center;
      border-radius: 8px;
    }
    
    .fullscreen-container:fullscreen {
      width: 100vw;
      height: 100vh;
    }
    
    button {
      padding: 12px 24px;
      font-size: 16px;
      cursor: pointer;
      background: white;
      border: none;
      border-radius: 4px;
    }
  </style>
</head>
<body>
  <div class="fullscreen-container" id="fsElement">
    <button id="fsBtn">Enter Fullscreen</button>
  </div>

  <script>
    // Get references to DOM elements
    const fsElement = document.getElementById('fsElement');
    const fsBtn = document.getElementById('fsBtn');
    
    // Check if fullscreen API is supported
    function isFullscreenSupported() {
      return document.fullscreenEnabled || 
             document.webkitFullscreenEnabled ||
             document.mozFullScreenEnabled ||
             document.msFullscreenEnabled;
    }
    
    // Request fullscreen on the element
    function requestFullscreen(element) {
      if (element.requestFullscreen) {
        return element.requestFullscreen();
      } else if (element.webkitRequestFullscreen) {
        return element.webkitRequestFullscreen();
      } else if (element.mozRequestFullScreen) {
        return element.mozRequestFullScreen();
      } else if (element.msRequestFullscreen) {
        return element.msRequestFullscreen();
      }
      return Promise.reject(new Error('Fullscreen API not supported'));
    }
    
    // Handle button click
    fsBtn.addEventListener('click', async () => {
      if (isFullscreenSupported()) {
        try {
          await requestFullscreen(fsElement);
          fsBtn.textContent = 'In Fullscreen Mode';
        } catch (err) {
          console.error('Fullscreen error:', err);
        }
      } else {
        alert('Fullscreen not supported on this browser');
      }
    });
    
    // Listen for fullscreen changes
    document.addEventListener('fullscreenchange', () => {
      if (!document.fullscreenElement) {
        fsBtn.textContent = 'Enter Fullscreen';
      }
    });
  </script>
</body>
</html>
```

**Why:** The Fullscreen API allows elements to occupy the entire screen, useful for videos, games, and immersive content. Handle vendor prefixes for cross-browser support.

## Page Visibility API

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Page Visibility API</title>
</head>
<body>
  <h1>Page Visibility Demo</h1>
  <p>Switch tabs to see the effect</p>
  <p id="status">Page is visible</p>
  <p>Time on page: <span id="timer">0</span> seconds</p>

  <script>
    /* PROBLEM: Detect when page becomes visible/hidden */
    /* TIME: O(1) SPACE: O(1) */
    
    let seconds = 0;
    let intervalId = null;
    const statusEl = document.getElementById('status');
    const timerEl = document.getElementById('timer');
    
    // Get current visibility state
    function getVisibilityState() {
      return document.visibilityState; // 'visible' or 'hidden'
    }
    
    // Check if page is visible
    function isPageVisible() {
      return !document.hidden;
    }
    
    // Start timer when visible
    function startTimer() {
      if (intervalId === null) {
        intervalId = setInterval(() => {
          seconds++;
          timerEl.textContent = seconds;
        }, 1000);
      }
    }
    
    // Stop timer when hidden
    function stopTimer() {
      if (intervalId !== null) {
        clearInterval(intervalId);
        intervalId = null;
      }
    }
    
    // Handle visibility change
    function handleVisibilityChange() {
      if (isPageVisible()) {
        statusEl.textContent = 'Page is visible';
        statusEl.style.color = 'green';
        startTimer();
        console.log('User returned to page');
      } else {
        statusEl.textContent = 'Page is hidden';
        statusEl.style.color = 'red';
        stopTimer();
        console.log('User left the page');
      }
    }
    
    // Add event listener for visibility change
    document.addEventListener('visibilitychange', handleVisibilityChange);
    
    // Initialize: start timer if page is visible
    if (isPageVisible()) {
      startTimer();
    }
  </script>
</body>
</html>
```

**Why:** The Page Visibility API helps optimize performance by pausing animations, videos, or expensive operations when the user isn't viewing the page. Essential for resource management.

    /* PROBLEM: Create custom error types for better error handling */
    /* TIME: O(1) SPACE: O(1) */
    
    const output = document.getElementById('output');
    const logs = [];
    
    function log(msg) {
      logs.push(msg);
      output.textContent = logs.join('\n');
    }
    
    // Base custom error class
    class ApplicationError extends Error {
      constructor(message, code) {
        super(message);
        this.name = 'ApplicationError';
        this.code = code || 'APP_ERROR';
        this.timestamp = new Date().toISOString();
        
        // Maintains proper stack trace in V8 environments
        if (Error.captureStackTrace) {
          Error.captureStackTrace(this, this.constructor);
        }
      }
      
      toJSON() {
        return {
          name: this.name,
          message: this.message,
          code: this.code,
          timestamp: this.timestamp,
          stack: this.stack
        };
      }
    }
    
    // Validation error
    class ValidationError extends ApplicationError {
      constructor(message, field) {
        super(message, 'VALIDATION_ERROR');
        this.name = 'ValidationError';
        this.field = field || null;
      }
    }
    
    // Network error
    class NetworkError extends ApplicationError {
      constructor(message, statusCode) {
        super(message, 'NETWORK_ERROR');
        this.name = 'NetworkError';
        this.statusCode = statusCode || null;
      }
    }
    
    // Authentication error
    class AuthenticationError extends ApplicationError {
      constructor(message) {
        super(message, 'AUTH_ERROR');
        this.name = 'AuthenticationError';
      }
    }
    
    // Demo usage
    function validateUser(user) {
      if (!user.email) {
        throw new ValidationError('Email is required', 'email');
      }
      if (!user.password || user.password.length < 8) {
        throw new ValidationError('Password must be at least 8 characters', 'password');
      }
      return true;
    }
    
    // Test the errors
    try {
      validateUser({ email: '', password: '123' });
    } catch (error) {
      if (error instanceof ValidationError) {
        log(`Validation failed on field: ${error.field}`);
        log(`Error: ${error.message}`);
      }
    }
    
    try {
      throw new NetworkError('Connection timeout', 408);
    } catch (error) {
      log(`\nNetwork Error (code ${error.statusCode}): ${error.message}`);
    }
    
    try {
      throw new AuthenticationError('Invalid token');
    } catch (error) {
      log(`\nAuth Error: ${error.message}`);
      log(`Error code: ${error.code}`);
    }
    
    // Serialize error to JSON

## Error Stack Trace

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Error Stack Trace</title>
</head>
<body>
  <h1>Error Stack Trace Analysis</h1>
  <pre id="stackOutput" style="background: #1e1e1e; color: #d4d4d4; padding: 20px; border-radius: 8px;"></pre>

  <script>
    /* PROBLEM: Access and analyze error stack traces */
    /* TIME: O(n) where n = stack depth SPACE: O(n) */
    
    const output = document.getElementById('stackOutput');
    
    // Function to format and display stack trace
    function displayStack(error) {
      const lines = [];
      lines.push('=== ERROR INFO ===');
      lines.push(`Name: ${error.name}`);
      lines.push(`Message: ${error.message}`);
      lines.push('\n=== STACK TRACE ===');
      
      if (error.stack) {
        // Parse stack trace
        const stackLines = error.stack.split('\n');
        
        stackLines.forEach((line, index) => {
          if (index === 0) {
            lines.push(`\x1b[31m${line}\x1b[0m`); // Red for error name
          } else {
            // Clean up and format stack line
            const match = line.match(/at\s+(.+?)\s+\((.+?):(\d+):(\d+)\)/);
            if (match) {
              const [, func, file, lineNum, col] = match;
              lines.push(`  at ${func}`);
              lines.push(`    → ${file}:${lineNum}:${col}`);
            } else {
              lines.push(`  ${line}`);
            }
          }
        });
      }
      
      output.textContent = lines.join('\n');
      return error.stack;
    }
    
    // Deep function call chain to demonstrate stack
    function level4() {
      throw new Error('Something went wrong at level 4!');
    }
    
    function level3() {
      level4();
    }
    
    function level2() {
      level3();
    }
    
    function level1() {
      level2();
    }
    
    // Capture stack without throwing
    function captureCurrentStack() {
      const err = new Error('Stack capture');
      return err.stack;
    }
    
    // Stack trace analysis utilities
    function parseStackTrace(stack) {
      const lines = stack.split('\n').slice(1);
      return lines.map(line => {
        const match = line.match(/at\s+(?:(.+?)\s+)?\(?(.+?):(\d+):(\d+)\)?/);
        if (match) {
          return {
            function: match[1] || 'anonymous',
            file: match[2],
            line: parseInt(match[3]),
            column: parseInt(match[4])
          };
        }
        return null;
      }).filter(Boolean);
    }
    
    // Demo
    try {
      level1();
    } catch (error) {
      displayStack(error);
      
      // Parse and analyze
      console.log('\nParsed stack frames:');
      const frames = parseStackTrace(error.stack);
      frames.forEach((frame, i) => {
        console.log(`${i}: ${frame.function} at line ${frame.line}`);
      });
    }
    
    // Demonstrate stack capture without throw
    console.log('\nCurrent stack (no throw):');
    console.log(captureCurrentStack());
  </script>
</body>
</html>
```

**Why:** Stack traces provide crucial debugging information showing the call sequence leading to an error. Essential for error monitoring, logging systems, and debugging production issues.

## RegExp Case-Insensitive Flag

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>RegExp Case-Insensitive Flag</title>
</head>
<body>
  <h1>RegExp Case-Insensitive Flag (i)</h1>
  <input type="text" id="searchInput" placeholder="Search text..." style="padding: 8px; width: 300px;">
  <div id="results" style="margin-top: 20px;"></div>

  <script>
    /* PROBLEM: Match patterns regardless of case */
    /* TIME: O(n*m) where n = string length, m = pattern length SPACE: O(1) */
    
    const searchInput = document.getElementById('searchInput');
    const resultsDiv = document.getElementById('results');
    
    // Sample data to search through
    const items = [
      'JavaScript is Awesome',
      'javascript tutorials',
      'JAVASCRIPT fundamentals',
      'Java is different from JavaScript',
      'TypeScript vs JavaScript',
      'Learn javascript today'
    ];
    
    // Using the 'i' flag for case-insensitive matching
    function searchItems(query) {
      if (!query.trim()) return items;
      
      // Create regex with 'i' flag
      const regex = new RegExp(query, 'i');
      
      return items.filter(item => regex.test(item));
    }
    
    // Alternative: using inline flag
    function searchWithInline(query) {
      // (?i) is not supported in JS, use flags parameter instead
      const regex = new RegExp(query, 'i');
      return items.filter(item => regex.test(item));
    }
    
    // Highlight matches
    function highlightMatch(text, query) {
      if (!query.trim()) return text;
      const regex = new RegExp(`(${query})`, 'gi');
      return text.replace(regex, '<mark>$1</mark>');
    }
    
    // Display results
    function displayResults(query) {
      const matches = searchItems(query);
      
      resultsDiv.innerHTML = matches.length > 0
        ? matches.map(item => `<p>${highlightMatch(item, query)}</p>`).join('')
        : '<p>No matches found</p>';
    }
    
    // Live search
    searchInput.addEventListener('input', (e) => {
      displayResults(e.target.value);
    });
    
    // Initial display
    displayResults('');
    
    // Demo various 'i' flag usages
    console.log('=== Case-Insensitive Flag Demo ===\n');
    
    // Literal notation with flag
    const pattern1 = /hello/i;
    console.log('/hello/i.test("HELLO"):', pattern1.test('HELLO')); // true
    console.log('/hello/i.test("HeLLo"):', pattern1.test('HeLLo')); // true
    
    // Constructor with flag
    const pattern2 = new RegExp('world', 'i');
    console.log('new RegExp("world", "i").test("WORLD"):', pattern2.test('WORLD')); // true
    
    // Multiple flags combined
    const pattern3 = /test/gi; // global + case-insensitive
    const text = 'Test TEST test TeSt';
    console.log('Matches with /gi:', text.match(pattern3)); // all 4
    
    // Case-insensitive replace
    const result = 'JavaScript JAVASCRIPT javascript'.replace(/javascript/gi, 'JS');
    console.log('Replaced:', result); // 'JS JS JS'
  </script>
</body>
</html>
```

**Why:** The `i` flag enables case-insensitive matching, essential for user-facing search, validation, and text processing where case shouldn't affect results.

## RegExp Quantifiers

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>RegExp Quantifiers</title>
</head>
<body>
  <h1>RegExp Quantifiers</h1>
  <div id="demo"></div>

  <script>
    /* PROBLEM: Match patterns with specific repetition counts */
    /* TIME: O(n) SPACE: O(1) */
    
    const demo = document.getElementById('demo');
    const results = [];
    
    function testPattern(name, pattern, testStrings) {
      const output = [`\n=== ${name} ===`];
      output.push(`Pattern: ${pattern}`);
      
      testStrings.forEach(str => {
        const regex = new RegExp(pattern);
        const matches = str.match(regex);
        output.push(`  "${str}" → ${matches ? '✓ Match: ' + matches[0] : '✗ No match'}`);
      });
      
      results.push(output.join('\n'));
    }
    
    // * = 0 or more
    testPattern('Asterisk * (0 or more)', 'ab*c', [
      'ac',       // a + 0 b's + c
      'abc',      // a + 1 b + c
      'abbbc',    // a + 3 b's + c
      'ab',       // no c
      'aabbcc'    // matches 'abbc'
    ]);
    
    // + = 1 or more
    testPattern('Plus + (1 or more)', 'ab+c', [
      'ac',       // no b
      'abc',      // 1 b
      'abbbc',    // 3 b's
      'ab'        // no c
    ]);
    
    // ? = 0 or 1 (optional)
    testPattern('Question ? (0 or 1)', 'colou?r', [
      'color',    // without u
      'colour',   // with u
      'colouur'   // 2 u's
    ]);
    
    // {n} = exactly n
    testPattern('{n} (exactly n)', 'a{3}', [
      'aa',       // 2 a's
      'aaa',      // exactly 3
      'aaaa'      // 4 a's (matches first 3)
    ]);
    
    // {n,} = n or more
    testPattern('{n,} (n or more)', 'a{2,}', [
      'a',        // 1 a
      'aa',       // 2 a's
      'aaaaa'     // 5 a's
    ]);
    
    // {n,m} = between n and m
    testPattern('{n,m} (between n and m)', 'a{2,4}', [
      'a',        // 1 a
      'aa',       // 2 a's
      'aaa',      // 3 a's
      'aaaaa'     // 5 a's (matches first 4)
    ]);
    
    // Practical examples
    results.push('\n=== PRACTICAL EXAMPLES ===');
    
    // Phone number: 3-3-4 digits
    const phone = /^\d{3}-\d{3}-\d{4}$/;
    results.push(`Phone (\\d{3}-\\d{3}-\\d{4}):`);
    results.push(`  "123-456-7890" → ${phone.test('123-456-7890')}`);
    results.push(`  "12-345-6789" → ${phone.test('12-345-6789')}`);
    
    // Username: 3-16 alphanumeric
    const username = /^[a-zA-Z0-9]{3,16}$/;
    results.push(`\nUsername (3-16 chars):`);
    results.push(`  "ab" → ${username.test('ab')}`);
    results.push(`  "john_doe" → ${username.test('john_doe')}`);
    
    // Password: at least 8 chars
    const password = /^.{8,}$/;
    results.push(`\nPassword (8+ chars):`);
    results.push(`  "pass" → ${password.test('pass')}`);
    results.push(`  "password123" → ${password.test('password123')}`);
    
    // HTML tag
    const htmlTag = /<\/?[a-z][a-z0-9]*>/gi;
    const html = '<div><p>Hello</p></div>';
    results.push(`\nHTML tags in "${html}":`);
    results.push(`  ${html.match(htmlTag).join(', ')}`);
    
    demo.innerHTML = `<pre>${results.join('\n')}</pre>`;
  </script>
</body>
</html>
```

**Why:** Quantifiers control repetition in patterns, enabling precise matching of phone numbers, passwords, usernames, and any format with specific length requirements.
