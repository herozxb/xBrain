<!--

## html_video_preload_metadata

```html

  PROBLEM: How to balance video loading performance vs bandwidth
  TIME: O(1) SPACE: O(metadata_size)
-->

<video preload="metadata" controls>
  <source src="video.mp4" type="video/mp4">
</video>

<!-- Load entire video -->
<video preload="auto" controls>
  <source src="video.mp4" type="video/mp4">

  <source src="video.mp4" type="video/mp4">
</video>

console.log('Preload strategy:', video.preload);
console.log('Duration:', video.duration); // Available with metadata

// Listen for metadata load

  console.log('Metadata loaded:', {
    duration: video.duration,
    width: video.videoWidth,
    height: video.videoHeight
  });

</script>

<!-- 
  WHY: Preload strategies optimize the balance between user experience and bandwidth.
  "metadata" loads minimal data (duration, dimensions) without downloading video content.
  "auto" lets browser decide (usually loads entire video).
  "none" delays all loading until user interaction.
-->
```

**Why**: Preloading metadata provides instant duration/size information without consuming bandwidth for video content, enabling responsive UI while conserving resources.

## html_aria_role_link

```html
<!-- 
  PROBLEM: How to create accessible custom interactive elements
  TIME: O(1) SPACE: O(1)
-->
<!-- Standard link (native semantics) -->
<a href="/about">About Us</a>

<!-- Custom element with link role -->
<span 
  role="link" 
  tabindex="0"
  onclick="navigate('/about')"
  onkeydown="handleKeyNav(event, '/about')"
  class="custom-link">
  About Us
</span>

<!-- Div acting as link -->
<div 
  role="link"
  tabindex="0"
  onclick="window.location.href='/contact'"
  onkeypress="if(event.key==='Enter')window.location.href='/contact'"
  style="cursor: pointer;">
  Contact Page
</div>

<script>
function navigate(url) {
  window.location.href = url;
}

function handleKeyNav(event, url) {
  // Links should respond to Enter and Space
  if (event.key === 'Enter' || event.key === ' ') {
    event.preventDefault();
    navigate(url);
  }
}

// Verify ARIA role
const customLink = document.querySelector('[role="link"]');
console.log('Role:', customLink.getAttribute('role')); // "link"
console.log('Tabbable:', customLink.tabIndex >= 0); // true
</script>

<!-- 
  WHY: role="link" tells screen readers this element behaves like a link.
  MUST include: tabindex="0" (keyboard accessible), click handler, keyboard handler.
  Only use when you can't use native <a> element for technical reasons.
  Native elements are always preferred over ARIA roles.
-->
```

**Why**: ARIA link role enables custom elements to be announced as links by screen readers, but requires manual implementation of all link behaviors (keyboard navigation, focus management) that native links provide automatically.

## html_meta_charset

```html
<!-- 
  PROBLEM: How to ensure proper character encoding for international text
  TIME: O(1) SPACE: O(1)
-->
<!DOCTYPE html>
<html lang="en">
<head>
  <!-- Must be first in <head> for security and correctness -->
  <meta charset="UTF-8">
  
  <!-- Other meta tags -->
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Character Encoding Demo</title>
</head>
<body>
  <!-- UTF-8 supports all characters -->
  <p>English: Hello World</p>
  <p>Chinese: 你好世界</p>
  <p>Arabic: مرحبا بالعالم</p>

</body>
</html>

<script>
// Check document encoding
console.log('Character set:', document.characterSet); // "UTF-8"
console.log('Input encoding:', document.inputEncoding); // "UTF-8" (deprecated)

// Verify meta tag

console.log('Charset:', charsetMeta.getAttribute('charset')); // "UTF-8"
</script>

<!-- 
  WHY: UTF-8 is the universal character encoding standard for the web.
  Must be declared in first 1024 bytes to prevent XSS attacks via encoding mismatches.
  Supports all 1,112,064 Unicode code points efficiently.
  Without it, browsers may misinterpret special characters (mojibake).
-->
```

**Why**: Declaring UTF-8 charset prevents encoding vulnerabilities, ensures proper rendering of international characters and symbols, and is required for modern web standards compliance.

## html_link_rel_preconnect

```html
<!-- 
  PROBLEM: How to reduce latency for third-party origins
  TIME: O(1) SPACE: O(1)
-->
<head>
  <!-- Early DNS + TCP + TLS handshake for critical origins -->
  <link rel="preconnect" href="https://fonts.googleapis.com">
  <link rel="preconnect" href="https://fonts.gstatic.com" crossorigin>
  
  <!-- Preconnect to CDN -->
  <link rel="preconnect" href="https://cdn.example.com">
  
  <!-- Preconnect to API endpoint -->
  <link rel="preconnect" href="https://api.example.com">
  
  <!-- Then use the resources -->
  <link rel="stylesheet" href="https://fonts.googleapis.com/css2?family=Roboto">
</head>

<script>
// Measure connection timing
const observer = new PerformanceObserver((list) => {
  list.getEntries().forEach((entry) => {
    console.log(`${entry.name}:`, {
      dnsTime: entry.domainLookupEnd - entry.domainLookupStart,
      tcpTime: entry.connectEnd - entry.connectStart,
      totalTime: entry.duration
    });
  });
});

observer.observe({ entryTypes: ['resource'] });

// Preconnect via JavaScript (less effective than HTML)
const link = document.createElement('link');
link.rel = 'preconnect';
link.href = 'https://example.com';
document.head.appendChild(link);
</script>

<!-- 
  WHY: Preconnect initiates early connection to origins, saving 100-300ms.
  Useful for: CDNs, fonts, APIs, analytics services.
  crossorigin attribute needed for CORS requests.
  Use sparingly (3-4 max) to avoid wasting bandwidth on unused connections.
-->
```

**Why**: Preconnect eliminates connection latency by establishing DNS, TCP, and TLS handshakes before resources are requested, significantly improving load times for critical third-party resources.

## js_promise_race

```html
<!DOCTYPE html>
<html>
<head>
  <title>Promise.race() Demo</title>
</head>
<body>
  <script>
    /*
      PROBLEM: How to handle multiple promises and use the first settled result
      TIME: O(n) SPACE: O(n) where n = number of promises
    */

    // Example 1: Timeout pattern
    function fetchWithTimeout(url, timeoutMs) {
      const fetchPromise = fetch(url).then(r => r.json());
      
      const timeoutPromise = new Promise((_, reject) => {
        setTimeout(() => reject(new Error('Timeout')), timeoutMs);
      });
      
      // Returns whichever settles first
      return Promise.race([fetchPromise, timeoutPromise]);
    }

    // Usage
    fetchWithTimeout('https://api.example.com/data', 5000)
      .then(data => console.log('Success:', data))
      .catch(err => console.error('Error:', err.message));

    // Example 2: Multiple APIs - use fastest response
    function fetchFromMultipleSources(sources) {
      const promises = sources.map(url => 
        fetch(url)
          .then(r => r.json())
          .catch(err => ({ error: err, url }))
      );
      
      return Promise.race(promises);
    }

    // Example 3: Racing async operations
    const delay = (ms, value) => 
      new Promise(resolve => setTimeout(() => resolve(value), ms));

    Promise.race([
      delay(1000, 'fast'),
      delay(2000, 'medium'),
      delay(3000, 'slow')
    ]).then(result => {
      console.log('Winner:', result); // "fast" after 1 second
    });

    // Example 4: Cancellation pattern
    let cancelled = false;
    const cancelPromise = new Promise((_, reject) => {
      if (cancelled) reject(new Error('Cancelled'));
    });

    Promise.race([
      fetch('/api/data'),
      cancelPromise
    ]).catch(err => console.log('Operation cancelled'));

    // Performance observation
    console.time('Promise.race');
    Promise.race([
      fetch('https://jsonplaceholder.typicode.com/posts/1'),
      fetch('https://jsonplaceholder.typicode.com/users/1')
    ])
      .then(r => r.json())
      .then(data => {
        console.timeEnd('Promise.race');
        console.log('First response:', data);
      });
  </script>
</body>
</html>

<!--
  WHY: Promise.race() returns a promise that settles when the first input promise settles.
  Use cases: timeouts, fallback APIs, cancellation, competitive fetching.
  Note: Returns first settled (resolved OR rejected), not just first resolved.
  Other promises continue running (not cancelled).
-->
```

**Why**: Promise.race() enables timeout patterns, competitive resource fetching, and cancellation logic by returning the first settled promise result, regardless of success or failure.

## js_array_flat

```html
<!DOCTYPE html>
<html>
<head>
  <title>Array.prototype.flat() Demo</title>
</head>
<body>
  <script>
    /*
      PROBLEM: How to flatten nested arrays
      TIME: O(n) SPACE: O(n) where n = total elements
    */

    // Basic flattening
    const nested = [1, [2, 3], [4, [5, 6]]];
    
    console.log(nested.flat());      // [1, 2, 3, 4, [5, 6]] - depth 1 (default)
    console.log(nested.flat(2));     // [1, 2, 3, 4, 5, 6] - depth 2
    console.log(nested.flat(Infinity)); // [1, 2, 3, 4, 5, 6] - fully flat

    // Real-world examples
    // Example 1: Flatten API responses
    const apiResponse = {
      users: [
        { name: 'Alice', tags: [['admin', 'user'], ['premium']] },
        { name: 'Bob', tags: [['user'], ['trial']] }
      ]
    };

    const allTags = apiResponse.users
      .map(u => u.tags)
      .flat(2);
    
    console.log('All tags:', allTags);
    // ["admin", "user", "premium", "user", "trial"]

    // Example 2: Remove empty slots (sparse arrays)
    const sparse = [1, , 3, , , 6];
    console.log(sparse.flat()); // [1, 3, 6]

    // Example 3: Flatten and map (use flatMap instead)
    const sentences = ['Hello World', 'Hi There'];
    const words = sentences
      .map(s => s.split(' '))
      .flat();
    
    console.log(words); // ["Hello", "World", "Hi", "There"]

    // Better: use flatMap
    const words2 = sentences.flatMap(s => s.split(' '));
    console.log(words2); // ["Hello", "World", "Hi", "There"]

    // Example 4: Deep flattening utility
    function deepFlatten(arr) {
      return arr.flat(Infinity);
    }

    const deeplyNested = [1, [2, [3, [4, [5]]]]];
    console.log(deepFlatten(deeplyNested)); // [1, 2, 3, 4, 5]

    // Example 5: Flatten with filtering
    const mixed = [1, [2, null], [3, [undefined, 4]]];
    const clean = mixed
      .flat(2)
      .filter(x => x != null);
    
    console.log(clean); // [1, 2, 3, 4]

    // Performance comparison
    const largeNested = Array(1000).fill([1, [2, [3]]]);
    
    console.time('flat(2)');
    largeNested.flat(2);
    console.timeEnd('flat(2)');

    console.time('flat(Infinity)');
    largeNested.flat(Infinity);
    console.timeEnd('flat(Infinity)');
  </script>
</body>
</html>

<!--
  WHY: Array.flat() simplifies nested array structures without manual recursion.
  Default depth is 1, can specify any depth or Infinity for complete flattening.
  Removes empty slots (holes) in arrays.
  Usually paired with map() -> prefer flatMap() for map-then-flatten pattern.
-->
```

**Why**: Array.flat() provides declarative array flattening with configurable depth, eliminating recursive functions and handling sparse arrays cleanly, while flatMap() offers optimized map+flatten operations.

## js_string_padStart

```html
<!DOCTYPE html>
<html>
<head>
  <title>String.prototype.padStart() Demo</title>
</head>
<body>
  <h2>String.padStart() Examples</h2>
  <div id="output"></div>

  <script>
    /*
      PROBLEM: How to pad strings to a fixed length
      TIME: O(n) SPACE: O(n) where n = target length
    */

    const output = document.getElementById('output');
    
    // Basic padding
    const str = '5';
    console.log(str.padStart(3, '0'));    // "005"
    console.log(str.padStart(5, '0'));    // "00005"
    console.log(str.padStart(2, '0'));    // "05"
    console.log(str.padStart(1, '0'));    // "5" (already length 1)

    // Real-world examples
    // Example 1: Format numbers with leading zeros
    function formatNumber(num, digits = 2) {
      return String(num).padStart(digits, '0');
    }

    console.log(formatNumber(5));      // "05"
    console.log(formatNumber(123, 5)); // "00123"

    // Example 2: Format time
    function formatTime(hours, minutes, seconds) {
      return [
        String(hours).padStart(2, '0'),
        String(minutes).padStart(2, '0'),
        String(seconds).padStart(2, '0')
      ].join(':');
    }

    console.log(formatTime(9, 5, 3));   // "09:05:03"
    console.log(formatTime(14, 30, 0)); // "14:30:00"

    // Example 3: Format currency
    function formatCurrency(amount, width = 10) {
      return '$' + String(amount.toFixed(2)).padStart(width);
    }

    console.log(formatCurrency(5.5));      // "$      5.50"
    console.log(formatCurrency(1234.56));  // "$   1234.56"

    // Example 4: ID generation
    function generateID(number) {
      const prefix = 'ID';
      const paddedNum = String(number).padStart(6, '0');
      return prefix + paddedNum;
    }

    console.log(generateID(42));    // "ID000042"
    console.log(generateID(99999)); // "ID099999"

    // Example 5: Table alignment in console
    function createTableRow(items, columnWidth = 10) {
      return items
        .map(item => String(item).padStart(columnWidth))
        .join(' | ');
    }

    const table = [
      createTableRow(['Name', 'Score', 'Rank']),
      createTableRow(['Alice', 95, 1]),
      createTableRow(['Bob', 87, 3]),
      createTableRow(['Charlie', 92, 2])
    ].join('\n');

    console.log(table);
    /*
         Name |     Score |      Rank
       Alice |        95 |         1
         Bob |        87 |         3
     Charlie |        92 |         2
    */

    // Example 6: Custom padding characters
    console.log('hello'.padStart(10, '.'));   // ".....hello"
    console.log('test'.padStart(8, '=-'));    // "-=-=-test"
    console.log('abc'.padStart(6, '123'));    // "123abc"

    // Example 7: Binary/hex formatting
    function formatBinary(num, bits = 8) {
      return num.toString(2).padStart(bits, '0');
    }

    function formatHex(num, digits = 4) {
      return '0x' + num.toString(16).toUpperCase().padStart(digits, '0');
    }

    console.log(formatBinary(5));     // "00000101"
    console.log(formatHex(255));      // "0x00FF"

    // Display in HTML
    output.innerHTML = `
      <pre>
Time Format: ${formatTime(9, 5, 3)}
ID Format: ${generateID(42)}
Binary: ${formatBinary(5)}
Hex: ${formatHex(255)}
      </pre>
    `;
  </script>
</body>
</html>

<!--
  WHY: padStart() adds characters to the beginning of strings to reach target length.
  Perfect for: formatting numbers, times, IDs, aligning text, binary/hex display.
  Parameters: targetLength (required), padString (optional, defaults to space).
  If string already >= targetLength, returns unchanged string.
  Counterpart: padEnd() adds to the end.
-->
```

**Why**: String.padStart() simplifies string formatting for fixed-width displays, number formatting with leading zeros, and text alignment without manual loop or slice operations.
