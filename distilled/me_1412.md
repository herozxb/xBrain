## Tim Sort
```python
# PROBLEM: Implement Tim Sort, Python's hybrid sorting algorithm
# TIME: O(n log n) worst, O(n) best SPACE: O(n)

def tim_sort(arr):
    """
    Tim Sort is a hybrid sorting algorithm derived from merge sort
    and insertion sort. It's optimized for real-world data with
    existing order patterns.
    
    Key features:
    - Uses insertion sort for small runs (minrun)
    - Merges runs using merge sort
    - Adaptive: exploits existing order in data
    - Stable sorting algorithm
    """
    
    MIN_RUN = 32  # Minimum run size for insertion sort
    
    def insertion_sort(arr, left, right):
        """Sort small subarrays with insertion sort."""
        for i in range(left + 1, right + 1):
            key = arr[i]
            j = i - 1
            # Shift elements greater than key to the right
            while j >= left and arr[j] > key:
                arr[j + 1] = arr[j]
                j -= 1
            arr[j + 1] = key
    
    def merge(arr, left, mid, right):
        """Merge two sorted subarrays."""
        left_arr = arr[left:mid + 1]
        right_arr = arr[mid + 1:right + 1]
        
        i = j = 0
        k = left
        
        # Merge while both arrays have elements
        while i < len(left_arr) and j < len(right_arr):
            if left_arr[i] <= right_arr[j]:
                arr[k] = left_arr[i]
                i += 1
            else:
                arr[k] = right_arr[j]
                j += 1
            k += 1
        
        # Copy remaining elements
        while i < len(left_arr):
            arr[k] = left_arr[i]
            i += 1
            k += 1
        
        while j < len(right_arr):
            arr[k] = right_arr[j]
            j += 1
            k += 1
    
    def calc_min_run(n):
        """Calculate minimum run size (32-64)."""
        min_run = MIN_RUN
        while n >= min_run:
            n //= 2
        return n + 1
    
    n = len(arr)
    if n < 2:
        return arr
    
    min_run = calc_min_run(n)
    
    # Sort individual runs with insertion sort
    for start in range(0, n, min_run):
        end = min(start + min_run - 1, n - 1)
        insertion_sort(arr, start, end)
    
    # Merge runs in increasing sizes
    size = min_run
    while size < n:
        for left in range(0, n, 2 * size):
            mid = min(n - 1, left + size - 1)
            right = min(n - 1, left + 2 * size - 1)
            if mid < right:
                merge(arr, left, mid, right)
        size *= 2
    
    return arr
```
**Why**: Python's built-in sort uses this algorithm; hybrid approach balances simplicity and performance.

## Red-Black Tree
```python
# PROBLEM: Implement a self-balancing Red-Black Tree
# TIME: O(log n) for insert/search/delete SPACE: O(n)

class Color:
    """Node colors for Red-Black Tree properties."""
    RED = 0
    BLACK = 1

class RBNode:
    """Red-Black Tree node with color attribute."""
    
    def __init__(self, value):
        self.value = value
        self.color = Color.RED  # New nodes are always red
        self.left = None
        self.right = None
        self.parent = None

class RedBlackTree:
    """
    Red-Black Tree is a self-balancing BST with these properties:
    1. Every node is either red or black
    2. Root is always black
    3. Red nodes can't have red children (no consecutive red)
    4. Every path from root to leaf has same number of black nodes
    5. Leaves (None) are considered black
    
    Guarantees O(log n) operations through balance maintenance.
    """
    
    def __init__(self):
        self.NIL = RBNode(None)
        self.NIL.color = Color.BLACK
        self.root = self.NIL
    
    def insert(self, value):
        """Insert value and maintain Red-Black properties."""
        node = RBNode(value)
        node.left = self.NIL
        node.right = self.NIL
        
        parent = None
        current = self.root
        
        # Standard BST insertion
        while current != self.NIL:
            parent = current
            if node.value < current.value:
                current = current.left
            else:
                current = current.right
        
        node.parent = parent
        
        if parent is None:
            self.root = node
        elif node.value < parent.value:
            parent.left = node
        else:
            parent.right = node
        
        node.color = Color.RED
        self._fix_insert(node)
    
    def _fix_insert(self, node):
        """Fix Red-Black violations after insertion."""
        while node != self.root and node.parent.color == Color.RED:
            if node.parent == node.parent.parent.left:
                uncle = node.parent.parent.right
                
                if uncle.color == Color.RED:
                    # Case 1: Uncle is red - recolor
                    node.parent.color = Color.BLACK
                    uncle.color = Color.BLACK
                    node.parent.parent.color = Color.RED
                    node = node.parent.parent
                else:
                    if node == node.parent.right:
                        # Case 2: Left rotation needed
                        node = node.parent
                        self._left_rotate(node)
                    
                    # Case 3: Recolor and right rotate
                    node.parent.color = Color.BLACK
                    node.parent.parent.color = Color.RED
                    self._right_rotate(node.parent.parent)
            else:
                # Mirror cases for right side
                uncle = node.parent.parent.left
                
                if uncle.color == Color.RED:
                    node.parent.color = Color.BLACK
                    uncle.color = Color.BLACK
                    node.parent.parent.color = Color.RED
                    node = node.parent.parent
                else:
                    if node == node.parent.left:
                        node = node.parent
                        self._right_rotate(node)
                    
                    node.parent.color = Color.BLACK
                    node.parent.parent.color = Color.RED
                    self._left_rotate(node.parent.parent)
        
        self.root.color = Color.BLACK
    
    def _left_rotate(self, x):
        """Left rotation around node x."""
        y = x.right
        x.right = y.left
        
        if y.left != self.NIL:
            y.left.parent = x
        
        y.parent = x.parent
        
        if x.parent is None:
            self.root = y
        elif x == x.parent.left:
            x.parent.left = y
        else:
            x.parent.right = y
        
        y.left = x
        x.parent = y
    
    def _right_rotate(self, x):
        """Right rotation around node x."""
        y = x.left
        x.left = y.right
        
        if y.right != self.NIL:
            y.right.parent = x
        
        y.parent = x.parent
        
        if x.parent is None:
            self.root = y
        elif x == x.parent.right:
            x.parent.right = y
        else:
            x.parent.left = y
        
        y.right = x
        x.parent = y
    
    def search(self, value):
        """Search for value in tree, returns node or None."""
        current = self.root
        while current != self.NIL:
            if value == current.value:
                return current
            elif value < current.value:
                current = current.left
            else:
                current = current.right
        return None
```
**Why**: Self-balancing BST used in many systems (Linux scheduler, Java TreeMap); guarantees logarithmic operations.

## LFU Cache
```python
# PROBLEM: Implement LFU (Least Frequently Used) cache with O(1) operations
# TIME: O(1) for get/put SPACE: O(capacity)

from collections import defaultdict, OrderedDict

class LFUCache:
    """
    LFU Cache evicts the least frequently used item when full.
    Uses frequency counts and LRU as tiebreaker for same frequency.
    
    Structure:
    - key_node: maps key -> (value, freq)
    - freq_lists: maps frequency -> OrderedDict of keys
    - min_freq: tracks minimum frequency for eviction
    """
    
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.key_node = {}  # key -> (value, frequency)
        self.freq_lists = defaultdict(OrderedDict)  # freq -> {key: value}
        self.min_freq = 0
    
    def get(self, key: int) -> int:
        """
        Get value and update frequency.
        Move key to next frequency list.
        """
        if key not in self.key_node:
            return -1
        
        value, freq = self.key_node[key]
        
        # Remove from current frequency list
        del self.freq_lists[freq][key]
        
        # Update minimum frequency if needed
        if freq == self.min_freq and not self.freq_lists[freq]:
            self.min_freq += 1
        
        # Add to next frequency list
        new_freq = freq + 1
        self.key_node[key] = (value, new_freq)
        self.freq_lists[new_freq][key] = value
        
        return value
    
    def put(self, key: int, value: int) -> None:
        """
        Put value into cache.
        Evict LFU item if capacity exceeded.
        """
        if self.capacity <= 0:
            return
        
        # Update existing key
        if key in self.key_node:
            _, freq = self.key_node[key]
            del self.freq_lists[freq][key]
            
            if freq == self.min_freq and not self.freq_lists[freq]:
                self.min_freq += 1
            
            new_freq = freq + 1
            self.key_node[key] = (value, new_freq)
            self.freq_lists[new_freq][key] = value
            return
        
        # Evict if at capacity
        if len(self.key_node) >= self.capacity:
            # Remove LFU+LRU item (first in OrderedDict)
            evict_key, _ = self.freq_lists[self.min_freq].popitem(last=False)
            del self.key_node[evict_key]
        
        # Insert new key
        self.min_freq = 1
        self.key_node[key] = (value, 1)
        self.freq_lists[1][key] = value
```
**Why**: Advanced caching strategy; used when access patterns favor frequently accessed data over recently accessed.

## Composite Pattern
```python
# PROBLEM: Compose objects into tree structures to represent part-whole hierarchies
# TIME: O(n) for operations SPACE: O(n) for structure

from abc import ABC, abstractmethod
from typing import List

class FileSystemComponent(ABC):
    """
    Component interface for composite pattern.
    Defines common operations for both files and directories.
    """
    
    @abstractmethod
    def display(self, indent: str = "") -> str:
        """Display the component."""
        pass
    
    @abstractmethod
    def get_size(self) -> int:
        """Get size of component."""
        pass

class File(FileSystemComponent):
    """Leaf component - represents individual file."""
    
    def __init__(self, name: str, size: int):
        self.name = name
        self.size = size
    
    def display(self, indent: str = "") -> str:
        return f"{indent}📄 {self.name} ({self.size}KB)"
    
    def get_size(self) -> int:
        return self.size

class Directory(FileSystemComponent):
    """
    Composite component - can contain files and other directories.
    Implements tree structure for part-whole hierarchy.
    """
    
    def __init__(self, name: str):
        self.name = name
        self.children: List[FileSystemComponent] = []
    
    def add(self, component: FileSystemComponent) -> None:
        """Add child component."""
        self.children.append(component)
    
    def remove(self, component: FileSystemComponent) -> None:
        """Remove child component."""
        self.children.remove(component)
    
    def display(self, indent: str = "") -> str:
        result = f"{indent}📁 {self.name}/\n"
        for child in self.children:
            result += child.display(indent + "  ") + "\n"
        return result.rstrip()
    
    def get_size(self) -> int:
        """Recursively calculate total size."""
        return sum(child.get_size() for child in self.children)

# Usage example
def build_file_system():
    """Demonstrate composite pattern with file system."""
    root = Directory("root")
    
    # Create files
    readme = File("README.md", 5)
    config = File("config.json", 2)
    main_py = File("main.py", 10)
    test_py = File("test.py", 8)
    
    # Create directories
    src = Directory("src")
    tests = Directory("tests")
    
    # Build hierarchy
    src.add(main_py)
    tests.add(test_py)
    
    root.add(readme)
    root.add(config)
    root.add(src)
    root.add(tests)
    
    return root
```
**Why**: Treats individual and composite objects uniformly; essential for UI frameworks, file systems, and organizational hierarchies.

## Async Generators
```python
# PROBLEM: Stream data asynchronously with async generators
# TIME: O(1) per iteration SPACE: O(1) per batch

import asyncio
from typing import AsyncIterator, List
import aiohttp

class AsyncDataStreamer:
    """
    Async generators allow yielding values asynchronously.
    Perfect for streaming data, paginated APIs, and real-time feeds.
    """
    
    def __init__(self, base_url: str):
        self.base_url = base_url
    
    async def fetch_pages(self, endpoint: str, page_size: int = 100) -> AsyncIterator[List[dict]]:
        """
        Async generator that yields pages of data.
        Fetches next page only when needed (lazy evaluation).
        """
        page = 1
        has_more = True
        
        while has_more:
            # Async HTTP request
            async with aiohttp.ClientSession() as session:
                url = f"{self.base_url}/{endpoint}?page={page}&size={page_size}"
                async with session.get(url) as response:
                    data = await response.json()
                    
                    if not data:
                        has_more = False
                    else:
                        yield data
                        page += 1
    
    async def stream_lines(self, file_path: str) -> AsyncIterator[str]:
        """
        Async generator for reading large files line by line.
        Non-blocking I/O for concurrent operations.
        """
        with open(file_path, 'r') as file:
            for line in file:
                # Yield control to event loop
                await asyncio.sleep(0)
                yield line.strip()
    
    async def process_with_backpressure(
        self, 
        data_stream: AsyncIterator[dict],
        max_concurrent: int = 5
    ) -> AsyncIterator[dict]:
        """
        Process async stream with backpressure control.
        Limits concurrent processing to avoid overload.
        """
        semaphore = asyncio.Semaphore(max_concurrent)
        
        async def process_item(item: dict) -> dict:
            async with semaphore:
                # Simulate async processing
                await asyncio.sleep(0.1)
                return {**item, 'processed': True}
        
        async for batch in data_stream:
            # Process batch concurrently with backpressure
            tasks = [process_item(item) for item in batch]
            results = await asyncio.gather(*tasks)
            
            for result in results:
                yield result

async def main():
    """Demonstrate async generator usage."""
    streamer = AsyncDataStreamer("https://api.example.com")
    
    # Consume async generator
    async for page in streamer.fetch_pages("/users"):
        print(f"Received {len(page)} users")
        for user in page:
            print(f"  - {user['name']}")

# Usage with async for
async def consume_stream():
    """Pattern for consuming async generators."""
    streamer = AsyncDataStreamer("https://api.example.com")
    
    # Async comprehension
    results = [
        item 
        async for batch in streamer.fetch_pages("/data")
        for item in batch
    ]
    
    return results
```
**Why**: Essential for modern async Python; enables efficient streaming without blocking event loop.

## Sagas Pattern (Distributed Transactions)
```python
# PROBLEM: Coordinate distributed transactions across microservices
# TIME: O(n) for n steps SPACE: O(n) for compensation log

from dataclasses import dataclass
from typing import Callable, List, Optional, Any
from enum import Enum
import logging

class SagaStatus(Enum):
    """Saga execution states."""
    PENDING = "pending"
    COMPLETED = "completed"
    COMPENSATING = "compensating"
    FAILED = "failed"

@dataclass
class SagaStep:
    """
    Represents a single step in the saga.
    Contains forward action and compensation action.
    """
    name: str
    action: Callable[[], Any]  # Forward action
    compensate: Callable[[], Any]  # Rollback action

class SagaOrchestrator:
    """
    Orchestrates saga execution across multiple services.
    Implements compensating transaction pattern for distributed consistency.
    """
    
    def __init__(self, name: str):
        self.name = name
        self.steps: List[SagaStep] = []
        self.completed_steps: List[SagaStep] = []
        self.status = SagaStatus.PENDING
        self.logger = logging.getLogger(__name__)
    
    def add_step(self, name: str, action: Callable, compensate: Callable) -> 'SagaOrchestrator':
        """Add a step to the saga (builder pattern)."""
        self.steps.append(SagaStep(name, action, compensate))
        return self
    
    async def execute(self) -> bool:
        """
        Execute saga: run forward actions, compensate on failure.
        Returns True if successful, False if compensation was needed.
        """
        try:
            # Execute forward actions
            for step in self.steps:
                self.logger.info(f"Executing step: {step.name}")
                
                try:
                    result = await step.action()
                    self.completed_steps.append(step)
                    self.logger.info(f"Step completed: {step.name}")
                    
                except Exception as e:
                    self.logger.error(f"Step failed: {step.name}, error: {e}")
                    await self._compensate()
                    self.status = SagaStatus.FAILED
                    return False
            
            self.status = SagaStatus.COMPLETED
            self.logger.info(f"Saga completed: {self.name}")
            return True
            
        except Exception as e:
            self.logger.error(f"Saga execution failed: {e}")
            await self._compensate()
            self.status = SagaStatus.FAILED
            return False
    
    async def _compensate(self) -> None:
        """
        Execute compensation actions in reverse order.
        Runs all compensations even if some fail.
        """
        self.status = SagaStatus.COMPENSATING
        self.logger.info(f"Starting compensation for: {self.name}")
        
        # Reverse order compensation
        for step in reversed(self.completed_steps):
            try:
                self.logger.info(f"Compensating step: {step.name}")
                await step.compensate()
                self.logger.info(f"Compensation complete: {step.name}")
                
            except Exception as e:
                # Log but continue compensating other steps
                self.logger.error(f"Compensation failed for {step.name}: {e}")
```
