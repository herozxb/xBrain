## [Maximum Sum Subarray K]
```python
# PROBLEM: Find maximum sum of any contiguous subarray of size k
# TIME: O(n) SPACE: O(1)
def max_sum_subarray_k(arr, k):
    # Edge case: array smaller than window
    if len(arr) < k:
        return None
    
    # Compute sum of first window
    window_sum = sum(arr[:k])
    max_sum = window_sum
    
    # Slide window across array, updating sum in O(1)
    for i in range(len(arr) - k):
        # Subtract element leaving window, add element entering
        window_sum = window_sum - arr[i] + arr[i + k]
        max_sum = max(max_sum, window_sum)
    
    return max_sum
```
**Why**: Classic sliding window pattern—maintains running sum instead of recomputing O(k) each step.

## [Three Sum]
```python
# PROBLEM: Find all unique triplets that sum to zero
# TIME: O(n²) SPACE: O(1) excluding output
def three_sum(nums):
    result = []
    nums.sort()  # Sort to enable two-pointer technique
    
    for i in range(len(nums) - 2):
        # Skip duplicates for first element
        if i > 0 and nums[i] == nums[i - 1]:
            continue
        
        # Two pointers for remaining two elements
        left, right = i + 1, len(nums) - 1
        
        while left < right:
            total = nums[i] + nums[left] + nums[right]
            
            if total < 0:
                left += 1  # Need larger sum
            elif total > 0:
                right -= 1  # Need smaller sum
            else:
                result.append([nums[i], nums[left], nums[right]])
                # Skip duplicates for second and third elements
                while left < right and nums[left] == nums[left + 1]:
                    left += 1
                while left < right and nums[right] == nums[right - 1]:
                    right -= 1
                left += 1
                right -= 1
    
    return result
```
**Why**: Two pointers after sorting eliminates one nested loop, achieving O(n²) instead of O(n³).

## [Subsets]
```python
# PROBLEM: Generate all possible subsets of a set (power set)
# TIME: O(n * 2ⁿ) SPACE: O(n) for recursion depth
def subsets(nums):
    result = []
    
    def backtrack(start, current):
        # Add current subset to result (make a copy!)
        result.append(current[:])
        
        # Try adding each remaining element
        for i in range(start, len(nums)):
            current.append(nums[i])  # Choose
            backtrack(i + 1, current)  # Explore
            current.pop()  # Unchoose (backtrack)
    
    backtrack(0, [])
    return result
```
**Why**: Backtracking explores all 2ⁿ subsets systematically; each element is either included or excluded.

Topics: backtracking

## [Number of Connected Components]
```python
# PROBLEM: Count connected components in an undirected graph
# TIME: O(n * α(n)) ≈ O(n) SPACE: O(n)
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.components = n  # Track number of components
    
    def find(self, x):
        # Path compression: point to root directly
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        px, py = self.find(x), self.find(y)
        if px == py:
            return False  # Already connected
        
        # Union by rank for balanced trees
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        self.components -= 1
        return True

def count_components(n, edges):
    uf = UnionFind(n)
    for u, v in edges:
        uf.union(u, v)
    return uf.components
```
**Why**: Union-Find with path compression and union by rank gives near-constant time operations.

Topics: union_find

## [Longest Common Subsequence]
```python
# PROBLEM: Find length of longest common subsequence between two strings
# TIME: O(m*n) SPACE: O(min(m,n)) with space optimization
def longest_common_subsequence(text1, text2):
    # Ensure text2 is shorter for space optimization
    if len(text1) < len(text2):
        text1, text2 = text2, text1
    
    # Only keep previous and current row
    prev = [0] * (len(text2) + 1)
    
    for i in range(1, len(text1) + 1):
        curr = [0] * (len(text2) + 1)
        for j in range(1, len(text2) + 1):
            if text1[i - 1] == text2[j - 1]:
                # Characters match: extend LCS
                curr[j] = prev[j - 1] + 1
            else:
                # Take max of excluding either character
                curr[j] = max(prev[j], curr[j - 1])
        prev = curr
    
    return prev[len(text2)]
```
**Why**: DP with space optimization—only need previous row, reducing space from O(mn) to O(min(m,n)).

Topics: dp

## [BFS Shortest Path]
```python
# PROBLEM: Find shortest path length between two nodes in unweighted graph
# TIME: O(V + E) SPACE: O(V)
from collections import deque

def shortest_path(graph, start, end):
    if start == end:
        return 0
    
    queue = deque([(start, 0)])  # (node, distance)
    visited = {start}
    
    while queue:
        node, dist = queue.popleft()
        
        for neighbor in graph[node]:
            if neighbor == end:
                return dist + 1
            
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append((neighbor, dist + 1))
    
    return -1  # No path exists
```
**Why**: BFS guarantees shortest path in unweighted graphs—explores all nodes at distance d before d+1.

Topics: graphs

## [Maximum Depth of Binary Tree]
```python
# PROBLEM: Find maximum depth/height of binary tree
# TIME: O(n) SPACE: O(h) where h is height
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def max_depth(root):
    # Base case: empty tree has depth 0
    if not root:
        return 0
    
    # Recursively find depth of subtrees
    left_depth = max_depth(root.left)
    right_depth = max_depth(root.right)
    
    # Depth is max of subtrees plus current node
    return 1 + max(left_depth, right_depth)
```
**Why**: Simple recursive DFS—each node contributes 1 to depth, take max of children.

Topics: trees

## [Kth Largest Element]
```python
# PROBLEM: Find kth largest element in unsorted array
# TIME: O(n log k) SPACE: O(k)
import heapq

def find_kth_largest(nums, k):
    # Maintain min-heap of k largest elements
    heap = []
    
    for num in nums:
        heapq.heappush(heap, num)
        # Keep only k elements—smallest gets pushed out
        if len(heap) > k:
            heapq.heappop(heap)
    
    # Root of min-heap is kth largest
    return heap[0]
```
**Why**: Min-heap of size k efficiently tracks top k elements; root is the kth largest.

## [Word Dictionary with Wildcards]
```python
# PROBLEM: Implement dictionary supporting '.' wildcard matching any letter
# TIME: O(m) for normal, O(26^m) worst case with wildcards SPACE: O(alphabet * m * n)
class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_word = False

class WordDictionary:
    def __init__(self):
        self.root = TrieNode()
    
    def add_word(self, word):
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_word = True
    
    def search(self, word):
        def dfs(node, index):
            # Reached end of word
            if index == len(word):
                return node.is_word
            
            char = word[index]
            if char != '.':
                # Exact match required
                return char in node.children and dfs(node.children[char], index + 1)
            else:
                # Wildcard: try all possible children
                return any(dfs(child, index + 1) for child in node.children.values())
        
        return dfs(self.root, 0)
```
**Why**: Trie provides efficient prefix-based storage; DFS handles wildcard exploration.

Topics: trie

## [LRU Cache]
```python
# PROBLEM: Implement cache with O(1) get and put, evicting least recently used
# TIME: O(1) for both operations SPACE: O(capacity)
from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = OrderedDict()  # Maintains insertion order
    
    def get(self, key):
        if key not in self.cache:
            return -1
        
        # Move to end (most recently used)
        self.cache.move_to_end(key)
        return self.cache[key]
    
    def put(self, key, value):
        if key in self.cache:
            # Update and move to end
            self.cache.move_to_end(key)
        else:
            # Evict LRU if at capacity
            if len(self.cache) >= self.capacity:
                self.cache.popitem(last=False)  # Remove from front
        
        self.cache[key] = value
```
**Why**: OrderedDict combines hash table (O(1) lookup) with doubly-linked list (O(1) reorder), perfect for LRU.

Topics: lru_cache
