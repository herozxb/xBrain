## [Segment Tree - Range Sum Query]

```python
# PROBLEM: Efficiently handle range queries and point updates on an array
# TIME: O(n) build, O(log n) query/update SPACE: O(n)
class SegmentTree:
    """Segment tree for range sum queries with point updates"""
    
    def __init__(self, data):
        """Build segment tree from initial data array"""
        self.n = len(data)
        # Allocate 4*n space for tree (safe upper bound)
        self.tree = [0] * (4 * self.n)
        self._build(data, 0, 0, self.n - 1)
    
    def _build(self, data, node, start, end):
        """Recursively build tree: each node stores sum of its range"""
        if start == end:
            # Leaf node: store single element
            self.tree[node] = data[start]
            return
        
        mid = (start + end) // 2
        left_child = 2 * node + 1
        right_child = 2 * node + 2
        
        # Build left and right subtrees
        self._build(data, left_child, start, mid)
        self._build(data, right_child, mid + 1, end)
        
        # Internal node: sum of children
        self.tree[node] = self.tree[left_child] + self.tree[right_child]
    
    def query(self, l, r):
        """Query sum in range [l, r] (inclusive)"""
        return self._query(0, 0, self.n - 1, l, r)
    
    def _query(self, node, start, end, l, r):
        # No overlap: query range outside node range
        if r < start or end < l:
            return 0
        
        # Complete overlap: query range covers node range
        if l <= start and end <= r:
            return self.tree[node]
        
        # Partial overlap: recurse on both children
        mid = (start + end) // 2
        left_sum = self._query(2 * node + 1, start, mid, l, r)
        right_sum = self._query(2 * node + 2, mid + 1, end, l, r)
        return left_sum + right_sum
    
    def update(self, idx, value):
        """Update single element at index idx to new value"""
        self._update(0, 0, self.n - 1, idx, value)
    
    def _update(self, node, start, end, idx, value):
        if start == end:
            # Leaf node: update value
            self.tree[node] = value
            return
        
        mid = (start + end) // 2
        if idx <= mid:
            self._update(2 * node + 1, start, mid, idx, value)
        else:
            self._update(2 * node + 2, mid + 1, end, idx, value)
        
        # Update current node after child update
        self.tree[node] = self.tree[2 * node + 1] + self.tree[2 * node + 2]
```

**Why**: Segment trees enable efficient range queries (sum/min/max) with dynamic updates. Critical for competitive programming and scenarios requiring frequent range operations, like analytics dashboards or game leaderboards.

## [Load Balancer - Round Robin]

```python
# PROBLEM: Distribute incoming requests evenly across multiple servers
# TIME: O(1) per request SPACE: O(n) for n servers
class LoadBalancer:
    """Round-robin load balancer with health checking"""
    
    def __init__(self, servers):
        """Initialize with list of server addresses"""
        self.servers = servers
        self.current = 0  # Current server index
        self.healthy = {s: True for s in servers}  # Health status
    
    def get_next_server(self):
        """Get next available healthy server using round-robin"""
        if not self.servers:
            raise Exception("No servers available")
        
        attempts = 0
        n = len(self.servers)
        
        # Try all servers until finding a healthy one
        while attempts < n:
            server = self.servers[self.current]
            self.current = (self.current + 1) % n  # Circular increment
            
            if self.healthy.get(server, False):
                return server
            
            attempts += 1
        
        raise Exception("No healthy servers available")
    
    def mark_unhealthy(self, server):
        """Mark server as unhealthy (e.g., after failed health check)"""
        self.healthy[server] = False
    
    def mark_healthy(self, server):
        """Mark server as healthy after recovery"""
        self.healthy[server] = True
    
    def add_server(self, server):
        """Add new server to pool"""
        if server not in self.servers:
            self.servers.append(server)
            self.healthy[server] = True
    
    def remove_server(self, server):
        """Remove server from pool"""
        if server in self.servers:
            self.servers.remove(server)
            del self.healthy[server]
            # Adjust current index if needed
            if self.current >= len(self.servers):
                self.current = 0
```

**Why**: Load balancers are fundamental infrastructure for scaling web applications. Round-robin ensures fair distribution while health checking prevents routing to failed servers. Essential for high-availability systems.

## [Consistent Hashing]

```python
# PROBLEM: Distribute keys across servers with minimal redistribution on changes
# TIME: O(log n) lookup/add SPACE: O(n) for n nodes
import hashlib

class ConsistentHashing:
    """Consistent hashing for distributed caching/database sharding"""
    
    def __init__(self, virtual_nodes=150):
        """Initialize with virtual nodes per server for better distribution"""
        self.ring = {}  # hash -> server mapping
        self.sorted_keys = []  # Sorted hash values
        self.virtual_nodes = virtual_nodes
    
    def _hash(self, key):
        """Generate consistent hash using MD5"""
        return int(hashlib.md5(key.encode()).hexdigest(), 16)
    
    def add_server(self, server):
        """Add server with multiple virtual nodes to ring"""
        for i in range(self.virtual_nodes):
            # Create virtual node key
            virtual_key = f"{server}#{i}"
            hash_val = self._hash(virtual_key)
            
            self.ring[hash_val] = server
            
            # Maintain sorted keys for binary search
            if hash_val not in self.sorted_keys:
                self.sorted_keys.append(hash_val)
        
        self.sorted_keys.sort()
    
    def remove_server(self, server):
        """Remove server and all its virtual nodes"""
        for i in range(self.virtual_nodes):
            virtual_key = f"{server}#{i}"
            hash_val = self._hash(virtual_key)
            
            if hash_val in self.ring:
                del self.ring[hash_val]
                self.sorted_keys.remove(hash_val)
    
    def get_server(self, key):
        """Find server responsible for given key"""
        if not self.ring:
            return None
        
        hash_val = self._hash(key)
        
        # Binary search for first server with hash >= key hash
        # This is the "clockwise" next server on the ring
        for server_hash in self.sorted_keys:
            if server_hash >= hash_val:
                return self.ring[server_hash]
        
        # Wrap around to first server if past end
        return self.ring[self.sorted_keys[0]]
```

**Why**: Consistent hashing minimizes data movement when servers join/leave. Critical for distributed caches (Redis Cluster, Memcached), CDNs, and databases. Virtual nodes ensure even distribution and reduce hotspot issues.

## [Message Queue]

```python
# PROBLEM: Decouple producers and consumers for async processing
# TIME: O(1) enqueue/dequeue SPACE: O(n) for n messages
from collections import deque
import threading
import time

class MessageQueue:
    """Thread-safe message queue with multiple consumers"""
    
    def __init__(self, max_size=1000):
        """Initialize queue with optional max size"""
        self.queue = deque()
        self.max_size = max_size
        self.lock = threading.Lock()
        self.not_empty = threading.Condition(self.lock)
        self.not_full = threading.Condition(self.lock)
    
    def publish(self, message, priority=0):
        """Add message to queue (producer)"""
        with self.not_full:
            # Wait if queue is full
            while len(self.queue) >= self.max_size:
                self.not_full.wait()
            
            # Add message with timestamp for ordering
            self.queue.append({
                'data': message,
                'priority': priority,
                'timestamp': time.time()
            })
            
            # Notify waiting consumers
            self.not_empty.notify()
    
    def consume(self, timeout=None):
        """Retrieve message from queue (consumer)"""
        with self.not_empty:
            # Wait if queue is empty
            while not self.queue:
                if not self.not_empty.wait(timeout):
                    return None  # Timeout
            
            # Get message (FIFO)
            message = self.queue.popleft()
            
            # Notify waiting producers
            self.not_full.notify()
            
            return message['data']
    
    def size(self):
        """Get current queue size"""
        with self.lock:
            return len(self.queue)
    
    def is_empty(self):
        """Check if queue is empty"""
        with self.lock:
            return len(self.queue) == 0
```

**Why**: Message queues enable asynchronous communication between services, improving resilience and scalability. Used in event-driven architectures, job processing systems, and microservices. Enables buffering during traffic spikes.

## [Binary Search Tree Validator]

```python
# PROBLEM: Validate if a binary tree is a valid Binary Search Tree
# TIME: O(n) SPACE: O(h) where h is tree height
class TreeNode:
    """Binary tree node"""
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def is_valid_bst(root):
    """
    Validate BST using range-based approach.
    Each node must be within valid min/max range inherited from ancestors.
    """
    def validate(node, min_val, max_val):
        # Empty tree is valid BST
        if not node:
            return True
        
        # Check if current node violates BST property
        if min_val is not None and node.val <= min_val:
            return False
        if max_val is not None and node.val >= max_val:
            return False
        
        # Recursively validate subtrees with updated ranges
        # Left subtree: all values must be < current
        # Right subtree: all values must be > current
        return (validate(node.left, min_val, node.val) and
                validate(node.right, node.val, max_val))
    
    return validate(root, None, None)

def is_valid_bst_inorder(root):
    """
    Alternative: Use inorder traversal (should be sorted for BST).
    TIME: O(n) SPACE: O(h)
    """
    prev = None
    
    def inorder(node):
        nonlocal prev
        
        if not node:
            return True
        
        # Check left subtree
        if not inorder(node.left):
            return False
        
        # Current must be greater than previous
        if prev is not None and node.val <= prev:
            return False
        
        prev = node.val
        
        # Check right subtree
        return inorder(node.right)
    
    return inorder(root)
```

**Why**: BST validation is a common interview question testing tree traversal and BST property understanding. Essential for debugging tree structures and ensuring database indexes remain valid. Two approaches demonstrate different problem-solving strategies.

## [Rate Limiter - Token Bucket]

```python
# PROBLEM: Limit request rate to prevent abuse/overload
# TIME: O(1) check SPACE: O(n) for n users
import time
from collections import defaultdict

class TokenBucket:
    """Token bucket rate limiter for API throttling"""
    
    def __init__(self, rate=10, capacity=100):
        """
        rate: tokens added per second
        capacity: maximum bucket size (burst limit)
        """
        self.rate = rate
        self.capacity = capacity
        self.buckets = defaultdict(lambda: {
            'tokens': capacity,  # Start full
            'last_update': time.time()
        })
    
    def _refill(self, user_id):
        """Refill tokens based on elapsed time"""
        bucket = self.buckets[user_id]
        now = time.time()
        elapsed = now - bucket['last_update']
        
        # Add tokens for elapsed time
        refill_amount = elapsed * self.rate
        bucket['tokens'] = min(
            self.capacity,
            bucket['tokens'] + refill_amount
        )
        bucket['last_update'] = now
    
    def allow_request(self, user_id, tokens=1):
        """Check if request should be allowed"""
        self._refill(user_id)
        bucket = self.buckets[user_id]
        
        if bucket['tokens'] >= tokens:
            bucket['tokens'] -= tokens
            return True
        
        return False
    
    def get_wait_time(self, user_id, tokens=1):
        """Calculate time until enough tokens available"""
        self._refill(user_id)
        bucket = self.buckets[user_id]
        
        if bucket['tokens'] >= tokens:
            return 0
        
        needed = tokens - bucket['tokens']
        return needed / self.rate

# Usage example
limiter = TokenBucket(rate=5, capacity=20)  # 5 req/s, burst 20

# Check if user can make request
if limiter.allow_request('user_123'):
    print("Request allowed")
else:
    wait = limiter.get_wait_time('user_123')
    print(f"Rate limited. Retry in {wait:.2f}s")
```

**Why**: Rate limiting protects services from abuse and ensures fair resource allocation. Token bucket allows controlled bursts while maintaining average rate. Used in API gateways, authentication services, and DDoS protection.

## [State Pattern - Order Processing]

```python
# PROBLEM: Manage complex state transitions in order processing
# TIME: O(1) state transition SPACE: O(n) for n states
from abc import ABC, abstractmethod

class OrderState(ABC):
    """Abstract base class for order states"""
    
    @abstractmethod
    def pay(self, order):
        pass
    
    @abstractmethod
    def ship(self, order):
        pass
    
    @abstractmethod
    def deliver(self, order):
        pass
    
    @abstractmethod
    def cancel(self, order):
        pass

class NewOrderState(OrderState):
    """Initial state: order created but not paid"""
    
    def pay(self, order):
        print(f"Order {order.id}: Payment received")
        order.state = PaidState()
    
    def ship(self, order):
        print(f"Order {order.id}: Cannot ship unpaid order")
    
    def deliver(self, order):
        print(f"Order {order.id}: Cannot deliver unpaid order")
    
    def cancel(self, order):
        print(f"Order {order.id}: Order cancelled")
        order.state = CancelledState()

class PaidState(OrderState):
    """Order paid, ready for shipping"""
    
    def pay(self, order):
        print(f"Order {order.id}: Already paid")
    
    def ship(self, order):
        print(f"Order {order.id}: Order shipped")
        order.state = ShippedState()
    
    def deliver(self, order):
        print(f"Order {order.id}: Must ship first")
    
    def cancel(self, order):
        print(f"Order {order.id}: Refunding and cancelling")
        order.state = CancelledState()

class ShippedState(OrderState):
    """Order in transit"""
    
    def pay(self, order):
        print(f"Order {order.id}: Already paid")
    
    def ship(self, order):
        print(f"Order {order.id}: Already shipped")
    
    def deliver(self, order):
        print(f"Order {order.id}: Order delivered")
        order.state = DeliveredState()
    
    def cancel(self, order):
        print(f"Order {order.id}: Cannot cancel shipped order")

class DeliveredState(OrderState):
    """Order delivered to customer"""
    
    def pay(self, order):
        print(f"Order {order.id}: Already paid")
    
    def ship(self, order):
        print(f"Order {order.id}: Already delivered")
    
    def deliver(self, order):
        print(f"Order {order.id}: Already delivered")
    
    def cancel(self, order):
        print(f"Order {order.id}: Process return request")

class CancelledState(OrderState):
    """Order cancelled"""
    
    def pay(self, order):
        print(f"Order {order.id}: Order is cancelled")
    
    def ship(self, order):
        print(f"Order {order.id}: Order is cancelled")
    
    def deliver(self, order):
        print(f"Order {order.id}: Order is cancelled")
    
    def cancel(self, order):
        print(f"Order {order.id}: Already cancelled")

class Order:
    """Order with state management"""
    
    def __init__(self, order_id):
        self.id = order_id
        self.state = NewOrderState()
    
    def pay(self):
        self.state.pay(self)
    
    def ship(self):
        self.state.ship(self)
    
    def deliver(self):
        self.state.deliver(self)
    
    def cancel(self):
        self.state.cancel(self)
```

**Why**: State pattern encapsulates state-specific behavior and makes transitions explicit. Eliminates complex if-else chains. Perfect for workflows (orders, documents, games) where behavior depends on current state. Easy to add new states without modifying existing code.

## [Bloom Filter]

```python
# PROBLEM: Space-efficient probabilistic set membership test
# TIME: O(k) where k is number of hash functions SPACE: O(m) bits
import hashlib

class BloomFilter:
    """Probabilistic data structure for set membership"""
    
    def __init__(self, size=10000, hash_count=7):
        """
        size: bit array size (m)
        hash_count: number of hash functions (k)
        
        Optimal k ≈ (m/n) * ln(2) where n is expected elements
        """
        self.size = size
        self.hash_count = hash_count
        self.bit_array = [False] * size
    
    def _hashes(self, item):
        """Generate k hash values using double hashing"""
        # Convert item to bytes
        if isinstance(item, str):
            item = item.encode()
        
        # Get two base hashes
        h1 = int(hashlib.md5(item).hexdigest(), 16)
        h2 = int(hashlib.sha1(item).hexdigest(), 16)
        
        # Generate k hashes using: h(i) = h1 + i * h2
        hashes = []
        for i in range(self.hash_count):
            hash_val = (h1 + i * h2) % self.size
            hashes.append(hash_val)
        
        return hashes
    
    def add(self, item):
        """Add item to filter"""
        for hash_val in self._hashes(item):
            self.bit_array[hash_val] = True
    
    def contains(self, item):
        """
        Check if item might be in set.
        Returns False: definitely NOT in set
        Returns True: PROBABLY in set (may be false positive)
        """
        for hash_val in self._hashes(item):
            if not self.bit_array[hash_val]:
                return False  # Definitely not present
        return True  # Probably present
    
    def false_positive_rate(self, n_items):
        """Estimate false positive rate given n items inserted"""
        # P(false positive) ≈ (1 - e^(-kn/m))^k
        import math
        k = self.hash_count
        m = self.size
        n = n_items
        
        exponent = -k * n / m
        return (1 - math.exp(exponent)) ** k
```

**Why**: Bloom filters provide space-efficient membership testing with controllable false positive rate. Used in databases (avoiding disk lookups), web crawlers (URL deduplication), and caching systems. Can determine if something definitely doesn't exist.
