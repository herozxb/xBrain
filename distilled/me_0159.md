## Sliding Window - Minimum Window Substring
```python
# PROBLEM: Find minimum window in s containing all chars of t
# TIME: O(n) SPACE: O(k) where k is unique chars in t

def min_window(s: str, t: str) -> str:
    """
    Find smallest window in s containing all characters of t.
    
    Args:
        s: Source string
        t: Target string with required characters
    
    Returns:
        Minimum window substring, or empty string if not found
    """
    from collections import Counter
    
    if not s or not t:
        return ""
    
    # Count required characters
    need = Counter(t)
    have = {}
    
    left = 0
    min_len = float('inf')
    min_start = 0
    
    # Track how many unique chars we have enough of
    formed = 0
    required = len(need)
    
    for right, char in enumerate(s):
        # Add character to window
        have[char] = have.get(char, 0) + 1
        
        # Check if this char satisfies a requirement
        if char in need and have[char] == need[char]:
            formed += 1
        
        # Try to shrink window while valid
        while left <= right and formed == required:
            # Update minimum
            if right - left + 1 < min_len:
                min_len = right - left + 1
                min_start = left
            
            # Remove left character
            left_char = s[left]
            have[left_char] -= 1
            
            if left_char in need and have[left_char] < need[left_char]:
                formed -= 1
            
            left += 1
    
    return s[min_start:min_start + min_len] if min_len != float('inf') else ""
```

**Why**: Sliding window with two pointers achieves O(n) by expanding right and shrinking left when valid.

## Two Pointers - 3Sum
```python
# PROBLEM: Find all triplets that sum to zero
# TIME: O(n²) SPACE: O(1) excluding output

def three_sum(nums: list) -> list:
    """
    Find all unique triplets that sum to zero.
    
    Args:
        nums: List of integers
    
    Returns:
        List of triplet lists
    """
    nums.sort()
    result = []
    n = len(nums)
    
    for i in range(n - 2):
        # Skip duplicates
        if i > 0 and nums[i] == nums[i - 1]:
            continue
        
        left, right = i + 1, n - 1
        
        while left < right:
            total = nums[i] + nums[left] + nums[right]
            
            if total == 0:
                result.append([nums[i], nums[left], nums[right]])
                
                # Skip duplicates
                while left < right and nums[left] == nums[left + 1]:
                    left += 1
                while left < right and nums[right] == nums[right - 1]:
                    right -= 1
                
                left += 1
                right -= 1
            
            elif total < 0:
                left += 1
            else:
                right -= 1
    
    return result
```

**Why**: Sorting + two pointers avoids O(n³) brute force, achieving O(n²).

## Backtracking - Permutations
```python
# PROBLEM: Generate all permutations of array
# TIME: O(n! * n) SPACE: O(n) for recursion

def permute(nums: list) -> list:
    """
    Generate all permutations of given numbers.
    
    Args:
        nums: List of integers
    
    Returns:
        List of all permutation lists
    """
    result = []
    
    def backtrack(start: int):
        # Base case: all positions filled
        if start == len(nums):
            result.append(nums[:])
            return
        
        for i in range(start, len(nums)):
            # Swap to place nums[i] at position start
            nums[start], nums[i] = nums[i], nums[start]
            
            # Recurse for next position
            backtrack(start + 1)
            
            # Backtrack: restore original order
            nums[start], nums[i] = nums[i], nums[start]
    
    backtrack(0)
    return result
```

**Why**: Backtracking explores all n! permutations by swapping and recursing.

## Backtracking - Subsets
```python
# PROBLEM: Generate all possible subsets
# TIME: O(2^n * n) SPACE: O(n) for current path

def subsets(nums: list) -> list:
    """
    Generate all subsets of given numbers.
    
    Args:
        nums: List of integers
    
    Returns:
        List of all subset lists
    """
    result = []
    
    def backtrack(start: int, path: list):
        # Add current subset
        result.append(path[:])
        
        # Try adding each remaining element
        for i in range(start, len(nums)):
            path.append(nums[i])
            backtrack(i + 1, path)
            path.pop()
    
    backtrack(0, [])
    return result
```

**Why**: Each element can be included or not, giving 2^n subsets. Backtracking builds them efficiently.

## Union Find - Number of Islands
```python
# PROBLEM: Count islands in 2D grid
# TIME: O(m * n * α(mn)) SPACE: O(m * n)

class UnionFind:
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.count = 0
    
    def find(self, x: int) -> int:
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x: int, y: int) -> None:
        px, py = self.find(x), self.find(y)
        if px == py:
            return
        
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        self.count -= 1


def num_islands(grid: list) -> int:
    """
    Count number of islands in grid.
    
    Args:
        grid: 2D list of '1' and '0'
    
    Returns:
        Number of islands
    """
    if not grid:
        return 0
    
    rows, cols = len(grid), len(grid[0])
    uf = UnionFind(rows * cols)
    
    # Count land cells
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == '1':
                uf.count += 1
    
    # Union adjacent land
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == '1':
                idx = r * cols + c
                
                # Check right neighbor
                if c + 1 < cols and grid[r][c + 1] == '1':
                    uf.union(idx, r * cols + c + 1)
                
                # Check bottom neighbor
                if r + 1 < rows and grid[r + 1][c] == '1':
                    uf.union(idx, (r + 1) * cols + c)
    
    return uf.count
```

**Why**: Union Find efficiently groups connected components in near O(1) amortized time.

## Bloom Filter
```python
# PROBLEM: Probabilistic membership testing
# TIME: O(k) where k is number of hash functions
# SPACE: O(m) where m is bit array size

import mmh3
from bitarray import bitarray

class BloomFilter:
    """
    Space-efficient probabilistic data structure for membership tests.
    """
    
    def __init__(self, size: int, hash_count: int):
        """
        Initialize Bloom filter.
        
        Args:
            size: Size of bit array
            hash_count: Number of hash functions
        """
        self.size = size
        self.hash_count = hash_count
        self.bit_array = bitarray(size)
        self.bit_array.setall(False)
    
    def add(self, item: str) -> None:
        """Add item to filter."""
        for i in range(self.hash_count):
            index = mmh3.hash(item, i) % self.size
            self.bit_array[index] = True
    
    def contains(self, item: str) -> bool:
        """Check if item might be in filter."""
        for i in range(self.hash_count):
            index = mmh3.hash(item, i) % self.size
            if not self.bit_array[index]:
                return False
        return True


# Simplified version without external libs
class SimpleBloomFilter:
    def __init__(self, size: int = 1000, hash_count: int = 3):
        self.size = size
        self.hash_count = hash_count
        self.bits = [False] * size
    
    def _hashes(self, item: str) -> list:
        """Generate multiple hash values."""
        result = []
        for i in range(self.hash_count):
            h = hash(f"{item}_{i}") % self.size
            result.append(h)
        return result
    
    def add(self, item: str) -> None:
        for h in self._hashes(item):
            self.bits[h] = True
    
    def contains(self, item: str) -> bool:
        return all(self.bits[h] for h in self._hashes(item))
```

**Why**: Bloom filter uses O(1) space per element and O(k) time, trading accuracy for efficiency.

## LRU Cache
```python
# PROBLEM: Cache with least recently used eviction
# TIME: O(1) get/put SPACE: O(capacity)

from collections import OrderedDict

class LRUCache:
    """
    Least Recently Used cache with O(1) operations.
    """
    
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = OrderedDict()
    
    def get(self, key: int) -> int:
        """Get value and mark as recently used."""
        if key not in self.cache:
            return -1
        
        # Move to end (most recent)
        self.cache.move_to_end(key)
        return self.cache[key]
    
    def put(self, key: int, value: int) -> None:
        """Put value, evict LRU if at capacity."""
        if key in self.cache:
            self.cache.move_to_end(key)
        
        self.cache[key] = value
        
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)  # Remove oldest
```

**Why**: OrderedDict maintains insertion order, enabling O(1) access and eviction.

## Rate Limiter - Token Bucket
```python
# PROBLEM: Limit request rate with token bucket
# TIME: O(1) SPACE: O(1) per client

import time

class TokenBucket:
    """
    Token bucket rate limiter.
    """
    
    def __init__(self, rate: float, capacity: int):
        """
        Initialize token bucket.
        
        Args:
            rate: Tokens added per second
            capacity: Maximum tokens
        """
        self.rate = rate
        self.capacity = capacity
        self.tokens = capacity
        self.last_time = time.time()
    
    def consume(self, tokens: int = 1) -> bool:
        """
        Try to consume tokens.
        
        Args:
            tokens: Number of tokens to consume
        
        Returns:
            True if allowed, False if rate limited
        """
        now = time.time()
        
        # Add tokens based on elapsed time
        elapsed = now - self.last_time
        self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)
        self.last_time = now
        
        if self.tokens >= tokens:
            self.tokens -= tokens
            return True
        
        return False
```

**Why**: Token bucket allows burst traffic up to capacity while maintaining average rate.

## Consistent Hashing
```python
# PROBLEM: Distribute keys across servers
# TIME: O(log n) SPACE: O(n)

import hashlib
from bisect import bisect_left

class ConsistentHash:
    """
    Consistent hashing for distributed systems.
    """
    
    def __init__(self, replicas: int = 100):
        self.replicas = replicas
        self.ring = {}  # hash -> server
        self.sorted_keys = []
    
    def _hash(self, key: str) -> int:
        """Generate consistent hash for key."""
        return int(hashlib.md5(key.encode()).hexdigest(), 16)
    
    def add_server(self, server: str) -> None:
        """Add server to ring with replicas."""
        for i in range(self.replicas):
            key = self._hash(f"{server}:{i}")
            self.ring[key] = server
            self.sorted_keys.append(key)
        
        self.sorted_keys.sort()
    
    def remove_server(self, server: str) -> None:
        """Remove server from ring."""
        for i in range(self.replicas):
            key = self._hash(f"{server}:{i}")
            if key in self.ring:
                del self.ring[key]
                self.sorted_keys.remove(key)
    
    def get_server(self, key: str) -> str:
        """Find server responsible for key."""
        if not self.ring:
            return None
        
        h = self._hash(key)
        idx = bisect_left(self.sorted_keys, h)
        
        # Wrap around to first server
        if idx >= len(self.sorted_keys):
            idx = 0
        
        return self.ring[self.sorted_keys[idx]]
```

**Why**: Consistent hashing minimizes key redistribution when servers are added/removed.
