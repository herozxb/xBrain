    <script>
        // PROBLEM: Make custom objects iterable using Symbol.iterator
        // TIME: O(n) SPACE: O(1)
        
        // Example 1: Custom iterable object
        const numberRange = {
            start: 1,
            end: 5,
            
            // Define the iterator protocol
            [Symbol.iterator]() {
                let current = this.start;
                const end = this.end;
                
                return {
                    next() {
                        if (current <= end) {
                            return { value: current++, done: false };
                        } else {
                            return { done: true };
                        }
                    }
                };
            }
        };
        
        // Iterate using for...of
        console.log('Number Range:');
        for (const num of numberRange) {
            console.log(num);
        }
        
        // Example 2: Custom array-like iterable
        class SimpleList {
            constructor(...items) {
                this.items = items;
            }
            
            [Symbol.iterator]() {
                let index = 0;
                const items = this.items;
                
                return {
                    next() {
                        if (index < items.length) {
                            return { value: items[index++], done: false };
                        } else {
                            return { done: true };
                        }
                    }
                };
            }
        }
        
        const list = new SimpleList('apple', 'banana', 'cherry');
        console.log('\nSimple List:');
        for (const item of list) {
            console.log(item);
        }
        
        // Example 3: Infinite iterator with break condition
        const fibonacciIterator = {
            [Symbol.iterator]() {
                let prev = 0, curr = 1;
                
                return {
                    next() {
                        [prev, curr] = [curr, prev + curr];
                        return { value: prev, done: false };
                    }
                };
            }
        };
        
        console.log('\nFibonacci (first 10):');
        let count = 0;
        for (const fib of fibonacciIterator) {
            if (count++ >= 10) break;
            console.log(fib);
        }
        
        // Display in HTML
        const output = document.getElementById('output');
        output.innerHTML = `
            <p><strong>Number Range:</strong> ${[...numberRange].join(', ')}</p>
            <p><strong>Simple List:</strong> ${[...list].join(', ')}</p>
        `;
    </script>
</body>
</html>
```

**Why**: Symbol.iterator enables custom iteration behavior for objects, making them work with for...of loops, spread operators, and destructuring.

        // PROBLEM: Understand JavaScript callback queue execution order
        // TIME: O(1) SPACE: O(1)
        
        const output = document.getElementById('output');
        const messages = [];
        
        function log(message) {
            messages.push(message);
            console.log(message);
        }
        
        log('1. Synchronous start');
        
        // setTimeout adds callback to task queue
        setTimeout(() => {
            log('4. setTimeout 0ms callback (task queue)');
        }, 0);
        
        // Another task queue item
        setTimeout(() => {
            log('5. setTimeout 100ms callback (task queue)');
        }, 100);
        
        // Promise callbacks go to microtask queue (higher priority)
        Promise.resolve()
            .then(() => {
                log('3. Promise microtask 1');
            })
            .then(() => {
                log('3b. Promise microtask 2');
            });
        
        log('2. Synchronous end');
        
        // Display execution order after all callbacks complete
        setTimeout(() => {
            const html = messages.map((msg, i) => `<p>${i + 1}. ${msg}</p>`).join('');
            output.innerHTML = `
                <h3>Execution Order:</h3>
                ${html}
                <p><strong>Key:</strong> Synchronous → Microtasks → Task Queue</p>
            `;
        }, 200);
    </script>
</body>
</html>
```

**Why**: Understanding callback queues is essential for asynchronous JavaScript, explaining why setTimeout(..., 0) doesn't execute immediately and how promises get priority.

## js_addeventlistener_bubbling

```html
<!DOCTYPE html>
<html lang="en">

        }

            padding: 30px;
            background: #4caf50;
        }
        #child {
            padding: 20px;
            background: #2196f3;
            color: white;
        }
    </style>
</head>
<body>
    <h2>Event Bubbling Demo</h2>
    
    <div id="grandparent">
        Grandparent
        <div id="parent">
            Parent
            <div id="child">
                Child (click me)
            </div>
        </div>
    </div>
    
    <div id="output"></div>
    
    <script>
        // PROBLEM: Understand event bubbling in the DOM
        // TIME: O(1) SPACE: O(1)
        
        const grandparent = document.getElementById('grandparent');
        const parent = document.getElementById('parent');
        const child = document.getElementById('child');
        const output = document.getElementById('output');
        const events = [];
        
        // Event bubbling (default behavior - third parameter false or omitted)
        grandparent.addEventListener('click', function(e) {
            logEvent('Grandparent clicked (bubbling phase)');
            this.style.background = '#ff9800';
        }, false);
        
        parent.addEventListener('click', function(e) {
            logEvent('Parent clicked (bubbling phase)');
            this.style.background = '#8bc34a';
        }, false);
        
        child.addEventListener('click', function(e) {
            logEvent('Child clicked (bubbling phase)');
            this.style.background = '#03a9f4';
        }, false);
        
        // Event capturing (third parameter true)
        grandparent.addEventListener('click', function(e) {
            logEvent('Grandparent (capturing phase)', true);
        }, true);
        
        parent.addEventListener('click', function(e) {
            logEvent('Parent (capturing phase)', true);
        }, true);
        
        // Example: Stop propagation
        const stopButton = document.createElement('button');
        stopButton.textContent = 'Stop Propagation Mode';
        stopButton.onclick = function() {
            child.addEventListener('click', function(e) {
                e.stopPropagation();
                logEvent('Child - propagation stopped!');
            });
        };
        document.body.appendChild(stopButton);
        
        function logEvent(message, isCapture = false) {
            const prefix = isCapture ? '[CAPTURE]' : '[BUBBLE]';
            events.push(`${prefix} ${message}`);
            updateOutput();
        }
        
        function updateOutput() {
            output.innerHTML = '<h3>Event Flow:</h3><pre>' + 
                events.join('\n') + '</pre>';
        }
        
        // Reset button
        const resetButton = document.createElement('button');
        resetButton.textContent = 'Reset';
        resetButton.onclick = function() {
            events.length = 0;
            updateOutput();
        };
        document.body.appendChild(resetButton);
    </script>
</body>
</html>
```

**Why**: Event bubbling allows parent elements to receive events triggered on children, enabling event delegation and efficient event handling patterns.
